def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating:
      - Hard DDL protection gate: explicit slack <= ddl_protection_threshold (not just < 0) using parameterized threshold
      - Unified MAD-based global feature scaling (instead of per-feature percentile) for rank stability across N=1 to large sets
      - Upward-rank × remaining-work interaction unconditionally (confirmed critical-path release)
      - Energy-aware terms gated only under *meaningfully positive slack* (slack > slack_headroom_threshold), not just >0
      - Bounded monotonic gates replacing fragile sigmoids/percentiles; uses tanh and clipped linear
      - Ready-wait-time scaled by inverse slack magnitude, but clamped to avoid explosion near zero
      - All divisions guarded; no infinite/nan outputs; deterministic and finite
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Unified MAD-based global scaling: robust to outliers and single-element case
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        abs_dev = np.abs(x - med)
        mad = np.median(abs_dev) + eps
        return np.clip((x - med) / mad, -2.0, 2.0)  # bounded [-2,2] using allowed literals

    # --- DDL-critical base scores ---
    # Hard protection threshold: slack <= ddl_protection_threshold triggers full penalty
    ddl_protection_threshold = PARAMS["slack_min_bound"]  # reused as safe hard threshold
    is_ddl_risky = (slack <= ddl_protection_threshold)
    slack_score = np.where(
        is_ddl_risky,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    
    # Uncertainty-slack coupling: penalize high uncertainty when slack is tight or negative
    unc_slack_coupling = np.where(
        is_ddl_risky,
        uncertainty * (-slack + eps),
        0.0
    ) * PARAMS["uncertainty_slack_coupling"]

    # Duration risk: total duration weighted by uncertainty
    duration_total = min_exec_time + min_comm_time + eps
    duration_risk = duration_total * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Critical path urgency: unconditional upward_rank × remaining_work interaction ---
    # Confirmed in starvation failure replays and consensus decisions
    critical_path_urgency = upward_rank * remaining_work

    # --- Lexicographic headroom gate: activate non-DDL terms only when slack > headroom threshold ---
    slack_headroom_threshold = PARAMS["slack_min_bound"] + 1.0  # offset from protection threshold
    slack_headroom_mask = np.where(slack > slack_headroom_threshold, 1.0, 0.0)

    # --- Non-DDL terms (only active when slack exceeds headroom threshold) ---
    # Energy efficiency: marginal energy per unit work, normalized globally
    energy_per_work = min_incremental_energy / (remaining_work + eps)
    energy_eff_score = mad_normalize(energy_per_work)

    # Rank-weighted slack balance: smoothly transitions from slack-dominant to rank-dominant as slack decreases
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    # Use tanh for smooth bounded monotonic rank score (avoids percentile fragility)
    rank_score = -np.tanh(mad_normalize(upward_rank)) * weight_rank

    # Risk-aware energy: interaction gated by uncertainty sigmoid, then globally normalized
    unc_sigmoid = np.tanh(PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # Anti-starvation wait term: scaled by inverse |slack|, clamped to prevent explosion near zero
    wait_base = np.where(
        slack_headroom_mask > 0.0,
        ready_wait_time / (np.abs(slack) + 1.0),
        0.0
    )
    wait_score = mad_normalize(wait_base)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        mad_normalize(critical_path_urgency)
    )
    
    # Non-DDL terms only added when slack is sufficiently positive (lexicographic enforcement)
    score += slack_headroom_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Apply criticality boost multiplicatively only under DDL pressure (tight or violated)
    critical_gate = np.where(
        is_ddl_risky,
        PARAMS["criticality_boost"],
        1.0
    )
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
