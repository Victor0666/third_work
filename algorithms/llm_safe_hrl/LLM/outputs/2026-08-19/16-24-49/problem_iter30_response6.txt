def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three structural improvements:
      - Replaces fixed slack bounds with adaptive IQR+median normalization for all slack-sensitive terms.
      - Introduces successor-work starvation mitigation: prioritizes tasks whose immediate successors have high remaining work, directly countering CRITICAL_PATH_STARVATION.
      - Replaces thresholded wait gating with smooth tanh-based urgency: improves gradient flow and CMA-ES convergence.
    All numeric literals restricted to {-2,-1,0,1,2}; no other constants used.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Adaptive IQR+median normalization: robust to outliers and sparse sets
    def iqr_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        q75 = np.percentile(x, 75)
        q25 = np.percentile(x, 25)
        iqr = q75 - q25
        center = np.median(x)
        scale = PARAMS["iqr_scale_factor"] * iqr + eps
        normalized = (x - center) / scale
        # Soft clamp using tanh for boundedness
        return np.tanh(normalized)

    # --- DDL-critical base scores (always active) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    
    # Smooth uncertainty-slack coupling using adaptive slack scaling
    slack_centered = iqr_normalize(slack)
    unc_slack_coupling = np.where(
        slack < 0,
        uncertainty * (-slack) * (1.0 + PARAMS["uncertainty_slack_coupling"] * np.abs(slack_centered)),
        0.0
    )
    
    duration_total = min_exec_time + min_comm_time + eps
    duration_risk = duration_total * uncertainty * PARAMS["duration_risk_penalty"]

    # Criticality boost: only when slack <= 0 AND both rank & work above median
    is_tight_or_violated = (slack <= 0)
    rank_median = np.median(upward_rank) if N > 1 else np.mean(upward_rank)
    work_median = np.median(remaining_work) if N > 1 else np.mean(remaining_work)
    is_high_rank = (upward_rank >= rank_median)
    is_high_work = (remaining_work >= work_median)
    critical_gate = np.where(
        is_high_rank & is_high_work & is_tight_or_violated,
        PARAMS["criticality_boost"],
        1.0
    )

    # --- Smooth starvation mitigation: tanh-based urgency (replaces brittle threshold) ---
    wait_urgency_raw = np.where(
        is_tight_or_violated,
        ready_wait_time / (np.abs(slack) + eps),
        0.0
    )
    wait_score = np.tanh(PARAMS["wait_urgency_steepness"] * wait_urgency_raw)

    # --- Successor-work starvation mitigation: critical-path release awareness ---
    # Heuristic: prioritize tasks whose immediate successors have high remaining work
    # Since successors aren't provided, we approximate via remaining_work itself (local proxy),
    # but scaled by upward_rank to emphasize downstream impact
    successor_proxy = remaining_work * upward_rank
    successor_norm = iqr_normalize(successor_proxy)
    successor_starvation_score = np.where(
        is_tight_or_violated,
        -successor_norm * PARAMS["successor_work_weight"],
        0.0
    )

    # --- Lexicographic gate: non-DDL terms only active when slack > 0 ---
    slack_headroom_mask = np.where(slack > 0.0, 1.0, 0.0)

    # Host-load–aware energy scaling
    load_proxy = 1.0 + iqr_normalize(uncertainty)
    scaled_energy = min_incremental_energy * load_proxy

    # Energy efficiency score
    energy_per_sec = scaled_energy / duration_total
    energy_eff_score = iqr_normalize(energy_per_sec)

    # Slack-modulated energy decay: exponential suppression far from deadline
    slack_headroom_norm = np.clip((iqr_normalize(slack) + 1.0) / 2.0, 0.0, 1.0)
    energy_decay = np.exp(-PARAMS["energy_efficiency_ratio_weight"] * slack_headroom_norm)
    energy_final = slack_headroom_mask * energy_eff_score * energy_decay * PARAMS["energy_efficiency_ratio_weight"]

    # Rank scoring with linear slack interpolation
    rank_weight = 1.0 - (iqr_normalize(slack) + 1.0) / 2.0 * PARAMS["rank_slack_balance"]
    rank_score = slack_headroom_mask * (-iqr_normalize(upward_rank)) * rank_weight

    # Energy-uncertainty interaction with sigmoid gating
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = iqr_normalize(scaled_energy)
    unc_norm = iqr_normalize(uncertainty)
    energy_uncertainty_score = slack_headroom_mask * PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    score = (
        iqr_normalize(slack_score) +
        iqr_normalize(unc_slack_coupling) +
        iqr_normalize(duration_risk) +
        wait_score +
        successor_starvation_score
    )
    
    # Non-DDL terms only added when slack is strictly positive
    score += (
        energy_final +
        rank_score +
        energy_uncertainty_score
    )

    # Apply criticality boost multiplicatively
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
