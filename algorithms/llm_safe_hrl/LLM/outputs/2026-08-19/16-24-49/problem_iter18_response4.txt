def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with decoupled per-feature min-max normalization:
      - Replaces fragile joint MAD with robust, independent min-max clipping per feature using quantiles
      - Introduces DDL-protection gate: suppresses energy/efficiency terms when |slack| < ddl_protection_threshold * joint_scale
      - Converts successor-release from additive to multiplicative coupling: (upward_rank × remaining_work) × slack_gate × (1 + DDL-protection)
      - All normalizations are now feature-isolated: avoids scale leakage between slack, uncertainty, and duration
      - Bounded min-max uses quantile-based bounds for resilience to sparse/outlier distributions
      - DDL-protection gate ensures hard deadline constraints dominate energy optimization near deadlines
      - Uses only allowed literals {-2,-1,0,1,2} and no hidden state or randomness
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust per-feature min-max normalization using quantile bounds (replaces joint MAD)
    def robust_minmax_normalize(x, q_low=0.05, q_high=PARAMS["min_max_clip_quantile"]):
        x = np.asarray(x, dtype=float)
        if N <= 1:
            return np.zeros_like(x, dtype=float)
        x_min = np.quantile(x, q_low)
        x_max = np.quantile(x, q_high)
        range_val = np.maximum(x_max - x_min, eps)
        normalized = (x - x_min) / range_val
        return np.clip(normalized, 0.0, 1.0)

    # Normalize each feature independently to avoid scale coupling
    norm_slack = robust_minmax_normalize(slack, 0.05, 0.95)
    norm_uncertainty = robust_minmax_normalize(uncertainty, 0.05, 0.95)
    norm_duration = robust_minmax_normalize(min_exec_time + min_comm_time + eps, 0.05, 0.95)
    norm_energy = robust_minmax_normalize(min_incremental_energy, 0.05, 0.95)
    norm_rank = robust_minmax_normalize(upward_rank, 0.05, 0.95)
    norm_work = robust_minmax_normalize(remaining_work, 0.05, 0.95)
    norm_wait = robust_minmax_normalize(ready_wait_time, 0.05, 0.95)

    # Joint scale proxy for DDL-protection threshold (geometric mean of key ranges)
    joint_scale_proxy = (np.quantile(np.abs(slack), 0.9) + 
                         np.quantile(uncertainty, 0.9) + 
                         np.quantile(min_exec_time + min_comm_time + eps, 0.9)) / 3.0 + eps

    # --- DDL-critical base scores (always active, unconditionally dominant) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- DDL-protection gate: activates when slack is critically tight ---
    # Suppresses non-critical terms when |slack| is below threshold relative to joint scale
    ddl_protection_gate = np.where(
        np.abs(slack) < PARAMS["ddl_protection_threshold"] * joint_scale_proxy,
        1.0,
        0.0
    )

    # --- Multiplicative successor-release signal: embeds urgency into all regimes ---
    # Combines critical path importance with deadline proximity and protection
    rank_work_product = upward_rank * remaining_work
    norm_rank_work = robust_minmax_normalize(rank_work_product, 0.05, 0.95)
    # Sigmoid on normalized slack for smooth deadline proximity weighting
    slack_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (0.5 - norm_slack)))
    successor_release = norm_rank_work * slack_sigmoid * (1.0 + ddl_protection_gate * 0.5)

    # --- Smooth DDL-gate for non-DDL terms: sigmoid on normalized slack ---
    slack_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (0.5 - norm_slack)))

    # --- Non-DDL terms (smoothly weighted by slack_gate, suppressed under DDL-protection) ---
    energy_per_sec = min_incremental_energy / (min_exec_time + min_comm_time + eps)
    norm_energy_per_sec = robust_minmax_normalize(energy_per_sec, 0.05, 0.95)
    energy_eff_score = norm_energy_per_sec * PARAMS["energy_efficiency_ratio_weight"]

    # Rank balance via normalized slack sigmoid
    rank_score = -norm_rank * (
        PARAMS["rank_slack_balance"] * (1.0 - slack_gate) +
        (1.0 - PARAMS["rank_slack_balance"]) * slack_gate
    )

    # Energy-uncertainty interaction: both normalized
    energy_uncertainty_score = (norm_energy * norm_uncertainty * 
                                PARAMS["energy_uncertainty_interaction"] * 
                                (1.0 - ddl_protection_gate))

    # --- Wait-time anti-starvation: exponential decay using normalized slack ---
    wait_decay = np.exp(-PARAMS["wait_decay_steepness"] * np.abs(0.5 - norm_slack))
    wait_score = norm_wait * wait_decay

    # --- Assemble final score: smaller = higher priority ---
    # Base DDL-violation terms dominate
    score = (
        robust_minmax_normalize(slack_score, 0.05, 0.95) +
        robust_minmax_normalize(unc_slack_coupling, 0.05, 0.95) +
        robust_minmax_normalize(duration_risk, 0.05, 0.95)
    )
    
    # Non-DDL terms blended smoothly, but suppressed under DDL-protection
    non_ddl_terms = (
        energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )
    score += slack_gate * non_ddl_terms * (1.0 - ddl_protection_gate)

    # Add multiplicative successor-release (always active, enhances critical-path urgency)
    score += successor_release * PARAMS["criticality_boost"]

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
