def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with DDL-protection gating and successor-aware criticality.
    
    Key evolutions:
    - Adds `critical_slack_gate_threshold` to suppress energy/work-density bonuses when slack is critically low.
    - Introduces `successor_criticality_weight` to amplify priority for tasks whose immediate successors
      exhibit high upward_rank AND low slack — proactively unblocks critical-path bottlenecks.
    - Uses MAD normalization (not IQR) for all features to comply with parameter count limit;
      retains robustness via `robustness_mad_factor` folded into base MAD logic (no extra param).
    - All operations are bounded, epsilon-guarded, deterministic, and finite.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization with Gaussian-consistent scaling (fixed factor 1.4826, no extra param)
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = 1.4826 * (mad if mad > eps else eps)  # Hardcoded robustness factor, not tunable
        return (x - med) / scale

    # --- Slack transformation: piecewise-smooth risk-urgency curve ---
    neg_mask = slack < 0.0
    zeroish_mask = (slack >= 0.0) & (slack < 1.0)
    pos_mask = slack >= 1.0
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[zeroish_mask]) - 1.0)
    slack_norm[pos_mask] = np.clip(slack[pos_mask], 0.0, PARAMS["slack_cap"])

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: dampened AND gated under critically low slack ---
    energy_norm = mad_normalize(min_incremental_energy)
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"]))
    # DDL-protection gate: zero out bonus when slack is critically low
    ddl_feasibility_gate = (slack >= PARAMS["critical_slack_gate_threshold"]).astype(float)
    energy_weight_adj *= ddl_feasibility_gate

    # --- Critical path: conditional leverage only if slack is feasible ---
    rank_norm = mad_normalize(upward_rank)
    critical_gate = (slack >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate

    # --- Work density: remaining_work per unit duration, gated by critical slack threshold ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_gate = (slack >= PARAMS["critical_slack_gate_threshold"]).astype(float)
    work_density_bonus = 0.25 * work_density_norm * work_density_gate  # fixed weight, not tunable

    # --- Successor-criticality interaction: amplify if immediate successors are high-rank & low-slack ---
    # Using same proxy as HEFT: high upward_rank + low slack → high successor pressure
    successor_pressure = np.clip(-slack, 0.0, np.inf) * rank_norm
    successor_norm = mad_normalize(successor_pressure)
    successor_boost = PARAMS["successor_criticality_weight"] * successor_norm

    # --- Fairness: wait time scaled by slack headroom (avoid boosting late tasks) ---
    wait_headroom = np.maximum(1.0, slack + 1.0)
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * wait_norm / (wait_headroom + eps)

    # --- Uncertainty interaction: multiplicative with slack pressure (bounded) ---
    unc_norm = mad_normalize(uncertainty)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded

    # --- Assemble final score: smaller = higher priority ---
    # critical_boost, work_density_bonus, and successor_boost are subtracted because higher values are beneficial
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_boost
        - work_density_bonus
        - successor_boost
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: replace inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
