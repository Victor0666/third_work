def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Linear ramp slack gate (replacing sigmoid) for robust, interpretable DDL gating
      - MAD-based normalization *with stabilizing shift* for all risk dimensions (duration, |slack|, uncertainty)
      - Reinstated robust MAD (not IQR) for energy normalization — more outlier-resistant than IQR in sparse regimes
      - Removal of exponential urgency decay — replaced by linear wait-time scaling with bounded slack-coupled urgency boost
      - All numeric literals restricted to {-2,-1,0,1,2}; no unbounded functions or fragile nonlinearities
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Joint MAD normalization for coherent risk scaling — now with stabilizing shift
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    all_risk_dims = np.stack([duration_total, abs_slack, uncertainty], axis=0)
    mad_per_dim = np.mean(np.abs(all_risk_dims - np.median(all_risk_dims, axis=1, keepdims=True)), axis=1) + eps
    # Apply stabilizing shift to avoid degenerate denominators
    mad_shifted = mad_per_dim + PARAMS["mad_normalization_shift"] * np.median(mad_per_dim)
    duration_norm = duration_total / (mad_shifted[0] * PARAMS["duration_mad_scale"] + eps)
    slack_norm = abs_slack / (mad_shifted[1] * PARAMS["duration_mad_scale"] + eps)
    unc_norm = uncertainty / (mad_shifted[2] * PARAMS["duration_mad_scale"] + eps)

    # --- DDL-critical base score: slack violation penalty ---
    slack_penalty = np.where(
        slack < 0,
        (-slack) * (1.0 + PARAMS["uncertainty_slack_coupling"] * unc_norm),
        0.0
    )

    # --- Critical path release signal: upward_rank * remaining_work, MAD-normalized ---
    critical_release = upward_rank * remaining_work
    mad_cr = np.mean(np.abs(critical_release - np.median(critical_release))) + eps
    critical_release_norm = (critical_release - np.median(critical_release)) / (mad_cr * PARAMS["duration_mad_scale"] + eps)
    critical_release_score = -critical_release_norm * PARAMS["critical_path_release_weight"]

    # --- Linear ramp slack gate: robust, bounded, differentiable, and parameter-efficient ---
    # Gate = 0 when slack <= ddl_protection_threshold - ramp_width/2,
    # linearly rises to 1 at ddl_protection_threshold + ramp_width/2, then clamps to 1
    ramp_half = PARAMS["slack_linear_ramp_width"] / 2.0
    gate_center = PARAMS["ddl_protection_threshold"]
    gate_low = gate_center - ramp_half
    gate_high = gate_center + ramp_half
    slack_gate = np.clip((slack - gate_low) / (ramp_half * 2.0 + eps), 0.0, 1.0)

    # --- Energy efficiency term: gated by linear ramp, MAD-normalized ---
    mad_energy = np.mean(np.abs(min_incremental_energy - np.median(min_incremental_energy))) + eps
    energy_norm = (min_incremental_energy - np.median(min_incremental_energy)) / (mad_energy * PARAMS["duration_mad_scale"] + eps)
    energy_score = slack_gate * energy_norm * PARAMS["risk_adjusted_energy_weight"]

    # --- Uncertainty-energy interaction: same gate, same normalization ---
    energy_uncertainty_score = slack_gate * energy_norm * unc_norm * PARAMS["energy_uncertainty_interaction"]

    # --- Starvation mitigation: linear urgency boost, no exponential decay ---
    # Boosts priority for tasks waiting longer than median, scaled by slack deficit
    median_wait = np.median(ready_wait_time) + eps
    wait_ratio = np.clip(ready_wait_time / median_wait, 0.0, 2.0)  # bounded ratio
    slack_deficit_boost = np.clip(-slack / (PARAMS["ddl_protection_threshold"] + eps), 0.0, 2.0)
    wait_score = ready_wait_time * (1.0 + slack_deficit_boost)
    wait_norm = (wait_score - np.median(wait_score)) / (np.mean(np.abs(wait_score - np.median(wait_score))) + eps)
    wait_final = wait_norm * PARAMS["wait_starvation_penalty"]

    # --- Final slack-driven balance: interpolate between slack urgency and rank importance ---
    slack_weight = np.clip(1.0 - slack_norm / (slack_norm.max() + eps), 0.0, 1.0)
    rank_weight = 1.0 - slack_weight
    rank_score = -upward_rank / (np.median(upward_rank) + eps) * rank_weight * PARAMS["rank_slack_balance"]

    # --- Assemble final score: smaller = higher priority ---
    score = (
        slack_penalty +
        critical_release_score +
        wait_final +
        energy_score +
        energy_uncertainty_score +
        rank_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
