def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: piecewise slack gating + IQR-normalized features + uncertainty-coupled criticality.
    
    Key innovations:
    - Uses robust IQR-based normalization (not mean/std) to suppress outlier distortion.
    - Applies smooth, bounded slack penalty: exponential decay for positive slack, linear penalty for negative.
    - Introduces slack-uncertainty interaction: high uncertainty * tight slack → boost criticality.
    - Anti-starvation via saturating exponential wait bonus (avoids unbounded growth).
    - Energy prioritization only after slack feasibility is confirmed (soft feasibility gate).
    - All operations are finite, epsilon-guarded, and deterministic.
    """
    eps = PARAMS["epsilon"]
    
    # Safe IQR-based normalization: robust to outliers, always finite
    def iqr_normalize(x):
        x = np.asarray(x, dtype=float)
        q1, q3 = np.quantile(x, [0.25, 0.75], method='midpoint')
        iqr = q3 - q1 + eps
        med = np.median(x)
        return (x - med) / iqr
    
    # Slack handling: prioritize strongly near/under deadline; smooth transition at zero
    # For slack <= 0: linear penalty (more negative → higher priority → lower score)
    # For slack > 0: decaying exponential priority loss (less urgent → higher score)
    slack_raw = np.asarray(slack, dtype=float)
    slack_penalty = np.where(
        slack_raw <= 0,
        PARAMS["slack_penalty_scale"] * (-slack_raw),
        PARAMS["slack_penalty_scale"] * np.exp(-PARAMS["slack_sensitivity"] * slack_raw)
    )
    
    # Uncertainty-slack interaction: amplifies criticality when both are high
    # Scaled to [0,1] range first to avoid explosion
    unc_norm = np.clip(uncertainty / (np.maximum(np.abs(uncertainty).max(), eps) + eps), 0, 1)
    slack_pressure_norm = np.clip((np.maximum(-slack_raw, 0)) / (np.maximum(np.abs(slack_raw).max(), eps) + eps), 0, 1)
    interaction_gate = unc_norm * slack_pressure_norm
    
    # Criticality boost: upward_rank weighted by interaction_gate
    rank_raw = np.asarray(upward_rank, dtype=float)
    criticality = rank_raw * (1.0 + PARAMS["criticality_boost"] * interaction_gate)
    
    # Anti-starvation: saturating exponential bonus for long-waiting tasks
    wait_raw = np.asarray(ready_wait_time, dtype=float)
    wait_bonus = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait_raw)  # [0,1), increases then saturates
    
    # Robust feature normalizations (all finite & bounded)
    norm_slack = iqr_normalize(slack_penalty)
    norm_energy = iqr_normalize(min_incremental_energy)
    norm_crit = iqr_normalize(criticality)
    norm_wait = iqr_normalize(wait_bonus)
    norm_duration = iqr_normalize(min_exec_time + min_comm_time)
    
    # Final score: smaller = higher priority
    # Prioritize slack penalty most strongly, then energy (only if slack feasible),
    # then critical path, duration, and anti-starvation
    score = (
        1.0 * norm_slack
        + PARAMS["energy_norm_factor"] * np.where(slack_raw > 0, norm_energy, 0.0)
        + 0.8 * norm_crit
        + PARAMS["duration_balance"] * norm_duration
        - 0.6 * norm_wait  # subtract because wait_bonus improves priority → lowers score
    )
    
    # Ensure finite output: replace inf/nan with large finite fallback
    score = np.nan_to_num(score, nan=np.finfo(float).max, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
