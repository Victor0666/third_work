def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's smoothness and Parent 1's critical-path congestion awareness,
       enhanced with clipped tanh gating, MAD-normalized work-density as core term, and unified risk suppression.
    
    Structural improvements:
    - Replaces all sigmoid gates with `np.clip(np.tanh(x), 0, 1)` to enforce zero-saturated, monotonic risk suppression.
    - Introduces `work_density = remaining_work / (min_exec_time + min_comm_time + eps)` as a *standalone priority term*,
      normalized via MAD and weighted — not just a bonus subtracted from score.
    - Removes empirical slack bounds; uses MAD-normalized `slack` directly in all interactions for robustness.
    - Unifies uncertainty gating: `unc_gate = np.clip(np.tanh(steepness * uncertainty), 0, 1)` applied consistently
      to *all* non-deadline terms (energy, duration, wait, work-density) to suppress risky optimization.
    - Retains smooth slack penalty `max(0,-slack)**p` (Parent 2) but adds Parent 1’s `upward_rank × remaining_work`
      interaction scaled by `unc_gate * (1.0 - slack_sigmoid)` as explicit critical-path congestion signal.
    - Anti-starvation `wait_boost` now gated by both `unc_gate` and `slack_headroom_mask` (Parent 1 style) for DDL-safe fairness.
    - All normalizations use MAD with N=1 fallback; no percentile or IQR dependencies.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs without mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)
    
    # Robust MAD normalization: handles N=1 via fallback to zero-mean
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        denom = mad + eps
        normalized = (x - center) / denom
        # Clip to [-2, 2] to prevent outlier explosion — uses allowed literals
        return np.clip(normalized, -2.0, 2.0)

    # --- Unified uncertainty gate: clipped tanh for zero-saturated, monotonic suppression ---
    unc_gate = np.clip(np.tanh(PARAMS["uncertainty_tanh_steepness"] * uncertainty), 0.0, 1.0)

    # --- Smooth slack penalty (Parent 2) + deadline pressure for coupling ---
    slack_penalty = np.maximum(0.0, -slack) ** PARAMS["slack_penalty_exponent"]
    deadline_pressure = np.maximum(0.0, -slack)

    # --- Duration risk: exec+comm normalized, then modulated by clipped tanh gate ---
    duration_total = min_exec_time + min_comm_time + eps
    duration_norm = mad_normalize(duration_total)
    duration_risk = duration_norm * (1.0 + unc_gate * PARAMS["duration_risk_penalty"])

    # --- Energy efficiency ratio: marginal energy per second, normalized and gated ---
    energy_per_sec = min_incremental_energy / (duration_total + eps)
    energy_eff_norm = mad_normalize(energy_per_sec)
    energy_eff_score = energy_eff_norm * PARAMS["energy_efficiency_ratio_weight"] * (1.0 + unc_gate)

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    unc_slack_coupling = mad_normalize(uncertainty * deadline_pressure) * PARAMS["uncertainty_slack_coupling"]

    # --- Slack-aware normalized criticality boost: upward_rank scaled by clipped negative slack ---
    rank_norm = mad_normalize(upward_rank)
    clipped_slack = np.maximum(0.0, -slack)
    critical_bonus = rank_norm * (1.0 + PARAMS["slack_aware_criticality_scale"] * clipped_slack)

    # --- Work-density as standalone term: remaining_work / duration, MAD-normalized, weighted, and gated ---
    work_density = remaining_work / (duration_total + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_term = PARAMS["work_density_weight"] * work_density_norm * (1.0 + unc_gate)

    # --- Energy-uncertainty interaction: tanh-gated multiplicative coupling ---
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_gate

    # --- Base energy fallback: direct normalized energy, also gated ---
    energy_base_score = mad_normalize(min_incremental_energy) * (1.0 + unc_gate)

    # --- Fairness: tanh-scaled wait boost, DDL-safe via slack_headroom_mask and unc_gate ---
    slack_headroom_mask = np.where(slack > 0.0, 1.0, 0.0)
    wait_norm = mad_normalize(ready_wait_time)
    wait_boost = PARAMS["tanh_bias"] * (1.0 + np.tanh(PARAMS["wait_fairness_gain"] * wait_norm)) * slack_headroom_mask * unc_gate

    # --- Critical-path congestion signal: upward_rank × remaining_work, gated by slack safety and uncertainty ---
    rank_work_product = upward_rank * remaining_work
    # Sigmoid gate on slack: soft transition from violation to safety
    slack_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_tanh_steepness"] * (slack + 1.0)))
    # Strength decays as slack increases; capped at 1.0
    congestion_strength = np.clip(1.0 - slack_sigmoid, 0.0, 1.0)
    congestion_term = mad_normalize(rank_work_product) * congestion_strength * unc_gate * PARAMS["slack_aware_criticality_scale"]

    # --- Assemble final score: smaller = higher priority ---
    # Slack dominates; all penalties added, bonuses subtracted
    score = (
        mad_normalize(slack_penalty) +
        duration_risk +
        energy_eff_score +
        unc_slack_coupling +
        energy_uncertainty_score +
        energy_base_score +
        wait_boost -
        critical_bonus -
        work_density_term -
        congestion_term
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
