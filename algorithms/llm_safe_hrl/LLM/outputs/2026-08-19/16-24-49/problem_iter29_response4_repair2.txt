def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's lexicographic DDL gating and percentile normalization with Parent 1's bounded exponential urgency.
      - Strict lexicographic separation: all non-DDL terms fully suppressed when slack <= slack_min_bound.
      - Robust clipped percentile normalization replaces MAD, avoiding degeneracy in sparse sets (N ≤ 3).
      - Novel hybrid urgency term: exponential decay over *bounded slack distance*, applied only to critical-path release score, preserving DDL dominance.
      - Criticality boost now gated jointly by slack violation, high upward_rank (> median), and high remaining_work (> median) — same as Parent 2.
      - Anti-starvation wait-term uses robust linear form `ready_wait_time / (|slack| + 1)` but only in feasible region (slack > slack_min_bound).
      - All numeric literals strictly limited to {-2,-1,0,1,2}; all tunables accessed via PARAMS; deterministic and finite-output guaranteed.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Clipped percentile normalization: q_low and symmetric upper bound = 2*median - q_low (2 is allowed literal)
    def percentile_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        q_low = np.percentile(x, PARAMS["percentile_clip_low"])
        med = np.median(x)
        q_high = 2.0 * med - q_low  # uses allowed literal 2.0
        denom = q_high - q_low + eps
        normalized = (x - q_low) / denom
        return np.clip(normalized, 0.0, 1.0)

    # --- DDL-critical base scores (always active, lexicographically dominant) ---
    # Bounded exponential urgency for slack-modulated critical-path release (hybrid innovation)
    slack_distance = np.clip(PARAMS["slack_min_bound"] - slack, 0.0, PARAMS["bounded_urgency_distance_cap"])
    urgency_factor = np.exp(-slack_distance / (PARAMS["criticality_boost"] + eps))  # reuse criticality_boost as rate proxy
    # Higher urgency → higher priority → lower score → use negative sign
    slack_rank_score = -upward_rank * urgency_factor * PARAMS["criticality_boost"]

    # Standard negative-slack penalty: (-slack)**exponent, zero otherwise
    slack_score = np.where(
        slack < PARAMS["slack_min_bound"],
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_total = min_exec_time + min_comm_time + eps
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]
    # Escalated duration risk when slack is deeply negative
    slack_ramp = np.clip(
        (-slack - PARAMS["slack_min_bound"]) / (0.0 - PARAMS["slack_min_bound"] + eps),
        0.0, 1.0
    )
    escalated_duration_risk = duration_risk * (1.0 + slack_ramp)

    # --- Criticality boost: activated only under strict DDL violation (slack ≤ slack_min_bound), requires high rank AND high remaining work ---
    is_violated = (slack <= PARAMS["slack_min_bound"])
    rank_median = np.median(upward_rank) if N > 1 else np.mean(upward_rank)
    work_median = np.median(remaining_work) if N > 1 else np.mean(remaining_work)
    is_high_rank = (upward_rank >= rank_median)
    is_high_work = (remaining_work >= work_median)
    critical_gate = np.where(
        is_high_rank & is_high_work & is_violated,
        PARAMS["criticality_boost"],
        1.0
    )

    # --- Lexicographic gate: non-DDL terms only active when slack > slack_min_bound (strictly feasible region) ---
    feasible_mask = np.where(slack > PARAMS["slack_min_bound"], 1.0, 0.0)

    # --- Non-DDL terms (only active when slack > slack_min_bound) ---
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = percentile_normalize(energy_per_sec)

    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -percentile_normalize(upward_rank) * weight_rank

    unc_projected = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = percentile_normalize(min_incremental_energy)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_projected

    # Anti-starvation: linear wait-time term gated by strict feasibility
    wait_normalized = np.where(
        feasible_mask > 0.0,
        ready_wait_time / (np.abs(slack) + 1.0),
        0.0
    )
    wait_score = percentile_normalize(wait_normalized)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute and dominate
    score = (
        percentile_normalize(slack_score) +
        percentile_normalize(unc_slack_coupling) +
        percentile_normalize(escalated_duration_risk) +
        percentile_normalize(slack_rank_score)
    )
    
    # Non-DDL terms only added when slack is strictly above slack_min_bound (strict lexicographic separation)
    score += feasible_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Apply criticality boost multiplicatively to DDL-violating tasks only
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
