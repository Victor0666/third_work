def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """
    Hybrid priority rule: combines Parent 2's relative-MAD normalization and fairness-gated energy
    with Parent 1's rank-energy interaction penalty and unified urgency-criticality gating.
    Key innovations:
      - Replaces separate urgency/criticality branches with *single unified slack-gated rank-energy penalty*
        that activates only under tight deadlines, avoiding bottleneck scheduling.
      - Uses relative-MAD normalization for *all* features (not just slack/uncertainty) for consistent outlier robustness.
      - Introduces *normalized slack-pressure ratio* (|slack| / (|slack| + ready_wait_time + eps)) as fairness-aware
        headroom proxy — more stable than raw ratio under zero/near-zero wait time.
      - Retains tanh-bounded slack pressure for uncertainty coupling but applies it *after* fairness gating.
      - All numeric literals are strictly in {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(np.float64)
    
    # Relative-MAD normalization: robust, preserves ordinal structure across all features
    def relative_mad_normalize(x):
        x = np.asarray(x, dtype=np.float64)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = np.where(mad > eps, mad, eps)
        return (x - med) / scale
    
    # Normalize all inputs
    norm_slack = relative_mad_normalize(slack)
    norm_energy = relative_mad_normalize(min_incremental_energy)
    norm_duration = relative_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = relative_mad_normalize(upward_rank)
    norm_wait = relative_mad_normalize(ready_wait_time)
    norm_uncert = relative_mad_normalize(uncertainty)
    
    # --- Unified slack transformation: smooth, bounded, differentiable ---
    slack_arr = np.asarray(slack, dtype=np.float64)
    neg_mask = slack_arr < 0.0
    pos_mask = slack_arr >= 0.0
    
    slack_norm = np.zeros_like(slack_arr)
    # Negative slack → linear lateness penalty
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack_arr[neg_mask])
    # Positive slack → sigmoid urgency (1 - exp(-k*slack)) capped & linearly decayed beyond slack_cap
    urgency_pos = 1.0 - np.exp(-PARAMS["slack_sigmoid_k"] * slack_arr[pos_mask])
    urgency_scaled = PARAMS["slack_sigmoid_k"] * urgency_pos  # reuse k as scale for simplicity & identifiability
    slack_clipped = np.clip(slack_arr[pos_mask], 0.0, PARAMS["slack_cap"])
    linear_decay = slack_clipped / (PARAMS["slack_cap"] + eps)
    slack_norm[pos_mask] = urgency_scaled * linear_decay + (1.0 - linear_decay) * slack_clipped
    
    # --- Fairness headroom: stable ratio |slack| / (|slack| + ready_wait_time + eps) ∈ [0,1] ---
    # This avoids division-by-zero and extreme values when wait_time ≈ 0 or slack ≈ 0
    headroom_ratio = np.abs(slack_arr) / (np.abs(slack_arr) + ready_wait_time + eps)
    # Sigmoid gate: high energy weight when headroom_ratio ≈ 1 (i.e., wait_time << |slack|)
    fairness_gate = 1.0 / (1.0 + np.exp(-PARAMS["energy_fairness_gate_k"] * (headroom_ratio - PARAMS["fairness_headroom_offset"])))
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * fairness_gate
    
    # --- Critical path leverage: only active when slack >= 0 ---
    critical_gate = (slack_arr >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * norm_rank * critical_gate
    
    # --- Anti-starvation: relative wait score, headroom-scaled ---
    wait_headroom = np.maximum(1.0, np.abs(slack_arr) + 1.0)
    wait_score = PARAMS["wait_fairness_gain"] * norm_wait / (wait_headroom + eps)
    
    # --- Uncertainty coupling: tanh-bounded slack pressure × normalized uncertainty × fairness gate ---
    slack_pressure = np.clip(-slack_arr, 0.0, np.inf)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * norm_uncert * slack_pressure_bounded * fairness_gate
    
    # --- Rank-energy interaction penalty: activated only under deadline pressure (slack < 0) ---
    # Discourages scheduling high-energy, high-rank tasks when lateness risk exists
    rank_energy_penalty = PARAMS["rank_energy_interaction"] * norm_energy * norm_rank * (1.0 - critical_gate)
    
    # --- Assemble final score: smaller = higher priority ---
    score = (
        1 * slack_norm
        + PARAMS["duration_balance"] * norm_duration
        + energy_weight_adj * norm_energy
        - critical_boost
        + wait_score
        + uncertainty_amplifier
        + rank_energy_penalty
    )
    
    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
