def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's DDL-first robustness with Parent 1's resilience-aware interactions:
      - Retains Parent 2's conditional DDL-protection gate and clipped empirical slack scaling for stability.
      - Integrates Parent 1's *joint feasibility-inspired* wait-term: `ready_wait_time / (|slack| + 1)` scaled by adaptive factor,
        but now gated by empirical slack safety (`slack_scaled > wait_safety_threshold`) to avoid unfair boosts near deadlines.
      - Replaces Parent 2's successor-release interaction with Parent 1's *work-density bonus*: `remaining_work / duration`,
        normalized and multiplied by joint slack/uncertainty gate (`slack_scaled * (1 - unc_norm)`) — favors throughput-critical
        tasks only when slack is positive and uncertainty low.
      - Adds robustness: all MAD normalizations use [-2, 2] clipping and explicit N=1 handling.
      - Introduces novel *uncertainty-aware energy efficiency*: replaces simple `energy_per_sec` with `min_incremental_energy / (duration * (1 + uncertainty))`,
        capturing risk-adjusted efficiency without introducing new parameters.
      - All numeric literals are in {-2,-1,0,1,2}; no unbounded loops or state; fully deterministic and finite.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust clipped MAD normalization: median ± MAD, with epsilon-guarded scale and [-2, 2] bounds
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x) if N > 1 else np.mean(x)
        mad = np.median(np.abs(x - med)) if N > 1 else eps
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return np.clip((x - med) / scale, -2.0, 2.0)

    # --- Empirical slack scaling: bounded, interpretable, stable ---
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)

    # --- Slack scoring: power penalty for violation, zero otherwise ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )

    # --- Conditional DDL-protection gate: only boost critical path when safe conditions hold ---
    median_unc = np.median(uncertainty) if N > 1 else np.mean(uncertainty)
    median_energy = np.median(min_incremental_energy) if N > 1 else np.mean(min_incremental_energy)
    ddl_protection_gate = (
        (slack <= 0.0) &
        (uncertainty <= median_unc + eps) &
        (min_incremental_energy <= median_energy + eps)
    ).astype(float)

    # --- Critical-path release boost: simplified & stabilized ---
    rank_norm = mad_normalize(upward_rank)
    critical_release_boost = (
        ddl_protection_gate *
        rank_norm *
        (1.0 + PARAMS["criticality_boost"] * (1.0 - slack_scaled))
    )

    # --- Wait-time anti-starvation: adaptive and safety-gated ---
    # Only activate fairness when slack is reasonably safe (slack_scaled > wait_safety_threshold)
    wait_base = ready_wait_time / (np.abs(slack) + 1.0)
    wait_gated = wait_base * (slack_scaled > PARAMS["wait_safety_threshold"]).astype(float)
    wait_score = mad_normalize(wait_gated) * PARAMS["wait_slack_adaptive_scale"]

    # --- Uncertainty-aware energy efficiency: joules per second per unit uncertainty ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_efficiency = min_incremental_energy / (duration_total * (1.0 + uncertainty))
    energy_eff_norm = mad_normalize(energy_efficiency)

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]

    # --- Duration risk penalty: long duration × uncertainty ---
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Rank-slack balance: minimal influence unless slack critical ---
    weight_rank = PARAMS["rank_slack_balance"] * (1.0 - slack_scaled)
    rank_score = -rank_norm * weight_rank

    # --- Work-density bonus: favors throughput-critical tasks under safe conditions ---
    work_density = np.divide(remaining_work, duration_total + eps)
    work_density_norm = mad_normalize(work_density)
    # Joint gate: active only when slack is positive AND uncertainty is low
    unc_norm = mad_normalize(uncertainty)
    work_density_gate = slack_scaled * (1.0 - np.clip(unc_norm, 0.0, 1.0))
    work_density_bonus = work_density_norm * work_density_gate

    # --- Hard-bounded uncertainty gate for energy-uncertainty interaction ---
    unc_threshold_gate = (uncertainty <= median_unc + eps).astype(float)
    energy_norm = mad_normalize(min_incremental_energy)
    energy_uncertainty_score = (
        PARAMS["energy_uncertainty_interaction"] *
        energy_norm *
        unc_norm *
        unc_threshold_gate
    )

    # --- Assemble final score: smaller = higher priority ---
    # Penalties added; beneficial terms subtracted (critical_release_boost, work_density_bonus)
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_norm +
        rank_score +
        energy_uncertainty_score +
        wait_score -
        critical_release_boost -
        work_density_bonus
    )
    
    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
