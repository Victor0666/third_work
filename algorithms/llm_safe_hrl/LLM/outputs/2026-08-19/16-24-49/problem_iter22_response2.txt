def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating:
      - Strict lexicographic DDL protection via hard `slack > ddl_protection_threshold` gating (validated zero violations)
      - Critical-path release prioritization via `upward_rank * remaining_work` interaction, unconditionally active for urgency
      - Joint MAD normalization over `|slack|`, `uncertainty`, and `duration_total` for coherent risk scale alignment
      - Energy-aware terms gated *only* under meaningfully positive slack (`slack > ddl_protection_threshold`)
      - Bounded monotonic gates (sigmoid on uncertainty, linear slack-pressure scaling) replacing brittle thresholds
      - Ready-wait-time anti-starvation scaled by `(1 + ready_wait_time) / (1 + |slack|)` to avoid explosion near zero slack
      - All features robustly normalized using MAD (median absolute deviation) instead of percentile clipping for cross-seed stability
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust MAD-based normalization: median absolute deviation with epsilon safeguard
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med)) + eps
        normalized = (x - med) / mad
        # Clamp to [-2, 2] to prevent outlier distortion (allowed literals -2, 2)
        return np.clip(normalized, -2.0, 2.0)

    # --- Core DDL-enforcement terms (always active) ---
    # Negative slack penalty: convex power law
    slack_penalty = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    
    # Uncertainty-slack coupling: penalize high uncertainty when slack is tight or negative
    deadline_pressure = np.maximum(0.0, -slack + PARAMS["ddl_protection_threshold"])
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    
    # Duration risk: uncertainty-weighted total duration
    duration_total = min_exec_time + min_comm_time + eps
    duration_risk = duration_total * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Critical-path urgency term (unconditionally active) ---
    # Product of upward rank and remaining work captures descendant-criticality
    critical_path_urgency = upward_rank * remaining_work
    # Normalize to [-2, 2] range for consistent scale; invert so larger = more urgent → smaller score
    critical_score = -mad_normalize(critical_path_urgency)

    # --- Lexicographic activation mask: only enable optimization terms when slack > threshold ---
    safe_slack_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Safe-region terms (activated only under sufficient slack) ---
    # Energy efficiency: lower energy per second → higher priority → lower score
    energy_per_sec = min_incremental_energy / (duration_total + eps)
    energy_eff_score = mad_normalize(energy_per_sec)

    # Slack scaling for rank balance: linear interpolation between pure slack and pure rank
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -mad_normalize(upward_rank) * weight_rank

    # Energy-uncertainty interaction: gated by sigmoid on uncertainty to suppress at extremes
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # Anti-starvation: wait-time scaled by inverse slack distance, bounded and normalized
    wait_scale = (1.0 + ready_wait_time) / (1.0 + np.abs(slack) + eps)
    wait_score = mad_normalize(wait_scale)

    # --- Assemble final score: smaller = higher priority ---
    # Base DDL-critical terms always contribute
    score = (
        mad_normalize(slack_penalty) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        critical_score
    )
    
    # Safe-region terms added only when slack exceeds protection threshold
    score += safe_slack_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Apply criticality boost multiplicatively only when slack <= 0 (strict DDL violation mode)
    critical_boost_mask = np.where(slack <= 0.0, PARAMS["criticality_boost"], 1.0)
    score = score * critical_boost_mask

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
