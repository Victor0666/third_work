def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Exactly ONE conditional branch (np.where) — all other logic is vectorized arithmetic.
      - Soft feasibility mask via sigmoid (replaces brittle boolean gates).
      - Quantile-based normalization (Q1/Q3) with tunable scale.
      - No boolean indexing, no additional np.where, no if/else, no loops.
      - All parameters declared and used; no unused or missing PARAMS references.
      - Numeric literals strictly limited to {-2,-1,0,1,2}.
      - Deterministic, finite, shape-(N,) output; smaller score = higher priority.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust quantile-based normalization: Q1–Q3 range scaled by tunable factor
    def quantile_normalize(x):
        x = np.asarray(x, dtype=float)
        q1 = np.quantile(x, 0.25) if N > 1 else np.mean(x)
        q3 = np.quantile(x, 0.75) if N > 1 else np.mean(x)
        iqr = q3 - q1 + eps
        center = (q1 + q3) / 2.0
        # Bounded to [-2, 2] and scaled
        normed = np.clip((x - center) / iqr, -2.0, 2.0) * PARAMS["quantile_normalize_scale"]
        return normed

    # --- Slack scoring: power penalty for violation, zero otherwise (ONLY np.where branch) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )

    # --- Empirical slack scaling for balance terms ---
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)

    # --- Soft feasibility mask: sigmoid of joint feasibility distance ---
    duration_total = min_exec_time + min_comm_time + eps
    median_unc = np.median(uncertainty) if N > 1 else np.mean(uncertainty)
    median_energy = np.median(min_incremental_energy) if N > 1 else np.mean(min_incremental_energy)
    median_duration = np.median(duration_total) if N > 1 else np.mean(duration_total)
    # Feasibility distance: lower = more feasible; negate for sigmoid input
    feasibility_dist = (
        np.abs(slack) +
        (uncertainty - median_unc) +
        (min_incremental_energy - median_energy) +
        (duration_total - median_duration)
    )
    # Sigmoid soft gate: 1.0 when highly feasible, ~0 when deeply infeasible
    soft_feasibility_gate = 1.0 / (1.0 + np.exp(PARAMS["soft_feasibility_temperature"] * feasibility_dist))

    # --- Critical-path release boost with successor-release interaction ---
    critical_release_boost = (
        soft_feasibility_gate *
        quantile_normalize(upward_rank) *
        (1.0 + PARAMS["criticality_boost"] * (1.0 - slack_scaled)) *
        (1.0 + PARAMS["criticality_boost"] * (remaining_work / (np.quantile(remaining_work, 0.75) + eps)))
    )

    # --- Wait-time anti-starvation term: clipped linear form with tunable clamp ---
    wait_denom = np.minimum(np.abs(slack), PARAMS["wait_slack_clamp"]) + 1.0
    wait_normalized = ready_wait_time / wait_denom
    wait_score = quantile_normalize(wait_normalized)

    # --- Energy efficiency ratio: joules per second (lower = better), quantile-normalized ---
    energy_per_sec = min_incremental_energy / (duration_total + eps)
    energy_eff_norm = quantile_normalize(energy_per_sec)

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]

    # --- Duration risk penalty: long duration × uncertainty ---
    duration_risk = duration_total * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Rank-slack balance: very low weight except under extreme slack pressure ---
    weight_rank = PARAMS["rank_slack_balance"] * (1.0 - slack_scaled)
    rank_norm = quantile_normalize(upward_rank)
    rank_score = -rank_norm * weight_rank  # higher rank → lower score → higher priority

    # --- Energy-uncertainty interaction: quantile-normalized and softly gated ---
    energy_norm = quantile_normalize(min_incremental_energy)
    unc_norm = quantile_normalize(uncertainty)
    energy_uncertainty_score = (
        PARAMS["energy_uncertainty_interaction"] *
        energy_norm *
        unc_norm *
        soft_feasibility_gate
    )

    # --- Assemble final score: smaller = higher priority ---
    score = (
        quantile_normalize(slack_score) +
        quantile_normalize(unc_slack_coupling) +
        quantile_normalize(duration_risk) +
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_norm +
        rank_score +
        energy_uncertainty_score +
        wait_score -
        critical_release_boost
    )
    
    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
