def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaces brittle `slack > 0.0` lexicographic gate with smooth, differentiable sigmoid feasibility gate: 
        `1 / (1 + exp(-steepness * slack))`, enabling gradient-based optimization and robust interpolation.
      - Introduces load-aware anti-starvation term: activated only when ready_wait_time > percentile-threshold of wait times,
        coupled with min_incremental_energy to prioritize energy-efficient execution for long-waiting tasks — but *only* when slack permits.
      - Switches from percentile to rank-based (ordinal) normalization: computes rank order then maps to [0,1] via `(rank - 1) / (N - 1)` for N>1,
        eliminating degeneracy in flat/duplicate distributions while preserving monotonicity and boundedness.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Rank-based normalization: monotonic, bounded [0,1], immune to duplicates/flatness
    def rank_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        # Compute descending rank (largest gets rank 1), then map to [0,1]
        sorted_idx = np.argsort(-x)  # descending order
        ranks = np.empty_like(sorted_idx, dtype=float)
        ranks[sorted_idx] = np.arange(1, N + 1)
        return (ranks - 1.0) / (N - 1.0)  # maps rank 1→0.0, rank N→1.0

    # --- DDL-critical base scores (always active) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Criticality boost: activated only under DDL pressure (slack <= 0), requires high rank AND high remaining work ---
    is_tight_or_violated = (slack <= 0)
    rank_median = np.median(upward_rank) if N > 1 else np.mean(upward_rank)
    work_median = np.median(remaining_work) if N > 1 else np.mean(remaining_work)
    is_high_rank = (upward_rank >= rank_median)
    is_high_work = (remaining_work >= work_median)
    critical_gate = np.where(
        is_high_rank & is_high_work & is_tight_or_violated,
        PARAMS["criticality_boost"],
        1.0
    )

    # --- Smooth feasibility gate: replaces hard `slack > 0.0` with differentiable sigmoid ---
    # Enables continuous optimization and reduces threshold brittleness
    feasibility_gate = 1.0 / (1.0 + np.exp(-PARAMS["feasibility_sigmoid_steepness"] * slack))

    # --- Non-DDL terms (activated smoothly as slack increases) ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = rank_normalize(energy_per_sec)

    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -rank_normalize(upward_rank) * weight_rank

    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = rank_normalize(min_incremental_energy)
    unc_norm = rank_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Load-aware anti-starvation: activates only when wait time exceeds adaptive threshold ---
    # Uses quantile instead of median to avoid sensitivity to outliers
    wait_threshold = np.percentile(ready_wait_time, PARAMS["load_aware_wait_threshold"] * 100.0) if N > 1 else np.mean(ready_wait_time)
    load_aware_mask = np.where(ready_wait_time > wait_threshold, 1.0, 0.0)
    # Coupled with energy efficiency to favor low-energy execution for starved tasks
    load_aware_score = load_aware_mask * energy_eff_score * feasibility_gate

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        rank_normalize(slack_score) +
        rank_normalize(unc_slack_coupling) +
        rank_normalize(duration_risk)
    )
    
    # Non-DDL terms scaled smoothly by feasibility_gate (not binary mask)
    score += feasibility_gate * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        load_aware_score
    )

    # Apply criticality boost multiplicatively
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
