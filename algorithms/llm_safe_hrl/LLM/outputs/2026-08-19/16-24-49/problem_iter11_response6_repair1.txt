def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """
    Hybrid priority rule combining Parent 2's robust Q1/Q3 normalization and piecewise slack modeling
    with Parent 1's explicit fairness gating and joint feasibility awareness.
    Key improvements:
      - Adds bounded, threshold-gated fairness term: saturating linear boost above wait_threshold in normalized wait time,
        preserving anti-starvation without over-parameterizing.
      - Replaces triple-gated critical boost with dual-gated (slack >= 0 ∧ uncertainty <= median) + upward_rank interaction,
        removing brittle energy-median coupling that degraded under fuzzy energy variance.
      - Introduces work-density gating using joint slack/uncertainty feasibility (tanh(slack/(uncertainty+eps))) instead of binary slack gate,
        enabling smooth ramp-down under risk.
      - Retains multiplicative uncertainty coupling for energy and urgency, but applies it *after* normalization for stability.
      - All parameters used; no numeric literals beyond {-2,-1,0,1,2}; fully deterministic and finite.
    """
    eps = PARAMS["q1_q3_epsilon"]
    finfo = np.finfo(np.float64)
    
    # Cast and sanitize inputs — ensure no NaN/inf persists
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=np.float64), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=np.float64), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust Q1/Q3 normalization: uses interquartile range for stability under skew/small-N
    def q1_q3_normalize(x):
        x = np.asarray(x, dtype=np.float64)
        q1 = np.quantile(x, PARAMS["quantile_q1"])
        q3 = np.quantile(x, PARAMS["quantile_q3"])
        iqr = q3 - q1
        scale = iqr if iqr > eps else eps
        return (x - np.median(x)) / scale

    # --- Slack transformation: piecewise-smooth, bounded ---
    neg_mask = slack < 0.0
    tight_mask = (slack >= 0.0) & (slack <= PARAMS["slack_cap"])
    loose_mask = slack > PARAMS["slack_cap"]
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])  # linear penalty
    slack_norm[tight_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[tight_mask]) - 1.0)  # saturating urgency
    slack_norm[loose_mask] = PARAMS["slack_urgency_scale"] * (np.exp(PARAMS["slack_cap"]) - 1.0)  # flat decay cap

    # --- Duration: exec + comm, Q1/Q3 normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = q1_q3_normalize(duration)

    # --- Energy: monotonic uncertainty-weighted (not gated) ---
    unc_norm = q1_q3_normalize(uncertainty)
    # Smoothly scale energy impact by uncertainty confidence: (1 + |unc_norm|) ≥ 1
    energy_base = min_incremental_energy * (1.0 + np.abs(unc_norm))
    energy_norm = q1_q3_normalize(energy_base)
    energy_score = PARAMS["energy_sensitivity"] * energy_norm

    # --- Rank: upward_rank Q1/Q3 normalized ---
    rank_norm = q1_q3_normalize(upward_rank)

    # --- Critical path: dual-gated leverage (slack >= 0 ∧ low uncertainty) — removes brittle energy-median gate ---
    unc_med = np.median(uncertainty)
    critical_gate = ((slack >= 0.0) & (uncertainty <= unc_med)).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate

    # --- Work density: remaining_work per unit duration, gated by smooth joint feasibility ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = q1_q3_normalize(work_density)
    # Smooth joint feasibility: tanh(slack/(uncertainty+eps)) ∈ [-1,1]; apply offset and clamp to [0,1] for gating
    joint_feasibility = PARAMS["tanh_offset"] * (1.0 + np.tanh((slack / (uncertainty + eps)))) 
    joint_feasibility = np.clip(joint_feasibility, 0.0, 1.0)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * joint_feasibility

    # --- Fairness: threshold-gated linear boost in normalized wait time ---
    wait_norm = q1_q3_normalize(ready_wait_time)
    fairness_boost = np.clip(wait_norm - PARAMS["fairness_wait_threshold"], 0.0, np.inf)
    fairness_score = fairness_boost  # No multiplication — direct penalty reduction (subtracted later)

    # --- Uncertainty coupling: multiplicative modulation of *both* energy_score and slack_norm ---
    unc_mod = 1.0 + PARAMS["uncertainty_sensitivity"] * np.abs(unc_norm)

    # Apply uncertainty modulation to key components
    modulated_energy_score = energy_score * unc_mod
    modulated_slack_norm = slack_norm * unc_mod

    # --- Assemble final score: smaller = higher priority ---
    # Penalties added; beneficial boosts subtracted
    score = (
        modulated_slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + modulated_energy_score
        - critical_boost
        - work_density_bonus
        - fairness_score
    )

    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    
    return score
