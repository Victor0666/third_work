def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """
    Hybrid priority rule: piecewise slack scoring (exponential lateness penalty + linear earliness reward),
    criticality boost gated by *both* rank and slack tightness, uncertainty-coupled deadline pressure,
    hybrid anti-starvation (exponential soft boost + saturation cap), and rank/energy interpolation
    weighted by slack tightness. All numeric literals are -2,-1,0,1,2; all tunables declared.
    """
    eps = PARAMS["epsilon"]
    
    # Safe, finite, deterministic normalization: mean absolute + epsilon
    def robust_norm(x):
        x = np.asarray(x, dtype=np.float64)
        denom = np.mean(np.abs(x)) + eps
        return x / denom
    
    # Normalize base features
    norm_slack = robust_norm(slack)
    norm_energy = robust_norm(min_incremental_energy)
    norm_duration = robust_norm(min_exec_time + min_comm_time)
    norm_rank = robust_norm(upward_rank)
    norm_wait = robust_norm(ready_wait_time)
    norm_uncert = robust_norm(uncertainty)
    
    # === 1. Piecewise slack scoring ===
    slack_sign_mask_neg = (slack < 0).astype(float)
    slack_sign_mask_pos = (slack > 0).astype(float)
    
    # Negative slack → bounded exponential urgency
    urgency_base = -norm_slack * slack_sign_mask_neg
    urgency_clipped = np.clip(urgency_base, 0.0, PARAMS["urgency_clip_max"])
    urgency_penalty = np.exp(PARAMS["slack_penalty_exponent"] * urgency_clipped)
    
    # Positive slack → linear reward
    early_reward = slack * slack_sign_mask_pos * PARAMS["early_slack_reward_factor"]
    early_reward_norm = robust_norm(early_reward)
    
    slack_score = (
        urgency_penalty * slack_sign_mask_neg
        + early_reward_norm * slack_sign_mask_pos
    )
    
    # === 2. Criticality boost: only if high rank AND tight slack (<= 2*eps) ===
    rank_median = np.median(norm_rank) if len(norm_rank) > 1 else np.mean(norm_rank)
    is_high_rank = (norm_rank >= rank_median)
    is_tight_slack = (slack <= 2 * eps)  # literal 2 allowed
    critical_gate = np.where(is_high_rank & is_tight_slack, PARAMS["criticality_boost"], 1.0)
    critical_boost = norm_rank * critical_gate
    
    # === 3. Uncertainty-modulated slack pressure ===
    deadline_pressure = np.maximum(0.0, -norm_slack)
    uncertainty_coupled_pressure = PARAMS["uncertainty_slack_coupling"] * deadline_pressure * norm_uncert
    
    # === 4. Anti-starvation: hybrid exponential + saturation ===
    wait_exp_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * norm_wait)
    wait_clipped = np.clip(ready_wait_time, 0, PARAMS["wait_saturation_threshold"])
    wait_sat_norm = robust_norm(wait_clipped)
    wait_score = -wait_sat_norm + wait_exp_boost
    
    # === 5. Energy-rank interaction penalty ===
    rank_energy_penalty = PARAMS["rank_energy_interaction"] * norm_energy * norm_rank
    
    # === 6. Slack-weighted rank/energy tradeoff ===
    # Use global slack range for robust interpolation
    slack_min = np.min(slack)
    slack_max = np.max(slack)
    slack_range = np.maximum(eps, slack_max - slack_min)
    slack_normalized = np.clip((slack - slack_min) / (slack_range + eps), 0, 1)
    weight_rank = 1.0 - slack_normalized * (1.0 - PARAMS["rank_slack_balance"])
    rank_energy_tradeoff = weight_rank * norm_rank + (1.0 - weight_rank) * norm_energy
    
    # === Final score: smaller = better ===
    score = (
        robust_norm(slack_score)
        + uncertainty_coupled_pressure
        + rank_energy_penalty
        + PARAMS["energy_sensitivity"] * norm_energy
        + PARAMS["duration_balance"] * norm_duration
        - critical_boost
        + wait_score
        + rank_energy_tradeoff
    )
    
    # Final safeguard
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=-finfo.max)
    
    return score
