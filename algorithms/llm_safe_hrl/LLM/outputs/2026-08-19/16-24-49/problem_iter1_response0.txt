def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: deadline-risk gating + robustness-aware energy scaling + anti-starvation saturation.
    
    Key innovations:
    - Uses piecewise slack gating: strong linear penalty for negative slack, smooth saturation for large positive slack
    - Critical-path boost only activates above upward_rank threshold (avoids over-prioritizing shallow nodes)
    - Energy weight is dynamically scaled by slack pressure (higher urgency → higher energy sensitivity)
    - Ready-wait time uses bounded logarithmic saturation to prevent starvation without dominating
    - Uncertainty interacts multiplicatively with slack pressure and robustness bias, not additively
    - All features normalized via median-IQR (more robust than mean/std under skew/outliers)
    """
    epsilon = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(epsilon, finfo.tiny)

    # Robust normalization: median + IQR, avoids outlier distortion
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        q75, q25 = np.percentile(x, [75, 25], method='midpoint')
        iqr = q75 - q25
        scale = iqr if iqr > eps_safe else 1.0
        return (x - med) / scale

    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # --- Slack-driven risk gating ---
    # Strong penalty for negative slack; saturating soft penalty for large positive slack
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_penalty_factor"] * (-slack) / (np.abs(slack) + eps_safe),  # bounded near zero
        np.log1p(slack) * 0.3  # slow growth for positive slack
    )
    
    # Critical slack threshold gate: enables rank boost only when slack is tight
    slack_tight_mask = (slack <= PARAMS["critical_slack_threshold"]).astype(float)
    
    # --- Dynamic energy weighting: more sensitive when deadlines are at risk ---
    energy_weight_dynamic = (
        PARAMS["energy_sensitivity"] 
        * (1.0 + 0.5 * np.maximum(-norm_slack, 0.0))
    )

    # --- Critical-path boost: only apply upward_rank bonus when both slack is tight AND rank is high ---
    rank_boost = np.where(
        (upward_rank >= PARAMS["rank_urgency_gate"] * np.median(upward_rank + eps_safe)) & (slack_tight_mask > 0),
        -0.8 * norm_rank,  # higher rank → lower score (better priority)
        np.zeros_like(norm_rank)
    )

    # --- Anti-starvation: saturating log-based wait-time priority ---
    # Prevents long waits from overwhelming other signals
    wait_priority = -(
        (1.0 - np.exp(-PARAMS["wait_decay_rate"] * (ready_wait_time + eps_safe))) 
        * norm_wait
    )

    # --- Uncertainty-robustness coupling: penalize high uncertainty only under deadline pressure ---
    # Reduces preference for uncertain tasks when slack is negative or small
    uncertainty_penalty = (
        PARAMS["robustness_bias"] 
        * norm_uncert 
        * np.maximum(-norm_slack, 0.0)
        * slack_tight_mask
    )

    # --- Duration-uncertainty coupling: longer durations amplify uncertainty impact ---
    dur_uncert_interaction = (
        PARAMS["duration_uncertainty_coupling"] 
        * norm_duration 
        * norm_uncert
    )

    # Composite score: smaller = higher priority
    score = (
        slack_penalty
        + energy_weight_dynamic * norm_energy
        + 0.4 * norm_duration
        + rank_boost
        + 0.2 * norm_work
        + wait_priority
        + uncertainty_penalty
        + dur_uncert_interaction
    )

    # Final safeguard: ensure finite, bounded output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
