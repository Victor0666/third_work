def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with unified DDL-protection gating and percentile-based robust normalization.
    
    Key structural improvements:
    - Replaces MAD normalization with percentile-based scaling (q_low/q_high) for `upward_rank`, `remaining_work`,
      `uncertainty`, and `ready_wait_time` to better handle skewed distributions without Gaussian assumptions.
    - Introduces hard DDL-protection gate: all non-slack terms are multiplied by `(1.0 - np.clip(-slack, 0, 1))`,
      strictly suppressing energy/fairness/uncertainty contributions when slack < 0 — enforces hard deadline adherence.
    - Combines work density and successor release into single term: `(upward_rank * remaining_work) / (duration + eps)`,
      normalized and gated — avoids double-penalization while preserving critical-path unblocking insight.
    - All numeric literals remain in {-2,-1,0,1,2}; no hidden constants; full epsilon and finfo safeguards applied.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Percentile-based normalization: q_low = (1-p)/2, q_high = (1+p)/2 → span covers middle p-fraction
    p_span = PARAMS["robustness_percentile_span"]
    q_low = (1.0 - p_span) / 2.0
    q_high = 1.0 - q_low
    
    def percentile_normalize(x):
        x = np.asarray(x, dtype=float)
        q_lo = np.percentile(x, 100.0 * q_low)
        q_hi = np.percentile(x, 100.0 * q_high)
        iqr = q_hi - q_lo
        scale = iqr if iqr > eps else eps
        return (x - (q_lo + q_hi) / 2.0) / scale

    # --- Slack transformation: piecewise-smooth risk-urgency curve ---
    neg_mask = slack < 0.0
    zeroish_mask = (slack >= 0.0) & (slack < 1.0)
    pos_mask = slack >= 1.0
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[zeroish_mask]) - 1.0)
    slack_norm[pos_mask] = np.clip(slack[pos_mask], 0.0, PARAMS["slack_cap"])

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = percentile_normalize(duration)

    # --- Energy: dampened and DDL-gated ---
    energy_norm = percentile_normalize(min_incremental_energy)
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - slack_pressure_bounded)
    ddl_protection_gate = (1.0 - np.clip(-slack, 0.0, 1.0))

    # --- Critical path leverage: only active when slack allows, and DDL-gated ---
    rank_norm = percentile_normalize(upward_rank)
    critical_gate = (slack >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate * ddl_protection_gate

    # --- Unified critical-density term: (upward_rank * remaining_work) / duration, percentile-normalized and gated ---
    critical_density = np.divide(upward_rank * remaining_work, duration + eps)
    critical_density_norm = percentile_normalize(critical_density)
    critical_density_bonus = PARAMS["work_density_weight"] * critical_density_norm * critical_gate * ddl_protection_gate

    # --- Fairness: wait score scaled by slack headroom AND DDL-protection gate ---
    wait_headroom = np.maximum(1.0, slack + 1.0)
    wait_norm = percentile_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * wait_norm / (wait_headroom + eps) * ddl_protection_gate

    # --- Uncertainty interaction: multiplicative with same bounded slack pressure AND DDL gate ---
    unc_norm = percentile_normalize(uncertainty)
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded * ddl_protection_gate

    # --- Assemble final score: smaller = higher priority ---
    # Note: critical_boost and critical_density_bonus are subtracted (higher values improve schedule)
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_boost
        - critical_density_bonus
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
