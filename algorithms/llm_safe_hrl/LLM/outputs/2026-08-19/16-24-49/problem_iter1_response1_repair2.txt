def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: replaces hardcoded 75/25 with tunable percentiles.
    Uses robust IQR-normalized features, piecewise slack penalty, and risk-modulated criticality.
    Smaller score = higher priority."""
    eps = PARAMS["epsilon"]
    
    # Convert all inputs to float arrays, ensure no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)
    
    # --- Robust normalization using tunable percentiles (replaces hardcoded 75/25) ---
    def robust_normalize(x):
        x = np.asarray(x)
        q75_val = PARAMS["q75_percentile"]
        q25_val = PARAMS["q25_percentile"]
        q75, q25 = np.percentile(x, [q75_val, q25_val], method='midpoint')
        iqr = q75 - q25
        # Fall back to mean abs dev if iqr ~ 0
        mad = np.mean(np.abs(x - np.median(x)))
        scale = np.where(iqr > eps, iqr, mad + eps)
        center = np.median(x)
        return (x - center) / (scale + eps)
    
    # --- Slack pressure: piecewise with strong negative-penalty and smooth transition ---
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_powered = np.where(
        slack >= 0,
        slack_abs ** PARAMS["slack_sensitivity"],
        - (slack_abs ** PARAMS["slack_sensitivity"])
    )
    # Apply sharp penalty scaling only to negative slack; non-negative slack uses tunable weight
    slack_pressure = np.where(
        slack < 0,
        slack_powered * PARAMS["slack_penalty_scale"],
        slack_powered * PARAMS["slack_positive_weight"]
    )
    
    # --- Energy: IQR-normalized marginal fuzzy energy (risk-adjusted) ---
    energy_norm = robust_normalize(min_incremental_energy)
    energy_score = PARAMS["energy_norm_factor"] * energy_norm
    
    # --- Duration: exec + comm, normalized robustly ---
    duration = min_exec_time + min_comm_time
    duration_norm = robust_normalize(duration)
    duration_score = PARAMS["duration_balance"] * duration_norm
    
    # --- Criticality boost: upward_rank weighted by combined risk signal ---
    risk_signal = np.maximum(-slack, 0.0) * uncertainty
    risk_clamped = np.tanh(risk_signal / (np.maximum(np.mean(np.abs(risk_signal)) + eps, eps)))
    rank_score = -upward_rank * (1.0 + PARAMS["criticality_boost"] * risk_clamped)
    
    # --- Anti-starvation: exponential wait boost (smooth, bounded, monotonic) ---
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    wait_score = -wait_boost
    
    # --- Uncertainty-slack interaction: additive penalty for high uncertainty AND low slack ---
    unc_norm = robust_normalize(uncertainty)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * unc_norm * slack_pressure
    
    # --- Assemble final score ---
    score = (
        slack_pressure +
        energy_score +
        duration_score +
        rank_score +
        wait_score +
        unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf with finite values, preserve shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
