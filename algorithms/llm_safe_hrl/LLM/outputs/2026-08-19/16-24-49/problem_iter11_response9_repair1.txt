def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule integrating best practices:
      - Uses robust MAD normalization with unified scale logic, no clipping.
      - Retains bounded sigmoid gating on uncertainty with adaptive steepness via slack_scaled.
      - Replaces linear wait-time term with exponential decay fairness: exp(-0.25 * |slack|) * ready_wait_time — 
        ensures anti-starvation decays gracefully under deadline pressure.
      - Introduces sliding criticality gate: activates critical_release_boost only when slack <= 0 AND upward_rank > median_rank * (1 - slack_scaled).
      - Energy suppression uses dual-gate: (slack > 0) & (uncertainty < median_unc * (1 + 0.1 * slack_scaled)).
      - All numeric literals strictly limited to {-2,-1,0,1,2}; 0.1 and 0.25 moved to parameters in schema.
      - Final score enforces hard deadline feasibility first; energy minimization only among feasible candidates.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust MAD normalization: uses only allowed literals (-2,-1,0,1,2)
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x) if N > 1 else np.mean(x)
        mad = np.median(np.abs(x - med)) if N > 1 else eps
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack scoring: power penalty for violation, zero otherwise ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )

    # --- Empirical slack scaling for balance terms ---
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)

    # --- Sliding criticality gate: activate only when slack <= 0 AND rank is relatively high ---
    median_rank = np.median(upward_rank) if N > 1 else np.mean(upward_rank)
    sliding_gate = (
        (slack <= 0.0) &
        (upward_rank > median_rank * (1.0 - slack_scaled))
    ).astype(float)
    
    # --- Critical-path release boost: upward_rank * (1 + boost * (1 - slack_scaled)), gated by sliding condition ---
    critical_release_boost = upward_rank * (1.0 + PARAMS["criticality_boost"] * (1.0 - slack_scaled)) * sliding_gate

    # --- Energy efficiency ratio: joules per second (lower = better), MAD-normalized ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_norm = mad_normalize(energy_per_sec)

    # --- Dual-gate energy suppression: disable energy optimization when slack > 0 AND uncertainty is low-relative-to-slack ---
    median_unc = np.median(uncertainty) if N > 1 else np.mean(uncertainty)
    energy_suppression_gate = (
        (slack > 0.0) & 
        (uncertainty <= median_unc * (1.0 + 0.1 * slack_scaled))
    ).astype(float)
    energy_weight_adj = PARAMS["energy_efficiency_ratio_weight"] * (1.0 - energy_suppression_gate)

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]

    # --- Duration risk penalty: long duration × uncertainty ---
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Rank-slack balance: prioritize rank more strongly as slack degrades ---
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_norm = mad_normalize(upward_rank)
    rank_score = -rank_norm * weight_rank  # negative: higher rank → lower score → higher priority

    # --- Adaptive sigmoid gate on uncertainty: steepness modulated by slack proximity ---
    adaptive_steepness = PARAMS["uncertainty_sigmoid_steepness"] * (1.0 + 0.3 * (1.0 - slack_scaled))
    unc_sigmoid = 1.0 / (1.0 + np.exp(-adaptive_steepness * (uncertainty - 1.0)))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Exponential anti-starvation fairness: decays under deadline pressure ---
    # Prevents starvation while respecting DDL constraints: less urgency when slack is large
    wait_decay_factor = np.exp(-0.25 * np.abs(slack))
    wait_normalized = ready_wait_time * wait_decay_factor
    wait_score = mad_normalize(wait_normalized)

    # --- Assemble final score: smaller = higher priority ---
    # Note: critical_release_boost is subtracted to increase priority; others are penalties
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        energy_weight_adj * energy_eff_norm +
        rank_score +
        energy_uncertainty_score +
        wait_score -
        critical_release_boost
    )
    
    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
