def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """
    Self-evolved priority rule featuring a *unified joint feasibility gate*:
      - Replaces separate `feasibility_gate` and `unc_sigmoid_gate` with smooth, bounded `tanh(slack / (uncertainty + eps))`
        → captures 'risk-adjusted slack density': high slack + low uncertainty = full boost; negative slack or high uncertainty = strong suppression.
      - Introduces `rank_slack_uncertainty_coupling`: direct three-way interaction term `upward_rank * tanh(slack / (uncertainty + eps))`,
        enabling critical-path awareness that *naturally degrades* under uncertainty or deadline pressure — no fragile conditionals.
      - Removes all hard caps and sigmoid gates; relies entirely on bounded tanh for stability and differentiability.
      - Keeps MAD-normalized inputs throughout; preserves decoupled energy-slack coupling and saturating fairness logic.
      - All parameters used; no numeric literals beyond {-2,-1,0,1,2}; fully deterministic and finite.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(np.float64)
    
    # Sanitize inputs — ensure no NaN/inf persists
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=np.float64), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=np.float64), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization with Gaussian-consistent scaling
    def mad_normalize(x):
        x = np.asarray(x, dtype=np.float64)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Unified bounded slack urgency: tanh(-slack * scale) → strong penalty when late, zero when slack abundant ---
    slack_pressure = -slack  # positive when late
    slack_urgency = PARAMS["slack_urgency_scale"] * np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: base normalized energy, with slack-coupled amplification ---
    energy_norm = mad_normalize(min_incremental_energy)
    energy_base = PARAMS["energy_efficiency_bias"] * energy_norm
    energy_slack_gate = np.clip(slack_pressure, 0.0, np.inf)
    energy_score = energy_base * (1.0 + PARAMS["energy_slack_coupling"] * np.tanh(energy_slack_gate * PARAMS["slack_pressure_tanh_scale"]))

    # --- Joint feasibility gate: tanh-based risk-adjusted slack density (replaces dual gating) ---
    # Higher value when slack is large AND uncertainty is small → enables critical path boosting only when safe
    # Bounded, smooth, differentiable, and avoids division-by-zero via eps
    joint_feasibility = np.tanh((slack / (uncertainty + eps)) * PARAMS["joint_feasibility_tanh_scale"])

    # --- Critical path: rank boosted only under joint feasibility — no separate unc/feasibility logic ---
    rank_norm = mad_normalize(upward_rank)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * joint_feasibility

    # --- Work density: remaining_work per unit duration, gated by joint feasibility (not just slack) ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * joint_feasibility

    # --- Fairness: saturating exponential wait boost, gated by joint feasibility ---
    wait_norm = mad_normalize(ready_wait_time)
    wait_boost = 1.0 - np.exp(-PARAMS["wait_fairness_gain"] * np.maximum(wait_norm, 0.0))
    wait_score = wait_boost * joint_feasibility

    # --- Resilience: three-way interaction — rank × slack × uncertainty, smoothly bounded via tanh ---
    # Captures how critical-path importance should *modulate* with both urgency and risk simultaneously
    unc_norm = mad_normalize(uncertainty)
    # Use signed slack to preserve directionality: negative slack enhances penalty; positive slack with low uncertainty supports boost
    rank_slack_uncertainty_interaction = rank_norm * np.tanh(slack * (1.0 / (uncertainty + eps)))
    resilience_boost = PARAMS["rank_slack_uncertainty_coupling"] * rank_slack_uncertainty_interaction

    # --- Assemble final score: smaller = higher priority ---
    # Penalties added (slack_urgency, duration_norm, energy_score); beneficial terms subtracted (critical_boost, etc.)
    score = (
        slack_urgency
        + PARAMS["duration_balance"] * duration_norm
        + energy_score
        - critical_boost
        - work_density_bonus
        + wait_score
        - resilience_boost
    )

    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    
    return score
