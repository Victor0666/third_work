def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with stabilized risk alignment and hardened DDL semantics:
      - Replaces joint MAD with *per-feature* median absolute deviation + max-based robust scaling
        → eliminates fragile weighted fusion while preserving cross-feature scale coherence
      - Uses binary feasibility gate `slack >= ddl_safety_margin` (not > threshold) → aligns with hard deadline constraint semantics
      - Removes nonlinear slack exponent → ensures lexicographic DDL priority remains linear, monotonic, and interpretable
      - Tightens uncertainty sigmoid center to `uncertainty_sigmoid_center` for sharper suppression of energy terms at moderate-high risk
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no hidden constants or domain-specific scales
      - Final score preserves strict DDL-first ordering: violation penalty dominates; all other terms are gated or scaled
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Compute per-feature MADs for robust, decoupled normalization
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack)
    
    # Per-feature MADs: |slack|, uncertainty, duration_total
    mad_abs_slack = np.median(np.abs(abs_slack - np.median(abs_slack))) + eps
    mad_uncertainty = np.median(np.abs(uncertainty - np.median(uncertainty))) + eps
    mad_duration = np.median(np.abs(duration_total - np.median(duration_total))) + eps
    
    # Robust joint scale: max of scaled per-feature MADs → stable against outlier domination
    joint_scale = np.max([
        mad_abs_slack * PARAMS["per_feature_mad_scale"],
        mad_uncertainty * PARAMS["per_feature_mad_scale"],
        mad_duration * PARAMS["per_feature_mad_scale"]
    ]) + eps

    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        x_median = np.median(x)
        norm = (x - x_median) / (joint_scale + eps)
        return np.clip(norm, -2.0, 2.0)

    # --- DDL-critical base penalty: lexicographically dominant, linear slack deficit ---
    # Feasibility defined by slack >= ddl_safety_margin (hard deadline semantics)
    slack_deficit = np.maximum(0.0, PARAMS["ddl_safety_margin"] - slack)
    ddl_violation_penalty = slack_deficit  # linear, monotonic, interpretable

    # --- Critical-path release: unconditional, linearly scaled by slack pressure ---
    critical_release_score = upward_rank * remaining_work
    # Linear slack pressure: 0 when slack >= ddl_safety_margin, 1 when slack = -inf
    slack_pressure = np.clip((PARAMS["ddl_safety_margin"] - slack) / (np.abs(PARAMS["ddl_safety_margin"]) + eps), 0.0, 1.0)
    critical_score = -mad_normalize(critical_release_score) * slack_pressure * PARAMS["critical_path_release_weight"]

    # --- Feasibility mask: activate non-critical terms only when slack >= ddl_safety_margin ---
    feasible_mask = np.where(slack >= PARAMS["ddl_safety_margin"], 1.0, 0.0)

    # --- Risk-adjusted energy term: only active in feasible region ---
    energy_norm = mad_normalize(min_incremental_energy)
    energy_term = feasible_mask * PARAMS["risk_adjusted_energy_weight"] * energy_norm

    # --- Uncertainty-slack interaction: penalizes high uncertainty near deadline ---
    unc_slack_interaction = uncertainty * slack_deficit * PARAMS["uncertainty_slack_interaction"]
    unc_slack_norm = mad_normalize(unc_slack_interaction)

    # --- Starvation mitigation: wait time normalized by duration ---
    wait_efficiency = np.where(duration_total > eps, ready_wait_time / duration_total, 0.0)
    wait_norm = mad_normalize(wait_efficiency)
    wait_term = feasible_mask * PARAMS["wait_starvation_penalty"] * wait_norm

    # --- Energy-uncertainty coupling: sharp sigmoid centered at tunable `uncertainty_sigmoid_center` ---
    # Suppresses energy-aware terms earlier and more decisively than unit-centered sigmoid
    unc_sigmoid = 1.0 / (1.0 + np.exp(-2.0 * (PARAMS["uncertainty_sigmoid_center"] - uncertainty)))
    energy_uncertainty_term = feasible_mask * PARAMS["energy_uncertainty_coupling"] * energy_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    # DDL violation dominates lexicographically
    score = (
        mad_normalize(ddl_violation_penalty) +
        critical_score +
        unc_slack_norm
    )
    
    # Add feasible-only terms
    score += energy_term + wait_term + energy_uncertainty_term

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
