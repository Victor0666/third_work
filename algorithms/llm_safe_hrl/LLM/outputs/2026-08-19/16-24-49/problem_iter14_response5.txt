def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining best practices:
      - Uses robust MAD-based normalization (from Parent 2) for stability across N.
      - Retains unconditional upward_rank × remaining_work critical-path interaction (Parent 2).
      - Integrates smooth feasibility gate (from Parent 1) using feasibility_sigmoid_steepness, enabling differentiable optimization.
      - Introduces novel wait_slack_coupling_exponent: replaces linear (1 - slack_sigmoid) with power-law anti-starvation that intensifies near slack=0, improving responsiveness to incipient deadline pressure.
      - All divisions guarded; all outputs finite, shape-(N,), deterministic; only literals {-2,-1,0,1,2} used.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust MAD-based normalization: more stable than percentile for sparse/variable N
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        abs_dev = np.abs(x - med)
        mad = np.median(abs_dev) + eps
        normalized = (x - med) / mad
        # Clip to [-2, 2] — uses allowed literals only
        return np.clip(normalized, -2.0, 2.0)

    # --- DDL-critical base scores (always active) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Criticality interaction: upward_rank × remaining_work — unconditionally included ---
    rank_work_interaction = upward_rank * remaining_work
    rank_work_med = np.median(rank_work_interaction) if N > 1 else np.mean(rank_work_interaction)
    rank_work_normalized = mad_normalize(rank_work_interaction) / (np.abs(rank_work_med) + eps)

    # --- Smooth feasibility gate: replaces hard `slack > 0.0` with differentiable sigmoid ---
    feasibility_gate = 1.0 / (1.0 + np.exp(-PARAMS["feasibility_sigmoid_steepness"] * slack))
    # Anti-starvation term: non-linearly boosted near slack=0 via exponentiation
    wait_slack_coupling = (1.0 - feasibility_gate) ** PARAMS["wait_slack_coupling_exponent"]
    wait_score = mad_normalize(ready_wait_time) * wait_slack_coupling

    # --- Non-DDL terms (smoothly gated by feasibility_gate) ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = mad_normalize(energy_per_sec)

    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -mad_normalize(upward_rank) * weight_rank

    # Uncertainty-aware energy: sigmoid-gated to suppress at high uncertainty
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        rank_work_normalized
    )
    
    # Non-DDL terms smoothly gated by feasibility_gate
    score += feasibility_gate * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Apply criticality boost multiplicatively only when slack <= 0 (tight or violated)
    is_tight_or_violated = (slack <= 0)
    critical_gate = np.where(
        is_tight_or_violated,
        PARAMS["criticality_boost"],
        1.0
    )
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
