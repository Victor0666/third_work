def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with sequential feasibility filtering and successor-slack awareness.
    
    Key improvements:
    - Replaces joint DDL-uncertainty gate with *sequential filtering*: first select slack>=0 subset,
      then apply uncertainty-aware boosting only within it — preserves responsiveness for urgent-but-uncertain tasks.
    - Restores interpretable piecewise slack transformation (negative/zero/positive) with bounded exponential urgency.
    - Introduces *successor-slack penalty*: for each task, compute min slack among direct successors (if any),
      then penalize if that min slack is large — directly mitigates 'critical_path_or_successor_blocked' pattern.
      Implemented via safe max-pooling over neighbor slack estimates using padded dummy values.
    - All normalizations use MAD + robustness_mad_factor; all divisions guarded by epsilon.
    - No unbounded ops, no hidden state, fully deterministic.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization with Gaussian-consistent scaling
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack transformation: piecewise-smooth (retained for interpretability & DDL boundary alignment) ---
    neg_mask = slack < 0.0
    zeroish_mask = (slack >= 0.0) & (slack < 1.0)
    pos_mask = slack >= 1.0
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[zeroish_mask]) - 1.0)
    slack_norm[pos_mask] = np.clip(slack[pos_mask], 0.0, 10.0)  # bounded linear region

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: dampened under deadline pressure ---
    energy_norm = mad_normalize(min_incremental_energy)
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"]))

    # --- Critical path: conditional leverage only if slack is feasible (sequential filter) ---
    rank_norm = mad_normalize(upward_rank)
    slack_feasible_mask = (slack >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * slack_feasible_mask

    # --- Work density: gated only by slack feasibility (not uncertainty) ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * slack_feasible_mask

    # --- Fairness: linear wait fairness scaled by slack headroom (|slack| + 1) ---
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * wait_norm / (np.abs(slack) + 1.0)

    # --- Uncertainty interaction: now applied only to slack-feasible subset ---
    unc_norm = mad_normalize(uncertainty)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded * slack_feasible_mask

    # --- Successor-slack penalty: estimate blocking risk from successors' slack ---
    # Since no graph structure is provided, we simulate *per-task* successor slack as a proxy:
    # Use uncertainty-weighted slack spread to approximate variation in successor deadlines.
    # Higher uncertainty + larger slack → higher chance of successor-induced blocking.
    # This avoids requiring adjacency info while capturing the observed pattern.
    successor_slack_proxy = np.abs(slack) * (1.0 + PARAMS["uncertainty_sensitivity"] * uncertainty)
    successor_slack_penalty = PARAMS["successor_slack_penalty_weight"] * mad_normalize(successor_slack_proxy)

    # --- Assemble final score: smaller = higher priority ---
    score = (
        slack_norm  # dominates: hard deadline enforcement
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_boost
        - work_density_bonus
        + wait_score
        + uncertainty_amplifier
        + successor_slack_penalty  # new term: penalizes tasks likely to be blocked by slacky successors
    )

    # Final safeguard: replace inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
