def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with bounded piecewise uncertainty-slack coupling,
       cleaned joint MAD (excludes ready_wait_time), and sign-invariant critical-path scoring.
    
    Key improvements:
      - Removed ready_wait_time from joint MAD normalization per self-reflection → preserves risk-signal coherence
      - Replaced linear unc_slack_coupling with bounded piecewise: linear ramp for slack > slack_saturation_threshold,
        saturated constant below it → prevents over-penalization near deadline
      - Critical path release uses sign-invariant normalization: |upward_rank * remaining_work| → avoids numerical instability
        when either term approaches zero, while preserving monotonic priority inversion via explicit negation
      - All numeric literals strictly limited to {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Joint MAD normalization over unified risk signal space: |slack|, uncertainty, duration_total ONLY
    # Excludes ready_wait_time per self-reflection to avoid distorting risk alignment
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    risk_signals = np.concatenate([
        abs_slack,
        uncertainty,
        duration_total
    ])
    risk_median = np.median(risk_signals)
    mad = np.median(np.abs(risk_signals - risk_median)) + eps
    joint_scale = PARAMS["joint_mad_scale_factor"] * mad

    def joint_mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        # Normalize using shared scale; clip to [-2,2] for stability and bounded interaction
        return np.clip((x - risk_median) / joint_scale, -2.0, 2.0)

    # --- DDL-enforcement core: always active ---
    slack_violation = np.maximum(0.0, -slack)
    
    # Bounded piecewise uncertainty-slack coupling: linear above saturation threshold, saturated below
    # Prevents explosive penalties near slack = 0 while preserving sensitivity to violation magnitude
    unc_slack_mask = np.where(slack > PARAMS["slack_saturation_threshold"], 1.0, 0.0)
    unc_slack_linear = PARAMS["uncertainty_slack_coupling"] * uncertainty * slack_violation
    unc_slack_saturated = PARAMS["uncertainty_slack_coupling"] * uncertainty * np.abs(PARAMS["slack_saturation_threshold"])
    unc_slack_coupling = unc_slack_linear * unc_slack_mask + unc_slack_saturated * (1.0 - unc_slack_mask)
    
    # Duration risk activated under tight slack (not just violation) → proactive throttling
    tight_slack_mask = np.where(slack <= PARAMS["ddl_protection_threshold"], 1.0, 0.0)
    duration_risk = PARAMS["duration_risk_penalty"] * duration_total * tight_slack_mask * uncertainty

    # --- Critical path release: unconditionally prioritized bottleneck unlock ---
    # Use absolute value to ensure robustness near zero; sign inversion handled separately
    critical_path_release_score = np.abs(upward_rank * remaining_work)

    # --- Lexicographic gate for non-DDL optimization terms ---
    slack_headroom_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Non-DDL terms: normalized, gated, and interactively coupled ---
    energy_norm = joint_mad_normalize(min_incremental_energy)
    energy_score = PARAMS["risk_adjusted_energy_weight"] * energy_norm * slack_headroom_mask

    unc_norm = joint_mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * slack_headroom_mask

    # Hybrid wait term: combines global penalty and exponent decay
    wait_base = ready_wait_time / (np.abs(slack) + 1.0)
    wait_decay = (wait_base ** PARAMS["ready_wait_decay_exponent"]) * slack_headroom_mask
    wait_score = PARAMS["wait_starvation_penalty"] * wait_decay
    wait_norm = joint_mad_normalize(wait_score)

    # --- Assemble final score: smaller = higher priority ---
    # Base DDL enforcement: violation + coupling + duration risk (all normalized)
    score = (
        joint_mad_normalize(slack_violation) +
        joint_mad_normalize(unc_slack_coupling) +
        joint_mad_normalize(duration_risk)
    )

    # Critical path release: highest priority, always subtracted (higher → lower score)
    score += -joint_mad_normalize(critical_path_release_score)

    # Non-DDL optimization: additive contributions only under safety margin
    score += energy_score + energy_uncertainty_score + wait_norm

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
