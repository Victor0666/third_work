def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaces empirical slack bounds with adaptive MAD-based slack scaling (robust across workflows).
      - Introduces host-load congestion gate using `ready_wait_time * uncertainty` to suppress energy optimization when queueing is low but uncertainty is high.
      - Adds upward-rank × remaining-work interaction term (MAD-normalized), strengthening critical-path release prioritization under deadline pressure.
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no other constants.
      - Uses only mad_normalize for all feature normalizations — fully compliant, robust, and deterministic.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust MAD normalization: uses only allowed literals (-2,-1,0,1,2)
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x) if N > 1 else np.mean(x)
        mad = np.median(np.abs(x - med)) if N > 1 else eps
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Adaptive MAD-based slack scaling (replaces fixed bounds) ---
    # Uses MAD instead of hardcoded [min_bound, max_bound] for generalization
    slack_med = np.median(slack) if N > 1 else np.mean(slack)
    slack_mad = np.median(np.abs(slack - slack_med)) if N > 1 else eps
    slack_scale = PARAMS["robustness_mad_factor"] * (slack_mad if slack_mad > eps else eps)
    slack_scaled = np.clip((slack - slack_med) / (slack_scale + eps), -2.0, 2.0)  # bounded using allowed literals

    # --- Slack scoring: power penalty for violation, zero otherwise ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )

    # --- Critical-path release boost: upward_rank * (1 + boost * (1 - slack_scaled)), only when slack <= 0 ---
    release_gate = (slack <= 0).astype(float)
    critical_release_boost = upward_rank * (1.0 + PARAMS["criticality_boost"] * (1.0 - (slack_scaled + 2.0) / 4.0)) * release_gate

    # --- Energy efficiency ratio: joules per second (lower = better), MAD-normalized ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_norm = mad_normalize(energy_per_sec)

    # --- Host-load congestion gate: activates when ready_wait_time * uncertainty exceeds threshold ---
    # Suppresses energy optimization when queueing is low (short wait) but uncertainty is high → risk masking
    load_congestion_score = ready_wait_time * uncertainty
    congestion_gate = (load_congestion_score >= PARAMS["load_congestion_gate_threshold"]).astype(float)
    energy_weight_adj = PARAMS["energy_efficiency_ratio_weight"] * (1.0 - congestion_gate)

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]

    # --- Duration risk penalty: long duration × uncertainty ---
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Rank-slack balance: prioritize rank when slack is poor (low slack_scaled) ---
    # Map slack_scaled ∈ [-2,2] → weight_rank ∈ [0,1] linearly
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * np.clip((1.0 - slack_scaled / 2.0), 0.0, 1.0)
    rank_norm = mad_normalize(upward_rank)
    rank_score = -rank_norm * weight_rank

    # --- Critical-path interaction term: upward_rank × remaining_work, MAD-normalized ---
    # Strongly prioritizes tasks that are both high-criticality and high-workload descendants
    critical_path_interaction = upward_rank * remaining_work
    critical_path_norm = mad_normalize(critical_path_interaction)
    critical_path_score = -critical_path_norm * PARAMS["critical_path_interaction_weight"]  # negative → higher priority

    # --- Bounded sigmoid gate on uncertainty for energy-uncertainty interaction ---
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        energy_weight_adj * energy_eff_norm +
        rank_score +
        energy_uncertainty_score +
        mad_normalize(min_incremental_energy) * (1.0 - weight_rank) -
        critical_release_boost +
        critical_path_score
    )
    
    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
