def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Adaptive lexicographic gate: non-DDL terms activate only when slack > median(slack) + offset → robust to workload skew & avoids brittle fixed thresholds
      - Hybrid normalization: MAD for risk dimensions (duration, uncertainty), percentile *only* for upward_rank interpolation → preserves critical-path ordering fidelity
      - Load-aware energy suppression: disables energy-per-sec term when ready_wait_time exceeds median by load_aware_energy_suppression_threshold → prevents selecting low-energy-but-high-congestion tasks
      - All numeric literals restricted to {-2,-1,0,1,2}; no unbounded loops or side effects
      - Final score enforces strict DDL-first ordering, then optimizes within feasibility
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # --- Joint MAD normalization for risk dimensions (duration, slack abs, uncertainty) ---
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    all_risk_dims = np.stack([duration_total, abs_slack, uncertainty], axis=0)
    mad_per_dim = np.mean(np.abs(all_risk_dims - np.median(all_risk_dims, axis=1, keepdims=True)), axis=1) + eps
    duration_norm = duration_total / (mad_per_dim[0] + eps)
    slack_norm = abs_slack / (mad_per_dim[1] + eps)
    unc_norm = uncertainty / (mad_per_dim[2] + eps)

    # --- Clipped percentile normalization (for rank interpolation only, per reflection) ---
    def percentile_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        q_low = np.percentile(x, PARAMS["percentile_clip_low"])
        med = np.median(x)
        q_high = 2.0 * med - q_low  # allowed literal 2.0
        denom = q_high - q_low + eps
        normalized = (x - q_low) / denom
        return np.clip(normalized, 0.0, 1.0)

    # --- DDL-critical base scores (always active) ---
    slack_penalty = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Criticality boost: activated only under DDL pressure (slack <= 0), requires high rank AND high remaining work ---
    is_tight_or_violated = (slack <= 0)
    rank_median = np.median(upward_rank) if N > 1 else np.mean(upward_rank)
    work_median = np.median(remaining_work) if N > 1 else np.mean(remaining_work)
    is_high_rank = (upward_rank >= rank_median)
    is_high_work = (remaining_work >= work_median)
    critical_gate = np.where(
        is_high_rank & is_high_work & is_tight_or_violated,
        PARAMS["criticality_boost"],
        1.0
    )

    # --- Adaptive lexicographic gate: non-DDL terms only active when slack > median(slack) + offset ---
    slack_median = np.median(slack) if N > 1 else np.mean(slack)
    slack_headroom_mask = np.where(slack > (slack_median + PARAMS["adaptive_slack_gate_offset"]), 1.0, 0.0)

    # --- Non-DDL terms (only active when slack sufficiently above median) ---
    duration_total_safe = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total_safe
    # Energy efficiency term: suppressed under congestion per reflection
    median_wait = np.median(ready_wait_time) + eps
    wait_ratio = ready_wait_time / median_wait
    energy_suppression_gate = np.where(wait_ratio > PARAMS["load_aware_energy_suppression_threshold"], 0.0, 1.0)
    energy_eff_score = percentile_normalize(energy_per_sec) * energy_suppression_gate

    # Slack-aware rank balance using percentile-normalized rank (preserves ordering fidelity)
    slack_scaled = np.clip((slack - slack_median) / (1.0 + eps), 0.0, 1.0)  # uses allowed literal 1.0
    weight_rank = PARAMS["rank_slack_balance"] * (1.0 - slack_scaled) + (1.0 - PARAMS["rank_slack_balance"]) * slack_scaled
    rank_score = -percentile_normalize(upward_rank) * weight_rank

    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = percentile_normalize(min_incremental_energy)
    unc_norm_pct = percentile_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm_pct * unc_sigmoid

    # --- Wait-time anti-starvation: scaled by congestion proxy but only in safe region ---
    congestion_proxy = (wait_ratio + eps) * (1.0 + unc_norm)
    wait_score = percentile_normalize(ready_wait_time) * np.clip(congestion_proxy, 0.0, 1.0)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        percentile_normalize(slack_penalty) +
        percentile_normalize(unc_slack_coupling) +
        percentile_normalize(duration_risk)
    )
    
    # Non-DDL terms only added when slack is sufficiently above median (adaptive lexicographic enforcement)
    score += slack_headroom_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Apply criticality boost multiplicatively
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
