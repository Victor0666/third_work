def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Per-task criticality-aware activation: critical_release_score only when slack ≤ 0 AND (upward_rank * remaining_work) in top-k percentile.
      - Quantile-based normalization (clipped percentiles) replacing MAD — eliminates fragile median/MAD dependence on small N and outliers.
      - Removed inactive `uncertainty_slack_coupling`; reallocated budget to strengthen critical interaction term.
      - Critical release now includes multiplicative coupling: upward_rank * remaining_work * uncertainty * weight, enhancing risk-aware bottleneck targeting.
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no hidden constants.
      - Strict lexicographic ordering preserved: DDL-violating terms dominate unconditionally.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Quantile-based normalization: robust for small-N, avoids median/MAD fragility
    def quantile_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        # Compute clipped percentiles for stability
        q_low = np.clip(PARAMS["quantile_clip_low"], 0.001, 0.1)
        q_high = np.clip(PARAMS["quantile_clip_high"], 0.9, 0.999)
        p_low = np.quantile(x, q_low)
        p_high = np.quantile(x, q_high)
        scale = p_high - p_low + eps
        # Center at median, clip extremes
        med = np.median(x)
        normed = (x - med) / scale
        return np.clip(normed, -2.0, 2.0)  # Bounded output per structural constraint

    # --- DDL-critical base scores (unconditional, lexically dominant) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Critical-path release: activated only for truly critical tasks ---
    # Compute criticality metric: upward_rank * remaining_work
    criticality_metric = upward_rank * remaining_work
    # Per-task activation: slack ≤ 0 AND metric ≥ top-k percentile of ready set
    if N > 1:
        critical_threshold = np.quantile(criticality_metric[slack <= 0], PARAMS["critical_percentile_threshold"])
        # Only activate where both conditions hold; avoids global gating instability
        is_critical = (slack <= 0) & (criticality_metric >= critical_threshold - eps)
    else:
        is_critical = (slack <= 0) & (criticality_metric >= criticality_metric[0] - eps)
    # Enhanced coupling: include uncertainty for risk-aware critical release
    critical_release_score = (
        criticality_metric * uncertainty * PARAMS["criticality_release_weight"] * is_critical.astype(float)
    )

    # --- Slack safety signal for non-DDL blending ---
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_centered = slack - slack_lb
    slack_range = slack_ub - slack_lb + eps
    slack_normalized = np.clip(slack_centered / slack_range, -2.0, 2.0)
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * slack_normalized))

    # --- Energy efficiency ratio: joules per second, quantile-normalized ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_norm = quantile_normalize(energy_per_sec)

    # --- Rank-driven score: weighted by slack safety ---
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - ddl_gate)
    rank_norm = quantile_normalize(upward_rank)
    rank_score = -rank_norm * weight_rank  # higher rank → lower score → higher priority

    # --- Energy-uncertainty interaction: gated by uncertainty sigmoid ---
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = quantile_normalize(min_incremental_energy)
    unc_norm = quantile_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Anti-starvation: clamped sigmoid on normalized wait time ---
    wait_norm = quantile_normalize(ready_wait_time)
    wait_sigmoid = 1.0 - np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * np.maximum(wait_norm, 0.0))
    wait_fairness = np.clip(wait_sigmoid, 0.0, 1.0)
    wait_gated = wait_fairness * ddl_gate

    # --- Work density bonus: only when slack is safe ---
    work_density = np.divide(remaining_work, duration_total + eps)
    work_density_norm = quantile_normalize(work_density)
    work_density_bonus = (
        PARAMS["energy_efficiency_ratio_weight"] * 
        work_density_norm * 
        (np.abs(work_density_norm) <= PARAMS["energy_efficiency_ratio_weight"]) *
        (1.0 - ddl_gate)
    )

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms dominate unconditionally
    score = (
        quantile_normalize(slack_score) +
        quantile_normalize(duration_risk) +
        (-critical_release_score)  # negative: promotes release → lowers score
    )
    
    # Non-DDL terms smoothly blended via ddl_gate
    score += ddl_gate * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_norm +
        rank_score +
        energy_uncertainty_score +
        wait_gated +
        work_density_bonus
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    score = np.clip(score, finfo.min + eps, finfo.max - eps)
    
    return score
