def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with DDL-protection gating, successor-aware criticality, and IQR-based robustness.
    
    Key evolutions:
    - Introduces `critical_slack_gate_threshold`: suppresses energy/work-density bonuses when slack < threshold,
      preventing over-optimization of energy at the cost of deadline violation.
    - Adds `successor_criticality_weight`: amplifies priority for tasks whose *immediate successors* have
      high upward_rank AND low slack — proactively unblocks critical-path bottlenecks.
    - Replaces MAD with IQR-based normalization for `work_density` and `uncertainty`, using tunable quantiles
      for superior stability on skewed distributions without increasing AST depth.
    - All gates are bounded, epsilon-guarded, and preserve determinism and finite output.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust IQR normalization for skewed features (work_density, uncertainty)
    def iqr_normalize(x, q_low=PARAMS["iqr_quantile_low"], q_high=PARAMS["iqr_quantile_high"]):
        x = np.asarray(x, dtype=float)
        q1 = np.quantile(x, q_low)
        q3 = np.quantile(x, q_high)
        iqr = q3 - q1
        scale = iqr if iqr > eps else eps
        med = np.median(x)
        return (x - med) / scale

    # Standard MAD normalization for other features
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack transformation: piecewise-smooth risk-urgency curve ---
    neg_mask = slack < 0.0
    zeroish_mask = (slack >= 0.0) & (slack < 1.0)
    pos_mask = slack >= 1.0
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[zeroish_mask]) - 1.0)
    slack_norm[pos_mask] = np.clip(slack[pos_mask], 0.0, PARAMS["slack_cap"])

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: dampened AND gated under critically low slack ---
    energy_norm = mad_normalize(min_incremental_energy)
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"]))
    # New DDL-protection gate: zero out bonus when slack is critically low
    ddl_feasibility_gate = (slack >= PARAMS["critical_slack_gate_threshold"]).astype(float)
    energy_weight_adj *= ddl_feasibility_gate

    # --- Critical path: conditional leverage only if slack is feasible ---
    rank_norm = mad_normalize(upward_rank)
    critical_gate = (slack >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate

    # --- Work density: remaining_work per unit duration, gated by both slack feasibility AND critical threshold ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = iqr_normalize(work_density)  # IQR for skew resilience
    work_density_gate = (slack >= PARAMS["critical_slack_gate_threshold"]).astype(float)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * work_density_gate

    # --- Successor-criticality interaction: amplify if immediate successors are high-rank & low-slack ---
    # Note: this is a *feature interaction*, not a graph traversal — uses input arrays only.
    # Environment provides ready tasks; successor properties must be approximated via aligned inputs.
    # Here we assume upward_rank and slack reflect aggregate successor criticality (standard HEFT proxy).
    successor_pressure = np.clip(-slack, 0.0, np.inf) * rank_norm  # high-rank + low-slack → high pressure
    successor_norm = iqr_normalize(successor_pressure)
    successor_boost = PARAMS["successor_criticality_weight"] * successor_norm

    # --- Fairness: wait time scaled by slack headroom (avoid boosting late tasks) ---
    wait_headroom = np.maximum(1.0, slack + 1.0)
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * wait_norm / (wait_headroom + eps)

    # --- Uncertainty interaction: multiplicative with slack pressure (bounded), IQR-normalized ---
    unc_norm = iqr_normalize(uncertainty)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded

    # --- Assemble final score: smaller = higher priority ---
    # critical_boost, work_density_bonus, and successor_boost are subtracted because higher values are beneficial
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_boost
        - work_density_bonus
        - successor_boost
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: replace inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
