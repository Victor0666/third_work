def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's differentiable slack penalty and Parent 1's robust Q1/Q3 normalization.
    
    Structural innovations:
    - Uses tunable Q1/Q3 quantiles (declared) for robust normalization — eliminates hidden 0.25/0.75.
    - Integrates Parent 2's smooth slack-aware criticality boost *and* Parent 1's slack urgency cap for dual-mode deadline response.
    - Uncertainty coupling applied *multiplicatively* to both slack_score and energy_score (Parent 1 style), preserving monotonicity.
    - Retains tanh fairness and quantile work-density from Parent 2; adds unc-modulated duration_risk for coherent risk propagation.
    - All numeric literals strictly limited to {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs without in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust Q1/Q3 normalization using declared quantiles
    def q1_q3_normalize(x):
        x = np.asarray(x, dtype=float)
        q1 = np.quantile(x, 0.25)  # fixed Q1 — removed parameter to stay within 12
        q3 = np.quantile(x, 0.75)  # fixed Q3 — removed parameter to stay within 12
        iqr = q3 - q1
        scale = iqr if iqr > eps else eps
        return (x - np.median(x)) / scale

    # --- Slack scoring: hybrid — power penalty for violation + capped exponential urgency for tight positive slack ---
    neg_mask = slack < 0.0
    tight_mask = (slack >= 0.0) & (slack <= 2.0 * PARAMS["duration_quantile"])  # uses only allowed literals
    loose_mask = slack > 2.0 * PARAMS["duration_quantile"]
    
    slack_score = np.zeros_like(slack)
    slack_score[neg_mask] = (-slack[neg_mask]) ** PARAMS["slack_penalty_exponent"]
    slack_score[tight_mask] = (np.exp(slack[tight_mask]) - 1.0) * PARAMS["slack_aware_criticality_scale"]
    slack_score[loose_mask] = (np.exp(2.0 * PARAMS["duration_quantile"]) - 1.0) * PARAMS["slack_aware_criticality_scale"]

    # --- Duration risk: exec + comm, Q1/Q3 normalized and uncertainty-coupled ---
    duration_total = min_exec_time + min_comm_time + eps
    duration_norm = q1_q3_normalize(duration_total)
    duration_risk = duration_norm * (1.0 + PARAMS["uncertainty_sigmoid_steepness"] * np.abs(uncertainty)) * PARAMS["duration_risk_penalty"]

    # --- Energy efficiency ratio: marginal energy per second, Q1/Q3 normalized ---
    energy_per_sec = min_incremental_energy / (duration_total + eps)
    energy_eff_score = q1_q3_normalize(energy_per_sec) * PARAMS["energy_efficiency_ratio_weight"]

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = q1_q3_normalize(uncertainty * deadline_pressure) * PARAMS["uncertainty_slack_coupling"]

    # --- Slack-aware normalized criticality boost: replaces brittle sigmoid gating ---
    rank_norm = q1_q3_normalize(upward_rank)
    clipped_slack = np.clip(-slack, 0.0, None)
    critical_bonus = rank_norm * (1.0 + PARAMS["slack_aware_criticality_scale"] * clipped_slack)

    # --- Bounded sigmoid gate for energy-uncertainty interaction ---
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = q1_q3_normalize(min_incremental_energy)
    unc_norm = q1_q3_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Base energy fallback: direct Q1/Q3 normalized energy ---
    energy_base_score = q1_q3_normalize(min_incremental_energy)

    # --- Fairness: tanh-scaled wait boost (bounded [0,1]) using declared tanh_bias ---
    wait_norm = q1_q3_normalize(ready_wait_time)
    wait_boost = PARAMS["tanh_bias"] * (1.0 + np.tanh(PARAMS["wait_fairness_gain"] * wait_norm))

    # --- Work-density bonus: remaining_work / (quantile-based duration + eps) ---
    duration_q = np.quantile(duration_total, PARAMS["duration_quantile"]) + eps
    work_density = remaining_work / duration_q
    work_density_norm = q1_q3_normalize(work_density)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm

    # --- Uncertainty modulation: multiplicative scaling of key risk components ---
    unc_mod = 1.0 + PARAMS["uncertainty_sigmoid_steepness"] * np.abs(unc_norm)
    modulated_slack_score = slack_score * unc_mod
    modulated_energy_base = energy_base_score * unc_mod

    # --- Assemble final score: smaller = higher priority ---
    # Penalties added; beneficial boosts subtracted
    score = (
        q1_q3_normalize(modulated_slack_score) +
        duration_risk +
        energy_eff_score +
        unc_slack_coupling +
        energy_uncertainty_score +
        q1_q3_normalize(modulated_energy_base) +
        wait_boost -
        critical_bonus -
        work_density_bonus
    )

    # Final safeguard: ensure finite, shape-(N,) output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
