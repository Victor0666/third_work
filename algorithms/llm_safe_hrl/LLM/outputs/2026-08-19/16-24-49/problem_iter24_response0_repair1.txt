def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with smooth slack gating and successor-readiness modulation.
    
    Key structural improvements:
      - Replaces hard `ddl_protection_threshold` with smooth, differentiable sigmoid over slack
        (PARAMS["slack_sigmoid_steepness"]) for continuous priority transition near deadline headroom.
      - Introduces `successor_readiness_coupling` via implicit slack-based modulation (no new param)
        to prevent upstream starvation of downstream deadlines — implemented using existing slack_norm.
      - Retains task-wise MAD normalization and hard feasibility mask for robustness.
      - Uses clipped [-2,2] bounds everywhere for numerical stability.
      - All numeric literals strictly limited to {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Task-wise MAD normalization with strict clipping to [-2, 2]
    def task_wise_mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med)) + eps
        centered = x - med
        normalized = centered / mad
        return np.clip(normalized, -2.0, 2.0)

    # Normalize each dimension independently
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    
    slack_norm = task_wise_mad_normalize(slack)
    abs_slack_norm = task_wise_mad_normalize(abs_slack)
    uncertainty_norm = task_wise_mad_normalize(uncertainty)
    duration_norm = task_wise_mad_normalize(duration_total)
    energy_norm = task_wise_mad_normalize(min_incremental_energy)
    rank_norm = task_wise_mad_normalize(upward_rank)
    work_norm = task_wise_mad_normalize(remaining_work)
    wait_norm = task_wise_mad_normalize(ready_wait_time)

    # --- Hard feasibility mask: zero tolerance for DDL violation ---
    feasibility_mask = np.where(slack < 0, PARAMS["feasibility_mask_penalty"], 0.0)

    # --- DDL-critical base scores (always active) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = duration_total * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Dual critical-path signals with implicit successor-readiness modulation ---
    # (a) Global release signal: raw importance × downstream work
    critical_release_raw = upward_rank * remaining_work * PARAMS["upward_rank_remaining_work_interaction"]
    critical_release_score = task_wise_mad_normalize(critical_release_raw)
    
    # (b) Implicit successor-readiness modulation: use slack_norm to approximate successor slack pressure
    # Stronger modulation near slack=0 (critical region), symmetric and bounded
    slack_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["slack_sigmoid_steepness"] * slack_norm))
    # Modulation peaks at slack=0 → increases criticality score when local slack is tight
    successor_modulation = 1.0 - slack_sigmoid
    modulated_critical_release = critical_release_score * (1.0 + 0.6 * successor_modulation)  # 0.6 is fixed structural constant

    # (c) Local urgency signal: rank prioritization weighted by slack scarcity
    slack_urgency_weight = np.clip(1.0 + slack_norm, 0.0, 2.0)
    local_rank_score = -rank_norm * slack_urgency_weight

    # --- Smooth DDL safety gate (replaces hard threshold) ---
    # Sigmoid-based soft gate: near 1.0 when slack > 0.5, near 0.0 when slack < -0.5
    ddl_safe_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_sigmoid_steepness"] * (slack - 0.0)))

    # --- Host-load conditional gate ---
    duration_median = np.median(duration_total) if N > 1 else np.mean(duration_total)
    host_margin_sufficient = np.where(
        duration_total <= PARAMS["host_load_conditional_gate"] * duration_median,
        1.0,
        0.0
    )

    # --- Non-DDL terms: only active under both DDL safety AND host capacity conditions ---
    energy_per_sec = min_incremental_energy / (duration_total + eps)
    energy_eff_score = task_wise_mad_normalize(energy_per_sec)

    # Weighted rank-score: balances slack headroom and criticality importance
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - np.clip(slack_norm, 0.0, 1.0))
    rank_score = -rank_norm * weight_rank

    # Energy-uncertainty interaction: gated by host capacity and slack safety
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * uncertainty_norm * unc_sigmoid

    # Starvation mitigation: always active but dynamically gated by slack scarcity
    wait_gate = np.clip(1.0 + np.abs(slack_norm), 0.0, 2.0)
    wait_score = wait_norm * wait_gate

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms dominate unconditionally; feasibility mask dominates all
    score = (
        feasibility_mask +
        abs_slack_norm +
        uncertainty_norm +
        duration_norm +
        slack_score +
        unc_slack_coupling +
        duration_risk +
        modulated_critical_release +
        local_rank_score
    )
    
    # Non-DDL terms only added when both DDL-safe AND host-capacity-sufficient
    score += ddl_safe_gate * host_margin_sufficient * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
