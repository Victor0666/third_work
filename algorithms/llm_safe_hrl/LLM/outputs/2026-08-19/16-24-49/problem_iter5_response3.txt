def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining DDL-protection gating (Parent 2) with smooth sigmoid energy gating (Parent 1),
       plus novel uncertainty-aware rank coupling and tanh-bounded slack-pressure normalization.
    
    Key innovations:
    - Replaces piecewise slack scoring with unified tanh-bounded slack-pressure signal (more stable than linear/exp branches).
    - Integrates sigmoid energy gating *inside* DDL-protection logic: energy term is smoothly suppressed *and* softened near DDL boundary.
    - Introduces uncertainty-weighted rank coupling: upward_rank * (1 + uncertainty * PARAMS["uncertainty_sensitivity"]) 
      normalized via MAD, enhancing critical-path focus under risk — even during protection mode.
    - Uses robust tanh-based normalization instead of MAD where possible for better gradient flow and boundedness.
    - All numeric literals are strictly in {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs: ensure finite, non-NaN, bounded values
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Tanh-based bounded normalization: preserves sign & monotonicity, strict [-1,1] output
    def tanh_norm(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = mad if mad > eps else eps
        normed = (x - med) / scale
        return np.tanh(normed)  # Smooth, bounded, differentiable

    # Unified slack-pressure signal: bounded, differentiable, captures both risk and urgency
    # Negative slack → large positive pressure; small positive slack → medium pressure; large positive → saturated low pressure
    slack_pressure = -slack
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])

    # Slack-driven urgency score: replaces Parent 2's piecewise and Parent 1's separate penalty/reward
    # Directly proportional to slack_pressure_bounded, scaled for sensitivity
    slack_score = slack_pressure_bounded * PARAMS["slack_risk_penalty"]

    # Duration term: exec + comm, tanh-normalized
    duration = min_exec_time + min_comm_time
    duration_norm = tanh_norm(duration)
    duration_term = PARAMS["duration_balance"] * duration_norm

    # Energy term: softly gated by sigmoid AND modulated by slack pressure
    energy_norm = tanh_norm(min_incremental_energy)
    energy_gate = 1.0 / (1.0 + np.exp(-PARAMS["energy_feasibility_sigmoid_steepness"] * 
                                     (slack - PARAMS["energy_feasibility_sigmoid_offset"])))
    # Energy weight decays smoothly as slack decreases (via tanh pressure)
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - slack_pressure_bounded)
    energy_term = energy_gate * energy_weight_adj * energy_norm

    # Critical path leverage: always active but strengthened under feasibility
    rank_uncertainty_coupled = upward_rank * (1.0 + uncertainty * PARAMS["uncertainty_sensitivity"])
    rank_norm = tanh_norm(rank_uncertainty_coupled)
    critical_gate = (slack >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate

    # Successor release impact: gated to slack >= 0, tanh-normalized
    successor_impact = upward_rank * remaining_work
    successor_norm = tanh_norm(successor_impact)
    successor_gate = (slack >= 0.0).astype(float)
    successor_bonus = PARAMS["successor_release_weight"] * successor_norm * successor_gate

    # Fairness: saturating wait bonus, applied only when slack >= 0
    wait_norm = tanh_norm(ready_wait_time)
    wait_score = (1.0 - np.exp(-PARAMS["wait_fairness_gain"] * np.abs(wait_norm))) * (slack >= 0.0).astype(float)

    # Uncertainty interaction: additive penalty when slack pressure is high
    unc_norm = tanh_norm(uncertainty)
    uncertainty_penalty = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded

    # DDL protection mode: soft blend (not hard switch) for CMA-ES stability
    ddl_protection_mode = (slack < PARAMS["ddl_protection_threshold"]).astype(float)
    # In protection mode: suppress energy, fairness, successor; retain urgency, duration, critical, and uncertainty terms
    base_score = (
        slack_score
        + duration_term
        + energy_term
        - critical_boost
        - successor_bonus
        + wait_score
        + uncertainty_penalty
    )
    
    # Protection fallback: enhanced critical boost (factor 2), no energy/fairness/successor
    protection_score = (
        slack_score
        + duration_term
        - (PARAMS["critical_path_leverage"] * 2) * rank_norm * critical_gate
        + uncertainty_penalty
    )

    # Soft blending ensures differentiability and avoids optimization cliffs
    score = ddl_protection_mode * protection_score + (1.0 - ddl_protection_mode) * base_score

    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
