def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """
    Self-evolved priority rule: replaces brittle threshold gates with smooth, bounded sigmoid slack gating;
    eliminates piecewise discontinuity and unstable early-reward terms; unifies urgency/criticality via monotonic risk scaling;
    retains robust normalization and finite safeguards.
    """
    eps = PARAMS["epsilon"]
    
    # Safe, finite, deterministic normalization: mean absolute + epsilon
    def robust_norm(x):
        x = np.asarray(x, dtype=np.float64)
        denom = np.mean(np.abs(x)) + eps
        return x / denom
    
    # Normalize base features
    norm_slack = robust_norm(slack)
    norm_energy = robust_norm(min_incremental_energy)
    norm_duration = robust_norm(min_exec_time + min_comm_time)
    norm_rank = robust_norm(upward_rank)
    norm_wait = robust_norm(ready_wait_time)
    norm_uncert = robust_norm(uncertainty)
    
    # === 1. Monotonic risk-aware slack gating: bounded sigmoid interpolation ===
    # Maps slack → [0,1] smoothly: 0 = neutral, near 1 = high urgency, near 0 = high criticality (long slack)
    # Uses parametrized sigmoid: 1 / (1 + exp(-steepness * (slack - offset)))
    # Offset negative to trigger urgency earlier; steepness controls transition sharpness
    sigmoid_input = PARAMS["slack_sigmoid_steepness"] * (norm_slack - PARAMS["slack_sigmoid_offset"])
    # Clamp input to avoid overflow in exp()
    sigmoid_input_clipped = np.clip(sigmoid_input, -PARAMS["urgency_clip_max"], PARAMS["urgency_clip_max"])
    slack_gate = 1.0 / (1.0 + np.exp(-sigmoid_input_clipped))
    
    # === 2. Urgency-driven score: exponential penalty scaled by sigmoid gate ===
    # Only active where slack is negative or slightly positive → avoids late scheduling
    urgency_base = np.maximum(0.0, -norm_slack)
    urgency_clipped = np.clip(urgency_base, 0.0, PARAMS["urgency_clip_max"])
    urgency_penalty = np.exp(PARAMS["slack_sigmoid_steepness"] * urgency_clipped) * slack_gate
    
    # === 3. Criticality-driven score: rank amplification under low-urgency regime (1 - slack_gate) ===
    critical_boost = PARAMS["criticality_weight"] * norm_rank * (1.0 - slack_gate)
    
    # === 4. Uncertainty-modulated deadline pressure: only in high-urgency region ===
    deadline_pressure = np.maximum(0.0, -norm_slack)
    uncertainty_coupled_pressure = PARAMS["uncertainty_slack_coupling"] * deadline_pressure * norm_uncert * slack_gate
    
    # === 5. Anti-starvation: pure exponential soft boost (simplified, stable) ===
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * norm_wait)
    
    # === 6. Energy-rank interaction penalty: discourage high-energy + high-rank scheduling ===
    rank_energy_penalty = PARAMS["rank_energy_interaction"] * norm_energy * norm_rank
    
    # === Final score: smaller = better ===
    score = (
        urgency_penalty
        + critical_boost
        + uncertainty_coupled_pressure
        + rank_energy_penalty
        + PARAMS["energy_sensitivity"] * norm_energy
        + PARAMS["duration_balance"] * norm_duration
        - wait_boost
    )
    
    # Final safeguard: ensure finite output
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=-finfo.max)
    
    return score
