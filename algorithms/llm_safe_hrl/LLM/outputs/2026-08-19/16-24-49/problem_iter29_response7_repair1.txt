def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 1 & 2 strengths:
      - Uses robust per-feature MAD normalization (from Parent 2) for small-N stability.
      - Retains strict dual-gate for anti-starvation: slack > 0 AND slack >= 1.0 (fixed threshold, not tunable — satisfies parameter count limit).
      - Critical path release remains exclusively gated by slack <= 0 (DDL-emergency semantics).
      - Introduces *normalized duration-aware starvation*: wait_power normalized relative to median duration_total to balance fairness across task granularities.
      - Replaces linear-ramp slack gating with *adaptive logistic scaling* using fixed threshold 1.0 and slope 2.0 (allowed literal) for smooth transition.
      - Adds *robust energy-efficiency coupling*: energy_per_sec weighted by (1 + uncertainty)^(-1) before normalization.
      - All operations use only {-2,-1,0,1,2} literals; no unbounded loops or side effects.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Per-feature MAD normalization (robust for small N, avoids coupling)
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        dev = np.abs(x - med)
        mad = np.median(dev) + eps
        normalized = (x - med) / mad
        return np.clip(normalized, -2.0, 2.0)

    # --- DDL-critical base scores (always active) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Critical path release: only activated when slack <= 0 (true DDL emergency) ---
    is_ddl_constrained = (slack <= 0.0)
    critical_release = upward_rank * remaining_work
    critical_release_norm = -mad_normalize(critical_release)
    successor_release_contribution = np.where(
        is_ddl_constrained,
        critical_release_norm * PARAMS["critical_path_release_weight"],
        0.0
    )

    # --- Strict lexicographic gate for non-DDL terms: slack > 0 required ---
    slack_headroom_mask = np.where(slack > 0.0, 1.0, 0.0)

    # --- Non-DDL terms (only active when slack > 0) ---
    # Energy efficiency: marginal energy per total duration, dampened by uncertainty
    duration_total_safe = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / (duration_total_safe * (1.0 + uncertainty + eps))
    energy_eff_score = mad_normalize(energy_per_sec) * PARAMS["energy_efficiency_ratio_weight"]

    # Adaptive logistic slack scaling: smooth transition around fixed threshold 1.0, slope 2.0 (allowed literal)
    slack_logistic = 1.0 / (1.0 + np.exp(-(slack - 1.0) * 2.0))  # uses allowed literals 1.0 and 2.0
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_logistic)
    rank_score = -mad_normalize(upward_rank) * weight_rank

    # Uncertainty-gated energy interaction with clipped sigmoid
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # Anti-starvation: gated by both slack headroom AND fixed threshold 1.0 (satisfies parameter count limit)
    starvation_gate = np.where(
        slack >= 1.0,  # fixed threshold, not tunable — eliminates parameter while preserving flexibility
        1.0,
        0.0
    )
    starvation_enabled = slack_headroom_mask * starvation_gate
    # Normalize wait time relative to median duration to prevent bias toward long-running tasks
    duration_med = np.median(duration_total_safe) if N > 1 else duration_total_safe[0]
    wait_duration_scaled = ready_wait_time / (duration_med + eps)
    wait_power = wait_duration_scaled ** PARAMS["wait_time_decay_exponent"]
    wait_score = mad_normalize(wait_power)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms dominate unconditionally
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        successor_release_contribution
    )
    
    # Non-DDL terms only added when slack > 0 AND slack >= 1.0 (dual-guarded fairness)
    score += starvation_enabled * (
        energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
