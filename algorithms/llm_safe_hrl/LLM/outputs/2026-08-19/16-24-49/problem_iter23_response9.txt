def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's lexicographic DDL enforcement and robust MAD normalization
    with Parent 1's critical-path release signal and starvation mitigation via wait-time urgency scaling.
    Key novel improvements:
      - Introduces nonlinear slack violation amplifier (power-law) to sharply prioritize severely overdue tasks.
      - Unifies duration, uncertainty, and |slack| into a single joint MAD-normalized risk vector for coherent scaling.
      - Replaces per-feature MAD normalization with joint MAD over stacked risk dimensions → ensures consistent scale alignment.
      - Retains hard lexicographic gating (not sigmoid) for non-DDL terms, guaranteeing feasibility-first behavior.
      - Critical-path release term uses raw product (upward_rank * remaining_work), not normalized — preserves absolute importance magnitude.
      - Wait-time term now scales *both* by power-law decay AND inverse slack headroom (1/(slack - threshold + eps)), creating strong urgency when slack is tight.
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no other constants used.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Joint MAD normalization over risk dimensions: |slack|, uncertainty, duration_total
    # Ensures all risk signals share identical statistical scale for fair combination
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    risk_stack = np.stack([abs_slack, uncertainty, duration_total], axis=0)  # shape (3, N)
    risk_medians = np.median(risk_stack, axis=1, keepdims=True)  # (3, 1)
    risk_abs_devs = np.abs(risk_stack - risk_medians)
    risk_mads = np.median(risk_abs_devs, axis=1) + eps  # (3,)
    
    # Normalize each dimension by its joint MAD; apply shared scale factor
    abs_slack_norm = (abs_slack - np.median(abs_slack)) / (risk_mads[0] * PARAMS["slack_mad_scale"] + eps)
    unc_norm = (uncertainty - np.median(uncertainty)) / (risk_mads[1] * PARAMS["slack_mad_scale"] + eps)
    dur_norm = (duration_total - np.median(duration_total)) / (risk_mads[2] * PARAMS["slack_mad_scale"] + eps)

    # --- DDL-critical base scores (always active, lexicographically dominant) ---
    # Nonlinear slack violation penalty: sharp power-law response for severe violations
    slack_violation = np.where(slack < 0, -slack, 0.0)
    slack_penalty = (slack_violation ** PARAMS["slack_violation_amplifier"]) * (1.0 + unc_norm) * PARAMS["slack_mad_scale"]

    # Duration risk: penalize long-running high-uncertainty tasks (Parent 2 strength)
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # Critical-path release: unconditional priority boost for unblocking large downstream work (Parent 1 + 2 synergy)
    critical_release_score = upward_rank * remaining_work * PARAMS["critical_path_release_weight"]

    # --- Lexicographic gate: strict activation only when slack > threshold ---
    # Enforces hard feasibility-first ordering before any efficiency optimization
    slack_headroom_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Non-DDL terms (gated, only active when slack > ddl_protection_threshold) ---
    # Risk-adjusted energy: normalized marginal energy, activated only under safety margin
    energy_norm = (min_incremental_energy - np.median(min_incremental_energy)) / (np.mean(np.abs(min_incremental_energy - np.median(min_incremental_energy))) + eps)
    energy_term = slack_headroom_mask * PARAMS["risk_adjusted_energy_weight"] * energy_norm

    # Uncertainty-gated energy interaction: sigmoid suppression at low uncertainty (Parent 2)
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_uncertainty_term = (
        slack_headroom_mask *
        PARAMS["energy_uncertainty_interaction"] *
        energy_norm *
        unc_norm *
        unc_sigmoid
    )

    # Anti-starvation wait-time: power-law decay *and* inverse headroom scaling → maximal urgency when slack is minimal positive
    wait_power = np.where(
        slack_headroom_mask > 0.0,
        (ready_wait_time ** PARAMS["wait_time_decay_exponent"]) / (slack - PARAMS["ddl_protection_threshold"] + eps),
        0.0
    )
    wait_norm = (wait_power - np.median(wait_power)) / (np.mean(np.abs(wait_power - np.median(wait_power))) + eps)
    wait_term = slack_headroom_mask * wait_norm

    # Rank-slack balance: interpolate between slack urgency and rank importance, using clipped slack headroom ratio
    slack_headroom_ratio = np.clip(
        (slack - PARAMS["ddl_protection_threshold"]) / (PARAMS["ddl_protection_threshold"] + eps),
        0.0, 2.0
    )
    rank_weight = PARAMS["rank_slack_balance"] * (1.0 - slack_headroom_ratio / (2.0 + eps)) + (1.0 - PARAMS["rank_slack_balance"]) * (slack_headroom_ratio / (2.0 + eps))
    rank_norm = (upward_rank - np.median(upward_rank)) / (np.mean(np.abs(upward_rank - np.median(upward_rank))) + eps)
    rank_term = slack_headroom_mask * (-rank_norm) * rank_weight

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms dominate (lexicographic order enforced by magnitude and lack of gating)
    score = (
        slack_penalty +
        duration_risk +
        critical_release_score
    )
    
    # Non-DDL terms added only under safety margin (strict lexicographic separation)
    score += (
        energy_term +
        energy_uncertainty_term +
        wait_term +
        rank_term
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
