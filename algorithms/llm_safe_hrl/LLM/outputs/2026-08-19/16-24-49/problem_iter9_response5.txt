def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with two key structural improvements:
    1. Replaces fragile critical_path_leverage with stable slack-aware normalization — decouples scaling
       for deadline-critical (slack < 0) vs. slack-rich (slack >= 0) tasks using alpha-mixed medians/MADs.
    2. Introduces bounded sigmoidal energy headroom gating (smooth, numerically stable, no power-law overflow).
    All features are normalized *within their slack context*, ensuring fairness across deadline regimes.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # --- Slack-aware robust normalization: mix global and slack-feasible subset statistics ---
    # Identify feasible subset (non-negative slack) for conditional centering/scaling
    feasible_mask = (slack >= 0.0)
    has_feasible = np.any(feasible_mask)
    
    def slack_aware_mad_normalize(x):
        x = np.asarray(x, dtype=float)
        # Global stats
        global_med = np.median(x)
        global_mad = np.median(np.abs(x - global_med))
        
        # Feasible-subset stats (if any exist)
        if has_feasible:
            x_feas = x[feasible_mask]
            feasible_med = np.median(x_feas)
            feasible_mad = np.median(np.abs(x_feas - feasible_med)) if len(x_feas) > 1 else eps
        else:
            feasible_med = global_med
            feasible_mad = global_mad
        
        # Alpha-weighted blend: higher alpha → more feasible-subset influence
        med = PARAMS["slack_aware_normalization_alpha"] * feasible_med + (1.0 - PARAMS["slack_aware_normalization_alpha"]) * global_med
        mad = PARAMS["slack_aware_normalization_alpha"] * feasible_mad + (1.0 - PARAMS["slack_aware_normalization_alpha"]) * global_mad
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack transformation: unified pressure via tanh with shared scale ---
    slack_pressure = np.tanh((-slack) * PARAMS["slack_decay_scale"])
    slack_norm = PARAMS["slack_risk_penalty"] * slack_pressure + PARAMS["slack_urgency_scale"] * (1.0 - np.tanh(slack * PARAMS["slack_decay_scale"]))

    # --- Duration: exec + comm, slack-aware normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = slack_aware_mad_normalize(duration)

    # --- Energy: gated by fairness headroom using bounded sigmoid (stable, smooth, no overflow) ---
    fairness_headroom = np.divide(ready_wait_time, np.abs(slack) + eps)
    energy_norm = slack_aware_mad_normalize(min_incremental_energy)
    # Sigmoidal gate: 1/(1 + exp(-k*(headroom - 0.5))) → ~0 at headroom=0, ~1 at headroom>=1
    energy_headroom_gate = 1.0 / (1.0 + np.exp(-PARAMS["energy_headroom_sigmoid_k"] * (fairness_headroom - 0.5)))
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * energy_headroom_gate

    # --- Fairness: saturating exponential wait boost, slack-aware normalized ---
    wait_norm = slack_aware_mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * (1.0 - np.exp(-PARAMS["wait_fairness_gain"] * (wait_norm + eps)))

    # --- Uncertainty interaction: multiplicative only under high slack pressure ---
    unc_norm = slack_aware_mad_normalize(uncertainty)
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure

    # --- Work density bonus: remaining_work / (duration + eps), slack-aware normalized and gated ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = slack_aware_mad_normalize(work_density)
    # Gate bonus only where slack permits waiting (positive headroom) to avoid starving critical paths
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * (fairness_headroom > eps).astype(float)

    # --- Assemble final score: smaller = higher priority ---
    # No critical_path_leverage → removed per self-reflection; replaced by slack-aware scaling of all features
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - work_density_bonus
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: replace inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
