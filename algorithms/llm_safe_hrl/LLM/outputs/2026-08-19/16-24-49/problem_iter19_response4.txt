def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule emphasizing:
      - Strict hard gate: only activate non-DDL terms when slack > ddl_protection_threshold (not just >0)
      - Critical path release via upward_rank × remaining_work product — unconditionally prioritized for urgency
      - Joint MAD normalization over |slack|, uncertainty, and duration_total for coherent risk scaling
      - Wait-time penalty scaled by (1 + ready_wait_time / (max(1, median(slack[slack>0]) + eps))) to avoid dominance
      - Risk-aware energy term only active in safe region, coupled with uncertainty via product
      - All numeric literals restricted to {-2,-1,0,1,2}; no other constants
      - Lexicographic DDL enforcement preserved: violation penalties dominate; safe-region terms are additive and gated
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Joint MAD normalization over key risk dimensions: |slack|, uncertainty, duration_total
    # Ensures comparable scale for robust interaction — avoids fragile percentile clipping
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    all_risk_dims = np.stack([abs_slack, uncertainty, duration_total], axis=0)
    mad_per_dim = np.mean(np.abs(all_risk_dims - np.median(all_risk_dims, axis=1, keepdims=True)), axis=1) + eps
    # Normalize each dimension by its MAD, then take element-wise max for unified risk scale
    norm_slack = abs_slack / (mad_per_dim[0] * PARAMS["duration_mad_scale"] + eps)
    norm_uncert = uncertainty / (mad_per_dim[1] * PARAMS["duration_mad_scale"] + eps)
    norm_duration = duration_total / (mad_per_dim[2] * PARAMS["duration_mad_scale"] + eps)
    joint_risk_scale = np.maximum.reduce([norm_slack, norm_uncert, norm_duration])

    # --- DDL-violation penalty: always active, monotonic, bounded away from zero ---
    # Uses signed slack directly: negative slack → large penalty; positive slack → small base
    ddl_penalty = np.where(
        slack < 0,
        (-slack) * (1.0 + norm_uncert * PARAMS["uncertainty_slack_interaction"]),
        0.0
    )

    # --- Critical path release urgency: unconditionally active, based on HEFT criticality × work ---
    # Prioritizes tasks that unlock large downstream computation — prevents critical-path starvation
    critical_release_score = -(upward_rank * remaining_work)

    # --- Safe-region activation mask: only enable energy/wait terms when slack > threshold ---
    safe_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Energy efficiency term: only active in safe region, normalized and weighted ---
    energy_norm = (min_incremental_energy + eps) / (joint_risk_scale + eps)
    energy_term = safe_mask * PARAMS["risk_adjusted_energy_weight"] * (energy_norm / (np.median(energy_norm) + eps))

    # --- Uncertainty-coupled energy term: multiplicative interaction under safety ---
    energy_uncert_term = safe_mask * PARAMS["energy_uncertainty_coupling"] * (energy_norm * norm_uncert)

    # --- Anti-starvation wait term: scaled by relative wait time, capped at 1.0 per task ---
    # Prevents long waits without overwhelming DDL signals
    safe_slack_median = np.median(slack[slack > PARAMS["ddl_protection_threshold"]]) if np.any(slack > PARAMS["ddl_protection_threshold"]) else 1.0
    wait_scale = 1.0 + ready_wait_time / (np.maximum(safe_slack_median, 1.0) + eps)
    wait_term = safe_mask * PARAMS["wait_starvation_penalty"] * (ready_wait_time / (wait_scale + eps))

    # --- Assemble final score: smaller = higher priority ---
    # DDL penalty dominates; critical release is always subtracted (higher rank/work → lower score)
    # All other terms are additive contributions to priority score
    score = (
        ddl_penalty +
        critical_release_score * PARAMS["critical_path_release_weight"] +
        energy_term +
        energy_uncert_term +
        wait_term
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
