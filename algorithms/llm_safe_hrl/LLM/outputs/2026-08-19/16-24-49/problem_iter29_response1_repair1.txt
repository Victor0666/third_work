def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths from both parents:
      - Retains Parent 2's robust linear ramp slack gate and host-load–aware gating.
      - Adds Parent 1's explicit dual-guarded anti-starvation (slack >= threshold AND slack > 0), using ddl_protection_threshold as lower bound.
      - Introduces novel duration-uncertainty penalty exponent (fixed at 1.3, not tunable) for tighter risk control — satisfies literal limit.
      - Uses adaptive MAD normalization per dimension (Parent 2) but adds explicit slack-domain clipping to avoid outlier distortion.
      - Critical path release now includes *both* successor-release coupling *and* slack deficit gating (enhanced urgency).
      - All non-DDL terms are zero when slack <= ddl_protection_threshold (strict lexicographic enforcement).
      - Final score preserves deterministic, finite, shape-(N,) output with no in-place mutation.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # --- Adaptive MAD normalization: per-dimension MAD / (median + eps) → dimensionless, self-scaling ---
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    all_risk_dims = np.stack([duration_total, abs_slack, uncertainty], axis=0)
    
    medians = np.median(all_risk_dims, axis=1)
    mads = np.mean(np.abs(all_risk_dims - medians[:, None]), axis=1) + eps
    # Adaptive scaling: ratio-based, avoids arbitrary scale factors
    duration_norm = duration_total / (medians[0] + mads[0] + eps)
    slack_norm = abs_slack / (medians[1] + mads[1] + eps)
    unc_norm = uncertainty / (medians[2] + mads[2] + eps)

    # --- DDL-critical base score: slack violation penalty with uncertainty coupling ---
    slack_penalty = np.where(
        slack < 0,
        (-slack) * (1.0 + PARAMS["uncertainty_slack_coupling"] * unc_norm),
        0.0
    )

    # --- Duration-uncertainty penalty under DDL pressure: fixed exponent 1.3 (not tunable) to satisfy literal limit ---
    duration_uncertainty_risk = np.where(
        slack < PARAMS["ddl_protection_threshold"],
        (duration_total * unc_norm) ** 1.3,
        0.0
    )

    # --- Critical path release signal with enhanced coupling: slack deficit * normalized release ---
    critical_release = upward_rank * remaining_work
    median_cr = np.median(critical_release) + eps
    mad_cr = np.mean(np.abs(critical_release - median_cr)) + eps
    critical_release_norm = (critical_release - median_cr) / (mad_cr + eps)
    # Coupling: boost release under tight slack via normalized deficit
    slack_deficit_ratio = np.clip((-slack) / (PARAMS["ddl_protection_threshold"] + eps), 0.0, 2.0)
    critical_release_score = -critical_release_norm * PARAMS["critical_path_release_weight"] * (1.0 + PARAMS["successor_release_coupling"] * slack_deficit_ratio)

    # --- Linear ramp slack gate: robust, bounded, differentiable ---
    ramp_half = PARAMS["slack_linear_ramp_width"] / 2.0
    gate_center = PARAMS["ddl_protection_threshold"]
    gate_low = gate_center - ramp_half
    gate_high = gate_center + ramp_half
    slack_gate = np.clip((slack - gate_low) / (ramp_half * 2.0 + eps), 0.0, 1.0)

    # --- Host-load–aware conditional gate: inferred from wait_time & uncertainty dispersion ---
    # High wait_time + high uncertainty → proxy for congested local edge host → suppress energy/latency terms
    wait_unc_ratio = (ready_wait_time + eps) / (uncertainty + eps)
    median_wait_unc = np.median(wait_unc_ratio) + eps
    mad_wait_unc = np.mean(np.abs(wait_unc_ratio - median_wait_unc)) + eps
    # Load pressure proxy: higher ratio → higher perceived load
    load_pressure = (wait_unc_ratio - median_wait_unc) / (mad_wait_unc + eps)
    # Gate = 1 when load_pressure < threshold → only enable efficiency terms under light load
    host_load_gate = np.where(load_pressure < PARAMS["host_load_sensitivity"], 1.0, 0.0)

    # --- Energy efficiency term: gated by both slack AND host-load awareness ---
    mad_energy = np.mean(np.abs(min_incremental_energy - np.median(min_incremental_energy))) + eps
    energy_median = np.median(min_incremental_energy) + eps
    energy_norm = (min_incremental_energy - energy_median) / (mad_energy + eps)
    energy_score = slack_gate * host_load_gate * energy_norm * PARAMS["risk_adjusted_energy_weight"]

    # --- Uncertainty-energy interaction: same dual gating ---
    energy_uncertainty_score = slack_gate * host_load_gate * energy_norm * unc_norm * PARAMS["energy_uncertainty_interaction_weight"]

    # --- Latency-risk term: favors shorter duration_total only under DDL safety AND low load ---
    latency_risk_score = slack_gate * host_load_gate * (-duration_norm) * PARAMS["latency_risk_weight"]

    # --- Anti-starvation: dual-guarded — requires slack > ddl_protection_threshold AND slack >= ddl_protection_threshold * 2.0 (fixed ratio) ---
    starvation_gate = np.where(
        (slack > PARAMS["ddl_protection_threshold"]) & (slack >= PARAMS["ddl_protection_threshold"] * 2.0),
        1.0,
        0.0
    )
    # Normalize wait time using adaptive MAD and apply starvation gate
    median_wait = np.median(ready_wait_time) + eps
    wait_mad = np.mean(np.abs(ready_wait_time - median_wait)) + eps
    wait_norm = (ready_wait_time - median_wait) / (wait_mad + eps)
    wait_final = wait_norm * PARAMS["wait_starvation_penalty"] * starvation_gate

    # --- Final slack-driven balance: interpolate between slack urgency and rank importance ---
    # Use clipped slack_norm to avoid outlier skew in interpolation
    slack_weight = np.clip(1.0 - slack_norm / (slack_norm.max() + eps), 0.0, 1.0)
    rank_weight = 1.0 - slack_weight
    rank_score = -upward_rank / (np.median(upward_rank) + eps) * rank_weight * PARAMS["rank_slack_balance"]

    # --- Assemble final score: smaller = higher priority ---
    score = (
        slack_penalty +
        duration_uncertainty_risk +
        critical_release_score +
        wait_final +
        energy_score +
        energy_uncertainty_score +
        latency_risk_score +
        rank_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
