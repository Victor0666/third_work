def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Hard feasibility-aware switch: below `critical_slack_threshold`, score prioritizes slack urgency (|slack| when negative, else 0);
        above it, defaults to critical-path release — eliminates fragile interpolation.
      - Bounded min-max normalization for slack, uncertainty, duration_total (replacing MAD) to stabilize small-N ready sets.
      - Decoupled anti-starvation: uses normalized `ready_wait_time` directly (not scaled by headroom), gated only by DDL feasibility.
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no other constants.
      - Final score is deterministic, finite, shape-(N,), and satisfies 'smaller = higher priority'.
    """
    eps = PARAMS["epsilon"]
    mm_eps = PARAMS["minmax_norm_epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Bounded min-max normalization: robust for small-N, avoids outlier distortion
    def minmax_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        x_min = np.min(x)
        x_max = np.max(x)
        denom = x_max - x_min + mm_eps
        normalized = (x - x_min) / denom
        return np.clip(normalized, 0.0, 1.0)

    # Joint risk features for coherent scale alignment
    duration_total = min_exec_time + min_comm_time + eps
    slack_abs = np.abs(slack)

    # --- DDL-critical base scores (always active, lexicographically dominant) ---
    # Exponentiated slack penalty for lateness risk
    slack_penalty = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    # Duration risk
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # Unconditional critical-path release
    critical_release_score = upward_rank * remaining_work * PARAMS["critical_path_release_weight"]

    # --- Hard feasibility-aware switch: urgency vs. critical-path ---
    # Below critical_slack_threshold: prioritize urgency (absolute slack magnitude when negative)
    # Above: default to critical-path release (no blending)
    urgency_mask = np.where(slack < PARAMS["critical_slack_threshold"], 1.0, 0.0)
    urgency_score = np.where(
        slack < 0,
        slack_abs,
        0.0
    )
    # Normalize urgency score only over negative-slack subset for fairness
    if np.any(slack < 0):
        urgency_norm = minmax_normalize(urgency_score)
    else:
        urgency_norm = np.zeros_like(urgency_score)
    # Apply hard switch: urgency dominates below threshold; critical release dominates above
    switched_base_score = (
        urgency_mask * urgency_norm +
        (1.0 - urgency_mask) * minmax_normalize(critical_release_score)
    )

    # --- Lexicographic gate: non-DDL terms only active when slack > threshold ---
    slack_headroom_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Non-DDL terms (gated only, not scaled by headroom) ---
    # Risk-adjusted energy term
    energy_norm = minmax_normalize(min_incremental_energy)
    energy_term = slack_headroom_mask * PARAMS["risk_adjusted_energy_weight"] * energy_norm

    # Anti-starvation: normalized wait time, gated only by feasibility — no headroom scaling
    wait_norm = minmax_normalize(ready_wait_time)
    wait_term = slack_headroom_mask * (wait_norm ** PARAMS["wait_time_decay_exponent"])

    # Uncertainty-gated energy interaction
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    unc_norm = minmax_normalize(uncertainty)
    energy_uncertainty_term = (
        slack_headroom_mask *
        PARAMS["energy_uncertainty_interaction"] *
        energy_norm *
        unc_norm *
        unc_sigmoid
    )

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms dominate unconditionally
    score = (
        minmax_normalize(slack_penalty) +
        minmax_normalize(duration_risk) +
        switched_base_score
    )
    
    # Non-DDL terms added only when slack allows
    score += (
        energy_term +
        wait_term +
        energy_uncertainty_term
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
