def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating conditional DDL protection gate and successor-release-aware interactions.
    
    Key structural mutations:
    - Replaces linear wait fairness with smooth anti-starvation term: 1 - exp(-rate * norm_wait)
      to avoid clipping artifacts and improve cross-seed stability.
    - Introduces successor-release interaction: multiplies critical path boost by normalized remaining_work,
      prioritizing tasks whose execution unblocks high-work descendants — addresses 'successor_blocked' patterns.
    - Adds host-load-aware energy dampening: energy weight decays exponentially with slack pressure instead of tanh,
      improving gradient continuity near zero slack.
    - Uses robust mean-absolute scaling (not MAD) for all feature normalizations to resist outliers and align with
      counterfactual evidence recommending improved cross-seed stability.
    - Applies strict conditional DDL protection gate: critical path leverage and work-density bonus are only active
      when slack >= 0 AND uncertainty <= median_uncertainty, preventing over-commitment to risky paths.
    - Removes duration_balance from raw duration; instead applies it to *normalized* duration after robust scaling.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust mean-absolute deviation (MAD) equivalent via mean absolute deviation (not median) for smoother gradients
    # Uses np.mean(abs(x - mean)) instead of np.median(abs(x - median)) — more differentiable and stable under small N
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        mu = np.mean(x)
        mad = np.mean(np.abs(x - mu))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - mu) / scale

    # --- Slack transformation: bounded exponential urgency for negative/zero slack, capped linear for large positive ---
    # Avoids infinite growth; uses clipped linear beyond slack_cap for stability
    slack_norm = np.where(
        slack < 0.0,
        PARAMS["slack_risk_penalty"] * (-slack),
        np.where(
            slack < 1.0,
            PARAMS["slack_urgency_scale"] * (np.exp(slack) - 1.0),
            np.clip(slack, 0.0, PARAMS["slack_cap"])
        )
    )

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = robust_normalize(duration)

    # --- Energy: exponentially dampened under deadline pressure (smoother than tanh) ---
    energy_norm = robust_normalize(min_incremental_energy)
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    # Exponential decay: weight → 0 as slack_pressure increases, avoids sharp transitions
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * np.exp(-slack_pressure * PARAMS["slack_pressure_tanh_scale"])

    # --- Critical path & successor release interaction: conditional on slack feasibility AND low uncertainty ---
    rank_norm = robust_normalize(upward_rank)
    work_norm = robust_normalize(remaining_work)
    # DDL protection gate: only activate if slack >= 0 AND uncertainty below median (low-risk context)
    median_unc = np.median(uncertainty)
    ddl_protection_gate = ((slack >= 0.0) & (uncertainty <= median_unc + eps)).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * work_norm * ddl_protection_gate

    # --- Work density: remaining_work per unit duration, gated by DDL protection ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = robust_normalize(work_density)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * ddl_protection_gate

    # --- Fairness: smooth saturating anti-starvation term (1 - exp(-rate * norm_wait)) ---
    wait_norm = robust_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * (1.0 - np.exp(-PARAMS["wait_fairness_gain"] * wait_norm))

    # --- Uncertainty interaction: additive coupling with slack pressure (bounded) ---
    unc_norm = robust_normalize(uncertainty)
    # Use soft sigmoid instead of tanh for gentler saturation and better gradient behavior
    slack_pressure_bounded = 1.0 / (1.0 + np.exp(-slack_pressure * PARAMS["slack_pressure_tanh_scale"]))
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded

    # --- Assemble final score: smaller = higher priority ---
    # All beneficial terms (critical_boost, work_density_bonus) subtracted; others added
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_boost
        - work_density_bonus
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
