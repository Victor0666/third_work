def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robustness from Parent 2 and structural clarity from Parent 1:
      - Joint MAD normalization over |slack|, uncertainty, and duration_total (robust scaling)
      - Lexicographic DDL enforcement: violation penalty raised to power >1 for strict dominance
      - Critical-path urgency uses smooth pressure decay (not binary gating) to avoid discontinuities
      - Wait-term uses relative wait efficiency (ready_wait_time / duration_total) instead of absolute time
      - Energy-uncertainty coupling gated by bounded sigmoid on uncertainty (centered at 1.0)
      - All numeric literals restricted to {-2,-1,0,1,2}; no other constants
      - No percentile clipping or fragile medians on empty subsets; safe fallbacks used
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Joint MAD normalization: robustly aligns |slack|, uncertainty, and duration_total scales
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    all_risk_features = np.stack([abs_slack, uncertainty, duration_total], axis=0)
    mad_per_dim = np.mean(np.abs(all_risk_features - np.median(all_risk_features, axis=1, keepdims=True)), axis=1) + eps
    joint_mad = np.median(mad_per_dim) + eps

    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        centered = x - np.median(x)
        denom = joint_mad * PARAMS["duration_mad_scale"] + eps
        return np.clip(centered / denom, -2.0, 2.0)

    # --- DDL-critical base penalty: always active, exponentiated for strict dominance ---
    # Stronger nonlinear penalty for negative slack; zero when slack >= threshold
    ddl_violation_base = np.maximum(0.0, PARAMS["ddl_protection_threshold"] - slack)
    ddl_violation_penalty = np.power(ddl_violation_base + eps, PARAMS["ddl_violation_exponent"])

    # --- Critical-path release urgency: smooth pressure decay instead of hard gate ---
    # HEFT-inspired criticality: upward_rank * remaining_work reflects descendant work importance
    critical_release_score = upward_rank * remaining_work
    # Decay urgency smoothly when slack > threshold: full weight at violation, reduced above threshold
    slack_ratio = np.clip((PARAMS["ddl_protection_threshold"] - slack) / (PARAMS["ddl_protection_threshold"] + eps), 0.0, 1.0)
    # Apply decay factor to reduce urgency in safe region
    critical_weight = PARAMS["critical_path_release_weight"] * (PARAMS["critical_pressure_decay"] + (1.0 - PARAMS["critical_pressure_decay"]) * slack_ratio)
    critical_score = -robust_normalize(critical_release_score) * critical_weight

    # --- Feasibility mask: only enable non-DDL terms when slack > threshold ---
    feasible_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Risk-adjusted energy term: only active in feasible region, normalized ---
    energy_norm = robust_normalize(min_incremental_energy)
    energy_term = feasible_mask * PARAMS["risk_adjusted_energy_weight"] * energy_norm

    # --- Uncertainty-slack interaction: penalizes high uncertainty near deadline ---
    unc_slack_interaction = uncertainty * (PARAMS["ddl_protection_threshold"] - slack) * PARAMS["uncertainty_slack_interaction"]
    unc_slack_norm = robust_normalize(unc_slack_interaction)

    # --- Starvation mitigation: wait efficiency (relative to duration) instead of absolute wait time ---
    wait_efficiency = np.where(duration_total > eps, ready_wait_time / duration_total, 0.0)
    wait_norm = robust_normalize(wait_efficiency)
    wait_term = feasible_mask * PARAMS["wait_starvation_penalty"] * wait_norm

    # --- Energy-uncertainty coupling: bounded sigmoid ensures suppression at extremes ---
    # Sigmoid centered at uncertainty = 1.0: suppresses energy optimization when uncertainty is very low or very high
    unc_sigmoid = 1.0 / (1.0 + np.exp(-2.0 * (1.0 - uncertainty)))
    energy_uncertainty_term = feasible_mask * PARAMS["energy_uncertainty_coupling"] * energy_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    # DDL violation dominates lexicographically via exponentiation
    score = (
        robust_normalize(ddl_violation_penalty) +
        critical_score +
        unc_slack_norm
    )
    
    # Add feasible-only terms
    score += energy_term + wait_term + energy_uncertainty_term

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
