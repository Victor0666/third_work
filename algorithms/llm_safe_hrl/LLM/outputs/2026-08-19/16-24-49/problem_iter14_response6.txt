def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining strengths from both parents:
      - Retains Parent 2's DDL-aware critical_release_score and robustness_mad_factor.
      - Adopts Parent 1's unified slack-normalized ddl_gate for smooth blending of non-DDL terms.
      - Replaces hard binary gates (e.g., slack <= 0) with continuous, bounded sigmoid activation (ddl_gate) to avoid instability.
      - Introduces novel wait_fairness_steepness parameter to tune anti-starvation response — decoupled from other parameters for independent optimization.
      - Uses single robust MAD normalization function with PARAMS["robustness_mad_factor"] for all features — improves stability across variable N.
      - Enforces strict lexicographic ordering: DDL-critical terms dominate; non-DDL terms smoothly fade in only when slack is safe.
      - All numeric literals are strictly {-2,-1,0,1,2}; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust MAD normalization with configurable scale factor
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad + eps)
        return (x - med) / scale

    # --- DDL-critical base scores ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Critical-path release: activated via continuous ddl_gate, not binary mask ---
    # Prevents abrupt transitions seen in Parent 2's binary gate while preserving urgency under pressure
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_centered = slack - slack_lb
    slack_range = slack_ub - slack_lb + eps
    slack_normalized = np.clip(slack_centered / slack_range, -2.0, 2.0)
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * slack_normalized))
    critical_release_score = (
        upward_rank * remaining_work * PARAMS["criticality_release_weight"] * (1.0 - ddl_gate)
    )

    # --- Energy efficiency ratio: joules per second, normalized ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_norm = mad_normalize(energy_per_sec)

    # --- Rank-driven score: weighted by slack safety (higher weight when slack is tight) ---
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - ddl_gate)
    rank_norm = mad_normalize(upward_rank)
    rank_score = -rank_norm * weight_rank  # higher rank → lower score → higher priority

    # --- Energy-uncertainty interaction: gated by uncertainty sigmoid ---
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Anti-starvation: clamped sigmoid on normalized wait time, tuned by new steepness ---
    wait_norm = mad_normalize(ready_wait_time)
    # Smooth saturation using exp-based sigmoid with tunable steepness
    wait_sigmoid = 1.0 - np.exp(-PARAMS["wait_fairness_steepness"] * np.maximum(wait_norm, 0.0))
    wait_fairness = np.clip(wait_sigmoid, 0.0, 1.0)
    # Fade out under DDL pressure to preserve deadline feasibility
    wait_gated = wait_fairness * ddl_gate

    # --- Work density bonus: saturating contribution only when slack is safe ---
    work_density = np.divide(remaining_work, duration_total + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_bonus = (
        PARAMS["energy_efficiency_ratio_weight"] * 
        work_density_norm * 
        (np.abs(work_density_norm) <= PARAMS["energy_efficiency_ratio_weight"]) *
        (1.0 - ddl_gate)
    )

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute strongly
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        critical_release_score  # negative contribution: promotes release
    )
    
    # Non-DDL terms smoothly blended via ddl_gate (0→1 as slack improves)
    score += ddl_gate * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_norm +
        rank_score +
        energy_uncertainty_score +
        wait_gated +
        work_density_bonus
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    score = np.clip(score, finfo.min + eps, finfo.max - eps)
    
    return score
