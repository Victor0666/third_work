def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaces unstable inverse-headroom scaling with a bounded linear ramp (0→1 over [0, 2*threshold]) for wait-time urgency.
      - Introduces explicit two-stage DDL violation handling: first isolate all slack <= 0 tasks with raw penalty dominance, then apply nonlinear amplifier only to severely overdue ones.
      - Adds robust energy-uncertainty coupling activation gate: term only engages when both normalized energy and uncertainty exceed per-task MAD-thresholded levels, eliminating spurious signals in low-risk regimes.
      - Joint MAD normalization retained for coherent risk scaling, but now applied consistently across all normalized features using shared robust centering/scaling logic.
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no other constants used.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust joint MAD normalization function for consistent scale alignment
    def robust_mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        abs_dev = np.abs(x - med)
        mad = np.median(abs_dev) + eps
        # Normalize and clip to [-2, 2] for stability (allowed literals only)
        z = (x - med) / (mad + eps)
        return np.clip(z, -2.0, 2.0)

    # Compute joint risk features for coherent scaling
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    risk_stack = np.stack([abs_slack, uncertainty, duration_total], axis=0)
    risk_medians = np.median(risk_stack, axis=1, keepdims=True)
    risk_abs_devs = np.abs(risk_stack - risk_medians)
    risk_mads = np.median(risk_abs_devs, axis=1) + eps
    
    # Normalize each risk dimension using its joint MAD
    abs_slack_norm = robust_mad_normalize(abs_slack)
    unc_norm = robust_mad_normalize(uncertainty)
    dur_norm = robust_mad_normalize(duration_total)

    # --- Stage 1: Strict DDL-violation isolation (lexicographically highest priority) ---
    # All tasks with slack <= 0 receive dominant penalty before any normalization or interaction
    slack_violation_mask = (slack <= 0).astype(float)
    # Base linear penalty for any violation
    base_violation_penalty = (-slack) * PARAMS["slack_mad_scale"] * (1.0 + unc_norm)
    # Nonlinear amplifier only for severe violations (slightly more aggressive than linear)
    severe_violation = np.where(slack < -PARAMS["ddl_protection_threshold"], -slack, 0.0)
    severe_penalty = (severe_violation ** PARAMS["slack_violation_amplifier"]) * (1.0 + unc_norm * 0.5)
    slack_penalty = slack_violation_mask * (base_violation_penalty + severe_penalty)

    # --- Stage 2: Duration risk (always active, but secondary to DDL violation) ---
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Stage 3: Critical-path release (unconditional, preserves absolute magnitude) ---
    critical_release_score = upward_rank * remaining_work * PARAMS["critical_path_release_weight"]

    # --- Lexicographic gate for non-DDL terms: only when slack > ddl_protection_threshold ---
    slack_headroom_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Non-DDL terms (gated, only active under safety margin) ---
    # Energy normalization
    energy_norm = robust_mad_normalize(min_incremental_energy)
    energy_term = slack_headroom_mask * PARAMS["risk_adjusted_energy_weight"] * energy_norm

    # Uncertainty-gated energy interaction with activation threshold
    # Compute per-task MAD thresholds for energy and uncertainty
    energy_mad = np.median(np.abs(min_incremental_energy - np.median(min_incremental_energy))) + eps
    unc_mad = np.median(np.abs(uncertainty - np.median(uncertainty))) + eps
    energy_above_thresh = (min_incremental_energy > np.median(min_incremental_energy) + PARAMS["energy_uncertainty_activation_threshold"] * energy_mad).astype(float)
    unc_above_thresh = (uncertainty > np.median(uncertainty) + PARAMS["energy_uncertainty_activation_threshold"] * unc_mad).astype(float)
    joint_activation = energy_above_thresh * unc_above_thresh
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_uncertainty_term = (
        slack_headroom_mask *
        joint_activation *
        PARAMS["energy_uncertainty_interaction"] *
        energy_norm *
        unc_norm *
        unc_sigmoid
    )

    # Anti-starvation wait-time with bounded linear ramp instead of inverse headroom
    # Ramp: 0 at slack == ddl_protection_threshold, linearly increases to 1.0 at slack == 3 * ddl_protection_threshold
    ramp_end = 3.0 * PARAMS["ddl_protection_threshold"] + eps
    wait_ramp = np.clip(
        (slack - PARAMS["ddl_protection_threshold"]) / (ramp_end - PARAMS["ddl_protection_threshold"]),
        0.0, 1.0
    )
    wait_power = np.where(
        slack_headroom_mask > 0.0,
        (ready_wait_time ** PARAMS["wait_time_decay_exponent"]) * wait_ramp,
        0.0
    )
    wait_norm = robust_mad_normalize(wait_power)
    wait_term = slack_headroom_mask * wait_norm

    # Rank-slack balance with improved interpolation
    slack_headroom_ratio = np.clip(
        (slack - PARAMS["ddl_protection_threshold"]) / (2.0 * PARAMS["ddl_protection_threshold"] + eps),
        0.0, 1.0
    )
    rank_weight = PARAMS["rank_slack_balance"] * (1.0 - slack_headroom_ratio) + (1.0 - PARAMS["rank_slack_balance"]) * slack_headroom_ratio
    rank_norm = robust_mad_normalize(upward_rank)
    rank_term = slack_headroom_mask * (-rank_norm) * rank_weight

    # --- Assemble final score: smaller = higher priority ---
    # DDL-violation terms dominate lexicographically
    score = (
        slack_penalty +
        duration_risk +
        critical_release_score
    )
    
    # Non-DDL terms added only under safety margin
    score += (
        energy_term +
        energy_uncertainty_term +
        wait_term +
        rank_term
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
