def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL-risk gating, critical-path-aware slack balancing, 
    and starvation-robust efficiency scoring. Uses bounded piecewise interactions and 
    monotonic normalizations to avoid instability."""
    eps = PARAMS["epsilon"]
    
    # --- Robust normalization: use mean absolute + eps, never zero-variance collapse
    def robust_norm(x):
        x = np.asarray(x, dtype=float)
        denom = np.mean(np.abs(x)) + eps
        return x / denom
    
    # --- Slack processing: strong nonlinear penalty for negative slack, smooth transition near zero
    # Use signed power: negative slack → large positive penalty (bad), positive slack → small penalty
    slack_signed = np.copy(slack)
    slack_abs = np.abs(slack_signed)
    slack_sign = np.sign(slack_signed)  # -1 for late, +1 for early
    # Penalize lateness quadratically; reward earliness linearly but weakly
    slack_score = np.where(
        slack_signed < 0,
        -slack_sign * (slack_abs ** PARAMS["slack_penalty_exponent"]),
        slack_signed * 0.2  # mild reward for buffer
    )
    
    # --- Criticality boost: only activate when both upward_rank is high AND slack is tight (< 2*eps or negative)
    rank_median = np.median(upward_rank) if len(upward_rank) > 1 else np.mean(upward_rank)
    is_high_rank = (upward_rank >= rank_median)
    is_tight_slack = (slack_signed <= 2 * eps)
    critical_gate = np.where(is_high_rank & is_tight_slack, PARAMS["criticality_boost"], 1.0)
    
    # --- Energy efficiency ratio: joules per second (lower = more efficient)
    # Avoid div-by-zero: add eps to denominator
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = robust_norm(energy_per_sec)
    
    # --- Uncertainty-slack coupling: penalize high uncertainty * high deadline pressure
    # Deadline pressure = max(0, -slack) — only active when at risk
    deadline_pressure = np.maximum(0.0, -slack_signed)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    
    # --- Wait-time saturation: anti-starvation term, bounded by threshold
    wait_norm = np.clip(ready_wait_time, 0, PARAMS["wait_saturation_threshold"])
    wait_score = -robust_norm(wait_norm)  # negative → higher priority for older tasks
    
    # --- Duration risk penalty: amplify long-duration tasks under high uncertainty
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]
    
    # --- Rank-slack balance: interpolate between slack-driven and rank-driven priority
    # When slack is tight → lean toward rank; when slack is ample → lean toward slack
    slack_range = np.maximum(eps, np.percentile(slack, 90) - np.percentile(slack, 10))
    slack_normalized = np.clip((slack_signed - np.min(slack)) / (slack_range + eps), 0, 1)
    # Interpolate: weight_rank = rank_slack_balance when slack is tight (low slack_normalized)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * slack_normalized
    rank_score = -robust_norm(upward_rank) * weight_rank
    
    # --- Assemble final score: smaller = better
    # All components are normalized or bounded; no unbounded terms
    score = (
        robust_norm(slack_score)  # primary driver: lateness penalty dominates
        + robust_norm(unc_slack_coupling)
        + robust_norm(duration_risk)
        + PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score
        + rank_score
        + wait_score
        + robust_norm(min_incremental_energy) * (1.0 - weight_rank)  # fallback energy term
    )
    
    # Apply criticality boost multiplicatively only to core deadline-sensitive terms
    score = score * critical_gate
    
    # Final safeguard: ensure finite, replace NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
