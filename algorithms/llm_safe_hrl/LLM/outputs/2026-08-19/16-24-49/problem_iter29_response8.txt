def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's stability with novel starvation modeling:
      - Decoupled per-feature MAD normalization (robust, no cross-contamination)
      - Hard DDL-protection gate + smooth tanh coupling (strict feasibility-first)
      - Novel starvation urgency exponentiation: wait_ratio^p strengthens priority for extreme waiting cases
      - Critical path release decoupled from slack normalization to avoid dilution under deadline pressure
      - All gates and interactions bounded, differentiable, and numerically safeguarded
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # --- Decoupled per-feature MAD normalization (robust, no scale leakage) ---
    def mad_normalize(x):
        x_med = np.median(x)
        x_mad = np.mean(np.abs(x - x_med)) + eps
        return (x - x_med) / (x_mad * PARAMS["duration_mad_scale"] + eps)

    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    
    duration_norm = mad_normalize(duration_total)
    slack_norm = mad_normalize(abs_slack)
    unc_norm = mad_normalize(uncertainty)
    energy_norm = mad_normalize(min_incremental_energy)
    rank_norm = mad_normalize(upward_rank)
    work_norm = mad_normalize(remaining_work)
    wait_norm = mad_normalize(ready_wait_time)

    # --- DDL-critical base scores (lexicographically dominant, always active) ---
    # Smooth tanh-based slack-uncertainty coupling for violation penalty
    slack_sign = np.tanh(slack / (np.abs(slack) + eps))
    slack_penalty = np.where(
        slack < 0,
        (-slack) * (1.0 + PARAMS["uncertainty_slack_coupling"] * unc_norm * 
                   np.abs(slack_sign) ** PARAMS["normalized_slack_coupling_exponent"]),
        0.0
    )

    # Critical path release: use raw product (not normalized) for scale-invariant dominance
    # Prevents dilution when all ranks are similar; preserves absolute importance signal
    critical_release_raw = upward_rank * remaining_work
    critical_release_score = -critical_release_raw * PARAMS["critical_path_release_weight"]

    # --- Hard DDL-protection gate: disables *all* non-critical terms when slack < 0 ---
    ddl_safe_mask = np.where(slack >= 0.0, 1.0, 0.0)

    # --- Non-DDL terms: only active when ddl_safe_mask == 1.0 ---
    # Linear ramp slack gate centered at ddl_protection_threshold
    ramp_half = PARAMS["slack_linear_ramp_width"] / 2.0
    gate_center = PARAMS["ddl_protection_threshold"]
    gate_low = gate_center - ramp_half
    gate_high = gate_center + ramp_half
    slack_gate = np.clip((slack - gate_low) / (ramp_half * 2.0 + eps), 0.0, 1.0)
    safe_gate = ddl_safe_mask * slack_gate

    # Energy efficiency term: gated and normalized
    energy_score = safe_gate * energy_norm * PARAMS["risk_adjusted_energy_weight"]

    # Uncertainty-energy interaction: same gate and normalization
    energy_uncertainty_score = safe_gate * energy_norm * unc_norm * PARAMS["energy_uncertainty_interaction"]

    # Starvation mitigation: non-linear urgency boost using tunable exponent
    median_wait = np.median(ready_wait_time) + eps
    wait_ratio = np.clip(ready_wait_time / median_wait, 0.0, 2.0)
    # Apply exponent to strengthen priority for longest-waiting tasks without unbounded growth
    wait_urgency = wait_ratio ** PARAMS["starvation_urgency_exponent"]
    wait_score = ready_wait_time * (1.0 + np.clip(-slack / (PARAMS["ddl_protection_threshold"] + eps), 0.0, 2.0))
    wait_final = safe_gate * mad_normalize(wait_score) * PARAMS["wait_starvation_penalty"] * wait_urgency

    # Rank-slack balance: interpolate only in safe region, using raw rank for dominance
    slack_weight = np.clip(1.0 - slack_norm / (slack_norm.max() + eps), 0.0, 1.0)
    rank_weight = 1.0 - slack_weight
    # Use raw rank (not normalized) to preserve signal strength in low-slack scenarios
    rank_score = safe_gate * (-upward_rank / (np.median(upward_rank) + eps)) * rank_weight * PARAMS["rank_slack_balance"]

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute (dominant lexicographic layer)
    score = (
        slack_penalty +
        critical_release_score
    )
    
    # Non-DDL terms added only under hard+soft gate
    score += (
        energy_score +
        energy_uncertainty_score +
        wait_final +
        rank_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
