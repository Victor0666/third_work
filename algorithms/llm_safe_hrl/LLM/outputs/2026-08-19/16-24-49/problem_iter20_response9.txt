def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's lexicographic reliability with Parent 1's differentiable urgency:
       - Introduces convex hybrid DDL gate: alpha * sigmoid_urgency + (1-alpha) * hard_mask → preserves CMA-ES gradients while retaining feasibility robustness
       - Keeps joint MAD normalization over |slack|, uncertainty, duration_total for coherent risk alignment
       - Retains explicit starvation mitigation via wait/duration ratio, gated by hybrid DDL signal
       - Critical-path release now uses *both* slack_pressure (Parent 2) *and* sigmoid_urgency (Parent 1) for dual-mode criticality scaling
       - Energy-uncertainty coupling uses bounded sigmoid centered at uncertainty=1.0 (Parent 2) but scaled by hybrid gate
       - All numeric literals restricted to {-2,-1,0,1,2}; no other constants used
       - Final score prioritizes DDL violation penalty first, then critical release, then feasible terms
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Joint MAD normalization: robustly aligns |slack|, uncertainty, and duration_total scales
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack)
    all_risk_features = np.stack([abs_slack, uncertainty, duration_total], axis=0)
    mad = np.mean(np.abs(all_risk_features - np.median(all_risk_features, axis=1, keepdims=True)), axis=1)
    joint_mad = np.median(mad) + eps
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        return np.clip((x - np.median(x)) / (joint_mad * PARAMS["duration_mad_scale"] + eps), -2.0, 2.0)

    # --- Hybrid DDL gate: convex blend of hard threshold and soft sigmoid urgency ---
    hard_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)
    sigmoid_urgency = 1.0 / (1.0 + np.exp(PARAMS["slack_urgency_sigmoid_steepness"] * slack))
    hybrid_gate = (
        PARAMS["hybrid_ddl_gate_alpha"] * sigmoid_urgency +
        (1.0 - PARAMS["hybrid_ddl_gate_alpha"]) * hard_mask
    )
    # Invert for priority: smaller score = higher priority → use 1 - sigmoid_urgency for urgency
    ddl_urgency = 1.0 - sigmoid_urgency  # [0,1], 1 when slack <= 0, decays smoothly

    # --- DDL-critical base penalty: always active ---
    ddl_violation_penalty = np.maximum(0.0, PARAMS["ddl_protection_threshold"] - slack)

    # --- Critical-path release: scaled by *both* hard pressure and soft urgency ---
    critical_release_score = upward_rank * remaining_work
    # Slack pressure from Parent 2 (normalized to [0,1])
    slack_pressure = np.clip((PARAMS["ddl_protection_threshold"] - slack) / (PARAMS["ddl_protection_threshold"] + eps), 0.0, 1.0)
    # Combine with soft urgency for dual-mode sensitivity
    combined_critical_scale = (slack_pressure + ddl_urgency) / 2.0
    critical_score = -mad_normalize(critical_release_score) * combined_critical_scale * PARAMS["critical_path_release_weight"]

    # --- Feasibility mask for non-DDL terms: hybrid-gated ---
    feasible_mask = hybrid_gate

    # --- Risk-adjusted energy term: only active in feasible region ---
    energy_norm = mad_normalize(min_incremental_energy)
    energy_term = feasible_mask * PARAMS["risk_adjusted_energy_weight"] * energy_norm

    # --- Uncertainty-slack interaction: penalizes high uncertainty near deadline ---
    unc_slack_interaction = uncertainty * (PARAMS["ddl_protection_threshold"] - slack) * PARAMS["uncertainty_slack_interaction"]
    unc_slack_norm = mad_normalize(unc_slack_interaction)

    # --- Starvation mitigation: wait time normalized by duration (faster tasks get higher boost) ---
    wait_efficiency = np.where(duration_total > eps, ready_wait_time / duration_total, 0.0)
    wait_norm = mad_normalize(wait_efficiency)
    wait_term = feasible_mask * PARAMS["wait_starvation_penalty"] * wait_norm

    # --- Energy-uncertainty coupling: only active when feasible and uncertainty is moderate ---
    unc_sigmoid = 1.0 / (1.0 + np.exp(-2.0 * (1.0 - uncertainty)))  # bounded sigmoid centered at 1.0
    energy_uncertainty_term = feasible_mask * PARAMS["energy_uncertainty_coupling"] * energy_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    # DDL violation penalty dominates lexicographically
    score = (
        mad_normalize(ddl_violation_penalty) +
        critical_score +
        unc_slack_norm
    )
    
    # Add feasible-only terms multiplicatively gated by hybrid signal
    score += energy_term + wait_term + energy_uncertainty_term

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
