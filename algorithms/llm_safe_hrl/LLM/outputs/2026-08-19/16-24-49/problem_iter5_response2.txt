def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's stability and Parent 1's successor-aware leverage.
    
    Key structural improvements:
    - Retains Parent 2's fixed empirical slack bounds and bounded sigmoid uncertainty gate for robustness.
    - Integrates Parent 1's *conditional critical path leverage* gated by both slack>=0 AND uncertainty<=median_unc,
      but re-purposed as a *bonus term* (not multiplier) to avoid over-amplification while preserving DDL protection.
    - Replaces Parent 2's linear rank_slack_balance interpolation with *piecewise linear mapping*: 
      prioritizes rank only in the critical zone [slack_min_bound, 0], zero outside — sharpens deadline enforcement.
    - Uses robust mean-absolute normalization (with PARAMS["robustness_mad_factor"]) instead of simple mean-abs scaling
      for better outlier resistance and cross-seed stability.
    - Removes all wait-time terms (confirmed inactive) and replaces energy fallback with direct normalized energy
      weighted by slack pressure — cleaner and more interpretable.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs without in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)
    
    # Robust normalization: mean absolute deviation scaled by robustness_mad_factor
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        mu = np.mean(x)
        mad = np.mean(np.abs(x - mu))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - mu) / (scale + eps)

    # --- Slack scoring: power penalty for violation, zero reward for positive slack ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )

    # --- Duration risk: exec + comm, normalized and coupled with uncertainty ---
    duration_total = min_exec_time + min_comm_time + eps
    duration_norm = robust_normalize(duration_total)
    duration_risk = duration_norm * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Energy efficiency ratio: marginal energy per second, normalized ---
    energy_per_sec = min_incremental_energy / (duration_total + eps)
    energy_eff_score = robust_normalize(energy_per_sec) * PARAMS["energy_efficiency_ratio_weight"]

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = robust_normalize(uncertainty * deadline_pressure) * PARAMS["uncertainty_slack_coupling"]

    # --- Critical path leverage: bonus only when slack >= 0 AND uncertainty <= median_unc (DDL protection gate) ---
    median_unc = np.median(uncertainty)
    ddl_protection_gate = ((slack >= 0.0) & (uncertainty <= median_unc + eps)).astype(float)
    rank_norm = robust_normalize(upward_rank)
    # Bonus increases linearly with rank only in safe region; no penalty elsewhere
    critical_bonus = PARAMS["criticality_boost"] * rank_norm * ddl_protection_gate

    # --- Piecewise rank-slack balance: active only in critical slack zone [slack_min_bound, 0] ---
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = 0.0  # Upper bound of critical zone is zero slack (tightest feasible)
    in_critical_zone = (slack >= slack_lb) & (slack <= slack_ub)
    rank_weight = np.where(in_critical_zone, 
                          PARAMS["rank_slack_balance"] * (1.0 - (slack - slack_lb) / (slack_ub - slack_lb + eps)), 
                          0.0)
    rank_score = -robust_normalize(upward_rank) * rank_weight

    # --- Bounded sigmoid gate for energy-uncertainty interaction ---
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = robust_normalize(min_incremental_energy)
    unc_norm = robust_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Slack-pressure-weighted base energy term: fallback priority when slack is highly negative ---
    slack_pressure = np.maximum(0.0, -slack)
    energy_base_weight = np.exp(-slack_pressure * 0.1)  # Soft decay for extreme violations
    energy_base_score = robust_normalize(min_incremental_energy) * energy_base_weight

    # --- Assemble final score: smaller = higher priority ---
    # All beneficial terms (critical_bonus, rank_score) subtracted; penalties added
    score = (
        robust_normalize(slack_score) +
        duration_risk +
        energy_eff_score +
        unc_slack_coupling +
        energy_uncertainty_score +
        energy_base_score -
        critical_bonus -
        rank_score
    )

    # Final safeguard: ensure finite, shape-(N,) output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
