def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: deadline-risk-gated criticality + uncertainty-aware fairness.
    
    Key innovations:
    - Slack is transformed via piecewise smooth function: strong penalty for negative slack,
      exponential urgency for small positive slack (< 1s), linear decay beyond.
    - Critical path leverage is *conditional*: only applied when slack >= 0, avoiding over-prioritizing 
      doomed tasks.
    - Energy term is dampened under high deadline pressure (via slack-based gate).
    - Wait fairness uses relative waiting time scaled by remaining slack headroom to avoid biasing late tasks.
    - Uncertainty interacts multiplicatively with slack pressure (not raw slack) for bounded risk amplification.
    - All features use MAD-normalization (more robust than mean/std) with epsilon safeguards.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Robust normalization using Median Absolute Deviation (MAD) to resist outliers
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = mad if mad > eps else eps
        return (x - med) / scale
    
    # --- Slack transformation: piecewise-smooth risk-urgency curve ---
    # Negative slack → strong linear penalty; near-zero positive → exponential urgency; larger positive → linear decay
    slack_arr = np.asarray(slack, dtype=float)
    neg_mask = slack_arr < 0.0
    zeroish_mask = (slack_arr >= 0.0) & (slack_arr < 1.0)
    pos_mask = slack_arr >= 1.0
    
    slack_norm = np.zeros_like(slack_arr)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack_arr[neg_mask])
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack_arr[zeroish_mask]) - 1.0)
    slack_norm[pos_mask] = np.clip(slack_arr[pos_mask], 0.0, 10.0)  # cap large slack to avoid dominance
    
    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)
    
    # --- Energy: dampened under deadline pressure ---
    energy_norm = mad_normalize(min_incremental_energy)
    # Dampen energy weight when slack is highly negative (doomed) or extremely urgent
    slack_pressure = np.clip(-slack_arr, 0.0, np.inf)  # only pressure, not penalty
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - np.tanh(slack_pressure * 0.5))
    
    # --- Critical path: conditional leverage only if slack is feasible ---
    rank_norm = mad_normalize(upward_rank)
    critical_gate = (slack_arr >= 0.0).astype(float)  # 1 if slack OK, 0 otherwise
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate
    
    # --- Fairness: wait time scaled by slack headroom (avoid boosting late tasks) ---
    # Use max(1, slack+1) to prevent division-by-zero and give baseline fairness even when slack < 0
    wait_headroom = np.maximum(1.0, slack_arr + 1.0)
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * wait_norm / (wait_headroom + eps)
    
    # --- Uncertainty interaction: multiplicative with slack pressure (bounded) ---
    unc_norm = mad_normalize(uncertainty)
    # Cap pressure to avoid explosion: tanh ensures [0,1] scaling
    slack_pressure_bounded = np.tanh(slack_pressure * 0.3)
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded
    
    # --- Assemble final score: smaller = higher priority ---
    # Note: all components are additive; sign conventions ensure lower score = more urgent/better
    score = (
        slack_norm  # dominates: high penalty for negative slack, urgency for tight deadlines
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_boost  # subtract because higher rank is beneficial → lowers score
        + wait_score  # add fairness boost (smaller wait_score is better, but wait_score itself is positive boost)
        + uncertainty_amplifier
    )
    
    # Final safeguard: replace inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
