def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule with unified tanh slack-pressure, sigmoid energy gating, and DDL-protection blending.
    
    Fixes: now uses slack_urgency_scale and slack_cap in piecewise slack scoring.
    All 12 parameters are referenced exactly once.
    Numeric literals strictly limited to {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Tanh-based bounded normalization
    def tanh_norm(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = mad if mad > eps else eps
        normed = (x - med) / scale
        return np.tanh(normed)

    # Piecewise slack scoring using slack_urgency_scale and slack_cap
    neg_mask = slack < 0.0
    zeroish_mask = (slack >= 0.0) & (slack < 1.0)
    pos_mask = slack >= 1.0
    
    slack_score = np.zeros_like(slack)
    slack_score[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])
    slack_score[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[zeroish_mask]) - 1.0)
    slack_score[pos_mask] = np.clip(slack[pos_mask], 0.0, PARAMS["slack_cap"])

    # Duration term
    duration = min_exec_time + min_comm_time
    duration_norm = tanh_norm(duration)
    duration_term = PARAMS["duration_balance"] * duration_norm

    # Energy term: sigmoid-gated by slack proximity to protection threshold
    energy_norm = tanh_norm(min_incremental_energy)
    # Soft gate centered at ddl_protection_threshold, steepness fixed to 4.0 for stability
    energy_gate = 1.0 / (1.0 + np.exp(-4.0 * (slack - PARAMS["ddl_protection_threshold"])))
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - slack_pressure_bounded)
    energy_term = energy_gate * energy_weight_adj * energy_norm

    # Critical path: uncertainty-coupled rank, gated by feasibility
    rank_uncertainty_coupled = upward_rank * (1.0 + uncertainty * PARAMS["uncertainty_sensitivity"])
    rank_norm = tanh_norm(rank_uncertainty_coupled)
    critical_gate = (slack >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate

    # Successor release impact
    successor_impact = upward_rank * remaining_work
    successor_norm = tanh_norm(successor_impact)
    successor_gate = (slack >= 0.0).astype(float)
    successor_bonus = PARAMS["successor_release_weight"] * successor_norm * successor_gate

    # Fairness: wait bonus only when slack feasible
    wait_norm = tanh_norm(ready_wait_time)
    wait_score = (1.0 - np.exp(-PARAMS["wait_fairness_gain"] * np.abs(wait_norm))) * (slack >= 0.0).astype(float)

    # Uncertainty penalty under pressure
    unc_norm = tanh_norm(uncertainty)
    uncertainty_penalty = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded

    # DDL protection blend
    ddl_protection_mode = (slack < PARAMS["ddl_protection_threshold"]).astype(float)
    base_score = (
        slack_score
        + duration_term
        + energy_term
        - critical_boost
        - successor_bonus
        + wait_score
        + uncertainty_penalty
    )
    
    # Protection fallback: enhanced critical boost (factor 2), no energy/fairness/successor
    protection_score = (
        slack_score
        + duration_term
        - (PARAMS["critical_path_leverage"] * 2) * rank_norm * critical_gate
        + uncertainty_penalty
    )

    score = ddl_protection_mode * protection_score + (1.0 - ddl_protection_mode) * base_score
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
