def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robustness with Parent 1's structured risk decomposition:
      - Retains Parent 2's stable min-max normalization with empirical slack bounds and DDL-gated starvation.
      - Integrates Parent 1's bounded duration-risk penalty (linear, not power-law) to avoid over-penalization under tight slack.
      - Unifies critical path release signal using upward_rank * remaining_work (Parent 1) but normalizes via Parent 2's robust min-max.
      - Uses lexicographic gating: non-DDL terms only active when slack > wait_starvation_activation_threshold (strictly safe region).
      - All numeric literals restricted to {-2,-1,0,1,2}; fully deterministic, finite-output guaranteed.
    """
    eps = PARAMS["epsilon"]
    mm_eps = PARAMS["minmax_stability_eps"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust min-max normalization: bounded by task-set extrema + stability epsilon
    def minmax_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        x_min = np.min(x)
        x_max = np.max(x)
        denom = x_max - x_min + mm_eps
        normalized = (x - x_min) / denom
        return np.clip(normalized, 0.0, 1.0)

    # --- DDL-critical base scores (always active) ---
    # Slack violation penalty: power-law for strong lateness pressure
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    # Uncertainty-slack coupling: amplifies penalty for uncertain tasks near deadline
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    # Bounded duration-risk penalty: linear penalty on (exec+comm)*uncertainty under tight slack
    duration_total = min_exec_time + min_comm_time + eps
    duration_risk_base = duration_total * uncertainty
    duration_risk_penalty = np.where(
        slack <= PARAMS["wait_starvation_activation_threshold"],
        duration_risk_base * PARAMS["bounded_duration_risk_factor"],
        0.0
    )

    # --- Critical path release signal: upward_rank * remaining_work, robustly normalized ---
    critical_release = upward_rank * remaining_work
    critical_release_norm = minmax_normalize(critical_release)
    critical_release_score = -critical_release_norm * PARAMS["critical_path_release_weight"]

    # --- Lexicographic gate: non-DDL terms only active when slack > activation threshold ---
    slack_headroom_mask = np.where(slack > PARAMS["wait_starvation_activation_threshold"], 1.0, 0.0)

    # --- Host-load–aware energy scaling: synthetic load proxy = 1 + normalized uncertainty ---
    load_proxy = 1.0 + minmax_normalize(uncertainty)
    scaled_energy = min_incremental_energy * load_proxy

    # --- Non-DDL terms (only active when slack is sufficiently positive) ---
    # Energy efficiency: energy per unit duration
    energy_per_sec = scaled_energy / duration_total
    energy_eff_score = minmax_normalize(energy_per_sec)

    # Slack-scaled rank importance: interpolates between slack-driven and rank-driven
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -minmax_normalize(upward_rank) * weight_rank

    # Energy-uncertainty interaction with bounded sigmoid gate
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = minmax_normalize(scaled_energy)
    unc_norm = minmax_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- DDL-gated starvation mitigation: ONLY active when slack <= activation threshold ---
    wait_normalized = np.where(
        slack <= PARAMS["wait_starvation_activation_threshold"],
        ready_wait_time / (np.abs(slack) + 1.0),
        0.0
    )
    wait_score = minmax_normalize(wait_normalized)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        minmax_normalize(slack_score) +
        minmax_normalize(unc_slack_coupling) +
        minmax_normalize(duration_risk_penalty) +
        critical_release_score +
        wait_score
    )
    
    # Non-DDL terms only added when slack is sufficiently positive (lexicographic enforcement)
    score += slack_headroom_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
