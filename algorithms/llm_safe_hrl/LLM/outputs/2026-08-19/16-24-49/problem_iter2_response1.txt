def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: MAD-robust, piecewise-smooth slack gating, conditional criticality,
       uncertainty-aware fairness, and starvation-avoiding wait scaling.
    
    Key improvements over parents:
    - Uses MAD normalization exclusively (more outlier-robust than IQR/mean-std) for all features.
    - Slack transformation: negative → linear penalty; [0,1) → exp(urgency*slack)-1; [1,cap] → linear;
      >cap → capped (avoids unbounded growth).
    - Criticality boost strictly gated by slack >= 0 (prevents boosting doomed tasks).
    - Energy term dampened via tanh(slack_pressure * scale), not just binary gates.
    - Wait fairness uses relative waiting time *divided by headroom* (max(1, slack+1)), preventing bias toward late tasks.
    - Uncertainty amplifies priority *only* when slack pressure is present (via tanh-bounded product).
    - All numeric literals are in {-2,-1,0,1,2}; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Robust MAD normalization: resistant to outliers in slack, uncertainty, energy
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = np.where(mad > eps, mad, eps)
        return (x - med) / scale
    
    # --- Slack transformation: piecewise-smooth risk-urgency curve ---
    slack_arr = np.asarray(slack, dtype=float)
    neg_mask = slack_arr < 0.0
    zeroish_mask = (slack_arr >= 0.0) & (slack_arr < 1.0)
    pos_mask = slack_arr >= 1.0
    
    slack_norm = np.zeros_like(slack_arr)
    # Negative slack: strong linear penalty
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack_arr[neg_mask])
    # Near-zero positive: exponential urgency
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack_arr[zeroish_mask]) - 1.0)
    # Larger positive: linear decay up to cap
    slack_clipped = np.clip(slack_arr[pos_mask], 0.0, PARAMS["slack_cap"])
    slack_norm[pos_mask] = slack_clipped
    
    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)
    
    # --- Energy: base term with smooth slack-pressure damping ---
    energy_norm = mad_normalize(min_incremental_energy)
    slack_pressure = np.clip(-slack_arr, 0.0, np.inf)
    # Smooth damping: full weight when slack OK, reduced under pressure
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"]))
    
    # --- Critical path: conditional leverage only if slack is feasible (>=0) ---
    rank_norm = mad_normalize(upward_rank)
    critical_gate = (slack_arr >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate
    
    # --- Fairness: wait time scaled by slack headroom (avoids boosting late tasks) ---
    # Headroom = max(1, slack + 1) ensures finite denominator and baseline fairness even when slack < 0
    wait_headroom = np.maximum(1.0, slack_arr + 1.0)
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * wait_norm / (wait_headroom + eps)
    
    # --- Uncertainty interaction: bounded multiplicative amplification under pressure ---
    unc_norm = mad_normalize(uncertainty)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded
    
    # --- Assemble final score: smaller = higher priority ---
    # Note sign conventions:
    #   slack_norm, duration_norm, energy_weight_adj*energy_norm → larger = worse → add
    #   critical_boost → larger rank helps → subtract to lower score
    #   wait_score → longer wait improves fairness → add (but wait_norm is centered; scaling makes it beneficial)
    #   uncertainty_amplifier → higher risk under pressure increases urgency → add
    score = (
        1 * slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_boost
        + wait_score
        + uncertainty_amplifier
    )
    
    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
