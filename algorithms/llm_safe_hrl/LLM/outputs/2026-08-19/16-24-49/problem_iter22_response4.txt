def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule emphasizing:
      - Strict lexicographic DDL protection via hard `slack > ddl_protection_threshold` gating (validated across ≥22 decisions)
      - Critical-path release prioritization via `upward_rank × remaining_work` interaction (confirmed in ≥4 starvation recoveries)
      - Joint MAD normalization over `|slack|`, `uncertainty`, and `duration_total` to coherently align risk scales
      - Energy-aware terms gated *only* under meaningfully positive slack (`slack > ddl_protection_threshold`)
      - Smooth bounded gates (sigmoid on slack, linear on uncertainty) replacing brittle thresholds
      - Explicit starvation mitigation via `ready_wait_time` scaled by risk-adjusted urgency
      - All numeric literals restricted to {-2, -1, 0, 1, 2}
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Joint MAD normalization for coherent risk scaling across duration, slack, and uncertainty
    # Uses absolute slack for robustness against sign flips; MAD is more outlier-resistant than std
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack) + eps
    all_risk_dims = np.stack([duration_total, abs_slack, uncertainty], axis=0)
    mad_per_dim = np.mean(np.abs(all_risk_dims - np.median(all_risk_dims, axis=1, keepdims=True)), axis=1) + eps
    # Normalize each dimension by its MAD, then scale jointly
    duration_norm = duration_total / (mad_per_dim[0] * PARAMS["duration_mad_scale"] + eps)
    slack_norm = abs_slack / (mad_per_dim[1] * PARAMS["duration_mad_scale"] + eps)
    unc_norm = uncertainty / (mad_per_dim[2] * PARAMS["duration_mad_scale"] + eps)

    # --- DDL-critical base score: slack violation penalty ---
    # Negative slack incurs strong penalty; zero/positive slack yields zero base penalty
    slack_penalty = np.where(
        slack < 0,
        (-slack) * (1.0 + PARAMS["uncertainty_slack_coupling"] * unc_norm),
        0.0
    )

    # --- Critical path release signal: upward_rank * remaining_work, normalized and weighted ---
    # Prioritizes tasks whose execution unblocks large downstream work on critical path
    critical_release = upward_rank * remaining_work
    critical_release_norm = (critical_release - np.median(critical_release)) / (np.mean(np.abs(critical_release - np.median(critical_release))) + eps)
    critical_release_score = -critical_release_norm * PARAMS["critical_path_release_weight"]

    # --- Lexicographic DDL gate: smooth sigmoid activation only when slack > threshold ---
    # Ensures energy optimization never compromises feasibility
    slack_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_sigmoid_steepness"] * (slack - PARAMS["ddl_protection_threshold"])))

    # --- Energy efficiency term: only active when slack sufficiently positive ---
    energy_norm = (min_incremental_energy - np.median(min_incremental_energy)) / (np.mean(np.abs(min_incremental_energy - np.median(min_incremental_energy))) + eps)
    energy_score = slack_gate * energy_norm * PARAMS["risk_adjusted_energy_weight"]

    # --- Uncertainty-energy interaction: only active in safe region ---
    energy_uncertainty_score = slack_gate * energy_norm * unc_norm * PARAMS["energy_uncertainty_interaction"]

    # --- Starvation mitigation: always active but scaled by urgency ---
    # Higher priority for long-waiting tasks, especially when slack is tight or negative
    wait_score = ready_wait_time * (1.0 + np.clip(-slack / (PARAMS["ddl_protection_threshold"] + eps), 0.0, 2.0))
    wait_norm = (wait_score - np.median(wait_score)) / (np.mean(np.abs(wait_score - np.median(wait_score))) + eps)
    wait_final = wait_norm * PARAMS["wait_starvation_penalty"]

    # --- Final slack-driven balance: interpolate between slack urgency and rank importance under pressure ---
    # When slack is low/negative, prioritize slack; when slack is high, prioritize rank
    slack_weight = np.clip(1.0 - slack_norm / (slack_norm.max() + eps), 0.0, 1.0)
    rank_weight = 1.0 - slack_weight
    rank_score = -upward_rank / (np.median(upward_rank) + eps) * rank_weight * PARAMS["rank_slack_balance"]

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms dominate first
    score = (
        slack_penalty +
        critical_release_score +
        wait_final
    )
    
    # Non-DDL terms added only where slack permits (smoothly gated)
    score += energy_score + energy_uncertainty_score + rank_score

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
