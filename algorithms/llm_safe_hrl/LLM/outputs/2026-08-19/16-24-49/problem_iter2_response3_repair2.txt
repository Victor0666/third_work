def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with soft median-ratio gating and work-energy balance.
    
    Structural changes vs Parent 2:
    - Replaces hard criticality gate with smooth activation based on slack magnitude vs median.
    - Adds work-energy balance term: when slack > 0, favors high-work & low-energy tasks.
    - Uses robust MAE-based normalization with IQR fallback only if MAE ≈ 0 (avoids quantile fragility).
    - All numeric literals are in {-2,-1,0,1,2}; no hidden constants.
    - Smaller score = higher priority; output shape is (N,); finite and deterministic.
    """
    eps = PARAMS["epsilon"]
    
    # Cast and copy inputs safely
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust normalization: use MAE; fall back to IQR only if MAE is near zero
    def robust_norm(x):
        x_abs = np.abs(x)
        mae = np.mean(x_abs)
        # If MAE is tiny, use IQR to avoid instability
        if mae < eps:
            q1, q3 = np.quantile(x_abs, [PARAMS["iqr_q1"], PARAMS["iqr_q3"]], method='midpoint')
            denom = q3 - q1 + eps
        else:
            denom = mae + eps
        return x / denom
    
    # Normalize core features
    norm_slack = robust_norm(slack)
    norm_energy = robust_norm(min_incremental_energy)
    norm_duration = robust_norm(min_exec_time + min_comm_time)
    norm_rank = robust_norm(upward_rank)
    norm_work = robust_norm(remaining_work)
    norm_wait = robust_norm(ready_wait_time)
    norm_uncert = robust_norm(uncertainty)
    
    # --- Smooth slack risk penalty: sign-preserving power, safe at zero ---
    slack_power = np.where(
        np.abs(norm_slack) < eps,
        0.0,
        np.sign(norm_slack) * np.abs(norm_slack) ** PARAMS["slack_risk_exponent"]
    )
    
    # --- Soft criticality gate: activate smoothly when slack magnitude < threshold × median magnitude ---
    slack_magnitude = np.abs(norm_slack)
    median_mag = np.median(slack_magnitude) + eps
    gate_input = PARAMS["criticality_gate_threshold"] * median_mag - slack_magnitude
    gate_activation = np.clip(gate_input, 0.0, 1.0)  # smooth [0,1] ramp
    
    # --- Duration-uncertainty coupling: bounded product, capped at 1.0
    dur_uncert_coupling = np.clip(norm_duration * norm_uncert, 0.0, 1.0) * PARAMS["duration_uncertainty_coupling"]
    
    # --- Wait-time boost: exponential saturation with epsilon offset
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * (norm_wait + eps))
    
    # --- Uncertainty-slack interaction: only penalize under deadline stress
    slack_pressure_mask = np.where(norm_slack < 0, 1.0, 0.0)
    uncert_slack_penalty = norm_uncert * slack_pressure_mask * PARAMS["uncertainty_slack_interaction"]
    
    # --- Work-energy balance: when slack > 0, prefer high-work & low-energy tasks
    safe_slack_mask = np.where(norm_slack > 0, 1.0, 0.0)
    work_energy_term = safe_slack_mask * (norm_work - norm_energy)
    
    # --- Final score composition ---
    score = slack_power
    
    # Energy matters more when slack is safe — now enhanced by work-awareness
    score += safe_slack_mask * PARAMS["energy_sensitivity"] * norm_energy
    score -= work_energy_term  # Favor high-work/low-energy pairs
    
    # Critical path dominates under deadline pressure (smoothly gated)
    score -= gate_activation * PARAMS["rank_fallback_weight"] * norm_rank
    
    # Add duration–uncertainty coupling as cost
    score += dur_uncert_coupling
    
    # Anti-starvation: reduce score for long-waiting tasks
    score -= wait_boost * PARAMS["wait_boost_weight"]
    
    # Add uncertainty-driven penalty under deadline stress
    score += uncert_slack_penalty
    
    # Replace NaN and clip infinities using declared parameters
    score = np.nan_to_num(
        score,
        nan=PARAMS["nan_replacement"],
        posinf=PARAMS["posinf_clip"],
        neginf=PARAMS["neginf_clip"]
    )
    
    return score
