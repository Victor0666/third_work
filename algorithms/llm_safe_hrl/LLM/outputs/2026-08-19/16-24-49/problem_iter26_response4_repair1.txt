def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's superior bounded sigmoid and exclusive coupling
    with Parent 1's robust bounded duration risk and slack-modulated rank importance.
    
    Key structural improvements:
      - Bounded duration-risk penalty (Parent 1) replaces unbounded duration_risk_penalty (Parent 2)
        to prevent over-penalization under extreme uncertainty, active only when slack <= threshold.
      - Slack-modulated rank importance (Parent 1) replaces rank-slack interpolation (Parent 2),
        directly coupling upward_rank with slack distance for sharper DDL-aware critical-path modulation.
      - Retains Parent 2's bounded sigmoid slack penalty, exclusive tight-slack coupling, and normalized
        wait-headroom scaling — all validated for stability and performance.
      - All normalizations use clipped MAD to [-2,2] for bounded influence; no numeric literals beyond {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust MAD-based normalization: clip to [-2,2] for stability and bounded influence
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med)) + eps
        normalized = (x - med) / mad
        return np.clip(normalized, -2.0, 2.0)

    # Joint risk-feature base: |slack|, uncertainty, duration_total for coherent scaling
    duration_total = min_exec_time + min_comm_time + eps
    slack_abs_norm = mad_normalize(np.abs(slack))
    unc_norm = mad_normalize(uncertainty)
    dur_norm = mad_normalize(duration_total)

    # --- DDL-critical base scores (always active, lexicographically dominant) ---
    # Bounded sigmoid slack penalty: steep for negative slack, smooth & bounded for positive
    slack_sigmoid_input = -slack_abs_norm  # flip so negative slack → large positive input
    slack_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["slack_sigmoid_steepness"] * slack_sigmoid_input))
    slack_penalty = slack_sigmoid * PARAMS["slack_mad_scale"] * (1.0 + unc_norm)

    # Tight-slack uncertainty coupling: *only* activated when slack <= threshold (exclusive condition)
    tight_slack_mask = np.where(slack <= PARAMS["ddl_protection_threshold"], 1.0, 0.0)
    tight_slack_uncertainty_coupling = (
        tight_slack_mask *
        uncertainty *
        slack_abs_norm *
        PARAMS["tight_slack_uncertainty_coupling"]
    )
    
    # Bounded duration-risk penalty (Parent 1): linear, active only under tight slack
    duration_risk_base = (min_exec_time + min_comm_time) * uncertainty
    duration_risk_penalty = tight_slack_mask * duration_risk_base * PARAMS["bounded_duration_risk_factor"]

    # --- Critical-path release term: unconditional, validated for starvation recovery ---
    critical_release_score = upward_rank * remaining_work * PARAMS["critical_path_release_weight"]

    # --- Slack-modulated rank importance (Parent 1): replaces interpolation with direct coupling ---
    # Scales upward_rank by slack distance — higher weight when slack is tight (more urgent to release successors)
    slack_distance = np.clip(PARAMS["ddl_protection_threshold"] - slack, 0.0, np.inf)
    slack_rank_score = -upward_rank * (1.0 + slack_distance / (PARAMS["ddl_protection_threshold"] + eps)) * PARAMS["rank_slack_balance"]

    # --- Lexicographic gate: non-DDL terms only active when slack strictly exceeds threshold ---
    slack_headroom_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Non-DDL terms (only active when slack > ddl_protection_threshold) ---
    # Risk-adjusted energy term: normalized marginal energy
    energy_norm = mad_normalize(min_incremental_energy)
    energy_term = slack_headroom_mask * PARAMS["risk_adjusted_energy_weight"] * energy_norm

    # Wait-time anti-starvation: power-law decay scaled by *normalized headroom ratio*, not raw slack
    wait_power = np.where(
        slack_headroom_mask > 0.0,
        ready_wait_time ** PARAMS["wait_time_decay_exponent"],
        0.0
    )
    wait_norm = mad_normalize(wait_power)
    # Scaled by parameter-controlled headroom ratio to avoid over-amplification
    wait_term = slack_headroom_mask * wait_norm * PARAMS["normalized_wait_headroom_ratio"]

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute (lexicographically dominant)
    score = (
        mad_normalize(slack_penalty) +
        mad_normalize(tight_slack_uncertainty_coupling) +
        mad_normalize(duration_risk_penalty) +
        mad_normalize(critical_release_score) +
        mad_normalize(slack_rank_score)
    )
    
    # Non-DDL terms only added when slack exceeds threshold (strict lexicographic separation)
    score += (
        energy_term +
        wait_term
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
