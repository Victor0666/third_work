def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robustness from Parent 1 and expressiveness from Parent 2:
      - Joint MAD normalization over [abs(slack), uncertainty, duration_total] using fused weighted median absolute deviation
      - Critical-path release scaled by *nonlinear* slack pressure: (max(0, ddl_th - slack))^exponent → sharper penalty near threshold
      - Bounded sigmoid on uncertainty (centered at 1.0, slope=2) to smoothly suppress energy-aware terms under high risk
      - Starvation mitigation via wait/duration ratio, *only activated when feasible*, avoiding starvation without compromising DDL safety
      - Lexicographic DDL enforcement: violation penalty dominates; all other terms are gated or scaled by feasibility
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no hidden constants
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Compute joint risk features for MAD normalization
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack)
    
    # Stack risk signals and compute *weighted* joint MAD: fuse scales using parameterized fusion weight
    risk_signals = np.stack([abs_slack, uncertainty, duration_total], axis=0)
    risk_medians = np.median(risk_signals, axis=1, keepdims=True)
    abs_deviations = np.abs(risk_signals - risk_medians)
    
    # Weighted MAD: apply fusion weight to each signal's MAD before global median
    mad_per_signal = np.median(abs_deviations, axis=1)
    weighted_mads = mad_per_signal * np.array([
        PARAMS["joint_risk_mad_fusion"],  # weight for |slack|
        1.0,                               # weight for uncertainty (reference)
        2.0 - PARAMS["joint_risk_mad_fusion"]  # complementary weight for duration_total
    ])
    joint_mad = np.median(weighted_mads) + eps

    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        x_median = np.median(x)
        norm = (x - x_median) / (joint_mad * PARAMS["duration_mad_scale"] + eps)
        return np.clip(norm, -2.0, 2.0)

    # --- DDL-critical base penalty: lexicographically dominant ---
    # Violation penalty: zero unless slack < threshold; grows nonlinearly near boundary
    slack_deficit = np.maximum(0.0, PARAMS["ddl_protection_threshold"] - slack)
    ddl_violation_penalty = slack_deficit ** PARAMS["slack_pressure_exponent"]

    # --- Critical-path release: unconditional but *nonlinearly* scaled by slack pressure ---
    critical_release_score = upward_rank * remaining_work
    # Nonlinear slack pressure: steeper penalty near threshold, smooth decay beyond
    slack_pressure = np.clip(slack_deficit / (PARAMS["ddl_protection_threshold"] + eps), 0.0, 1.0)
    critical_score = -mad_normalize(critical_release_score) * (slack_pressure ** PARAMS["slack_pressure_exponent"]) * PARAMS["critical_path_release_weight"]

    # --- Feasibility mask: only enable non-critical terms when slack > threshold ---
    feasible_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Risk-adjusted energy term: only active in feasible region ---
    energy_norm = mad_normalize(min_incremental_energy)
    energy_term = feasible_mask * PARAMS["risk_adjusted_energy_weight"] * energy_norm

    # --- Uncertainty-slack interaction: penalizes high uncertainty near deadline ---
    unc_slack_interaction = uncertainty * slack_deficit * PARAMS["uncertainty_slack_interaction"]
    unc_slack_norm = mad_normalize(unc_slack_interaction)

    # --- Starvation mitigation: wait time normalized by duration (faster tasks get higher boost) ---
    wait_efficiency = np.where(duration_total > eps, ready_wait_time / duration_total, 0.0)
    wait_norm = mad_normalize(wait_efficiency)
    wait_term = feasible_mask * PARAMS["wait_starvation_penalty"] * wait_norm

    # --- Energy-uncertainty coupling: only active when feasible and uncertainty is moderate ---
    # Bounded sigmoid centered at uncertainty=1.0 with slope=2 → suppresses energy use when uncertainty > 1
    unc_sigmoid = 1.0 / (1.0 + np.exp(-2.0 * (1.0 - uncertainty)))
    energy_uncertainty_term = feasible_mask * PARAMS["energy_uncertainty_coupling"] * energy_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    # DDL violation dominates lexicographically
    score = (
        mad_normalize(ddl_violation_penalty) +
        critical_score +
        unc_slack_norm
    )
    
    # Add feasible-only terms
    score += energy_term + wait_term + energy_uncertainty_term

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
