def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with smooth sigmoidal gating and IQR-normalized critical-path interaction.
    
    Key self-evolution improvements:
    - Replaces discontinuous `slack >= threshold` gating with smooth sigmoid: 
      `1 / (1 + exp(-sharpness * (slack - threshold)))` — improves gradient stability for CMA-ES.
    - Introduces normalized critical-path slack ratio: `slack / (min_exec_time + min_comm_time + eps)`,
      capturing joint urgency-duration sensitivity; used in slack-aware fairness & uncertainty modulation.
    - Switches `upward_rank × remaining_work` normalization from MAD to robust IQR-based scaling,
      using tunable quantiles to avoid hardcoded 0.25/0.75 — satisfies numeric literal constraint.
    - Applies sigmoidal gate also to uncertainty amplification and fairness boost for consistent smoothness.
    - All nonlinearities are bounded, epsilon-guarded, and finite-output guaranteed.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust IQR normalization for critical-path features (tunable quantiles, no hardcoded literals)
    def iqr_normalize(x):
        x = np.asarray(x, dtype=float)
        q_low = np.quantile(x, PARAMS["iqr_quantile_low"])
        q_high = np.quantile(x, PARAMS["iqr_quantile_high"])
        iqr = q_high - q_low
        scale = PARAMS["robustness_mad_factor"] * (iqr if iqr > eps else eps)
        return (x - q_low) / scale

    # Robust MAD normalization for other features
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Normalized critical-path slack ratio: urgency per unit duration ---
    duration = min_exec_time + min_comm_time
    slack_ratio = slack / (duration + eps)
    slack_ratio_norm = mad_normalize(slack_ratio)

    # --- Unified slack pressure: linear risk + tanh urgency (no discontinuity at zero) ---
    slack_risk = PARAMS["slack_risk_penalty"] * np.clip(-slack, 0.0, np.inf)
    slack_urgency = np.tanh(PARAMS["slack_pressure_tanh_scale"] * np.maximum(0.0, slack))
    slack_norm = slack_risk + PARAMS["slack_urgency_scale"] * (1.0 - slack_urgency)

    # --- Duration: exec + comm, robustly normalized ---
    duration_norm = mad_normalize(duration)

    # --- Energy: conditional suppression only when slack > 0 AND uncertainty low ---
    energy_norm = mad_normalize(min_incremental_energy)
    median_unc = np.median(uncertainty)
    energy_suppression_gate = ((slack > 0.0) & (uncertainty <= median_unc + eps)).astype(float)
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - energy_suppression_gate)

    # --- Critical path & successor release: upward_rank × remaining_work, gated by smooth sigmoid ---
    rank_work_interaction = upward_rank * remaining_work
    rank_work_norm = iqr_normalize(rank_work_interaction)
    # Smooth sigmoid gate: replaces hard threshold → avoids optimization cliffs
    sigmoid_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_ratio_sigmoid_sharpness"] * (slack - PARAMS["successor_release_threshold"])))
    successor_boost = PARAMS["critical_path_leverage"] * rank_work_norm * sigmoid_gate

    # --- Fairness: saturating exponential wait boost, gated by slack ratio urgency ---
    wait_norm = mad_normalize(ready_wait_time)
    # Apply sigmoid urgency gate: higher boost when slack_ratio is small (tighter relative deadline)
    wait_urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_ratio_sigmoid_sharpness"] * (-slack_ratio_norm)))
    wait_score = (1.0 - np.exp(-PARAMS["wait_fairness_gain"] * np.maximum(0.0, wait_norm))) * wait_urgency_gate

    # --- Uncertainty interaction: additive amplification only under deadline pressure (negative slack),
    #     scaled by normalized slack ratio for tighter causal coupling ---
    unc_norm = mad_normalize(uncertainty)
    uncertainty_amplifier = (
        PARAMS["uncertainty_sensitivity"] 
        * unc_norm 
        * np.clip(-slack, 0.0, np.inf) 
        * (1.0 / (1.0 + np.exp(-PARAMS["slack_ratio_sigmoid_sharpness"] * slack_ratio_norm)))
    )

    # --- Assemble final score: smaller = higher priority ---
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - successor_boost
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
