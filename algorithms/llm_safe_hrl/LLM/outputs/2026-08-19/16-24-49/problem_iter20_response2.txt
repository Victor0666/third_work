def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule synthesizing best practices from both parents:
      - Joint MAD normalization over |slack|, uncertainty, duration_total for robust cross-signal alignment
      - Hard lexicographic DDL protection via `slack > ddl_protection_threshold` (not > 0)
      - Unconditional critical-path release term: upward_rank * remaining_work * weight
      - Dual-gated non-DDL optimization: requires BOTH sufficient slack AND host capacity margin
      - Replaces fragile sigmoid/percentile gates with bounded linear interactions and MAD clipping
      - Eliminates wait-time decay (redundant under critical-path + DDL pressure) and anti-starvation normalization (handled by critical release + DDL gate)
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no other constants used
      - Final score structure: [DDL-violation] + [critical-release] + [gated-efficiency-terms]
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Joint MAD normalization basis: robust scale alignment across risk dimensions
    duration_total = min_exec_time + min_comm_time + eps
    abs_slack = np.abs(slack)
    all_risk = np.concatenate([abs_slack, uncertainty, duration_total])
    risk_median = np.median(all_risk)
    mad = np.median(np.abs(all_risk - risk_median)) + eps
    joint_scale = PARAMS["joint_mad_scale_factor"] * mad

    def joint_mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        centered = x - risk_median
        normalized = centered / joint_scale
        return np.clip(normalized, -2.0, 2.0)

    # --- DDL-critical base terms (always active) ---
    # Slack violation: nonlinear penalty for lateness risk
    slack_violation = np.where(slack < 0, (-slack) ** PARAMS["slack_penalty_exponent"], 0.0)
    # Uncertainty-slack coupling: linear, stable, monotonic
    unc_slack_coupling = PARAMS["uncertainty_slack_coupling"] * uncertainty * np.maximum(0.0, -slack)
    # Duration risk: penalize high-duration-high-uncertainty under deadline pressure
    duration_risk = PARAMS["duration_risk_penalty"] * duration_total * uncertainty * np.where(slack <= PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Critical-path release (unconditional, highest structural priority) ---
    # Directly rewards unlocking large descendant work on high-rank paths
    critical_release = PARAMS["upward_rank_remaining_work_interaction"] * upward_rank * remaining_work

    # --- Dual gating for non-DDL optimization terms ---
    # Gate 1: DDL safety
    ddl_safe_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)
    # Gate 2: Host capacity margin (avoids energy optimization when resource is saturated)
    duration_median = np.median(duration_total) if N > 1 else duration_total[0]
    host_margin_mask = np.where(duration_total <= PARAMS["host_load_margin_gate"] * duration_median, 1.0, 0.0)
    optimization_mask = ddl_safe_mask * host_margin_mask

    # --- Non-DDL optimization terms (only active under dual gate) ---
    # Energy efficiency: marginal energy per second — lower is better
    energy_per_sec = min_incremental_energy / (duration_total + eps)
    energy_eff_score = joint_mad_normalize(energy_per_sec) * PARAMS["energy_efficiency_ratio_weight"]

    # Energy-uncertainty interaction: captures risk-aware energy tradeoff
    energy_norm = joint_mad_normalize(min_incremental_energy)
    unc_norm = joint_mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm

    # --- Assemble final score: smaller = higher priority ---
    # DDL-violation penalties (always contribute)
    score = (
        joint_mad_normalize(slack_violation) +
        joint_mad_normalize(unc_slack_coupling) +
        joint_mad_normalize(duration_risk)
    )
    # Critical release: always contributes, negative because higher importance → lower priority score
    score += -joint_mad_normalize(critical_release)
    # Non-DDL terms: only added when both DDL-safe AND host-capacity-sufficient
    score += optimization_mask * (energy_eff_score + energy_uncertainty_score)

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
