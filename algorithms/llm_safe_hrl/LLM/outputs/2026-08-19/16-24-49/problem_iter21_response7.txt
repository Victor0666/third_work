def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with successor-release urgency and slack-aware feature scaling:
      - Replaces joint-risk MAD with task-level slack-aware scaling: preserves sharp deadline signals without over-smoothing.
      - Introduces `successor_release_factor`-weighted successor-release term: `upward_rank * (1 + remaining_work/(slack+eps))`
        to amplify critical-path urgency under tight deadlines — directly addressing reflection's concern.
      - Removes `joint_risk_mad_scale` and associated joint normalization (structural redundancy removed).
      - All DDL-critical terms remain unconditional and lexicographically dominant.
      - Non-DDL terms strictly gated by `ddl_safe_mask`, with host-load-aware scaling via linear slack headroom.
      - Uses per-feature median absolute deviation (MAD) only where needed — avoids cross-dimension coupling that weakens DDL safety.
      - No numeric literals beyond {-2,-1,0,1,2}; all tunables declared and used exactly once.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Per-feature robust normalization (MAD) — stable for N=1, avoids joint coupling
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        abs_dev = np.abs(x - med)
        mad = np.median(abs_dev) + eps
        normalized = (x - med) / mad
        return np.clip(normalized, -2.0, 2.0)

    # --- DDL-critical base scores (always active, lexicographically dominant) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Unconditional critical-path release term: enhanced with successor-release urgency under tight slack ---
    # New structural change: amplifies upward_rank when slack is small → promotes early release of critical descendants
    slack_safe = np.where(slack > 0, slack, eps)
    successor_release_gain = 1.0 + (remaining_work / slack_safe)
    critical_path_release = upward_rank * successor_release_gain * PARAMS["successor_release_factor"]
    # Keep original critical-path term as base (unamplified) for stability
    critical_path_base = upward_rank * remaining_work * PARAMS["upward_rank_remaining_work_interaction"]
    critical_path_total = critical_path_base + critical_path_release

    # --- Lexicographic DDL protection gate: non-DDL terms only active if slack > ddl_protection_threshold ---
    ddl_safe_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Non-DDL terms (only active when slack exceeds protection threshold) ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = mad_normalize(energy_per_sec)

    # Linear slack headroom scaling: [ddl_protection_threshold, max_slack] → [0,1]
    slack_headroom = np.maximum(0.0, slack - PARAMS["ddl_protection_threshold"])
    max_slack_headroom = np.max(slack_headroom) + eps
    slack_headroom_normalized = slack_headroom / max_slack_headroom

    # Host-load-aware scaling: activates energy terms more strongly when headroom is ample
    host_load_scale = slack_headroom_normalized * PARAMS["host_load_conditional_gate"]

    # Rank score: MAD-normalized upward_rank, weighted by headroom — preserves urgency for high-rank tasks
    rank_score = -mad_normalize(upward_rank) * (1.0 + slack_headroom_normalized)

    # Energy-uncertainty interaction: sigmoid-gated, using normalized components
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (mad_normalize(uncertainty) - 0.0)))
    energy_uncertainty_score = ddl_safe_mask * host_load_scale * PARAMS["energy_uncertainty_interaction"] * energy_eff_score * mad_normalize(uncertainty) * unc_sigmoid

    # Wait-time anti-starvation: linearly scaled by headroom, using normalized wait time
    wait_score = ddl_safe_mask * mad_normalize(ready_wait_time) * slack_headroom_normalized * PARAMS["wait_starvation_suppression"]

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute (lexicographically dominant)
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        mad_normalize(critical_path_total)
    )
    
    # Non-DDL terms only added when slack exceeds protection threshold, with host-load-aware energy scaling
    score += ddl_safe_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * host_load_scale * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
