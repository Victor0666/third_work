def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule integrating critical-path-and-workload coupling and host-load-aware energy dampening.
    
    Key structural improvements:
    - Replaces isolated upward_rank leverage with *slack-gated critical_work_leverage*: 
      upward_rank × remaining_work interaction — directly models 'critical path or successor blocked' diagnostic.
      This couples importance and computational scale, improving DDL safety for high-impact successors.
    - Introduces *load-aware energy dampening*: uses ready_wait_time as a proxy for VM queue pressure to further suppress 
      marginal energy when local congestion is high — aligning with 'avoidable_marginal_energy_or_load_cost' evidence.
    - Restores hard (slack >= 0) gating for work_density_bonus per reflection: avoids soft transitions harming DDL feasibility.
    - All normalizations remain MAD-based and epsilon-guarded; no unbounded literals beyond {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs: replace NaN/inf with safe values, ensure finite dtype
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization: consistent scaling across all features
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack transformation: piecewise-smooth risk-urgency curve ---
    neg_mask = slack < 0.0
    zeroish_mask = (slack >= 0.0) & (slack < 1.0)
    pos_mask = slack >= 1.0
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[zeroish_mask]) - 1.0)
    slack_norm[pos_mask] = np.clip(slack[pos_mask], 0.0, PARAMS["slack_cap"])

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: dampened under deadline pressure AND host load (queue pressure proxy) ---
    energy_norm = mad_normalize(min_incremental_energy)
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    # Load-aware dampening: longer wait → higher queue pressure → stronger energy suppression
    load_pressure = np.clip(ready_wait_time, 0.0, np.inf)
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (
        1.0 - np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"]) *
        np.tanh(load_pressure * PARAMS["load_aware_energy_dampen"])
    )

    # --- Critical-work leverage: slack-feasibility-gated upward_rank × remaining_work interaction ---
    # Captures 'critical path or successor blocked' diagnostic: high rank + large descendant work = high DDL risk if delayed
    rank_work_product = upward_rank * remaining_work
    rank_work_norm = mad_normalize(rank_work_product)
    critical_work_gate = (slack >= 0.0).astype(float)
    critical_work_boost = PARAMS["critical_work_leverage"] * rank_work_norm * critical_work_gate

    # --- Work density: remaining_work per unit duration, hard-gated by slack feasibility ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_gate = (slack >= 0.0).astype(float)  # Hard gate per reflection
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * work_density_gate

    # --- Fairness: wait time scaled by slack headroom ---
    wait_headroom = np.maximum(1.0, slack + 1.0)
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * wait_norm / (wait_headroom + eps)

    # --- Uncertainty interaction: multiplicative with slack pressure (bounded) ---
    unc_norm = mad_normalize(uncertainty)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded

    # --- Assemble final score: smaller = higher priority ---
    # Note: critical_work_boost and work_density_bonus are subtracted because higher values indicate higher benefit
    score = (
        slack_norm  # dominates: high penalty for negative slack, urgency for tight deadlines
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_work_boost
        - work_density_bonus
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
