def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing risk-aware slack gating, adaptive criticality weighting,
    and bounded uncertainty-duration coupling. Prioritizes tasks threatening deadline first,
    then balances energy vs. path importance conditionally. Uses exponential wait boost with decay
    to prevent starvation without overshooting. All operations are epsilon-stabilized and finite."""
    eps = PARAMS["epsilon"]
    
    # Safely cast and copy inputs
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust normalization: use mean absolute value + epsilon; avoid std on small N or zeros
    def robust_norm(x):
        x_abs = np.abs(x)
        denom = np.mean(x_abs) + eps
        return x / denom
    
    # Normalize core features
    norm_slack = robust_norm(slack)
    norm_energy = robust_norm(min_incremental_energy)
    norm_duration = robust_norm(min_exec_time + min_comm_time)
    norm_rank = robust_norm(upward_rank)
    norm_work = robust_norm(remaining_work)
    norm_wait = robust_norm(ready_wait_time)
    norm_uncert = robust_norm(uncertainty)
    
    # --- Risk-aware slack penalty: stronger nonlinearity for negative slack ---
    # Use signed power: preserves sign, penalizes negativity quadratically/cubically
    slack_power = np.sign(norm_slack) * np.abs(norm_slack) ** PARAMS["slack_risk_exponent"]
    
    # --- Adaptive criticality gate: only amplify rank priority when slack is critically low ---
    # Gate: [0,1] weight that activates upward_rank dominance under deadline pressure
    slack_normalized_for_gate = (norm_slack - PARAMS["criticality_gate_threshold"]) / (1.0 + eps)
    gate_activation = np.clip(-slack_normalized_for_gate, 0.0, 1.0)  # 1 when norm_slack << threshold
    
    # --- Duration-uncertainty coupling: amplifies duration impact when uncertainty is high ---
    # Bounded product, capped at 1.0 to prevent explosion
    dur_uncert_coupling = np.clip(norm_duration * norm_uncert, 0.0, 1.0) * PARAMS["duration_uncertainty_coupling"]
    
    # --- Wait-time boost: exponential decay ensures early urgency but avoids runaway priority ---
    # Scaled by normalized wait time and decays smoothly: exp(-decay * wait) → 0 as wait grows large
    # But we want *higher* priority for longer wait → use (1 - exp(-...)) for saturation boost
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * (norm_wait + eps))
    
    # --- Uncertainty-slack interaction: extra penalty only when both slack < 0 AND uncertainty > 0 ---
    # Use soft sign: negative slack → positive penalty term
    slack_pressure_mask = np.where(norm_slack < 0, 1.0, 0.0)
    uncert_slack_penalty = norm_uncert * slack_pressure_mask * PARAMS["uncertainty_slack_interaction"]
    
    # --- Final score composition ---
    # Primary driver: slack risk (negative = high priority → low score)
    score = slack_power
    
    # Energy matters more when slack is safe (positive), less when critical
    safe_slack_mask = np.where(norm_slack > 0, 1.0, 0.0)
    score += safe_slack_mask * PARAMS["energy_sensitivity"] * norm_energy
    
    # Critical path dominates under deadline pressure
    score -= gate_activation * PARAMS["rank_fallback_weight"] * norm_rank
    
    # Add duration-uncertainty coupling as cost (longer + uncertain → higher score → lower priority)
    score += dur_uncert_coupling
    
    # Anti-starvation: boost priority for long-waiting tasks (reduces score)
    score -= wait_boost * 0.5  # moderate strength
    
    # Add uncertainty-driven penalty under deadline stress
    score += uncert_slack_penalty
    
    # Ensure all values are finite and replace NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=1e6, neginf=-1e6)
    
    return score
