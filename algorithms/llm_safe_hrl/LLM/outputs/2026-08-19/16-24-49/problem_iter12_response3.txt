def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with smooth slack penalty, tanh-based uncertainty gating, and MAD-normalization.
    
    Key structural improvements:
    - Replaces piecewise slack scoring with continuous `max(0,-slack)**p` — eliminates discontinuities and instability.
    - Replaces IQR normalization with robust MAD (mean absolute deviation) to avoid outlier sensitivity of median/IQR.
    - Replaces sigmoid uncertainty gates with bounded `tanh(steepness * uncertainty)` for smoother, monotonic modulation.
    - Applies uncertainty gating *only* to energy/duration terms (not slack_score), decoupling deadline urgency from risk scaling.
    - All inputs sanitized with `np.nan_to_num`; no in-place mutation; strict shape-(N,) output.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs without mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)
    
    # Robust MAD normalization: less sensitive to outliers than IQR or std
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.mean(x)
        abs_dev = np.abs(x - center)
        scale = np.mean(abs_dev) + eps
        return (x - center) / (scale + eps)

    # --- Smooth slack penalty: continuous, monotonic, no piecewise boundaries ---
    slack_penalty = np.maximum(0.0, -slack) ** PARAMS["slack_penalty_exponent"]

    # --- Duration risk: exec+comm normalized, then modulated by tanh-bounded uncertainty ---
    duration_total = min_exec_time + min_comm_time + eps
    duration_norm = mad_normalize(duration_total)
    unc_gate = np.tanh(PARAMS["uncertainty_tanh_steepness"] * uncertainty)
    duration_risk = duration_norm * (1.0 + unc_gate * PARAMS["duration_risk_penalty"])

    # --- Energy efficiency ratio: marginal energy per second, normalized and gated ---
    energy_per_sec = min_incremental_energy / (duration_total + eps)
    energy_eff_norm = mad_normalize(energy_per_sec)
    energy_eff_score = energy_eff_norm * PARAMS["energy_efficiency_ratio_weight"] * (1.0 + unc_gate * 0.5)

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = mad_normalize(uncertainty * deadline_pressure) * PARAMS["uncertainty_slack_coupling"]

    # --- Slack-aware normalized criticality boost: upward_rank scaled by clipped negative slack ---
    rank_norm = mad_normalize(upward_rank)
    clipped_slack = np.maximum(0.0, -slack)
    critical_bonus = rank_norm * (1.0 + PARAMS["slack_aware_criticality_scale"] * clipped_slack)

    # --- Energy-uncertainty interaction: tanh-gated multiplicative coupling ---
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_gate

    # --- Base energy fallback: direct normalized energy, also gated ---
    energy_base_score = mad_normalize(min_incremental_energy) * (1.0 + unc_gate * 0.3)

    # --- Fairness: tanh-scaled wait boost (bounded [0,1]) using declared tanh_bias ---
    wait_norm = mad_normalize(ready_wait_time)
    wait_boost = PARAMS["tanh_bias"] * (1.0 + np.tanh(PARAMS["wait_fairness_gain"] * wait_norm))

    # --- Work-density bonus: remaining_work / (quantile-based duration + eps) ---
    duration_q = np.quantile(duration_total, PARAMS["robust_quantile"]) + eps
    work_density = remaining_work / duration_q
    work_density_norm = mad_normalize(work_density)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm

    # --- Assemble final score: smaller = higher priority ---
    # Slack dominates; all penalties added, bonuses subtracted
    score = (
        mad_normalize(slack_penalty) +
        duration_risk +
        energy_eff_score +
        unc_slack_coupling +
        energy_uncertainty_score +
        energy_base_score +
        wait_boost -
        critical_bonus -
        work_density_bonus
    )

    # Final safeguard
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
