def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
       - Smooth sigmoidal slack urgency (replacing both hard DDL gate AND slack_score's piecewise branch) 
         for differentiable, full-range deadline pressure — improves CMA-ES convergence and avoids feasibility cliffs.
       - Unified urgency signal: `1 / (1 + exp(-k * slack))` maps slack ∈ ℝ → (0,1), inverted for priority (smaller = tighter).
       - Retains bias-stabilized critical_path_release (monotonic across all slack) and drops wait-time entirely.
       - All DDL-critical terms now share a common differentiable foundation — no discontinuous branches.
       - MAD-normalization remains for robustness; N=1 fallback preserved.
       - No numeric literals outside {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Unified MAD-based global feature scaling (robust to outliers, stable for N=1)
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        abs_dev = np.abs(x - med)
        mad = np.median(abs_dev) + eps
        normalized = (x - med) / mad
        # Clip to [-2, 2] for numerical stability (allowed literals)
        return np.clip(normalized, -2.0, 2.0)

    # --- Smooth, differentiable slack urgency: replaces piecewise slack_score AND ddl_safe_mask ---
    # Maps slack → [0,1] where 0 = urgent (tight/late), 1 = relaxed; inverted for priority (smaller = better)
    # Uses sigmoid: 1 - sigmoid(slack) = 1 / (1 + exp(slack * k)) → monotonic decreasing in slack
    slack_urgency = 1.0 / (1.0 + np.exp(PARAMS["slack_urgency_sigmoid_steepness"] * slack))

    # --- DDL-critical base scores (all now differentiable & active for all tasks) ---
    # Replaces previous piecewise `slack_score`: now smooth, bounded, and gradient-friendly
    # Also replaces `ddl_safe_mask`: urgency naturally decays as slack increases → no brittle threshold
    deadline_pressure = slack_urgency  # Already [0,1], high when slack ≤ 0, low when slack ≫ 0
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Unconditional critical-path release term with bias-stabilized denominator ---
    critical_release_denom = np.maximum(0.0, -slack) + PARAMS["critical_release_denom_bias"]
    critical_path_release = (upward_rank * remaining_work) / critical_release_denom * PARAMS["upward_rank_remaining_work_interaction"]

    # --- Non-DDL terms: now gated smoothly by slack_urgency instead of hard threshold ---
    # High urgency → suppress non-DDL terms; low urgency → fully activate them
    ddl_safe_scale = 1.0 - slack_urgency  # 0 when urgent, 1 when relaxed

    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = mad_normalize(energy_per_sec)

    # Host-load-aware scaling: now proportional to relaxation level
    host_load_scale = ddl_safe_scale * PARAMS["host_load_conditional_gate"]

    # Slack-conditioned rank boost: same logic, now smoothly activated
    rank_boost_weight = slack_urgency  # 1.0 when urgent → prioritize rank; 0.0 when relaxed
    rank_score = -mad_normalize(upward_rank) * rank_boost_weight

    # Energy-uncertainty interaction: smoothly activated under relaxation
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = host_load_scale * PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    # All terms now differentiable and active for all tasks
    score = (
        mad_normalize(slack_urgency) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        mad_normalize(critical_path_release)
    )
    
    # Non-DDL terms smoothly scaled by relaxation level (not hard gate)
    score += ddl_safe_scale * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
