def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: smooth sigmoid slack gating, relative-MAD normalization,
       and fairness-gated energy dampening.
    
    Key evolutions:
    - Replaced piecewise slack transformation with *smooth, differentiable sigmoid*:
      urgency = 1 - exp(-k * max(0, slack)) for positive slack, preserving monotonicity
      and enabling gradient-aware optimization — eliminates discontinuities at slack=0/1.
    - Introduced *relative MAD normalization*: for slack and uncertainty, normalize w.r.t.
      their own distribution *and* scale by median(|slack|+ε) to preserve risk ordering
      under outlier-heavy workloads (e.g., one task with extreme slack dominates ranking).
    - Energy dampening now gated by *fairness headroom*: uses sigmoid of
      (ready_wait_time / (|slack| + ε)), not raw slack pressure — prioritizes energy
      efficiency only when waiting time is low *relative* to deadline margin, avoiding
      starvation while preserving energy savings where feasible.
    - All numeric literals are strictly in {-2,-1,0,1,2}; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Relative-MAD normalization: robust + preserves ordinal risk structure
    def relative_mad_normalize(x, ref_scale=None):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        # Use ref_scale (e.g., median |slack|) if provided for relative scaling
        scale = np.where(mad > eps, mad, eps)
        if ref_scale is not None:
            scale = np.maximum(scale, np.abs(ref_scale) + eps)
        return (x - med) / scale
    
    # --- Smooth slack transformation: differentiable sigmoid urgency + linear penalty ---
    slack_arr = np.asarray(slack, dtype=float)
    neg_mask = slack_arr < 0.0
    pos_mask = slack_arr >= 0.0
    
    slack_norm = np.zeros_like(slack_arr)
    # Negative slack: linear penalty (unchanged for interpretability & stability)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack_arr[neg_mask])
    # Positive slack: smooth sigmoid urgency → 1 - exp(-k * slack), bounded in [0,1)
    # Then scale by slack_urgency_scale and cap at slack_cap for large values
    urgency_pos = 1.0 - np.exp(-PARAMS["slack_sigmoid_k"] * slack_arr[pos_mask])
    urgency_scaled = PARAMS["slack_urgency_scale"] * urgency_pos
    # Apply linear decay cap beyond threshold for numerical stability
    slack_clipped = np.clip(slack_arr[pos_mask], 0.0, PARAMS["slack_cap"])
    linear_decay = slack_clipped / (PARAMS["slack_cap"] + eps)  # [0,1]
    slack_norm[pos_mask] = urgency_scaled * linear_decay + (1.0 - linear_decay) * slack_clipped
    
    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = relative_mad_normalize(duration)
    
    # --- Energy: fairness-gated dampening (not slack-pressure-gated) ---
    energy_norm = relative_mad_normalize(min_incremental_energy)
    # Fairness headroom: ratio of wait time to absolute slack margin — low ratio means safe to optimize energy
    wait_headroom_ratio = ready_wait_time / (np.abs(slack_arr) + eps)
    # Sigmoid gate: high energy weight when headroom is abundant (ratio ≈ 0); low when starved
    fairness_gate = 1.0 - (1.0 / (1.0 + np.exp(-PARAMS["energy_fairness_gate_k"] * (1.0 - wait_headroom_ratio))))
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * fairness_gate
    
    # --- Critical path: conditional leverage only if slack is feasible (>=0) ---
    rank_norm = relative_mad_normalize(upward_rank)
    critical_gate = (slack_arr >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate
    
    # --- Fairness: relative wait score — uses relative MAD and headroom-scaled denominator ---
    wait_norm = relative_mad_normalize(ready_wait_time)
    # Headroom = max(1, |slack| + 1) ensures finite, stable scaling even under extreme lateness
    wait_headroom = np.maximum(1.0, np.abs(slack_arr) + 1.0)
    wait_score = PARAMS["wait_fairness_gain"] * wait_norm / (wait_headroom + eps)
    
    # --- Uncertainty interaction: bounded multiplicative amplification under pressure ---
    unc_norm = relative_mad_normalize(uncertainty, ref_scale=np.median(np.abs(slack_arr) + eps))
    slack_pressure = np.clip(-slack_arr, 0.0, np.inf)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded
    
    # --- Assemble final score: smaller = higher priority ---
    score = (
        1 * slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_boost
        + wait_score
        + uncertainty_amplifier
    )
    
    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
