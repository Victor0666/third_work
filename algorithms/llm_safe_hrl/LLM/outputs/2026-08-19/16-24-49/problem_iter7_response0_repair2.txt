def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating counterfactual evidence:
    - Adds conditional DDL protection gate (slack >= 0 AND uncertainty <= median_uncertainty)
    - Replaces linear wait fairness with saturating exponential: 1 - exp(-gain * norm_wait)
    - Introduces successor-release interaction: upward_rank * remaining_work, gated by slack feasibility
    - Uses fairness headroom (wait_time / (|slack| + eps)) instead of raw slack to gate energy savings
    - Robustly normalizes all features using MAD scaled by robustness_mad_factor
    - All operations are epsilon-guarded and finite-value safe.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization with Gaussian-consistent scaling
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack transformation: smooth tanh-based pressure curve ---
    # Negative slack → strong monotonic penalty; positive slack → compressive decay
    slack_pressure = np.tanh((-slack) * PARAMS["slack_pressure_tanh_scale"])
    slack_norm = PARAMS["slack_risk_penalty"] * slack_pressure + PARAMS["slack_urgency_scale"] * (1.0 - np.tanh(slack * PARAMS["slack_decay_scale"]))

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: gated by fairness headroom: wait_time / (|slack| + eps) ---
    # Only allow energy optimization where slack permits waiting; avoids aggressive scheduling under risk
    fairness_headroom = np.divide(ready_wait_time, np.abs(slack) + eps)
    energy_norm = mad_normalize(min_incremental_energy)
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * np.clip(fairness_headroom, 0.0, 1.0)

    # --- Critical path & successor release: upward_rank × remaining_work, gated by slack feasibility and low uncertainty ---
    rank_work_interaction = upward_rank * remaining_work
    rank_work_norm = mad_normalize(rank_work_interaction)
    # Conditional DDL protection gate: only activate when slack is feasible AND uncertainty is below median
    median_uncertainty = np.median(uncertainty)
    ddl_protection_gate = ((slack >= 0.0) & (uncertainty <= median_uncertainty + eps)).astype(float)
    critical_successor_boost = PARAMS["critical_path_leverage"] * rank_work_norm * ddl_protection_gate

    # --- Fairness: saturating exponential wait boost ---
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * (1.0 - np.exp(-PARAMS["wait_fairness_gain"] * (wait_norm + eps)))

    # --- Uncertainty interaction: multiplicative only under high slack pressure and critical path context ---
    unc_norm = mad_normalize(uncertainty)
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure * ddl_protection_gate

    # --- Assemble final score: smaller = higher priority ---
    # Note: critical_successor_boost is subtracted because higher interaction indicates greater criticality
    score = (
        slack_norm  # dominates: unified pressure signal for deadline risk
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_successor_boost
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: replace inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
