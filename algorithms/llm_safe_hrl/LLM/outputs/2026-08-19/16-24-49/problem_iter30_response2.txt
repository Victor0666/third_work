def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Directional slack normalization: preserves sign-awareness for violation detection while enabling coherent magnitude scaling via convex blend of signed/absolute slack.
      - Restored robust z-score normalization (not MAD) for all features — avoids cross-dimension signal leakage; slack retains directional semantics.
      - Replaced overscaled successor urgency with bounded linear headroom gate: `np.clip(slack, 0, PARAMS['slack_max_bound'])` ensures monotonic, interpretable scaling without saturation.
      - Unified criticality boost condition now uses *directionally normalized slack* (blended signed+abs) instead of raw deficit — improves sensitivity at marginal slack (e.g., slack ≈ -0.1 vs 0.1).
      - All non-DDL terms strictly gated by `slack > 0.0`; no leakage into violation regime — preserves hard deadline guarantees.
      - Wait-time anti-starvation remains power-law dampened but now anchored to directional slack norm for consistent behavior across violation/margin regimes.
      - Uses only {-2,-1,0,1,2} literals; all tunables via PARAMS; shape (N,) guaranteed.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust z-score normalization with clipping (avoids percentile fragility for sparse sets)
    def z_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        mean_x = np.mean(x)
        std_x = np.std(x, ddof=0) + eps
        z = (x - mean_x) / std_x
        return np.clip(z, -2.0, 2.0)

    # --- Directional slack normalization: convex blend preserving sign and magnitude coherence ---
    abs_slack = np.abs(slack) + eps
    # Blend signed slack (for violation polarity) and absolute slack (for risk magnitude) using tunable weight
    slack_dir_norm = PARAMS["slack_directional_normalization"] * slack + (1.0 - PARAMS["slack_directional_normalization"]) * abs_slack

    # --- DDL-critical base scores (always active) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Criticality boost: activated only under DDL pressure (slack <= 0), using directionally normalized slack ---
    is_tight_or_violated = (slack <= 0)
    rank_z = z_normalize(upward_rank)
    # Use directionally normalized slack for deficit sensitivity near zero-crossing
    slack_dir_z = z_normalize(slack_dir_norm)
    is_high_rank_norm = (rank_z >= 0.0)
    is_high_deficit_norm = (slack_dir_z <= 0.0)  # negative directionally normalized slack → stronger deficit signal
    critical_gate = np.where(
        is_high_rank_norm & is_high_deficit_norm & is_tight_or_violated,
        PARAMS["criticality_boost"],
        1.0
    )

    # --- Lexicographic gate: non-DDL terms only active when slack > 0 (strictly safe) ---
    slack_headroom_mask = np.where(slack > 0.0, 1.0, 0.0)

    # --- Non-DDL terms (only active when slack is strictly positive) ---
    duration_total_safe = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total_safe
    energy_eff_score = z_normalize(energy_per_sec)

    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -z_normalize(upward_rank) * weight_rank

    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = z_normalize(min_incremental_energy)
    unc_norm = z_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Successor-release urgency: bounded linear headroom gate replaces power-law — monotonic, stable, no saturation ---
    successor_urgency = np.where(
        slack_headroom_mask > 0.0,
        (remaining_work + eps) / (upward_rank + eps),
        0.0
    )
    # Bounded linear gate: clip slack to [0, slack_ub] → prevents dominance in large-headroom regimes
    bounded_headroom = np.clip(slack, 0.0, slack_ub)
    successor_score = z_normalize(successor_urgency) * slack_headroom_mask * bounded_headroom

    # --- Wait-time anti-starvation: power-law dampened using directionally normalized slack norm for consistency ---
    # Uses |slack_dir_norm| to anchor dampening across both violation and margin regimes
    wait_score_raw = ready_wait_time / np.power(np.abs(slack_dir_norm) + 1.0, PARAMS["wait_starvation_dampening_power"])
    wait_score = np.clip(z_normalize(wait_score_raw), 0.0, 1.0)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        z_normalize(slack_score) +
        z_normalize(unc_slack_coupling) +
        z_normalize(duration_risk)
    )
    
    # Non-DDL terms only added when slack is strictly positive (lexicographic enforcement)
    score += slack_headroom_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        successor_score
    )

    # Add unconditional wait-score (anti-starvation), but bounded to prevent dominance
    score += wait_score

    # Apply criticality boost multiplicatively
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
