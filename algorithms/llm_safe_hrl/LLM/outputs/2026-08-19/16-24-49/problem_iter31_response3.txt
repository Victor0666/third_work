def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating:
      - Strict lexicographic DDL gating (slack > 0) preserved.
      - Successor-release coupling: (upward_rank × remaining_work) scaled by normalized slack deficit, activated only under tight/late conditions.
      - Host-load–aware energy scaling via (1 + normalized_uncertainty), applied only when slack > 0 to avoid energy optimization under violation.
      - Bounded exponential urgency decay (-slack)**exponent for critical-path prioritization near deadlines.
      - Robust MAD-based normalization (not percentile) for improved stability in sparse ready sets (N ≤ 5).
      - Ready-wait anti-starvation now uses exponential decay: exp(-ready_wait_decay * normalized_wait), reducing dominance in congested queues.
      - All divisions guarded; no unbounded ops; shape (N,) enforced.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust MAD normalization: median absolute deviation, avoids percentile fragility in small N
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med)) + eps
        normalized = (x - med) / mad
        # Clip to [-2, 2] to prevent outliers dominating scores (allowed literals: -2, -1, 0, 1, 2)
        return np.clip(normalized, -2.0, 2.0)

    # --- DDL-critical base scores (always active) ---
    # Negative slack penalty: bounded exponential urgency
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    
    # Deadline pressure & coupling terms
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Successor-release coupling: (upward_rank * remaining_work) * normalized_slack_deficit ---
    # Only active under DDL pressure (slack <= 0), directly addresses CRITICAL_PATH_STARVATION
    is_tight_or_violated = (slack <= 0)
    rank_work_product = upward_rank * remaining_work
    # Normalize slack deficit for coupling strength: map [-∞, 0] → [0, 1]
    slack_deficit_norm = np.where(
        is_tight_or_violated,
        1.0 / (1.0 + np.exp(-slack)),  # Sigmoid from 0→1 as slack goes from 0→-∞
        0.0
    )
    successor_release_score = np.where(
        is_tight_or_violated,
        rank_work_product * slack_deficit_norm * PARAMS["successor_release_weight"],
        0.0
    )

    # --- Lexicographic gate: non-DDL terms only active when slack > 0 ---
    slack_headroom_mask = np.where(slack > 0.0, 1.0, 0.0)

    # --- Non-DDL terms (only active when slack > 0) ---
    # Energy efficiency: min_incremental_energy / (min_exec_time + min_comm_time + eps)
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = mad_normalize(energy_per_sec)

    # Rank score: inverted upward_rank, weighted by slack headroom
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -mad_normalize(upward_rank) * weight_rank

    # Uncertainty-gated energy: host-load–aware scaling (1 + normalized_uncertainty) only when slack > 0
    unc_norm = mad_normalize(uncertainty)
    energy_norm = mad_normalize(min_incremental_energy)
    # Sigmoid gate on uncertainty to suppress energy-uncertainty interaction at extremes
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_uncertainty_score = (
        PARAMS["energy_uncertainty_interaction"] * 
        energy_norm * 
        (1.0 + unc_norm) * 
        unc_sigmoid
    )

    # Anti-starvation: exponential decay on ready_wait_time, normalized and masked
    wait_normalized = mad_normalize(ready_wait_time)
    wait_score = np.where(
        slack_headroom_mask > 0.0,
        np.exp(-PARAMS["ready_wait_decay"] * np.abs(wait_normalized)),
        0.0
    )

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        mad_normalize(successor_release_score)
    )
    
    # Non-DDL terms only added when slack is strictly positive (lexicographic enforcement)
    score += slack_headroom_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
