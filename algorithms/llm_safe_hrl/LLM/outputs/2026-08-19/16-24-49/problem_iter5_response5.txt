def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust normalization and work-density with Parent 1's successor-release insight,
       enhanced by novel priority stabilization and unified slack-pressure gating.
    
    Structural innovations:
    - Introduces `priority_stabilization_alpha`: applies lightweight exponential smoothing over *logical* scheduling steps
      (via deterministic pseudo-historical bias using task index position) to suppress jitter-induced priority inversion.
    - Unifies all slack-dependent modulations (energy dampening, uncertainty amplification, fairness scaling) under a single
      bounded `slack_pressure_bounded = tanh(slack_pressure * scale)` signal for coherence and differentiability.
    - Adds `successor_release_weight` term: uses `upward_rank * remaining_work` as proxy for critical path unlocking power,
      gated by slack feasibility (`slack >= 0`) to avoid rewarding late releases.
    - Replaces brittle hard-mode fallbacks with smooth, monotonic slack-pressure modulation — no discrete branches beyond piecewise slack shape.
    - All numeric literals are strictly -2, -1, 0, 1, or 2; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs — ensure finite, non-NaN, bounded values
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization with Gaussian-consistent scaling
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack transformation: piecewise-smooth risk-urgency curve ---
    # Negative slack → strong linear penalty; near-zero positive → exponential urgency; larger positive → capped linear decay
    neg_mask = slack < 0.0
    zeroish_mask = (slack >= 0.0) & (slack < 1.0)
    pos_mask = slack >= 1.0
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[zeroish_mask]) - 1.0)
    slack_norm[pos_mask] = np.clip(slack[pos_mask], 0.0, PARAMS["slack_cap"])

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: dampened under deadline pressure via unified slack pressure ---
    energy_norm = mad_normalize(min_incremental_energy)
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - slack_pressure_bounded)

    # --- Critical path leverage: only active when slack allows ---
    rank_norm = mad_normalize(upward_rank)
    critical_gate = (slack >= 0.0).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate

    # --- Work density bonus: remaining_work per unit duration, slack-gated ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_gate = (slack >= 0.0).astype(float)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * work_density_gate

    # --- Successor release impact: upward_rank * remaining_work as proxy for unblocking power, slack-gated ---
    successor_impact = upward_rank * remaining_work
    successor_norm = mad_normalize(successor_impact)
    successor_gate = (slack >= 0.0).astype(float)
    successor_bonus = PARAMS["successor_release_weight"] * successor_norm * successor_gate

    # --- Fairness: wait score scaled by slack headroom and stabilized ---
    wait_headroom = np.maximum(1.0, slack + 1.0)
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * wait_norm / (wait_headroom + eps)

    # --- Uncertainty interaction: multiplicative with same bounded slack pressure ---
    unc_norm = mad_normalize(uncertainty)
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded

    # --- Priority stabilization: deterministic pseudo-temporal smoothing using task index order ---
    # Simulates EMA without state: applies alpha-weighted bias toward median priority of earlier indices
    # Ensures deterministic repeatability while reducing oscillation — critical for DAG stability
    N = len(slack)
    if N > 1:
        # Create deterministic index-based weighting: earlier indices slightly favored (smoothing anchor)
        # Equivalent to applying EMA with fixed horizon: weight[i] ∝ exp(-alpha * i), normalized
        idx = np.arange(N, dtype=float)
        idx_weights = np.exp(-PARAMS["priority_stabilization_alpha"] * idx)
        idx_weights /= np.sum(idx_weights) + eps
        # Compute weighted median reference (robust against outliers)
        ref_priority = np.sum(idx_weights * slack_norm)
        # Stabilize by pulling extreme scores toward reference — soft shrinkage
        stab_offset = (slack_norm - ref_priority) * (1.0 - slack_pressure_bounded)
        slack_norm = slack_norm - stab_offset
    # For N == 1, no stabilization needed (no relative ordering possible)

    # --- Assemble final score: smaller = higher priority ---
    # Note: critical_boost, work_density_bonus, successor_bonus are subtracted (higher values improve schedule)
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - critical_boost
        - work_density_bonus
        - successor_bonus
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: replace inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
