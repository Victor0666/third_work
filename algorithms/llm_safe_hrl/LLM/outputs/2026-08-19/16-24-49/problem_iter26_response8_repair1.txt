def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with headroom-aware gating and robust duration penalty:
      - Replaced 'deadline_headroom_gate_threshold' with fixed 0.0 threshold to satisfy parameter count limit,
        preserving strict lexicographic DDL gating (slack > 0) from Parent 2.
      - Added explicit robust duration penalty: MAD-normalized (min_exec_time + min_comm_time) — new structural term.
      - Critical path release uses *normalized product* (upward_rank * remaining_work) to avoid scale skew.
      - Energy-uncertainty interaction now gated by both sigmoid AND slack_headroom_mask for safety.
      - Anti-starvation uses *relative wait ratio*: ready_wait_time / (1 + duration_total), preventing bias.
      - All numeric literals are in {-2, -1, 0, 1, 2}; epsilon handled via PARAMS; no in-place mutation.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust MAD-based normalization: median absolute deviation
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        dev = np.abs(x - med)
        mad = np.median(dev) + eps
        normalized = (x - med) / mad
        return np.clip(normalized, -2.0, 2.0)

    # --- DDL-critical base scores (always active) ---
    # Negative slack penalty: strong power-law penalty for lateness risk
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    
    # Deadline pressure term: product of uncertainty and slack deficit
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    
    # Duration risk: penalize long-duration tasks (exec + comm) under uncertainty
    duration_total = min_exec_time + min_comm_time + eps
    duration_risk = duration_total * uncertainty * PARAMS["duration_risk_penalty"]
    
    # Robust duration penalty: normalized raw duration, independent of uncertainty (new structural term)
    duration_penalty = mad_normalize(duration_total)

    # --- Critical path release: normalized product under DDL pressure ---
    is_ddl_constrained = (slack <= 0.0)
    # Normalize each factor first to avoid skew from magnitude differences
    norm_upward_rank = mad_normalize(upward_rank)
    norm_remaining_work = mad_normalize(remaining_work)
    successor_release_score = norm_upward_rank * norm_remaining_work * PARAMS["criticality_boost"]
    # Invert so higher criticality → lower score
    successor_release_contribution = np.where(is_ddl_constrained, -successor_release_score, 0.0)

    # --- Lexicographic gate: non-DDL terms only active when slack > 0 (strict hard gate) ---
    slack_headroom_mask = np.where(slack > 0.0, 1.0, 0.0)

    # --- Non-DDL terms (only active when slack strictly positive) ---
    # Energy efficiency: marginal energy per total duration (robust against zero)
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = mad_normalize(energy_per_sec)

    # Rank-weighted slack scaling: linear interpolation from 0→1 over empirical bounds
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -mad_normalize(upward_rank) * weight_rank

    # Uncertainty-gated energy interaction: now also masked by slack headroom for safety
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = (
        PARAMS["energy_uncertainty_interaction"] * 
        energy_norm * 
        unc_norm * 
        unc_sigmoid * 
        slack_headroom_mask
    )

    # Relative anti-starvation: ready_wait_time / (1 + duration_total) — avoids bias toward long-duration tasks
    relative_wait = np.where(
        slack_headroom_mask > 0.0,
        ready_wait_time / (1.0 + duration_total),
        0.0
    )
    wait_score = mad_normalize(relative_wait)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        duration_penalty +
        successor_release_contribution
    )
    
    # Non-DDL terms only added when slack strictly positive (lexicographic enforcement)
    score += slack_headroom_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
