def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """
    Hybrid priority rule combining Parent 2's robust piecewise slack modeling and triple gating
    with Parent 1's successor-release interaction and uncertainty-coupled energy penalty.
    
    Key structural improvements:
    - Retains Parent 2's interpretable piecewise slack transformation (linear risk + saturating exponential urgency).
    - Adds Parent 1's successor-release interaction (upward_rank * remaining_work), but now *quadruple-gated*:
      (slack >= 0) ∧ (uncertainty <= median) ∧ (energy <= median) ∧ (duration > 0) → avoids degenerate cases.
    - Introduces novel 'energy_uncertainty_coupling': extra penalty on energy score when uncertainty is high,
      enforcing that high-risk tasks are not scheduled on energy-efficient but unreliable VMs.
    - Uses unified MAD-normalized features throughout; all operations epsilon-guarded and inf-safe.
    - All numeric literals are in {-2,-1,0,1,2}; no hidden constants or randomization.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(np.float64)
    
    # Cast and sanitize inputs — ensure no NaN/inf persists
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=np.float64), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=np.float64), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=np.float64), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization with Gaussian-consistent scaling
    def mad_normalize(x):
        x = np.asarray(x, dtype=np.float64)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack transformation: piecewise-smooth, interpretable, bounded ---
    neg_mask = slack < 0.0
    tight_mask = (slack >= 0.0) & (slack <= PARAMS["slack_cap"])
    loose_mask = slack > PARAMS["slack_cap"]
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])  # linear penalty
    slack_norm[tight_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[tight_mask]) - 1.0)  # saturating urgency
    slack_norm[loose_mask] = PARAMS["slack_urgency_scale"] * (np.exp(PARAMS["slack_cap"]) - 1.0)  # flat decay cap

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: base normalized energy ---
    energy_norm = mad_normalize(min_incremental_energy)
    energy_score = PARAMS["energy_sensitivity"] * energy_norm

    # --- Uncertainty-coupled energy penalty: extra cost when uncertainty is high ---
    unc_med = np.median(uncertainty)
    unc_high_mask = uncertainty > (unc_med + eps)
    energy_uncertainty_penalty = PARAMS["energy_uncertainty_coupling"] * energy_norm * unc_high_mask.astype(float)

    # --- Rank: upward_rank normalized ---
    rank_norm = mad_normalize(upward_rank)

    # --- Critical path: triple-gated leverage (slack ≥ 0 ∧ low uncertainty ∧ low energy) ---
    unc_med = np.median(uncertainty)
    eng_med = np.median(min_incremental_energy)
    unc_gate = (uncertainty <= unc_med).astype(float)
    eng_gate = (min_incremental_energy <= eng_med).astype(float)
    critical_gate = ((slack >= 0.0) * unc_gate * eng_gate).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate

    # --- Successor release: upward_rank × remaining_work, quadruple-gated ---
    # Gate: (slack >= 0) ∧ (uncertainty <= median) ∧ (energy <= median) ∧ (duration > 0)
    duration_safe = (duration > eps).astype(float)
    successor_interaction = upward_rank * remaining_work
    successor_norm = mad_normalize(successor_interaction)
    successor_gate = critical_gate * duration_safe
    successor_boost = PARAMS["successor_release_strength"] * successor_norm * successor_gate

    # --- Work density: remaining_work per unit duration, gated by slack feasibility ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_gate = (slack >= 0.0).astype(float)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * work_density_gate

    # --- Fairness: saturating exponential wait boost, gated by slack feasibility ---
    wait_norm = mad_normalize(ready_wait_time)
    wait_boost = 1.0 - np.exp(-PARAMS["wait_fairness_gain"] * np.maximum(wait_norm, 0.0))
    wait_score = wait_boost * (slack >= 0.0).astype(float)

    # --- Uncertainty interaction: additive amplification under slack pressure ---
    unc_norm = mad_normalize(uncertainty)
    slack_pressure = np.clip(-slack, 0.0, np.inf)  # only pressure from lateness risk
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])

    # --- Assemble final score: smaller = higher priority ---
    # Penalties added; beneficial boosts subtracted
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_score
        + energy_uncertainty_penalty
        - critical_boost
        - successor_boost
        - work_density_bonus
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    
    return score
