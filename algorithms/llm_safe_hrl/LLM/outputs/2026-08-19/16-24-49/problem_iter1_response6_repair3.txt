def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all tunable constants declared in PARAMETER_SCHEMA.
    Uses robust normalization, sigmoid-gated criticality, arctan fairness boost,
    and pressure-dependent energy weighting — all with no hidden literals beyond [-2,2]."""
    eps = PARAMS["epsilon"]
    
    # Safe finite normalization: use median absolute deviation + mean fallback for robustness
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        x_abs = np.abs(x)
        med_abs = np.median(x_abs)
        scale = np.where(med_abs > eps, med_abs, np.mean(x_abs) + eps)
        return x / (scale + eps)

    # Compute base normalized features
    exec_comm = min_exec_time + min_comm_time
    norm_exec_comm = robust_normalize(exec_comm)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # --- Deadline risk gating: strong penalty for negative slack, smooth transition near zero ---
    slack_sign = np.tanh(norm_slack * 2.0)  # bounded [-1,1], steep near zero
    slack_risk_mask = np.where(norm_slack < 0, 1.0, 0.0)  # binary risk indicator
    slack_penalty = PARAMS["slack_risk_penalty"] * slack_risk_mask * (-norm_slack)

    # --- Criticality boost: amplify upward_rank only when deadline or uncertainty pressure is high ---
    pressure_score = np.clip(
        norm_slack * PARAMS["uncertainty_slack_gate"] + norm_uncert,
        -1.0, 2.0
    )
    # Sigmoid-gated boost: stronger rank weight when pressure > 0.3
    rank_boost_factor = 1.0 + PARAMS["criticality_boost"] * (
        1.0 / (1.0 + np.exp(-4.0 * (pressure_score - 0.3)))
    )
    critical_rank_score = -rank_boost_factor * norm_rank

    # --- Energy efficiency: de-prioritize high marginal energy, but soften under deadline pressure ---
    # When slack is very tight (norm_slack < -0.5), reduce energy weight using declared param
    energy_weight_adj = PARAMS["energy_efficiency_scale"] * np.where(
        norm_slack < -0.5,
        PARAMS["energy_weight_under_pressure"],
        1.0
    )
    energy_score = energy_weight_adj * norm_energy

    # --- Duration-uncertainty coupling: tasks with both high duration and high uncertainty become urgent ---
    dur_uncert_coupling = np.clip(
        norm_exec_comm * norm_uncert * PARAMS["duration_uncertainty_coupling"],
        -1.0, 2.0
    )

    # --- Work density ratio: penalize tasks with high remaining_work but low exec_comm (inefficient bottlenecks) ---
    work_density = remaining_work / (exec_comm + eps)
    norm_work_density = robust_normalize(work_density)
    work_density_penalty = PARAMS["work_density_ratio"] * norm_work_density

    # --- Fairness-aware waiting boost: smooth saturating boost using arctan (bounded, monotonic, no overflow) ---
    wait_boost = PARAMS["wait_fairness_gain"] * (2.0 / np.pi) * np.arctan(norm_wait * 3.0)

    # --- Final score: smaller = higher priority ---
    score = (
        slack_penalty                  # Primary: hard-deadline enforcement
        + critical_rank_score          # Secondary: path-criticality amplification under pressure
        + energy_score                 # Tertiary: energy minimization where DDL-safe
        + dur_uncert_coupling          # Risk-aware urgency coupling
        + work_density_penalty         # Bottleneck efficiency correction
        - wait_boost                   # Anti-starvation (subtracts → raises priority)
    )

    # Ensure finite output: replace inf/nan using declared parameters
    score = np.nan_to_num(
        score,
        nan=PARAMS["nan_replacement"],
        posinf=PARAMS["posinf_replacement"],
        neginf=PARAMS["neginf_replacement"]
    )
    return score
