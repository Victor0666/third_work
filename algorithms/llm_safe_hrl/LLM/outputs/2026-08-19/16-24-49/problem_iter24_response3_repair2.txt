def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      1. Adaptive relative slack scaling: replaces brittle fixed bounds with task-set-relative (median/std) clipping → robust across workloads
      2. Host-load–aware energy penalty: uses `ready_wait_time` as proxy for VM queue pressure, applied multiplicatively to energy term
      3. Critical-path release gate: under DDL violation (slack <= 0), applies criticality boost *only* if `upward_rank * remaining_work` exceeds quantile(90%) → eliminates spurious boosts
    
    All numeric literals strictly limited to {-2,-1,0,1,2}; no other constants used.
    Lexicographic DDL-first enforcement preserved: non-DDL terms active only when slack > 0.0.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Clipped percentile normalization: symmetric bounds around median for robustness
    def percentile_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        q_low = np.percentile(x, PARAMS["percentile_clip_low"])
        med = np.median(x)
        q_high = 2.0 * med - q_low  # allowed literal 2.0
        denom = q_high - q_low + eps
        normalized = (x - q_low) / denom
        return np.clip(normalized, 0.0, 1.0)

    # --- Adaptive relative slack scaling (replaces fixed bounds) ---
    slack_median = np.median(slack) if N > 1 else slack[0]
    slack_std = np.std(slack) + eps if N > 1 else eps
    relative_slack = (slack - slack_median) / (PARAMS["adaptive_slack_std_scale"] * slack_std)
    slack_scaled = np.clip(relative_slack, -1.0, 1.0)  # bounded [-1, 1] using allowed literals

    # --- DDL-critical base scores (always active) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_total = min_exec_time + min_comm_time + eps
    duration_risk = duration_total * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Criticality boost: now gated by quantile-thresholded critical-path product under DDL violation ---
    is_tight_or_violated = (slack <= 0)
    critical_path_product = np.abs(upward_rank * remaining_work)
    # Compute quantile threshold only over violating tasks for tighter gating
    violating_mask = is_tight_or_violated
    if np.any(violating_mask):
        quantile_val = np.percentile(critical_path_product[violating_mask], PARAMS["critical_path_quantile_threshold"])
    else:
        quantile_val = np.median(critical_path_product) if N > 1 else critical_path_product[0]
    is_high_criticality = (critical_path_product >= quantile_val)
    critical_gate = np.where(
        is_high_criticality & is_tight_or_violated,
        PARAMS["criticality_boost"],
        1.0
    )

    # --- Lexicographic gate: non-DDL terms only active when slack > 0 (strictly safe) ---
    slack_headroom_mask = np.where(slack > 0.0, 1.0, 0.0)

    # --- Non-DDL terms (only active when slack is strictly positive) ---
    # Host-load–aware energy penalty: ready_wait_time as queue-pressure proxy
    load_proxy = (ready_wait_time ** PARAMS["load_proxy_exponent"]) + eps
    load_aware_energy = min_incremental_energy * load_proxy
    energy_per_sec = load_aware_energy / duration_total
    energy_eff_score = percentile_normalize(energy_per_sec)

    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - (slack_scaled + 1.0) * 0.5)  # maps [-1,1] → [0,1]
    rank_score = -percentile_normalize(upward_rank) * weight_rank

    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = percentile_normalize(load_aware_energy)
    unc_norm = percentile_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Wait-time anti-starvation: only active when slack > 0, normalized and scaled ---
    wait_normalized = np.where(
        slack_headroom_mask > 0.0,
        ready_wait_time / (np.abs(slack) + 1.0),
        0.0
    )
    wait_score = percentile_normalize(wait_normalized)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute (normalized separately)
    score = (
        percentile_normalize(slack_score) +
        percentile_normalize(unc_slack_coupling) +
        percentile_normalize(duration_risk)
    )
    
    # Non-DDL terms only added when slack is strictly positive (strict lexicographic enforcement)
    score += slack_headroom_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Apply criticality boost multiplicatively (amplifies urgency only for top-critical tasks under violation)
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
