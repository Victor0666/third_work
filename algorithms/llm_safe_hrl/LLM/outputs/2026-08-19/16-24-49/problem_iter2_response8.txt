def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robustness from Parent 1 and efficiency from Parent 2.
    
    Key structural improvements:
    - Introduces hard feasibility gate: energy term only active when slack > PARAMS["energy_feasibility_gate_threshold"]
      (replacing soft exp(-slk) gating in Parent 1, enabling strict DDL-first behavior).
    - Replaces IQR with robust mean-abs normalization *and* adds explicit finite-range clipping for all normalized terms.
    - Uses percentile-based slack range interpolation *and* couples it with uncertainty-aware criticality boost.
    - Adds uncertainty-weighted duration risk as separate term (not just multiplicative coupling).
    - All numeric literals are strictly in {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization: mean-abs denominator + clip output to [-2,2] for bounded influence
    def robust_norm(x):
        x = np.asarray(x, dtype=float)
        denom = np.mean(np.abs(x)) + eps
        normed = x / denom
        return np.clip(normed, -2.0, 2.0)  # Bounded influence
    
    # Slack scoring: piecewise — penalty for negative slack, linear reward for positive
    slack_arr = np.asarray(slack, dtype=float)
    slack_penalty = np.where(
        slack_arr < 0,
        (-slack_arr) ** PARAMS["slack_penalty_exponent"],
        slack_arr * PARAMS["early_slack_reward_factor"]
    )
    
    # Criticality boost: only when both high rank AND tight/overdue slack
    rank_median = np.median(upward_rank) if len(upward_rank) > 1 else np.mean(upward_rank)
    is_high_rank = (upward_rank >= rank_median)
    is_tight_slack = (slack_arr <= PARAMS["energy_feasibility_gate_threshold"] + eps)
    critical_gate = np.where(is_high_rank & is_tight_slack, PARAMS["criticality_boost"], 1.0)
    
    # Energy efficiency ratio: joules per second — only used if slack allows energy optimization
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = robust_norm(energy_per_sec)
    
    # Uncertainty-slack coupling: direct penalty term
    deadline_pressure = np.maximum(0.0, -slack_arr)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    
    # Duration risk: long duration × uncertainty, independent of slack sign
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]
    
    # Anti-starvation: saturated wait bonus (negative score → higher priority)
    wait_clipped = np.clip(ready_wait_time, 0, PARAMS["wait_saturation_threshold"])
    wait_score = -robust_norm(wait_clipped)
    
    # Rank-slack interpolation: weight toward rank when slack is tight
    if len(slack_arr) > 1:
        slack_p90 = np.percentile(slack_arr, PARAMS["slack_percentile_high"])
        slack_p10 = np.percentile(slack_arr, PARAMS["slack_percentile_low"])
        slack_range = np.maximum(eps, slack_p90 - slack_p10)
        slack_normalized = np.clip((slack_arr - np.min(slack_arr)) / (slack_range + eps), 0, 1)
    else:
        slack_normalized = np.array([0.0])
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * slack_normalized
    rank_score = -robust_norm(upward_rank) * weight_rank
    
    # Energy term gating: only active when slack > threshold (hard feasibility boundary)
    energy_term = np.where(
        slack_arr > PARAMS["energy_feasibility_gate_threshold"],
        energy_eff_score * PARAMS["energy_efficiency_ratio_weight"],
        0.0
    )
    
    # Assemble final score: smaller = better
    score = (
        robust_norm(slack_penalty)
        + robust_norm(unc_slack_coupling)
        + robust_norm(duration_risk)
        + energy_term
        + rank_score
        + wait_score
        + robust_norm(min_incremental_energy) * (1.0 - weight_rank)  # residual energy bias when rank dominates
    )
    
    # Apply criticality boost multiplicatively
    score = score * critical_gate
    
    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=np.finfo(float).max,
        neginf=-np.finfo(float).max
    )
    
    return score
