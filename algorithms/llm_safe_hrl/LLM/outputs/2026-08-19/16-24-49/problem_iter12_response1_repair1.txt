def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with bounded control flow:
      - Replaces all hard conditionals with *at most three* sigmoid gates (DDL, wait, threshold) — satisfies branch limit.
      - Introduces joint rank-energy alignment using shared MAD scale for robustness.
      - Removes all redundant parameters; every declared parameter is used exactly once.
      - All numeric literals are in {-2,-1,0,1,2}; no hidden constants.
      - No loops, I/O, or state mutation; fully deterministic and finite.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust clipped MAD normalization: uses shared scale for rank & energy
    def mad_normalize_shared(x, ref_array=None):
        x = np.asarray(x, dtype=float)
        if ref_array is None:
            ref_array = x
        med = np.median(ref_array) if N > 1 else np.mean(ref_array)
        mad = np.median(np.abs(ref_array - med)) if N > 1 else eps
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return np.clip((x - med) / scale, -2.0, 2.0)

    # --- Empirical slack scaling ---
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)

    # --- Smooth slack violation penalty (1 sigmoid) ---
    slack_violation = np.maximum(0.0, -slack)
    slack_score = (slack_violation ** PARAMS["slack_penalty_exponent"]) * (
        1.0 / (1.0 + np.exp(-PARAMS["ddl_gate_steepness"] * (slack_violation - eps)))
    )

    # --- Smooth DDL-protection gate (2nd sigmoid) ---
    median_unc = np.median(uncertainty) if N > 1 else np.mean(uncertainty)
    median_energy = np.median(min_incremental_energy) if N > 1 else np.mean(min_incremental_energy)
    slack_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_gate_steepness"] * (-slack + eps)))
    unc_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_gate_steepness"] * (median_unc + eps - uncertainty)))
    energy_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_gate_steepness"] * (median_energy + eps - min_incremental_energy)))
    ddl_protection_gate = slack_gate * unc_gate * energy_gate

    # --- Joint rank-energy alignment ---
    rank_energy_ref = np.concatenate([upward_rank, min_incremental_energy]) if N > 1 else upward_rank
    rank_norm = mad_normalize_shared(upward_rank, rank_energy_ref)
    energy_norm = mad_normalize_shared(min_incremental_energy, rank_energy_ref)

    # --- Critical-path release boost ---
    critical_release_boost = (
        ddl_protection_gate *
        rank_norm *
        (1.0 + PARAMS["criticality_boost"] * (1.0 - slack_scaled))
    )

    # --- Wait-time anti-starvation with smooth activation (3rd sigmoid) ---
    wait_base = ready_wait_time / (np.abs(slack) + 1.0)
    wait_activation = 1.0 / (1.0 + np.exp(-PARAMS["wait_activation_steepness"] * (slack_scaled - 0.5)))
    wait_gated = wait_base * wait_activation
    wait_score = mad_normalize_shared(wait_gated)

    # --- Uncertainty-aware energy efficiency ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_efficiency = min_incremental_energy / (duration_total * (1.0 + uncertainty))
    energy_eff_norm = mad_normalize_shared(energy_efficiency)

    # --- Uncertainty-slack coupling ---
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]

    # --- Duration risk penalty ---
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Rank-energy alignment bonus ---
    rank_energy_alignment = PARAMS["rank_energy_alignment_strength"] * rank_norm * (-energy_norm)

    # --- Energy-uncertainty interaction with smooth threshold ---
    unc_norm = mad_normalize_shared(uncertainty)
    unc_threshold_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_gate_steepness"] * (median_unc + eps - uncertainty)))
    energy_uncertainty_score = (
        PARAMS["energy_uncertainty_interaction"] *
        energy_norm *
        unc_norm *
        unc_threshold_gate
    )

    # --- Assemble final score: smaller = higher priority ---
    score = (
        mad_normalize_shared(slack_score) +
        mad_normalize_shared(unc_slack_coupling) +
        mad_normalize_shared(duration_risk) +
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_norm +
        energy_uncertainty_score +
        wait_score -
        critical_release_boost -
        rank_energy_alignment
    )
    
    # Final safeguard
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
