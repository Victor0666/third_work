def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating:
      - Conditional DDL-protection gate (activated when slack_fraction <= threshold) to prioritize critical-path release.
      - Successor-release interaction: product of upward_rank and remaining_work, gated by slack pressure.
      - Linear wait-normalized fairness: ready_wait_time / (|slack| + 1), avoiding starvation without compromising deadlines.
      - Robust clipped MAD-based slack deviation scoring instead of power penalty.
      - Host-load-aware marginal energy: energy × (1 + uncertainty × max(0, -slack)).
      - All features robustly normalized using MAD; no percentile-based scaling.
      - Final score preserves lexicographic DDL-first ordering.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust MAD normalization: uses only allowed literals (-2,-1,0,1,2)
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x) if N > 1 else np.mean(x)
        mad = np.median(np.abs(x - med)) if N > 1 else eps
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Empirical slack scaling for balance terms ---
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)

    # --- Slack fraction for DDL protection gate: [0,1] mapped from slack to [-1,1] then clipped ---
    slack_fraction = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    
    # --- Conditional DDL-protection gate: activates only when slack_fraction is low (tight deadline) ---
    ddl_protection_gate = (slack_fraction <= 0.25).astype(float)

    # --- Critical-path successor-release interaction: upward_rank * remaining_work, gated by DDL pressure ---
    successor_release_interaction = upward_rank * remaining_work * ddl_protection_gate * 0.87

    # --- Linear wait-normalized fairness: prevents starvation without dominating under deadline pressure ---
    # Scaled by |slack| + 1 to smoothly decay priority boost as slack increases
    wait_norm = ready_wait_time / (np.abs(slack) + 1.0)
    wait_fairness_term = wait_norm * 0.62

    # --- Sliding-window slack deviation: robust alternative to power penalty ---
    # Uses MAD-normalized absolute slack deviation centered at median slack
    slack_deviation = np.abs(slack - np.median(slack)) if N > 1 else np.abs(slack)
    slack_dev_norm = mad_normalize(slack_deviation)

    # --- Host-load-aware marginal energy: energy × (1 + uncertainty × max(0, -slack)) ---
    # Amplifies energy penalty for high-uncertainty tasks with negative slack
    slack_pressure = np.maximum(0.0, -slack)
    load_aware_energy = min_incremental_energy * (1.0 + uncertainty * slack_pressure * 0.1)
    energy_norm = mad_normalize(load_aware_energy)

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    unc_slack_coupling = uncertainty * slack_pressure * PARAMS["uncertainty_slack_coupling"]

    # --- Duration risk penalty: long duration × uncertainty ---
    duration_total = min_exec_time + min_comm_time + eps
    duration_risk = duration_total * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Rank-slack balance: prioritize rank when slack is poor (low slack_scaled) ---
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_norm = mad_normalize(upward_rank)
    rank_score = -rank_norm * weight_rank  # negative: higher rank → lower score → higher priority

    # --- Bounded sigmoid gate on uncertainty for energy-uncertainty interaction ---
    # Prevents explosion at high uncertainty while preserving sensitivity near median
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms come first and dominate via direct addition
    # Fairness and energy terms follow, weighted but secondary
    score = (
        # Primary DDL-risk drivers (non-negotiable)
        slack_dev_norm +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk) +
        
        # Critical-path release (conditional)
        -successor_release_interaction +  # subtract to boost priority
        
        # Secondary objectives: fairness & energy efficiency
        wait_fairness_term +
        energy_norm * PARAMS["energy_efficiency_ratio_weight"] +
        energy_uncertainty_score +
        rank_score
    )
    
    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
