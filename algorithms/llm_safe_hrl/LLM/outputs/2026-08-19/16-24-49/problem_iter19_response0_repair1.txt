def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with lexicographic DDL enforcement via hard gate, joint MAD risk normalization,
       critical-path release emphasis, and anti-starvation with slack-decayed wait-time term.
    
    Key structural changes:
      - Replaces percentile normalization with robust joint MAD normalization over |slack|, uncertainty, and duration_total
      - Introduces critical_path_release_score = upward_rank * remaining_work → unconditionally prioritizes DAG bottlenecks
      - Uses hard gate `slack > ddl_protection_threshold` (not just > 0) to separate feasibility enforcement from optimization
      - Normalizes ready_wait_time with exponent decay: (ready_wait_time / (|slack| + 1)) ** ready_wait_decay_exponent
      - Applies uncertainty_slack_coupling linearly (not multiplicatively) to avoid explosive penalties at extreme negatives
      - Removes fragile sigmoid gates; uses bounded linear interactions and MAD scaling for stability
      - All numeric literals strictly limited to {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Joint MAD normalization: compute median absolute deviation over concatenated risk signals
    # Signals: |slack|, uncertainty, duration_total — all represent risk dimensions
    duration_total = min_exec_time + min_comm_time + eps
    risk_signals = np.concatenate([
        np.abs(slack),
        uncertainty,
        duration_total
    ])
    risk_median = np.median(risk_signals)
    mad = np.median(np.abs(risk_signals - risk_median)) + eps
    joint_scale = PARAMS["joint_mad_scale_factor"] * mad

    def joint_mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        # Normalize using shared scale for coherent risk alignment
        return np.clip((x - risk_median) / joint_scale, -2.0, 2.0)

    # --- DDL-enforcement core: always active ---
    # Slack violation penalty: linear ramp for negative slack (stable, monotonic)
    slack_violation = np.maximum(0.0, -slack)
    # Uncertainty coupling: linear, not exponential — avoids fragility at boundary
    unc_slack_coupling = PARAMS["uncertainty_slack_coupling"] * uncertainty * slack_violation
    # Duration risk: penalize long uncertain durations when slack is tight
    duration_risk = PARAMS["duration_risk_penalty"] * duration_total * np.where(slack <= PARAMS["ddl_protection_threshold"], uncertainty, 0.0)

    # --- Critical path release score: unconditionally active, no gating ---
    # Prioritizes tasks that unlock large amounts of remaining work on high-rank paths
    critical_path_release_score = upward_rank * remaining_work

    # --- Lexicographic gate: non-DDL optimization terms only when slack > ddl_protection_threshold ---
    slack_headroom_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Non-DDL terms (only active when slack exceeds safety threshold) ---
    # Risk-adjusted energy: normalized and weighted
    energy_norm = joint_mad_normalize(min_incremental_energy)
    energy_score = PARAMS["risk_adjusted_energy_weight"] * energy_norm * slack_headroom_mask

    # Energy-uncertainty interaction: only active under sufficient headroom
    unc_norm = joint_mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * slack_headroom_mask

    # Anti-starvation: wait time scaled by slack headroom with exponent decay
    # Prevents starvation while respecting DDL safety — decays smoothly as slack grows
    wait_base = ready_wait_time / (np.abs(slack) + 1.0)
    wait_score = (wait_base ** PARAMS["ready_wait_decay_exponent"]) * slack_headroom_mask
    wait_norm = joint_mad_normalize(wait_score)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical base: violation + coupling + duration risk
    score = (
        joint_mad_normalize(slack_violation) +
        joint_mad_normalize(unc_slack_coupling) +
        joint_mad_normalize(duration_risk)
    )

    # Critical path release: always contributes, highest priority for bottleneck release
    score += -joint_mad_normalize(critical_path_release_score)  # negative because higher release importance → lower score

    # Non-DDL optimization terms: only added when slack is safely above threshold
    score += energy_score + energy_uncertainty_score + wait_norm

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
