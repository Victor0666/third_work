def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaces fragile sigmoid slack penalty with robust, bounded piecewise-linear penalty:
        distinct slopes for overdue/tight (<=0), critical (0 < slack <= threshold), and safe (slack > threshold) regimes.
      - Upgrades congestion gating to *task-level* normalized wait-ratio: (ready_wait_time / (mad_normalize(ready_wait_time) + eps)) 
        instead of global median, improving load-awareness consistency across varying ready-set sizes.
      - Introduces tunable exponent on congestion signal to control nonlinearity sensitivity without overfitting.
      - Removes redundant bounded_duration_risk_factor (per reflection); its role absorbed by tighter piecewise slack coupling + exponential urgency.
      - All numeric literals strictly {-2,-1,0,1,2}; all tunables via PARAMS; deterministic, finite, shape-(N,) guaranteed.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Conditional MAD-based normalization: only applied when N > threshold to avoid degeneracy
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N <= PARAMS["conditional_mad_min_size"]:
            return x  # skip normalization for sparse ready sets
        if N == 1:
            return np.zeros_like(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med)) + eps
        normalized = (x - med) / mad
        return np.clip(normalized, -2.0, 2.0)

    # Joint risk-feature base
    duration_total = min_exec_time + min_comm_time + eps
    slack_abs = np.abs(slack)
    slack_abs_norm = mad_normalize(slack_abs)
    unc_norm = mad_normalize(uncertainty)
    dur_norm = mad_normalize(duration_total)

    # --- Bounded piecewise-linear slack penalty (replaces sigmoid: more robust & interpretable) ---
    # Regime 1: overdue or critically tight (slack <= 0) → steep linear penalty
    tight_mask = np.where(slack <= 0.0, 1.0, 0.0)
    # Regime 2: critical headroom (0 < slack <= threshold) → medium slope
    critical_mask = np.where((slack > 0.0) & (slack <= PARAMS["ddl_protection_threshold"]), 1.0, 0.0)
    # Regime 3: safe headroom (slack > threshold) → shallow slope
    safe_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)
    
    slack_penalty = (
        tight_mask * slack_abs * PARAMS["slack_piecewise_slope_tight"] +
        critical_mask * slack_abs * PARAMS["slack_piecewise_slope_loose"] * 2.0 +
        safe_mask * slack_abs * PARAMS["slack_piecewise_slope_loose"]
    )
    # Apply uncertainty amplification uniformly
    slack_penalty = slack_penalty * (1.0 + unc_norm)

    # Tight-slack uncertainty coupling: *only* activated when slack <= threshold
    tight_slack_mask = np.where(slack <= PARAMS["ddl_protection_threshold"], 1.0, 0.0)
    tight_slack_uncertainty_coupling = (
        tight_slack_mask *
        uncertainty *
        slack_abs_norm *
        PARAMS["tight_slack_uncertainty_coupling"]
    )

    # --- Critical-path release term: unconditional ---
    critical_release_score = upward_rank * remaining_work * PARAMS["critical_path_release_weight"]

    # --- Bounded exponential urgency for slack-modulated rank ---
    slack_distance = np.clip(PARAMS["ddl_protection_threshold"] - slack, 0.0, PARAMS["slack_distance_upper_bound"])
    urgency_factor = np.exp(-slack_distance / (PARAMS["urgency_exponential_decay_rate"] + eps))
    slack_rank_score = -upward_rank * urgency_factor * PARAMS["critical_path_release_weight"]

    # --- Lexicographic gate: non-DDL terms only active when slack > threshold ---
    slack_headroom_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Task-level congestion-aware energy gating (evolved from Parent 1) ---
    # Replace global median with task-level MAD-normalized wait-ratio: more consistent under N variation
    wait_norm_base = mad_normalize(ready_wait_time) + eps
    wait_ratio = np.clip(ready_wait_time / wait_norm_base, 0.1, 10.0)  # avoid extreme ratios
    # Apply tunable sensitivity exponent and uncertainty coupling
    congestion_signal = (wait_ratio ** PARAMS["task_level_congestion_sensitivity"]) * (1.0 + unc_norm)
    congestion_gate = np.clip(congestion_signal * PARAMS["congestion_aware_energy_gate"], 0.0, 1.0)
    
    # Energy term: gated by both slack headroom AND task-level congestion
    energy_norm = mad_normalize(min_incremental_energy)
    energy_term = (
        slack_headroom_mask *
        congestion_gate *
        PARAMS["risk_adjusted_energy_weight"] *
        energy_norm
    )

    # Wait-time anti-starvation: power-law decay (fixed exponent 2 for stability)
    wait_power = np.where(
        slack_headroom_mask > 0.0,
        ready_wait_time ** 2,
        0.0
    )
    wait_norm = mad_normalize(wait_power)
    wait_term = slack_headroom_mask * wait_norm * PARAMS["normalized_wait_headroom_ratio"]

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        mad_normalize(slack_penalty) +
        mad_normalize(tight_slack_uncertainty_coupling) +
        mad_normalize(critical_release_score) +
        mad_normalize(slack_rank_score)
    )
    
    # Non-DDL terms only added when slack exceeds threshold
    score += (
        energy_term +
        wait_term
    )

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
