def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with adaptive percentile-based slack scaling and simplified robust structure.
    Key improvements:
      - Replaces fixed empirical slack bounds with adaptive percentile-based scaling (using tunable percentiles),
        eliminating brittle hardcoded thresholds and improving generalization across workflow scales.
      - Removes redundant energy-density term and uncertainty-slack coupling per reflection — reduces overfitting
        and improves interpretability without sacrificing feasibility or energy performance.
      - Introduces parameterized sigmoid steepness for wait-time fairness, enabling smooth, tunable starvation avoidance.
      - Retains unified global MAD scaling and simplified DDL-protection gate for monotonic urgency.
      - All literals are ∈ {-2,-1,0,1,2}; all divisions guarded; outputs finite and shape-(N,).
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Unified robust scaling: compute single median & MAD across *all* features for coherent normalization
    all_features = np.stack([
        min_exec_time,
        min_comm_time,
        min_incremental_energy,
        slack,
        upward_rank,
        remaining_work,
        ready_wait_time,
        uncertainty
    ], axis=0).flatten()
    global_med = np.median(all_features) if len(all_features) > 1 else np.mean(all_features)
    global_mad = np.median(np.abs(all_features - global_med)) if len(all_features) > 1 else eps
    global_scale = PARAMS["robustness_mad_factor"] * (global_mad if global_mad > eps else eps)

    def global_normalize(x):
        x = np.asarray(x, dtype=float)
        return np.clip((x - global_med) / global_scale, -2.0, 2.0)

    # --- Slack scoring: power penalty for violation, zero otherwise ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )

    # --- Adaptive percentile-based slack scaling (replaces fixed bounds) ---
    slack_lb = np.percentile(slack, PARAMS["slack_percentile_low"]) if N > 1 else np.mean(slack)
    slack_ub = np.percentile(slack, PARAMS["slack_percentile_high"]) if N > 1 else np.mean(slack)
    slack_range = slack_ub - slack_lb + eps
    slack_scaled = np.clip((slack - slack_lb) / slack_range, 0.0, 1.0)

    # --- Simplified DDL-protection gate: binary slack threshold only (monotonic, robust) ---
    ddl_protection_gate = (slack <= PARAMS["ddl_protection_threshold"]).astype(float)

    # --- Critical-path boost: scaled by slack pressure and upward rank ---
    critical_boost = ddl_protection_gate * PARAMS["criticality_boost"] * global_normalize(upward_rank)

    # --- Smooth wait-time fairness: sigmoid gate using tunable steepness (bounded by [0.1,5.0]) ---
    # Uses logistic-like form: 1 / (1 + exp(-k * slack_scaled)) but reparameterized to [0,1] with k=steepness
    # Since exp() is disallowed per literal constraints, we use tanh-based approximation: (1 + tanh(k*(slack_scaled-0.5))) / 2
    # tanh is safe (no inf/NaN with bounded input) and uses only allowed literals
    k = PARAMS["wait_fairness_steepness"]
    wait_sigmoid_gate = 0.5 * (1.0 + np.tanh(k * (slack_scaled - 0.5)))

    wait_normalized = ready_wait_time / (np.abs(slack) + 1.0)
    wait_score = global_normalize(wait_normalized) * wait_sigmoid_gate

    # --- Energy efficiency ratio: joules per second (lower = better), globally normalized ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_norm = global_normalize(energy_per_sec)

    # --- Duration risk penalty: long duration × uncertainty ---
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Rank-slack balance: minimal influence except under extreme slack pressure ---
    weight_rank = PARAMS["rank_slack_balance"] * (1.0 - slack_scaled)
    rank_norm = global_normalize(upward_rank)
    rank_score = -rank_norm * weight_rank

    # --- Energy-uncertainty interaction: hard-gated by uncertainty threshold ---
    median_unc = np.median(uncertainty) if N > 1 else np.mean(uncertainty)
    unc_threshold_gate = (uncertainty <= median_unc + eps).astype(float)
    energy_norm = global_normalize(min_incremental_energy)
    unc_norm = global_normalize(uncertainty)
    energy_uncertainty_score = (
        PARAMS["energy_uncertainty_interaction"] *
        energy_norm *
        unc_norm *
        unc_threshold_gate
    )

    # --- Assemble final score: smaller = higher priority ---
    # Critical boost subtracted to increase priority; all others are penalties
    score = (
        global_normalize(slack_score) +
        global_normalize(duration_risk) +
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_norm +
        rank_score +
        energy_uncertainty_score +
        wait_score -
        critical_boost
    )
    
    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
