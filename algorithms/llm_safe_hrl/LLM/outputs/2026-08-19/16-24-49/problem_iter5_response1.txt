def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's stability with Parent 1's energy gating insight.
    
    Structural improvements:
    - Retains Parent 2's robust MAD normalization and conditional critical-path boosting (slack≥0 ∧ low-uncertainty).
    - Integrates Parent 1's smooth sigmoid energy gating (now applied *before* normalization) to softly activate
      energy minimization only when slack permits feasible optimization — avoids penalizing energy on urgent tasks.
    - Replaces linear work_density bonus with uncertainty-gated version: prioritizes throughput-critical tasks
      only when uncertainty is low enough to trust their estimated duration/work ratio.
    - Uses unified epsilon-guarded sanitization across all inputs to prevent NaN/inf propagation.
    - All numeric literals are strictly in {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs uniformly
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization (Parent 2 strength)
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack transformation: piecewise-smooth risk-urgency curve (Parent 2) ---
    neg_mask = slack < 0.0
    zeroish_mask = (slack >= 0.0) & (slack < 1.0)
    pos_mask = slack >= 1.0
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[zeroish_mask]) - 1.0)
    slack_norm[pos_mask] = np.clip(slack[pos_mask], 0.0, PARAMS["slack_cap"])

    # --- Soft sigmoid energy gating: activates energy optimization only when slack allows (Parent 1) ---
    # Applied *before* normalization to preserve relative ordering among feasible tasks
    energy_gate = 1.0 / (1.0 + np.exp(-PARAMS["energy_feasibility_sigmoid_steepness"] * 
                                     (slack - PARAMS["energy_feasibility_sigmoid_offset"])))
    gated_energy = min_incremental_energy * energy_gate

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: base weight applied to *gated* energy (novel integration) ---
    energy_norm = mad_normalize(gated_energy)
    energy_score = PARAMS["energy_efficiency_bias"] * energy_norm

    # --- Critical path: conditional leverage only if slack >= 0 AND uncertainty is low (Parent 2) ---
    rank_norm = mad_normalize(upward_rank)
    unc_median = np.median(uncertainty)
    critical_gate = ((slack >= 0.0) & (uncertainty <= unc_median + eps)).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate

    # --- Work density: remaining_work per unit duration, now uncertainty-gated (novel improvement) ---
    # Avoids over-prioritizing speculative high-density tasks when uncertainty is high
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_gate = (uncertainty <= unc_median + eps).astype(float)  # Gate by uncertainty, not slack
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * work_density_gate

    # --- Fairness: saturating exponential wait boost (Parent 2) ---
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * (1.0 - np.exp(-wait_norm))

    # --- Uncertainty interaction: multiplicative with slack pressure (Parent 2) ---
    unc_norm = mad_normalize(uncertainty)
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    slack_pressure_bounded = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure_bounded

    # --- Assemble final score: smaller = higher priority ---
    # critical_boost and work_density_bonus are subtracted (higher rank/density → better → lower score)
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_score
        - critical_boost
        - work_density_bonus
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
