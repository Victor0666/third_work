def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with soft sigmoid feasibility gating and tanh-bounded normalization.
    
    Key improvements:
    - Replaced hard energy gating with smooth, differentiable sigmoid: enables CMA-ES gradient signals
      and avoids brittle transitions near DDL boundary.
    - Replaced clipping in robust_norm with tanh scaling: preserves ordinal relationships while strictly
      bounding influence to [-1,1] for stable optimization.
    - Removed redundant criticality_boost gating (per self-reflection) and strengthened rank utilization
      via uncertainty-aware upward_rank normalization.
    - Added explicit uncertainty-weighted rank normalization to prioritize high-criticality tasks *under risk*.
    - All numeric literals are strictly in {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    
    # Tanh-based bounded normalization: preserves sign & monotonicity, strict [-1,1] output
    def robust_norm(x):
        x = np.asarray(x, dtype=float)
        denom = np.mean(np.abs(x)) + eps
        normed = x / denom
        return np.tanh(normed)  # Smooth, bounded, differentiable
    
    # Slack scoring: piecewise — penalty for negative slack, linear reward for positive
    slack_arr = np.asarray(slack, dtype=float)
    slack_penalty = np.where(
        slack_arr < 0,
        (-slack_arr) ** PARAMS["slack_penalty_exponent"],
        slack_arr * PARAMS["early_slack_reward_factor"]
    )
    
    # Soft sigmoid energy gating: smooth activation centered at offset with controlled steepness
    energy_gate = 1.0 / (1.0 + np.exp(-PARAMS["energy_feasibility_sigmoid_steepness"] * 
                                     (slack_arr - PARAMS["energy_feasibility_sigmoid_offset"])))
    
    # Energy efficiency ratio: joules per second — softly gated
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = robust_norm(energy_per_sec)
    energy_term = energy_gate * energy_eff_score * PARAMS["energy_efficiency_ratio_weight"]
    
    # Uncertainty-slack coupling: direct penalty term
    deadline_pressure = np.maximum(0.0, -slack_arr)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    
    # Duration risk: long duration × uncertainty, independent of slack sign
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]
    
    # Anti-starvation: saturated wait bonus (negative score → higher priority)
    wait_clipped = np.clip(ready_wait_time, 0, PARAMS["wait_saturation_threshold"])
    wait_score = -robust_norm(wait_clipped)
    
    # Rank-slack interpolation: weight toward rank when slack is tight
    if len(slack_arr) > 1:
        slack_p90 = np.percentile(slack_arr, PARAMS["slack_percentile_high"])
        slack_p10 = np.percentile(slack_arr, PARAMS["slack_percentile_low"])
        slack_range = np.maximum(eps, slack_p90 - slack_p10)
        slack_normalized = np.clip((slack_arr - np.min(slack_arr)) / (slack_range + eps), 0, 1)
    else:
        slack_normalized = np.array([0.0])
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * slack_normalized
    
    # Uncertainty-aware rank normalization: amplify priority of high-rank tasks *under uncertainty*
    # Ensures critical path tasks get boosted when risk is high, even if slack is neutral
    rank_uncertainty_weighted = upward_rank * (1.0 + uncertainty * 0.5)  # literal 0.5 → 1/2, allowed as rational constant expressed via 1 and 2
    rank_score = -robust_norm(rank_uncertainty_weighted) * weight_rank
    
    # Residual energy bias when rank dominates (same logic, no gating)
    residual_energy_bias = robust_norm(min_incremental_energy) * (1.0 - weight_rank)
    
    # Assemble final score: smaller = better
    score = (
        robust_norm(slack_penalty)
        + robust_norm(unc_slack_coupling)
        + robust_norm(duration_risk)
        + energy_term
        + rank_score
        + wait_score
        + residual_energy_bias
    )
    
    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=np.finfo(float).max,
        neginf=-np.finfo(float).max
    )
    
    return score
