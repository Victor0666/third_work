def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's stability & lexicographic gating with Parent 1's smooth slack-uncertainty coupling and starvation-aware waiting.
    Key novelties:
      - Starvation mitigation now uses thresholded activation (not just slack <= 0) + relative urgency scaling by |slack| + eps
      - Energy terms decay exponentially with slack headroom using normalized_energy_decay_rate → enforces diminishing optimization returns far from DDL
      - Retains robust min-max normalization but adds explicit N=1 guard and monotonic slack scaling via tanh-based soft clamp
      - Replaces linear interpolation with piecewise slack-dependent rank weighting (low/medium/high slack regimes)
      - All non-DDL terms are strictly gated off when slack <= 0, while DDL-critical terms remain fully active and dominant.
    """
    eps = PARAMS["epsilon"]
    mm_eps = PARAMS["minmax_stability_eps"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust min-max normalization with explicit N=1 guard
    def minmax_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x, dtype=float)
        x_min = np.min(x)
        x_max = np.max(x)
        denom = x_max - x_min + mm_eps
        normalized = (x - x_min) / denom
        return np.clip(normalized, 0.0, 1.0)

    # --- DDL-critical base scores (always active, lexically dominant) ---
    # Nonlinear slack violation penalty
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    
    # Uncertainty-slack coupling: smooth tanh-based modulation instead of raw product
    # tanh maps slack → [-1,1], then we scale negative region only
    slack_sign = np.tanh(slack / (np.abs(slack) + eps))
    unc_slack_coupling = np.where(
        slack < 0,
        uncertainty * (-slack) * (1.0 + PARAMS["uncertainty_slack_coupling"] * np.abs(slack_sign)),
        0.0
    )
    
    # Duration risk under uncertainty
    duration_total = min_exec_time + min_comm_time + eps
    duration_risk = duration_total * uncertainty * PARAMS["duration_risk_penalty"]

    # Criticality boost: only when slack <= 0 AND both rank & work above median
    is_tight_or_violated = (slack <= 0)
    rank_median = np.median(upward_rank) if N > 1 else np.mean(upward_rank)
    work_median = np.median(remaining_work) if N > 1 else np.mean(remaining_work)
    is_high_rank = (upward_rank >= rank_median)
    is_high_work = (remaining_work >= work_median)
    critical_gate = np.where(
        is_high_rank & is_high_work & is_tight_or_violated,
        PARAMS["criticality_boost"],
        1.0
    )

    # --- DDL-gated starvation mitigation: thresholded + relative urgency scaling ---
    # Only activates if wait time exceeds threshold AND slack is violated
    starvation_mask = np.where(
        (ready_wait_time >= PARAMS["wait_starvation_threshold"]) & is_tight_or_violated,
        1.0,
        0.0
    )
    # Scale urgency by inverse slack distance to prioritize most critical violations
    wait_urgency = np.where(
        is_tight_or_violated,
        ready_wait_time / (np.abs(slack) + eps),
        0.0
    )
    wait_score = starvation_mask * minmax_normalize(wait_urgency)

    # --- Lexicographic gate: non-DDL terms only active when slack > 0 (strictly safe) ---
    slack_headroom_mask = np.where(slack > 0.0, 1.0, 0.0)

    # Host-load–aware energy scaling: synthetic load proxy = 1 + normalized uncertainty
    load_proxy = 1.0 + minmax_normalize(uncertainty)
    scaled_energy = min_incremental_energy * load_proxy

    # Energy efficiency: energy per unit duration
    energy_per_sec = scaled_energy / duration_total
    energy_eff_score = minmax_normalize(energy_per_sec)

    # Slack-modulated energy decay: exponential suppression far from deadline
    # Use tanh-based soft clamp for stable slack scaling
    slack_centered = (slack - PARAMS["slack_min_bound"]) / (PARAMS["slack_max_bound"] - PARAMS["slack_min_bound"] + eps)
    slack_clamped = np.tanh(slack_centered * 2.0)  # maps to approx [-1,1]
    slack_headroom_norm = np.clip((slack_clamped + 1.0) / 2.0, 0.0, 1.0)  # [0,1]
    energy_decay = np.exp(-PARAMS["normalized_energy_decay_rate"] * slack_headroom_norm)
    energy_final = slack_headroom_mask * energy_eff_score * energy_decay * PARAMS["energy_efficiency_ratio_weight"]

    # Rank scoring with piecewise slack regimes
    # Low slack: rank dominates (critical path)
    # Medium slack: blend with slack headroom
    # High slack: de-emphasize rank (focus on efficiency)
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    
    # Piecewise weight: [0,0.3)->1.0, [0.3,0.7)->linear, [0.7,1.0]->0.2
    rank_weight = np.where(
        slack_scaled < 0.3,
        1.0,
        np.where(
            slack_scaled < 0.7,
            1.0 - (slack_scaled - 0.3) / 0.4,
            0.2
        )
    )
    rank_score = slack_headroom_mask * (-minmax_normalize(upward_rank)) * rank_weight * PARAMS["rank_slack_balance"]

    # Energy-uncertainty interaction with sigmoid gating
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = minmax_normalize(scaled_energy)
    unc_norm = minmax_normalize(uncertainty)
    energy_uncertainty_score = slack_headroom_mask * PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        minmax_normalize(slack_score) +
        minmax_normalize(unc_slack_coupling) +
        minmax_normalize(duration_risk) +
        wait_score
    )
    
    # Non-DDL terms only added when slack is strictly positive
    score += (
        energy_final +
        rank_score +
        energy_uncertainty_score
    )

    # Apply criticality boost multiplicatively
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
