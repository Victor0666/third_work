def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule integrating successor-release interaction and host-load conditional gating.
    
    Key structural mutations:
    - Replaces piecewise slack transformation with smooth tanh-scaled urgency + linear risk penalty
      for unified DDL pressure modeling (no discontinuities at zero)
    - Adds successor-release interaction: upward_rank * remaining_work, gated by slack feasibility
      to unblock critical paths early (counterfactual evidence: 'successor_release_interaction')
    - Introduces host-load conditional gate: energy term is suppressed only when both slack > 0 AND 
      uncertainty <= median_uncertainty, preventing premature energy optimization under risk
    - Replaces wait fairness with saturating exponential: 1 - exp(-gain * norm_wait), bounded and robust
    - Uses MAD-normalized features throughout; all operations epsilon-guarded and inf-safe
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization with Gaussian-consistent scaling
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Unified slack pressure: smooth tanh urgency + linear risk penalty ---
    # Negative slack → linear penalty; positive slack → tanh-scaled urgency (bounded [0,1])
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    slack_urgency = np.tanh(slack * PARAMS["slack_pressure_tanh_scale"])  # monotonic, bounded
    slack_risk = PARAMS["slack_risk_penalty"] * slack_pressure
    slack_norm = slack_risk + PARAMS["slack_urgency_scale"] * (1.0 - slack_urgency)

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: conditional suppression only when slack > 0 AND uncertainty low ---
    energy_norm = mad_normalize(min_incremental_energy)
    median_unc = np.median(uncertainty)
    energy_suppression_gate = ((slack > 0.0) & (uncertainty <= median_unc + eps)).astype(float)
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * (1.0 - energy_suppression_gate)

    # --- Critical path & successor release: upward_rank × remaining_work, gated by slack feasibility ---
    rank_work_interaction = upward_rank * remaining_work
    rank_work_norm = mad_normalize(rank_work_interaction)
    successor_gate = (slack >= 0.0).astype(float)
    successor_boost = PARAMS["critical_path_leverage"] * rank_work_norm * successor_gate

    # --- Fairness: saturating exponential wait boost (bounded, no starvation) ---
    wait_norm = mad_normalize(ready_wait_time)
    wait_score = PARAMS["wait_fairness_gain"] * (1.0 - np.exp(-PARAMS["wait_fairness_gain"] * wait_norm))

    # --- Uncertainty interaction: additive amplification only under deadline pressure ---
    unc_norm = mad_normalize(uncertainty)
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure

    # --- Assemble final score: smaller = higher priority ---
    # Note: successor_boost is subtracted (higher interaction = more critical path release)
    score = (
        slack_norm  # dominates: captures both risk and urgency smoothly
        + PARAMS["duration_balance"] * duration_norm
        + energy_weight_adj * energy_norm
        - successor_boost
        + wait_score
        + uncertainty_amplifier
    )

    # Final safeguard: replace inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
