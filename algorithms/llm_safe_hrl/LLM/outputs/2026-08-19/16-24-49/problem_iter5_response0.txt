def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's stability and Parent 1's risk-aware gating:
    - Retains robust MAD normalization (Parent 2) for outlier resistance.
    - Keeps dual-gated critical-path boosting (slack ≥ 0 ∧ low uncertainty).
    - Introduces novel energy-slack coupling: energy penalty exponentiated by slack pressure.
    - Adds rank_uncertainty_amplifier: boosts critical path *under uncertainty* when slack is non-negative,
      improving resilience without violating DDL feasibility.
    - Replaces hard wait gating with smooth slack-feasibility weighting (continuous [0,1] gate).
    - All operations epsilon-guarded, finite, and deterministic.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs: ensure no NaN/inf; preserve sign semantics
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization with Gaussian-consistent scaling
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Slack transformation: piecewise-smooth risk-urgency curve ---
    neg_mask = slack < 0.0
    zeroish_mask = (slack >= 0.0) & (slack < 1.0)
    pos_mask = slack >= 1.0
    
    slack_norm = np.zeros_like(slack)
    slack_norm[neg_mask] = PARAMS["slack_risk_penalty"] * (-slack[neg_mask])
    slack_norm[zeroish_mask] = PARAMS["slack_urgency_scale"] * (np.exp(slack[zeroish_mask]) - 1.0)
    slack_norm[pos_mask] = np.clip(slack[pos_mask], 0.0, PARAMS["slack_cap"])

    # --- Duration: exec + comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_norm = mad_normalize(duration)

    # --- Energy: base normalized energy, coupled nonlinearly with slack pressure ---
    energy_norm = mad_normalize(min_incremental_energy)
    # Slack pressure: soft [0,1] gate from tanh, scaled by slack pressure magnitude
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    slack_pressure_gate = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])
    # Coupled energy term: stronger penalty for high energy when slack is tight
    energy_score = PARAMS["energy_efficiency_bias"] * energy_norm * (
        1.0 + slack_pressure_gate ** PARAMS["energy_slack_coupling_exponent"]
    )

    # --- Critical path: conditional leverage only if slack >= 0 AND uncertainty is low ---
    rank_norm = mad_normalize(upward_rank)
    unc_med = np.median(uncertainty)
    unc_gate = np.clip(1.0 - (uncertainty - unc_med) / (np.maximum(unc_med, eps)), 0.0, 1.0)
    critical_gate = ((slack >= 0.0) * unc_gate).astype(float)
    critical_boost = PARAMS["critical_path_leverage"] * rank_norm * critical_gate

    # --- Work density: remaining_work per unit duration, gated by slack feasibility ---
    work_density = np.divide(remaining_work, duration + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_gate = (slack >= 0.0).astype(float)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm * work_density_gate

    # --- Fairness: saturating exponential wait boost, weighted by continuous slack feasibility ---
    wait_norm = mad_normalize(ready_wait_time)
    wait_boost = 1.0 - np.exp(-PARAMS["wait_fairness_gain"] * np.maximum(wait_norm, 0.0))
    # Smooth slack feasibility gate: tanh-based [0,1] transition around slack=0
    slack_feasibility_gate = 0.5 * (1.0 + np.tanh((slack + eps) / (PARAMS["slack_pressure_tanh_scale"] + eps)))
    wait_score = wait_boost * slack_feasibility_gate

    # --- Uncertainty interaction: additive amplification under slack pressure ---
    unc_norm = mad_normalize(uncertainty)
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"])

    # --- Rank-uncertainty resilience: boost critical path when slack is feasible AND uncertainty exists ---
    # Encourages early scheduling of critical tasks even with moderate uncertainty — improves robustness
    resilience_gate = (slack >= 0.0).astype(float)
    resilience_boost = PARAMS["rank_uncertainty_amplifier"] * rank_norm * unc_norm * resilience_gate

    # --- Assemble final score: smaller = higher priority ---
    # Note: critical_boost, work_density_bonus, and resilience_boost are subtracted (beneficial)
    score = (
        slack_norm
        + PARAMS["duration_balance"] * duration_norm
        + energy_score
        - critical_boost
        - work_density_bonus
        + wait_score
        + uncertainty_amplifier
        - resilience_boost
    )

    # Final safeguard: replace inf/nan, ensure finite output with machine-precision bounds
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
