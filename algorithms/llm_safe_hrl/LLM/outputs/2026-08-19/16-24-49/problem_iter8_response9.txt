def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust slack bounds and percentile normalization with Parent 1's fairness and work-density awareness.
    
    Key structural innovations:
    - Retains Parent 2's empirically bounded slack clipping [slack_min_bound, slack_max_bound] for stability under DDL pressure.
    - Uses symmetric percentile normalization: [q_low, 2*median - q_low] (2 is allowed literal) — robust to outliers, no MAD dependency.
    - Integrates Parent 1's wait_fairness_gain as bounded exponential fairness boost, but gated by slack headroom to preserve DDL-first order.
    - Adds work_density_weight term: remaining_work / (exec+comm+eps), normalized and subtracted (favoring high-throughput critical paths).
    - All DDL-violation penalties dominate; fairness/throughput terms only activate where slack permits (no lexicographic violation).
    - No tanh, no fragile uncertainty gating: uses simple sigmoid on uncertainty for energy coupling, and strict slack<=0 for criticality.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Clipped percentile normalization: [q_low, 2*median - q_low] — uses allowed literal 2.0
    def percentile_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x)
        q_low = np.percentile(x, PARAMS["percentile_clip_low"])
        med = np.median(x)
        q_high = 2.0 * med - q_low
        denom = q_high - q_low + eps
        normalized = (x - q_low) / denom
        return np.clip(normalized, 0.0, 1.0)

    # --- Slack scoring: power penalty for violation only; zero otherwise ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )

    # --- Criticality boost: activated ONLY when slack <= 0 (DDL violation or tight) ---
    is_tight_or_violated = (slack <= 0)
    rank_median = np.median(upward_rank) if N > 1 else np.mean(upward_rank)
    is_high_rank = (upward_rank >= rank_median)
    critical_gate = np.where(
        is_high_rank & is_tight_or_violated,
        PARAMS["criticality_boost"],
        1.0
    )

    # --- Energy efficiency ratio: joules per second (lower = better), percentile-normalized ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = percentile_normalize(energy_per_sec)

    # --- Uncertainty-slack coupling: additive penalty for high uncertainty × deadline pressure ---
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]

    # --- Duration risk penalty: long duration × uncertainty ---
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Rank-slack balance: fixed empirical bounds for stability ---
    slack_lb = PARAMS["slack_min_bound"]
    slack_ub = PARAMS["slack_max_bound"]
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    # Weight toward rank when slack is poor (low slack_scaled)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -percentile_normalize(upward_rank) * weight_rank

    # --- Bounded sigmoid gate on uncertainty for energy-uncertainty interaction ---
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - 1.0)))
    energy_norm = percentile_normalize(min_incremental_energy)
    unc_norm = percentile_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Fairness boost: saturating exponential wait boost, but gated by slack headroom ---
    # Only apply fairness where slack allows: headroom = max(0, slack) / (|slack| + eps) → [0,1]
    slack_headroom = np.maximum(0.0, slack) / (np.abs(slack) + eps)
    wait_norm = percentile_normalize(ready_wait_time)
    wait_boost = (1.0 - np.exp(-PARAMS["wait_fairness_gain"] * np.maximum(0.0, wait_norm))) * slack_headroom

    # --- Work-density bonus: remaining_work / (exec+comm+eps), normalized and subtracted (favoring throughput) ---
    work_density = remaining_work / (duration_total + eps)
    work_density_norm = percentile_normalize(work_density)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm

    # --- Assemble final score: smaller = higher priority ---
    # Slack violation and coupling terms dominate; fairness and work density are beneficial (subtracted)
    score = (
        percentile_normalize(slack_score) +
        percentile_normalize(unc_slack_coupling) +
        percentile_normalize(duration_risk) +
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score +
        rank_score +
        energy_uncertainty_score +
        percentile_normalize(min_incremental_energy) * (1.0 - weight_rank) +
        wait_boost
    ) - work_density_bonus
    
    # Apply criticality boost multiplicatively (amplifies urgency for critical late tasks)
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
