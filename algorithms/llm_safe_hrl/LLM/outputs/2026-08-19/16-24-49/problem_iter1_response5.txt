def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """
    Novel priority rule: prioritizes deadline-criticality first via robust slack gating,
    couples uncertainty with slack to amplify risk awareness, uses exponential urgency for negative slack,
    applies conditional criticality boost, and penalizes high-energy + high-rank combinations.
    Anti-starvation uses soft exponential wait boost instead of linear scaling.
    All features are robustly normalized using mean-abs + epsilon; no unbounded ops.
    """
    eps = PARAMS["epsilon"]
    
    # Safe, finite, deterministic normalization: use mean absolute value + epsilon
    def robust_norm(x):
        x = np.asarray(x, dtype=np.float64)
        denom = np.mean(np.abs(x)) + eps
        return x / denom
    
    # Normalize base features
    norm_slack = robust_norm(slack)
    norm_energy = robust_norm(min_incremental_energy)
    norm_duration = robust_norm(min_exec_time + min_comm_time)
    norm_rank = robust_norm(upward_rank)
    norm_wait = robust_norm(ready_wait_time)
    norm_uncert = robust_norm(uncertainty)
    
    # === 1. Risk-urgent slack: exponential penalty for negative slack, bounded smoothness ===
    # Use softplus-like shape: exp(-norm_slack) for negative slack → large positive score (bad priority)
    # But we want *smaller* score = higher priority → so flip sign and clip extreme growth
    # Instead: define 'urgency' = -norm_slack if slack < 0, else 0; then apply exponent
    slack_sign_mask = (slack < 0).astype(float)
    urgency_base = -norm_slack * slack_sign_mask
    # Clamp to avoid overflow: max exponent ~4.0 (since exp(4)=54.6, exp(8)=2980 → too volatile)
    urgency_clipped = np.clip(urgency_base, 0.0, 4.0)
    urgency_penalty = np.exp(PARAMS["slack_penalty_exponent"] * urgency_clipped)
    
    # === 2. Criticality boost: only activate for top-half upward_rank tasks ===
    rank_median = np.median(norm_rank)
    critical_mask = (norm_rank >= rank_median).astype(float)
    critical_boost = PARAMS["criticality_boost"] * norm_rank * critical_mask
    
    # === 3. Uncertainty-modulated slack pressure: stronger penalty when both slack is tight AND uncertainty high ===
    # Use product of normalized slack pressure (max(0,-norm_slack)) and normalized uncertainty
    slack_pressure = np.maximum(0.0, -norm_slack)
    uncertainty_coupled_pressure = PARAMS["uncertainty_slack_coupling"] * slack_pressure * norm_uncert
    
    # === 4. Anti-starvation: soft exponential boost for long-waiting tasks (not linear)
    # Wait boost = 1 - exp(-wait_decay * norm_wait), increasing smoothly from 0 to 1
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * norm_wait)
    
    # === 5. Energy-rank interaction penalty: discourage scheduling high-energy tasks that are also on critical path
    # This avoids overloading bottleneck VMs with expensive + critical work
    rank_energy_penalty = PARAMS["rank_energy_interaction"] * norm_energy * norm_rank
    
    # === Final score: sum of all components — smaller = better ===
    # Base terms (all contribute positively to score → hurt priority when large):
    #   - urgency_penalty (↑ when slack negative → ↑ score → ↓ priority → CORRECT: punish delay risk)
    #   - uncertainty_coupled_pressure (↑ when uncertain & tight → ↑ score)
    #   - rank_energy_penalty (↑ when costly + critical → ↑ score)
    # Boost terms (subtract from score → improve priority when active):
    #   - critical_boost (↑ for critical tasks → ↓ score → ↑ priority)
    #   - wait_boost (↑ for waiting tasks → ↓ score → ↑ priority)
    # Neutral weighted terms:
    #   + energy_sensitivity * norm_energy
    #   + duration_balance * norm_duration
    
    score = (
        urgency_penalty
        + uncertainty_coupled_pressure
        + rank_energy_penalty
        + PARAMS["energy_sensitivity"] * norm_energy
        + PARAMS["duration_balance"] * norm_duration
        - critical_boost
        - wait_boost
    )
    
    # Ensure finite output: replace NaN/inf with safe large finite value (preserves argmin ordering)
    score = np.nan_to_num(score, nan=np.finfo(np.float64).max, posinf=np.finfo(np.float64).max, neginf=-np.finfo(np.float64).max)
    
    return score
