def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule integrating successor-release interaction and host-load conditional gating.
    
    Key structural mutations:
    - Replaces piecewise slack transformation with smooth tanh-based urgency curve for unified risk modeling.
    - Introduces 'successor_release_interaction': upward_rank × remaining_work, gated by slack feasibility
      and uncertainty threshold (to unblock critical paths only when risk allows).
    - Adds 'host_load_conditional_gate': energy term scaled by fairness headroom (wait_time / (|slack|+eps)),
      enabling energy savings only where deadlines permit—no unconditional energy minimization.
    - Removes duration_norm from main score; instead uses duration-aware work density and load-aware energy.
    - Uses saturating exponential wait boost (1 - exp(-gain * norm_wait)) for bounded fairness.
    - All interactions respect DDL-first lexicographic ordering: no energy or fairness terms dominate slack risk.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust MAD normalization with Gaussian-consistent scaling
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_factor"] * (mad if mad > eps else eps)
        return (x - med) / scale

    # --- Smooth slack-pressure gating: unified tanh-based urgency ---
    # Maps slack → [0,1] where 0 = max urgency (negative/zero slack), 1 = low urgency (large positive slack)
    slack_centered = slack - np.median(slack)
    slack_pressure = np.tanh(PARAMS["slack_pressure_tanh_scale"] * (-slack_centered + eps))
    # Convert to urgency penalty: higher penalty for high pressure (low slack)
    slack_penalty = PARAMS["slack_risk_penalty"] * (1.0 - slack_pressure) + PARAMS["slack_urgency_scale"] * slack_pressure

    # --- Successor-release interaction: critical path unlocking ---
    # Only activate when slack >= 0 AND uncertainty <= median (risk-controlled release)
    unc_med = np.median(uncertainty)
    successor_gate = ((slack >= 0.0) & (uncertainty <= unc_med + eps)).astype(float)
    successor_release = upward_rank * remaining_work
    successor_norm = mad_normalize(successor_release)
    successor_bonus = PARAMS["critical_path_leverage"] * successor_norm * successor_gate

    # --- Host-load conditional gate for energy: enable savings only where fairness headroom exists ---
    # Fairness headroom = wait_time / (|slack| + eps); large headroom → safe to optimize energy
    fairness_headroom = ready_wait_time / (np.abs(slack) + eps)
    energy_headroom_gate = np.clip(fairness_headroom, 0.0, 1.0)  # bounded [0,1]
    energy_norm = mad_normalize(min_incremental_energy)
    energy_score = PARAMS["energy_efficiency_bias"] * energy_norm * energy_headroom_gate

    # --- Bounded fairness: saturating exponential wait boost ---
    wait_norm = mad_normalize(ready_wait_time)
    wait_boost = 1.0 - np.exp(-PARAMS["wait_fairness_gain"] * np.maximum(0.0, wait_norm))

    # --- Uncertainty-modulated urgency amplification ---
    # Amplify slack penalty where uncertainty is high, but only under deadline pressure
    unc_norm = mad_normalize(uncertainty)
    uncertainty_amplifier = PARAMS["uncertainty_sensitivity"] * unc_norm * slack_pressure

    # --- Work-density bonus: remaining_work / (exec_time + comm_time + eps), robustly normalized ---
    duration = min_exec_time + min_comm_time
    work_density = remaining_work / (duration + eps)
    work_density_norm = mad_normalize(work_density)
    work_density_bonus = PARAMS["work_density_weight"] * work_density_norm

    # --- Assemble final score: smaller = higher priority ---
    # Critical-path unlocking and work density are subtracted (beneficial); others added (penalties/costs)
    score = (
        slack_penalty
        + energy_score
        + wait_boost
        + uncertainty_amplifier
        - successor_bonus
        - work_density_bonus
    )

    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
