def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule emphasizing conditional successor-release interaction and host-load-aware energy gating.
    
    Key structural mutations:
    - Replaces unconditional work-density bonus with *successor-release interaction*: 
      upward_rank × remaining_work, gated by slack feasibility AND low uncertainty to unblock critical paths early.
    - Introduces *host-load conditional gate* on energy: energy contribution suppressed only when both slack > 0 AND 
      ready_wait_time is low (i.e., no fairness pressure), enabling energy savings only where DDL headroom exists.
    - Uses saturating exponential wait boost: 1 - exp(-gain * norm_wait) → bounded, monotonic, robust fairness.
    - Normalizes all features via MAD but applies *robust range scaling* (IQR-based fallback) for zero-MAD safety.
    - Combines slack pressure and uncertainty into a unified *risk-saturation gate*: tanh(slack_pressure + uncertainty_scaled).
    - Removes direct duration_norm term; instead embeds duration influence implicitly via work_density and comm-aware slack headroom.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Cast and sanitize inputs
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    # Robust feature normalization: MAD primary, IQR fallback for degenerate cases
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        if mad < eps:
            q1, q3 = np.quantile(x, [0.25, 0.75])
            iqr = q3 - q1
            scale = max(iqr, eps) if iqr > 0 else eps
        else:
            scale = mad
        # Scale by robustness_mad_factor only when using MAD path
        scale = PARAMS["robustness_mad_factor"] * scale if mad >= eps else scale
        return (x - med) / (scale + eps)

    # --- Slack transformation: smooth, bounded, and risk-aware ---
    # Negative slack → strong linear penalty; positive slack → capped linear decay + saturation
    slack_norm = np.where(
        slack < 0.0,
        PARAMS["slack_risk_penalty"] * (-slack),
        np.clip(slack, 0.0, PARAMS["slack_cap"])
    )

    # --- Energy: host-load conditional gating (suppress energy cost only when slack > 0 AND wait time is low) ---
    energy_norm = robust_normalize(min_incremental_energy)
    wait_headroom = np.maximum(1.0, slack + 1.0)
    wait_normalized = robust_normalize(ready_wait_time)
    # Gate: enable energy optimization only when deadline headroom exists AND task isn't starved
    energy_gate = (slack > 0.0) & (wait_normalized < 0.0)  # wait_normalized < 0 means below median wait → not starved
    energy_weight_adj = PARAMS["energy_efficiency_bias"] * np.where(energy_gate, 1.0, 0.0)

    # --- Successor-release interaction: upward_rank × remaining_work, gated by slack feasibility AND low uncertainty ---
    # This directly addresses 'critical_path_or_successor_blocked' evidence
    successor_interaction = upward_rank * remaining_work
    successor_norm = robust_normalize(successor_interaction)
    # Gate: only activate when slack permits (>=0) AND uncertainty is below median (reliable estimate)
    unc_med = np.median(uncertainty)
    successor_gate = (slack >= 0.0) & (uncertainty <= unc_med + eps)
    successor_bonus = PARAMS["critical_path_leverage"] * successor_norm * successor_gate.astype(float)

    # --- Fairness: saturating exponential wait boost (bounded, smooth, avoids infinite growth) ---
    # Boost priority for long-waiting tasks, but asymptotically capped at 1.0
    wait_score = 1.0 - np.exp(-PARAMS["wait_fairness_gain"] * (ready_wait_time / (wait_headroom + eps) + eps))

    # --- Unified risk-saturation gate: combines slack pressure and uncertainty ---
    slack_pressure = np.clip(-slack, 0.0, np.inf)
    unc_norm = robust_normalize(uncertainty)
    risk_saturation = np.tanh(slack_pressure * PARAMS["slack_pressure_tanh_scale"] + unc_norm * PARAMS["uncertainty_sensitivity"])

    # --- Assemble final score: smaller = higher priority ---
    # Note: successor_bonus is subtracted because higher interaction implies greater critical-path impact
    score = (
        slack_norm  # primary DDL risk signal
        + energy_weight_adj * energy_norm
        - successor_bonus
        + wait_score
        + risk_saturation
    )

    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
