def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric constants declared; no hidden literals.
    
    Key features:
    - Slack gating: strong linear penalty for negative slack, log-scaled soft growth for positive slack
    - Critical-path boost only activates above rank threshold AND under deadline pressure
    - Energy weight dynamically scaled by slack pressure using declared coupling factor
    - Anti-starvation via saturating exponential decay on wait time
    - Uncertainty penalized only under tight slack, coupled with duration
    - All normalizations use robust median/IQR; all divisions guarded
    """
    epsilon = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(epsilon, finfo.tiny)

    # Robust normalization: median + IQR using declared percentiles
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        q75 = np.percentile(x, PARAMS["iqr_percentile_q75"], method='midpoint')
        q25 = np.percentile(x, PARAMS["iqr_percentile_q25"], method='midpoint')
        iqr = q75 - q25
        scale = iqr if iqr > eps_safe else 1.0
        return (x - med) / scale

    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # --- Slack-driven risk gating ---
    # Strong penalty for negative slack; bounded log growth for positive slack
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_penalty_factor"] * (-slack) / (np.abs(slack) + eps_safe),
        PARAMS["positive_slack_log_scale"] * np.log1p(slack)
    )
    
    # Critical slack threshold gate: enables rank boost only when slack is tight
    slack_tight_mask = (slack <= PARAMS["critical_slack_threshold"]).astype(float)
    
    # --- Dynamic energy weighting: more sensitive when deadlines are at risk ---
    energy_weight_dynamic = (
        PARAMS["energy_sensitivity"] 
        * (1 + PARAMS["energy_slack_coupling_factor"] * np.maximum(-norm_slack, 0.0))
    )

    # --- Critical-path boost: only apply when both slack is tight AND rank is high ---
    rank_boost = np.where(
        (upward_rank >= PARAMS["rank_urgency_gate"] * (np.median(upward_rank) + eps_safe)) & (slack_tight_mask > 0),
        -PARAMS["rank_boost_coeff"] * norm_rank,
        np.zeros_like(norm_rank)
    )

    # --- Anti-starvation: saturating exponential decay on wait time ---
    wait_priority = -(
        (1 - np.exp(-PARAMS["wait_decay_rate"] * (ready_wait_time + eps_safe))) 
        * norm_wait
    )

    # --- Uncertainty-robustness coupling: penalize high uncertainty only under deadline pressure ---
    uncertainty_penalty = (
        PARAMS["robustness_bias"] 
        * norm_uncert 
        * np.maximum(-norm_slack, 0.0)
        * slack_tight_mask
    )

    # --- Duration-uncertainty coupling: longer durations amplify uncertainty impact ---
    dur_uncert_interaction = (
        PARAMS["duration_uncertainty_coupling"] 
        * norm_duration 
        * norm_uncert
    )

    # Composite score: smaller = higher priority
    score = (
        slack_penalty
        + energy_weight_dynamic * norm_energy
        + PARAMS["duration_weight_base"] * norm_duration
        + rank_boost
        + PARAMS["work_weight_base"] * norm_work
        + wait_priority
        + uncertainty_penalty
        + dur_uncert_interaction
    )

    # Final safeguard: ensure finite, bounded output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
