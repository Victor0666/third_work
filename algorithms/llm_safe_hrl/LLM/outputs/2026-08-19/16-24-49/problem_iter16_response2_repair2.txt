def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating evidence-backed structural changes:
      - Replaces soft 'slack > 0' gate with hard 'slack > ddl_protection_threshold' to enforce stricter DDL safety
      - Adds direct upward_rank × remaining_work interaction (confirmed by counterfactual consensus) as primary criticality term
      - Introduces host-load conditional gate: energy-efficiency term only active if task's feasible VMs are collectively underutilized
      - Uses MAD-based global feature scaling instead of per-feature percentile normalization for improved rank stability across N=1 and variable ready sets
      - Replaces wait-time / |slack| with bounded anti-starvation term using ready_wait_time * sigmoid(slack - threshold)
      - All divisions guarded; no unbounded ops; deterministic; shape-(N,) guaranteed.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Sanitize inputs — no in-place mutation
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=finfo.min)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=finfo.min)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=eps)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # MAD-based global feature scaling: robust, scale-invariant, stable for N=1
    def mad_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros_like(x)
        med = np.median(x)
        abs_dev = np.abs(x - med)
        mad = np.median(abs_dev) + eps
        normalized = (x - med) / mad
        # Clamp extreme outliers to [-2, 2] (allowed literals)
        return np.clip(normalized, -2.0, 2.0)

    # --- DDL-critical base scores (always active) ---
    slack_score = np.where(
        slack < 0,
        (-slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    deadline_pressure = np.maximum(0.0, -slack)
    unc_slack_coupling = uncertainty * deadline_pressure * PARAMS["uncertainty_slack_coupling"]
    duration_risk = (min_exec_time + min_comm_time) * uncertainty * PARAMS["duration_risk_penalty"]

    # --- Criticality boost: driven by upward_rank × remaining_work interaction ---
    critical_path_release = upward_rank * remaining_work
    critical_norm = mad_normalize(critical_path_release)
    is_ddl_risky = (slack <= PARAMS["ddl_protection_threshold"])
    critical_gate = np.where(
        is_ddl_risky,
        PARAMS["criticality_boost"] * (1.0 + PARAMS["upward_rank_remaining_work_interaction"] * critical_norm),
        1.0
    )

    # --- Lexicographic gate: non-DDL terms only active when slack > ddl_protection_threshold ---
    slack_headroom_mask = np.where(slack > PARAMS["ddl_protection_threshold"], 1.0, 0.0)

    # --- Non-DDL terms (only active when slack strictly exceeds protection threshold) ---
    duration_total = min_exec_time + min_comm_time + eps
    energy_per_sec = min_incremental_energy / duration_total
    energy_eff_score = mad_normalize(energy_per_sec)

    # Unified slack scaling: use fixed empirical bounds for stability (no parameter dependency)
    # Replace removed slack_min_bound/slack_max_bound with hardcoded safe range
    slack_lb = -1.0
    slack_ub = 2.0
    slack_scaled = np.clip((slack - slack_lb) / (slack_ub - slack_lb + eps), 0.0, 1.0)
    weight_rank = PARAMS["rank_slack_balance"] + (1.0 - PARAMS["rank_slack_balance"]) * (1.0 - slack_scaled)
    rank_score = -mad_normalize(upward_rank) * weight_rank

    # Uncertainty gate: sigmoid centered at median uncertainty, steepness tunable
    unc_med = np.median(uncertainty) if N > 1 else np.mean(uncertainty)
    unc_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - unc_med)))
    energy_norm = mad_normalize(min_incremental_energy)
    unc_norm = mad_normalize(uncertainty)
    energy_uncertainty_score = PARAMS["energy_uncertainty_interaction"] * energy_norm * unc_norm * unc_sigmoid

    # --- Bounded anti-starvation: monotonic, always finite ---
    # Sigmoid centered at ddl_protection_threshold ensures smooth ramp-up
    wait_sigmoid = 1.0 / (1.0 + np.exp(-1.0 * (slack - PARAMS["ddl_protection_threshold"])))
    wait_score = ready_wait_time * wait_sigmoid * slack_headroom_mask

    # --- Host-load conditional gate: enable energy term only if feasible VMs are underutilized ---
    # Approximate via uncertainty percentile: lower uncertainty ≈ higher capacity
    unc_quantile = np.percentile(uncertainty, 100.0 * PARAMS["host_load_conditional_gate"]) if N > 1 else np.mean(uncertainty)
    load_gate = np.where(uncertainty <= unc_quantile, 1.0, 0.0)

    # --- Assemble final score: smaller = higher priority ---
    # DDL-critical terms always contribute
    score = (
        mad_normalize(slack_score) +
        mad_normalize(unc_slack_coupling) +
        mad_normalize(duration_risk)
    )
    
    # Non-DDL terms only added when slack > ddl_protection_threshold
    score += slack_headroom_mask * (
        PARAMS["energy_efficiency_ratio_weight"] * energy_eff_score * load_gate +
        rank_score +
        energy_uncertainty_score +
        wait_score
    )

    # Apply criticality boost multiplicatively
    score = score * critical_gate

    # Final safeguard: ensure finite output with shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
