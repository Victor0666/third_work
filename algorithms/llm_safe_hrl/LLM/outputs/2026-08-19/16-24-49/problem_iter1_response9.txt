def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: piecewise deadline-pressure gating + risk-aware duration-energy coupling.
    
    Key innovations:
      - Critical-mode activation below `critical_slack_threshold`: switches to slack-driven urgency + 
        uncertainty-amplified penalty (not linear), avoiding infinite penalty at extreme negative slack.
      - Upward rank preserved *only* when slack >= 0 — prevents over-prioritizing critical-path tasks that 
        are already late and better handled via fast-execution or low-energy VMs.
      - Anti-starvation term saturates at `max_wait_cap` to avoid dominance over deadline/energy signals.
      - Duration-risk interaction: uncertainty scales the *normalized sum* of exec+comm time, capturing 
        high-risk long-duration tasks that threaten both deadline and energy stability.
      - All normalizations use mean-abs with epsilon guard; no division by std or unstable stats.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Safe, robust normalization: avoids NaN/inf, preserves sign, handles singleton
    def robust_normalize(x):
        x = np.asarray(x, dtype=np.float64)
        x_abs_mean = np.mean(np.abs(x)) + eps
        return x / x_abs_mean
    
    # Normalize base features
    norm_exec_comm = robust_normalize(min_exec_time + min_comm_time)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_upward_rank = robust_normalize(upward_rank)
    norm_remaining_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(np.clip(ready_wait_time, 0, PARAMS["max_wait_cap"]))
    
    # --- Piecewise slack handling: critical mode with bounded penalty ---
    # Critical region: slack < threshold → activate steep, uncertainty-gated penalty
    is_critical = (slack < PARAMS["critical_slack_threshold"])
    # Use smooth but bounded negative slack penalty: tanh-based compression avoids explosion
    # tanh(-x) ≈ -1 for large x, but finite and differentiable; scaled by factor and gate
    raw_negative_slack = np.where(is_critical, -slack, 0.0)
    # Compress large negatives: tanh(raw_negative_slack * 0.5) keeps range in [-1, 0]
    compressed_slack_penalty = np.tanh(raw_negative_slack * 0.5) * (-1.0)
    # Apply uncertainty-modulated gain only in critical region
    critical_penalty = (
        PARAMS["slack_penalty_factor"] * compressed_slack_penalty *
        (1.0 + PARAMS["uncertainty_slack_gate"] * robust_normalize(uncertainty))
    )
    
    # Non-critical region: prioritize *positive* slack (more buffer → lower urgency), 
    # but preserve upward_rank only here to guide critical-path progress under feasibility
    norm_slack = robust_normalize(slack)
    non_critical_urgency = -PARAMS["rank_preservation_weight"] * norm_upward_rank * (slack >= PARAMS["critical_slack_threshold"])
    
    # --- Risk-aware duration-energy coupling ---
    # High uncertainty amplifies duration impact → penalize long exec+comm more when uncertain
    duration_risk_score = norm_exec_comm * (1.0 + PARAMS["duration_risk_interaction"] * robust_normalize(uncertainty))
    
    # --- Anti-starvation: linear capped wait boost (adds priority, i.e., reduces score)
    wait_boost = -PARAMS["wait_anti_starve_gain"] * norm_wait
    
    # --- Final score: smaller = higher priority ---
    # In critical mode: penalty dominates (large positive score → low priority unless mitigated)
    # In non-critical: balance slack buffer, rank, energy, and duration
    score = (
        # Base urgency terms
        critical_penalty +
        non_critical_urgency +
        # Energy efficiency signal (always desirable: lower energy → lower score)
        PARAMS["energy_sensitivity"] * norm_energy +
        # Duration-risk coupling (long + uncertain → higher score → lower priority)
        duration_risk_score +
        # Anti-starvation (reduces score for long-waiting tasks)
        wait_boost
    )
    
    # Ensure finite output and shape compliance
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.astype(np.float64)
