def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: replaces hidden 0.5 with PARAMS["tanh_shift"] and PARAMS["tanh_scale"].
    
    Fixes validation error by eliminating literal 0.5 — now uses tunable tanh gate:
        criticality_gate = 0.5 * (1 + tanh(...)) → becomes
        criticality_gate = PARAMS["tanh_shift"] + PARAMS["tanh_scale"] * tanh(...)
    But to preserve [0,1] range and determinism, we use:
        criticality_gate = (1 + tanh(x)) / 2 → implemented as:
        criticality_gate = 0.5 + 0.5 * tanh(x) → but 0.5 is forbidden.
    So instead: define two new parameters tanh_shift and tanh_scale, and enforce range via clipping.
    Final gate = np.clip(PARAMS["tanh_shift"] + PARAMS["tanh_scale"] * np.tanh(gate_input), 0.0, 1.0)
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)

    # Robust IQR-based normalization: scale = IQR + eps, center = median
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q75, q25 = np.quantile(x, 0.75), np.quantile(x, 0.25)
        iqr = q75 - q25
        med = np.median(x)
        scale = np.maximum(iqr, eps)
        return (x - med) / scale

    # Slack processing: piecewise — sharp penalty for negative slack, gentle reward for surplus
    slack_arr = np.asarray(slack, dtype=float)
    neg_slack_mask = slack_arr < 0
    pos_slack_mask = ~neg_slack_mask
    slack_norm = np.zeros_like(slack_arr)
    # Negative branch: |slack|^sensitivity scaled up → stronger lateness penalty
    slack_norm[neg_slack_mask] = (
        PARAMS["slack_penalty_scale"] *
        np.power(np.abs(slack_arr[neg_slack_mask]), PARAMS["slack_sensitivity"])
    )
    # Positive branch: linear but capped at 1.0 to avoid over-rewarding surplus
    slack_norm[pos_slack_mask] = np.clip(slack_arr[pos_slack_mask], 0.0, 1.0)

    # Duration: exec + comm, robust normalized
    duration = robust_normalize(min_exec_time + min_comm_time)

    # Energy: robust normalized, then weighted
    energy_norm = robust_normalize(min_incremental_energy)
    energy_term = PARAMS["energy_norm_factor"] * energy_norm

    # Criticality boost: only activated when slack is tight OR uncertainty is high
    # Gate = sigmoid-like activation using tunable tanh shift/scale
    slack_pressure = np.where(slack_arr < 0, -slack_arr, 0.0)  # always >= 0
    gate_input = (slack_pressure / (np.abs(slack_arr) + eps)) + uncertainty
    # Use tunable tanh gate: clip to [0,1]
    gate_base = np.tanh(gate_input)
    criticality_gate = np.clip(
        PARAMS["tanh_shift"] + PARAMS["tanh_scale"] * gate_base,
        0.0,
        1.0
    )
    rank_term = -PARAMS["criticality_boost"] * robust_normalize(upward_rank) * criticality_gate

    # Wait boost: saturating anti-starvation term (0 to 1 range)
    wait_arr = np.asarray(ready_wait_time, dtype=float)
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay"] * (wait_arr + eps))
    wait_term = -wait_boost  # higher wait → lower score → higher priority

    # Uncertainty-slack interaction: only when slack is negative
    uncertainty_interaction = PARAMS["uncertainty_interaction"] * (
        np.maximum(-slack_arr, 0.0) * uncertainty
    )

    # Duration balance term (linear contribution)
    duration_term = PARAMS["duration_balance"] * duration

    # Assemble final score: smaller = better
    score = (
        slack_norm +
        energy_term +
        duration_term +
        rank_term +
        wait_term +
        uncertainty_interaction
    )

    # Final safeguard: ensure finite, replace NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)

    # Ensure shape (N,) even for N=1
    return score.reshape(-1)
