def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing risk-aware urgency, critical path amplification under deadline pressure,
    robust duration-energy coupling, and bounded starvation relief — all with stable normalization."""
    eps = PARAMS["epsilon"]
    
    # Robust per-feature normalization: use median absolute deviation for outlier resilience
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        spread = np.median(np.abs(x - center)) + eps
        return (x - center) / spread

    # Normalize core features independently
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # --- Risk-aware slack penalty: strong nonlinearity for negative slack ---
    # Apply exponent only to negative slack; positive slack gets diminishing reward (soft cap)
    slack_sign = np.sign(slack)
    slack_abs = np.abs(slack)
    # Use smooth piecewise: penalty = |slack|^p for slack < 0, reward = -|slack|/(1+|slack|) for slack >= 0
    negative_penalty = np.where(slack < 0, 
                                np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"]), 
                                0.0)
    positive_relief = np.where(slack >= 0, 
                               -slack_abs / (1.0 + slack_abs + eps), 
                               0.0)
    slack_risk_score = negative_penalty + positive_relief

    # --- Criticality amplification: boost rank only when slack is critically low ---
    # Gate: activate only when normalized slack < -0.5 (i.e., highly urgent)
    critical_gate = np.where(norm_slack < -0.5, 1.0, 0.0)
    amplified_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * critical_gate)

    # --- Duration-uncertainty coupling: penalize high uncertainty *only* when slack is tight ---
    # Bounded gate: active when norm_slack < -0.3 AND norm_uncert > 0.2
    dur_uncert_gate = np.where((norm_slack < -0.3) & (norm_uncert > 0.2), 1.0, 0.0)
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Bounded starvation relief: saturating benefit from wait time ---
    # Soft saturation: wait_benefit = 1 - exp(-wait_decay * norm_wait)
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (np.maximum(norm_wait, 0.0) + eps))

    # --- Work saturation: avoid over-penalizing huge workloads via soft log-like scaling ---
    # Map remaining_work into [0,1] via sigmoid on log-scale reference
    work_ref = PARAMS["work_saturation_point"]
    work_ratio = np.clip(remaining_work / (work_ref + eps), 1e-6, 1e6)
    work_scaled = 1.0 / (1.0 + 1.0 / (work_ratio + eps))

    # --- Assemble final score: smaller = better ---
    # Primary drivers: slack risk (penalty), energy (minimize), duration-uncert (penalty)
    # Secondary: suppress rank unless urgent, suppress work unless small, reward wait (anti-starve)
    score = (
        slack_risk_score                              # Dominant urgency signal: high penalty if late
        + PARAMS["energy_sensitivity"] * norm_energy  # Minimize marginal fuzzy energy
        + dur_uncert_interaction                      # Penalize uncertain long tasks under pressure
        - amplified_rank                              # Favor rank only when needed — avoids blind criticality bias
        + work_scaled                                 # Small work → higher priority; large work → capped relief
        - wait_benefit                                # Anti-starvation: longer wait → higher priority (bounded)
    )

    # Ensure finite output, no NaN/infs, shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
