def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces additive feasibility guard with conditional DDL-protection gate,
    introduces upward_rank × remaining_work interaction as primary criticality signal,
    removes redundant work suppression and starvation arctan, replaces with linear wait benefit,
    and uses sigmoid-based slack gating for smoother risk pressure.

    Structural changes:
    - Replaces hard feasibility_guard with soft `ddl_protection_gate` that modulates *all* terms
      when slack < slack_negative_threshold AND uncertainty > uncertainty_activation_threshold.
    - Uses upward_rank × remaining_work (not just upward_rank) as core criticality signal,
      scaled by criticality_scale only under DDL pressure.
    - Removes work_suppression: empirical evidence shows no benefit; instead uses raw norm_work
      as a weak anti-criticality term only when work is large *and* slack is positive.
    - Replaces arctan wait_benefit with bounded linear scaling (simpler, more interpretable).
    - Applies sigmoid gating to slack_risk_score to ensure monotonic, bounded pressure.
    - All normalizations use mean-abs scaling (more robust than median/MAD for small N).
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Robust per-feature normalization using mean absolute deviation (MAD) with fallback
    def mean_abs_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.mean(x)
        spread = np.mean(np.abs(x - center)) + eps
        return (x - center) / (spread + eps)

    # Normalize all features stably
    norm_slack = mean_abs_normalize(slack)
    norm_energy = mean_abs_normalize(min_incremental_energy)
    norm_duration = mean_abs_normalize(min_exec_time + min_comm_time)
    norm_rank = mean_abs_normalize(upward_rank)
    norm_work = mean_abs_normalize(remaining_work)
    norm_wait = mean_abs_normalize(ready_wait_time)
    norm_uncert = mean_abs_normalize(uncertainty)

    # --- Slack risk score: smooth, bounded, monotonic via sigmoid-modulated power penalty
    # Use raw slack for penalty (not normalized) to preserve physical meaning and avoid distortion
    slack_abs = np.abs(slack)
    negative_penalty = np.where(
        slack < 0,
        np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"]),
        0.0
    )
    # Sigmoid gate ensures gradual ramp-up of urgency near zero slack
    slack_gate = 1.0 / (1.0 + np.exp(-(norm_slack - PARAMS["slack_negative_threshold"]) / eps))
    slack_risk_score = negative_penalty * slack_gate

    # --- Critical path leverage: upward_rank × remaining_work interaction, activated only under DDL stress
    # This captures both importance and downstream workload — stronger signal than rank alone
    ddl_pressure_gate = np.where(
        (norm_slack < PARAMS["slack_negative_threshold"]) &
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]),
        1.0, 0.0
    )
    critical_leverage = norm_rank * norm_work * ddl_pressure_gate * PARAMS["criticality_scale"]

    # --- Duration-uncertainty coupling: same condition as before, but applied directly to duration
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) & 
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]), 
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Bounded starvation relief: linear scaling up to saturation time, then flat
    wait_benefit = np.clip(ready_wait_time / (PARAMS["wait_saturation_time"] + eps), 0.0, 1.0)

    # --- Work-based moderation: only suppress when slack is *positive* AND work is large
    # Avoids penalizing high-work tasks that are already urgent (negative slack)
    work_moderation = np.where(
        (slack >= 0) & (remaining_work > PARAMS["work_leverage_threshold"]),
        norm_work * 0.5,
        0.0
    )

    # --- Final score: smaller = higher priority
    # DDL risk dominates (slack_risk_score), then critical leverage boosts urgency,
    # then energy minimization and uncertainty penalties follow.
    # Wait benefit reduces score (improves priority); work moderation increases it (reduces priority).
    score = (
        slack_risk_score                              # Primary deadline risk driver
        - critical_leverage                           # Boost for high-leverage critical-path tasks
        + PARAMS["energy_sensitivity"] * norm_energy  # Energy cost (secondary objective)
        + dur_uncert_interaction                      # Uncertainty-aware delay penalty
        - wait_benefit                                # Anti-starvation boost
        + work_moderation                             # Suppress non-urgent large-work tasks
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
