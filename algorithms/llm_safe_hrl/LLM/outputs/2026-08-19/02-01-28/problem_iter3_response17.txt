def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with hard DDL protection and piecewise urgency gating.
    
    Key structural improvements:
    - Replaces sigmoid urgency with bounded piecewise-linear gate: zero penalty above threshold,
      linear ramp below — enforces crisp deadline boundary without gradient erosion.
    - Introduces conditional DDL protection: when slack < 0, energy/duration penalties are scaled
      by PARAMS["ddl_protection_factor"] to strictly prioritize deadline compliance over energy.
    - Retains robust rank-work interaction and uncertainty-slack coupling for critical-path focus.
    - Uses MAD-based normalization for outlier resilience; all operations guarded against NaN/inf.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Defensive copy & cast
    min_exec_time = np.asarray(min_exec_time, dtype=float).copy()
    min_comm_time = np.asarray(min_comm_time, dtype=float).copy()
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float).copy()
    slack = np.asarray(slack, dtype=float).copy()
    upward_rank = np.asarray(upward_rank, dtype=float).copy()
    remaining_work = np.asarray(remaining_work, dtype=float).copy()
    ready_wait_time = np.asarray(ready_wait_time, dtype=float).copy()
    uncertainty = np.asarray(uncertainty, dtype=float).copy()

    # Robust normalization: median + MAD
    def normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        dev = np.median(np.abs(x - center))
        scale = np.maximum(dev, eps)
        return (x - center) / scale

    norm_slack = normalize(slack)
    norm_energy = normalize(min_incremental_energy)
    norm_rank = normalize(upward_rank)
    norm_work = normalize(remaining_work)
    norm_wait = normalize(ready_wait_time)
    norm_uncert = normalize(uncertainty)
    duration = min_exec_time + min_comm_time
    norm_duration = normalize(duration)

    # --- Piecewise-linear urgency gate: zero above threshold, linear ramp below ---
    # Enforces hard boundary: no urgency penalty when slack > threshold → avoids false pressure
    slack_urgency = np.where(
        norm_slack <= PARAMS["critical_slack_threshold"],
        PARAMS["slack_risk_penalty"] * (PARAMS["critical_slack_threshold"] - norm_slack),
        0.0
    )

    # --- Uncertainty-slack coupling gate: soft saturation via tanh ---
    risk_gate = np.tanh(norm_uncert * slack_urgency)

    # --- Conditional DDL protection: suppress energy/duration penalties under violation ---
    ddl_violated = (slack < 0.0).astype(float)
    protection_scale = (1.0 - ddl_violated) + ddl_violated * PARAMS["ddl_protection_factor"]

    # --- Amplified energy penalty: base energy × protection scale + uncertainty-coupled marginal increase ---
    energy_penalty = protection_scale * norm_energy * (1.0 + PARAMS["uncertainty_slack_interaction"] * risk_gate)

    # --- Amplified duration penalty: same protection logic ---
    duration_penalty = protection_scale * norm_duration * (1.0 + PARAMS["duration_coupling"] * risk_gate)

    # --- Critical-path workload interaction: rank * work → bottleneck impact magnitude ---
    rank_work_interaction = norm_rank * norm_work * PARAMS["rank_work_interaction_weight"]

    # --- Anti-starvation boost: subtracted (lowers score) for long-waiting tasks ---
    wait_boost = PARAMS["wait_boost_weight"] * np.sqrt(np.maximum(norm_wait, eps))

    # --- Final priority score: smaller = higher priority ---
    # DDL urgency dominates; energy/duration suppressed only under violation; fairness preserved
    score = (
        slack_urgency
        + PARAMS["energy_sensitivity"] * energy_penalty
        + duration_penalty
        + rank_work_interaction
        - wait_boost
    )

    # Final numerical safeguards
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    score = np.clip(score, PARAMS["clip_lower_bound_log_scale"], PARAMS["clip_upper_bound"])

    # Enforce shape (N,)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"

    return score
