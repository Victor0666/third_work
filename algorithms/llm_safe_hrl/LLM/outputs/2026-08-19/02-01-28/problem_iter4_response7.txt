def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces hard feasibility guard with *conditional energy suppression*,
    promotes upward_rank × remaining_work as primary critical-path signal,
    introduces sigmoid-based slack gating for energy sensitivity,
    and uses mean-abs scaling for robust cross-feature mixing.
    
    Structural changes:
    - Removed feasibility_guard_weight (no hard binary penalty; instead, suppress energy under joint risk)
    - Replaced amplified_rank with upward_rank × remaining_work interaction (validated by counterfactual consensus)
    - Energy term now gated by sigmoid(slack) — active only when slack > slack_positive_gate
    - Critical-path interaction activated only when both slack < slack_negative_threshold AND remaining_work > work_leverage_threshold
    - All normalizations use mean-abs (more stable than median/MAD for small N)
    - Added bounded uncertainty-modulated slack gating for duration-robustness
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Robust per-feature normalization using mean-abs scaling (stable for all N ≥ 1)
    def mean_abs_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.mean(x)
        spread = np.mean(np.abs(x - center)) + eps
        return (x - center) / (spread + eps)

    norm_slack = mean_abs_normalize(slack)
    norm_energy = mean_abs_normalize(min_incremental_energy)
    norm_duration = mean_abs_normalize(min_exec_time + min_comm_time)
    norm_rank = mean_abs_normalize(upward_rank)
    norm_work = mean_abs_normalize(remaining_work)
    norm_wait = mean_abs_normalize(ready_wait_time)
    norm_uncert = mean_abs_normalize(uncertainty)

    # --- Slack risk score: bounded sigmoid penalty (smooth, monotonic, avoids exponent instability)
    # Maps slack ∈ ℝ → [0, 1], with steep rise near zero and asymptotic saturation
    slack_sigmoid = 1.0 / (1.0 + np.exp(-norm_slack))
    slack_risk_score = 2.0 * (1.0 - slack_sigmoid)  # Higher score for tighter/negative slack

    # --- Critical-path interaction: upward_rank × remaining_work, gated by joint slack+work condition
    critical_gate = np.where(
        (norm_slack < PARAMS["slack_negative_threshold"]) & 
        (remaining_work > PARAMS["work_leverage_threshold"]),
        1.0, 0.0
    )
    critical_path_score = norm_rank * norm_work * critical_gate * PARAMS["criticality_scale"]

    # --- Duration-robustness gating: uncertainty-modulated slack mask for duration penalty
    # Active only under high uncertainty AND tight slack → penalizes duration when risk is high
    dur_robust_gate = np.where(
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]) & 
        (norm_slack < PARAMS["slack_positive_gate"]),
        1.0, 0.0
    )
    dur_penalty = norm_duration * dur_robust_gate * PARAMS["duration_robustness"]

    # --- Energy sensitivity gating: sigmoid of slack enables energy minimization only when slack is safe
    # Avoids optimizing energy for tasks already at risk — DDL first
    energy_gate = 1.0 / (1.0 + np.exp(-(norm_slack - PARAMS["slack_positive_gate"])))
    energy_term = norm_energy * energy_gate * PARAMS["energy_sensitivity"]

    # --- Bounded starvation relief: soft arctan benefit scaled by wait time
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_benefit = (2.0 / np.pi) * np.arctan(wait_scaled)

    # --- Final priority score: smaller = higher priority
    # Order of terms reflects lexicographic priority: DDL risk > critical path > duration robustness > energy > starvation relief
    score = (
        slack_risk_score           # Primary DDL urgency (bounded, smooth)
        + critical_path_score      # Critical-path leverage (validated by consensus)
        + dur_penalty              # Uncertainty-aware duration penalty
        + energy_term              # Energy minimization (only when slack permits)
        - wait_benefit             # Anti-starvation (soft, saturating)
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
