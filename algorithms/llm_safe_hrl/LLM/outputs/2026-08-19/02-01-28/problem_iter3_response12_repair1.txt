def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with bounded DDL-protection gate and robust successor-blocking.

    Key structural improvements:
    - Introduces a conditional DDL-protection gate: when slack is critically tight AND uncertainty is high,
      energy minimization is down-weighted and deadline-safety terms are amplified via ddl_protection_strength.
    - Replaces ad-hoc residual_slack with a normalized, critical-path-aware successor blocking proxy:
      uses upward_rank-weighted slack deficit to estimate downstream criticality pressure.
    - All numeric literals are only -2,-1,0,1,2; all tunable constants exposed via PARAMS.
    - Uses robust MAD-based normalization throughout.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Robust normalization using median absolute deviation (MAD)
    def normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        dev = np.median(np.abs(x - center))
        scale = np.maximum(dev, eps)
        return (x - center) / scale

    # Normalize base features
    norm_slack = normalize(slack)
    norm_energy = normalize(min_incremental_energy)
    norm_rank = normalize(upward_rank)
    norm_work = normalize(remaining_work)
    norm_wait = normalize(ready_wait_time)
    norm_uncert = normalize(uncertainty)
    norm_duration = normalize(min_exec_time + min_comm_time)

    # --- Slack urgency: logistic gate for deadline pressure ---
    slack_urgency = 1.0 / (1.0 + np.exp(-PARAMS["slack_risk_penalty"] * norm_slack))

    # --- Tight-slack mask: smooth ramp near threshold ---
    tight_mask = np.clip(
        1.0 - (norm_slack - PARAMS["critical_slack_threshold"]) / (np.abs(norm_slack).max() + eps),
        0.0, 1.0
    )

    # --- Uncertainty-slack interaction: multiplicative risk coupling ---
    risk_gate = np.clip(norm_uncert * slack_urgency, 0.0, 1.0)

    # --- DDL-protection gate: activates only under joint high-risk condition ---
    ddl_protection_active = np.where(
        (norm_slack < 0.0) & (norm_uncert > 0.5),
        PARAMS["ddl_protection_strength"],
        1.0
    )

    # --- Duration-uncertainty coupling: penalize uncertain long tasks under pressure ---
    coupled_duration = norm_duration * (1.0 + PARAMS["duration_coupling"] * risk_gate)

    # --- Adaptive rank weighting: only under tight slack, scaled by DDL protection ---
    adaptive_rank = norm_rank * tight_mask * PARAMS["rank_urgency_ratio"] * ddl_protection_active

    # --- Work saturation: sigmoidal diminishing return on large remaining_work ---
    work_deviation = remaining_work - PARAMS["work_threshold"]
    work_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["work_scale"] * work_deviation))
    work_contribution = work_sigmoid * norm_work

    # --- Critical-path-aware successor-blocking proxy: weighted by upward_rank ---
    # Estimate how much slack deficit propagates to critical path successors
    slack_deficit = np.maximum(0.0, -(slack - (min_exec_time + min_comm_time)))
    successor_block_proxy = slack_deficit * (1.0 + norm_rank)  # amplify for high-rank tasks
    successor_block_penalty = normalize(successor_block_proxy)
    successor_block_penalty = successor_block_penalty * PARAMS["successor_release_weight"]

    # --- Stabilized anti-starvation wait boost (bounded subtraction) ---
    rel_wait = np.clip(norm_wait / (np.abs(norm_wait).max() + eps), 0.0, 2.0)
    wait_boost = rel_wait + (1.0 - np.exp(-PARAMS["wait_decay_rate"] * rel_wait))

    # --- Final raw score: smaller = higher priority ---
    # Energy term is *down-weighted* under DDL-protection mode to avoid sacrificing deadline safety
    energy_term = norm_energy * (2.0 - ddl_protection_active)  # 1.0 → 1.0, 1.2 → 0.8, 2.0 → 0.0
    score = (
        slack_urgency * ddl_protection_active
        + PARAMS["energy_sensitivity"] * energy_term
        + coupled_duration * ddl_protection_active
        + adaptive_rank
        + work_contribution
        + successor_block_penalty
        - wait_boost
        + risk_gate * ddl_protection_active
    )

    # --- Soft saturation to preserve ranking while bounding extremes ---
    strength = PARAMS["soft_saturation_strength"]
    score = score - np.log(1.0 + np.exp(-strength * score) + eps)

    # Ensure finite, shape-(N,) output; handle edge case N=1
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).smallest_subnormal,
                          neginf=-np.finfo(float).smallest_subnormal)
    if score.ndim == 0:
        score = np.full(1, score.item(), dtype=float)

    return score
