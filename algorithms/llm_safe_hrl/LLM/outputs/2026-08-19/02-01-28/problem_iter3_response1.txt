def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with hard deadline gating, uncertainty-coupled energy, and piecewise urgency.

    Key improvements:
      - Replaces sigmoid/soft gates with sharp piecewise-linear urgency near slack threshold → stronger DDL enforcement.
      - Introduces uncertainty-weighted energy term: `min_incremental_energy * (1 + uncertainty)` → favors low-uncertainty low-energy VMs.
      - Uses threshold-triggered hard penalty: if normalized slack < urgency_threshold, assign capped max penalty → strict feasibility-first.
      - All feature interactions are bounded, deterministic, and preserve monotonic priority ordering under risk.
      - No uncertainty×urgency coupling (removed per reflection); instead couples uncertainty directly to energy.
      - Final score is clipped *before* nan_to_num to guarantee shape-(N,) and finite output.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Cast to float without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Mean-absolute normalization (stable, robust to outliers)
    def normalize(x):
        abs_mean = np.mean(np.abs(x))
        scale = np.maximum(abs_mean, eps)
        return x / scale

    norm_exec = normalize(min_exec_time)
    norm_comm = normalize(min_comm_time)
    norm_energy = normalize(min_incremental_energy)
    norm_slack = normalize(slack)
    norm_rank = normalize(upward_rank)
    norm_work = normalize(remaining_work)
    norm_wait = normalize(ready_wait_time)
    norm_uncert = normalize(uncertainty)

    # Deadline risk: magnitude of negative slack (only when slack < 0)
    deadline_risk = np.maximum(-slack, 0.0)
    norm_risk = normalize(deadline_risk)

    # Piecewise-linear urgency gate: flat at 0 until threshold, then linear ramp up to 1 over steepness width
    # Ensures crisp transition — no smoothing that dilutes hard constraint signaling
    slack_diff = norm_slack - PARAMS["urgency_threshold"]
    urgency_gate = np.clip(
        np.where(slack_diff < 0.0, 0.0, slack_diff * PARAMS["urgency_steepness"]),
        0.0, 1.0
    )

    # Uncertainty-coupled energy: penalize high-uncertainty high-energy options more
    # This replaces removed uncertainty×urgency coupling with physically grounded interaction
    coupled_energy = norm_energy * (1.0 + PARAMS["energy_uncertainty_coupling"] * norm_uncert)

    # Base linear score (smaller = better), but *only* valid when slack >= threshold
    base_score = (
        PARAMS["exec_time_weight"] * norm_exec
        + PARAMS["comm_time_weight"] * norm_comm
        + PARAMS["energy_weight"] * coupled_energy
        + PARAMS["work_weight"] * norm_work
        + PARAMS["upward_rank_weight"] * norm_rank
        + PARAMS["wait_weight"] * norm_wait
    )

    # Hard gating: if slack too tight (urgency_gate == 0), assign maximum penalty; else use base_score
    # This enforces strict deadline feasibility first — no soft trade-off with energy
    score = np.where(
        urgency_gate == 0.0,
        PARAMS["penalty_cap"],
        base_score + PARAMS["deadline_risk_penalty"] * norm_risk
    )

    # Ensure finite and shape-(N,)
    finfo = np.finfo(float)
    score_final = np.nan_to_num(
        score,
        nan=finfo.smallest_subnormal,
        posinf=PARAMS["penalty_cap"],
        neginf=-PARAMS["penalty_cap"]
    )

    # Enforce shape (N,)
    if score_final.ndim == 0:
        score_final = np.full(1, score_final.item(), dtype=float)

    return score_final
