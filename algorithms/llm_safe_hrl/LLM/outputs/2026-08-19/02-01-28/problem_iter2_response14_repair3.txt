def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: signed slack power law + dual-gated criticality + adaptive energy discounting.
    
    Structural improvements:
    - Added `rank_slack_gate_threshold` for smooth, tunable gating of upward_rank.
    - Replaced hidden 0.1 literal with explicit `energy_discount_floor_factor`.
    - Replaced median-based uncertainty threshold with relative `uncertainty / (mean + eps)` for robust coupling.
    - Unified robust normalization using `np.quantile` with normalized quantiles [q1_quantile, q3_quantile].
    - All numeric literals are only -2,-1,0,1,2; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization: uses tunable quantiles, avoids percentile ambiguity
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q1, q3 = np.quantile(x, [PARAMS["q1_quantile"], PARAMS["q3_quantile"]])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Signed slack pressure: smooth, differentiable, avoids zero discontinuity
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_power = np.where(
        slack_abs < eps,
        0.0,
        slack_sign * (slack_abs ** PARAMS["slack_sensitivity"])
    )
    base_slack_penalty = PARAMS["slack_penalty_scale"] * slack_power
    
    # Uncertainty-slack coupling: multiplicative gate on base slack penalty
    unc_mean = np.mean(uncertainty) + eps
    unc_ratio = uncertainty / unc_mean
    gated_slack = base_slack_penalty * (1.0 + PARAMS["uncertainty_slack_coupling"] * unc_ratio)
    
    # Upward rank gating: soft threshold instead of binary condition
    slack_risk_mask = np.clip(
        (PARAMS["rank_slack_gate_threshold"] - slack) / (PARAMS["rank_slack_gate_threshold"] + eps),
        0.0, 1.0
    )
    boosted_rank = upward_rank * (1.0 + PARAMS["criticality_boost"] * slack_risk_mask)
    
    # Energy efficiency: discounted under negative slack via linear interaction; floor derived from parameter
    energy_norm = robust_normalize(min_incremental_energy)
    slack_discount = 1.0 - PARAMS["energy_slack_interaction"] * np.maximum(0.0, -slack)
    energy_floor = PARAMS["energy_discount_floor_factor"]  # now declared, not literal
    energy_score = PARAMS["energy_normalizer"] * energy_norm * np.clip(slack_discount, energy_floor, 1.0)
    
    # Duration ratio: favors compute-heavy tasks for edge placement; avoids division-by-zero
    exec_plus_comm = min_exec_time + min_comm_time + eps
    duration_ratio = min_exec_time / exec_plus_comm
    ratio_score = PARAMS["duration_ratio_weight"] * (1.0 - duration_ratio)
    
    # Anti-starvation: monotonic exponential decay (larger wait → smaller score → higher priority)
    wait_boost = np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # Assemble final priority score: smaller = higher priority
    score = (
        gated_slack
        + energy_score
        - robust_normalize(boosted_rank)
        + ratio_score
        - wait_boost
    )
    
    # Ensure finite output, shape (N,), and numerical safety
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
