def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: uses task-relative percentile ranking;
    work-based prioritization gated by positive slack only; smooth tanh hinge.
    All numeric literals are -2,-1,0,1,2 — no hidden constants."""
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    tiny = finfo.tiny
    huge = finfo.max

    # Ensure float64 and no mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # --- Task-relative percentile ranking (replaces robust_normalize) ---
    def percentile_rank(x):
        x = np.asarray(x, dtype=float)
        if np.all(x == x[0]):
            return np.full_like(x, 0.5, dtype=float)
        sorted_idx = np.argsort(x)
        ranks = np.empty_like(sorted_idx, dtype=float)
        ranks[sorted_idx] = np.arange(len(x), dtype=float) / max(len(x)-1, 1)
        return ranks

    rank_slack = percentile_rank(slack)
    rank_energy = percentile_rank(min_incremental_energy)
    rank_duration = percentile_rank(min_exec_time + min_comm_time)
    rank_rank = percentile_rank(upward_rank)
    rank_work = percentile_rank(remaining_work)
    rank_wait = percentile_rank(ready_wait_time)
    rank_uncert = percentile_rank(uncertainty)

    # --- Bounded slack risk score: smooth tanh hinge at zero slack ---
    # tanh_hinge = 0.5 * (1.0 + tanh((slack - center) / scale))
    tanh_arg = (slack - PARAMS["tanh_hinge_center"]) / (PARAMS["tanh_hinge_scale"] + eps)
    tanh_hinge = 0.5 * (1.0 + np.tanh(tanh_arg))
    slack_abs = np.abs(slack)
    negative_penalty = (1.0 - tanh_hinge) * np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"])
    positive_relief = tanh_hinge * (-slack_abs / (1.0 + slack_abs + eps))
    slack_risk_score = negative_penalty + positive_relief

    # --- Uncertainty-modulated slack penalty under deadline pressure only ---
    pressure_mask = (slack < 0) & (uncertainty > (np.median(uncertainty) + eps))
    slack_penalty_mod = np.where(pressure_mask,
                                PARAMS["uncertainty_slack_coupling_strength"] * rank_uncert,
                                0.0)
    slack_risk_score = slack_risk_score * (1.0 + slack_penalty_mod)

    # --- Criticality amplification: only when slack is critically low (using raw slack) ---
    critical_gate = np.where(slack < PARAMS["slack_negative_threshold"] * (np.abs(np.median(slack)) + eps), 1.0, 0.0)
    amplified_rank = rank_rank * (1.0 + PARAMS["criticality_scale"] * critical_gate)

    # --- Duration-uncertainty coupling: active only under joint tight slack AND high uncertainty ---
    dur_uncert_gate = np.where(
        (rank_slack < PARAMS["slack_positive_gate"]) &
        (rank_uncert > PARAMS["uncertainty_activation_threshold"]),
        1.0, 0.0
    )
    dur_uncert_interaction = rank_duration * rank_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Bounded starvation relief with hard cap under deadline pressure ---
    wait_benefit_raw = 1.0 - np.exp(-PARAMS["wait_decay"] * (rank_wait + eps))
    wait_cap = np.where(slack < 0,
                       PARAMS["wait_cap_factor_under_pressure"] * negative_penalty,
                       0.0)
    wait_benefit = np.clip(wait_benefit_raw, a_min=0.0, a_max=wait_cap + eps)

    # --- Work saturation: activated ONLY when slack is sufficiently positive ---
    work_gate = np.where(rank_slack > PARAMS["work_activation_threshold"], 1.0, 0.0)
    work_ref = PARAMS["work_saturation_point"]
    work_ratio = np.clip(remaining_work / (work_ref + eps), tiny, huge)
    work_scaled = work_gate * (1.0 / (1.0 + 1.0 / (work_ratio + eps)))

    # --- Final score: smaller = better ---
    score = (
        slack_risk_score                              # Primary: risk-adjusted urgency (smooth hinge)
        + PARAMS["energy_sensitivity"] * rank_energy  # Secondary: energy efficiency
        + dur_uncert_interaction                      # Tertiary: uncertain long tasks under pressure
        - amplified_rank                              # Suppressive: critical path leverage only when urgent
        + work_scaled                                 # Conditional: small work → higher priority (only if slack safe)
        - wait_benefit                                # Longer wait → higher priority (capped)
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=huge/2, neginf=-huge/2)
    score = np.clip(score, a_min=-huge/2, a_max=huge/2)

    return score
