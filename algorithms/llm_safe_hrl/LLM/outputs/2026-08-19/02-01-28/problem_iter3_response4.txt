def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with adaptive piecewise-linear slack gating and slack-aware work-density amplification.
    
    Key improvements:
    - Replaces tanh/soft gates with *adaptive piecewise-linear slack penalty*: 
      enforces hard feasibility by zeroing slack gain below a robustly estimated threshold (median |slack| * factor),
      while preserving smoothness and monotonicity above it.
    - Introduces *slack-aware work-density scaling*: multiplies work_density by (1 + positive_slack_amplification * I(slack > threshold))
      only when slack is sufficiently positive — avoids false urgency under pressure and sharpens criticality focus.
    - Removes energy_efficiency_exponent: preserves strict ordinal ranking of energy/duration efficiency under tight deadlines.
    - Uses robust median-based thresholding instead of fixed zero → adapts to workflow-scale slack distribution.
    - All operations remain vectorized, bounded, and deterministic; no conditionals beyond np.where.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)
    N = len(slack)

    # Defensive copy & cast
    min_exec_time = np.asarray(min_exec_time, dtype=float).copy()
    min_comm_time = np.asarray(min_comm_time, dtype=float).copy()
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float).copy()
    slack = np.asarray(slack, dtype=float).copy()
    upward_rank = np.asarray(upward_rank, dtype=float).copy()
    remaining_work = np.asarray(remaining_work, dtype=float).copy()
    ready_wait_time = np.asarray(ready_wait_time, dtype=float).copy()
    uncertainty = np.asarray(uncertainty, dtype=float).copy()

    # Duration and derived features in raw space
    duration = min_exec_time + min_comm_time
    work_density = remaining_work / np.maximum(duration, eps_safe)

    # Adaptive slack threshold: median absolute slack scaled by tunable factor
    abs_slack = np.abs(slack)
    median_abs_slack = np.median(abs_slack) if N > 1 else abs_slack[0]
    slack_threshold = PARAMS["slack_zero_threshold_factor"] * (median_abs_slack + eps_safe)

    # Piecewise-linear slack gating: zero gain/penalty below threshold; linear ramp above
    # For slack <= -threshold: full penalty (linear in |slack|)
    # For |slack| < threshold: zero contribution (hard feasibility zone)
    # For slack >= threshold: linear gain (proportional to slack)
    slack_penalty_mask = (slack <= -slack_threshold)
    slack_gain_mask = (slack >= slack_threshold)
    slack_neutral_mask = ~(slack_penalty_mask | slack_gain_mask)

    # Linear penalty: slope = slack_risk_penalty / threshold → ensures continuity at -threshold
    penalty_slope = PARAMS["slack_risk_penalty"] / (slack_threshold + eps_safe)
    slack_penalty_term = penalty_slope * (-slack) * slack_penalty_mask

    # Linear gain: slope = slack_benefit_gain / threshold → ensures continuity at +threshold
    gain_slope = PARAMS["slack_benefit_gain"] / (slack_threshold + eps_safe)
    slack_gain_term = gain_slope * slack * slack_gain_mask

    # Uncertainty coupling applied only to magnitude of active slack terms
    slack_magnitude_active = (
        (-slack) * slack_penalty_mask + 
        slack * slack_gain_mask
    )
    norm_uncert = (uncertainty / (np.mean(np.abs(uncertainty)) + eps_safe))
    coupled_slack = (slack_penalty_term + slack_gain_term) * (
        1.0 + PARAMS["uncertainty_slack_coupling"] * norm_uncert
    )

    # Robust normalization: mean absolute value + epsilon
    def robust_normalize(x):
        x_abs = np.abs(x)
        denom = np.mean(x_abs) + eps_safe
        return x / denom

    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_duration = robust_normalize(duration)

    # Slack-aware work-density amplification: only activate when slack is sufficiently positive
    work_density_amplified = work_density * (
        1.0 + PARAMS["positive_slack_amplification"] * slack_gain_mask
    )
    norm_work_density = robust_normalize(work_density_amplified)

    # Critical path: higher upward_rank → higher priority → negate for minimization
    critical_term = -PARAMS["critical_path_weight"] * norm_rank

    # Work density efficiency: prefer high compute per time → negate
    work_density_term = -PARAMS["work_density_weight"] * norm_work_density

    # Energy minimization: use raw normalized energy (no exponent distortion)
    energy_term = PARAMS["energy_scale"] * norm_energy

    # Starvation relief: longer wait → higher priority → negate wait term
    starvation_term = -PARAMS["wait_starvation_relief"] * norm_wait

    # Duration flexibility bias: quadratic penalty around unit-normalized duration
    flex_penalty = PARAMS["duration_flexibility_bias"] * ((np.abs(norm_duration) - 1.0) ** 2)

    # Assemble final score: smaller = better
    score = (
        coupled_slack
        + critical_term
        + work_density_term
        + energy_term
        + starvation_term
        + flex_penalty
    )

    # Final safeguard: ensure finite, shape-(N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    return score
