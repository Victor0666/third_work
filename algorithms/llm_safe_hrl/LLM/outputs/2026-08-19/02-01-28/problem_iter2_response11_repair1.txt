def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's robustness with Parent 1's slack-gated energy/rank weighting.
    
    Key structural improvements:
      - Dual-gate criticality boost (slack + uncertainty) preserved from Parent 2
      - Slack-based gating for energy & rank contributions (from Parent 1) prevents over-prioritizing efficiency when deadlines are at risk
      - Robust mean-based normalization (Parent 2) with fallback to epsilon-scaled identity for near-zero variance
      - Explicit anti-starvation piecewise linear relief (Parent 2), now applied *before* normalization to preserve physical scale meaning
      - All numeric literals strictly limited to {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Safe normalization: handles zero-variance by falling back to scaled identity
    def safe_normalize(x):
        x = np.asarray(x, dtype=float)
        abs_x = np.abs(x)
        mean_abs = np.mean(abs_x)
        scale = np.where(mean_abs == 0.0, eps, mean_abs)
        return x / (scale + eps)
    
    # Normalize base features after applying domain-aware transformations
    norm_slack = safe_normalize(slack)
    norm_energy = safe_normalize(min_incremental_energy)
    norm_rank = safe_normalize(upward_rank)
    norm_work = safe_normalize(remaining_work)
    norm_wait = safe_normalize(ready_wait_time)
    norm_uncert = safe_normalize(uncertainty)
    norm_duration = safe_normalize(min_exec_time + min_comm_time)
    
    # === SLACK RISK PENALTY (nonlinear & signed) ===
    raw_slack_penalty = np.power(np.maximum(0.0, -slack), PARAMS["slack_penalty_exponent"])
    
    # === CRITICALITY BOOST GATE (dual condition) ===
    slack_critical = (slack < PARAMS["critical_slack_threshold"])
    uncert_high = (uncertainty > PARAMS["uncertainty_high_threshold"])
    critical_gate = np.where(slack_critical | uncert_high, 1.0, 0.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_boost"] * critical_gate)
    
    # === SLACK-GATED ENERGY & RANK CONTRIBUTIONS (from Parent 1) ===
    # Only reward energy efficiency and rank importance when slack is safe
    slack_safe_mask = (slack > PARAMS["critical_slack_threshold"]).astype(float)
    energy_contribution = -PARAMS["energy_efficiency_sensitivity"] * norm_energy * slack_safe_mask
    rank_contribution = -PARAMS["upward_rank_weight"] * norm_rank * slack_safe_mask
    
    # === DURATION-UNCERTAINTY COUPLING ===
    coupled_duration = norm_duration * (1.0 + PARAMS["duration_uncertainty_coupling"] * uncertainty)
    
    # === STARVATION RELIEF (piecewise linear, physical-scale first) ===
    wait_excess = np.maximum(0.0, ready_wait_time - PARAMS["wait_decay_threshold"])
    starvation_relief = PARAMS["starvation_relief_slope"] * wait_excess
    norm_relief = safe_normalize(starvation_relief)
    
    # === COMPOSITE SCORE: smaller = higher priority ===
    # All numeric literals are strictly -2,-1,0,1,2
    score = (
        raw_slack_penalty
        + boosted_rank
        + energy_contribution
        + rank_contribution
        + coupled_duration
        - norm_relief
        + 1 * PARAMS["remaining_work_weight"] * norm_work
    )
    
    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
