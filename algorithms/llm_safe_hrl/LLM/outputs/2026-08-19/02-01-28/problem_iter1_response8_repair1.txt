def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: removes unused 'uncertainty_slack_gate', uses all declared parameters,
    enforces contract-compliant literals (-2,-1,0,1,2), and applies robust normalization."""
    eps = PARAMS["epsilon"]
    
    # Robust per-feature normalization using median absolute deviation
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        spread = np.median(np.abs(x - center)) + eps
        return (x - center) / spread

    # Normalize core features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # Risk-aware slack penalty: strong nonlinearity for negative slack only
    slack_abs = np.abs(slack)
    negative_penalty = np.where(slack < 0, 
                                np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"]), 
                                0.0)
    positive_relief = np.where(slack >= 0, 
                               -slack_abs / (1.0 + slack_abs + eps), 
                               0.0)
    slack_risk_score = negative_penalty + positive_relief

    # Criticality amplification: boost rank only under high deadline pressure
    critical_gate = np.where(norm_slack < -0.5, 1.0, 0.0)
    amplified_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * critical_gate)

    # Duration-uncertainty coupling: penalize high uncertainty when slack is tight
    # Use sigmoid(-norm_slack) for smooth gating
    slack_sigmoid = 1.0 / (1.0 + np.exp(norm_slack))
    duration_uncert_coupling = norm_uncert * slack_sigmoid * PARAMS["duration_robustness"]

    # Bounded starvation relief via saturating log1p
    wait_saturation = np.log1p(norm_wait * PARAMS["wait_decay"]) / (PARAMS["wait_decay"] + eps)

    # Work saturation: dampen influence using log-scale reference
    work_ref = PARAMS["work_saturation_point"]
    work_scaled = np.log1p(remaining_work / (work_ref + eps))

    # Assemble final score: smaller = higher priority
    # Components ordered by DDL-risk dominance: slack risk > criticality > energy > duration >
    # starvation relief (negative contribution) > uncertainty coupling > work coverage
    score = (
        slack_risk_score                              # Primary DDL risk term
        + amplified_rank                             # Critical path boost under urgency
        + PARAMS["energy_sensitivity"] * norm_energy  # Energy minimization (secondary)
        + norm_duration                              # Latency preference
        - wait_saturation                            # Anti-starvation (reduces score)
        + duration_uncert_coupling                   # Uncertainty × urgency penalty
        - work_scaled                                # Prefer tasks guarding large remaining work
    )

    # Final numerical safeguard
    finfo = np.finfo(float)
    return np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
