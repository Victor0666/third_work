def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL-risk gating, critical-path leverage, and starvation-robust normalization.
    
    Key innovations:
    - Slack is split into linear urgency (for positive slack) and *superlinear penalty* (for negative slack)
    - Criticality boost activates only when both upward_rank > median AND slack <= slack_threshold
    - Duration sensitivity is gated by slack sign: favors fast execution *only* when slack > 0; otherwise suppresses it
    - Ready wait uses saturating arctan-based anti-starvation (bounded, smooth, no threshold jumps)
    - Uncertainty couples *only* with negative slack via a bounded product gate (no amplification when slack >= 0)
    - All features normalized robustly using mean-abs + epsilon; no std-based norm to avoid NaN on singleton or flat arrays
    - Energy term scaled *after* normalization to preserve risk-adjusted marginal interpretation
    - Remaining work leveraged fractionally to avoid overweighting coarse-grained tasks
    """
    epsilon = PARAMS["epsilon"]
    N = len(slack)

    # Robust normalization: use mean absolute value + epsilon, safe for zeros and singletons
    def robust_norm(x):
        x = np.asarray(x, dtype=float)
        denom = np.mean(np.abs(x)) + epsilon
        return x / denom

    # Normalize base features
    norm_slack = robust_norm(slack)
    norm_energy = robust_norm(min_incremental_energy)
    norm_duration = robust_norm(min_exec_time + min_comm_time)
    norm_rank = robust_norm(upward_rank)
    norm_work = robust_norm(remaining_work)
    norm_wait = robust_norm(ready_wait_time)
    norm_uncert = robust_norm(uncertainty)

    # === Slack-driven nonlinear risk gating ===
    # Positive slack → mild linear urgency; Negative slack → quadratic penalty (stronger push)
    slack_urgency = np.where(
        slack >= 0,
        norm_slack,
        -np.power(-norm_slack, PARAMS["slack_penalty_exponent"])
    )

    # === Criticality boost: activate only if high rank AND tight/overdue slack ===
    median_rank = np.median(upward_rank) if N > 1 else upward_rank[0]
    slack_threshold = np.percentile(slack, 30) if N > 1 else slack[0]
    critical_gate = np.where(
        (upward_rank > median_rank) & (slack <= slack_threshold),
        PARAMS["criticality_boost"],
        1.0
    )

    # === Duration sensitivity: favor short duration only when slack > 0; suppress otherwise ===
    duration_weighted = np.where(
        slack > 0,
        PARAMS["duration_sensitivity"] * norm_duration,
        -0.1 * norm_duration  # slight suppression (not penalty) when overdue
    )

    # === Anti-starvation: saturating arctan on ready_wait_time (smooth, bounded [0, π/2]) ===
    # Scale wait time first to avoid arctan saturation too early
    scaled_wait = ready_wait_time / (PARAMS["wait_saturation_threshold"] + epsilon)
    wait_anti_starve = (2.0 / np.pi) * np.arctan(scaled_wait)  # maps to [0, 1]

    # === Uncertainty-slack coupling: only active under deadline pressure (slack <= 0) ===
    uncert_slack_coupling = np.where(
        slack <= 0,
        PARAMS["uncertainty_slack_coupling"] * norm_uncert * (-norm_slack),
        0.0
    )

    # === Work leverage: fractional use to prevent domination by large-work tasks ===
    work_leveraged = PARAMS["work_leverage_factor"] * norm_work

    # Assemble final score: smaller = higher priority
    # Note: critical_gate multiplies *rank term*, not entire score — preserves sign semantics
    score = (
        -slack_urgency                            # higher urgency → lower score
        + PARAMS["energy_efficiency_scale"] * norm_energy
        + duration_weighted
        - critical_gate * norm_rank             # boost critical path *only when needed*
        + work_leveraged                        # add (not subtract) because large work often implies downstream impact
        - wait_anti_starve                      # reduce score for long-waiting tasks (anti-starvation)
        + uncert_slack_coupling                 # adds penalty under uncertainty + lateness
    )

    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=1e6, neginf=-1e6)
    return score
