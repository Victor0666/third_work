def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust slack modeling and conditional gating 
    with Parent 1's starvation capping and uncertainty-slack coupling — all via bounded, stable ops.
    
    Key structural improvements:
      - Adds uncertainty_slack_coupling_strength: modulates slack penalty *only* under negative slack
        (not global modulation), improving risk focus without destabilizing positive slack behavior.
      - Introduces wait_cap_factor_under_pressure: caps starvation relief by slack penalty magnitude
        to prevent overcompensation during deadline violations — a safety guard absent in both parents.
      - Uses robust per-feature normalization (Parent 2) but adds epsilon-guarded median scaling
        (enhanced from Parent 1) for improved outlier resilience.
      - Keeps Parent 2’s superior bounded slack penalty (+ reward) and joint conditional gates.
      - All numeric literals are strictly -2, -1, 0, 1, or 2; no unbounded exponents or divisions.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    tiny = finfo.tiny
    huge = finfo.max

    # Ensure float64 and no mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust per-feature normalization: median-center + MAD scale, epsilon-guarded
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        abs_devs = np.abs(x - center)
        spread = np.median(abs_devs) + eps
        return (x - center) / (spread + eps)

    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # --- Bounded slack risk score: strong penalty for negative slack, soft relief for positive ---
    slack_abs = np.abs(slack)
    negative_penalty = np.where(slack < 0,
                                np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"]),
                                0.0)
    positive_relief = np.where(slack >= 0,
                               -slack_abs / (1.0 + slack_abs + eps),
                               0.0)
    slack_risk_score = negative_penalty + positive_relief

    # --- Uncertainty-modulated slack penalty under deadline pressure only ---
    # Active only when slack < 0 AND uncertainty is high → tighter coupling than Parent 1's gate
    pressure_mask = (slack < 0) & (uncertainty > (np.median(uncertainty) + eps))
    slack_penalty_mod = np.where(pressure_mask,
                                PARAMS["uncertainty_slack_coupling_strength"] * uncertainty,
                                0.0)
    slack_risk_score = slack_risk_score * (1.0 + slack_penalty_mod)

    # --- Criticality amplification: only when norm_slack is critically low ---
    critical_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    amplified_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * critical_gate)

    # --- Duration-uncertainty coupling: active only under joint tight slack AND high uncertainty ---
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) &
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]),
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Bounded starvation relief with hard cap under deadline pressure ---
    wait_benefit_raw = 1.0 - np.exp(-PARAMS["wait_decay"] * (np.maximum(norm_wait, 0.0) + eps))
    # Cap starvation relief to avoid overriding slack penalties
    wait_cap = np.where(slack < 0,
                       PARAMS["wait_cap_factor_under_pressure"] * negative_penalty,
                       0.0)
    wait_benefit = np.clip(wait_benefit_raw, a_min=0.0, a_max=wait_cap + eps)

    # --- Work saturation: sigmoid-like scaling against reference point ---
    work_ref = PARAMS["work_saturation_point"]
    work_ratio = np.clip(remaining_work / (work_ref + eps), tiny, huge)
    work_scaled = 1.0 / (1.0 + 1.0 / (work_ratio + eps))

    # --- Final score: smaller = better ---
    # Dominant terms first: slack risk (penalty-driven), then energy minimization
    # Then tie-breakers: duration-uncert (penalty), rank (suppressive under stress), work (smaller favored), wait (anti-starve)
    score = (
        slack_risk_score                              # Primary: risk-adjusted urgency
        + PARAMS["energy_sensitivity"] * norm_energy  # Secondary: energy efficiency
        + dur_uncert_interaction                      # Tertiary: uncertain long tasks under pressure
        - amplified_rank                              # Suppressive: critical path leverage only when urgent
        + work_scaled                                 # Smaller work → higher priority
        - wait_benefit                                # Longer wait → higher priority (capped)
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=huge/2, neginf=-huge/2)
    score = np.clip(score, a_min=-huge/2, a_max=huge/2)

    return score
