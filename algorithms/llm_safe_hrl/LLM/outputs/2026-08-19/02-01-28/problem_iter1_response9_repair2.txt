def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing deadline risk mitigation, critical-path awareness,
    work-density efficiency, and starvation relief — all with robust normalization."""
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Normalize each feature robustly using mean absolute value + epsilon
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        denom = np.mean(np.abs(x)) + eps_safe
        return x / denom

    # Duration: execution + communication; used in work density and flexibility bias
    duration = min_exec_time + min_comm_time
    norm_duration = robust_normalize(duration)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # Work density: remaining_work / (duration + eps) → higher = more compute per time
    work_density = remaining_work / (duration + eps_safe)
    norm_work_density = robust_normalize(work_density)

    # Slack gating: strong penalty for negative slack, mild gain for positive slack
    slack_sign = np.sign(slack)
    slack_magnitude = np.abs(slack)
    # Use logistic-like bounded gate for smooth transition near zero slack
    slack_gate = 1.0 / (1.0 + np.exp(-2.0 * norm_slack))  # Sigmoid centered at 0
    slack_penalty_term = PARAMS["slack_risk_penalty"] * (1.0 - slack_gate) * norm_slack
    slack_gain_term = PARAMS["slack_benefit_gain"] * slack_gate * norm_slack

    # Uncertainty-coupled slack modulation: amplifies penalty/gain where uncertainty is high
    coupled_slack = (slack_penalty_term + slack_gain_term) * (
        1.0 + PARAMS["uncertainty_slack_coupling"] * norm_uncert
    )

    # Critical path importance: higher upward_rank → higher priority (so negate for min-score)
    critical_term = -PARAMS["critical_path_weight"] * norm_rank

    # Work density efficiency: prefer tasks delivering more work per time unit
    work_density_term = -PARAMS["work_density_weight"] * norm_work_density

    # Energy minimization: lower incremental energy → higher priority
    energy_term = PARAMS["energy_scale"] * norm_energy

    # Starvation relief: longer wait → higher priority (so negate wait term)
    starvation_term = -PARAMS["wait_starvation_relief"] * norm_wait

    # Duration flexibility bias: penalize extremes (very short or very long durations)
    # Quadratic penalty around median behavior: (|norm_duration| - 1)^2 → minimized at |x|=1
    flex_penalty = PARAMS["duration_flexibility_bias"] * ((np.abs(norm_duration) - 1.0) ** 2)

    # Assemble final score: smaller = better
    score = (
        coupled_slack
        + critical_term
        + work_density_term
        + energy_term
        + starvation_term
        + flex_penalty
    )

    # Ensure finite output and clip extreme values to avoid overflow-induced NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
