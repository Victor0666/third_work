def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: piecewise slack gating + robust median-root normalization + critical-path/urgency coupling.
    
    Key innovations:
      - Uses smooth sigmoid + hard penalty for negative slack to sharply escalate priority under deadline pressure
      - Replaces mean-based normalization with median-root scaling (more robust to outliers in energy/duration)
      - Introduces rank_slack_coupling: multiplies upward_rank by max(0,-slack), boosting high-rank tasks *only* when overdue
      - Combines uncertainty with duration via multiplicative interaction — high uncertainty amplifies scheduling urgency for long ops
      - Applies power-law decay to ready_wait_time to prevent starvation without over-promoting stale tasks
      - All operations guarded against NaN/inf/zero using epsilon and np.finfo
    """
    eps = np.finfo(float).tiny + PARAMS["epsilon"]
    
    # Robust median-root normalization: avoids mean skew from outliers; uses root power for compressive scaling
    def robust_normalize(x, power=1.0):
        x = np.asarray(x, dtype=float)
        med = np.median(np.abs(x))
        scale = np.power(np.maximum(med, eps), power)
        return x / (scale + eps)

    # Normalize core features with learned power scaling for energy
    norm_energy = robust_normalize(min_incremental_energy, PARAMS["energy_norm_power"])
    norm_duration = robust_normalize(min_exec_time + min_comm_time, 1.0)
    norm_rank = robust_normalize(upward_rank, 1.0)
    norm_work = robust_normalize(remaining_work, 1.0)
    norm_wait = robust_normalize(ready_wait_time, 1.0)
    norm_uncert = robust_normalize(uncertainty, 1.0)

    # Slack: soft sigmoid for near-zero transition + strong linear penalty for negative values
    # Sigmoid centered at 0 with steepness k → maps slack ∈ ℝ → (0,1), then invert & shift for priority
    sigmoid_slack = 1.0 / (1.0 + np.exp(-PARAMS["slack_sigmoid_k"] * slack))
    # Penalty region: strong linear boost for negative slack, smoothly faded near zero
    slack_penalty = PARAMS["slack_penalty_scale"] * np.maximum(-slack, 0.0)
    # Final slack score: low (good) when slack > 0, rises sharply when slack < 0
    slack_score = (1.0 - sigmoid_slack) * 10.0 + slack_penalty * 0.1

    # Critical-path + urgency coupling: only activates when slack is negative
    rank_urgency = norm_rank * np.maximum(-slack, 0.0) * PARAMS["rank_slack_coupling"]

    # Uncertainty-duration interaction: high uncertainty makes long durations more urgent
    dur_uncert_interaction = norm_duration * norm_uncert * PARAMS["duration_risk_interaction"]

    # Anti-starvation: wait time raised to sublinear power to avoid dominance
    wait_effect = np.power(np.maximum(norm_wait, eps), PARAMS["wait_decay"])

    # Assemble score: smaller = better
    # Prioritize: (1) meeting deadlines (slack_score), (2) critical+overdue work (rank_urgency),
    #             (3) energy efficiency (norm_energy), (4) uncertainty-aware duration (dur_uncert_interaction),
    #             (5) anti-starvation (wait_effect), (6) suppress non-critical work (norm_work)
    score = (
        slack_score
        + rank_urgency
        + norm_energy
        + dur_uncert_interaction
        + wait_effect
        - norm_work  # higher remaining work → lower priority *unless* coupled with urgency
    )

    # Clamp extreme values and ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=1e6, neginf=-1e6)
    return score
