def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric literals are -2,-1,0,1,2; no hidden constants.
    
    Key structural features:
      - Quadratic+ slack penalty for negative slack (DDL risk first)
      - Criticality gate: only boosts upward_rank when both high-rank AND slack < 0
      - Energy density (joules/sec) as efficiency signal — lower is better
      - Saturating exponential wait gain with tunable decay
      - Uncertainty-slack coupling only active under deadline pressure
      - All coefficients now declared; no literals beyond {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Defensive casting and safe zero/Nan handling
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)

    # Robust normalization using mean absolute value + epsilon
    def robust_normalize(x):
        x_abs = np.abs(x)
        denom = np.mean(x_abs) + eps
        return x / denom

    # 1. Slack penalty: linear for positive, exponentiated for negative
    slack_abs = np.abs(slack)
    slack_penalty = np.where(
        slack >= 0,
        slack_abs,
        slack_abs ** PARAMS["slack_penalty_exponent"]
    )
    norm_slack_penalty = robust_normalize(slack_penalty)

    # 2. Duration: weighted blend of exec and comm
    duration = (PARAMS["duration_balance"] * min_exec_time +
                (1.0 - PARAMS["duration_balance"]) * min_comm_time)
    norm_duration = robust_normalize(duration + eps)

    # 3. Energy efficiency density: marginal energy per time unit → lower is better
    energy_density = min_incremental_energy / (duration + eps)
    norm_energy_density = robust_normalize(energy_density)

    # 4. Criticality gate: boost rank only if high-rank AND slack < 0
    rank_threshold = PARAMS["rank_threshold_ratio"] * np.max(upward_rank + eps)
    critical_gate = np.where(
        (upward_rank >= rank_threshold) & (slack < 0),
        PARAMS["criticality_boost"],
        1.0
    )
    norm_upward_rank = robust_normalize(upward_rank)
    boosted_rank_score = critical_gate * norm_upward_rank

    # 5. Ready wait time: saturating exponential gain
    wait_gain = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    norm_wait_gain = robust_normalize(wait_gain)

    # 6. Uncertainty-slack coupling: only active when slack < 0
    coupled_risk = np.where(
        slack < 0,
        uncertainty * slack_abs,
        0.0
    )
    norm_coupled_risk = robust_normalize(coupled_risk + eps)

    # Composite score: smaller = higher priority
    # Lexicographic emphasis: DDL risk > energy efficiency > duration > criticality > fairness > risk coupling
    score = (
        # Primary: deadline risk dominates
        1.0 * norm_slack_penalty
        # Secondary: favor low energy density
        + PARAMS["energy_efficiency_gain"] * norm_energy_density
        # Tertiary: prefer shorter durations
        + PARAMS["duration_weight"] * norm_duration
        # Quaternary: suppress (not boost) rank unless gated — use suppression coefficient
        - PARAMS["rank_suppression"] * boosted_rank_score
        # Quinary: anti-starvation via wait gain
        - PARAMS["wait_gain_weight"] * norm_wait_gain
        # Senary: amplify uncertainty impact only under DDL stress
        + PARAMS["uncertainty_slack_coupling"] * norm_coupled_risk
    )

    # Final safeguard
    return np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
