def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's sharp quadratic slack penalty and criticality gating
    with Parent 1's starvation-aware capped wait boost and robust L1-median normalization.
    
    Key structural innovations:
      - Slack penalty uses exponentiated negative slack *and* capped linear urgency for positive slack
      - Criticality gating only activates when (upward_rank > threshold) AND (slack < 0) — tight coupling
      - Wait boost is saturating exponential but capped by slack penalty to prevent violation-risk overcompensation
      - Robust L1-median normalization (more outlier-resistant than mean-based) applied uniformly
      - Uncertainty amplifies slack penalty *only* when slack < 0, using safe exponential gating
      - Energy density (joules/sec) remains core efficiency signal; no raw energy alone
      - All numeric literals are -2,-1,0,1,2 or from PARAMS; no hidden constants
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    tiny = finfo.tiny
    huge = finfo.max

    # Defensive casting without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust L1-median normalization: more stable than mean under outliers
    def robust_norm(x):
        x_abs = np.abs(x)
        med = np.median(x_abs)
        scale = np.where(med > eps, med, eps)
        return x / (scale + eps)

    # Duration = exec + comm, normalized
    duration = min_exec_time + min_comm_time
    norm_duration = robust_norm(duration + eps)

    # Slack handling: quadratic penalty for negative, capped sublinear urgency for positive
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_penalty = np.where(
        slack < 0,
        slack_abs ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    slack_urgency = np.where(
        slack > 0,
        np.clip(slack_abs ** 0.5, a_min=0.0, a_max=huge),  # sqrt for diminishing returns
        0.0
    )
    # Cap urgency to avoid weakening penalty dominance
    slack_score = slack_penalty - np.clip(slack_urgency, a_min=0.0, a_max=PARAMS["slack_urgency_cap"] * np.maximum(slack_penalty, eps))

    # Uncertainty-amplified slack penalty: only active when slack < 0, gated by exp(-|slack|/tau)
    pos_slack_med = np.where(np.any(slack > eps), np.median(slack[slack > eps]), eps)
    tau = np.maximum(pos_slack_med, eps)
    slack_gate_arg = np.clip(slack_abs / (tau + eps), a_min=0.0, a_max=PARAMS["exp_clip_max"])
    slack_gate = np.exp(-slack_gate_arg)
    amplified_penalty = slack_penalty * (1.0 + PARAMS["uncertainty_risk_amplifier"] * uncertainty * slack_gate)
    slack_score = amplified_penalty - np.clip(slack_urgency, a_min=0.0, a_max=PARAMS["slack_urgency_cap"] * np.maximum(amplified_penalty, eps))

    # Criticality gate: only boosts high-rank tasks under deadline stress
    rank_threshold = PARAMS["criticality_gate_threshold"] * np.max(upward_rank + eps)
    critical_mask = (upward_rank >= rank_threshold) & (slack < 0)
    norm_upward_rank = robust_norm(upward_rank)
    boosted_rank = np.where(critical_mask,
                           PARAMS["rank_suppression_factor"] * norm_upward_rank,
                           norm_upward_rank)

    # Energy density: marginal energy per effective duration → lower is better
    energy_density = min_incremental_energy / (duration + eps)
    norm_energy_density = robust_norm(energy_density)

    # Saturating wait gain with cap tied to slack penalty (anti-starvation under safety)
    wait_gain_raw = 1.0 - np.exp(-PARAMS["wait_saturation_rate"] * ready_wait_time)
    wait_cap = np.where(slack < 0,
                       PARAMS["slack_urgency_cap"] * slack_penalty,
                       np.clip(PARAMS["slack_urgency_cap"] * slack_urgency, a_min=0.0, a_max=huge))
    wait_boost = np.clip(wait_gain_raw, a_min=0.0, a_max=wait_cap)
    norm_wait_boost = robust_norm(wait_boost)

    # Final composite score: smaller = higher priority
    # Order: DDL risk dominates → energy density → duration → critical rank → fairness
    score = (
        1.0 * robust_norm(slack_score)  # Primary: hard-DDL enforcement
        + PARAMS["energy_density_gain"] * norm_energy_density  # Favor efficiency
        + PARAMS["duration_weight"] * norm_duration  # Prefer low-latency tasks
        - boosted_rank  # Promote critical path *only* when at risk
        - PARAMS["wait_gain_weight"] * norm_wait_boost  # Anti-starvation within safety bounds
    )

    # Final numerical safeguards
    score = np.nan_to_num(score, nan=0.0, posinf=huge/2, neginf=-huge/2)
    score = np.clip(score, a_min=-huge/2, a_max=huge/2)

    return score
