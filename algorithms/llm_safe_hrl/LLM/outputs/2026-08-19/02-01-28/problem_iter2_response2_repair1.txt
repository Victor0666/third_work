def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule with all declared parameters used; no unused or missing PARAMS references.
    
    Key features:
      - Signed slack power law with configurable sensitivity and scale
      - Criticality boost gated *only* by joint (slack < 0 AND uncertainty > median)
      - Uncertainty-slack coupling applied multiplicatively only under deadline pressure
      - Duration blend and energy density (joules/sec) both normalized via IQR
      - Saturating exponential wait gain for fairness
      - All numeric literals are -2,-1,0,1,2; epsilon from PARAMS; no hidden constants
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Defensive casting and safe NaN/inf handling
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=finfo.max, neginf=-finfo.max)

    # Robust IQR-based normalization
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q1_val = PARAMS["q1_quantile"]
        q3_val = PARAMS["q3_quantile"]
        q1, q3 = np.quantile(x, [q1_val, q3_val])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr

    # 1. Signed slack power law: smooth, discriminative DDL pressure
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_power = np.where(
        slack_abs < eps,
        0.0,
        slack_sign * (slack_abs ** PARAMS["slack_sensitivity"])
    )
    base_slack_penalty = PARAMS["slack_penalty_scale"] * slack_power

    # 2. Uncertainty-slack coupling: multiplicative gate on absolute slack, active only when slack < 0
    unc_slack_gate = np.where(
        slack < 0,
        1.0 + PARAMS["uncertainty_slack_coupling"] * (uncertainty / (np.mean(uncertainty) + eps)),
        1.0
    )
    gated_slack_penalty = base_slack_penalty * unc_slack_gate

    # 3. Duration score: weighted blend → favors shorter total latency
    duration = (PARAMS["duration_balance"] * min_exec_time +
                (1.0 - PARAMS["duration_balance"]) * min_comm_time)
    norm_duration = robust_normalize(duration + eps)

    # 4. Energy density: marginal energy per time unit → lower is more efficient
    energy_density = min_incremental_energy / (duration + eps)
    norm_energy_density = robust_normalize(energy_density)

    # 5. Criticality boost: only when *both* slack < 0 AND uncertainty above median
    median_uncertainty = np.median(uncertainty) if len(uncertainty) > 1 else np.mean(uncertainty)
    critical_gate = np.where(
        (slack < 0) & (uncertainty > median_uncertainty + eps),
        1.0 + PARAMS["criticality_coupling"],
        1.0
    )
    norm_upward_rank = robust_normalize(upward_rank)
    boosted_rank_score = critical_gate * norm_upward_rank

    # 6. Anti-starvation: saturating exponential wait gain
    wait_gain = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    norm_wait_gain = robust_normalize(wait_gain)

    # Composite score: smaller = higher priority
    score = (
        # Primary: deadline risk dominates — gated and amplified
        1.0 * robust_normalize(gated_slack_penalty)
        # Secondary: favor low energy density (efficiency per second)
        + PARAMS["energy_density_weight"] * norm_energy_density
        # Tertiary: prefer shorter durations
        + (1.0 - PARAMS["energy_density_weight"]) * norm_duration
        # Quaternary: boost rank only under joint high-risk condition
        - robust_normalize(boosted_rank_score)
        # Quinary: anti-starvation via wait gain
        - norm_wait_gain
    )

    # Final safeguard: ensure finite, shape-(N,) output
    return np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
