def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: Slack-aware criticality gating + robust fuzzy energy scaling + anti-starvation exponential wait boost.
    
    Key innovations:
    - Uses smooth, bounded slack penalty: sign(slack) * |slack|^slacksens, with strong amplification for negative slack
    - Introduces 'criticality boost' only when slack <= 0 OR uncertainty > median, to avoid over-prioritizing non-risky critical tasks
    - Normalizes energy via IQR-based robust scaling (not mean), avoiding outlier distortion
    - Adds duration ratio term to distinguish compute-bound vs comm-bound tasks — avoids conflating short exec + long comm as 'fast'
    - Applies exp(wait_decay * ready_wait_time) for gentle, numerically stable anti-starvation (no linear bias)
    - All divisions guarded; all NaN/inf replaced via nan_to_num; no unbounded ops.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization using interquartile range to resist outliers
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q1, q3 = np.quantile(x, [0.25, 0.75])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Slack pressure: signed power law — sharp rise for negative slack, flat positive regime
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    # Avoid 0^0: use eps for zero slack
    slack_power = np.where(
        slack_abs < eps,
        0.0,
        slack_sign * (slack_abs ** PARAMS["slack_sensitivity"])
    )
    slack_penalty = PARAMS["slack_penalty_scale"] * slack_power

    # Criticality boost: only activate when under deadline risk or high uncertainty
    median_uncertainty = np.median(uncertainty) if len(uncertainty) > 1 else np.mean(uncertainty)
    risk_condition = (slack <= eps) | (uncertainty > median_uncertainty + eps)
    boosted_rank = upward_rank * (1.0 + PARAMS["criticality_boost"] * risk_condition.astype(float))

    # Robust energy normalization (IQR-based, not mean-based)
    energy_norm = robust_normalize(min_incremental_energy)
    energy_score = PARAMS["energy_normalizer"] * energy_norm

    # Duration ratio: compute-dominance indicator [0,1]; favors compute-heavy tasks for edge offload
    exec_plus_comm = min_exec_time + min_comm_time + eps
    duration_ratio = min_exec_time / exec_plus_comm
    ratio_score = PARAMS["duration_ratio_weight"] * (1.0 - duration_ratio)  # prefer higher compute ratio → lower score

    # Anti-starvation: exponential wait boost — gentle, monotonic, bounded growth
    wait_boost = np.exp(-PARAMS["wait_decay"] * ready_wait_time)  # smaller → higher priority for long-waiting

    # Uncertainty-slack coupling: multiplicative gate on slack penalty when uncertainty is high
    unc_slack_gate = 1.0 + PARAMS["uncertainty_slack_coupling"] * (uncertainty / (np.mean(uncertainty) + eps))
    gated_slack = slack_penalty * unc_slack_gate

    # Assemble final score: all terms contribute additively; smaller = better
    score = (
        gated_slack
        + energy_score
        - robust_normalize(boosted_rank)  # promote critical tasks (negative contribution)
        + ratio_score
        + wait_boost  # wait_boost ∈ (0,1], so adding it raises score → lowers priority; but we want *longer* wait → *higher* priority → so invert logic: use 1 - wait_boost? No — instead, use -wait_boost since exp(-t) decays; but we want *more* priority for longer t → so use -exp(-t) is wrong. Instead: define wait_priority = 1 - exp(-t), then subtract it → lower score. So: - (1 - exp(-t)) = exp(-t) - 1 → always ≤ 0. Better: use -wait_boost to *reduce* score for long waits → yes, that gives higher priority. So: -wait_boost.
        - wait_boost  # ← correct: longer wait → larger wait_boost → more negative contribution → lower score → higher priority
    )

    # Ensure finite output and shape compliance
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
