def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: restores discriminative power of upward_rank via non-normalized scaling,
       enforces hard deadline gating with tunable threshold, and eliminates unsafe energy discounting.
       
       Key structural improvements:
       - Replaces soft `rank_slack_gate_threshold` clipping with hard binary gate at `slack_gate_threshold`
         to preserve urgency when slack ≤ threshold — directly addressing reflection's DDL feasibility concern.
       - Applies `rank_preservation_factor` *before* any normalization to retain raw rank magnitude;
         `upward_rank` is now scaled but NOT normalized, preserving its inherent critical-path ordering.
       - Removes all energy discounting logic (`energy_slack_interaction`, `energy_discount_floor_factor`)
         to prevent masking of deadline risk — strictly separates feasibility enforcement from energy optimization.
       - Keeps robust quantile normalization only for energy and slack-derived terms; upward_rank remains unnormalized.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization: only for energy and slack-related terms — not for rank
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q1, q3 = np.quantile(x, [PARAMS["q1_quantile"], PARAMS["q3_quantile"]])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr

    # Slack pressure: signed power law — sharp rise for negative slack, flat positive regime
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_power = np.where(
        slack_abs < eps,
        0.0,
        slack_sign * (slack_abs ** PARAMS["slack_sensitivity"])
    )
    base_slack_penalty = PARAMS["slack_penalty_scale"] * slack_power

    # Uncertainty-slack coupling: multiplicative gate on slack penalty
    unc_mean = np.mean(uncertainty) + eps
    unc_ratio = uncertainty / unc_mean
    gated_slack = base_slack_penalty * (1.0 + PARAMS["uncertainty_slack_coupling"] * unc_ratio)

    # Hard deadline gating for criticality: activate boost only when slack <= threshold
    # This preserves sharp urgency at zero/negative slack — no dilution near deadline
    slack_risk_mask = (slack <= PARAMS["slack_gate_threshold"]).astype(float)
    boosted_rank = upward_rank * (1.0 + PARAMS["criticality_boost"] * slack_risk_mask)
    # Preserve raw scale of boosted_rank: apply only linear scaling, no normalization
    rank_score = -PARAMS["rank_preservation_factor"] * boosted_rank  # negative → higher priority

    # Energy term: robustly normalized, no slack discounting (avoids feasibility violation)
    energy_norm = robust_normalize(min_incremental_energy)
    energy_score = PARAMS["energy_normalizer"] * energy_norm

    # Duration ratio: compute-dominance indicator [0,1]; favors compute-heavy tasks for edge offload
    exec_plus_comm = min_exec_time + min_comm_time + eps
    duration_ratio = min_exec_time / exec_plus_comm
    ratio_score = PARAMS["duration_ratio_weight"] * (1.0 - duration_ratio)

    # Anti-starvation: exponential wait boost — gentle, monotonic, bounded growth
    wait_boost = np.exp(-PARAMS["wait_decay"] * ready_wait_time)

    # Assemble final score: smaller = higher priority
    score = (
        gated_slack
        + energy_score
        + rank_score
        + ratio_score
        - wait_boost
    )

    # Ensure finite output and shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
