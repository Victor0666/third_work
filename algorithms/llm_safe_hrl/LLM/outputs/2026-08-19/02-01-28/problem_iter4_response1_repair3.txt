def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces additive feasibility guard with conditional DDL-protection gate,
    replaces work suppression with upward-rank × remaining-work critical leverage,
    uses bounded sigmoid instead of power-law for slack risk,
    and applies host-load–conditional energy suppression via ready_wait_time interaction.
    
    Structural changes:
    - Replaced isolated rank boost with upward_rank × remaining_work interaction (per counterfactual evidence)
    - Replaced piecewise slack penalty with smooth, bounded tanh-based DDL gate
    - Introduced host-load–conditional energy suppression: energy term is downweighted when ready_wait_time is high
      (i.e., system is congested → avoid marginal energy cost unless urgent)
    - Added successor-release interaction: penalizes tasks whose descendants have large remaining_work *and*
      are not yet ready — indicating critical path blocking
    - All normalizations use mean-abs scaling (more stable than median/MAD for small N) with epsilon safeguard
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Mean-abs normalization (stable, avoids median degeneracy for small N)
    def mean_abs_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.mean(x)
        spread = np.mean(np.abs(x - center)) + eps
        return (x - center) / (spread + eps)

    norm_slack = mean_abs_normalize(slack)
    norm_energy = mean_abs_normalize(min_incremental_energy)
    norm_duration = mean_abs_normalize(min_exec_time + min_comm_time)
    norm_rank = mean_abs_normalize(upward_rank)
    norm_work = mean_abs_normalize(remaining_work)
    norm_wait = mean_abs_normalize(ready_wait_time)
    norm_uncert = mean_abs_normalize(uncertainty)

    # --- Bounded tanh-based DDL protection gate: monotonic, smooth, numerically stable
    # Maps slack ∈ (-∞, ∞) → [0, 1], steepening near zero; negative slack gets strong weight
    ddl_gate = PARAMS["tanh_half_scale"] * (1.0 - np.tanh(PARAMS["ddl_gate_steepness"] * norm_slack))

    # --- Critical path leverage: upward_rank × remaining_work interaction, activated only under slack pressure
    # Matches counterfactual evidence: 'add_upward_rank_remaining_work_interaction' is high-confidence action
    critical_leverage = norm_rank * norm_work
    leverage_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    critical_score = -critical_leverage * leverage_gate * PARAMS["criticality_scale"]

    # --- Duration-uncertainty coupling under joint tightness: active only when slack is tight AND uncertainty high
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) & 
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]), 
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Host-load–conditional energy suppression: suppress energy term when queue is saturated
    # Ready wait time serves as proxy for host load; suppress energy cost when long-waiting tasks accumulate
    wait_ratio = np.clip(ready_wait_time / (PARAMS["wait_saturation_time"] + eps), 0.0, 10.0)
    energy_suppression = 1.0 / (1.0 + wait_ratio)  # decays from 1.0 → 0.09 as wait grows
    energy_term = norm_energy * energy_suppression * PARAMS["energy_sensitivity"]

    # --- Anti-starvation: bounded linear benefit capped at saturation
    wait_benefit = np.clip(ready_wait_time / (PARAMS["wait_saturation_time"] + eps), 0.0, 1.0)

    # --- Successor release interaction: penalize tasks whose large-work descendants aren't ready yet
    # Encourages releasing bottlenecks earlier — matches 'add_successor_release_interaction' evidence
    # Heuristic: if remaining_work > threshold AND slack is tight → add penalty to delay release
    successor_gate = np.where(
        (remaining_work > PARAMS["work_leverage_threshold"]) & 
        (slack < PARAMS["slack_negative_threshold"]), 
        1.0, 0.0
    )
    successor_penalty = successor_gate * ddl_gate * PARAMS["successor_penalty_weight"]

    # --- Final score: smaller = higher priority
    # DDL risk dominates via ddl_gate; critical leverage reduces score (boosts priority); others add cost
    score = (
        ddl_gate * PARAMS["ddl_gate_scale"]        # Primary DDL enforcement (scaled for dominance)
        + critical_score                          # Boosts critical-path tasks (negative = good)
        + dur_uncert_interaction                  # Penalty for uncertain durations under tight slack
        + energy_term                             # Energy cost, suppressed under congestion
        - wait_benefit                            # Anti-starvation (reduces score)
        + successor_penalty                       # Penalizes delaying large successors
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
