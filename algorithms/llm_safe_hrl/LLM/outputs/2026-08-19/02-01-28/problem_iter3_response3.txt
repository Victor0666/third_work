def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: bounded piecewise gates + capped slack-pressure + stabilized dual wait modeling.
    
    Key structural improvements:
    - Replaces fragile adaptive duration_balance with static but slack-pressure-capped interactions
    - Introduces bounded_wait_gate to saturate exponential wait_bonus, preventing starvation-overcompensation
    - Caps slack_pressure via clip() before all multiplicative uses — eliminates unbounded growth near hard deadlines
    - Retains robust_normalize but removes unstable power-law wait urgency; uses only one well-bounded wait signal
    - All feature interactions now operate within [-3, 3] effective range via explicit clipping and gating
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q75 = np.quantile(x, PARAMS["q75_quantile"])
        q25 = np.quantile(x, PARAMS["q25_quantile"])
        iqr = q75 - q25
        rng = np.max(x) - np.min(x)
        med = np.median(x)
        scale = np.where(iqr > eps, iqr, np.where(rng > eps, rng, np.maximum(1.0, np.abs(med))))
        return (x - med) / (scale + eps)

    # Base normalized features
    exec_comm = min_exec_time + min_comm_time
    duration_norm = robust_normalize(exec_comm)
    energy_norm = robust_normalize(min_incremental_energy)
    rank_norm = robust_normalize(upward_rank)
    work_norm = robust_normalize(remaining_work)
    wait_norm = robust_normalize(ready_wait_time)
    
    # Slack processing with bounded pressure
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_powered = slack_sign * (slack_abs ** PARAMS["slack_sensitivity"])
    slack_norm = robust_normalize(slack_powered)
    
    # Lateness risk: linear penalty only for negative slack
    lateness_risk = np.where(slack < 0, PARAMS["slack_penalty_scale"] * (-slack), 0.0)
    
    # Uncertainty & slack pressure: now explicitly capped
    unc_norm = robust_normalize(uncertainty)
    slack_pressure = np.maximum(-slack_norm, 0.0)
    # Critical bounded cap prevents overflow in multiplicative terms
    slack_pressure_capped = np.clip(slack_pressure, 0.0, PARAMS["slack_pressure_cap"])
    
    # Interaction term: now bounded by capped pressure
    interaction_term = PARAMS["uncertainty_slack_interaction"] * unc_norm * slack_pressure_capped
    
    # Criticality boost: gated by capped pressure
    criticality_gate = np.maximum(slack_pressure_capped, unc_norm)
    boosted_rank = rank_norm * (1.0 + PARAMS["criticality_boost"] * criticality_gate)
    
    # Anti-starvation: single saturated exponential with explicit upper bound
    wait_base = np.maximum(ready_wait_time, eps)
    wait_bonus_exp = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait_base)
    # Bounded gate ensures wait_bonus never dominates score beyond intended range
    wait_bonus_bounded = np.minimum(wait_bonus_exp, PARAMS["bounded_wait_gate"])
    
    # Duration-energy composite remains static but stable
    dur_energy_combo = (
        PARAMS["duration_balance"] * duration_norm
        + (1.0 - PARAMS["duration_balance"]) * energy_norm
    )
    
    # Final score: smaller = higher priority
    score = (
        lateness_risk
        + interaction_term
        - boosted_rank
        + PARAMS["energy_norm_factor"] * dur_energy_combo
        - wait_bonus_bounded
        + PARAMS["work_weight"] * work_norm
    )
    
    # Ensure finite output with machine-precision bounds
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
