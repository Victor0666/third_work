def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces additive feasibility guard with conditional DDL-protection gate,
    introduces upward_rank × remaining_work interaction (validated by counterfactual evidence),
    introduces successor release pressure, uses mean-abs scaling, and replaces arctan wait benefit with sigmoid.
    
    Key structural changes:
      - Replaces hard feasibility_guard with soft 'ddl_protection' term activated only when slack < slack_negative_threshold.
      - Introduces 'successor_release_pressure' term: upward_rank × (negative_slack_mask) weighted by successor_release_weight.
      - Uses mean-abs normalization for speed and determinism.
      - Replaces arctan wait benefit with smooth sigmoid centered at wait_sigmoid_center and scaled by wait_sigmoid_scale.
      - All numeric literals are -2,-1,0,1,2 or derived from PARAMS; no hidden constants."""
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Mean-abs normalization: faster, deterministic, and stable under small N (e.g., N=1)
    def mean_abs_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.mean(x)
        spread = np.mean(np.abs(x - center)) + eps
        return (x - center) / (spread + eps)

    norm_slack = mean_abs_normalize(slack)
    norm_energy = mean_abs_normalize(min_incremental_energy)
    norm_duration = mean_abs_normalize(min_exec_time + min_comm_time)
    norm_rank = mean_abs_normalize(upward_rank)
    norm_work = mean_abs_normalize(remaining_work)
    norm_wait = mean_abs_normalize(ready_wait_time)
    norm_uncert = mean_abs_normalize(uncertainty)

    # --- Slack risk score: bounded sigmoid gating for monotonic DDL pressure
    # Smooth, differentiable, and bounded (-1,1) → maps negative slack to strong positive penalty
    slack_denom = np.abs(np.mean(slack)) + eps
    slack_sigmoid = 1.0 / (1.0 + np.exp(-slack / slack_denom))
    slack_risk_score = (1.0 - slack_sigmoid) * PARAMS["slack_penalty_exponent"]

    # --- Critical path leverage: upward_rank × remaining_work interaction (counterfactual-validated)
    critical_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    critical_leverage = norm_rank * norm_work * critical_gate * PARAMS["criticality_scale"]

    # --- Successor release pressure: prioritize tasks that unblock many high-risk descendants
    negative_slack_mask = np.where(slack < 0, 1.0, 0.0)
    successor_release_pressure = norm_rank * negative_slack_mask * PARAMS["criticality_scale"] * PARAMS["successor_release_weight"]

    # --- Duration-uncertainty coupling: active only under joint tightness (tight slack AND high uncertainty)
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) & 
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]), 
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Bounded starvation relief: sigmoid-saturated wait benefit
    wait_scaled = np.clip(ready_wait_time / (PARAMS["wait_saturation_time"] + eps), 0.0, PARAMS["wait_sigmoid_scale"])
    wait_benefit = 1.0 / (1.0 + np.exp(-wait_scaled + PARAMS["wait_sigmoid_center"]))

    # --- DDL protection term: soft but decisive boost for tasks threatening deadline feasibility
    ddl_protection = np.where(slack < PARAMS["slack_negative_threshold"] * slack_denom,
                              1.0 + np.abs(slack) / slack_denom,
                              0.0)

    # --- Final score: smaller = higher priority; DDL terms dominate; no unbounded components
    score = (
        slack_risk_score                          # Primary DDL urgency (bounded sigmoid)
        + PARAMS["energy_sensitivity"] * norm_energy  # Secondary energy minimization
        + dur_uncert_interaction                  # Uncertainty-coupled duration penalty
        - critical_leverage                       # Leverage-sensitive critical path boost
        - successor_release_pressure              # Bottleneck-release pressure
        - wait_benefit                            # Anti-starvation benefit
        + ddl_protection                          # Soft DDL protection
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
