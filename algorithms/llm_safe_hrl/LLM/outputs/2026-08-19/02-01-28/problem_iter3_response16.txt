def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with deadline-aware structural gating and topological urgency.

    Key improvements:
    - Replaces smooth ramps with bounded linear ramp over [critical_slack_threshold - epsilon, critical_slack_threshold]
      for interpretable, numerically stable DDL boundary enforcement.
    - Introduces hard structural gating: under critical slack (slack < critical_slack_threshold),
      priority is *dominated* by energy efficiency ratio and critical-path fraction (upward_rank / max_upward_rank),
      ignoring fairness and robustness to enforce feasibility first.
    - Uses min-max normalized critical-path fraction instead of log1p(remaining_work) — directly encodes
      topological urgency without arbitrary reference scaling or outlier sensitivity.
    - All feature normalizations use median/MAD; fallbacks avoid degenerate stats.
    - No unbounded ops: all divisions guarded, all exponents bounded, no hidden constants beyond [-2,2].
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Defensive cast & copy
    min_exec_time = np.asarray(min_exec_time, dtype=float).copy()
    min_comm_time = np.asarray(min_comm_time, dtype=float).copy()
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float).copy()
    slack = np.asarray(slack, dtype=float).copy()
    upward_rank = np.asarray(upward_rank, dtype=float).copy()
    remaining_work = np.asarray(remaining_work, dtype=float).copy()
    ready_wait_time = np.asarray(ready_wait_time, dtype=float).copy()
    uncertainty = np.asarray(uncertainty, dtype=float).copy()
    
    # === Robust normalization: MAD-based, with safe fallbacks ===
    def robust_normalize(x):
        x = np.abs(x)
        if N == 1:
            center = x[0]
            scale = eps
        else:
            center = np.median(x)
            mad = np.median(np.abs(x - center))
            scale = np.maximum(mad, eps)
        return (x - center) / scale
    
    # === Critical-path fraction: upward_rank normalized by global max → topological urgency ===
    max_upward_rank = np.max(upward_rank) if N > 0 else 1.0
    cp_fraction = upward_rank / np.maximum(max_upward_rank, eps)
    
    # === 1. Slacker-pressure: bounded linear ramp near threshold (replaces quadratic) ===
    # Linear ramp from 0 to slack_risk_penalty over interval [critical_slack_threshold - eps, critical_slack_threshold]
    ramp_start = PARAMS["critical_slack_threshold"] - eps
    ramp_region = np.logical_and(
        slack >= ramp_start,
        slack <= PARAMS["critical_slack_threshold"]
    )
    # Base penalty: zero above threshold, full penalty below ramp_start, linear in between
    slack_penalty_base = np.where(
        slack < ramp_start,
        -slack * PARAMS["slack_risk_penalty"],
        np.where(
            ramp_region,
            PARAMS["slack_risk_penalty"] * (PARAMS["critical_slack_threshold"] - slack) / eps,
            0.0
        )
    )
    # Coupling: only amplify where uncertainty is high AND slack is tight
    slack_pressure = slack_penalty_base * (
        1.0 + PARAMS["uncertainty_slack_coupling"] * np.tanh(uncertainty)
    )
    
    # === 2. Energy efficiency: joules per second, uncertainty-modulated ===
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_efficiency = min_incremental_energy / duration
    energy_eff_score = energy_efficiency * (1.0 + PARAMS["energy_uncertainty_interaction"] * uncertainty) if "energy_uncertainty_interaction" in PARAMS else energy_efficiency
    energy_eff_norm = robust_normalize(energy_eff_score)
    
    # === 3. Critical path: use cp_fraction directly (no sqrt compression) — now properly scaled [0,1] ===
    rank_score = PARAMS["rank_sensitivity"] * cp_fraction
    
    # === 4. Anti-starvation: sublinear wait boost ===
    wait_boost = np.power(np.maximum(ready_wait_time, eps), PARAMS["wait_fairness_exponent"])
    wait_score = robust_normalize(wait_boost)
    
    # === 5. Duration robustness: CV of exec+comm → invert for preference ===
    duration_cv = np.std(np.stack([min_exec_time, min_comm_time]), axis=0) / duration
    duration_robust_score = -robust_normalize(duration_cv)
    
    # === 6. Critical-mode gating: binary switch at critical_slack_threshold ===
    is_critical = slack < PARAMS["critical_slack_threshold"]
    
    # In critical mode: prioritize *only* energy efficiency and critical-path fraction
    critical_score = (
        PARAMS["critical_mode_energy_weight"] * energy_eff_norm
        + PARAMS["critical_mode_work_weight"] * rank_score
    )
    
    # Outside critical mode: blend fairness, robustness, and rank with tunable balance
    blended_score = (
        (1.0 - PARAMS["fairness_robustness_blend"]) * wait_score
        + PARAMS["fairness_robustness_blend"] * duration_robust_score
        + rank_score
    )
    
    # Assemble final score: smaller = better
    score = np.where(
        is_critical,
        critical_score,
        slack_pressure + PARAMS["energy_efficiency_bias"] * energy_eff_norm + blended_score
    )
    
    # Safeguard: ensure finite, shape-(N,), no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Final shape assertion
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    
    return score
