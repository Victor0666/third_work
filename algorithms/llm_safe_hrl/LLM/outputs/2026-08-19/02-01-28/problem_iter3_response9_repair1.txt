def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with deadline-aware work-density normalization,
    contextual [0,1] min-max scaling, and feasibility-preserving slack guard.
    
    Key self-evolution improvements:
      - Replaces global robust_normalize with bounded min-max scaling per feature → prevents rank inversion under skew
      - Introduces deadline-aware work density: (work_density) / (|slack| + eps)^exponent → strongly boosts high-work tasks *only* when slack is tight
      - Adds feasibility slack guard: clamps coupled_slack to zero when |norm_slack| > threshold → enforces hard deadline safety before energy tradeoffs
      - All feature scalings are contextually bounded [0,1]; divisions guarded; no literals beyond {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Defensive casting and safe NaN/inf handling
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)

    # Contextual min-max normalization: maps each feature to [0, 1] using its own range
    def minmax_normalize(x):
        x = np.asarray(x, dtype=float)
        x_min = np.min(x)
        x_max = np.max(x)
        denom = x_max - x_min
        denom_safe = np.where(denom == 0.0, eps_safe, denom)
        return np.clip((x - x_min) / denom_safe, 0.0, 1.0)

    # Duration and derived signals
    duration = min_exec_time + min_comm_time
    norm_duration = minmax_normalize(duration)
    norm_slack = minmax_normalize(slack)
    norm_energy = minmax_normalize(min_incremental_energy)
    norm_rank = minmax_normalize(upward_rank)
    norm_work = minmax_normalize(remaining_work)
    norm_wait = minmax_normalize(ready_wait_time)
    norm_uncert = minmax_normalize(uncertainty)

    # Work density: compute work per time unit
    work_density = remaining_work / (duration + eps_safe)
    norm_work_density = minmax_normalize(work_density)

    # Deadline-aware work density: amplifies priority for high-work tasks *only* under tight slack
    # Uses exponent to sharpen sensitivity near deadline
    deadline_aware_work_density = work_density / ((np.abs(slack) + eps_safe) ** PARAMS["rank_slack_interaction_exponent"])
    norm_deadline_aware_work_density = minmax_normalize(deadline_aware_work_density)

    # Slack gating: smooth sigmoid transition at zero slack
    slack_gate = 1.0 / (1.0 + np.exp(-2.0 * (2.0 * norm_slack - 1.0)))  # Sigmoid centered at norm_slack=0.5 for better resolution in [0,1]
    slack_penalty_term = PARAMS["slack_risk_penalty"] * (1.0 - slack_gate) * np.abs(norm_slack)
    slack_gain_term = PARAMS["slack_benefit_gain"] * slack_gate * np.abs(norm_slack)
    coupled_slack = (slack_penalty_term + slack_gain_term) * (
        1.0 + PARAMS["uncertainty_slack_coupling"] * norm_uncert
    )

    # Feasibility-preserving slack guard: suppress slack penalty when slack is ample (avoid over-penalizing safe tasks)
    slack_feasibility_mask = np.where(np.abs(norm_slack) <= PARAMS["slack_gate_threshold"], 1.0, 0.0)
    coupled_slack = coupled_slack * slack_feasibility_mask

    # Critical path importance: higher upward_rank → higher priority → negate for min-score
    critical_term = -PARAMS["critical_path_weight"] * norm_rank

    # Work density efficiency: prefer high work-per-time, now deadline-aware
    work_density_term = -PARAMS["work_density_weight"] * norm_deadline_aware_work_density

    # Energy density: marginal energy per duration → lower is better
    energy_density = min_incremental_energy / (duration + eps_safe)
    norm_energy_density = minmax_normalize(energy_density)
    # Uncertainty-coupled energy preference: reduce weight where uncertainty is high
    energy_density_term = PARAMS["energy_density_weight"] * norm_energy_density * (
        1.0 - PARAMS["energy_uncertainty_coupling"] * norm_uncert
    )

    # Starvation relief: longer wait → higher priority
    starvation_term = -PARAMS["wait_starvation_relief"] * norm_wait

    # Duration flexibility bias: quadratic penalty around |norm_duration| == 0.5 (center of [0,1])
    flex_penalty = PARAMS["duration_flexibility_bias"] * ((norm_duration - 0.5) ** 2)

    # Rank-slack interaction: amplifies priority for high-rank tasks *under deadline pressure*
    # Only active when normalized slack magnitude is below threshold
    rank_slack_interaction = norm_rank * (np.abs(norm_slack) ** PARAMS["rank_slack_interaction_exponent"])
    interaction_gate = np.where(np.abs(norm_slack) < PARAMS["slack_gate_threshold"], 1.0, 0.0)
    interaction_term = -PARAMS["critical_path_weight"] * rank_slack_interaction * interaction_gate

    # Assemble final score: smaller = higher priority
    score = (
        coupled_slack
        + critical_term
        + work_density_term
        + energy_density_term
        + starvation_term
        + flex_penalty
        + interaction_term
    )

    # Final safeguard: ensure finite, bounded output
    return np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
