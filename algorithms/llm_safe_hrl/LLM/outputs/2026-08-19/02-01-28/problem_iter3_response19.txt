def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces unstable power-law couplings with a hard feasibility guard,
    simplifies starvation relief to single saturating form, removes redundant work leverage,
    and enforces strict DDL-constraint prioritization before energy minimization.
    
    Key structural improvement: introduces `feasibility_guard` — a deterministic binary gate
    that elevates priority (via large additive penalty) for any task whose slack < -eps AND
    remaining_work > work_leverage_threshold AND uncertainty > uncertainty_activation_threshold,
    i.e., high-risk, high-work, high-uncertainty tasks that threaten deadline feasibility.
    This ensures hard constraint satisfaction dominates energy optimization."""
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Robust per-feature normalization using median/MAD
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        spread = np.median(np.abs(x - center)) + eps
        return (x - center) / (spread + eps)

    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # --- Slack risk score: piecewise smooth penalty (same as v0 but more stable)
    slack_abs = np.abs(slack)
    negative_penalty = np.where(slack < 0, 
                                np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"]), 
                                0.0)
    positive_relief = np.where(slack >= 0, 
                               -slack_abs / (1.0 + slack_abs + eps), 
                               0.0)
    slack_risk_score = negative_penalty + positive_relief

    # --- Criticality amplification: only under severe slack pressure
    critical_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    amplified_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * critical_gate)

    # --- Duration-uncertainty coupling: active only under tight slack AND high uncertainty
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) & 
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]), 
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Bounded starvation relief: single saturating arctan form (more stable than hybrid)
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_benefit = (2.0 / np.pi) * np.arctan(wait_scaled)

    # --- Feasibility guard: hard penalty for high-risk, high-work, high-uncertainty tasks
    # Activates only when all three danger conditions hold — prevents deadline violation
    guard_condition = (
        (slack < -eps) &
        (remaining_work > PARAMS["work_leverage_threshold"]) &
        (uncertainty > PARAMS["uncertainty_activation_threshold"])
    )
    feasibility_guard = np.where(guard_condition, 
                                 PARAMS["feasibility_guard_weight"], 
                                 0.0)

    # --- Work-based suppression: deprioritize large remaining_work *unless* guarded
    # Smooth sigmoid suppression to avoid step discontinuities
    work_ratio = np.clip(remaining_work / (PARAMS["work_leverage_threshold"] + eps), 
                         np.finfo(float).tiny, 
                         np.finfo(float).max)
    work_suppression = 1.0 / (1.0 + work_ratio)

    # --- Final score: smaller = better; feasibility guard dominates
    score = (
        slack_risk_score                              # Primary urgency signal
        + PARAMS["energy_sensitivity"] * norm_energy  # Energy minimization (secondary)
        + dur_uncert_interaction                      # Uncertainty-aware duration penalty
        - amplified_rank                              # Rank boost under stress
        - wait_benefit                                # Anti-starvation
        + work_suppression                            # Large work → lower priority (soft)
        + feasibility_guard                           # Hard feasibility enforcement
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
