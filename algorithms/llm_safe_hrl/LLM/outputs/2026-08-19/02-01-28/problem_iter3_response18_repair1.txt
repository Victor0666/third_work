def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with hard DDL feasibility guard and simplified critical-path coupling.
    
    Key improvements:
      - Hard feasibility guard: all non-urgency terms scaled by max(0, slack)^p → strictly zero when slack ≤ 0.
      - Linear rank-work coupling (upward_rank * remaining_work) preserves discriminative power under tight deadlines.
      - Explicit normalized uncertainty penalty for risk-aware scheduling.
      - All declared parameters are used; no unused or missing entries.
      - No numeric literals except -2,-1,0,1,2; epsilon handled via PARAMS["epsilon"] and np.finfo for inf/nan guards.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Robust normalization using median absolute deviation (MAD)
    def normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        dev = np.median(np.abs(x - center))
        scale = np.maximum(dev, eps)
        return (x - center) / scale

    # Normalize inputs
    norm_slack = normalize(slack)
    norm_energy = normalize(min_incremental_energy)
    norm_rank = normalize(upward_rank)
    norm_work = normalize(remaining_work)
    norm_wait = normalize(ready_wait_time)
    norm_uncert = normalize(uncertainty)
    norm_duration = normalize(min_exec_time + min_comm_time)

    # --- Smooth urgency: sigmoid response to slack ---
    urgency = 1.0 / (1.0 + np.exp(-PARAMS["slack_urgency_scale"] * norm_slack))

    # --- Hard feasibility guard: zero-out non-urgency contributions when slack <= 0 ---
    feasibility_guard = np.clip(slack, 0.0, np.inf) ** PARAMS["feasibility_guard_power"]
    guard_max = np.maximum(feasibility_guard.max(), eps)
    guard_norm = feasibility_guard / guard_max

    # --- Linear critical-path coupling: upward_rank * remaining_work (no saturation) ---
    rank_work_coupling = norm_rank * norm_work * PARAMS["rank_work_linear_coupling"]

    # --- Duration-uncertainty coupling under urgency pressure ---
    dur_uncert_interaction = (
        norm_duration * norm_uncert *
        urgency * PARAMS["duration_uncertainty_coupling"]
    )

    # --- Standalone uncertainty penalty (risk-aware even without coupling) ---
    uncert_penalty = norm_uncert * PARAMS["uncertainty_penalty_weight"]

    # --- Anti-starvation boost: bounded, saturating benefit from wait time ---
    wait_boost = np.minimum(norm_wait, 2.0) * PARAMS["wait_boost_weight"]

    # --- Final priority score: smaller = higher priority ---
    score = (
        urgency
        + guard_norm * (
            PARAMS["energy_sensitivity"] * norm_energy
            + dur_uncert_interaction
            + uncert_penalty
            + rank_work_coupling
        )
        - wait_boost
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    score = np.clip(score, PARAMS["clip_lower_bound_log_scale"], PARAMS["clip_upper_bound"])

    # Enforce shape (N,)
    if score.ndim == 0:
        score = np.full(1, score.item(), dtype=float)

    return score
