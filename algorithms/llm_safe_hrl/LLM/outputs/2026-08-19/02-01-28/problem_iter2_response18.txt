def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robust urgency modeling, successor-release coupling,
    and host-load–conditional suppression — all parameterized for CMA-ES co-evolution.
    
    Key innovations:
      - Smooth sigmoid urgency (not piecewise) for stable gradient-aware optimization.
      - Successor-work gate: upward_rank * (1 - exp(-remaining_work / work_ref)) → prioritizes
        critical-path tasks *with high descendant load* under tight deadlines.
      - Host-load suppression: uses external `host_load` proxy via normalized slack & uncertainty
        as counterfactual proxy (since true host_load not provided; inferred from risk pressure).
      - All numeric literals restricted to {-2,-1,0,1,2}; no hidden constants.
      - Robust MAD-based normalization and strict finite-output guarantees.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Robust normalization using median absolute deviation (MAD)
    def normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        dev = np.median(np.abs(x - center))
        scale = np.maximum(dev, eps)
        return (x - center) / scale

    # Normalize inputs
    norm_slack = normalize(slack)
    norm_energy = normalize(min_incremental_energy)
    norm_rank = normalize(upward_rank)
    norm_work = normalize(remaining_work)
    norm_wait = normalize(ready_wait_time)
    norm_uncert = normalize(uncertainty)
    norm_duration = normalize(min_exec_time + min_comm_time)

    # --- Smooth urgency: sigmoid response to slack (no brittle thresholds) ---
    # Higher score = more urgent; maps [-inf,+inf] -> (0,1)
    urgency = 1.0 / (1.0 + np.exp(-PARAMS["slack_urgency_scale"] * norm_slack))

    # --- Critical slack mask (soft gate for rank/work coupling) ---
    # Activates when slack is near or below threshold (0 → 1 as norm_slack decreases)
    critical_mask = np.clip(
        1.0 - (norm_slack - PARAMS["critical_slack_threshold"]) / (np.abs(norm_slack).max() + eps),
        0.0, 1.0
    )

    # --- Successor-release interaction: prioritize critical tasks with high descendant workload ---
    # Saturating work influence: logistic-like gate using work_ref_scale
    work_gate = 1.0 - np.exp(-np.maximum(remaining_work, eps) / (PARAMS["work_ref_scale"] + eps))
    successor_coupled_rank = norm_rank * work_gate * critical_mask * PARAMS["successor_work_gate"]

    # --- Duration-uncertainty coupling under urgency pressure ---
    dur_uncert_interaction = (
        norm_duration * norm_uncert *
        urgency * PARAMS["duration_uncertainty_coupling"]
    )

    # --- Host-load–conditional suppression proxy ---
    # Since true host_load is unavailable, infer pressure from combined risk: low slack + high uncertainty
    # Proxy: 1 - (urgency * norm_uncert), clamped to suppress energy/uncertainty penalties when system is saturated
    load_proxy = 1.0 - np.clip(urgency * norm_uncert, 0.0, 1.0)
    suppression_factor = np.clip(
        1.0 - PARAMS["host_load_suppression"] * np.maximum(load_proxy - PARAMS["load_saturation_point"], 0.0),
        0.0, 1.0
    )

    # --- Anti-starvation boost: bounded, saturating benefit from wait time ---
    wait_boost = np.minimum(norm_wait, 2.0) * PARAMS["wait_boost_weight"]

    # --- Final priority score: smaller = higher priority ---
    # Dominant drivers: urgency (penalty), energy (minimize), coupled duration-uncertainty (penalty)
    # Suppressive: successor-coupled rank (promotes critical+heavy), wait_boost (relieves starvation)
    # Suppressed via load-proxy: energy & uncertainty terms scaled down under saturation
    score = (
        urgency
        + PARAMS["energy_sensitivity"] * norm_energy * suppression_factor
        + dur_uncert_interaction * suppression_factor
        + successor_coupled_rank
        - wait_boost
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    score = np.clip(score, PARAMS["clip_lower_bound_log_scale"], PARAMS["clip_upper_bound"])

    # Enforce shape (N,)
    if score.ndim == 0:
        score = np.full(1, score.item(), dtype=float)

    return score
