def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with rank-based normalization for outlier robustness,
    hard-threshold slack gating for strict deadline adherence, and simplified structure.
    Removes latency-energy coupling and sigmoid smoothness in favor of crisp DDL enforcement."""
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Rank-based normalization: map each feature to [0,1] via percentile ranks → invariant to outliers
    def rank_normalize(x):
        x = np.asarray(x, dtype=float)
        # Use quantile-based scaling: values below q_low → 0, above q_high → 1, linear in between
        q_low = np.quantile(x, 1.0 - PARAMS["rank_normalization_quantile"], method='linear')
        q_high = np.quantile(x, PARAMS["rank_normalization_quantile"], method='linear')
        denom = q_high - q_low + eps_safe
        normed = (x - q_low) / denom
        return np.clip(normed, 0.0, 1.0)

    # Compute core features
    duration = min_exec_time + min_comm_time
    norm_duration = rank_normalize(duration)
    norm_slack = rank_normalize(slack)
    norm_energy = rank_normalize(min_incremental_energy)
    norm_rank = rank_normalize(upward_rank)
    norm_work = rank_normalize(remaining_work)
    norm_wait = rank_normalize(ready_wait_time)
    norm_uncert = rank_normalize(uncertainty)

    # Work density: compute efficiency — remaining_work per time unit, rank-normalized
    work_density = remaining_work / (duration + eps_safe)
    norm_work_density = rank_normalize(work_density)

    # Hard-threshold slack gating: crisp transition at normalized slack = slack_hard_threshold
    # Ensures strong, unambiguous priority shift when slack falls below safety margin
    slack_below_threshold = (norm_slack < PARAMS["slack_hard_threshold"]).astype(float)
    slack_penalty_term = PARAMS["slack_risk_penalty"] * slack_below_threshold * (1.0 - norm_slack)
    slack_gain_term = 0.0  # Remove benefit gain: prioritize safety over flexibility under deadline pressure

    # Uncertainty-coupled slack penalty: amplifies only the penalty (not gain) where uncertainty is high
    coupled_slack = slack_penalty_term * (1.0 + PARAMS["uncertainty_slack_coupling"] * norm_uncert)

    # Critical path term: higher upward_rank → higher priority → subtract
    critical_term = -PARAMS["critical_path_weight"] * norm_rank

    # Work density efficiency: prefer high-compute-per-time tasks
    work_density_term = -PARAMS["work_density_weight"] * norm_work_density

    # Energy minimization: lower incremental energy → higher priority
    energy_term = PARAMS["energy_scale"] * norm_energy

    # Starvation relief: linear in normalized wait time (simpler, more interpretable than log1p)
    starvation_term = -PARAMS["wait_starvation_relief"] * norm_wait

    # Duration flexibility bias: quadratic penalty around target normalized duration
    flex_penalty = PARAMS["duration_flexibility_bias"] * ((norm_duration - PARAMS["duration_target_normalized"]) ** 2)

    # Assemble final priority score: smaller = higher priority
    score = (
        coupled_slack
        + critical_term
        + work_density_term
        + energy_term
        + starvation_term
        + flex_penalty
    )

    # Final numerical safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
