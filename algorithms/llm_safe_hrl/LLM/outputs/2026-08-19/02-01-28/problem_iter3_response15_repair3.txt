def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with DDL-protection gate and task-level relative scaling.
    
    Key structural changes:
    - Replaced global IQR normalization with task-level relative scaling: x / (median(x) + eps).
    - Added hard DDL-protection gate that forces top priority when normalized slack is critically low.
    - All numeric literals are now only -2, -1, 0, 1, 2; epsilon and override strength declared in schema.
    - Uses np.finfo for immutable safeguards instead of hardcoded constants.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Task-level relative scaling: x / (median(x) + eps) — replaces IQR normalization
    def relative_scale(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x) if len(x) > 1 else x[0]
        return x / (med + eps)
    
    # Slack pressure: signed power law, zero at slack=0
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_power = np.where(
        slack_abs < eps,
        0.0,
        slack_sign * (slack_abs ** PARAMS["slack_sensitivity"])
    )
    base_slack_penalty = PARAMS["slack_penalty_scale"] * slack_power
    
    # DDL-protection gate: binary override activated when normalized slack is critically low
    abs_slack_med = np.median(slack_abs) if len(slack_abs) > 1 else slack_abs[0]
    rel_slack = slack / (abs_slack_med + eps)
    ddl_gate_active = (rel_slack < -PARAMS["ddl_protection_threshold"]).astype(float)
    
    # Uncertainty-slack coupling: multiplicative gate scaled by relative uncertainty
    unc_med = np.median(uncertainty) if len(uncertainty) > 1 else uncertainty[0]
    unc_rel = uncertainty / (unc_med + eps)
    unc_slack_gate = 1.0 + PARAMS["uncertainty_slack_coupling"] * unc_rel
    gated_slack = base_slack_penalty * unc_slack_gate
    
    # Criticality boost: applied only under joint risk (tight slack OR high uncertainty)
    slack_risky = (slack <= eps).astype(float)
    unc_risky = (uncertainty > unc_med + eps).astype(float)
    risk_mask = np.clip(slack_risky + unc_risky, 0.0, 1.0)  # union, not product
    boosted_rank = upward_rank * (1.0 + PARAMS["criticality_boost"] * risk_mask)
    
    # Energy efficiency term — using relative scaling
    energy_scaled = relative_scale(min_incremental_energy)
    energy_score = PARAMS["energy_normalizer"] * energy_scaled
    
    # Remaining work × energy coupling: penalizes high-marginal-energy gateways to heavy subtrees
    work_energy_interaction = min_incremental_energy * (remaining_work + eps)
    work_energy_scaled = relative_scale(work_energy_interaction)
    work_energy_penalty = PARAMS["remaining_work_energy_coupling"] * work_energy_scaled
    
    # Duration ratio: compute vs communication dominance [0,1]; higher favors compute-heavy → edge preference
    exec_plus_comm = min_exec_time + min_comm_time + eps
    duration_ratio = min_exec_time / exec_plus_comm
    ratio_score = PARAMS["duration_ratio_weight"] * (1.0 - duration_ratio)
    
    # Anti-starvation: monotonic, bounded exponential decay
    wait_boost = np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # Assemble final score: smaller = higher priority
    # DDL-protection gate forces top priority (lowest score) when active: subtract large boost
    # Use finfo.max * PARAMS["ddl_override_strength"] to avoid hidden constants
    ddl_override_boost = ddl_gate_active * (-finfo.max * PARAMS["ddl_override_strength"])
    
    score = (
        gated_slack
        + energy_score
        + work_energy_penalty
        + ratio_score
        - relative_scale(boosted_rank)
        - wait_boost
        + ddl_override_boost
    )
    
    # Ensure finite output and shape compliance
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
