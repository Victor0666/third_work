def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: piecewise slack gating, IQR-normalized features, 
    criticality-boosted upward rank, and uncertainty-modulated deadline pressure.
    Prioritizes latency-critical tasks first, then balances energy and starvation."""
    
    # Safeguard epsilon from inputs
    eps = np.finfo(float).tiny if PARAMS["epsilon"] < np.finfo(float).tiny else PARAMS["epsilon"]
    
    # Convert to float arrays, avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # Robust IQR-based normalization: resistant to outliers, bounded, deterministic
    def iqr_normalize(x):
        x = np.clip(x, PARAMS["iqr_clip_lower"], PARAMS["iqr_clip_upper"])
        q1, q3 = np.quantile(x, [0.25, 0.75])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Duration = exec + comm; normalized robustly
    duration = min_exec_time + min_comm_time
    norm_duration = iqr_normalize(duration)
    
    # Slack normalization with *asymmetric* penalty: sharp rise for negative slack
    # Use soft sign + power law: |slack|^s * sign(slack), but prioritize negatives more
    slack_abs = np.abs(slack)
    # Avoid log(0); use clipped linear near zero for stability
    slack_sign = np.sign(slack)
    slack_magnitude = np.where(
        slack_abs < eps,
        0.0,
        np.power(np.clip(slack_abs, eps, PARAMS["iqr_clip_upper"]), PARAMS["slack_sensitivity"])
    )
    # Negative slack gets amplified; positive slack attenuated (less urgent)
    slack_pressure = -slack_sign * slack_magnitude  # now: smaller = better (negative → high priority)
    
    # Normalize slack_pressure for stable scaling
    norm_slack_pressure = iqr_normalize(slack_pressure)
    
    # Energy: already risk-adjusted; normalize robustly
    norm_energy = iqr_normalize(min_incremental_energy)
    
    # Upward rank: boost when slack is tight (≤ 0) OR uncertainty is high (> median)
    median_uncert = np.median(uncertainty) if N > 1 else uncertainty[0]
    criticality_gate = np.where(
        (slack <= 0.0) | (uncertainty > median_uncert + eps),
        PARAMS["criticality_boost"],
        1.0
    )
    norm_rank = iqr_normalize(upward_rank) * criticality_gate
    
    # Remaining work: higher work → lower priority (unless critical), so invert
    norm_work = -iqr_normalize(remaining_work)
    
    # Ready wait time: anti-starvation — but decayed for long waits to avoid domination
    # Use soft saturation: arctan(wait * decay) → bounded, smooth, monotonic
    wait_scaled = ready_wait_time * PARAMS["wait_decay_rate"]
    norm_wait = np.arctan(wait_scaled)  # ∈ [0, π/2), monotonically increasing, bounded
    
    # Uncertainty-slack interaction: amplify pressure when both are high
    # Use product of normalized uncertainty and slack_pressure (already negative for risk)
    norm_uncert = iqr_normalize(uncertainty)
    uncertainty_slack_coupling = norm_uncert * norm_slack_pressure * PARAMS["uncertainty_slack_interaction"]
    
    # Assemble score: all terms contribute additively; smaller = better
    # Emphasize slack & risk first, then energy, then structure & fairness
    score = (
        PARAMS["slack_penalty_scale"] * norm_slack_pressure
        + PARAMS["energy_norm_factor"] * norm_energy
        + PARAMS["duration_balance"] * norm_duration
        - norm_rank  # higher rank → higher priority → subtract
        + norm_work  # higher remaining work → lower priority → add (since norm_work is negated)
        - norm_wait  # longer wait → higher priority → subtract
        + uncertainty_slack_coupling
    )
    
    # Final safeguard: replace NaN/inf with finite values, preserve shape
    score = np.nan_to_num(score, nan=0.0, posinf=1e12, neginf=-1e12)
    
    # Ensure shape (N,) even for N=1
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    
    return score
