def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing deadline risk containment first, then energy efficiency,
    with adaptive critical-path leverage and bounded fairness. Uses piecewise slack handling,
    robust range-based normalization, and uncertainty-modulated coupling."""
    eps = PARAMS["epsilon"]
    
    # Cast to float64 and ensure no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=np.float64)
    min_comm_time = np.asarray(min_comm_time, dtype=np.float64)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slack = np.asarray(slack, dtype=np.float64)
    upward_rank = np.asarray(upward_rank, dtype=np.float64)
    remaining_work = np.asarray(remaining_work, dtype=np.float64)
    ready_wait_time = np.asarray(ready_wait_time, dtype=np.float64)
    uncertainty = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust range-based normalization: avoids mean bias from outliers; uses IQR-like scale
    def normalize_range(x):
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        p25, p75 = np.percentile(x, [25, 75], method='midpoint')
        iqr = p75 - p25
        center = np.median(x)
        scale = np.maximum(iqr, eps)
        return (x - center) / scale
    
    # Duration: execution + communication — penalize long resource-holding tasks
    duration = min_exec_time + min_comm_time
    norm_duration = normalize_range(duration)
    
    # Slack: handle negative (violation-risk) and positive (urgency) separately
    # Negative slack → strong linear penalty; positive slack → diminishing-returns urgency
    slack_neg = np.clip(-slack, 0.0, None)  # >0 iff slack < 0
    slack_pos = np.clip(slack, 0.0, None)   # >0 iff slack > 0
    norm_slack_neg = normalize_range(slack_neg)
    norm_slack_pos = normalize_range(slack_pos) ** PARAMS["slack_urgency_scale"]
    
    # Risk-adjusted energy: lower is better → subtract from score (since smaller score = higher priority)
    norm_energy = normalize_range(min_incremental_energy)
    
    # Upward rank: importance on critical path — amplify only under risk pressure
    # Use bounded coupling: only boost when slack is tight OR uncertainty is high
    risk_pressure = np.clip((1.0 - np.tanh(slack / (1.0 + eps))) * (1.0 + uncertainty), 0.0, 2.0)
    norm_rank = normalize_range(upward_rank)
    coupled_rank = norm_rank * (1.0 + PARAMS["critical_path_leverage"] * risk_pressure)
    
    # Uncertainty-slack coupling: high uncertainty amplifies urgency for low slack
    # Smooth gate: tanh-based soft switch around slack ≈ 0
    slack_gate = 0.5 * (1.0 - np.tanh(slack / (5.0 + eps)))
    coupled_uncertainty = normalize_range(uncertainty) * slack_gate * PARAMS["uncertainty_slack_coupling"]
    
    # Bounded fairness: wait boost saturates at max_wait_cap
    capped_wait = np.minimum(ready_wait_time, PARAMS["max_wait_cap"])
    norm_wait = normalize_range(capped_wait)
    
    # Assemble priority score: minimize this → maximize DDL compliance, then energy, then fairness
    # All terms are normalized and dimensionless; sign chosen so smaller = better
    score = (
        # Strong penalty for violation risk (negative slack dominates)
        PARAMS["slack_risk_penalty"] * norm_slack_neg
        # Urgency for positive slack (sublinear)
        + norm_slack_pos
        # Energy efficiency gain: lower energy → lower score
        - PARAMS["energy_efficiency_gain"] * norm_energy
        # Duration penalty: longer duration → higher score
        + PARAMS["duration_sensitivity"] * norm_duration
        # Critical-path leverage: amplified rank under risk
        + coupled_rank
        # Uncertainty-modulated deadline pressure
        + coupled_uncertainty
        # Fairness: longer wait → lower score (anti-starvation)
        - PARAMS["wait_fairness_boost"] * norm_wait
    )
    
    # Final safeguard: replace NaN/inf with finite values, preserve shape
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(np.float64).max, neginf=-np.finfo(np.float64).max)
    
    return score
