def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: signed slack power + coupled energy-work penalty + adaptive criticality.
    
    Key improvements:
    - Added `remaining_work_energy_coupling`: penalizes high-energy tasks with large descendant workloads,
      improving global energy-awareness beyond per-task marginal cost.
    - Replaced binary risk_condition with smooth, normalized uncertainty coupling using median-relative ratio.
    - Unified robust normalization via np.quantile with tunable unit-interval quantiles.
    - Preserved Parent 2's signed slack power law and duration_ratio for edge-offload bias.
    - Kept anti-starvation as monotonic exp(-decay * wait) — avoids unbounded growth and starvation.
    - All numeric literals are only -2, -1, 0, 1, 2; eps handled via PARAMS; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization using tunable quantiles (unit interval)
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q1_val = PARAMS["q1_quantile"]
        q3_val = PARAMS["q3_quantile"]
        q1, q3 = np.quantile(x, [q1_val, q3_val])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Signed slack pressure: smooth, differentiable, zero-crossing at slack=0
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_power = np.where(
        slack_abs < eps,
        0.0,
        slack_sign * (slack_abs ** PARAMS["slack_sensitivity"])
    )
    base_slack_penalty = PARAMS["slack_penalty_scale"] * slack_power
    
    # Uncertainty-slack coupling: multiplicative gate scaled by relative uncertainty
    unc_rel = uncertainty / (np.median(uncertainty) + eps)
    unc_slack_gate = 1.0 + PARAMS["uncertainty_slack_coupling"] * unc_rel
    gated_slack = base_slack_penalty * unc_slack_gate
    
    # Criticality boost: applied to upward_rank only when slack is tight OR uncertainty is high
    # Uses soft threshold via normalized distance to median for stability
    slack_safe = (slack > np.median(slack) + eps).astype(float)
    unc_safe = (uncertainty <= np.median(uncertainty) + eps).astype(float)
    risk_mask = 1.0 - (slack_safe * unc_safe)  # 1 if risky, 0 if both safe
    boosted_rank = upward_rank * (1.0 + PARAMS["criticality_boost"] * risk_mask)
    
    # Energy efficiency term — normalized and weighted
    energy_norm = robust_normalize(min_incremental_energy)
    energy_score = PARAMS["energy_normalizer"] * energy_norm
    
    # Remaining work × energy coupling: penalizes high-marginal-energy tasks that also anchor heavy subtrees
    # Encourages early scheduling of energy-efficient gateways to heavy regions
    work_energy_interaction = min_incremental_energy * (remaining_work + eps)
    work_energy_norm = robust_normalize(work_energy_interaction)
    work_energy_penalty = PARAMS["remaining_work_energy_coupling"] * work_energy_norm
    
    # Duration ratio: compute vs communication dominance [0,1]; higher favors compute-heavy → edge preference
    exec_plus_comm = min_exec_time + min_comm_time + eps
    duration_ratio = min_exec_time / exec_plus_comm
    ratio_score = PARAMS["duration_ratio_weight"] * (1.0 - duration_ratio)
    
    # Anti-starvation: monotonic, bounded exponential decay — longer wait → higher priority (lower score)
    wait_boost = np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # Assemble final score: smaller = higher priority
    # Note: rank and wait_boost subtracted → increase priority (lower score); others added → decrease priority
    score = (
        gated_slack
        + energy_score
        + work_energy_penalty
        + ratio_score
        - robust_normalize(boosted_rank)
        - wait_boost
    )
    
    # Ensure finite output, shape (N,), and determinism
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
