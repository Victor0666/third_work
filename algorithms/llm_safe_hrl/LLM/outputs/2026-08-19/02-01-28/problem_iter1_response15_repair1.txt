def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing risk-aware DDL compliance first, then energy efficiency.
    
    Key innovations:
      - Uses piecewise slack penalty: linear for positive slack, quadratic for negative slack
      - Introduces criticality-boost gate: only activates high-rank boost when slack < 0 AND rank > threshold
      - Computes 'energy efficiency density' = energy / (exec + comm + epsilon), normalized
      - Models wait time with saturating exponential gain (anti-starvation without dominance)
      - Couples uncertainty *only* with negative slack via bounded product, avoiding unbounded growth
      - All normalizations use mean-abs + epsilon; no std-based scaling to avoid NaN from singleton
    """
    eps = PARAMS["epsilon"]
    posinf_rep = PARAMS["posinf_replacement"]
    neginf_rep = PARAMS["neginf_replacement"]
    
    # Defensive casting and safe zero/Nan handling — using declared parameters only
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps, posinf=posinf_rep, neginf=neginf_rep)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps, posinf=posinf_rep, neginf=neginf_rep)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps, posinf=posinf_rep, neginf=neginf_rep)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=posinf_rep, neginf=neginf_rep)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps, posinf=posinf_rep, neginf=neginf_rep)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps, posinf=posinf_rep, neginf=neginf_rep)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=posinf_rep, neginf=neginf_rep)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps, posinf=posinf_rep, neginf=neginf_rep)

    # Normalize all features robustly using mean absolute value
    def robust_normalize(x):
        x_abs = np.abs(x)
        denom = np.mean(x_abs) + eps
        return x / denom

    # 1. Slack penalty: prioritize tight/negative slack strongly, but penalize negativity quadratically
    slack_sign = np.sign(slack)
    slack_abs = np.abs(slack)
    # Linear for positive slack (less urgent), quadratic penalty for negative slack (urgent)
    slack_penalty = np.where(
        slack >= 0,
        slack_abs,
        slack_abs ** PARAMS["slack_penalty_exponent"]
    )
    norm_slack_penalty = robust_normalize(slack_penalty)

    # 2. Duration: weighted blend of exec and comm, then normalized
    duration = (PARAMS["duration_balance"] * min_exec_time +
                (1.0 - PARAMS["duration_balance"]) * min_comm_time)
    norm_duration = robust_normalize(duration + eps)

    # 3. Energy efficiency density: marginal energy per time unit — lower is better
    energy_density = min_incremental_energy / (duration + eps)
    norm_energy_density = robust_normalize(energy_density)

    # 4. Upward rank: only boost if both (a) rank is high *and* (b) slack is tight/negative
    rank_threshold = PARAMS["rank_threshold_ratio"] * np.max(upward_rank + eps)
    critical_gate = np.where(
        (upward_rank >= rank_threshold) & (slack < 0),
        PARAMS["criticality_boost"],
        1.0
    )
    norm_upward_rank = robust_normalize(upward_rank)
    boosted_rank_score = critical_gate * norm_upward_rank

    # 5. Ready wait time: saturating exponential gain to prevent starvation without dominating
    # Saturates near exp(-wait_decay * t) → 0, so we use 1 - exp(...) for increasing priority
    wait_gain = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    norm_wait_gain = robust_normalize(wait_gain)

    # 6. Uncertainty-slack coupling: only active under deadline pressure (slack < 0)
    coupled_risk = np.where(
        slack < 0,
        uncertainty * slack_abs,
        0.0
    )
    norm_coupled_risk = robust_normalize(coupled_risk + eps)

    # Composite score: minimize → higher priority
    # Order of terms reflects lexicographic intent: DDL risk first, then efficiency, then fairness
    score = (
        # Primary: deadline risk dominates — higher penalty for tighter/negative slack
        1.0 * norm_slack_penalty
        # Secondary: favor energy-efficient execution (low energy density)
        + PARAMS["energy_efficiency_gain"] * norm_energy_density
        # Tertiary: prefer shorter durations (but less than slack/energy)
        + 0.3 * norm_duration
        # Quaternary: boost critical tasks *only when at risk*
        - 0.8 * boosted_rank_score
        # Anti-starvation: add moderate gain for long-waiting tasks
        - 0.4 * norm_wait_gain
        # Risk coupling: small additive term to amplify uncertainty impact under DDL stress
        + PARAMS["uncertainty_slack_coupling"] * norm_coupled_risk
    )

    # Final safeguard: ensure finite, deterministic output using machine-precision bounds
    finfo = np.finfo(float)
    return np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
