def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: deadline-risk gated criticality + fairness-aware energy efficiency.
    
    Key innovations:
      - Slack is split into *risk-penalized* (negative) and *margin-rewarded* (positive) components
      - Upward rank leverage is dynamically gated by slack and uncertainty (not fixed weight)
      - Energy efficiency is conditionally emphasized only when slack is non-critical
      - Duration sensitivity adapts to urgency via slack-based scaling
      - All features are robustly normalized using mean-abs + epsilon, then clipped to avoid outliers
      - Ready-wait-time is linearly added (no normalization) to ensure monotonic anti-starvation
      - Final score is bounded and nan-cleaned for deterministic stability.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Safe cast & copy to prevent mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust normalization: use mean absolute value + epsilon; clip extremes to avoid outlier skew
    def robust_norm(x):
        x = np.clip(x, -1e6, 1e6)  # prevent overflow in mean abs
        denom = np.mean(np.abs(x)) + eps
        return np.clip(x / denom, -10.0, 10.0)
    
    # Normalize core features
    norm_energy = robust_norm(min_incremental_energy)
    norm_duration = robust_norm(min_exec_time + min_comm_time)
    norm_rank = robust_norm(upward_rank)
    norm_work = robust_norm(remaining_work)
    norm_uncert = robust_norm(uncertainty)
    
    # Slack decomposition: separate risk (negative) and margin (positive)
    slack_neg = np.clip(-slack, 0.0, np.inf)  # >0 iff slack < 0
    slack_pos = np.clip(slack, 0.0, np.inf)    # >0 iff slack > 0
    norm_slack_neg = robust_norm(slack_neg)
    norm_slack_pos = robust_norm(slack_pos)
    
    # Urgency signal: normalized negative slack dominates when present
    urgency = np.where(slack < 0, 
                       PARAMS["slack_risk_penalty"] * norm_slack_neg,
                       -PARAMS["slack_urgency_gain"] * norm_slack_pos)
    
    # Dynamic criticality leverage: full weight only when slack <= threshold * median slack
    # Compute normalized slack threshold relative to distribution
    slack_abs_med = np.median(np.abs(slack)) + eps
    slack_normalized_for_gate = np.abs(slack) / slack_abs_med
    rank_gate = np.where(slack_normalized_for_gate <= PARAMS["rank_saturation_threshold"],
                         1.0,
                         np.clip(slack_normalized_for_gate / PARAMS["rank_saturation_threshold"], 0.0, 1.0))
    leveraged_rank = rank_gate * norm_rank
    
    # Conditional energy emphasis: stronger weight when slack is safe (non-negative)
    energy_weight_cond = np.where(slack >= 0,
                                  PARAMS["energy_efficiency_bias"] * (1.0 - PARAMS["energy_slack_tradeoff"]),
                                  PARAMS["energy_efficiency_bias"] * PARAMS["energy_slack_tradeoff"])
    
    # Duration sensitivity scaled by urgency magnitude (higher when urgent)
    urgency_mag = np.abs(urgency)
    duration_weight_scaled = PARAMS["duration_sensitivity"] * (1.0 + 0.5 * urgency_mag)
    
    # Uncertainty-slack coupling: amplifies urgency when uncertainty is high
    coupling_term = PARAMS["uncertainty_coupling"] * norm_uncert * np.maximum(norm_slack_neg, 0)
    
    # Anti-starvation: linear additive boost (no normalization → monotonic fairness)
    wait_boost = PARAMS["wait_fairness"] * ready_wait_time
    
    # Assemble score: smaller = better
    score = (
        urgency
        + energy_weight_cond * norm_energy
        + duration_weight_scaled * norm_duration
        - PARAMS["criticality_leverage"] * leveraged_rank
        - 0.2 * norm_work  # light work-reduction bias (smaller work → less impact on total flow)
        + coupling_term
        - wait_boost  # subtract because longer wait should lower score (increase priority)
    )
    
    # Final stabilization: finite, bounded, shape-(N,)
    score = np.nan_to_num(score, nan=1e6, posinf=1e6, neginf=-1e6)
    score = np.clip(score, -1e5, 1e5)
    
    # Ensure shape (N,) — no squeezing
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    
    return score
