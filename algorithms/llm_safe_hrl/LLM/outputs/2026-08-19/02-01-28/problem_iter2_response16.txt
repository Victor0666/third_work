def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining v2's energy-efficiency ratio and smooth ramp with v1's work saturation and robust uncertainty coupling.
    
    Key improvements:
    - Replaces raw energy normalization with joules/second efficiency, then modulates it *multiplicatively* by uncertainty → penalizes high-uncertainty high-energy tasks more.
    - Keeps smooth quadratic ramp for slack near threshold (v2) but adds *asymmetric sigmoid gating* to slack pressure: stronger amplification for negative slack than relaxation for positive.
    - Uses log1p-saturated remaining_work (from v1) instead of raw or linearly normalized → prevents dominance by outliers while preserving monotonicity.
    - Robust normalization uses median absolute deviation (MAD) fallback for better outlier resilience vs std-based scaling.
    - All operations guarded against NaN/inf/zero; deterministic; shape-(N,) guaranteed.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Defensive cast & copy to avoid mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float).copy()
    min_comm_time = np.asarray(min_comm_time, dtype=float).copy()
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float).copy()
    slack = np.asarray(slack, dtype=float).copy()
    upward_rank = np.asarray(upward_rank, dtype=float).copy()
    remaining_work = np.asarray(remaining_work, dtype=float).copy()
    ready_wait_time = np.asarray(ready_wait_time, dtype=float).copy()
    uncertainty = np.asarray(uncertainty, dtype=float).copy()
    
    # === Robust normalization: MAD-based, fallback to mean if degenerate ===
    def robust_normalize(x):
        x = np.abs(x)
        center = np.median(x) if N > 1 else x[0]
        # Use MAD instead of std for outlier robustness
        mad = np.median(np.abs(x - center)) if N > 1 else eps
        scale = np.maximum(mad, eps)
        return (x - center) / scale
    
    # === 1. Slacker-pressure: smooth quadratic ramp + asymmetric sigmoid gating ===
    slack_centered = slack - PARAMS["critical_slack_threshold"]
    # Base convex penalty for negative slack
    slack_penalty_base = np.where(
        slack_centered < 0,
        -slack_centered * PARAMS["slack_risk_penalty"],
        0.0
    )
    # Quadratic ramp region: smooth transition from 0 to threshold
    ramp_region = np.logical_and(
        slack_centered >= -PARAMS["ramp_region_width"],
        slack_centered <= 0
    )
    slack_penalty_base = np.where(
        ramp_region,
        PARAMS["slack_risk_penalty"] * (slack_centered + PARAMS["ramp_region_width"])**2 / (PARAMS["quadratic_ramp_denom_factor"] * PARAMS["ramp_region_width"]),
        slack_penalty_base
    )
    # Asymmetric gating: tanh for negative slack (strong), exp(-|x|) for positive (gentle relief)
    gate_neg = np.tanh(np.maximum(-slack_centered, 0.0))
    gate_pos = np.exp(-np.maximum(slack_centered, 0.0))
    slack_gate = gate_neg + 0.5 * gate_pos  # Stronger negative response
    # Uncertainty-coupled pressure: only active where gate > 0.1
    slack_pressure = slack_penalty_base * slack_gate * (
        1.0 + PARAMS["uncertainty_slack_coupling"] * np.tanh(uncertainty)
    )
    
    # === 2. Energy efficiency: joules per second, uncertainty-modulated ===
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_efficiency = min_incremental_energy / duration  # lower = better
    # Modulate efficiency penalty by uncertainty: high-uncertainty high-energy tasks get extra penalty
    energy_eff_score = energy_efficiency * (1.0 + PARAMS["energy_uncertainty_interaction"] * uncertainty)
    # Normalize for consistent scaling
    energy_eff_norm = robust_normalize(energy_eff_score)
    
    # === 3. Critical path emphasis: sqrt-compressed upward rank ===
    rank_score = PARAMS["rank_sensitivity"] * np.sqrt(np.maximum(upward_rank, eps))
    
    # === 4. Anti-starvation: sublinear wait boost ===
    wait_boost = np.power(np.maximum(ready_wait_time, eps), PARAMS["wait_fairness_exponent"])
    wait_score = robust_normalize(wait_boost)
    
    # === 5. Duration robustness: coefficient-of-variation of exec+comm ===
    duration_cv = np.std(np.stack([min_exec_time, min_comm_time]), axis=0) / duration
    # Lower CV → higher priority → invert sign
    duration_robust_score = -robust_normalize(duration_cv)
    
    # === 6. Remaining work: log1p-saturated to prevent outlier dominance ===
    work_scaled = np.log1p(remaining_work / (PARAMS["work_saturation_ref"] + eps))
    work_score = -robust_normalize(work_scaled)  # Prefer tasks guarding large remaining work → lower score
    
    # === Assemble final score: smaller = higher priority ===
    # Prioritization order: DDL risk > energy efficiency > critical path > fairness > robustness > work scope
    score = (
        slack_pressure  # dominates when deadline at risk
        + PARAMS["energy_efficiency_bias"] * energy_eff_norm
        + rank_score
        - PARAMS["duration_robustness_weight"] * duration_robust_score
        - wait_score
        + work_score
    )
    
    # Final numerical safeguard: ensure finite, shape-(N,), no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure shape is exactly (N,) even for N=1
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    
    return score
