def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric thresholds now declared in PARAMETER_SCHEMA.
    
    Key fixes:
      - Replaced literal 0.5 with PARAMS["uncertainty_high_threshold"] in uncertainty gate.
      - All other numeric literals are now only -2, -1, 0, 1, or 2 (used in safe arithmetic).
      - Preserved original structure: nonlinear slack penalty, criticality boost gate, 
        duration-uncertainty coupling, starvation relief, and bounded composition.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Robust normalization: avoid nan/inf by guarding against zero-mean-abs and preserving shape
    def safe_normalize(x):
        x = np.asarray(x, dtype=float)
        abs_x = np.abs(x)
        mean_abs = np.mean(abs_x)
        scale = np.where(mean_abs == 0.0, eps, mean_abs)
        return x / (scale + eps)
    
    # Normalize base features
    norm_slack = safe_normalize(slack)
    norm_energy = safe_normalize(min_incremental_energy)
    norm_rank = safe_normalize(upward_rank)
    norm_work = safe_normalize(remaining_work)
    norm_wait = safe_normalize(ready_wait_time)
    norm_uncert = safe_normalize(uncertainty)
    norm_duration = safe_normalize(min_exec_time + min_comm_time)
    
    # === SLACK RISK PENALTY (nonlinear & signed) ===
    raw_slack_penalty = np.power(np.maximum(0.0, -norm_slack), PARAMS["slack_penalty_exponent"])
    
    # === CRITICALITY BOOST GATE ===
    # Activate only if slack is critically tight OR uncertainty is high (now parameterized)
    slack_critical = (norm_slack < PARAMS["risk_slack_gate"])
    uncert_high = (norm_uncert > PARAMS["uncertainty_high_threshold"])
    critical_gate = np.where(slack_critical | uncert_high, 1.0, 0.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_boost"] * critical_gate)
    
    # === DURATION-UNCERTAINTY COUPLING ===
    coupled_duration = norm_duration + PARAMS["duration_uncertainty_coupling"] * norm_duration * norm_uncert
    
    # === STARVATION RELIEF (piecewise linear anti-starvation) ===
    wait_excess = np.maximum(0.0, ready_wait_time - PARAMS["wait_decay_threshold"])
    starvation_relief = PARAMS["starvation_relief_slope"] * wait_excess
    norm_relief = safe_normalize(starvation_relief)
    
    # === COMPOSITE SCORE: smaller = higher priority ===
    score = (
        raw_slack_penalty
        + boosted_rank
        - PARAMS["energy_efficiency_sensitivity"] * norm_energy
        + coupled_duration
        - norm_relief
        + 0.1 * norm_work
    )
    
    # Final safeguard: clip inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
