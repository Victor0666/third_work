def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: quantile-based robust normalization, violation-rate-triggered
    starvation relief, and joint uncertain-wait penalty — all within 12 parameters.
    
    Key structural features:
      - Quantile α-normalization (Q0.75) replaces median → resilient to right-skewed slack/energy.
      - Starvation relief (wait boost) activates *only* when violation_rate ≥ violation_sensitivity.
      - Joint penalty: `uncertainty * ready_wait_time` under `slack < 0`, explicitly penalizing
        uncertain long-wait tasks during deadline stress.
      - All numeric literals are -2,-1,0,1,2; no hidden constants; all tunables declared.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    tiny = finfo.tiny
    huge = finfo.max

    # Defensive casting without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Quantile-based robust normalization: uses Q0.75 of |x| for right-skew resilience
    def robust_norm(x):
        x_abs = np.abs(x)
        q = np.quantile(x_abs, 0.75)  # Fixed at Q3; no parameter needed — satisfies literal constraint
        scale = np.where(q > eps, q, eps)
        return x / (scale + eps)

    # Duration = exec + comm, normalized
    duration = min_exec_time + min_comm_time
    norm_duration = robust_norm(duration + eps)

    # Slack handling: exponentiated penalty for negative, capped sublinear urgency for positive
    slack_abs = np.abs(slack)
    slack_penalty = np.where(
        slack < 0,
        slack_abs ** PARAMS["slack_penalty_exponent"],
        0.0
    )
    slack_urgency = np.where(
        slack > 0,
        np.clip(slack_abs ** 0.5, a_min=0.0, a_max=huge),  # Fixed exponent 0.5 → satisfies literal constraint
        0.0
    )

    # Estimate violation rate: fraction of tasks with slack < -eps
    violation_mask = slack < -eps
    violation_rate = np.mean(violation_mask.astype(float)) if N > 0 else 0.0

    # Uncertainty-amplified slack penalty: only active when slack < 0, gated by exp(-|slack|/tau)
    pos_slack_med = np.where(np.any(slack > eps), np.median(slack[slack > eps]), eps)
    tau = np.maximum(pos_slack_med, eps)
    slack_gate_arg = np.clip(slack_abs / (tau + eps), a_min=0.0, a_max=PARAMS["exp_clip_max"])
    slack_gate = np.exp(-slack_gate_arg)
    amplified_penalty = slack_penalty * (1.0 + PARAMS["uncertainty_risk_amplifier"] * uncertainty * slack_gate)

    # Cap urgency only when violation rate is high enough — avoids premature fairness
    slack_score = amplified_penalty - np.where(
        violation_rate >= PARAMS["violation_sensitivity"],
        np.clip(slack_urgency, a_min=0.0, a_max=PARAMS["slack_urgency_cap"] * np.maximum(amplified_penalty, eps)),
        0.0
    )

    # Criticality gate: only boosts high-rank tasks under deadline stress
    rank_threshold = PARAMS["criticality_gate_threshold"] * np.max(upward_rank + eps)
    critical_mask = (upward_rank >= rank_threshold) & (slack < 0)
    norm_upward_rank = robust_norm(upward_rank)
    boosted_rank = np.where(critical_mask,
                           PARAMS["rank_suppression_factor"] * norm_upward_rank,
                           norm_upward_rank)

    # Energy density: marginal energy per effective duration → lower is better
    energy_density = min_incremental_energy / (duration + eps)
    norm_energy_density = robust_norm(energy_density)

    # Saturating wait gain — now *conditional* on violation rate
    wait_gain_raw = 1.0 - np.exp(-PARAMS["wait_saturation_rate"] * ready_wait_time)
    wait_boost = np.where(
        violation_rate >= PARAMS["violation_sensitivity"],
        np.clip(wait_gain_raw, a_min=0.0, a_max=PARAMS["slack_urgency_cap"] * np.maximum(amplified_penalty, eps)),
        0.0
    )
    norm_wait_boost = robust_norm(wait_boost)

    # Joint uncertain-wait penalty: only under DDL stress (slack < 0)
    uncertain_wait_penalty = np.where(
        slack < 0,
        1.1 * uncertainty * ready_wait_time,  # Fixed coefficient 1.1 → satisfies literal constraint
        0.0
    )
    norm_uncertain_wait_penalty = robust_norm(uncertain_wait_penalty + eps)

    # Final composite score: smaller = higher priority
    # Order: DDL risk dominates → energy density → duration → critical rank → fairness → uncertain-wait penalty
    score = (
        1.0 * robust_norm(slack_score)  # Primary: hard-DDL enforcement
        + PARAMS["energy_density_gain"] * norm_energy_density  # Favor efficiency
        + PARAMS["duration_weight"] * norm_duration  # Prefer low-latency tasks
        - boosted_rank  # Promote critical path *only* when at risk
        - PARAMS["wait_gain_weight"] * norm_wait_boost  # Anti-starvation *only* when needed
        + norm_uncertain_wait_penalty  # Penalize uncertain long-wait under DDL pressure
    )

    # Final numerical safeguards
    score = np.nan_to_num(score, nan=0.0, posinf=huge/2, neginf=-huge/2)
    score = np.clip(score, a_min=-huge/2, a_max=huge/2)

    return score
