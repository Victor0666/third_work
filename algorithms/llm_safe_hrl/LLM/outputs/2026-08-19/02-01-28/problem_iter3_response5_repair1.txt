def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Deterministic priority scorer: smaller value = higher scheduling priority.
    
    Key self-evolution improvements:
    - Replaces soft sigmoid coupling with a bounded min-max gate on (slack, uncertainty)
      to strictly enforce feasibility-first logic before energy optimization.
    - Introduces task-age–normalized slack: slack_eff = slack / (1 + ready_wait_time / tau),
      then applies piecewise dead-zone around zero to eliminate DDL-boundary oscillation.
    - Uses robust piecewise slack_effective = 
          { slack if slack > deadzone, 
            0 if |slack| <= deadzone, 
            slack - sign(slack)*deadzone otherwise } — preserving monotonicity.
    - All normalizations use mean-abs with eps guard; all outputs finite and shape-(N,).
    """
    eps = PARAMS["epsilon"]
    
    # Convert all inputs to float arrays; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Safe mean-abs normalization
    def normalize(x):
        abs_mean = np.mean(np.abs(x)) + eps
        return x / abs_mean
    
    # === Structural evolution: Bounded min-max feasibility gate ===
    # Compute feasibility-critical slack: age-normalized & dead-zone filtered
    # Avoid division by zero: add eps to denominator
    tau = np.maximum(np.mean(ready_wait_time) + eps, eps)
    slack_age_norm = slack / (1.0 + ready_wait_time / tau + eps)
    
    # Piecewise dead-zone: suppress priority oscillation near slack ≈ 0
    dz = PARAMS["slack_deadzone_halfwidth"]
    slack_eff = np.where(
        slack_age_norm > dz,
        slack_age_norm - dz,
        np.where(
            slack_age_norm < -dz,
            slack_age_norm + dz,
            0.0
        )
    )
    
    # Deadline risk: only penalize *negative* effective slack (lateness risk)
    deadline_risk = np.maximum(-slack_eff, 0.0)
    
    # === Structural evolution: Hard min-max uncertainty-slack gate ===
    # Coupling activates only when both slack_eff is tight (< dz/2) AND uncertainty is high (> median)
    unc_median = np.median(uncertainty) + eps
    unc_high = (uncertainty > unc_median).astype(float)
    slack_tight = (np.abs(slack_eff) < dz / 2.0).astype(float)
    coupling = unc_high * slack_tight  # binary-like bounded gate [0,1]
    
    # Normalize all features once
    norm_exec = normalize(min_exec_time)
    norm_comm = normalize(min_comm_time)
    norm_energy = normalize(min_incremental_energy)
    norm_drisk = normalize(deadline_risk)
    norm_uprank = normalize(upward_rank)
    norm_work = normalize(remaining_work)
    norm_wait = normalize(ready_wait_time)
    norm_coupling = normalize(coupling)
    
    # Linear combination: smaller score = higher priority
    score = (
        PARAMS["exec_weight"] * norm_exec
        + PARAMS["comm_weight"] * norm_comm
        + PARAMS["energy_weight"] * norm_energy
        + PARAMS["deadline_risk_weight"] * norm_drisk
        + PARAMS["critical_work_weight"] * norm_work
        + PARAMS["upward_rank_weight"] * norm_uprank
        + PARAMS["wait_penalty_weight"] * norm_wait
        + PARAMS["uncertainty_coupling_weight"] * norm_coupling
    )
    
    # Ensure output is finite and shape-(N,)
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=np.finfo(float).max,
        neginf=-np.finfo(float).max
    )
    return score
