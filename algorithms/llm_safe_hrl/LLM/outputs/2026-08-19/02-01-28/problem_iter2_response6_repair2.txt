def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: exactly 12 parameters; all numeric literals are in {-2,-1,0,1,2}.
    Uses np.finfo for machine-precision safeguards. No hidden thresholds or coefficients."""
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Robust per-feature normalization: use median + clipped MAD to prevent degenerate spread
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        abs_devs = np.abs(x - center)
        spread = np.clip(np.median(abs_devs), eps, finfo.max) + eps
        return (x - center) / spread

    # Normalize core features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    norm_comm = robust_normalize(min_comm_time)

    # --- Risk-aware slack penalty: asymmetric & bounded ---
    slack_abs = np.abs(slack)
    negative_penalty = np.where(slack < 0, 
                                np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"]), 
                                0.0)
    # Soft reward for positive slack: avoids over-penalizing high slack tasks
    positive_relief = np.where(slack > 0, 
                               -np.log1p(slack_abs) / (1.0 + slack_abs + eps), 
                               0.0)
    slack_risk_score = negative_penalty + positive_relief

    # --- Multiplicative rank-slack interaction: captures joint criticality ---
    # Instead of step-gated amplification, use smooth power coupling: |norm_slack|^p * norm_rank
    slack_pressure = np.abs(norm_slack) ** PARAMS["rank_slack_interaction_power"]
    coupled_rank_score = norm_rank * slack_pressure

    # --- Criticality amplification gate preserved but refined ---
    critical_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    amplified_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * critical_gate)

    # --- Duration-uncertainty coupling: dual-threshold ANDed activation ---
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) & 
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]), 
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Communication-energy coupling: penalize high-comm tasks when slack is negative ---
    comm_energy_gate = np.where(slack < 0, 1.0, 0.0)
    comm_energy_coupling_term = norm_comm * norm_energy * comm_energy_gate * PARAMS["comm_energy_coupling"]

    # --- Bounded starvation relief: exponential decay with floor guard ---
    # Ensures monotonic improvement and avoids zero-gradient saturation
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (np.maximum(norm_wait, 0.0) + eps))
    # Floor guard prevents over-correction when norm_wait ≈ 0
    wait_benefit = np.maximum(wait_benefit, eps)

    # --- Work saturation: sigmoid-like scaling with explicit clipping ---
    work_ref = PARAMS["work_saturation_point"]
    work_ratio = np.clip(remaining_work / (work_ref + eps), 
                         finfo.tiny, 
                         finfo.max)
    work_scaled = 1.0 / (1.0 + 1.0 / (work_ratio + eps))

    # --- Final score assembly: smaller = higher priority ---
    # Dominant terms: slack risk, energy, duration-uncert, comm-energy
    # Suppressive terms: rank (via coupling), work (small favored), wait (anti-starve)
    score = (
        slack_risk_score                              # Primary DDL risk term
        + PARAMS["energy_sensitivity"] * norm_energy  # Energy minimization weight
        + dur_uncert_interaction                      # Uncertain long tasks under pressure
        + comm_energy_coupling_term                   # High-comm + high-energy under lateness risk
        + coupled_rank_score                          # Smooth rank-slack criticality coupling
        - amplified_rank                              # Step-gated critical rank boost (complementary)
        + work_scaled                                 # Prefer small-work tasks (guarding large subgraphs)
        - wait_benefit                                # Anti-starvation: longer wait → lower score
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    return score
