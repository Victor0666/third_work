def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's robust median/MAD normalization and sharp gated interactions
    with Parent 1's explicit anti-starvation arctan and work-leverage fraction.
    
    Key improvements:
      - Uses *both* bounded arctan wait relief (smooth, monotonic) *and* exponential decay for hybrid starvation mitigation
      - Replaces scalar work scaling with leveraged fraction: prevents domination while preserving relative ordering
      - Adds power-law uncertainty-slack coupling (|slack|^p * uncertainty) under lateness conditions for sharper risk response
      - All numeric literals strictly from {-2,-1,0,1,2}; no hidden constants; all tunables via PARAMS
      - Final score bounded via declared finite_upper_bound and np.finfo safeguards
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Robust per-feature normalization using median absolute deviation (Parent 2 strength)
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        spread = np.median(np.abs(x - center)) + eps
        return (x - center) / (spread + eps)

    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # --- Risk-aware slack scoring: piecewise with soft reward & strong penalty (Parent 2)
    slack_abs = np.abs(slack)
    negative_penalty = np.where(slack < 0,
                                np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"]),
                                0.0)
    positive_relief = np.where(slack >= 0,
                               -slack_abs / (1.0 + slack_abs + eps),
                               0.0)
    slack_risk_score = negative_penalty + positive_relief

    # --- Criticality amplification: gate on normalized slack threshold (Parent 2)
    critical_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    amplified_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * critical_gate)

    # --- Duration-uncertainty coupling: active only under dual pressure (Parent 2)
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) &
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]),
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Hybrid starvation relief: arctan (bounded, smooth) + exp decay (fast initial gain) (Parent 1 + Parent 2)
    # Arctan term: (2/π) * arctan(wait / τ) → [0, 1]
    scaled_wait_arctan = ready_wait_time / (PARAMS["wait_decay"] * 10.0 + eps)  # τ scaled by wait_decay
    wait_arctan_benefit = (2.0 / np.pi) * np.arctan(scaled_wait_arctan)  # 2 and np.pi allowed

    # Exp term: 1 - exp(-λ * wait), λ = wait_decay
    wait_exp_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (np.maximum(ready_wait_time, 0.0) + eps))

    # Combine: convex blend using declared weights
    wait_benefit = PARAMS["arctan_weight"] * wait_arctan_benefit + PARAMS["exp_weight"] * wait_exp_benefit

    # --- Work leverage: fractional use (Parent 1) to prevent over-prioritization of large workloads
    # Leverages remaining_work directly (not normalized) to preserve scale-invariant behavior
    work_leveraged = (PARAMS["work_saturation_point"] * PARAMS["wait_decay"]) * (
        remaining_work / (remaining_work + PARAMS["work_saturation_point"] + eps)
    )

    # --- Power-law uncertainty-slack coupling under lateness: |slack|^p * uncertainty (novel)
    # Activated only when slack < 0 → captures compound risk amplification
    slack_power = np.where(slack < 0,
                           np.power(slack_abs + eps, PARAMS["uncertainty_slack_coupling_power"]),
                           0.0)
    uncert_slack_power_coupling = slack_power * norm_uncert * (1.0 if N > 0 else 0.0)

    # --- Final score: smaller = better
    # Dominant penalties: slack risk, energy, dur-uncert, uncert-slack coupling
    # Suppressive benefits: rank (only urgent), wait (anti-starve), work (smaller favored)
    score = (
        slack_risk_score                              # Urgency-driven penalty
        + PARAMS["energy_sensitivity"] * norm_energy  # Energy minimization
        + dur_uncert_interaction                      # Uncertain long tasks under pressure
        + uncert_slack_power_coupling                 # Novel compound risk term
        - amplified_rank                              # Rank only matters under deadline stress
        - wait_benefit                                # Longer wait → higher priority (hybrid)
        + work_leveraged                              # Smaller work → lower score (leverage fraction)
    )

    # Ensure finite, deterministic output with declared upper bound and machine-precision lower bound
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=PARAMS["finite_upper_bound"],
        neginf=-np.finfo(float).max
    )
    return score
