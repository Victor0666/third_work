def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining best elements from both parents with novel structural enhancements.
    
    Key improvements:
    - Replaces fixed work weighting with bounded sigmoidal work saturation: prioritizes small-to-medium work
      but avoids over-prioritizing tiny tasks that may not amortize scheduling overhead.
    - Introduces explicit successor-release delay awareness via new logic pattern (no new input required — uses existing semantics).
      Since successor_release_delay is not provided, we proxy it via (slack - min_exec_time - min_comm_time) as an estimate
      of residual slack after this task's duration; negative values indicate successors will be blocked.
    - Applies soft saturation instead of hard clipping: preserves rank ordering while preventing outlier dominance.
    - Uses robust MAD-based normalization throughout.
    - All numeric literals are only -2,-1,0,1,2; all tunable constants exposed via PARAMS.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Robust normalization using median absolute deviation (MAD)
    def normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        dev = np.median(np.abs(x - center))
        scale = np.maximum(dev, eps)
        return (x - center) / scale

    # Normalize base features
    norm_slack = normalize(slack)
    norm_energy = normalize(min_incremental_energy)
    norm_rank = normalize(upward_rank)
    norm_work = normalize(remaining_work)
    norm_wait = normalize(ready_wait_time)
    norm_uncert = normalize(uncertainty)
    norm_duration = normalize(min_exec_time + min_comm_time)

    # --- Slack urgency: logistic gate for deadline pressure ---
    slack_urgency = 1.0 / (1.0 + np.exp(-PARAMS["slack_risk_penalty"] * norm_slack))

    # --- Tight-slack mask: smooth ramp near threshold ---
    tight_mask = np.clip(
        1.0 - (norm_slack - PARAMS["critical_slack_threshold"]) / (np.abs(norm_slack).max() + eps),
        0.0, 1.0
    )

    # --- Uncertainty-slack interaction: multiplicative risk coupling ---
    risk_gate = np.clip(norm_uncert * slack_urgency, 0.0, 1.0)

    # --- Duration-uncertainty coupling: penalize uncertain long tasks under pressure ---
    coupled_duration = norm_duration * (1.0 + PARAMS["duration_coupling"] * risk_gate)

    # --- Adaptive rank weighting: only under tight slack ---
    adaptive_rank = norm_rank * tight_mask * PARAMS["rank_urgency_ratio"]

    # --- Work saturation: sigmoidal diminishing return on large remaining_work ---
    # Proxy: prioritize medium-small work; saturate priority gain beyond work_threshold
    work_deviation = remaining_work - PARAMS["work_threshold"]
    work_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["work_scale"] * work_deviation))
    work_contribution = work_sigmoid * norm_work

    # --- Successor-release delay proxy: estimated blocking pressure ---
    # If slack < task duration, successors likely miss their earliest start → add penalty
    residual_slack = slack - (min_exec_time + min_comm_time)
    successor_block_penalty = np.where(residual_slack < 0.0,
                                       -residual_slack / (np.abs(slack).max() + eps),
                                       0.0)
    successor_block_penalty = normalize(successor_block_penalty)

    # --- Stabilized anti-starvation wait boost (bounded subtraction) ---
    rel_wait = np.clip(norm_wait / (np.abs(norm_wait).max() + eps), 0.0, 2.0)
    wait_boost = rel_wait + (1.0 - np.exp(-PARAMS["wait_decay_rate"] * rel_wait))

    # --- Final raw score: smaller = higher priority ---
    # Dominant drivers: slack urgency, energy minimization, duration-uncertainty, successor blocking
    # Suppressive: rank (only when urgent), work (smaller favored), wait (anti-starve)
    score = (
        slack_urgency
        + PARAMS["energy_sensitivity"] * norm_energy
        + coupled_duration
        + adaptive_rank
        + work_contribution
        + PARAMS["successor_release_weight"] * successor_block_penalty
        - wait_boost
        + PARAMS["uncertainty_slack_interaction"] * risk_gate
    )

    # --- Soft saturation to preserve ranking while bounding extremes ---
    # score ← score - log(1 + exp(-strength * score)) ensures monotonicity & boundedness
    strength = PARAMS["soft_saturation_strength"]
    score = score - np.log(1.0 + np.exp(-strength * score) + eps)

    # Ensure finite, shape-(N,) output; handle edge case N=1
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).smallest_subnormal,
                          neginf=-np.finfo(float).smallest_subnormal)
    if score.ndim == 0:
        score = np.full(1, score.item(), dtype=float)

    return score
