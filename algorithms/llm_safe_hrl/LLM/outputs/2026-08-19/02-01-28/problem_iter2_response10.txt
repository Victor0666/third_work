def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust gating with Parent 1's slack asymmetry and starvation capping.
    
    Key innovations:
      - Asymmetric slack handling: nonlinear penalty for negative slack AND diminishing-urgency for positive slack
      - Dual-gated criticality boost: activates only under *both* tight slack AND high uncertainty (AND, not OR)
      - Starvation relief capped by slack urgency/penalty to prevent overcompensation during deadline pressure
      - Robust L1-median normalization (more stable than mean) with explicit finite safeguards
      - All numeric literals strictly limited to {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    tiny = finfo.tiny
    huge = finfo.max

    # Ensure float64 and no mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust median-based normalization (L1-stable)
    def robust_norm(x):
        x_abs = np.abs(x)
        med = np.median(x_abs)
        scale = np.where(med > eps, med, eps)
        return x / (scale + eps)

    # Duration: exec + comm, normalized
    duration = robust_norm(min_exec_time + min_comm_time)

    # Slack decomposition: penalty for negative, diminishing urgency for positive
    slack_abs = np.abs(slack)
    slack_penalty = np.where(slack < 0,
                           np.power(-slack, PARAMS["slack_penalty_exponent"]),
                           0.0)
    slack_urgency = np.where(slack > 0,
                            np.clip(slack_abs ** PARAMS["slack_urgency_scale"],
                                   a_min=0.0, a_max=huge),
                            0.0)

    # Dual-gated criticality: boost only when *both* slack is tight AND uncertainty is high
    slack_critical = (slack_abs < np.maximum(np.median(slack_abs[slack_abs > eps]), eps))
    uncert_high = (uncertainty > np.median(uncertainty))
    critical_gate = np.where(slack_critical & uncert_high, 1.0, 0.0)
    boosted_rank = robust_norm(upward_rank) * (1.0 + PARAMS["criticality_boost"] * critical_gate)

    # Energy efficiency term
    energy_score = robust_norm(min_incremental_energy)

    # Duration-uncertainty coupling
    coupled_duration = duration + PARAMS["duration_uncertainty_coupling"] * duration * robust_norm(uncertainty)

    # Starvation relief: linear beyond threshold, but capped by slack urgency/penalty
    wait_excess = np.maximum(0.0, ready_wait_time - PARAMS["wait_decay_threshold"])
    raw_relief = PARAMS["starvation_relief_slope"] * wait_excess
    # Cap: use fraction of slack penalty for negative slack, fraction of urgency for positive
    cap_val = np.where(slack < 0,
                      PARAMS["starvation_relief_slope"] * slack_penalty,
                      np.clip(PARAMS["starvation_relief_slope"] * slack_urgency, a_min=0.0, a_max=huge))
    starvation_relief = np.clip(raw_relief, a_min=0.0, a_max=cap_val)

    # Remaining work bias (small positive weight to avoid starving small critical tasks)
    work_score = PARAMS["remaining_work_weight"] * robust_norm(remaining_work)

    # Final score: smaller = better
    # Slack penalty dominates → ensures DDL feasibility first
    # Then criticality, energy, duration, starvation, and work act as tie-breakers
    score = (
        1 * slack_penalty
        + 1 * boosted_rank
        - 1 * PARAMS["energy_efficiency_sensitivity"] * energy_score
        + 1 * coupled_duration
        - 1 * robust_norm(starvation_relief)
        + 1 * work_score
    )

    # Final numerical safeguards
    score = np.nan_to_num(score, nan=0.0, posinf=huge/2, neginf=-huge/2)
    score = np.clip(score, a_min=-huge/2, a_max=huge/2)

    return score
