def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL-risk gating, starvation-aware urgency,
    and uncertainty-modulated critical-path leverage — all via bounded, stable operations.
    
    Key innovations:
      - Slack is split into urgent (negative) and diminishing-urgency (positive) regimes
      - Critical-path leverage activates only under deadline pressure (soft gating)
      - Ready-wait boost is linear but capped by slack urgency/penalty to avoid overcompensation
      - Uncertainty modulates slack weight *before* nonlinear transformation (not additive)
      - All features normalized via robust L1-median scaling (more stable than mean)
      - No unbounded functions: uses np.clip, np.fmax, and safe divisions
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    tiny = finfo.tiny
    huge = finfo.max

    # Convert inputs to float64, ensure no mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust normalization: median absolute scaling with epsilon guard
    def robust_norm(x):
        x_abs = np.abs(x)
        med = np.median(x_abs)
        scale = np.where(med > eps, med, eps)
        return x / (scale + eps)

    # Duration: exec + comm, normalized
    duration = robust_norm(min_exec_time + min_comm_time)

    # Slack handling: strong penalty for negative slack; sublinear urgency for positive
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    # For negative slack: linear penalty
    slack_penalty = np.where(slack < 0, 
                           PARAMS["slack_risk_penalty"] * (-slack), 
                           0.0)
    # For positive slack: diminishing urgency via power law (capped at huge)
    slack_urgency = np.where(slack > 0,
                            np.clip(slack_abs ** PARAMS["slack_urgency_scale"], 
                                   a_min=0.0, a_max=huge),
                            0.0)
    # Combine: penalty dominates; urgency adds weak push for early scheduling
    slack_score = slack_penalty - slack_urgency

    # Uncertainty-modulated slack weight: tighter deadlines get stronger uncertainty coupling
    # Use soft gate: exp(-|slack|/tau) scaled by uncertainty — higher uncertainty *amplifies* urgency when slack is small
    tau = np.maximum(np.median(slack_abs[slack_abs > eps]), eps)
    slack_gate_arg = np.clip(slack_abs / (tau + eps), a_min=0.0, a_max=PARAMS["exp_clip_max"])
    slack_gate = np.exp(-slack_gate_arg)
    modulated_slack_weight = PARAMS["uncertainty_modulation"] * uncertainty * slack_gate
    # Apply modulation only to urgency component (penalty stays unconditional)
    slack_score = slack_penalty - slack_urgency * (1.0 + modulated_slack_weight)

    # Critical-path leverage: only active when slack is tight (|slack| < median positive slack or negative)
    pos_slack_mask = slack > eps
    pos_slack_med = np.where(np.any(pos_slack_mask), np.median(slack[pos_slack_mask]), eps)
    tight_deadline_mask = (slack < eps) | (slack_abs < np.maximum(pos_slack_med, eps))
    rank_boost = np.where(tight_deadline_mask,
                         PARAMS["critical_path_leverage"] * robust_norm(upward_rank),
                         robust_norm(upward_rank))

    # Energy efficiency: favor low marginal energy, but *only* after slack & rank constraints are satisfied
    energy_score = robust_norm(min_incremental_energy)

    # Starvation relief: linear wait boost, but capped so it never dominates tight-deadline tasks
    # Cap: max boost = wait_cap_factor_penalty * slack_penalty for negative slack, or wait_cap_factor_urgency * slack_urgency for positive
    cap_val = np.where(slack < 0,
                      PARAMS["wait_cap_factor_penalty"] * slack_penalty,
                      np.clip(PARAMS["wait_cap_factor_urgency"] * slack_urgency, a_min=0.0, a_max=huge))
    wait_boost = np.clip(PARAMS["wait_starvation_relief"] * ready_wait_time, 
                         a_min=0.0, a_max=cap_val)

    # Duration balance term — neutral baseline for latency-sensitive fairness
    duration_score = PARAMS["duration_balance"] * duration

    # Final score: smaller = better (minimize risk-adjusted energy *subject to DDL*)
    # Slack penalty dominates → ensures hard DDL feasibility first
    # Then leverage, energy, wait, and duration act as tie-breakers within feasible region
    score = (
        slack_score
        - rank_boost
        + PARAMS["energy_efficiency_gain"] * energy_score
        + wait_boost
        + duration_score
    )

    # Ensure finite output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=huge/2, neginf=-huge/2)
    score = np.clip(score, a_min=-huge/2, a_max=huge/2)

    return score
