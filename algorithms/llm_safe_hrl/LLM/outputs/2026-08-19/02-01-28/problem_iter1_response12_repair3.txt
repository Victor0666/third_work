def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: risk-aware, piecewise-gated, deadline-centric.
    
    Fixes: all numeric literals are in {-2,-1,0,1,2}; '0.5' moved to PARAMS["work_bias_weight"].
    Every declared parameter is used exactly once.
    Uses np.finfo for epsilon safety where needed.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    N = len(slack)
    if N == 0:
        return np.array([], dtype=float)

    # Robust normalization: use IQR + epsilon fallback
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q75, q25 = np.percentile(x, [75, 25], method='midpoint')
        iqr = q75 - q25
        center = np.median(x)
        scale = np.where(iqr > eps, iqr, np.std(x, ddof=0) + eps)
        return (x - center) / (scale + eps)

    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # === SLACK URGENCY: tanh-based with explicit negative penalty ===
    slack_urgency = np.tanh(PARAMS["slack_sensitivity"] * norm_slack)
    neg_slack_mask = (slack < 0).astype(float)
    slack_penalty = neg_slack_mask * PARAMS["slack_risk_penalty"] * (1.0 - np.tanh(norm_slack))

    # === UNCERTAINTY-SLACK COUPLING ===
    coupling_gate = np.tanh(norm_uncert * norm_slack)
    coupling_term = PARAMS["uncertainty_slack_coupling"] * np.abs(coupling_gate)

    # === CONDITIONAL CRITICAL PATH BOOST ===
    rank_boost_mask = (
        (slack >= 0).astype(float) *
        (norm_rank > PARAMS["rank_saturation_threshold"]).astype(float)
    )
    rank_boost = rank_boost_mask * PARAMS["critical_path_leverage"] * np.maximum(norm_rank, 0.0)

    # === ANTI-STARVATION BONUS (saturating exponential) ===
    wait_bonus = 1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + eps))
    wait_priority = -wait_bonus

    # === ENERGY EFFICIENCY TERM ===
    energy_priority = -PARAMS["energy_efficiency_scale"] * norm_energy

    # === DURATION BALANCE TERM ===
    duration_priority = -PARAMS["duration_balance"] * norm_duration

    # === WORK BIAS TERM ===
    work_bias = PARAMS["work_bias_weight"] * norm_work

    # Assemble final score: smaller = better
    score = (
        slack_urgency               # baseline urgency
        + slack_penalty             # explicit late-risk penalty
        + coupling_term             # joint pressure signal
        - rank_boost                # conditional critical path boost
        + energy_priority           # reward low-energy placement
        + duration_priority         # reward fast-executing tasks
        + wait_priority             # anti-starvation
        + work_bias                 # bias toward large remaining work
    )

    # Final safeguard: clip using declared finite bounds
    score = np.clip(score, PARAMS["score_clip_lower"], PARAMS["score_clip_upper"])
    score = np.nan_to_num(score, nan=0.0, posinf=PARAMS["score_clip_upper"], neginf=PARAMS["score_clip_lower"])

    return score
