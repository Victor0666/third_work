def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing deadline risk containment with adaptive criticality gating.
    
    Key innovations:
    - Uses a smooth, bounded sigmoid-like slack urgency gate (not raw linear) to strongly penalize negative slack
      while avoiding numerical explosion.
    - Applies HEFT upward rank *only* under tight-slack conditions — avoids over-prioritizing critical paths when
      deadlines are relaxed.
    - Introduces uncertainty-aware duration coupling: high uncertainty amplifies the importance of fast-executing
      or low-comm tasks *only* when slack is low.
    - Ready wait time uses a stabilized soft exponential decay: prevents starvation without overwhelming DDL pressure.
    - All features are robustly normalized using median absolute deviation (MAD) for outlier resilience.
    - No unbounded operations; all divisions guarded; all outputs finite and deterministic.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Robust normalization: use median absolute deviation (MAD) for outlier resilience
    def normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        dev = np.median(np.abs(x - center))
        scale = np.maximum(dev, eps)
        return (x - center) / scale

    # Normalize base features
    norm_slack = normalize(slack)
    norm_energy = normalize(min_incremental_energy)
    norm_rank = normalize(upward_rank)
    norm_work = normalize(remaining_work)
    norm_wait = normalize(ready_wait_time)
    norm_uncert = normalize(uncertainty)
    norm_duration = normalize(min_exec_time + min_comm_time)

    # --- Slack urgency gate: sharp but bounded penalty for negative slack ---
    # Sigmoid-like: near-zero for large positive slack, rises smoothly, saturates at ~1.0 for highly negative slack
    slack_urgency = 1.0 / (1.0 + np.exp(-PARAMS["slack_risk_penalty"] * norm_slack))

    # --- Tight-slack condition mask (soft binary) ---
    # Activates critical-path logic only when slack is below threshold (in normalized space)
    tight_mask = np.clip(
        1.0 - (norm_slack - PARAMS["critical_slack_threshold"]) / (np.abs(norm_slack).max() + eps),
        0.0, 1.0
    )

    # --- Uncertainty-slack interaction gate ---
    # High priority when both slack is tight AND uncertainty is high
    risk_gate = np.clip(norm_uncert * slack_urgency, 0.0, 1.0)

    # --- Duration-uncertainty coupling: prefer short-duration tasks more under high risk ---
    coupled_duration = norm_duration * (1.0 + PARAMS["duration_coupling"] * risk_gate)

    # --- Adaptive rank weighting: only activate under tight deadlines ---
    adaptive_rank = norm_rank * tight_mask * PARAMS["rank_urgency_ratio"]

    # --- Stabilized ready-wait anti-starvation: linear + soft exponential decay ---
    # Prevents infinite growth: capped exponential term based on relative wait
    rel_wait = np.clip(norm_wait / (np.abs(norm_wait).max() + eps), 0.0, 2.0)
    wait_boost = rel_wait + (1.0 - np.exp(-PARAMS["wait_decay_rate"] * rel_wait))

    # --- Final score: smaller = higher priority ---
    # Note: slack_urgency already encodes high priority for low/negative slack → keep sign positive
    # Energy, duration, rank, work, wait all contribute positively to score (i.e., increase priority cost)
    # So we add them — lower total score = higher priority
    score = (
        slack_urgency  # [0,1]: strong base urgency signal
        + PARAMS["energy_sensitivity"] * norm_energy
        + coupled_duration
        + adaptive_rank
        + 0.3 * norm_work  # fixed light weight: avoid biasing by workload size alone
        - 0.2 * wait_boost  # anti-starvation: subtract boost → lowers score for long-waiting tasks
        + PARAMS["uncertainty_slack_interaction"] * risk_gate
    )

    # Ensure finite, shape-(N,) output; handle edge cases (e.g., all identical inputs)
    score = np.nan_to_num(score, nan=0.0, posinf=0.0, neginf=0.0)
    
    # Clip to declared safe bounds — both bounds now positive for log transform compliance
    score = np.clip(score, PARAMS["clip_lower_bound_log_scale"], PARAMS["clip_upper_bound"])

    # Enforce shape (N,) even for scalar case
    if score.ndim == 0:
        score = np.full(1, score.item(), dtype=float)

    return score
