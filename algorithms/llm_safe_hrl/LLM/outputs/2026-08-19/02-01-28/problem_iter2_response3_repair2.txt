def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: hybrid slack gating + multi-scale robust normalization + dual wait modeling.
    
    Structural improvements:
    - Combines Parent 2's nonlinear slack_powered with Parent 1's wait_urgency_exponent for stronger anti-starvation
    - Replaces single wait_bonus with *two* complementary wait signals: saturating exponential (latency-aware) AND power-law urgency (starvation-robust)
    - Uses hybrid scale selection: iqr → rng → max(1, median) for even more degenerate-case stability
    - Introduces slack-aware duration weighting: duration_balance adapts based on slack pressure magnitude
    - All numeric literals are strictly -2,-1,0,1,2 — no hidden constants
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization with fallback hierarchy: iqr → range → median-based scale
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q75 = np.quantile(x, PARAMS["q75_quantile"])
        q25 = np.quantile(x, PARAMS["q25_quantile"])
        iqr = q75 - q25
        rng = np.max(x) - np.min(x)
        med = np.median(x)
        # Scale fallback: iqr > eps → iqr; else rng > eps → rng; else use max(1, |med|)
        scale = np.where(
            iqr > eps,
            iqr,
            np.where(
                rng > eps,
                rng,
                np.maximum(1.0, np.abs(med))
            )
        )
        return (x - med) / (scale + eps)

    # Compute base normalized features
    exec_comm = min_exec_time + min_comm_time
    duration_norm = robust_normalize(exec_comm)
    energy_norm = robust_normalize(min_incremental_energy)
    rank_norm = robust_normalize(upward_rank)
    work_norm = robust_normalize(remaining_work)
    wait_norm = robust_normalize(ready_wait_time)
    
    # Slack processing: nonlinear pressure using powered absolute slack with sign retention
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_powered = slack_sign * (slack_abs ** PARAMS["slack_sensitivity"])
    slack_norm = robust_normalize(slack_powered)
    
    # Lateness risk gate: linear penalty only when slack < 0
    lateness_risk = np.where(
        slack < 0,
        PARAMS["slack_penalty_scale"] * (-slack),
        0.0
    )
    
    # Uncertainty-slack interaction: multiplicative boost to urgency when both are high
    unc_norm = robust_normalize(uncertainty)
    slack_pressure = np.maximum(-slack_norm, 0.0)  # positive pressure under deadline stress
    interaction_term = PARAMS["uncertainty_slack_interaction"] * unc_norm * slack_pressure
    
    # Criticality boost: apply only under deadline pressure or high uncertainty
    criticality_gate = np.maximum(slack_pressure, unc_norm)
    boosted_rank = rank_norm * (1.0 + PARAMS["criticality_boost"] * criticality_gate)
    
    # Dual wait modeling:
    # (1) Saturating exponential bonus (bounded [0,1]) for latency-sensitive fairness
    wait_bonus_exp = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    # (2) Power-law urgency for starvation prevention — normalized and scaled
    wait_base = np.maximum(ready_wait_time, eps)
    wait_urgency_power = np.power(wait_base, PARAMS["wait_urgency_exponent"])
    wait_norm_power = robust_normalize(wait_urgency_power)
    
    # Adaptive duration-energy balance: tighten duration focus when slack is tight
    slack_pressure_mag = np.maximum(-slack, 0.0)
    # Use tanh for smooth, bounded adaptation (range [-1,1] → [0,1] via (tanh+1)/2), then scale by parameter
    adaptive_duration_weight = (
        PARAMS["duration_balance"] 
        + PARAMS["duration_balance"] * 0.5 * (np.tanh(slack_pressure_mag) + 1.0) * 0.5
    )
    dur_energy_combo = (
        adaptive_duration_weight * duration_norm
        + (1.0 - adaptive_duration_weight) * energy_norm
    )
    
    # Final score: smaller = higher priority
    score = (
        lateness_risk
        + interaction_term
        - boosted_rank
        + PARAMS["energy_norm_factor"] * dur_energy_combo
        - wait_bonus_exp
        - wait_norm_power
        + PARAMS["work_weight"] * work_norm
    )
    
    # Ensure finite output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
