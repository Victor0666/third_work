def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric thresholds declared; only -2,-1,0,1,2 used as literals."""
    eps = PARAMS["epsilon"]
    
    # Robust normalization using configurable quantiles
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q1_val = PARAMS["q1_quantile"]
        q3_val = PARAMS["q3_quantile"]
        q1, q3 = np.quantile(x, [q1_val, q3_val])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Slack pressure: signed power law — sharp rise for negative slack, flat positive regime
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    # Avoid 0^0: use eps for zero slack
    slack_power = np.where(
        slack_abs < eps,
        0.0,
        slack_sign * (slack_abs ** PARAMS["slack_sensitivity"])
    )
    slack_penalty = PARAMS["slack_penalty_scale"] * slack_power

    # Criticality boost: only activate when under deadline risk or high uncertainty
    median_uncertainty = np.median(uncertainty) if len(uncertainty) > 1 else np.mean(uncertainty)
    risk_condition = (slack <= eps) | (uncertainty > median_uncertainty + eps)
    boosted_rank = upward_rank * (1.0 + PARAMS["criticality_boost"] * risk_condition.astype(float))

    # Robust energy normalization (configurable quantile-based)
    energy_norm = robust_normalize(min_incremental_energy)
    energy_score = PARAMS["energy_normalizer"] * energy_norm

    # Duration ratio: compute-dominance indicator [0,1]; favors compute-heavy tasks for edge offload
    exec_plus_comm = min_exec_time + min_comm_time + eps
    duration_ratio = min_exec_time / exec_plus_comm
    ratio_score = PARAMS["duration_ratio_weight"] * (1.0 - duration_ratio)

    # Anti-starvation: exponential wait boost — gentle, monotonic, bounded growth
    wait_boost = np.exp(-PARAMS["wait_decay"] * ready_wait_time)

    # Uncertainty-slack coupling: multiplicative gate on slack penalty when uncertainty is high
    unc_slack_gate = 1.0 + PARAMS["uncertainty_slack_coupling"] * (uncertainty / (np.mean(uncertainty) + eps))
    gated_slack = slack_penalty * unc_slack_gate

    # Assemble final score: smaller = higher priority
    score = (
        gated_slack
        + energy_score
        - robust_normalize(boosted_rank)
        + ratio_score
        - wait_boost
    )

    # Ensure finite output and shape compliance
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
