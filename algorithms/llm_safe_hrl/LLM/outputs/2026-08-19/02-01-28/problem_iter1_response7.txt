def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: Slack-critical gating + risk-aware efficiency + anti-starvation.
    
    Key innovations:
    - Critical slack gate: applies strong penalty only when slack <= threshold (not linear everywhere)
    - Uncertainty-augmented duration: multiplies exec+comm by (1 + uncertainty) for risk-aware sizing
    - Nonlinear wait urgency: pow(ready_wait_time + epsilon, exponent) for gentle starvation relief
    - Upward rank only matters when slack is safe — avoids over-prioritizing critical path in violation risk
    - Energy efficiency bias activated only in safe slack region to avoid trading DDL for energy
    - All features normalized via robust IQR-based scaling (more outlier-resistant than mean-based)
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization: use interquartile range + median to resist outliers
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q1, q3 = np.percentile(x, [25, 75], method='midpoint')
        iqr = q3 - q1 + eps
        med = np.median(x)
        return (x - med) / iqr
    
    # --- Slack-driven criticality gating ---
    # Safe region: slack > threshold → enable efficiency & rank terms
    # Critical region: slack <= threshold → activate strong penalty and duration risk
    slack_safe_mask = (slack > PARAMS["critical_slack_threshold"]).astype(float)
    
    # Penalty term: large positive score for negative/low slack (i.e., low priority *unless* negated)
    # Since smaller score = higher priority, we want *large* scores for risky tasks → subtract penalty
    slack_penalty = np.where(
        slack <= PARAMS["critical_slack_threshold"],
        PARAMS["slack_penalty_factor"] * np.maximum(0.0, -slack + PARAMS["critical_slack_threshold"]),
        0.0
    )
    
    # --- Duration risk: exec + comm, amplified by uncertainty ---
    raw_duration = min_exec_time + min_comm_time
    risk_duration = raw_duration * (1.0 + uncertainty)
    
    # --- Energy efficiency: only prioritized when slack is safe ---
    energy_term = robust_normalize(min_incremental_energy)
    energy_contribution = PARAMS["energy_efficiency_bias"] * energy_term * slack_safe_mask
    
    # --- Upward rank: only weighted when slack is safe (no false criticality) ---
    rank_term = robust_normalize(upward_rank)
    rank_contribution = PARAMS["rank_sensitivity"] * rank_term * slack_safe_mask
    
    # --- Anti-starvation: nonlinear wait urgency (monotonic, bounded growth) ---
    # Use power law instead of linear to gently boost old tasks without exploding
    wait_base = np.maximum(ready_wait_time, eps)
    wait_urgency = np.power(wait_base, PARAMS["wait_urgency_exponent"])
    wait_term = robust_normalize(wait_urgency)
    
    # --- Uncertainty-slack coupling: amplifies duration weight when slack is tight ---
    # This makes uncertain-but-urgent tasks more expensive to delay
    duration_weight = (
        PARAMS["duration_risk_weight"] 
        + PARAMS["uncertainty_slack_coupling"] * np.maximum(0.0, PARAMS["critical_slack_threshold"] - slack)
    )
    
    # Normalize duration *after* applying uncertainty coupling
    duration_term = robust_normalize(risk_duration)
    
    # --- Assemble final score: minimize this → highest priority ---
    # Note: slack_penalty is *added* because it's a cost (larger = worse); others are base contributors
    # We want small total score → so critical slack pushes score up (low priority), while
    # efficiency/rank/wait push score down (higher priority) when beneficial.
    score = (
        slack_penalty
        + duration_weight * duration_term
        + energy_contribution
        - rank_contribution
        - wait_term
    )
    
    # Ensure finite output and handle edge cases (NaN, inf)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
