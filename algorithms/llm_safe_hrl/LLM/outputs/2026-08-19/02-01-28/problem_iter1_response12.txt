def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: risk-aware, piecewise-gated, and deadline-centric.
    
    Key innovations:
      - Uses smooth tanh-based slack urgency with explicit negative-slack penalty
      - Upward rank boost is conditional: only active when slack >= 0 AND normalized rank > threshold
      - Uncertainty-slack coupling via product gate (not additive), capped by tanh
      - Anti-starvation uses saturating exp(-wait_decay * wait) → bonus decays smoothly
      - All features normalized robustly using IQR + epsilon (more outlier-resistant than mean)
      - Energy term is inverted: lower energy → higher priority → subtracted from score
      - Final score is strictly finite, deterministic, and shaped (N,)
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    if N == 0:
        return np.array([], dtype=float)

    # Robust normalization: use IQR to resist outliers; fallback to std if degenerate
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q75, q25 = np.percentile(x, [75, 25], method='midpoint')
        iqr = q75 - q25
        center = np.median(x)
        scale = np.where(iqr > eps, iqr, np.std(x, ddof=0) + eps)
        return (x - center) / (scale + eps)

    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # === SLACK URGENCY: steep, bounded, sign-aware ===
    # tanh-based soft urgency: near zero slack → rapid rise; negative slack → strong penalty
    slack_urgency = np.tanh(PARAMS["slack_sensitivity"] * norm_slack)
    # Explicit penalty for negative slack: add scaled penalty only where slack < 0
    neg_slack_mask = (slack < 0).astype(float)
    slack_penalty = neg_slack_mask * PARAMS["slack_risk_penalty"] * (1.0 - np.tanh(norm_slack))

    # === UNCERTAINTY-SLACK COUPLING ===
    # Only activate coupling when both are high-pressure: product gate + tanh saturation
    coupling_gate = np.tanh(norm_uncert * norm_slack)  # negative when slack<0 → reduces coupling
    coupling_term = PARAMS["uncertainty_slack_coupling"] * np.abs(coupling_gate)

    # === CONDITIONAL CRITICAL PATH BOOST ===
    # Boost upward_rank only if slack is non-negative AND rank is above threshold percentile
    rank_boost_mask = (
        (slack >= 0).astype(float) *
        (norm_rank > PARAMS["rank_saturation_threshold"]).astype(float)
    )
    rank_boost = rank_boost_mask * PARAMS["critical_path_leverage"] * np.maximum(norm_rank, 0)

    # === ANTI-STARVATION BONUS (saturating exponential) ===
    # Higher wait → larger bonus → lowers score; saturates at ~95% of max after 3/decay steps
    wait_bonus = 1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + eps))
    # Normalize bonus to [0, 1] and invert priority effect: bonus reduces score
    wait_priority = -wait_bonus  # smaller score = higher priority

    # === ENERGY EFFICIENCY TERM ===
    # Lower incremental energy → higher priority → subtract from score
    energy_priority = -PARAMS["energy_efficiency_scale"] * norm_energy

    # === DURATION BALANCE TERM ===
    # Shorter duration → slightly higher priority (but secondary to deadline & criticality)
    duration_priority = -PARAMS["duration_balance"] * norm_duration

    # Assemble final score: smaller = better
    # All terms designed to be finite, stable, and bounded in [-10, 10] range
    score = (
        slack_urgency               # baseline urgency (tanh-scaled)
        + slack_penalty             # explicit late-risk penalty
        + coupling_term             # joint pressure signal
        - rank_boost                # conditional critical path boost
        + energy_priority           # reward low-energy placement
        + duration_priority         # reward fast-executing tasks
        + wait_priority             # anti-starvation
        + 0.5 * norm_work           # light bias toward large remaining work (avoids premature leaf execution)
    )

    # Final safeguard: clip extreme values and ensure finite output
    score = np.clip(score, -1e6, 1e6)
    score = np.nan_to_num(score, nan=0.0, posinf=1e6, neginf=-1e6)

    return score
