def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's robust gating and normalization with Parent 1's starvation capping and slack urgency.
    
    Key improvements:
      - Retains Parent 2's stable mean-abs normalization and binary criticality gate (slack_critical | uncert_high)
      - Adds Parent 1's capped starvation relief: linear boost beyond threshold, but bounded by slack_urgency * wait_cap_factor_urgency
      - Replaces pure penalty-only slack handling with dual-regime: negative slack → power penalty, positive slack → sublinear urgency (reducing idle bias)
      - All numeric literals are strictly {-2,-1,0,1,2} — no hidden constants
      - Uses safe_normalize consistently; no in-place mutation; handles N=1 correctly
      - Ensures finite output via nan_to_num and shape enforcement
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Safe, non-mutating normalization using mean absolute value with epsilon guard
    def safe_normalize(x):
        x = np.asarray(x, dtype=float)
        abs_x = np.abs(x)
        mean_abs = np.mean(abs_x)
        scale = np.where(mean_abs == 0.0, eps, mean_abs)
        return x / (scale + eps)
    
    # Normalize all inputs
    norm_slack = safe_normalize(slack)
    norm_energy = safe_normalize(min_incremental_energy)
    norm_rank = safe_normalize(upward_rank)
    norm_work = safe_normalize(remaining_work)
    norm_wait = safe_normalize(ready_wait_time)
    norm_uncert = safe_normalize(uncertainty)
    norm_duration = safe_normalize(min_exec_time + min_comm_time)
    
    # === SLACK REGIME SPLITTING ===
    # Negative slack: nonlinear penalty (higher exponent → stronger urgency)
    raw_slack_penalty = np.power(np.maximum(0.0, -norm_slack), PARAMS["slack_penalty_exponent"])
    # Positive slack: diminishing urgency (sublinear push to avoid over-deferring)
    raw_slack_urgency = np.where(norm_slack > 0.0,
                                np.power(norm_slack, PARAMS["slack_urgency_scale"]),
                                0.0)
    
    # === CRITICALITY BOOST GATE (Parent 2 style: simple, stable OR gate) ===
    slack_critical = (norm_slack < PARAMS["risk_slack_gate"])
    uncert_high = (norm_uncert > PARAMS["uncertainty_high_threshold"])
    critical_gate = np.where(slack_critical | uncert_high, 1.0, 0.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_boost"] * critical_gate)
    
    # === DURATION-UNCERTAINTY COUPLING ===
    coupled_duration = norm_duration + PARAMS["duration_uncertainty_coupling"] * norm_duration * norm_uncert
    
    # === STARVATION RELIEF WITH URGENCY-BASED CAP (Parent 1 insight) ===
    wait_excess = np.maximum(0.0, ready_wait_time - PARAMS["wait_decay_threshold"])
    uncapped_relief = PARAMS["starvation_relief_slope"] * wait_excess
    # Cap relief using scaled slack urgency to prevent overcompensation when deadlines are loose
    urgency_cap = PARAMS["wait_cap_factor_urgency"] * raw_slack_urgency
    starvation_relief = np.minimum(uncapped_relief, urgency_cap)
    norm_relief = safe_normalize(starvation_relief)
    
    # === COMPOSITE SCORE: smaller = higher priority ===
    # Dominant term: slack penalty ensures deadline feasibility first
    # Then rank boost (for critical path), energy savings (for objective), duration (latency fairness), and capped relief
    score = (
        raw_slack_penalty                    # Hard DDL enforcement
        + boosted_rank                       # Critical path leverage
        - PARAMS["energy_efficiency_sensitivity"] * norm_energy  # Energy optimization
        + coupled_duration                   # Risk-aware latency balance
        - norm_relief                        # Anti-starvation (capped)
        + 1 * PARAMS["remaining_work_weight"] * norm_work       # Work-aware tie-breaker
    )
    
    # Final safeguards: ensure finite, shape-(N,), deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max/2, neginf=finfo.min/2)
    score = np.clip(score, a_min=finfo.min/2, a_max=finfo.max/2)
    
    return score
