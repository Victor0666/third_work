def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces additive feasibility guard with conditional DDL-protection gate,
    introduces upward_rank × remaining_work interaction (not isolated rank), suppresses energy under
    negative slack, uses bounded sigmoid for starvation relief, and applies robust mean-abs scaling.
    
    Structural changes:
      - Replaces hard feasibility_guard_weight with conditional slack-risk gating: no additive penalty,
        but instead modulates criticality and energy terms via smooth sigmoid gates.
      - Upward rank now interacts multiplicatively with remaining_work — captures critical-path leverage,
        not just importance.
      - Energy term is suppressed when slack < 0 (DDL risk dominates), activated only when slack >= 0.
      - Starvation relief uses bounded sigmoid instead of arctan — smoother saturation at unit limit.
      - All feature normalizations use mean-abs (more stable than median/MAD for small N).
      - Duration-uncertainty coupling now directly modulates slack penalty, not added separately.
      - Removes redundant 'feasibility_guard_weight' parameter — aligns with evidence that it was inactive.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Robust per-feature normalization using mean & mean-abs deviation (stable for small N)
    def mean_abs_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.mean(x)
        spread = np.mean(np.abs(x - center)) + eps
        return (x - center) / (spread + eps)

    norm_slack = mean_abs_normalize(slack)
    norm_energy = mean_abs_normalize(min_incremental_energy)
    norm_duration = mean_abs_normalize(min_exec_time + min_comm_time)
    norm_rank = mean_abs_normalize(upward_rank)
    norm_work = mean_abs_normalize(remaining_work)
    norm_wait = mean_abs_normalize(ready_wait_time)
    norm_uncert = mean_abs_normalize(uncertainty)

    # --- Slack risk: bounded sigmoid pressure (monotonic, stable, avoids exponent overflow)
    # Maps slack → [0, 1] where lower slack → higher risk score; uses soft step near zero
    slack_risk = 1.0 / (1.0 + np.exp(-(norm_slack - PARAMS["slack_negative_threshold"]) / eps))
    
    # --- Critical path leverage: upward_rank × remaining_work, gated by slack risk
    critical_leverage = norm_rank * norm_work
    critical_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    leveraged_criticality = critical_leverage * (1.0 + PARAMS["criticality_scale"] * critical_gate)

    # --- Uncertainty-modulated slack penalty: only active when both slack and uncertainty are tight
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) & 
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]), 
        1.0, 0.0
    )
    # Apply uncertainty-weighted slack penalty: stronger penalty under joint tightness
    uncertainty_modulated_slack = slack_risk * (1.0 + PARAMS["duration_robustness"] * norm_uncert * dur_uncert_gate)

    # --- Energy term: only active when slack is non-negative (DDL-safe region)
    energy_term = np.where(slack >= 0,
                          PARAMS["energy_sensitivity"] * norm_energy,
                          0.0)

    # --- Bounded starvation relief: sigmoid saturation (smooth, finite, unit-bounded)
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_benefit = 1.0 - 1.0 / (1.0 + wait_scaled)

    # --- Work-based suppression: soft penalty for large remaining_work, scaled by slack risk
    # Larger work penalized more under high deadline risk
    work_ratio = np.clip(remaining_work / (PARAMS["work_leverage_threshold"] + eps),
                         np.finfo(float).tiny,
                         np.finfo(float).max)
    work_suppression = work_ratio / (1.0 + work_ratio) * slack_risk

    # --- Final score: smaller = better
    # Order of terms reflects lexicographic priority: DDL risk → critical path → energy → starvation → work
    score = (
        uncertainty_modulated_slack          # Primary DDL risk signal (gated & modulated)
        - leveraged_criticality              # Boost tasks that unblock large critical work
        + energy_term                        # Secondary: minimize energy only when safe
        - wait_benefit                       # Anti-starvation (additive benefit → lowers score)
        + work_suppression                   # Penalize large work under risk (soft, risk-scaled)
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
