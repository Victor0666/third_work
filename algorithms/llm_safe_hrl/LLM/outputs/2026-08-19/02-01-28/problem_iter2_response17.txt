def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining robust urgency gating, joint critical-path/workload emphasis,
       and uncertainty-amplified energy-duration penalties — all bounded and differentiable.
    
    Key improvements:
    - Replaces hard masks/clips with smooth sigmoidal gates for better CMA-ES gradient flow.
    - Introduces explicit upward_rank * remaining_work interaction to jointly prioritize bottleneck tasks
      with large descendant computational scope (supported by counterfactual evidence).
    - Uncertainty-slack interaction multiplicatively scales *both* energy and duration penalties,
      not just a gate — directly penalizing high marginal energy under deadline pressure.
    - Robustness penalty uses uncertainty^exponent instead of CV to avoid fragile std estimation on small N.
    - Uses median-centered normalization with MAD fallback (outlier-resilient), no mean/std degeneracy logic.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Defensive copy & cast
    min_exec_time = np.asarray(min_exec_time, dtype=float).copy()
    min_comm_time = np.asarray(min_comm_time, dtype=float).copy()
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float).copy()
    slack = np.asarray(slack, dtype=float).copy()
    upward_rank = np.asarray(upward_rank, dtype=float).copy()
    remaining_work = np.asarray(remaining_work, dtype=float).copy()
    ready_wait_time = np.asarray(ready_wait_time, dtype=float).copy()
    uncertainty = np.asarray(uncertainty, dtype=float).copy()

    # Robust normalization: median + MAD (more resilient than std for small N or outliers)
    def normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        dev = np.median(np.abs(x - center))
        scale = np.maximum(dev, eps)
        return (x - center) / scale

    norm_slack = normalize(slack)
    norm_energy = normalize(min_incremental_energy)
    norm_rank = normalize(upward_rank)
    norm_work = normalize(remaining_work)
    norm_wait = normalize(ready_wait_time)
    norm_uncert = normalize(uncertainty)
    duration = min_exec_time + min_comm_time
    norm_duration = normalize(duration)

    # --- Sigmoid urgency gate: smooth, bounded, centered at threshold ---
    # Higher score = higher urgency (to be added to priority)
    slack_urgency = 1.0 / (1.0 + np.exp(-PARAMS["slack_risk_penalty"] * (norm_slack - PARAMS["critical_slack_threshold"])))

    # --- Uncertainty-slack coupling gate: soft binary interaction for amplification ---
    # Clipped to [0,1] to avoid runaway scaling; tanh provides smoother saturation than clip
    risk_gate = np.tanh(norm_uncert * slack_urgency)

    # --- Amplified energy penalty: base energy + uncertainty-coupled marginal increase ---
    energy_penalty = norm_energy * (1.0 + PARAMS["uncertainty_slack_interaction"] * risk_gate)

    # --- Amplified duration penalty: base duration + uncertainty-coupled volatility cost ---
    duration_penalty = norm_duration * (1.0 + PARAMS["duration_coupling"] * risk_gate)

    # --- Critical-path workload interaction: rank * work → captures bottleneck impact magnitude ---
    rank_work_interaction = norm_rank * norm_work * PARAMS["rank_work_interaction_weight"]

    # --- Robustness penalty: power-law penalty on normalized uncertainty (avoids CV instability) ---
    robustness_penalty = np.power(np.maximum(norm_uncert, eps), PARAMS["robustness_exponent"])

    # --- Anti-starvation boost: subtracted (lowers score) for long-waiting tasks ---
    # Sublinear but monotonic: avoids dominance while ensuring fairness
    wait_boost = PARAMS["wait_boost_weight"] * np.sqrt(np.maximum(norm_wait, eps))

    # --- Final priority score: smaller = higher priority ---
    # Order of terms reflects scheduling hierarchy: DDL risk > energy > criticality > robustness > fairness
    score = (
        slack_urgency
        + PARAMS["energy_sensitivity"] * energy_penalty
        + duration_penalty
        + rank_work_interaction
        + robustness_penalty
        - wait_boost
    )

    # Final numerical safeguards
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    score = np.clip(score, PARAMS["clip_lower_bound_log_scale"], PARAMS["clip_upper_bound"])

    # Enforce shape (N,)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"

    return score
