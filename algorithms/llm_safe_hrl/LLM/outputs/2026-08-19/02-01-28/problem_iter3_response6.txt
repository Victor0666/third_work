def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: adds slack-energy coupling and wait-slack interaction.
    Removes redundant comm-energy term per reflection; restores rational positive_relief.
    Introduces bounded, context-aware energy weighting and starvation relief scaling."""
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Robust per-feature normalization: median + clipped MAD, immune to zero-spread
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        abs_devs = np.abs(x - center)
        spread = np.clip(np.median(abs_devs), eps, finfo.max) + eps
        return (x - center) / spread

    # Normalize core features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # --- Risk-aware slack penalty: asymmetric & rational ---
    slack_abs = np.abs(slack)
    negative_penalty = np.where(slack < 0, 
                                np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"]), 
                                0.0)
    # Rational anti-penalty for positive slack: monotonic, bounded, avoids log distortion
    positive_relief = np.where(slack > 0, 
                               -slack_abs / (1.0 + slack_abs + eps), 
                               0.0)
    slack_risk_score = negative_penalty + positive_relief

    # --- Step-gated criticality amplification (reverted per reflection) ---
    critical_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    amplified_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * critical_gate)

    # --- Duration-uncertainty coupling: dual-threshold ANDed activation ---
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) & 
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]), 
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Slack-modulated energy sensitivity: prioritize energy minimization more under pressure ---
    # Increases energy weight when slack is negative or near-zero, bounded by [0, 1.5]
    slack_pressure_weight = np.clip(1.0 + (-norm_slack) * PARAMS["slack_energy_coupling_weight"], 0.0, 1.5)
    modulated_energy_term = slack_pressure_weight * norm_energy * PARAMS["energy_sensitivity"]

    # --- Wait-slack interaction: anti-starvation scaled by deadline pressure ---
    # Stronger wait benefit when slack is tight (negative or low), using power law
    wait_slack_factor = np.abs(norm_slack) ** PARAMS["wait_slack_interaction_power"]
    wait_benefit = (1.0 - np.exp(-PARAMS["wait_decay"] * (np.maximum(norm_wait, 0.0) + eps))) * wait_slack_factor

    # --- Work saturation: sigmoid-like scaling with explicit clipping ---
    work_ref = PARAMS["work_saturation_point"]
    work_ratio = np.clip(remaining_work / (work_ref + eps), 
                         finfo.tiny, 
                         finfo.max)
    work_scaled = 1.0 / (1.0 + 1.0 / (work_ratio + eps))

    # --- Final score: smaller = higher priority ---
    # Dominant: slack risk, modulated energy, duration-uncert
    # Suppressive: rank (only when urgent), work (small favored), wait (anti-starve, pressure-scaled)
    score = (
        slack_risk_score                              # Primary DDL risk term
        + modulated_energy_term                       # Energy prioritization adapts to slack pressure
        + dur_uncert_interaction                      # Uncertain long tasks under pressure
        - amplified_rank                              # Critical rank boost only when needed
        + work_scaled                                 # Prefer small-work tasks
        - wait_benefit                                # Anti-starvation, enhanced under deadline stress
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    return score
