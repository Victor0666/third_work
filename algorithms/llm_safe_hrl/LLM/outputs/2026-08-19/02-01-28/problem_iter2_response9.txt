def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining robust deadline-risk modulation (Parent 2) with
    uncertainty-aware energy-efficiency gating and rank-slack interaction (novel).
    
    Key improvements:
      - Introduces rank-slack interaction term: upward_rank * |slack|^exponent → 
        strongly prioritizes high-criticality tasks *only* when slack is tight
      - Adds energy-uncertainty coupling: downweights energy density preference 
        where uncertainty is high (avoids over-committing to fragile low-energy options)
      - Keeps smooth sigmoid slack gating + quadratic duration flexibility bias (Parent 2 strength)
      - Uses work density (remaining_work / duration) as primary efficiency signal
      - All normalization is robust, all divisions guarded, no literals beyond {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Defensive casting and safe NaN/inf handling
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=eps_safe, posinf=finfo.max, neginf=-finfo.max)

    # Robust normalization using mean absolute value + epsilon
    def robust_normalize(x):
        x_abs = np.abs(x)
        denom = np.mean(x_abs) + eps_safe
        return x / denom

    # Duration and derived signals
    duration = min_exec_time + min_comm_time
    norm_duration = robust_normalize(duration)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # Work density: compute work per time unit — higher is more efficient
    work_density = remaining_work / (duration + eps_safe)
    norm_work_density = robust_normalize(work_density)

    # Slack gating: smooth sigmoid transition at zero slack
    slack_gate = 1.0 / (1.0 + np.exp(-2.0 * norm_slack))  # Sigmoid centered at 0
    slack_penalty_term = PARAMS["slack_risk_penalty"] * (1.0 - slack_gate) * np.abs(norm_slack)
    slack_gain_term = PARAMS["slack_benefit_gain"] * slack_gate * np.abs(norm_slack)
    coupled_slack = (slack_penalty_term + slack_gain_term) * (
        1.0 + PARAMS["uncertainty_slack_coupling"] * norm_uncert
    )

    # Critical path importance: higher upward_rank → higher priority → negate for min-score
    critical_term = -PARAMS["critical_path_weight"] * norm_rank

    # Work density efficiency: prefer high work-per-time
    work_density_term = -PARAMS["work_density_weight"] * norm_work_density

    # Energy density: marginal energy per duration → lower is better
    energy_density = min_incremental_energy / (duration + eps_safe)
    norm_energy_density = robust_normalize(energy_density)
    # Uncertainty-coupled energy preference: reduce weight where uncertainty is high
    energy_density_term = PARAMS["energy_density_weight"] * norm_energy_density * (
        1.0 - PARAMS["energy_uncertainty_coupling"] * norm_uncert
    )

    # Starvation relief: longer wait → higher priority
    starvation_term = -PARAMS["wait_starvation_relief"] * norm_wait

    # Duration flexibility bias: quadratic penalty around |norm_duration| == 1
    flex_penalty = PARAMS["duration_flexibility_bias"] * ((np.abs(norm_duration) - 1.0) ** 2)

    # Rank-slack interaction: amplifies priority for high-rank tasks *under deadline pressure*
    # Only active when slack is negative or near-zero (tight constraint)
    rank_slack_interaction = norm_rank * (np.abs(norm_slack) ** PARAMS["rank_slack_interaction_exponent"])
    # Apply gate to avoid dominance: only boost when slack is moderately negative
    interaction_gate = np.where(norm_slack < 0.5, 1.0, 0.0)
    interaction_term = -PARAMS["critical_path_weight"] * rank_slack_interaction * interaction_gate

    # Assemble final score: smaller = higher priority
    score = (
        coupled_slack
        + critical_term
        + work_density_term
        + energy_density_term
        + starvation_term
        + flex_penalty
        + interaction_term
    )

    # Final safeguard: ensure finite, bounded output
    return np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
