def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all quantiles and thresholds declared as parameters; no hidden literals."""
    eps = PARAMS["epsilon"]
    
    # Robust normalization using declared quantiles
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q75 = np.quantile(x, PARAMS["q75_quantile"])
        q25 = np.quantile(x, PARAMS["q25_quantile"])
        iqr = q75 - q25
        rng = np.max(x) - np.min(x)
        scale = np.where(iqr > eps, iqr, np.where(rng > eps, rng, 1.0))
        center = np.median(x)
        return (x - center) / (scale + eps)

    # Compute base normalized features
    exec_comm = min_exec_time + min_comm_time
    duration_norm = robust_normalize(exec_comm)
    energy_norm = robust_normalize(min_incremental_energy)
    rank_norm = robust_normalize(upward_rank)
    work_norm = robust_normalize(remaining_work)
    wait_norm = robust_normalize(ready_wait_time)
    
    # Slack processing: two regimes — aggressive penalty for negative slack, soft decay for positive
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_powered = np.where(
        slack >= 0,
        slack_abs ** PARAMS["slack_sensitivity"],
        - (slack_abs ** PARAMS["slack_sensitivity"])
    )
    slack_norm = robust_normalize(slack_powered)
    
    # Lateness risk gate: linear penalty only when slack < 0, scaled by slack_penalty_scale
    lateness_risk = np.where(
        slack < 0,
        PARAMS["slack_penalty_scale"] * (-slack),
        0.0
    )
    
    # Uncertainty-slack interaction: multiplicative boost to urgency when both are high
    unc_norm = robust_normalize(uncertainty)
    slack_pressure = np.maximum(-slack_norm, 0.0)
    interaction_term = PARAMS["uncertainty_slack_interaction"] * unc_norm * slack_pressure
    
    # Criticality boost: apply only under deadline pressure or high uncertainty
    criticality_gate = np.maximum(slack_pressure, unc_norm)
    boosted_rank = rank_norm * (1.0 + PARAMS["criticality_boost"] * criticality_gate)
    
    # Anti-starvation: saturating exponential bonus (bounded in [0,1]) for waiting time
    wait_bonus = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    
    # Balanced duration-energy composite: same scale, weighted average
    dur_energy_combo = (
        PARAMS["duration_balance"] * duration_norm
        + (1.0 - PARAMS["duration_balance"]) * energy_norm
    )
    
    # Final score: smaller = higher priority
    score = (
        lateness_risk
        + interaction_term
        - boosted_rank
        + PARAMS["energy_norm_factor"] * dur_energy_combo
        - wait_bonus
        + 0.3 * work_norm
    )
    
    # Ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
