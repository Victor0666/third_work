import numpy as np

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Robust priority rule: asymmetric slack penalty, IQR-normalized features,
    conditional criticality boost, and bounded anti-starvation term.
    Smaller score = higher priority; satisfies all contract constraints."""
    
    # Safeguard epsilon using machine epsilon, not literal
    eps = np.finfo(float).tiny if PARAMS["epsilon"] < np.finfo(float).tiny else PARAMS["epsilon"]
    
    # Convert to float arrays, avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # Robust IQR-based normalization with symmetric absolute clipping
    def iqr_normalize(x):
        x_clipped = np.clip(x, -PARAMS["iqr_clip_abs"], PARAMS["iqr_clip_abs"])
        q1, q3 = np.quantile(x_clipped, [0.25, 0.75])
        iqr = q3 - q1 + eps
        center = np.median(x_clipped)
        return (x_clipped - center) / iqr
    
    # Duration = exec + comm; normalized robustly
    duration = min_exec_time + min_comm_time
    norm_duration = iqr_normalize(duration)
    
    # Slack normalization with *asymmetric* penalty: sharp rise for negative slack
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    # Use power law only where magnitude is meaningful; linear near zero
    slack_magnitude = np.where(
        slack_abs < eps,
        slack_abs,
        np.power(slack_abs, PARAMS["slack_sensitivity"])
    )
    # Negative slack → large positive pressure → subtracted later → smaller score
    slack_pressure = -slack_sign * slack_magnitude
    norm_slack_pressure = iqr_normalize(slack_pressure)
    
    # Energy: normalize robustly
    norm_energy = iqr_normalize(min_incremental_energy)
    
    # Upward rank: boost when slack ≤ 0 OR uncertainty > median
    median_uncert = np.median(uncertainty) if N > 1 else uncertainty[0]
    criticality_gate = np.where(
        (slack <= 0.0) | (uncertainty > median_uncert + eps),
        PARAMS["criticality_boost"],
        1.0
    )
    norm_rank = iqr_normalize(upward_rank) * criticality_gate
    
    # Remaining work: higher work → lower priority → invert
    norm_work = -iqr_normalize(remaining_work)
    
    # Ready wait time: anti-starvation via bounded arctan
    wait_scaled = ready_wait_time * PARAMS["wait_decay_rate"]
    norm_wait = np.arctan(wait_scaled)  # ∈ [0, π/2), monotonic, finite
    
    # Uncertainty-slack coupling: product of normalized signals
    norm_uncert = iqr_normalize(uncertainty)
    uncertainty_slack_coupling = norm_uncert * norm_slack_pressure * PARAMS["uncertainty_slack_interaction"]
    
    # Final score: smaller = better
    score = (
        PARAMS["slack_penalty_scale"] * norm_slack_pressure
        + PARAMS["energy_norm_factor"] * norm_energy
        + PARAMS["duration_balance"] * norm_duration
        - norm_rank
        + norm_work
        - norm_wait
        + uncertainty_slack_coupling
    )
    
    # Ensure finiteness and shape
    score = np.nan_to_num(score, nan=0.0, posinf=1e12, neginf=-1e12)
    assert score.shape == (N,), f"Expected {(N,)}, got {score.shape}"
    
    return score
