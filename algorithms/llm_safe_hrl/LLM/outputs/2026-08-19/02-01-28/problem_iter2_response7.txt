def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's smooth slack gating and work density
    with Parent 1's starvation saturation and uncertainty-aware coupling, plus novel
    adaptive normalization and bounded latency-energy tradeoff."""
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Adaptive robust normalization: blends median-MAD and mean-abs using tunable rate
    def adaptive_normalize(x):
        x = np.asarray(x, dtype=float)
        # Median-based center & spread
        med = np.median(x)
        mad = np.median(np.abs(x - med)) + eps_safe
        norm_med = (x - med) / mad
        # Mean-abs-based normalization
        mean_abs = np.mean(np.abs(x)) + eps_safe
        norm_meanabs = x / mean_abs
        # Blend using robustness_adaptation_rate
        return PARAMS["robustness_adaptation_rate"] * norm_med + \
               (1.0 - PARAMS["robustness_adaptation_rate"]) * norm_meanabs

    # Core duration and derived features
    duration = min_exec_time + min_comm_time
    norm_duration = adaptive_normalize(duration)
    norm_slack = adaptive_normalize(slack)
    norm_energy = adaptive_normalize(min_incremental_energy)
    norm_rank = adaptive_normalize(upward_rank)
    norm_work = adaptive_normalize(remaining_work)
    norm_wait = adaptive_normalize(ready_wait_time)
    norm_uncert = adaptive_normalize(uncertainty)

    # Work density: compute efficiency — remaining_work per time unit, robustly normalized
    work_density = remaining_work / (duration + eps_safe)
    norm_work_density = adaptive_normalize(work_density)

    # Smooth sigmoid slack gating (centered at zero, slope controlled by fixed 2.0)
    slack_gate = 1.0 / (1.0 + np.exp(-2.0 * norm_slack))  # Sigmoid: 0→1 transition at slack=0
    slack_penalty_term = PARAMS["slack_risk_penalty"] * (1.0 - slack_gate) * norm_slack
    slack_gain_term = PARAMS["slack_benefit_gain"] * slack_gate * norm_slack

    # Uncertainty-coupled slack modulation: amplifies both penalty and gain where uncertainty is high
    coupled_slack = (slack_penalty_term + slack_gain_term) * (
        1.0 + PARAMS["uncertainty_slack_coupling"] * np.clip(norm_uncert, 0.0, 2.0)
    )

    # Critical path term: higher upward_rank → higher priority → subtract (since smaller score wins)
    critical_term = -PARAMS["critical_path_weight"] * norm_rank

    # Work density efficiency term: prefer high-compute-per-time tasks
    work_density_term = -PARAMS["work_density_weight"] * norm_work_density

    # Energy minimization: lower incremental energy → higher priority (add scaled norm_energy)
    energy_term = PARAMS["energy_scale"] * norm_energy

    # Starvation relief via saturating log1p (from Parent 1), prevents unbounded wait benefit
    starvation_term = -PARAMS["wait_starvation_relief"] * np.log1p(norm_wait * 0.5) / (0.5 + eps_safe)

    # Duration flexibility bias: quadratic penalty around |norm_duration| = 1.0 (balanced duration)
    flex_penalty = PARAMS["duration_flexibility_bias"] * ((np.abs(norm_duration) - 1.0) ** 2)

    # Bounded latency-energy tradeoff: penalize high duration * high energy coupling
    # Only active when both are above median-normalized threshold → avoids over-penalizing small tasks
    dur_energy_coupling = np.where(
        (norm_duration > 0.0) & (norm_energy > 0.0),
        norm_duration * norm_energy * 0.5,
        0.0
    )

    # Assemble final priority score: smaller = higher priority
    score = (
        coupled_slack
        + critical_term
        + work_density_term
        + energy_term
        + starvation_term
        + flex_penalty
        + dur_energy_coupling
    )

    # Final numerical safeguard: ensure finite, replace NaN/inf with safe bounds
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
