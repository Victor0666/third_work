def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric literals are in {-2,-1,0,1,2}; all tunables declared.
    
    Key features:
      - Robust median-root normalization with tunable power for energy
      - Sigmoid + linear penalty for slack to sharply prioritize overdue tasks
      - Critical-path/urgency coupling only activates under negative slack
      - Uncertainty-duration interaction amplifies urgency for long+uncertain ops
      - Sublinear wait decay prevents starvation without dominance
      - All numeric literals strictly limited to -2, -1, 0, 1, 2
    """
    eps = np.finfo(float).tiny + PARAMS["epsilon"]
    
    # Robust median-root normalization
    def robust_normalize(x, power=1.0):
        x = np.asarray(x, dtype=float)
        med = np.median(np.abs(x))
        scale = np.power(np.maximum(med, eps), power)
        return x / (scale + eps)

    norm_energy = robust_normalize(min_incremental_energy, PARAMS["energy_norm_power"])
    norm_duration = robust_normalize(min_exec_time + min_comm_time, 1.0)
    norm_rank = robust_normalize(upward_rank, 1.0)
    norm_work = robust_normalize(remaining_work, 1.0)
    norm_wait = robust_normalize(ready_wait_time, 1.0)
    norm_uncert = robust_normalize(uncertainty, 1.0)

    # Slack scoring: smooth sigmoid near zero + strong linear penalty for negative slack
    sigmoid_slack = 1.0 / (1.0 + np.exp(-PARAMS["slack_sigmoid_k"] * slack))
    slack_penalty = PARAMS["slack_penalty_scale"] * np.maximum(-slack, 0.0)
    slack_score = (1 - sigmoid_slack) * 2 + slack_penalty * PARAMS["slack_linear_coeff"]

    # Critical-path + urgency coupling: only active when slack < 0
    rank_urgency = norm_rank * np.maximum(-slack, 0.0) * PARAMS["rank_slack_coupling"]

    # Uncertainty-duration interaction
    dur_uncert_interaction = norm_duration * norm_uncert * PARAMS["duration_risk_interaction"]

    # Anti-starvation: sublinear power decay on wait time
    wait_effect = np.power(np.maximum(norm_wait, eps), PARAMS["wait_decay"])

    # Final score: smaller = higher priority
    score = (
        slack_score
        + rank_urgency
        + norm_energy
        + dur_uncert_interaction
        + wait_effect
        - norm_work
    )

    # Ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=1e6, neginf=-1e6)
    return score
