def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces scalar rank amplification with critical-path leverage term
    (upward_rank × remaining_work), removes redundant feasibility_guard, introduces conditional
    energy sensitivity (only active when slack >= 0), replaces arctan starvation relief with
    bounded sigmoid, and uses mean-abs scaling for stability over median/MAD (lower variance,
    faster convergence).
    
    Key structural changes:
      - Criticality now modeled as upward_rank * remaining_work interaction — captures both
        path importance and computational load impact on critical path.
      - Energy contribution is gated: only penalized when slack >= 0 (i.e., no DDL risk),
        avoiding conflict between urgency and efficiency.
      - Starvation relief uses smooth sigmoid: 1 / (1 + exp(-x)) scaled to [0,1], avoiding
        asymptotic saturation artifacts of arctan.
      - All feature normalizations use mean-abs (robust to outliers, avoids median cost).
      - No hard feasibility guard: counterfactual evidence shows it's inactive; replaced by
        tighter joint gating on slack/uncertainty/work.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Robust per-feature normalization using mean-abs deviation (faster & more stable than median/MAD)
    def mean_abs_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.mean(x)
        spread = np.mean(np.abs(x - center)) + eps
        return (x - center) / (spread + eps)

    norm_slack = mean_abs_normalize(slack)
    norm_energy = mean_abs_normalize(min_incremental_energy)
    norm_duration = mean_abs_normalize(min_exec_time + min_comm_time)
    norm_rank = mean_abs_normalize(upward_rank)
    norm_work = mean_abs_normalize(remaining_work)
    norm_wait = mean_abs_normalize(ready_wait_time)
    norm_uncert = mean_abs_normalize(uncertainty)

    # --- Slack risk score: bounded sigmoid gating for monotonic DDL pressure
    # Uses soft step: 1 / (1 + exp(-k * slack)) → high priority when slack << 0
    # Avoids piecewise and exponentials near zero; stable and differentiable
    slack_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_penalty_exponent"] * norm_slack))
    slack_risk_score = slack_gate * (1.0 - slack_gate) * PARAMS["slack_penalty_exponent"]

    # --- Critical-path leverage: upward_rank × remaining_work interaction
    # Activated only under severe slack pressure (joint condition)
    critical_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    critical_leverage = (norm_rank * norm_work) * critical_gate * PARAMS["criticality_scale"]

    # --- Duration-uncertainty coupling: only when slack is tight AND uncertainty is high
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) & 
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]), 
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Bounded starvation relief: smooth sigmoid (more numerically stable than arctan)
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_benefit = 1.0 / (1.0 + np.exp(-wait_scaled))  # maps [0,∞) → [0.5,1)

    # --- Conditional energy sensitivity: only penalize energy when slack >= 0
    # Prevents conflicting objectives: when deadline risk dominates, energy is secondary
    energy_gate = np.where(slack >= 0.0, 1.0, 0.0)
    energy_contribution = energy_gate * PARAMS["energy_sensitivity"] * norm_energy

    # --- Work-based suppression: only when work exceeds threshold AND slack is non-negative
    # Avoids suppressing high-work tasks that are already urgent (negative slack)
    work_suppression_gate = np.where(
        (remaining_work > PARAMS["work_leverage_threshold"]) & (slack >= 0.0),
        1.0, 0.0
    )
    work_ratio = np.clip(remaining_work / (PARAMS["work_leverage_threshold"] + eps), 
                         np.finfo(float).tiny, 
                         np.finfo(float).max)
    work_suppression = work_suppression_gate * (1.0 / (1.0 + work_ratio))

    # --- Final score: smaller = higher priority
    # DDL risk (slack_gate) dominates; critical leverage boosts blocking tasks;
    # energy only considered in safe region; starvation relief subtracted (improves priority)
    score = (
        slack_risk_score                              # Primary DDL risk signal (monotonic, bounded)
        + critical_leverage                           # Critical-path work leverage under stress
        + dur_uncert_interaction                      # Uncertainty-aware duration penalty
        + energy_contribution                         # Energy minimization — conditional & secondary
        - wait_benefit                                # Anti-starvation (subtracted → improves priority)
        + work_suppression                            # Suppress large non-urgent work
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
