def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL risk containment, critical-path fidelity,
    and starvation-aware fairness — all via bounded, normalized, piecewise interactions.
    
    Key innovations:
      - Uses *nonlinear slack penalty* (pow) for strong urgency response to negative slack.
      - Introduces *criticality boost gate*: upward_rank scaled only when slack is critically low OR uncertainty high.
      - *Duration-uncertainty coupling*: adds uncertainty-weighted duration to penalize risky long tasks.
      - *Starvation relief* activates linearly only after wait_decay_threshold, avoiding over-correction.
      - All features normalized robustly using mean absolute value + epsilon; no std/mean on possibly degenerate arrays.
      - No unbounded ops: pow uses clipped exponents; divisions use np.finfo guards; gates are bounded sigmoid-like.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Robust normalization: avoid nan/inf by guarding against zero-mean-abs and preserving shape
    def safe_normalize(x):
        x = np.asarray(x, dtype=float)
        abs_x = np.abs(x)
        mean_abs = np.mean(abs_x)
        scale = np.where(mean_abs == 0.0, eps, mean_abs)
        return x / (scale + eps)
    
    # Normalize base features
    norm_slack = safe_normalize(slack)
    norm_energy = safe_normalize(min_incremental_energy)
    norm_rank = safe_normalize(upward_rank)
    norm_work = safe_normalize(remaining_work)
    norm_wait = safe_normalize(ready_wait_time)
    norm_uncert = safe_normalize(uncertainty)
    norm_duration = safe_normalize(min_exec_time + min_comm_time)
    
    # === SLACK RISK PENALTY (nonlinear & signed) ===
    # Strong penalty for negative slack: (max(0, -slack))^exponent → higher priority (lower score) for urgent tasks
    raw_slack_penalty = np.power(np.maximum(0.0, -norm_slack), PARAMS["slack_penalty_exponent"])
    
    # === CRITICALITY BOOST GATE ===
    # Activate only if slack is critically tight (normalized < gate) OR uncertainty is high (>0.5 normalized)
    slack_critical = (norm_slack < PARAMS["risk_slack_gate"])
    uncert_high = (norm_uncert > 0.5)
    critical_gate = np.where(slack_critical | uncert_high, 1.0, 0.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_boost"] * critical_gate)
    
    # === DURATION-UNCERTAINTY COUPLING ===
    # Penalize tasks whose combined duration is both long *and* uncertain
    coupled_duration = norm_duration + PARAMS["duration_uncertainty_coupling"] * norm_duration * norm_uncert
    
    # === STARVATION RELIEF (piecewise linear anti-starvation) ===
    # Only applies beyond threshold; avoids boosting short-wait tasks
    wait_excess = np.maximum(0.0, ready_wait_time - PARAMS["wait_decay_threshold"])
    starvation_relief = PARAMS["starvation_relief_slope"] * wait_excess
    # Normalize relief to same scale as other terms (prevents dominance)
    norm_relief = safe_normalize(starvation_relief)
    
    # === COMPOSITE SCORE: smaller = higher priority ===
    # Components ordered by lexicographic importance: DDL risk > critical path > energy > fairness
    score = (
        # Primary: strict deadline risk containment
        raw_slack_penalty
        # Secondary: critical-path fidelity (boosted only under risk)
        + boosted_rank
        # Tertiary: energy efficiency (lower energy → lower score)
        - PARAMS["energy_efficiency_sensitivity"] * norm_energy
        # Quaternary: penalize risky long durations
        + coupled_duration
        # Quinary: anti-starvation — *reduces* score (i.e., improves priority) for long waits
        - norm_relief
        # Bonus: mild preference for less work (reduces downstream load) — bounded & weak
        + 0.1 * norm_work
    )
    
    # Final safeguard: clip inf/nan, ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
