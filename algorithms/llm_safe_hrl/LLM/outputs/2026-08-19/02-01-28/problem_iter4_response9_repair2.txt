def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces rank-only amplification with upward_rank × remaining_work interaction,
    removes feasibility_guard_weight (redundant per counterfactual evidence), replaces arctan wait with bounded sigmoid,
    and enforces strict DDL-first ordering via sigmoid-gated slack risk — smoother, more monotonic, less fragile.
    
    Key structural changes:
      - Replaces isolated `amplified_rank` with critical-path leverage term: `upward_rank * remaining_work`
      - Uses bounded sigmoid on normalized slack for monotonic DDL pressure instead of piecewise power/relief
      - Removes `feasibility_guard_weight` parameter and logic (evidence shows it's inactive and boundary-contacted)
      - Replaces arctan starvation relief with saturating sigmoid: avoids plateau artifacts and improves gradient flow
      - Adds joint slack-uncertainty gating for energy sensitivity to prevent over-penalizing uncertain tasks under deadline safety
      - All normalizations use mean-abs scaling (more stable than MAD for small N) with machine-epsilon safeguards
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Robust per-feature normalization using mean absolute deviation (stable for small N)
    def mean_abs_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.mean(x)
        spread = np.mean(np.abs(x - center)) + eps
        return (x - center) / (spread + eps)

    norm_slack = mean_abs_normalize(slack)
    norm_energy = mean_abs_normalize(min_incremental_energy)
    norm_duration = mean_abs_normalize(min_exec_time + min_comm_time)
    norm_rank = mean_abs_normalize(upward_rank)
    norm_work = mean_abs_normalize(remaining_work)
    norm_wait = mean_abs_normalize(ready_wait_time)
    norm_uncert = mean_abs_normalize(uncertainty)

    # --- Smooth, monotonic DDL pressure: bounded sigmoid on normalized slack
    # Maps large negative slack → near 1.0; positive slack → near 0.0; no discontinuity
    slack_pressure = 1.0 / (1.0 + np.exp(-norm_slack * PARAMS["slack_penalty_exponent"]))

    # --- Critical path leverage: product of rank and work, gated by severe slack
    # Captures both importance and computational weight — only active when truly urgent
    critical_leverage_gate = np.where(norm_slack < PARAMS["slack_negative_threshold"], 1.0, 0.0)
    critical_leverage = (norm_rank * norm_work) * critical_leverage_gate * PARAMS["criticality_scale"]

    # --- Duration-uncertainty coupling: only under joint tightness (tight slack AND high uncertainty)
    dur_uncert_gate = np.where(
        (norm_slack < PARAMS["slack_positive_gate"]) & 
        (norm_uncert > PARAMS["uncertainty_activation_threshold"]), 
        1.0, 0.0
    )
    dur_uncert_interaction = norm_duration * norm_uncert * dur_uncert_gate * PARAMS["duration_robustness"]

    # --- Bounded starvation relief: smooth sigmoid (replaces arctan for better saturation control)
    wait_ratio = np.clip(ready_wait_time / (PARAMS["wait_saturation_time"] + eps), 0.0, PARAMS["wait_upper_bound"])
    wait_benefit = 1.0 / (1.0 + np.exp(-wait_ratio + 2.0))  # Shifted to start benefit earlier

    # --- Energy sensitivity gating: suppress energy penalty when slack is safe
    # Prevents unnecessary energy minimization at the cost of deadline margin
    energy_gate = np.where(norm_slack > PARAMS["slack_positive_gate"], 0.0, 1.0)
    energy_term = energy_gate * PARAMS["energy_sensitivity"] * norm_energy

    # --- Final score: smaller = better; DDL pressure dominates, then leverage, then energy
    score = (
        slack_pressure                              # Primary hard-DDL signal (monotonic, bounded)
        - critical_leverage                         # Boost tasks blocking large critical work
        + dur_uncert_interaction                    # Penalize uncertain durations only when risky
        + energy_term                               # Energy minimization only under deadline pressure
        - wait_benefit                              # Anti-starvation (smooth, early-activating)
    )

    # Ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
