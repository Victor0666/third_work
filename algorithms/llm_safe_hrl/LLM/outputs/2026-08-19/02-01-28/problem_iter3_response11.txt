def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: unified slack-normalized gating + wait-time sensitivity + deadline-aware energy discounting.
    
    Key improvements:
      - Replaces physical slack threshold with *normalized* critical_slack_threshold_normalized (relative to |slack| distribution),
        enabling robustness across heterogeneous workflow deadlines.
      - Removes redundant uncertainty gating: criticality now triggered *only* by normalized slack — aligning with reflection's simplification goal.
      - Introduces slack_energy_discount_factor: smoothly attenuates energy-efficiency benefit as slack approaches zero (not binary gating),
        enabling graceful trade-off instead of hard cutoff.
      - Replaces starvation relief with direct wait_time_sensitivity on *normalized* ready_wait_time — simpler, bounded, and comparably effective.
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Safe normalization: handles zero-mean-abs by falling back to epsilon-scaled identity
    def safe_normalize(x):
        x = np.asarray(x, dtype=float)
        abs_x = np.abs(x)
        mean_abs = np.mean(abs_x)
        scale = np.where(mean_abs == 0.0, eps, mean_abs)
        return x / (scale + eps)
    
    # Normalize all features using robust mean-abs scaling
    norm_slack = safe_normalize(slack)
    norm_energy = safe_normalize(min_incremental_energy)
    norm_rank = safe_normalize(upward_rank)
    norm_work = safe_normalize(remaining_work)
    norm_wait = safe_normalize(ready_wait_time)
    norm_uncert = safe_normalize(uncertainty)
    norm_duration = safe_normalize(min_exec_time + min_comm_time)
    
    # === SLACK RISK PENALTY (nonlinear & signed) ===
    raw_slack_penalty = np.power(np.maximum(0.0, -slack), PARAMS["slack_penalty_exponent"])
    
    # === CRITICALITY BOOST (single normalized slack gate) ===
    # Compute slack safety margin: threshold relative to observed slack spread
    slack_abs_mean = np.mean(np.abs(slack))
    slack_norm_threshold = PARAMS["critical_slack_threshold_normalized"] * np.where(slack_abs_mean == 0.0, eps, slack_abs_mean)
    slack_critical = (slack < slack_norm_threshold)
    critical_gate = slack_critical.astype(float)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_boost"] * critical_gate)
    
    # === DEADLINE-AWARE ENERGY DISCOUNTING (smooth, not binary) ===
    # Discount energy benefit linearly from 1.0 (full) to slack_energy_discount_factor (min) as slack approaches threshold
    slack_margin = np.clip((slack - slack_norm_threshold) / (np.abs(slack_norm_threshold) + eps), 0.0, 1.0)
    energy_discount = PARAMS["slack_energy_discount_factor"] + (1.0 - PARAMS["slack_energy_discount_factor"]) * slack_margin
    energy_contribution = -PARAMS["energy_efficiency_sensitivity"] * norm_energy * energy_discount
    
    # === DURATION-UNCERTAINTY COUPLING ===
    coupled_duration = norm_duration * (1.0 + PARAMS["duration_uncertainty_coupling"] * norm_uncert)
    
    # === STARVATION MITIGATION (simplified: direct normalized wait penalty) ===
    # Penalize long waits proportionally — no piecewise thresholds, fully differentiable & bounded
    wait_penalty = PARAMS["wait_time_sensitivity"] * norm_wait
    
    # === COMPOSITE SCORE: smaller = higher priority ===
    # All numeric literals are strictly -2,-1,0,1,2
    score = (
        raw_slack_penalty
        + boosted_rank
        + energy_contribution
        + 1 * PARAMS["upward_rank_weight"] * norm_rank * (1.0 - critical_gate)  # base rank outside critical mode
        + coupled_duration
        + wait_penalty
        + 1 * PARAMS["remaining_work_weight"] * norm_work
    )
    
    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
