def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: DDL-risk-gated criticality + starvation-aware energy efficiency.
    
    Key innovations:
      - Slack is split into *risk penalty* (for slack < 0) and *urgency gain* (for slack >= 0)
      - Criticality (upward_rank) only activates under DDL-safe conditions (slack >= 0)
      - Robust IQR-based normalization avoids mean/std sensitivity to outliers
      - Uncertainty-slack coupling amplifies urgency when both deadline pressure AND resource fuzziness are high
      - Wait time uses soft clipping (tanh) with tunable scale to prevent dominance
      - All operations use epsilon-guarded, finite-safe arithmetic
      - Final score clipped to bounded range using declared parameters
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Ensure all inputs are float arrays, no in-place modification
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # --- Robust normalization via interquartile range (IQR) with epsilon guard ---
    def iqr_normalize(x):
        x = np.clip(x, -1e12, 1e12)  # Prevent overflow before stats
        q75, q25 = np.percentile(x, [75, 25], method='midpoint')
        iqr = q75 - q25
        center = np.median(x)
        denom = np.maximum(iqr, eps)
        return (x - center) / denom
    
    # Normalize base features (finite & bounded)
    norm_energy = iqr_normalize(min_incremental_energy)
    norm_duration = iqr_normalize(min_exec_time + min_comm_time)
    norm_uprank = iqr_normalize(upward_rank)
    norm_work = iqr_normalize(remaining_work)
    norm_wait = iqr_normalize(ready_wait_time)
    norm_uncert = iqr_normalize(uncertainty)
    norm_slack = iqr_normalize(slack)
    
    # --- Slack gating: separate risk (negative) and urgency (positive) regimes ---
    # Negative slack → hard penalty (urgent execution required)
    slack_risk_mask = (slack < 0).astype(float)
    slack_risk_score = slack_risk_mask * PARAMS["slack_risk_penalty"] * (-norm_slack)
    
    # Positive slack → gentle urgency gain (avoid over-scheduling too-early tasks)
    slack_safe_mask = (slack >= 0).astype(float)
    slack_urgency_score = slack_safe_mask * PARAMS["slack_urgency_gain"] * norm_slack
    
    # --- Criticality only matters when DDL-safe (slack >= 0) ---
    criticality_score = slack_safe_mask * PARAMS["criticality_weight"] * norm_uprank
    
    # --- Uncertainty-slack coupling: high uncertainty + low slack → amplified urgency ---
    # Use clipped product: max(0, -slack) * uncertainty to avoid NaNs and infinite growth
    slack_pressure = np.clip(-slack, 0.0, np.finfo(float).max)
    coupled_risk = (slack_pressure / (np.abs(np.mean(slack)) + eps + 1e-3)) * uncertainty
    coupled_risk_norm = iqr_normalize(coupled_risk)
    coupling_score = PARAMS["uncertainty_slack_coupling"] * coupled_risk_norm
    
    # --- Starvation relief: soft-clipped wait boost (prevents runaway priority) ---
    # Apply tanh with tunable scale to bound influence of extreme wait times
    wait_boost = np.tanh(PARAMS["wait_starvation_relief"] * PARAMS["wait_clipping_scale"] * norm_wait)
    
    # --- Energy efficiency only activated when DDL risk is low (slack >= 0) ---
    # Otherwise, energy is secondary to meeting deadline
    energy_score = slack_safe_mask * PARAMS["energy_efficiency_weight"] * norm_energy
    
    # --- Duration balance term: encourages shorter-duration tasks when DDL-safe ---
    duration_score = slack_safe_mask * PARAMS["duration_balance_factor"] * norm_duration
    
    # --- Final score: smaller = higher priority ---
    # All components are additive; negative contributions lower score (increase priority)
    score = (
        slack_risk_score           # ↑ penalty → ↑ score → ↓ priority
        - slack_urgency_score      # ↓ urgency → ↓ score → ↑ priority (gentle pull for early tasks)
        - criticality_score        # ↓ criticality score → ↑ priority (but only if safe)
        + energy_score             # ↑ energy → ↑ score → ↓ priority (only when safe)
        + duration_score           # ↑ duration → ↑ score → ↓ priority (only when safe)
        - wait_boost               # ↓ wait_boost → ↓ score → ↑ priority (anti-starvation)
        + coupling_score           # ↑ coupling → ↑ score → ↓ priority (risk amplification)
    )
    
    # Final safeguard: replace inf/nan, clip to declared finite bounds
    score = np.nan_to_num(score, nan=0.0, posinf=0.0, neginf=0.0)
    score = np.clip(score, PARAMS["score_clip_lower"], PARAMS["score_clip_upper"])
    
    # Ensure shape is (N,) — critical for single-task case
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    
    return score
