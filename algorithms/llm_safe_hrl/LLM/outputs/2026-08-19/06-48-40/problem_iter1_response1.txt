def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing deadline risk containment, critical-path safety,
    and robust anti-starvation — all with bounded, monotonic, and numerically stable interactions.
    
    Key innovations:
      - Uses signed slack power law: sharp penalty for negative slack, smooth ramp for positive slack
      - Criticality boost gated by both slack pressure and uncertainty (not linear)
      - Wait-time boost uses exponential decay (safe via np.clip) to avoid unbounded growth
      - Energy and duration terms normalized using IQR-like range scaling for outlier resilience
      - All operations protected against NaN/inf/zero via epsilon and np.nan_to_num
      - No unbounded functions: only powers in [0.5, 2.0], clamped exponentials, and piecewise-linear gates
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Safe cast & copy
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust normalization: use interquartile-range-inspired scaling (more outlier-resistant than mean/std)
    def robust_normalize(x):
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        q75 = np.percentile(np.abs(x), 75)
        q25 = np.percentile(np.abs(x), 25)
        iqr = q75 - q25 + eps
        center = np.median(x)
        return (x - center) / iqr

    # --- Slack handling: strong nonlinearity for risk containment ---
    # Signed power: (x)^p for x>0, -(abs(x))^q for x<0 → preserves sign and amplifies lateness
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_power = np.where(
        slack >= 0,
        np.power(np.maximum(slack, eps), PARAMS["slack_sensitivity"]),
        -np.power(np.maximum(slack_abs, eps), PARAMS["slack_sensitivity"])
    )
    # Apply penalty scale only to negative slack (lateness), keep positive slack soft
    slack_score = np.where(
        slack < 0,
        PARAMS["slack_penalty_scale"] * slack_power,
        slack_power
    )

    # --- Criticality gating: boost upward_rank only under pressure ---
    # Gate = sigmoid( ( -slack + uncertainty ) * coupling ) → active when slack low OR uncertainty high
    pressure_index = -slack + uncertainty
    gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_slack_coupling"] * (pressure_index + eps)))
    criticality_score = PARAMS["criticality_boost"] * upward_rank * gate

    # --- Energy term: normalized marginal fuzzy energy (already risk-adjusted) ---
    energy_norm = robust_normalize(min_incremental_energy)
    energy_score = PARAMS["energy_normalizer"] * energy_norm

    # --- Duration risk: uncertainty-weighted exec+comm, robustly normalized ---
    duration = min_exec_time + min_comm_time
    duration_risk = duration * (1.0 + uncertainty)  # amplify uncertain durations
    duration_norm = robust_normalize(duration_risk)
    duration_score = PARAMS["duration_risk_weight"] * duration_norm

    # --- Anti-starvation: clipped exponential wait boost (avoids overflow) ---
    # exp(-wait_decay * t) decays; we want *higher* priority for longer waits → use (1 - decay)
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay"] * np.maximum(ready_wait_time, 0.0))
    wait_boost = np.clip(wait_boost, 0.0, 0.999)  # prevent exact 1.0 saturation
    # Normalize wait_boost itself to align scale with other terms
    wait_norm = robust_normalize(wait_boost)
    wait_score = -wait_norm  # negative because higher wait → higher priority → lower score

    # --- Composite score: sum of prioritized components ---
    # All terms designed so smaller final score = higher priority
    score = (
        slack_score               # dominant: negative slack penalized sharply
        + energy_score            # secondary: lower marginal energy preferred
        + duration_score          # duration risk adds cost
        - criticality_score       # critical tasks get *negative* contribution → lower score
        + wait_score              # anti-starvation: longer wait → lower score
    )

    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=1e6, neginf=-1e6)
    
    # Enforce shape (N,) explicitly
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    
    return score
