def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing deadline risk isolation, coupled uncertainty-pressure,
       and fairness-constrained efficiency tradeoff — all with bounded, numerically stable forms."""
    eps = PARAMS["epsilon"]
    
    # --- Robust normalization: use median absolute deviation + epsilon for outlier resilience
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        spread = np.median(np.abs(x - center)) + eps
        return (x - center) / spread

    # --- Slack processing: isolate hard DDL violation (negative slack) with smooth, bounded penalty
    # Use signed power: negative slack → large positive penalty (bad), positive slack → mild decay
    slack_norm = robust_normalize(slack)
    # Soft sign: preserves direction but compresses magnitude via arctan-like shape (bounded derivative)
    slack_sign = np.tanh(slack_norm)
    # Stronger penalty for negative slack using parameterized exponent and factor
    slack_urgency = np.where(
        slack_norm < 0,
        PARAMS["slack_penalty_factor"] * (-slack_norm) ** PARAMS["slack_sensitivity"],
        slack_norm * 0.3  # gentle reward for slack margin
    )

    # --- Energy-duration efficiency: combine min_exec_time + min_comm_time and min_incremental_energy
    # into a single resource-efficiency ratio, avoiding direct addition across units
    duration = min_exec_time + min_comm_time
    # Efficiency score: joules per second → higher = worse; clamp denominator to avoid blowup
    eff_ratio = np.divide(
        min_incremental_energy,
        duration + eps,
        out=np.full_like(min_incremental_energy, np.inf, dtype=float),
        where=(duration + eps) != 0
    )
    # Normalize the efficiency ratio — lower ratio = better energy/time tradeoff
    eff_score = robust_normalize(eff_ratio)

    # --- Critical path leverage: boost upward_rank *only* when duration is below median
    # (i.e., short critical tasks get extra priority — avoids starving fast-but-important work)
    duration_med = np.median(duration) + eps
    short_task_gate = np.clip(duration / duration_med, 0.0, 1.0)  # [0,1] gate
    rank_boosted = upward_rank * (1.0 + PARAMS["critical_path_leverage"] * (1.0 - short_task_gate))

    # --- Uncertainty-slack coupling: penalize high uncertainty *and* low slack jointly
    # Bounded product: prevents explosion while preserving interaction signal
    unc_norm = robust_normalize(uncertainty)
    slack_pressure = np.maximum(-slack_norm, 0.0)  # only pressure from tight/negative slack
    coupled_risk = PARAMS["uncertainty_slack_coupling"] * unc_norm * slack_pressure

    # --- Fairness: bounded linear anti-starvation term — no diminishing returns, but capped at 1.0
    wait_norm = robust_normalize(ready_wait_time)
    # Cap contribution to avoid dominance: map [-2,2] → [0,1] via sigmoid-like piecewise
    wait_fairness = np.clip(PARAMS["wait_fairness_gain"] * wait_norm, -1.0, 1.0)

    # --- Composite score: smaller = higher priority
    # Deadline risk dominates (urgency term is largest contributor)
    # Then efficiency (energy/duration), then criticality, then fairness
    score = (
        slack_urgency
        + PARAMS["energy_efficiency_bias"] * eff_score
        - robust_normalize(rank_boosted)  # negative: higher rank → lower score
        + coupled_risk
        - wait_fairness  # negative fairness term: longer wait → lower score
    )

    # Final safeguard: ensure finite, deterministic output
    return np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
