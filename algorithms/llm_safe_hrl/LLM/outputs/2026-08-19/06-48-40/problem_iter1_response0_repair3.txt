def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric thresholds moved into PARAMETER_SCHEMA.
    Uses robust percentile-based gating, smooth risk penalties, and bounded interactions.
    Smaller score = higher priority.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization using median & IQR
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        q75, q25 = np.percentile(x, [75, 25], method='midpoint')
        iqr = q75 - q25
        scale = iqr if iqr > eps else (np.abs(med) + eps)
        return (x - med) / (scale + eps)

    # 1. Slack risk: smooth ramp for negative slack
    slack_risk = np.where(slack < 0, np.power(-slack, PARAMS["slack_sensitivity"]), 0.0)
    slack_score = PARAMS["slack_penalty_scale"] * slack_risk

    # 2. Uncertainty-modulated slack pressure
    norm_slack = robust_normalize(slack)
    norm_uncert = robust_normalize(uncertainty)
    coupling_factor = np.clip(norm_slack * norm_uncert, 0.0, 1.0)
    coupled_risk = slack_score * (1.0 + PARAMS["uncertainty_slack_coupling"] * coupling_factor)

    # 3. Criticality boost: activate only under high lateness risk OR high uncertainty
    q_thresh = np.percentile(uncertainty, PARAMS["uncertainty_percentile_threshold"], method='midpoint')
    high_risk_mask = (slack < 0) | (uncertainty > q_thresh + eps)
    boosted_rank = upward_rank * np.where(high_risk_mask, PARAMS["criticality_boost"], 1.0)
    rank_score = -robust_normalize(boosted_rank)

    # 4. Risk-adjusted energy
    energy_score = PARAMS["energy_normalizer"] * robust_normalize(min_incremental_energy)

    # 5. Duration risk: execution + communication, scaled by uncertainty
    duration_base = min_exec_time + min_comm_time
    duration_risk = duration_base * (1.0 + uncertainty * PARAMS["uncertainty_slack_coupling"])
    duration_score = PARAMS["duration_risk_weight"] * robust_normalize(duration_risk)

    # 6. Anti-starvation: inverse exponential wait boost
    clamped_wait = np.clip(ready_wait_time, 0.0, PARAMS["max_wait_clamp"])
    wait_boost = np.exp(-PARAMS["wait_decay"] * clamped_wait)
    wait_score = -wait_boost

    # 7. Remaining work penalty (mild, conditional)
    work_score = -0.3 * robust_normalize(remaining_work) * (1.0 - np.clip(norm_slack, 0.0, 1.0))

    # Aggregate
    score = (
        coupled_risk
        + energy_score
        + duration_score
        + rank_score
        + wait_score
        + work_score
    )

    # Final safeguard
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
