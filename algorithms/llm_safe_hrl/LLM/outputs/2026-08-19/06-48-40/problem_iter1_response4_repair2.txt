def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: Slack-driven criticality gating + uncertainty-coupled energy efficiency.
    
    Key innovations:
    - Uses piecewise slack gating: negative slack triggers strong penalty + criticality boost;
      non-negative slack enables energy/duration optimization.
    - Introduces exponential wait credit (not linear) to avoid starvation while bounding growth.
    - Combines uncertainty and slack via bounded product, not max/min, for smooth risk coupling.
    - Normalizes each feature robustly using IQR-based scaling + epsilon guard.
    - Avoids sign flips on upward_rank/work: higher values always indicate higher structural importance.
    """
    epsilon = PARAMS["epsilon"]
    N = len(slack)
    
    # Robust normalization: use interquartile range + mean absolute deviation for stability
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q75, q25 = np.percentile(x, [75, 25], method='midpoint')
        iqr = q75 - q25
        mad = np.median(np.abs(x - np.median(x)))
        scale = np.maximum(iqr, mad) + epsilon
        return (x - np.median(x)) / scale
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # Slack-based gating: critical regime (slack <= 0) vs. efficiency regime (slack > 0)
    is_critical = (slack <= 0).astype(float)
    is_efficient = 1.0 - is_critical
    
    # Critical regime: prioritize by slack penalty + upward_rank amplification
    # Penalty grows quadratically for more negative slack, bounded by epsilon
    slack_penalty = PARAMS["slack_penalty_factor"] * np.square(np.minimum(norm_slack, 0.0))
    
    # Amplify upward_rank only in critical regime, using soft-gated sigmoid-like boost
    rank_boost = PARAMS["criticality_boost"] * norm_rank * is_critical
    
    # Efficiency regime: favor low-energy, short-duration, high-work-balance tasks
    energy_efficiency = -PARAMS["energy_efficiency_gain"] * norm_energy * is_efficient
    duration_favor = -PARAMS["duration_sensitivity"] * norm_duration * is_efficient
    work_balance = -PARAMS["work_balance_factor"] * norm_work * is_efficient
    
    # Anti-starvation: exponential wait credit (bounded, monotonic, no overflow)
    # Clip ready_wait_time to avoid exp overflow; use scaled log for numerical safety
    wait_clip = np.clip(ready_wait_time, 0.0, PARAMS["wait_clip_threshold"])
    wait_credit = np.exp(-PARAMS["wait_decay"] * wait_clip) - 1.0  # negative → higher priority
    wait_credit = np.clip(wait_credit, -PARAMS["wait_credit_clip"], 0.0)
    
    # Uncertainty-slack coupling: stronger interaction when both are high-risk
    # Bounded product avoids explosion; uses normalized versions for consistent scale
    unc_slack_coupling = -PARAMS["uncertainty_slack_coupling"] * norm_uncert * np.abs(norm_slack)
    
    # Assemble score: smaller = better
    score = (
        slack_penalty
        + rank_boost
        + energy_efficiency
        + duration_favor
        + work_balance
        + wait_credit
        + unc_slack_coupling
    )
    
    # Final safeguard: ensure finite outputs, preserve shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score
