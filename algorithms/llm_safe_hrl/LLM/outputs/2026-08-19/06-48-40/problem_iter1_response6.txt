def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: prioritizes deadline-criticality first, then balances energy-efficiency, 
    structural importance, and starvation fairness via bounded, scale-invariant interactions.
    
    Key innovations:
      - Critical slack gate: exponential penalty only below `critical_slack_threshold`, avoiding 
        instability for large negative slack while strongly penalizing imminent violations.
      - Work-density ratio: (exec+comm)/remaining_work → low ratio = high computation density per MI;
        favors tasks that unlock large descendant work with small duration.
      - Upward rank normalized by dynamic range (not mean), making it robust to DAG skew.
      - Sublinear wait urgency: pow(ready_wait_time + epsilon, exponent) prevents dominance by long waits.
      - Uncertainty-slack coupling: multiplies uncertainty by sigmoid(-slack) to amplify risk awareness 
        only when slack is tight — avoids over-penalizing stable tasks.
      - All normalizations use median absolute scaling for robustness against outliers.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization: median-based, epsilon-guarded
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        med_abs = np.median(np.abs(x)) + eps
        return x / med_abs
    
    # --- 1. Deadline risk: critical slack gate with smooth exponential penalty ---
    # Only activate strong penalty when slack < threshold; avoid unbounded exp growth
    slack_deviation = np.clip(slack - PARAMS["critical_slack_threshold"], a_min=None, a_max=0.0)
    critical_slack_penalty = PARAMS["slack_penalty_factor"] * np.exp(
        np.clip(slack_deviation, a_min=-10.0, a_max=0.0)
    )
    
    # --- 2. Risk-adjusted energy: normalized marginal cost ---
    energy_score = PARAMS["energy_sensitivity"] * robust_normalize(min_incremental_energy)
    
    # --- 3. Duration efficiency: (exec + comm) / remaining_work → lower = better density ---
    # Avoid division by zero; cap denominator to prevent blowup
    denom = np.maximum(remaining_work, PARAMS["work_density_baseline"] * eps)
    duration_density = (min_exec_time + min_comm_time) / denom
    density_score = robust_normalize(duration_density)
    
    # --- 4. Structural importance: upward rank normalized by its own range (robust to outliers) ---
    urank = np.asarray(upward_rank, dtype=float)
    urank_range = np.ptp(urank) + eps
    normalized_rank = (urank - np.min(urank)) / urank_range
    # Invert: higher rank → higher priority → lower score
    rank_score = -PARAMS["rank_normalized_weight"] * normalized_rank
    
    # --- 5. Starvation mitigation: sublinear wait urgency ---
    wait_urgency = np.power(np.maximum(ready_wait_time, 0.0) + eps, PARAMS["wait_urgency_exponent"])
    wait_score = -robust_normalize(wait_urgency)  # negative → higher priority for longer waits
    
    # --- 6. Uncertainty-slack coupling: sigmoid(-slack) gates uncertainty impact ---
    # Sigmoid maps slack → [0,1], high weight when slack is negative/tight
    slack_sigmoid = 1.0 / (1.0 + np.exp(np.clip(-slack, a_min=-10.0, a_max=10.0)))
    coupled_uncertainty = PARAMS["uncertainty_slack_coupling"] * uncertainty * slack_sigmoid
    uncertainty_score = robust_normalize(coupled_uncertainty)
    
    # --- Assemble final priority score ---
    # Smaller score = higher priority
    score = (
        critical_slack_penalty
        + energy_score
        + density_score
        + rank_score
        + wait_score
        + uncertainty_score
    )
    
    # Final safeguard: ensure finite, deterministic output
    return np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
