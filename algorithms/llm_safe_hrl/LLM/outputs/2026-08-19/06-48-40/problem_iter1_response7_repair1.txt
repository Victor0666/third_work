def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL risk containment first, then energy efficiency,
    with bounded saturation, critical slack gating, and uncertainty-modulated robustness.
    
    Key innovations:
      - Critical slack gate: sharp but bounded penalty for slack <= threshold, using softplus
      - Energy-efficiency ratio: prioritizes low marginal energy per second (J/s), not raw energy
      - Upward rank compressed via power law to avoid dominance over DDL/energy terms
      - Ready wait time uses smooth saturation (tanh) to prevent starvation without over-promotion
      - Uncertainty couples *only* with negative slack via product gate — no amplification when slack > 0
      - Duration term gated by uncertainty to reduce reliance on fragile time estimates under high risk
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Defensive copy & cast
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Normalize features robustly using mean absolute value + epsilon
    def robust_normalize(x):
        x = np.abs(x)
        x_mean = np.mean(x) if N > 1 else x[0]
        denom = x_mean + eps
        return x / denom
    
    # 1. Critical slack penalty: strong, bounded, asymmetric
    # Softplus(-slack) approximates ReLU(-slack) but is differentiable and bounded below
    # Scaled by factor and threshold-shifted for sensitivity tuning
    slack_centered = slack - PARAMS["critical_slack_threshold"]
    critical_penalty = np.log1p(np.exp(-slack_centered / (eps + PARAMS["softplus_scale"])))  # softplus(-shifted_slack)
    slack_score = PARAMS["slack_penalty_factor"] * critical_penalty
    
    # 2. Energy efficiency: marginal energy per total duration (J/s), lower is better
    # Avoid division by zero: use max(duration, eps)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_sec = min_incremental_energy / duration
    energy_eff_score = robust_normalize(energy_per_sec) * PARAMS["energy_efficiency_ratio"]
    
    # 3. Upward rank: compress high values to prevent domination; higher rank → more critical
    # Apply power-law decay: rank^p grows slower than linear for p<1, but here p>1 compresses extremes
    # We want diminishing returns on very high rank → use rank^(1/p) instead; so invert exponent
    rank_exponent = 1.0 / PARAMS["rank_decay_power"]
    compressed_rank = np.power(np.maximum(upward_rank, eps), rank_exponent)
    rank_score = -robust_normalize(compressed_rank)  # negative: higher rank → higher priority
    
    # 4. Ready wait time: smooth saturation via tanh to avoid unbounded growth
    # tanh(wait / tau) ∈ [0,1), scaled and inverted for priority boost
    wait_sat = np.tanh(ready_wait_time / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -wait_sat  # smaller score = higher priority
    
    # 5. Uncertainty-slack coupling: only activates when slack is tight/negative
    # Gate: 1 if slack <= threshold, else decays smoothly → use sigmoid of (threshold - slack)
    slack_gap = PARAMS["critical_slack_threshold"] - slack
    coupling_gate = 1.0 / (1.0 + np.exp(-slack_gap / (eps + PARAMS["sigmoid_scale"])))
    coupled_uncertainty = coupling_gate * uncertainty
    unc_score = PARAMS["uncertainty_slack_coupling"] * robust_normalize(coupled_uncertainty)
    
    # 6. Duration term: only trusted when uncertainty is low → gated by (1 - normalized uncertainty)
    norm_unc = robust_normalize(uncertainty)
    duration_trust = np.maximum(0.0, 1.0 - norm_unc)
    duration_score = PARAMS["duration_sensitivity"] * robust_normalize(duration) * duration_trust
    
    # Assemble final score: all components contribute additively; smaller = better
    score = (
        slack_score +
        energy_eff_score +
        rank_score +
        wait_score +
        unc_score +
        duration_score
    )
    
    # Final safeguard: replace NaN/inf with finite values, preserve shape
    score = np.nan_to_num(score, nan=np.finfo(float).max, posinf=np.finfo(float).max, neginf=np.finfo(float).smallest_subnormal)
    
    return score
