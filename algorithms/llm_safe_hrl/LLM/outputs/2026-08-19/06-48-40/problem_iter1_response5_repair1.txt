def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL risk isolation, coupled duration-energy tradeoff,
    and conditional critical-path awareness. Uses bounded piecewise gates and robust L1-normalization.
    Smaller score = higher priority."""
    eps = PARAMS["epsilon"]
    
    # Robust L1-based normalization: avoids mean-of-abs sensitive to outliers
    def l1_normalize(x):
        x = np.asarray(x, dtype=float)
        norm = np.sum(np.abs(x)) + eps
        return x / norm
    
    # Compute base normalized features
    exec_comm = min_exec_time + min_comm_time
    norm_exec_comm = l1_normalize(exec_comm)
    norm_energy = l1_normalize(min_incremental_energy)
    norm_slack = l1_normalize(slack)
    norm_rank = l1_normalize(upward_rank)
    norm_work = l1_normalize(remaining_work)
    norm_wait = l1_normalize(ready_wait_time)
    norm_uncert = l1_normalize(uncertainty)
    
    # --- Critical slack risk gate: strong nonlinear penalty for negative slack ---
    # Use smooth clipped power: (max(0, -slack))^p → zero for slack >= 0, grows for negative slack
    slack_risk = np.power(np.maximum(0.0, -slack), PARAMS["slack_risk_exponent"])
    norm_slack_risk = l1_normalize(slack_risk)
    
    # --- Conditional rank activation: only reward high upward_rank when slack >= threshold ---
    # Avoids over-prioritizing critical tasks that are already doomed (negative slack)
    rank_active = (slack >= PARAMS["critical_slack_threshold"]).astype(float)
    effective_rank = rank_active * norm_rank
    
    # --- Uncertainty-slack coupling: uncertainty matters most when slack is tight or negative
    # Bounded sigmoid-like interaction: max(0, 1 + slack / |threshold|) scaled by uncertainty
    slack_pressure_ratio = np.clip(
        1.0 + slack / (np.abs(PARAMS["critical_slack_threshold"]) + eps),
        0.0, 2.0
    )
    coupled_uncert = norm_uncert * slack_pressure_ratio
    
    # --- Work density term: favors tasks with high remaining_work per estimated duration
    # Prevents starvation of large sub-DAGs; uses stable ratio with epsilon guard
    work_density = remaining_work / (exec_comm + eps)
    norm_work_density = l1_normalize(work_density)
    
    # --- Assemble priority score: lower = better
    # Primary drivers: deadline risk (penalty), then energy & duration efficiency, then fairness
    score = (
        # 1. Strong penalty for deadline risk — dominates when slack < 0
        PARAMS["slack_risk_exponent"] * norm_slack_risk
        
        # 2. Energy-efficiency incentive (smaller energy → smaller score)
        + PARAMS["energy_sensitivity"] * norm_energy
        
        # 3. Duration-efficiency incentive (shorter exec+comm → smaller score)
        + PARAMS["duration_coupling"] * norm_exec_comm
        
        # 4. Critical-path awareness (only active when slack permits)
        - PARAMS["rank_awareness"] * effective_rank
        
        # 5. Anti-starvation: longer wait → lower score (higher priority)
        - PARAMS["wait_fairness"] * norm_wait
        
        # 6. Uncertainty-modulated pressure: higher uncertainty + tighter slack → higher score (penalty)
        + PARAMS["uncertainty_slack_gate"] * coupled_uncert
        
        # 7. Work density bonus: high work-per-duration → lower score (prioritize throughput-critical nodes)
        - PARAMS["work_density_factor"] * norm_work_density
    )
    
    # Final safeguard: ensure finite, deterministic output using declared replacement
    nan_rep = PARAMS["nan_replacement"]
    return np.nan_to_num(score, nan=nan_rep, posinf=nan_rep, neginf=-nan_rep)
