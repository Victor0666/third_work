def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing deadline risk containment, critical-path-aware energy tradeoffs,
    and starvation-robust waiting-time compensation — all via bounded, normalized, and sign-aware interactions.
    
    Key innovations:
      - Uses a smooth, differentiable 'risk cliff' penalty: (max(0, -slack))^slacksensitivity
      - Criticality boost activates only under high lateness risk OR high uncertainty, avoiding over-prioritization
      - Energy term uses IQR-normalized score to suppress outlier sensitivity
      - Ready wait time modeled as exp(-wait_decay * wait) → inverse: higher wait → lower score (better priority)
      - Duration risk combines exec+comm with uncertainty gating: (exec+comm) * (1 + uncertainty * coupling)
      - All normalizations use median-IQR for robustness against skew and outliers
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization using median & IQR (more stable than mean/std for skewed distributions)
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        q75, q25 = np.percentile(x, [75, 25], method='midpoint')
        iqr = q75 - q25
        scale = iqr if iqr > eps else (np.abs(med) + eps)
        return (x - med) / (scale + eps)

    # --- Core components ---
    # 1. Slack risk: sharp penalty for negative slack; soft decay for positive slack
    # Use smooth ramp: (-slack)^p for negative, else 0 — avoids discontinuity at 0
    slack_risk = np.where(slack < 0, np.power(-slack, PARAMS["slack_sensitivity"]), 0.0)
    slack_score = PARAMS["slack_penalty_scale"] * slack_risk

    # 2. Uncertainty-modulated slack pressure: amplifies risk when both slack is low AND uncertainty high
    # Scaled to [0,1] then multiplied into slack_score to avoid explosion
    norm_slack = robust_normalize(slack)
    norm_uncert = robust_normalize(uncertainty)
    coupling_factor = np.clip(norm_slack * norm_uncert, 0.0, 1.0)
    coupled_risk = slack_score * (1.0 + PARAMS["uncertainty_slack_coupling"] * coupling_factor)

    # 3. Criticality boost: only activate when slack is tight (< 0) OR uncertainty is high (> 75th percentile)
    q75_uncert = np.percentile(uncertainty, 75, method='midpoint')
    high_risk_mask = (slack < 0) | (uncertainty > q75_uncert + eps)
    boosted_rank = upward_rank * np.where(high_risk_mask, PARAMS["criticality_boost"], 1.0)
    rank_score = -robust_normalize(boosted_rank)  # negative → higher priority for larger rank

    # 4. Risk-adjusted energy: already marginal & fuzzy; normalize robustly
    energy_score = PARAMS["energy_normalizer"] * robust_normalize(min_incremental_energy)

    # 5. Duration risk: execution + communication, scaled by uncertainty coupling
    duration_base = min_exec_time + min_comm_time
    duration_risk = duration_base * (1.0 + uncertainty * PARAMS["uncertainty_slack_coupling"])
    duration_score = PARAMS["duration_risk_weight"] * robust_normalize(duration_risk)

    # 6. Anti-starvation: longer wait → exponentially stronger priority boost (inverse exponential)
    # Compute exp(-wait_decay * wait), then invert so smaller score = higher priority
    # Clamp wait to avoid overflow in exp()
    clamped_wait = np.clip(ready_wait_time, 0.0, PARAMS["max_wait_clamp"])
    wait_boost = np.exp(-PARAMS["wait_decay"] * clamped_wait)
    wait_score = -wait_boost  # smaller score = higher priority

    # 7. Remaining work: mild penalty for large remaining work (to avoid blocking long tails)
    # But only if not already critical (prevents conflict with rank/latency)
    work_score = -0.3 * robust_normalize(remaining_work) * (1.0 - np.clip(norm_slack, 0.0, 1.0))

    # --- Aggregate final score (smaller = better) ---
    # Sum all contributions; all terms designed to be numerically balanced via robust normalization
    score = (
        coupled_risk
        + energy_score
        + duration_score
        + rank_score
        + wait_score
        + work_score
    )

    # Final safeguard: replace NaN/inf with finite values, preserve shape
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
