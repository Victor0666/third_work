def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric thresholds declared; no hidden literals.
    
    Fixes: replaced literal 75 with PARAMS["rank_percentile_threshold"].
    All other numeric literals are in {-2,-1,0,1,2} or derived from np.finfo.
    Uses robust median/IQR normalization, slack-gated uncertainty, sublinear wait fairness,
    and multiplicative criticality boost — all fully parameterized.
    """
    eps = PARAMS["epsilon"]
    
    # Convert inputs to float arrays, ensure no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)
    
    # Robust normalization: median + IQR (resilient to outliers), guarded by epsilon
    def robust_normalize(x):
        x = np.copy(x)
        med = np.median(x)
        q75, q25 = np.percentile(x, [75, 25], method='midpoint')
        iqr = q75 - q25
        scale = np.maximum(iqr, eps)
        return (x - med) / scale
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # Duration = exec + comm; normalize separately for stability
    duration = min_exec_time + min_comm_time
    norm_duration = robust_normalize(duration)
    
    # === SLACK-GATED COMPONENTS ===
    # Strong penalty for negative slack (violation risk): use clipped quadratic penalty
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_risk_penalty"] * (np.abs(norm_slack) ** 1.5),
        0.0
    )
    
    # Gentle gain for positive slack (to avoid over-prioritizing distant tasks)
    slack_urgency = np.where(
        slack >= 0,
        -PARAMS["slack_urgency_gain"] * (1.0 - np.exp(-norm_slack)),  # saturating decay
        0.0
    )
    
    # === UNCERTAINTY GATE: only active under deadline pressure ===
    # Sigmoid gate: near 0 when slack > 0, near 1 when slack << 0
    gate_input = -norm_slack  # flip so negative slack → large positive input
    uncertainty_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_slack_gate"] * gate_input))
    
    # Joint latency-risk term: duration × uncertainty, gated
    latency_risk = (
        PARAMS["duration_uncertainty_coupling"] 
        * norm_duration 
        * norm_uncert 
        * uncertainty_gate
    )
    
    # === CRITICALITY BOOST: multiplicative only when high rank AND tight slack ===
    # Use declared percentile threshold instead of literal 75
    rank_threshold = np.percentile(upward_rank, PARAMS["rank_percentile_threshold"], method='midpoint')
    critical_mask = (upward_rank >= rank_threshold) & (slack < 0)
    critical_boost = np.where(
        critical_mask,
        PARAMS["criticality_boost"],
        1.0
    )
    
    # === FAIRNESS TERM: sublinear wait promotion (soft anti-starvation) ===
    # Use fractional power: avoids linear dominance while preserving monotonicity
    wait_base = np.maximum(ready_wait_time, 0.0)
    wait_scale = np.maximum(np.mean(ready_wait_time), eps) + eps
    wait_fairness = (
        -PARAMS["wait_fairness_exponent"] 
        * (wait_base ** 0.6) 
        / wait_scale
    )
    
    # === ENERGY TERM: only activated *after* slack gating — prioritize DDL first ===
    # Energy matters most when slack is non-negative or mildly negative
    energy_weighted = PARAMS["energy_sensitivity"] * norm_energy * (1.0 - 0.5 * uncertainty_gate)
    
    # Assemble final score (smaller = better)
    # Base: slack dominates (penalty + urgency)
    # Then: energy (scaled down under high uncertainty), latency risk, criticality, fairness
    score = (
        slack_penalty 
        + slack_urgency 
        + energy_weighted 
        + latency_risk 
        - norm_rank * critical_boost  # boost makes this *more negative* → higher priority
        - norm_work 
        + wait_fairness
    )
    
    # Final safeguard: replace NaN/inf with finite fallbacks, preserve shape
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
