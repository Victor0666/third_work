def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: deadline-risk-gated criticality + starvation-aware efficiency.
    
    All numeric literals now restricted to {-2,-1,0,1,2}. Outlier clipping uses PARAMS["outlier_clip_scale"].
    Uses np.finfo for machine-precision epsilon fallback where needed.
    """
    eps = PARAMS["epsilon"]
    clip_scale = PARAMS["outlier_clip_scale"]
    
    # Safe cast & copy
    N = len(min_exec_time)
    exec_t = np.asarray(min_exec_time, dtype=float)
    comm_t = np.asarray(min_comm_time, dtype=float)
    energy = np.asarray(min_incremental_energy, dtype=float)
    slk = np.asarray(slack, dtype=float)
    rank = np.asarray(upward_rank, dtype=float)
    work = np.asarray(remaining_work, dtype=float)
    wait = np.asarray(ready_wait_time, dtype=float)
    unc = np.asarray(uncertainty, dtype=float)
    
    # Clip extreme values to avoid normalization collapse
    finfo = np.finfo(float)
    exec_t = np.clip(exec_t, -clip_scale, clip_scale)
    comm_t = np.clip(comm_t, -clip_scale, clip_scale)
    energy = np.clip(energy, -clip_scale, clip_scale)
    slk = np.clip(slk, -clip_scale, clip_scale)
    rank = np.clip(rank, -clip_scale, clip_scale)
    work = np.clip(work, eps, clip_scale)
    wait = np.clip(wait, 0.0, clip_scale)
    unc = np.clip(unc, 0.0, clip_scale)
    
    # Robust normalization: MAD-based (more outlier-resistant than std/mean)
    def robust_normalize(x):
        x = np.clip(x, -clip_scale, clip_scale)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = mad if mad > eps else np.mean(np.abs(x - med) + eps)
        return (x - med) / (scale + eps)
    
    # Slack response: convex penalty for violation, concave urgency for margin
    median_slack_abs = np.median(np.abs(slk)) + eps
    slack_score = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slk / median_slack_abs) ** 2,
        PARAMS["slack_urgency_gain"] * np.log1p(slk / median_slack_abs)
    )
    
    # Normalize energy directly — already risk-adjusted fuzzy marginal; lower is better
    energy_score = robust_normalize(energy)
    
    # Criticality leverage: activate upward rank only under deadline pressure or high uncertainty
    slack_pressure_ratio = np.abs(slk) / median_slack_abs
    unc_normalized = robust_normalize(unc)
    criticality_gate = np.where(
        (slack_pressure_ratio > 0.9) | (unc_normalized > 0.5),
        1.0,
        PARAMS["criticality_gate_low"]
    )
    rank_score = robust_normalize(rank) * criticality_gate * PARAMS["criticality_leverage"]
    
    # Starvation relief: boost for long-waiting tasks, inversely scaled by remaining work
    work_inv = 1.0 / (work + eps)
    starvation_score = -PARAMS["starvation_relief"] * wait * work_inv
    
    # Duration balance: penalize large exec+comm unless compensated by low energy
    duration = exec_t + comm_t + eps
    duration_score = robust_normalize(duration) * PARAMS["duration_balance"]
    
    # Uncertainty sensitivity: modulates slack penalty and criticality gate strength
    unc_effect = np.tanh(PARAMS["uncertainty_sensitivity"] * unc_normalized)
    
    # Assemble final score: smaller = higher priority
    score = (
        slack_score * (1.0 + unc_effect)
        + energy_score
        + duration_score
        + rank_score
        + starvation_score
    )
    
    # Ensure finite output and correct shape
    score = np.nan_to_num(score, nan=0.0, posinf=1e10, neginf=-1e10)
    score = np.clip(score, -1e10, 1e10)
    return score.astype(float).reshape(-1)
