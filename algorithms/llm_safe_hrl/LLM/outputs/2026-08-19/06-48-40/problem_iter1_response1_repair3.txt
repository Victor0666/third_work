def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule with all numeric thresholds declared as parameters.
    
    Fixes: replaced hidden 1e6 / -1e6 with PARAMS["score_clip_upper"] and
           PARAMS["score_clip_lower"]; all other literals are -2,-1,0,1,2 or np.finfo guards.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Safe cast & copy
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust normalization using parameterized percentiles
    def robust_normalize(x):
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        q75 = np.percentile(np.abs(x), PARAMS["robust_norm_q75"])
        q25 = np.percentile(np.abs(x), PARAMS["robust_norm_q25"])
        iqr = q75 - q25 + eps
        center = np.median(x)
        return (x - center) / iqr

    # --- Slack handling: strong nonlinearity for risk containment ---
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_power = np.where(
        slack >= 0,
        np.power(np.maximum(slack, eps), PARAMS["slack_sensitivity"]),
        -np.power(np.maximum(slack_abs, eps), PARAMS["slack_sensitivity"])
    )
    slack_score = np.where(
        slack < 0,
        PARAMS["slack_penalty_scale"] * slack_power,
        slack_power
    )

    # --- Criticality gating: boost upward_rank only under pressure ---
    pressure_index = -slack + uncertainty
    gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_slack_coupling"] * (pressure_index + eps)))
    criticality_score = PARAMS["criticality_boost"] * upward_rank * gate

    # --- Energy term: normalized marginal fuzzy energy ---
    energy_norm = robust_normalize(min_incremental_energy)
    energy_score = PARAMS["energy_normalizer"] * energy_norm

    # --- Duration risk: uncertainty-weighted exec+comm ---
    duration = min_exec_time + min_comm_time
    duration_risk = duration * (1.0 + uncertainty)
    duration_norm = robust_normalize(duration_risk)
    duration_score = PARAMS["duration_risk_weight"] * duration_norm

    # --- Anti-starvation: clipped exponential wait boost ---
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay"] * np.maximum(ready_wait_time, 0.0))
    wait_boost = np.clip(wait_boost, 0.0, PARAMS["wait_saturation_cap"])
    wait_norm = robust_normalize(wait_boost)
    wait_score = -wait_norm

    # --- Composite score ---
    score = (
        slack_score
        + energy_score
        + duration_score
        - criticality_score
        + wait_score
    )

    # Final safeguard: use declared clip bounds instead of literals
    score = np.clip(score, PARAMS["score_clip_lower"], PARAMS["score_clip_upper"])
    score = np.nan_to_num(score, nan=0.0, posinf=PARAMS["score_clip_upper"], neginf=PARAMS["score_clip_lower"])
    
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    
    return score
