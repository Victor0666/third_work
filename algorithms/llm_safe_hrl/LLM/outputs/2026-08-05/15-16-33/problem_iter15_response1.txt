def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
    1. Piecewise-linear urgency amplification: applies stronger bias *only* when slack ∈ [-δ, δ], 
       preserving bottleneck fidelity for non-critical tasks while sharpening zero-slack discrimination.
    2. Host-load-aware conditional gate: introduces `host_load_pressure` term that activates multiplicatively 
       only under high uncertainty *and* high upstream load pressure — decoupling energy from deadline logic.
    3. Unified risk-gated bottleneck: replaces additive DDL-risk amplification with multiplicative gating 
       of bottleneck pressure by urgency, ensuring critical-path release is *conditioned* on deadline risk.
    All normalizations use adaptive IQR→min-max fallback; no unbounded ops or hidden state.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max when dispersion is low
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR too small (low dispersion), else use IQR
        if iqr < eps:
            x_min, x_max = np.min(x), np.max(x)
            denom = x_max - x_min
            if denom < eps:
                return np.zeros_like(x)
            return (x - x_min) / (denom + eps)
        else:
            denom = iqr
            return (x - center) / (denom + eps)
    
    # Sigmoid urgency gate: maps slack → [0, 1] where negative slack → near 1.0 (high urgency)
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered,
                            -PARAMS["sigmoid_clip_bound"],
                            PARAMS["sigmoid_clip_bound"])
    urgency = 1.0 / (1.0 + np.exp(sigmoid_input))  # range [0, 1]
    
    # Normalize urgency
    norm_urgency = adaptive_normalize(urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Host-load-aware pressure: multiplicative gate activated only under joint high uncertainty & high critical load
    # Captures resource contention threat to critical path without global bias
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    host_load_condition = (
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty) &
        (upward_rank > np.quantile(upward_rank, 0.75, method='midpoint') + eps)
    ).astype(float)
    host_load_pressure = host_load_condition * upward_rank * remaining_work
    
    # Successor bottleneck: now gated by urgency (not additive DDL-ramp) → ensures release pressure scales with deadline risk
    # Includes remaining_work and host-load pressure for downstream load + contention awareness
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    bottleneck_pressure = duration * upward_rank * (1.0 + uncertainty + eps) * (remaining_work + eps) * (1.0 + norm_urgency)
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Adaptive linear ramp anti-starvation: saturates at median_wait * factor
    if N == 1:
        median_wait = ready_wait_time[0]
    else:
        median_wait = np.median(ready_wait_time)
    saturation_wait = median_wait * PARAMS["wait_ramp_saturation_factor"]
    wait_ramp = np.clip(ready_wait_time / (saturation_wait + eps), 0.0, 1.0)
    norm_wait = adaptive_normalize(wait_ramp)
    
    # Piecewise-linear urgency amplification window: active only in [-δ, δ] around slack=0
    # Enables precise zero-slack discrimination without distorting far-positive/negative slack behavior
    delta = PARAMS["urgency_slack_window_halfwidth"]
    in_critical_window = ((slack >= -delta) & (slack <= delta)).astype(float)
    # Linear ramp: 0 at slack=-delta, 1 at slack=0, 0 at slack=+delta → triangular kernel
    critical_urgency_bias = in_critical_window * (1.0 - np.abs(slack) / (delta + eps))
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline-critical window bias > urgency-gated bottleneck > host-load pressure > energy > fairness
    score = (
        critical_urgency_bias  # Sharp zero-slack discrimination only in critical window
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Release pressure conditioned on urgency
        + PARAMS["host_load_pressure_weight"] * adaptive_normalize(host_load_pressure)  # Load-aware critical-path acceleration
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Energy efficiency
        - norm_wait  # Favor recently arrived tasks (adaptive fairness)
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
