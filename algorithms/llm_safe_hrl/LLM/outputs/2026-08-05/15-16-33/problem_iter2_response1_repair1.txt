def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's structural strengths with Parent 1's robustness and fairness mechanisms.
    
    Key improvements:
      - Replaces exponential wait boost with bounded arctan saturation (more stable under long queues)
      - Introduces remaining_work × energy coupling to prioritize energy-critical bottlenecks
      - Uses tunable quantiles for all normalizations (consistent with Parent 1's robustness)
      - Retains Parent 2's percentile-gated critical-path bonus and duration-normalized energy efficiency
      - Adds slack-pressure gating to uncertainty coupling: only activates when slack < median
      - All numeric literals restricted to {-2,-1,0,1,2}; eps handled via np.finfo
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)
    N = len(slack)

    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_percentile_low"])
        q_high = np.percentile(x, PARAMS["iqr_percentile_high"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps_safe else (np.max(np.abs(x - center)) + eps_safe)
        return (x - center) / (denom + eps_safe)

    # Slack: piecewise — penalize negative slack strongly, reward positive slack gently
    slack_norm = iqr_normalize(slack)
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_risk_penalty"] * (-slack_norm),
        -PARAMS["slack_urgency_gain"] * slack_norm
    )

    # Duration-normalized energy efficiency: J/s, prevents bias toward trivial short tasks
    duration = np.maximum(min_exec_time + min_comm_time, eps_safe)
    energy_per_duration = min_incremental_energy / duration
    energy_eff_norm = iqr_normalize(energy_per_duration)

    # Critical-path bonus: activated only for top percentile of upward_rank (robust to scale)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps_safe)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_threshold"], 1.0, 0.0)
    rank_norm = iqr_normalize(upward_rank)
    critical_bonus = PARAMS["critical_path_bonus"] * rank_norm * critical_gate

    # Ready wait time: bounded arctan saturation (smooth, monotonic, avoids over-prioritization)
    wait_scaled = ready_wait_time / (PARAMS["iqr_percentile_high"] + eps_safe)  # reusing existing param as proxy for saturation scale
    wait_boost = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    wait_norm = iqr_normalize(wait_boost)

    # Uncertainty-slack coupling: only amplifies urgency when slack is tight (< median) AND uncertainty is high
    median_slack = np.median(slack)
    slack_pressure = np.where(slack < median_slack, 1.0, 0.0)
    unc_norm = iqr_normalize(uncertainty)
    coupled_urgency = PARAMS["uncertainty_slack_coupling"] * unc_norm * slack_pressure

    # Remaining-work × energy coupling: identifies high-energy bottlenecks in critical sub-DAGs
    # Normalized product emphasizes tasks that are both computationally heavy and energy-expensive
    rw_norm = iqr_normalize(remaining_work)
    energy_norm = iqr_normalize(min_incremental_energy)
    bottleneck_score = rw_norm * energy_norm * PARAMS["remaining_work_energy_coupling"]

    # Duration-efficiency ratio: balances duration cost vs energy cost
    dur_eff_ratio = PARAMS["duration_efficiency_ratio"] * iqr_normalize(duration)

    # Final score: smaller = higher priority
    # Ordering reflects objective hierarchy: deadline safety > energy efficiency > fairness > bottleneck mitigation
    score = (
        slack_penalty
        + PARAMS["energy_efficiency_weight"] * energy_eff_norm
        - critical_bonus
        - wait_norm
        + coupled_urgency
        + dur_eff_ratio
        + bottleneck_score
    )

    # Ensure finite output, shape compliance, and numerical stability
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
