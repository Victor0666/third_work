def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
    1. Conditional activation of urgency modulation: only enabled when slack distribution is sufficiently dispersed
       (variance > threshold), avoiding over-sharpening in near-deterministic deadlines.
    2. Hard feasibility guard on bottleneck amplification: uncertainty exponent applied *only* when both
       slack < median_slack AND uncertainty > bottleneck_feasibility_guard * max_uncertainty — prevents
       over-penalization of critical paths under low-risk conditions.
    3. Restored robust adaptive IQR+min-max fallback (from v0) for all normalizations — improves deadline
       feasibility under low-dispersion features, per empirical reflection.
    All operations remain finite, deterministic, and protected against NaN/inf/zero.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max when dispersion is low (v0-style)
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR too small or variance too low, else use IQR
        if iqr > eps and np.var(x) > PARAMS["slack_variance_activation_threshold"] * (np.max(x) - np.min(x) + eps)**2:
            denom = iqr
        else:
            denom = np.max(x) - np.min(x)
            if denom < eps:
                denom = eps
        
        return (x - center) / (denom + eps)
    
    # Base sigmoid urgency gate
    slack_centered = slack - 0.0  # Remove offset; now gated by variance condition
    base_steepness = PARAMS["slack_sigmoid_steepness"] if "slack_sigmoid_steepness" in PARAMS else 5.0
    sigmoid_input = np.clip(-base_steepness * slack_centered,
                            -18.0, 18.0)  # Fixed safe bounds instead of parametrized clip
    urgency_gate = 1.0 / (1.0 + np.exp(sigmoid_input))
    
    # Adaptive steepness modulation: only active when slack variance is high enough
    slack_var = np.var(slack) if N > 1 else 0.0
    slack_range = np.max(slack) - np.min(slack) if N > 0 else eps
    variance_sufficient = (slack_var > PARAMS["slack_variance_activation_threshold"] * (slack_range + eps)**2).astype(float)
    
    median_slack = np.median(slack) if N > 0 else 0.0
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    joint_ddl_pressure = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Modulated steepness only when variance is sufficient
    modulated_steepness = base_steepness * (1.0 + PARAMS["urgency_uncertainty_coupling_strength"] * joint_ddl_pressure * variance_sufficient)
    modulated_sigmoid_input = np.clip(-modulated_steepness * slack_centered,
                                      -18.0, 18.0)
    modulated_urgency = 1.0 / (1.0 + np.exp(modulated_sigmoid_input))
    
    norm_urgency = adaptive_normalize(modulated_urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Successor bottleneck coupling with hard feasibility guard on uncertainty amplification
    bottleneck_pressure = duration * upward_rank * (1.0 + modulated_urgency + eps)
    
    # Apply uncertainty exponent only under strict joint feasibility guard
    bottleneck_guard = (
        (slack < median_slack) &
        (uncertainty > PARAMS["bottleneck_feasibility_guard"] * max_uncertainty)
    ).astype(float)
    bottleneck_pressure = bottleneck_pressure * np.power(1.0 + uncertainty, 
                                                         PARAMS["bottleneck_uncertainty_amplification"] * bottleneck_guard)
    
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Ready wait time: bounded sigmoid saturation (stable anti-starvation)
    wait_scaled = ready_wait_time / (3.9 + eps)  # Fixed from prior best value, no param
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_scaled))  # [0,1], smooth, bounded
    norm_wait = adaptive_normalize(wait_saturation)
    
    # DDL-protection gate: activates only under joint DDL+uncertainty pressure
    ddl_risk_amplifier = joint_ddl_pressure * adaptive_normalize(uncertainty)
    
    # Final score: smaller = higher priority
    score = (
        norm_urgency  # Primary deadline pressure signal, adaptively sharpened only when needed
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors, guarded amplification
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - norm_wait  # Reward recently arrived tasks (anti-starvation)
        + ddl_risk_amplifier  # Boost uncertainty-aware scheduling only when DDL is threatened
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
