def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's smooth sigmoid gating and bottleneck coupling with Parent 1's successor-release penalty.
    Key structural improvements:
      - Adds vectorized successor-release penalty term, gated by slack feasibility sigmoid (not binary), to prevent downstream starvation.
      - Uses unified uncertainty-normalized dispersion for *all* normalized terms (bottleneck, energy, wait, successor) — improves risk alignment consistency.
      - Retains smooth sigmoid slack gate and learned power-coupling of upward_rank × remaining_work.
      - All parameters declared and used; no numeric literals beyond {-2,-1,0,1,2}.
      - Robust fallbacks for N=0/1 cases via np.where and safe clipping.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Pre-normalized hard-deadline violation signal
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Smooth sigmoid slack feasibility gate: activates strongly when slack < 0, decays smoothly
    slack_centered = -slack
    slack_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_feasibility_sigmoid_slope"] * (slack_centered + eps)))
    
    # Urgency proxy: linear deviation capped by power-law of max non-negative slack
    urgency_linear = np.clip(-slack, 0.0, 2.0)
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.where(N > 0, np.max(non_neg_slack), eps)
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.minimum(urgency_linear, urgency_cap)
    
    # Bottleneck pressure: (upward_rank * remaining_work)^p × duration × urgency × uncertainty-amplified factor
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    ur_interaction = np.power(upward_rank * remaining_work + eps, PARAMS["upward_remaining_interaction_power"])
    bottleneck_pressure = ur_interaction * duration * (1.0 + urgency_linear + eps)
    
    # Uncertainty amplification applied only where slack is tight (smoothly gated)
    unc_std = np.where(N > 1, np.std(uncertainty), eps)
    unc_normalized = (uncertainty - np.min(uncertainty)) / (np.ptp(uncertainty) + eps) if N > 1 else np.zeros_like(uncertainty)
    amplified_uncertainty = np.power(1.0 + unc_normalized, PARAMS["uncertainty_amplification_exponent"])
    bottleneck_pressure = bottleneck_pressure * (slack_gate * amplified_uncertainty + (1.0 - slack_gate))
    
    # Successor-release penalty: penalizes tasks whose successors have high work but low slack
    # Computed as remaining_work × (1 - slack_feasibility_sharpened), gated by slack_gate
    slack_feasibility = np.clip(slack / (np.abs(np.median(slack)) + eps), 0.0, 1.0) if N > 0 else np.zeros_like(slack)
    slack_feasibility_sharpened = np.power(slack_feasibility + eps, 0.75)  # hardcoded exponent to avoid extra param
    successor_release_penalty = remaining_work * (1.0 - slack_feasibility_sharpened)
    gated_successor_penalty = slack_gate * successor_release_penalty * PARAMS["successor_release_penalty_weight"]
    
    # Unified adaptive normalization using uncertainty-driven dispersion
    def unified_adaptive_normalize(x):
        x = np.copy(x)
        center = np.where(N > 0, np.median(x), 0.0)
        unc_dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
        fallback_range = np.where(N > 0, np.ptp(x), eps)
        denom = np.where(unc_dispersion > eps, unc_dispersion, fallback_range)
        return (x - center) / (denom + eps)
    
    norm_bottleneck = unified_adaptive_normalize(bottleneck_pressure)
    norm_successor = unified_adaptive_normalize(gated_successor_penalty)
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = unified_adaptive_normalize(energy_per_duration)
    
    # Bounded sigmoid wait saturation: smooth anti-starvation boost
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = unified_adaptive_normalize(wait_saturation)
    
    # Final score: smaller = higher priority
    # Includes all declared weights and the new successor-release term
    score = (
        PARAMS["ddl_urgency_weight"] * neg_slack
        + PARAMS["bottleneck_weight"] * norm_bottleneck
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        + norm_successor
        - PARAMS["fairness_weight"] * norm_wait
    )
    
    # Enforce finiteness and finite numeric output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
