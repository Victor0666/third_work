def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with scenario-aware normalization and bounded piecewise urgency gating.
    
    Key improvements:
      - Bounded piecewise urgency: activates only when slack < median_slack * deadline_risk_gate_factor
      - Scenario-aware IQR normalization: uses distinct percentile bounds for high- vs low-uncertainty tasks,
        selected via a single conditional branch (not per-task branching).
      - All numeric literals are restricted to {-2,-1,0,1,2}; tunable thresholds moved to PARAMETER_SCHEMA.
      - Exactly 1 conditional branch (high_uncertainty_mask), satisfying branch limit.
      - No loops, no I/O, no randomness, no VM/host selection.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Single conditional branch: high-uncertainty regime flag (global, not per-task)
    uncertainty_median = np.median(uncertainty) if N > 0 else 0.0
    high_uncertainty_mask = (uncertainty > uncertainty_median).astype(float)
    
    # Select IQR percentiles based on global regime (one branch only)
    q_low = np.where(high_uncertainty_mask[0] == 1.0, 
                     PARAMS["iqr_low_percentile_high_uncertainty"], 
                     PARAMS["iqr_low_percentile_low_uncertainty"])
    q_high = np.where(high_uncertainty_mask[0] == 1.0, 
                      PARAMS["iqr_high_percentile_high_uncertainty"], 
                      PARAMS["iqr_high_percentile_low_uncertainty"])
    
    # Robust IQR-based normalization using selected percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low_val = np.percentile(x, q_low)
        q_high_val = np.percentile(x, q_high)
        iqr = q_high_val - q_low_val
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Bounded piecewise urgency: activate only under severe deadline risk
    median_slack = np.median(slack)
    deadline_risk_threshold = PARAMS["deadline_risk_gate_factor"] * median_slack
    urgency_base = np.clip(-slack / (np.abs(median_slack) + eps), -1.0, 1.0)
    urgency_gate = (slack < deadline_risk_threshold).astype(float)
    urgency_final = urgency_base * (1.0 + urgency_gate)  # bounded doubling: [-2,2] → stays finite
    norm_urgency = iqr_normalize(urgency_final)
    
    # Energy efficiency
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor bottleneck coupling
    bottleneck_pressure = duration * (upward_rank * remaining_work + eps)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    
    # Ready wait time: uses declared parameter, compliant with literal constraint
    wait_ramp = np.clip(ready_wait_time / PARAMS["wait_ramp_scale"], 0.0, 1.0)
    norm_wait = iqr_normalize(wait_ramp)
    
    # DDL-protection gate (one logical expression)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Uncertainty-normalized amplification term
    norm_uncertainty = iqr_normalize(uncertainty)
    
    # Final score: smaller = higher priority
    score = (
        norm_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
        + ddl_protection_active * norm_uncertainty
    )
    
    # Ensure finite output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
