def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with structural improvements guided by counterfactual evidence:
      - Introduces conditional DDL protection gate: only activates bottleneck & successor terms when slack is tight AND uncertainty is high.
      - Replaces linear urgency with normalized neg_slack as dominant additive term (preserves hard deadline dominance).
      - Adds unconditional upward_rank × remaining_work interaction (verified to resolve starvation without quantile estimation).
      - Uses bounded sigmoid wait saturation with learned exponent for robust anti-starvation.
      - Removes all inactive parameters (energy_duration_ratio_weight, successor_bottleneck_coupling, urgency_cap_exponent) per diagnostics.
      - All normalizations use DDL-aware min-max clipping for small-N sets to preserve signal integrity.
      - No branching beyond boolean gates; all operations finite, deterministic, and epsilon-guarded.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Pre-normalized dominant DDL violation signal: raw negative slack (no scaling distortion)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Compute median_slack for gating logic — robust against outliers
    median_slack = np.median(slack) if N > 0 else 0.0
    
    # Uncertainty-normalized dispersion for adaptive scaling
    unc_std = np.std(uncertainty) if N > 1 else eps
    dispersion_base = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
    
    # DDL-protection activation condition: tight slack AND high normalized uncertainty
    # Use clipped range-based uncertainty normalization for stability
    unc_min, unc_max = np.min(uncertainty), np.max(uncertainty)
    unc_range = np.maximum(unc_max - unc_min, eps)
    unc_normalized = (uncertainty - unc_min) / (unc_range + eps)
    ddl_protection_active = (slack <= median_slack) & (unc_normalized > PARAMS["ddl_protection_threshold"])
    
    # Unconditional critical-path pressure: upward_rank × remaining_work (no gating)
    critical_path_pressure = upward_rank * remaining_work
    # Normalize using DDL-aware min-max (clipped range) for small-N robustness
    cp_min, cp_max = np.min(critical_path_pressure), np.max(critical_path_pressure)
    cp_range = np.maximum(cp_max - cp_min, eps)
    norm_critical_path = (critical_path_pressure - cp_min) / (cp_range + eps)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + uncertainty)^amplification
    # Only active under DDL protection
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    bottleneck_base = duration * upward_rank * remaining_work
    bottleneck_pressure = np.where(
        ddl_protection_active,
        bottleneck_base * np.power(1.0 + unc_normalized, PARAMS["bottleneck_uncertainty_amplification"]),
        0.0
    )
    # Normalize with same DDL-aware min-max
    bp_min, bp_max = np.min(bottleneck_pressure), np.max(bottleneck_pressure)
    bp_range = np.maximum(bp_max - bp_min, eps)
    norm_bottleneck = np.where(
        ddl_protection_active,
        (bottleneck_pressure - bp_min) / (bp_range + eps),
        0.0
    )
    
    # Bounded sigmoid wait saturation: smooth anti-starvation boost
    wait_scaled = ready_wait_time / (1.0 + eps)  # unitless scale
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_sigmoid = 1.0 / (1.0 + np.exp(-wait_clipped))
    # Apply learned exponent for saturation steepness
    wait_saturation = np.power(wait_sigmoid, PARAMS["wait_saturation_exponent"])
    # Normalize wait term separately for fairness
    w_min, w_max = np.min(wait_saturation), np.max(wait_saturation)
    w_range = np.maximum(w_max - w_min, eps)
    norm_wait = (wait_saturation - w_min) / (w_range + eps)
    
    # Final score: smaller = higher priority
    # Dominant ordering: DDL violation > critical path > bottleneck (conditional) > fairness
    score = (
        neg_slack  # Raw penalty — strongest signal for deadline violation
        + PARAMS["upward_rank_remaining_work_weight"] * norm_critical_path
        + norm_bottleneck
        - norm_wait
    )
    
    # Enforce finiteness and shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure output is (N,) float64 array
    return score.astype(float, copy=False)
