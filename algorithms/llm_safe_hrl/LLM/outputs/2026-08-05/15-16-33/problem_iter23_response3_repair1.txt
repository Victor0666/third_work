def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 1's robust IQR normalization and hard DDL guard 
    with Parent 2's successor-release pressure, feasibility-gated bottleneck, and blended fairness.
    
    Key structural improvements:
      - Dual DDL protection: hard +inf gate for slack < ddl_hard_gate_threshold (prevents violation),
        plus soft urgency suppression below urgency_suppression_threshold (avoids conflicting signals).
      - Unified robust normalization: IQR-based (from Parent 1) for stability on small N, enhanced with
        uncertainty-aware dispersion scaling (from Parent 2) via adaptive dispersion = max(IQR, std(uncertainty)).
      - Successor-release pressure: upward_rank * remaining_work / (1 + |slack|)^exponent — decays smoothly
        under slack surplus, dominates under deadline pressure.
      - Bottleneck pressure gated by feasibility AND amplified by uncertainty only when slack permits.
      - Urgency uses smooth sigmoid (Parent 1) *and* hard suppression (Parent 2) — sigmoid applied only to non-suppressed region.
      - Anti-starvation blends sigmoid and linear inverted wait terms (Parent 2), normalized robustly (Parent 1).
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization with uncertainty fallback dispersion
    def robust_normalize(x):
        x = np.copy(x)
        if N > 2:
            q25 = np.percentile(x, PARAMS["percentile_q25"])
            q50 = np.percentile(x, PARAMS["percentile_q50"])
            q75 = np.percentile(x, PARAMS["percentile_q75"])
        else:
            q25 = q50 = q75 = x[0] if N > 0 else 0.0
        iqr = q75 - q25
        unc_std = np.std(uncertainty) if N > 1 else eps
        dispersion = np.maximum(iqr, unc_std) + eps
        center = q50
        return (x - center) / (dispersion + eps)
    
    # Hard DDL gate: assign +inf priority to tasks violating hard deadline constraint
    is_hard_violated = slack < PARAMS["ddl_hard_gate_threshold"]
    
    # 1. Urgency: sigmoid of normalized slack, but zeroed where slack < suppression threshold
    slack_masked = np.where(slack >= PARAMS["urgency_suppression_threshold"], slack, np.nan)
    slack_norm = robust_normalize(slack_masked)
    # Invert so smaller slack → higher urgency; apply steep sigmoid only on valid region
    urgency_raw = np.where(
        np.isfinite(slack_norm),
        1.0 - 1.0 / (1.0 + np.exp(-slack_norm * PARAMS["slack_sigmoid_steepness"])),
        0.0
    )
    # Cap urgency using power-law cap derived from non-negative slack
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency = np.minimum(urgency_raw, urgency_cap)
    norm_urgency = robust_normalize(urgency)
    
    # 2. Successor-release pressure: critical-path work scaled by deadline pressure decay
    slack_abs = np.abs(slack) + eps
    release_decay = np.power(1.0 + slack_abs, -PARAMS["successor_release_pressure_exponent"])
    successor_release_pressure = upward_rank * remaining_work * release_decay
    norm_successor_release = robust_normalize(successor_release_pressure)
    
    # 3. Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency), 
    # amplified by uncertainty only when slack permits (feasibility gate)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    base_bottleneck = duration * upward_rank * remaining_work * (1.0 + urgency + eps)
    
    # Feasibility mask: activate amplification only when slack >= threshold
    feasibility_mask = (slack >= PARAMS["ddl_hard_gate_threshold"]).astype(float)
    
    # Uncertainty amplification: power-law boost using normalized uncertainty
    unc_norm = robust_normalize(uncertainty)
    amp_factor = np.power(1.0 + np.abs(unc_norm), PARAMS["uncertainty_amplification_exponent"])
    bottleneck_pressure = base_bottleneck * amp_factor * feasibility_mask
    norm_bottleneck = robust_normalize(bottleneck_pressure)
    
    # 4. Energy efficiency: marginal energy per unit duration (J/s), normalized
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = robust_normalize(energy_per_duration)
    
    # 5. Anti-starvation: blended sigmoid + linear inverted wait term
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_sigmoid = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait_sigmoid = robust_normalize(wait_sigmoid)
    norm_wait_linear = 1.0 - robust_normalize(ready_wait_time)
    norm_wait = (
        PARAMS["wait_fairness_blend_weight"] * norm_wait_sigmoid
        + (1.0 - PARAMS["wait_fairness_blend_weight"]) * norm_wait_linear
    )
    
    # 6. Critical-path coupling: pure structural importance (ungated, unamplified)
    cp_coupling = upward_rank * remaining_work
    norm_cp_coupling = robust_normalize(cp_coupling)
    
    # Final score: smaller = higher priority
    # Dominance order: urgency > successor-release > bottleneck > energy > fairness > cp-coupling
    score = (
        norm_urgency
        + PARAMS["bottleneck_coupling_strength"] * norm_successor_release
        + PARAMS["bottleneck_coupling_strength"] * norm_bottleneck
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        - norm_wait
        + PARAMS["upward_rank_remaining_work_coupling"] * norm_cp_coupling
    )
    
    # Apply hard DDL gate: set score to +inf for violated tasks
    finfo = np.finfo(float)
    score = np.where(is_hard_violated, finfo.max, score)
    
    # Enforce finiteness and return
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
