def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with structural improvements guided by counterfactual evidence:
      - Introduces joint DDL-risk protection gate: activates bottleneck & successor-release terms only when
        (slack <= median_slack * ddl_risk_gate_slack_ratio) AND (normalized_uncertainty > ddl_risk_gate_threshold)
      - Replaces sigmoid wait saturation with bounded power-law saturation for smoother, more robust anti-starvation.
      - Uses raw upward_rank × remaining_work as uncoupled critical-path pressure (no multiplicative urgency coupling).
      - Adds uncertainty-modulated energy penalty: higher uncertainty → stronger marginal energy penalty.
      - Removes all adaptive normalization on neg_slack and urgency_linear to preserve hard-deadline dominance.
      - All operations are finite, deterministic, and use only {-2,-1,0,1,2} literals.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Pre-normalized dominant deadline violation signal: raw negative slack
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Median-based DDL risk context for gating
    median_slack = np.median(slack) if N > 0 else 0.0
    high_risk_slack_condition = slack <= (median_slack * PARAMS["ddl_risk_gate_slack_ratio"] + eps)
    
    # Uncertainty normalization for gating: [0,1] scaled via robust min-max
    unc_min = np.min(uncertainty) if N > 0 else 0.0
    unc_max = np.max(uncertainty) if N > 0 else eps
    normalized_uncertainty = (uncertainty - unc_min) / (unc_max - unc_min + eps)
    high_uncertainty_condition = normalized_uncertainty > PARAMS["ddl_risk_gate_threshold"]
    
    # Joint DDL-risk protection gate: only active under both conditions
    ddl_risk_gate = np.where(high_risk_slack_condition & high_uncertainty_condition, 1.0, 0.0)
    
    # Critical-path pressure: upward_rank × remaining_work — uncoupled, unnormalized, additive
    critical_path_pressure = upward_rank * remaining_work
    
    # Uncertainty-modulated energy penalty: higher uncertainty → stronger penalty on incremental energy
    energy_penalty = min_incremental_energy * (1.0 + PARAMS["energy_uncertainty_coupling"] * uncertainty)
    
    # Bounded power-law wait saturation: smooth, monotonic anti-starvation boost
    wait_scaled = np.clip(ready_wait_time / (np.maximum(np.mean(ready_wait_time), eps) + eps), 0.0, 2.0)
    wait_saturation = np.power(wait_scaled + eps, PARAMS["wait_saturation_exponent"])
    
    # Final score: smaller = higher priority
    # Dominant ordering: neg_slack >> gated critical-path >> energy penalty >> wait saturation (as fairness boost)
    score = (
        neg_slack  # Strongest term: raw deadline violation penalty
        + ddl_risk_gate * PARAMS["upward_rank_remaining_work_weight"] * critical_path_pressure
        + energy_penalty
        - wait_saturation
    )
    
    # Enforce finiteness and avoid NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
