def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's successor-release penalty with Parent 1's robust DDL-aware normalization:
      - Retains Parent 2's evidence-backed successor-release penalty (SUCCESSOR_RELEASE_BLOCKING mitigation).
      - Integrates Parent 1's DDL-aware min-max normalization for neg_slack and critical terms to preserve dominance ordering under small-N or skewed distributions.
      - Adds anti-starvation offset to prevent wait_decay from overwhelming hard deadline signals.
      - Uses unified uncertainty-normalized gating (Parent 1 style) for more stable DDL-protection activation.
      - Energy normalization retains Parent 2's adaptive dispersion fallback but applies DDL-aware clipping to avoid outlier distortion.
      - All operations guarded; no loops, side effects, or non-determinism.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Pre-normalized hard-deadline violation signal: dominant term — clipped to avoid unbounded growth
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Compute median_slack for gating and normalization
    median_slack = np.median(slack) if N > 0 else 0.0
    
    # Unified uncertainty-normalized gating (Parent 1 style): robust against outliers
    unc_center = np.median(uncertainty) if N > 0 else 0.0
    unc_range = np.ptp(uncertainty) if N > 1 else eps
    norm_uncertainty = (uncertainty - unc_center) / (unc_range + eps)
    ddl_risk_gate = np.where(
        (slack <= median_slack) & (norm_uncertainty > PARAMS["ddl_protection_gate_threshold"]),
        1.0,
        0.0
    )
    
    # Critical-path pressure: upward_rank × remaining_work, gated by DDL risk
    critical_path_pressure = upward_rank * remaining_work
    
    # Successor-release blocking penalty: high remaining_work × low slack feasibility
    slack_feasibility = np.clip(slack / (np.abs(median_slack) + eps), 0.0, 1.0)
    slack_feasibility_sharpened = np.power(slack_feasibility + eps, PARAMS["slack_feasibility_exponent"])
    successor_release_penalty = remaining_work * (1.0 - slack_feasibility_sharpened)
    
    # Apply gating and weighting
    gated_critical_pressure = ddl_risk_gate * critical_path_pressure
    gated_successor_penalty = ddl_risk_gate * successor_release_penalty * PARAMS["successor_release_penalty_weight"]
    
    # DDL-aware min-max normalization (Parent 1 style) for stability on small-N and outliers
    def ddl_aware_normalize(x):
        x = np.copy(x)
        x_min = np.min(x) if N > 0 else 0.0
        x_max = np.max(x) if N > 0 else 1.0
        range_val = np.maximum(x_max - x_min, eps)
        # Clip to avoid extreme outliers dominating normalization
        clip_offset = PARAMS["anti_starvation_offset"] * range_val
        x_clipped = np.clip(x, x_min - clip_offset, x_max + clip_offset)
        return (x_clipped - x_min) / (range_val + eps)
    
    # Normalize critical and successor terms using DDL-aware scheme
    norm_critical = ddl_aware_normalize(critical_path_pressure)
    norm_successor = ddl_aware_normalize(successor_release_penalty)
    
    # Energy term: adaptive normalization with DDL-aware clipping
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    energy_center = np.median(energy_per_duration) if N > 0 else 0.0
    energy_std = np.std(energy_per_duration) if N > 1 else 0.0
    energy_range = np.ptp(energy_per_duration) if N > 0 else 0.0
    energy_dispersion = np.where(energy_std > eps, energy_std, energy_range)
    # Apply DDL-aware clipping before normalization
    energy_clipped = np.clip(energy_per_duration, energy_center - 2.0*energy_dispersion, energy_center + 2.0*energy_dispersion)
    norm_energy = (energy_clipped - energy_center) / (energy_dispersion * PARAMS["energy_normalization_scale"] + eps)
    
    # Bounded power-law wait decay with anti-starvation offset
    max_wait = np.max(ready_wait_time) if N > 0 else eps
    wait_ratio = np.clip(ready_wait_time / (max_wait + eps), 0.0, 1.0)
    wait_decay = np.power(wait_ratio + eps, PARAMS["wait_decay_power"])
    # Offset ensures fairness never overrides hard DDL violation
    norm_wait = wait_decay + PARAMS["anti_starvation_offset"]
    
    # Final score: smaller = higher priority
    # Dominant ordering: raw neg_slack > gated successor penalty > gated critical pressure > energy > wait decay
    score = (
        neg_slack
        + PARAMS["successor_release_penalty_weight"] * norm_successor
        + PARAMS["upward_rank_remaining_work_weight"] * norm_critical
        + norm_energy
        - norm_wait
    )
    
    # Enforce finiteness and finite numeric output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
