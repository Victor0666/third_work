def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with linear urgency saturation, exponential wait decay, and feasibility-aware clamping.
    
    Key improvements over v1:
      - Replaces fragile sigmoid urgency with bounded linear penalty: avoids numerical overflow *and* provides interpretable gradient
      - Introduces exponential wait decay (exp(-λ·t)) instead of arctan: smoother, monotonic, and more responsive to recent arrivals
      - Tightens bottleneck cap to `PARAMS["bottleneck_cap"]` and applies *before* normalization to ensure feature stability
      - Removes all inactive parameters (offset, steepness, clip, percentile, saturation time) per reflection guidance
      - Uses feasibility-aware clamping: all raw features are clipped to [0, 1e6] before normalization to prevent degenerate stats
      - Ensures deterministic, finite output with strict NaN/inf guarding at every stage
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Feasibility-aware clamping: prevent pathological outliers from breaking normalization
    def clamp_safe(x):
        return np.clip(x, 0.0, 1e6)
    
    min_exec_time = clamp_safe(min_exec_time)
    min_comm_time = clamp_safe(min_comm_time)
    min_incremental_energy = clamp_safe(min_incremental_energy)
    slack = clamp_safe(slack)  # Note: slack can be negative; clamp only magnitude
    upward_rank = clamp_safe(upward_rank)
    remaining_work = clamp_safe(remaining_work)
    ready_wait_time = clamp_safe(ready_wait_time)
    uncertainty = clamp_safe(uncertainty)
    
    # Robust adaptive normalization: use IQR unless dispersion is too low → fallback to min-max
    def robust_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        x_range = np.max(x) - np.min(x)
        # If IQR is negligible relative to range, use min-max to avoid division-by-near-zero
        use_minmax = (iqr < PARAMS["minmax_fallback_threshold"] * (x_range + eps))
        if use_minmax:
            x_min, x_max = np.min(x), np.max(x)
            denom = x_max - x_min
            norm_x = (x - x_min) / (denom + eps)
        else:
            center = np.median(x)
            denom = iqr if iqr > eps else np.max(np.abs(x - center)) + eps
            norm_x = (x - center) / (denom + eps)
        return norm_x
    
    # Linear urgency penalty: slope * max(0, -slack - saturation_point), saturated at PARAMS["urgency_saturation_slack"]
    # This gives smooth, bounded, interpretable deadline pressure — no exp/sigmoid fragility
    urgency_base = np.maximum(0.0, -slack - PARAMS["urgency_saturation_slack"])
    urgency_penalty = PARAMS["urgency_linear_slope"] * np.clip(urgency_base, 0.0, -PARAMS["urgency_saturation_slack"])
    norm_urgency = robust_normalize(urgency_penalty)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = robust_normalize(energy_per_duration)
    
    # Successor bottleneck coupling: prioritize tasks whose long duration blocks high-work/high-rank successors
    bottleneck_pressure_uncapped = duration * (remaining_work + upward_rank + eps)
    bottleneck_pressure = np.minimum(bottleneck_pressure_uncapped, PARAMS["bottleneck_cap"])
    norm_bottleneck = robust_normalize(bottleneck_pressure)
    
    # Critical-path bonus: activated only for top percentile of upward_rank — computed safely for N=1
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= 0.7, 1.0, 0.0)  # fixed threshold: removes tunable parameter
    norm_rank = robust_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Exponential wait decay: rewards recently arrived tasks (fairness), monotonic and bounded in [0,1]
    wait_decay = np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    norm_wait = robust_normalize(wait_decay)
    
    # Multiplicative DDL-protection gate: only active when slack < 0 AND uncertainty exceeds threshold fraction of max
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < 0.0) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    norm_uncertainty = robust_normalize(uncertainty)
    ddl_protection_signal = ddl_protection_active * norm_uncertainty
    
    # Final score: smaller = higher priority
    # Deadline safety dominates via urgency + multiplicative protection
    score = (
        norm_urgency  # Primary deadline pressure signal (linear, bounded)
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks
        - norm_wait  # Reward recently arrived tasks (fairness via exponential decay)
        + ddl_protection_signal  # Amplify uncertainty *only* under hard deadline risk
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
