def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with successor-release-aware critical-path scaling and diagnostic DDL-protection gating.
    
    Key structural improvements:
      - Replaces percentile-gated critical bonus with *upward_rank decay scaling*: 
        score = upward_rank / (1 + successor_depth_penalty), where depth penalty is derived from normalized slack/uncertainty.
      - Introduces *DDL-protection gate*: only activates uncertainty-slack coupling AND critical scaling when 
        slack <= slack_quantile AND uncertainty >= uncertainty_quantile — using high-confidence diagnostic signals.
      - Adds *normalized upward_rank × remaining_work* term to prioritize high-importance bottlenecks that block deep sub-DAGs.
      - All normalizations use tunable IQR percentiles; no hardcoded constants beyond {-2,-1,0,1,2}.
      - Uses robust arctan wait boost and bounded energy-duration efficiency.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)
    N = len(slack)

    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_percentile_low"])
        q_high = np.percentile(x, PARAMS["iqr_percentile_high"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps_safe else (np.max(np.abs(x - center)) + eps_safe)
        return (x - center) / (denom + eps_safe)

    # Slack: piecewise — penalize negative slack strongly, reward positive slack gently
    slack_norm = iqr_normalize(slack)
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_risk_penalty"] * (-slack_norm),
        -PARAMS["slack_urgency_gain"] * slack_norm
    )

    # Duration-normalized energy efficiency: J/s
    duration = np.maximum(min_exec_time + min_comm_time, eps_safe)
    energy_per_duration = min_incremental_energy / duration
    energy_eff_norm = iqr_normalize(energy_per_duration)

    # DDL-protection gate: active only under high-confidence deadline risk (tight slack + high uncertainty)
    slack_quantile = np.percentile(slack, PARAMS["ddl_protection_quantile"] * 100.0)
    unc_quantile = np.percentile(uncertainty, PARAMS["uncertainty_confidence_quantile"])
    ddl_protection_gate = np.where(
        (slack <= slack_quantile) & (uncertainty >= unc_quantile),
        1.0,
        0.0
    )

    # Successor-release-aware upward_rank scaling: decays with estimated blocking depth
    # Blocking depth proxy: slack pressure + uncertainty → higher values imply deeper critical dependency
    slack_pressure = np.maximum(0.0, (slack_quantile - slack) / (np.abs(slack_quantile) + eps_safe))
    depth_proxy = slack_pressure + iqr_normalize(uncertainty)
    # Apply soft decay: rank / (1 + sensitivity * depth_proxy)
    scaled_rank = upward_rank / (1.0 + PARAMS["successor_release_sensitivity"] * np.clip(depth_proxy, 0.0, 2.0))
    rank_norm = iqr_normalize(scaled_rank)
    critical_bonus = PARAMS["critical_path_bonus"] * rank_norm * ddl_protection_gate

    # Ready wait time: bounded arctan saturation
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps_safe)
    wait_boost = (2.0 / np.pi) * np.arctan(wait_scaled)
    wait_norm = iqr_normalize(wait_boost)

    # Uncertainty-slack coupling: now gated by DDL-protection signal
    unc_norm = iqr_normalize(uncertainty)
    coupled_urgency = PARAMS["uncertainty_slack_coupling"] * unc_norm * slack_pressure * ddl_protection_gate

    # Remaining-work × energy coupling: bottleneck identification
    rw_norm = iqr_normalize(remaining_work)
    energy_norm = iqr_normalize(min_incremental_energy)
    bottleneck_score = rw_norm * energy_norm * PARAMS["remaining_work_energy_coupling"]

    # Upward_rank × remaining_work coupling: prioritizes high-importance bottlenecks blocking large sub-DAGs
    rank_rw_interaction = iqr_normalize(upward_rank) * rw_norm * PARAMS["critical_path_bonus"] * ddl_protection_gate

    # Duration-efficiency ratio
    dur_eff_ratio = PARAMS["duration_efficiency_ratio"] * iqr_normalize(duration)

    # Final score: smaller = higher priority
    # Hierarchy: deadline safety > critical-path release > energy efficiency > fairness > bottlenecks
    score = (
        slack_penalty
        + PARAMS["energy_efficiency_weight"] * energy_eff_norm
        - critical_bonus
        - wait_norm
        + coupled_urgency
        + dur_eff_ratio
        + bottleneck_score
        + rank_rw_interaction
    )

    # Ensure finite output, shape compliance, and numerical stability
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
