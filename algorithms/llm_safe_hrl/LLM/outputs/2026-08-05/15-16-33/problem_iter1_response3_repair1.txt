def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing deadline risk mitigation, critical-path awareness,
    and energy-duration efficiency — with starvation prevention and uncertainty-coupled urgency.
    
    Key innovations:
      - Piecewise slack handling: strong penalty for negative slack, gentle reward for positive slack
      - Duration-normalized energy efficiency: min_incremental_energy / (min_exec_time + min_comm_time + epsilon)
      - Critical-path bonus gated by upward_rank percentile (not raw value) to avoid dominance
      - Ready-wait boost decays exponentially to prevent over-prioritization of old tasks
      - Uncertainty-slack coupling: amplifies urgency only when both slack is tight AND uncertainty is high
      - All features robustly normalized using interquartile range (IQR) for outlier resilience
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure finite inputs and avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization (more stable than mean/std for skewed distributions)
    def iqr_normalize(x):
        x = np.copy(x)
        q25, q75 = np.percentile(x, [25, 75])
        iqr = q75 - q25
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Slack: piecewise — penalize negative slack strongly, reward positive slack gently
    slack_norm = iqr_normalize(slack)
    slack_penalty = np.where(slack < 0, 
                            PARAMS["slack_risk_penalty"] * (-slack_norm), 
                            -PARAMS["slack_urgency_gain"] * slack_norm)
    
    # Energy efficiency: marginal energy per unit of total duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    energy_eff_norm = iqr_normalize(energy_per_duration)
    
    # Critical-path bonus: only activate for top percentile of upward_rank
    rank_norm = iqr_normalize(upward_rank)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        # Compute percentile rank: fraction of elements <= current element
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_threshold"], 1.0, 0.0)
    critical_bonus = PARAMS["critical_path_bonus"] * rank_norm * critical_gate
    
    # Waiting-time boost: exponential decay ensures fairness without starvation
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    wait_norm = iqr_normalize(wait_boost)
    
    # Uncertainty-slack coupling: only amplify urgency when both are high-risk
    unc_norm = iqr_normalize(uncertainty)
    slack_pressure = np.maximum(-slack_norm, 0.0)  # only pressure from tight/negative slack
    coupled_urgency = PARAMS["uncertainty_slack_coupling"] * unc_norm * slack_pressure
    
    # Duration-efficiency ratio: balance duration cost vs energy cost
    dur_eff_ratio = PARAMS["duration_efficiency_ratio"] * iqr_normalize(duration)
    
    # Final score: smaller = higher priority
    # Prioritize deadline safety first (slack_penalty dominates), then efficiency, then fairness
    score = (
        slack_penalty
        + PARAMS["energy_efficiency_weight"] * energy_eff_norm
        - critical_bonus
        - wait_norm
        + coupled_urgency
        + dur_eff_ratio
    )
    
    # Clamp to finite range and ensure no NaN/infs
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
