def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with corrected parameter usage:
      - Removes unused ddl_aware_minmax_clip and uncertainty_penalty_weight.
      - Uses robust min-max normalization with fixed 5% tail trimming (no parameter dependency).
      - Introduces DDL protection gate: multiplicative priority boost for slack < threshold.
      - Successor-release bottleneck interaction uses slack_mask derived from slack_sensitivity_threshold.
      - All numeric literals are restricted to {-2,-1,0,1,2}; epsilon guarded via PARAMS["epsilon"].
      - Zero conditional branches; fully vectorized and finite-valued.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust min-max normalization with fixed 5% tail trimming
    def robust_minmax_normalize(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros_like(x)
        # Trim 5% tails to suppress outliers while preserving signal integrity
        low_pct = np.percentile(x, 5) if N > 1 else np.min(x)
        high_pct = np.percentile(x, 95) if N > 1 else np.max(x)
        x_clipped = np.clip(x, low_pct, high_pct)
        x_min = np.min(x_clipped)
        x_max = np.max(x_clipped)
        range_val = x_max - x_min
        # Use fixed epsilon-guarded range
        norm_range = np.maximum(range_val, eps)
        center = (x_min + x_max) / 2.0
        return (x - center) / (norm_range + eps)
    
    # Dominant hard-deadline term: raw clipped negative slack, weighted
    neg_slack = np.clip(-slack, 0.0, None)
    ddl_penalty = PARAMS["ddl_violation_weight"] * neg_slack
    
    # Urgency proxy: linear distance from median slack, capped by feasibility-preserving power law
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, None)
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_clipped = np.minimum(urgency_linear, urgency_cap)
    norm_urgency = robust_minmax_normalize(urgency_clipped)
    
    # Unconditional critical-path pressure: upward_rank × remaining_work (no gating)
    critical_path_pressure = upward_rank * remaining_work
    norm_critical_path = robust_minmax_normalize(critical_path_pressure)
    
    # Energy efficiency: marginal energy per unit duration (J/s), weighted
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = robust_minmax_normalize(energy_per_duration)
    
    # Successor-release bottleneck interaction: duration × upward_rank × (1 + urgency_clipped) × (1 − slack_mask)
    # slack_mask = 1 when slack is safe (>= threshold), 0 when critical (< threshold)
    slack_mask = (slack >= PARAMS["slack_sensitivity_threshold"]).astype(float)
    bottleneck_base = duration * upward_rank * (1.0 + urgency_clipped + eps) * (1.0 - slack_mask + eps)
    norm_bottleneck = robust_minmax_normalize(bottleneck_base)
    
    # Bounded sigmoid wait saturation: smooth anti-starvation boost (branch-free)
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = robust_minmax_normalize(wait_saturation)
    
    # DDL protection gate: multiplicative priority boost for critically late tasks
    # Boost factor = 1 + weight × (1 − slack_mask), applied multiplicatively to base urgency + CP terms
    ddl_protection_boost = 1.0 + PARAMS["ddl_protection_gate_weight"] * (1.0 - slack_mask)
    
    # Final score: smaller = higher priority
    base_score = (
        ddl_penalty
        + norm_urgency
        + PARAMS["upward_rank_weight"] * norm_critical_path
        + PARAMS["remaining_work_weight"] * norm_critical_path
        + PARAMS["bottleneck_coupling_weight"] * norm_bottleneck
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        - norm_wait
    )
    score = base_score * ddl_protection_boost
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
