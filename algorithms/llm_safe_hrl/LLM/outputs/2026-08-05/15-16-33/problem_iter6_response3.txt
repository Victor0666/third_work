def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with bounded piecewise-linear slack penalty for strict DDL safety,
       reinstated robust successor-release delay (no wait scaling), and simplified critical-path gating.
    
    Key improvements:
      - Replaced hybrid/convex slack scoring with monotonic, bounded piecewise-linear penalty:
          * Negative slack: linear penalty with steeper slope (higher violation cost)
          * Positive slack: gentler linear penalty (avoids over-rewarding margin)
          * Zero-crossing offset shifts sensitivity leftward to trigger earlier intervention
      - Removed wait-gated critical boost per reflection — eliminates starvation risk and simplifies fairness
      - Restored original successor_delay formulation (un-scaled) — proven causal bottleneck unblocking
      - Simplified critical_bonus to percentile-gated only (no wait coupling), improving determinism
      - All normalizations remain IQR-based with tunable percentiles; no unbounded terms or fragile heuristics
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Bounded piecewise-linear slack penalty: monotonic, interpretable, and strictly increasing
    # Shifted zero-crossing improves early violation detection without convex instability
    shifted_slack = slack - PARAMS["slack_zero_crossing_offset"]
    slack_penalty = np.where(
        shifted_slack < 0,
        PARAMS["slack_piecewise_slope_negative"] * shifted_slack,
        PARAMS["slack_piecewise_slope_positive"] * shifted_slack
    )
    # Ensure penalty is non-negative: violation must dominate margin erosion
    slack_penalty = np.maximum(slack_penalty, 0.0)
    norm_slack_penalty = iqr_normalize(slack_penalty)
    
    # Duration-normalized energy efficiency: J / (s + eps)
    duration = min_exec_time + min_comm_time + eps
    energy_efficiency = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_efficiency)
    
    # Successor-release delay: robust, unscaled formulation — proven to unblock bottlenecks
    raw_successor_delay = remaining_work / (slack + eps)
    bounded_successor_delay = np.minimum(raw_successor_delay, PARAMS["successor_delay_upper_bound"])
    norm_successor_delay = iqr_normalize(bounded_successor_delay)
    
    # Critical-path bonus: percentile-gated only (no wait coupling) — simpler, fairer, starvation-free
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_upward_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_upward_rank
    
    # DDL-protection gate: activates only when slack is tight AND upward_rank is high
    median_slack = np.median(slack)
    ddl_protection_active = (
        (slack < median_slack) &
        (rank_percentile > PARAMS["ddl_protection_gate_threshold"])
    ).astype(float)
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline safety > bottleneck release > energy > critical path
    score = (
        norm_slack_penalty
        + PARAMS["successor_bottleneck_coupling"] * norm_successor_delay
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - critical_bonus
        + ddl_protection_active * norm_upward_rank
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
