def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: risk-aware, criticality-gated, efficiency-normalized, and starvation-robust.
    
    Key innovations:
    - Uses *signed* slack with smooth piecewise penalty: linear for positive slack, 
      quadratic exponential ramp for negative slack (stronger DDL risk emphasis).
    - Applies multiplicative criticality boost only to top-quartile upward_rank tasks,
      avoiding over-prioritization of all high-rank nodes.
    - Replaces raw energy with energy-efficiency ratio (J/s) — prioritizes low-energy-per-second
      tasks when durations are short, aligning with edge-VM constraints.
    - Introduces uncertainty-slack coupling: multiplies normalized uncertainty by max(0, -slack)
      to amplify scheduling urgency for high-risk, late-bound tasks.
    - Models ready_wait_time with saturating exponential benefit: early wait yields fast gain,
      diminishing returns beyond a soft threshold (prevents runaway starvation bias).
    - Normalizes all features using median absolute deviation (MAD) for outlier robustness,
      not mean-based scaling.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Safely cast inputs; avoid in-place mutation
    exec_t = np.asarray(min_exec_time, dtype=float)
    comm_t = np.asarray(min_comm_time, dtype=float)
    energy = np.asarray(min_incremental_energy, dtype=float)
    slk = np.asarray(slack, dtype=float)
    rank = np.asarray(upward_rank, dtype=float)
    work = np.asarray(remaining_work, dtype=float)
    wait = np.asarray(ready_wait_time, dtype=float)
    unc = np.asarray(uncertainty, dtype=float)
    
    # Robust normalization: MAD-based, epsilon-guarded
    def robust_normalize(x):
        x = np.clip(x, -1e12, 1e12)  # prevent overflow before stats
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = mad if mad > eps else np.mean(np.abs(x - med) + eps)
        return (x - med) / (scale + eps)
    
    # 1. Slack risk: piecewise — linear reward for slack > 0, exponential penalty for slack < 0
    slack_norm = robust_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        np.exp(PARAMS["slack_risk_exponent"] * np.abs(slack_norm)) - 1.0,
        -0.5 * slack_norm  # gentle reward for headroom
    )
    
    # 2. Criticality gating: boost only top quantile of upward_rank
    rank_quantile = PARAMS["rank_threshold_quantile"]
    rank_threshold = np.quantile(rank, rank_quantile)
    rank_boost_mask = (rank >= rank_threshold).astype(float)
    rank_score = -robust_normalize(rank) * (1.0 + PARAMS["criticality_boost"] * rank_boost_mask)
    
    # 3. Energy-efficiency ratio: J / (s) → lower is better (more efficient per time unit)
    # Avoid division by zero: use min_exec_time + min_comm_time + eps
    duration = exec_t + comm_t + eps
    energy_per_sec = energy / duration
    energy_eff_score = robust_normalize(energy_per_sec) * PARAMS["energy_efficiency_ratio"]
    
    # 4. Uncertainty-slack coupling: only activates under deadline pressure
    unc_norm = robust_normalize(unc)
    slack_pressure = np.maximum(-slk, 0.0)  # only when slack is negative
    # Normalize slack_pressure robustly too, then couple
    slack_pressure_norm = robust_normalize(slack_pressure + eps)
    coupling_term = unc_norm * slack_pressure_norm * PARAMS["uncertainty_slack_coupling"]
    
    # 5. Ready wait time: saturating exponential benefit (anti-starvation)
    # Benefit = 1 - exp(-wait_decay_rate * wait), bounded in [0, 1)
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -robust_normalize(wait_benefit)  # higher wait → lower score (better priority)
    
    # 6. Duration robustness: penalize tasks with high relative exec+comm variability
    # Use coefficient of variation (std/mean), but robustified
    dur_mean = np.mean(duration)
    dur_std = np.std(duration)
    cv_dur = dur_std / (dur_mean + eps)
    dur_var_score = robust_normalize(np.full(N, cv_dur)) * PARAMS["duration_sensitivity"]
    
    # Aggregate: all terms designed so smaller = better priority
    score = (
        slack_penalty
        + rank_score
        + energy_eff_score
        + coupling_term
        + wait_score
        + dur_var_score
    )
    
    # Final safeguard: replace NaN/inf with finite values, preserve shape (N,)
    score = np.nan_to_num(
        score,
        nan=PARAMS["nan_replacement"],
        posinf=PARAMS["inf_replacement"],
        neginf=PARAMS["neginf_replacement"]
    )
    
    return score
