def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust deadline enforcement with Parent 1's calibrated slack reward
       and improved fairness via wait-aware bottleneck mitigation.
    
    Key innovations:
      - Hybrid slack scoring: convex penalty (|slack|^p) for negative slack (p>2), linear reward for positive slack
        → preserves hard DDL violation dominance while retaining margin utility (unlike pure convex).
      - Wait-aware successor delay: scales bottleneck pressure by normalized wait time, preventing starvation-induced
        late scheduling of long-waiting tasks that block critical paths.
      - Critical-path bonus now includes wait-gated amplification: only high-rank *and* long-waiting tasks get boosted,
        reducing false positives from static rank outliers.
      - All normalizations use tunable IQR percentiles; no hardcoded thresholds or unbounded growth.
      - Removed fragile load-proxy and power-law wait terms — replaced by deterministic wait-aware coupling.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Hybrid slack scoring: convex penalty for violations, linear reward for margin
    # Avoids over-penalizing large negative slack while preserving strict violation dominance
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_penalty = np.where(
        slack < 0,
        np.power(slack_abs, PARAMS["slack_convex_penalty_exponent"]),
        PARAMS["slack_linear_reward_slope"] * slack
    )
    norm_slack_penalty = iqr_normalize(slack_penalty)
    
    # Duration-normalized energy efficiency: J / (s + eps)
    duration = min_exec_time + min_comm_time + eps
    energy_efficiency = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_efficiency)
    
    # Wait-aware successor-release delay: bottleneck pressure scaled by normalized wait time
    # Ensures long-waiting tasks blocking successors get prioritized even under moderate slack
    raw_successor_delay = remaining_work / (slack + eps)
    bounded_successor_delay = np.minimum(raw_successor_delay, PARAMS["successor_delay_upper_bound"])
    norm_wait = iqr_normalize(ready_wait_time + eps)
    wait_scaled_delay = bounded_successor_delay * (1.0 + norm_wait)
    norm_successor_delay = iqr_normalize(wait_scaled_delay)
    
    # Critical-path bonus: activated only for top percentile of upward_rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    
    # Wait-gated critical boost: only amplify rank for tasks that have waited significantly
    if N == 1:
        wait_percentile = np.array([1.0])
    else:
        sorted_wait = np.sort(ready_wait_time)
        wait_idx = np.searchsorted(sorted_wait, ready_wait_time, side='right')
        wait_percentile = wait_idx / (N + eps)
    wait_gate = np.where(wait_percentile >= PARAMS["wait_gate_percentile_threshold"], 1.0, 0.0)
    norm_upward_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * wait_gate * norm_upward_rank
    
    # DDL-protection gate: activates only when slack is tight AND upward_rank is high
    median_slack = np.median(slack)
    ddl_protection_active = (
        (slack < median_slack) &
        (rank_percentile > PARAMS["ddl_protection_gate_threshold"])
    ).astype(float)
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline safety > wait-aware bottlenecks > energy > gated critical path
    score = (
        norm_slack_penalty
        + PARAMS["successor_bottleneck_coupling"] * norm_successor_delay
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - critical_bonus
        + ddl_protection_active * norm_upward_rank
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
