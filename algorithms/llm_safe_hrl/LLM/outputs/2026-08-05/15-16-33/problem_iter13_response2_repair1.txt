def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with:
    - Replaced sigmoid urgency with smooth, bounded tanh-based slack mapping for stable gradient near zero.
    - Removed energy-duration ratio term: energy is now embedded in bottleneck via min_incremental_energy * (1 + urgency).
    - Unified bottleneck: (duration + comm) * upward_rank * (1 + tanh(slack)) * (1 + uncertainty) * (min_incremental_energy + eps)
      — directly couples risk-adjusted energy into critical-path pressure.
    - Anti-starvation via normalized ready_wait_time, scaled by inverse slack magnitude when slack < 0.
    - DDL-risk amplification now uses joint condition: tight slack AND high uncertainty → activates only under verified stress.
    - All normalizations use adaptive IQR; no fallback to min-max (IQR percentiles tuned to be robust across scenarios).
    - No inactive parameters: removed slack_sigmoid_* and wait_ramp_* per counterfactual evidence of redundancy.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization using IQR only (no min-max fallback — tuned percentiles ensure stability)
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Smooth, bounded urgency: tanh(-slack / scale) maps negative slack → +1, positive → ~0, zero → 0
    # Scale chosen to avoid overflow: use median abs slack or eps if all zero
    abs_slack = np.abs(slack)
    scale = np.median(abs_slack) if np.any(abs_slack > eps) else 1.0
    tanh_urgency = np.tanh(-slack / (scale + eps))  # range [-1, 1] → flip so negative slack → positive urgency
    # Clamp to [0, 1] for intuitive interpretation
    urgency = np.clip((tanh_urgency + 1.0) / 2.0, 0.0, 1.0)
    
    # Normalize urgency
    norm_urgency = adaptive_normalize(urgency)
    
    # Unified bottleneck pressure: integrates duration, communication, criticality, risk, and energy
    # (min_exec_time + min_comm_time) captures total latency exposure
    # upward_rank × (1 + urgency) emphasizes release of successors under deadline stress
    # (1 + uncertainty) weights pressure under capacity/bandwidth volatility
    # (min_incremental_energy + eps) embeds risk-adjusted energy cost directly into bottleneck
    duration = min_exec_time + min_comm_time
    bottleneck_pressure = (
        duration *
        upward_rank *
        (1.0 + urgency) *
        (1.0 + uncertainty) *
        (min_incremental_energy + eps)
    )
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Anti-starvation: linearly favor long-waiting tasks, but boost more when slack < 0 (DDL risk + starvation)
    # Use normalized ready_wait_time, then apply slack-conditioned multiplier
    norm_wait = adaptive_normalize(ready_wait_time)
    slack_penalty_mask = (slack < 0.0).astype(float)
    # Boost wait priority only when under DDL risk — avoids over-prioritizing idle tasks
    boosted_wait = norm_wait * (1.0 + slack_penalty_mask * PARAMS["wait_boost_under_ddl_risk"])
    
    # Joint DDL-risk gate: activated only when slack is tight AND uncertainty exceeds threshold
    # Uses max uncertainty across ready set as reference — ensures gate responds to system-wide risk level
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    ddl_risk_gate = (
        (slack < PARAMS["ddl_protection_gate_threshold"] * np.abs(np.median(slack) + eps)) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    # Amplify priority for high-upward-rank tasks under this joint condition
    ddl_risk_amplification = ddl_risk_gate * upward_rank
    
    # Final score: smaller = higher priority
    # Priority order: urgency (primary deadline signal) → bottleneck (critical path + energy) → DDL-risk amplification → fairness
    score = (
        norm_urgency +
        PARAMS["successor_bottleneck_coupling"] * norm_bottleneck +
        ddl_risk_amplification -
        boosted_wait  # subtract because higher wait → higher priority (lower score desired)
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
