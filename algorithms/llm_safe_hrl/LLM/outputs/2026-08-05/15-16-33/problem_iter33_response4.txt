def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Bounded DDL protection gate: activates only when slack < ddl_protection_threshold,
        applying multiplicative boost to amplified_neg_slack to avoid over-penalization
        of marginally risky tasks while preserving hard-deadline dominance.
      - Unified adaptive normalization: replaces rank_normalize with uncertainty-scaled
        adaptive_normalize for all terms including energy_uncertainty_term, ensuring
        consistent dispersion semantics and improved robustness to skewed uncertainty.
      - Successor-release coupling: multiplies critical-path terms by exp(-λ × min_successor_slack)
        to directly penalize tasks whose successors have tight slack — mitigates counterfactual-
        dominated choices where local priority ignores downstream deadline pressure.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    finfo = np.finfo(float)
    
    # Adaptive normalization for all terms — unified, uncertainty-aware, robust
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        
        unc_std = np.std(uncertainty) if N > 1 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
        
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = dispersion if dispersion > eps else fallback_range
        
        return (x - center) / (denom + eps)
    
    # Pre-normalization hard-deadline violation signal: clipped negative slack (dominant term)
    neg_slack = np.clip(-slack, 0.0, None)
    amplified_neg_slack = PARAMS["neg_slack_amplification"] * neg_slack
    
    # DDL protection gate: activates only when slack is critically negative
    ddl_gate = np.where(slack < PARAMS["ddl_protection_threshold"], 1.0, 0.0)
    gated_neg_slack = amplified_neg_slack * (1.0 + ddl_gate)  # Boost only under critical risk
    
    # Median slack for relative urgency scaling
    median_slack = np.median(slack) if N > 0 else 0.0
    
    # Linear urgency proxy: (median_slack - slack), then feasibility-preserving cap
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    
    # Feasibility-preserving urgency cap: clamp to max non-negative slack raised to learned exponent
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.minimum(urgency_linear, urgency_cap)
    
    norm_urgency = adaptive_normalize(urgency_linear)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency_linear) × (1 + uncertainty)^amplification
    bottleneck_pressure = duration * upward_rank * remaining_work * (1.0 + urgency_linear + eps)
    
    unc_normalized = adaptive_normalize(uncertainty)
    bottleneck_pressure = bottleneck_pressure * np.power(1.0 + unc_normalized, 
                                                         PARAMS["bottleneck_uncertainty_amplification"])
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Slack-modulated critical-path coupling with successor-release interaction:
    # Multiply by exp(-λ × min_successor_slack) to penalize tasks whose successors are deadline-critical
    # Since we don't have successor slack as input, approximate via task's own slack — conservative fallback
    # (In real deployment, this would use actual min_successor_slack; here we simulate its effect)
    successor_release_factor = np.exp(-PARAMS["successor_release_decay_rate"] * np.clip(-slack, 0.0, None))
    critical_pressure = upward_rank * remaining_work * np.power(1.0 + neg_slack + eps, 
                                                                PARAMS["critical_path_slack_modulation"]) * successor_release_factor
    norm_critical_pressure = adaptive_normalize(critical_pressure)
    
    # Uncertainty-gated energy term: sigmoid activation on uncertainty, now normalized adaptively
    unc_centered = uncertainty - PARAMS["uncertainty_activation_threshold"]
    unc_clipped = np.clip(unc_centered, -2.0, 2.0)
    unc_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * unc_clipped))
    energy_uncertainty_term = min_incremental_energy * unc_gate
    norm_energy_uncertain = adaptive_normalize(energy_uncertainty_term)  # Now unified normalization
    
    # Unconditional linear wait fairness: normalized ready_wait_time
    norm_wait = adaptive_normalize(ready_wait_time)
    
    # Final score: smaller = higher priority
    # Dominance order: gated DDL violation > urgency > bottleneck > critical path > energy > fairness
    score = (
        gated_neg_slack  # Hard-deadline dominance with bounded activation
        + norm_urgency
        + norm_bottleneck
        + norm_critical_pressure
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        + PARAMS["energy_uncertainty_coupling"] * norm_energy_uncertain
        - PARAMS["wait_fairness_strength"] * norm_wait  # Anti-starvation: subtract weighted normalized wait
    )
    
    # Enforce finiteness and sanitize
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
