def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule restoring hard deadline safety via binary DDL-protection gate,
    while enhancing robustness through *adaptive normalization scaling* — dynamically tightening
    IQR denominator when uncertainty is high, preventing over-normalization of risk-sensitive terms.
    
    Key structural improvement:
      - Adaptive IQR scaling: multiplies normalization denominator by (1 + norm_uncertainty),
        making normalized urgency/energy terms more discriminative under high uncertainty.
      - Restores binary DDL-protection gate (as per self-reflection) but now gated *after* adaptive scaling,
        ensuring urgency amplification only where both slack is tight AND uncertainty exceeds threshold.
      - All other successful elements retained: smooth sigmoid urgency, bottleneck coupling, arctan wait saturation,
        critical-path percentile gating, and finite-output safeguards.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust IQR-based normalization with adaptive scaling under uncertainty
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps_safe else (np.max(np.abs(x - center)) + eps_safe)
        # Adaptive scaling: tighten normalization under high uncertainty → preserve signal discriminability
        norm_uncertainty_local = iqr_normalize(uncertainty) if N > 1 else np.zeros_like(x)
        adaptive_denom = denom * (1.0 + np.abs(norm_uncertainty_local))
        return (x - center) / (adaptive_denom + eps_safe)

    # Sigmoid urgency gate: smooth, differentiable, centered at offset, steepness tunable
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered, 
                            -PARAMS["sigmoid_clip_bound"], 
                            PARAMS["sigmoid_clip_bound"])
    urgency_gate = 2.0 / (1.0 + np.exp(sigmoid_input)) - 1.0  # range [-1, 1]
    norm_urgency = iqr_normalize(urgency_gate)

    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps_safe)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)

    # Successor bottleneck coupling: prioritize tasks whose long duration blocks high-work/high-rank successors
    bottleneck_pressure = duration * (remaining_work + upward_rank + eps_safe)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)

    # Critical-path bonus: activated only for top percentile of upward_rank, applied to normalized rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps_safe)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank

    # Ready wait time: smooth saturation via arctan (bounded [0,1])
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps_safe)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = iqr_normalize(wait_saturation)

    # DDL-protection gate: binary activation only when slack is tight AND uncertainty is high
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)

    # Final score: smaller = higher priority
    # Urgency dominates; DDL-protection amplifies urgency only when both slack and uncertainty are critical
    score = (
        norm_urgency  # Primary deadline pressure
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks
        - norm_wait  # Reward recently arrived tasks (fairness)
        + ddl_protection_active * norm_urgency  # Hard-gated urgency amplification under DDL risk
    )

    # Ensure finite output, correct shape
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
