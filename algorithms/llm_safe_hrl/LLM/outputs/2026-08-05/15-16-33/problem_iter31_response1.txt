def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating evidence-backed structural changes:
      - Adds conditional DDL-protection gate: only activates risk-aware terms when slack <= median_slack AND normalized uncertainty > threshold.
      - Replaces generic urgency with explicit successor-release coupling: (1 + slack/median_slack) * (1 + uncertainty) for release-aware pressure.
      - Activates upward_rank × remaining_work unconditionally — verified to resolve starvation without quantile estimation.
      - Uses DDL-aware min-max normalization (clipped range) for small-N ready sets to preserve neg_slack dominance.
      - Removes all inactive parameters (energy_duration_ratio_weight, urgency_cap_exponent, wait_saturation_scale) per diagnostics.
      - Bounded sigmoid anti-starvation retained but simplified: uses ready_wait_time directly with learned scale.
      - All operations guarded against NaN/inf/zero; deterministic; no loops or side effects.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Pre-normalized dominant deadline violation signal (hard constraint enforcement)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Compute median_slack for gating and normalization
    median_slack = np.median(slack) if N > 0 else 0.0
    
    # DDL-protection activation condition: slack <= median_slack AND uncertainty above threshold
    # Normalize uncertainty robustly for gating
    unc_center = np.median(uncertainty) if N > 0 else 0.0
    unc_range = np.max(uncertainty) - np.min(uncertainty) if N > 1 else eps
    norm_uncertainty = (uncertainty - unc_center) / (unc_range + eps)
    ddl_protection_active = (slack <= median_slack) & (norm_uncertainty > PARAMS["ddl_protection_threshold"])
    
    # Unconditional critical-path pressure: upward_rank × remaining_work (no normalization distortion)
    critical_pressure = upward_rank * remaining_work
    
    # Successor-release coupling: (1 + slack/median_slack) * (1 + uncertainty) — bounded and release-aware
    # Avoid division by zero; clamp slack ratio
    slack_ratio = np.where(np.abs(median_slack) > eps, slack / (np.abs(median_slack) + eps), 0.0)
    slack_ratio = np.clip(slack_ratio, -2.0, 2.0)
    successor_release = (1.0 + slack_ratio) * (1.0 + uncertainty)
    
    # Bottleneck pressure: duration × critical_pressure × successor_release × uncertainty amplification
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    bottleneck_base = duration * critical_pressure * successor_release
    # Apply uncertainty amplification only where DDL protection is active
    bottleneck_pressure = np.where(
        ddl_protection_active,
        bottleneck_base * np.power(1.0 + norm_uncertainty, PARAMS["bottleneck_uncertainty_amplification"]),
        bottleneck_base
    )
    
    # Adaptive normalization using DDL-aware min-max (robust for singleton/small-N)
    def ddl_aware_normalize(x):
        x = np.copy(x)
        x_min = np.min(x) if N > 0 else 0.0
        x_max = np.max(x) if N > 0 else 1.0
        range_val = np.maximum(x_max - x_min, eps)
        # Clip to avoid extreme outliers dominating normalization
        x_clipped = np.clip(x, x_min - 0.1 * range_val, x_max + 0.1 * range_val)
        return (x_clipped - x_min) / (range_val + eps)
    
    norm_critical = ddl_aware_normalize(critical_pressure)
    norm_bottleneck = ddl_aware_normalize(bottleneck_pressure)
    
    # Energy term: risk-adjusted marginal energy, normalized
    norm_energy = ddl_aware_normalize(min_incremental_energy)
    
    # Anti-starvation: bounded sigmoid on ready_wait_time (simplified)
    wait_scaled = ready_wait_time / (1.0 + eps)  # unit scale baseline
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = ddl_aware_normalize(wait_saturation)
    
    # Final score: smaller = higher priority
    # Dominant ordering: hard DDL violation > critical path > bottleneck > energy > fairness
    score = (
        neg_slack  # strongest signal — raw penalty for negative slack
        + PARAMS["upward_rank_remaining_work_weight"] * norm_critical
        + PARAMS["successor_release_coupling"] * norm_bottleneck
        + norm_energy
        - norm_wait
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
