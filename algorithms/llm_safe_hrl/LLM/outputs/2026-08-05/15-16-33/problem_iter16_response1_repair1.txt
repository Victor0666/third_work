def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating:
    - Linear slack penalty (not sigmoid) with uncertainty coupling for sharper, numerically stable deadline risk.
    - Direct upward_rank × remaining_work interaction to capture critical-path computational bottleneck pressure.
    - Energy term shifted by offset to avoid over-penalizing low-energy tasks when slack is tight.
    - Removed anti-starvation ramp (replaced by robust wait-aware normalization in urgency term).
    - Unified bottleneck term now explicitly includes uncertainty-weighted slack penalty.
    - All normalizations use adaptive IQR→min-max fallback; no unbounded ops or hidden state.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max when dispersion is low
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR too small (low dispersion), else use IQR
        if iqr < eps:
            x_min, x_max = np.min(x), np.max(x)
            denom = x_max - x_min
            if denom < eps:
                return np.zeros_like(x)
            return (x - x_min) / (denom + eps)
        else:
            denom = iqr
            return (x - center) / (denom + eps)
    
    # Linear slack urgency: penalize negative slack more strongly, scaled by uncertainty
    # Avoids sigmoid instability near zero and provides interpretable gradient
    linear_slack_penalty = PARAMS["slack_linear_coeff"] * np.clip(-slack, 0.0, None)
    coupled_slack_penalty = linear_slack_penalty * (1.0 + PARAMS["uncertainty_slack_coupling"] * uncertainty)
    norm_slack_penalty = adaptive_normalize(coupled_slack_penalty)
    
    # Upward rank × remaining work interaction: captures downstream computational load on critical path
    # Strongly supported by counterfactual evidence for 'critical_path_or_successor_blocked'
    critical_load_pressure = upward_rank * (remaining_work + eps)
    norm_critical_load = adaptive_normalize(critical_load_pressure)
    
    # Energy term: normalized incremental energy, shifted to reduce dominance when slack is tight
    norm_energy = adaptive_normalize(min_incremental_energy)
    shifted_energy = norm_energy + PARAMS["energy_normalization_offset"]
    # Clamp to prevent excessive shift
    shifted_energy = np.clip(shifted_energy, 0.0, 2.0)
    
    # Duration-aware bottleneck: execution + comm time weighted by critical load and slack penalty
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    bottleneck_base = duration * (1.0 + norm_critical_load) * (1.0 + norm_slack_penalty)
    norm_bottleneck = adaptive_normalize(bottleneck_base)
    
    # Final score: smaller = higher priority
    # Priority hierarchy: deadline risk (linear + uncertainty-coupled) > critical load > bottleneck > energy
    score = (
        norm_slack_penalty  # Primary DDL risk signal — linear, stable, uncertainty-amplified
        + PARAMS["upward_remaining_interaction_weight"] * norm_critical_load  # Critical-path load pressure
        + norm_bottleneck  # Duration-weighted release pressure
        + shifted_energy  # Risk-adjusted energy, bias-corrected for fairness under tight deadlines
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
