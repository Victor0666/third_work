def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining structural strengths:
      - Joint DDL-risk protection gate from Parent 1 (slacks + uncertainty co-activation)
      - Adaptive exponential slack penalty from Parent 2 (linear for slack>0, exp for slack<=0)
      - Task-type–aware energy gating: only enable energy term for compute-bound tasks under low uncertainty
      - Bottleneck pressure with uncertainty amplification, gated by joint DDL-risk condition
      - Bounded power-law wait saturation (Parent 1 style) for robust anti-starvation
      - All normalizations use task-type–aware clipping (Parent 2 style) but applied to bottleneck & energy terms
      - Critical-path pressure remains uncoupled and unnormalized per Parent 1 guidance
      - No sigmoid; uses monotonic power-law for fairness boost
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Compute-bound vs I/O-bound ratio: robust against zero denominator
    exec_comm_sum = min_exec_time + min_comm_time
    exec_comm_sum_safe = np.maximum(exec_comm_sum, eps)
    exec_ratio = min_exec_time / exec_comm_sum_safe
    
    # Dominant hard-constraint signal: piecewise slack penalty
    linear_slack_penalty = np.clip(-slack, 0.0, None)
    exp_slack_penalty = np.where(
        slack <= 0.0,
        np.power(np.abs(slack) + eps, PARAMS["slack_sensitivity_exponent"]),
        0.0
    )
    neg_slack = linear_slack_penalty + exp_slack_penalty
    
    # Median-based DDL risk context for gating
    median_slack = np.median(slack) if N > 0 else 0.0
    high_risk_slack_condition = slack <= (median_slack * PARAMS["ddl_risk_gate_slack_ratio"] + eps)
    
    # Uncertainty normalization for gating: [0,1] scaled via robust min-max
    unc_min = np.min(uncertainty) if N > 0 else 0.0
    unc_max = np.max(uncertainty) if N > 0 else eps
    normalized_uncertainty = (uncertainty - unc_min) / (unc_max - unc_min + eps)
    high_uncertainty_condition = normalized_uncertainty > PARAMS["ddl_risk_gate_threshold"]
    
    # Joint DDL-risk protection gate: only active under both conditions
    ddl_risk_gate = np.where(high_risk_slack_condition & high_uncertainty_condition, 1.0, 0.0)
    
    # Critical-path pressure: upward_rank × remaining_work — uncoupled, unnormalized
    critical_path_pressure = upward_rank * remaining_work
    
    # Bottleneck pressure: duration × critical_pressure × (1+uncertainty)^amplification, gated
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    base_bottleneck = duration * critical_path_pressure * (1.0 + uncertainty)
    bottleneck_pressure = np.where(
        ddl_risk_gate,
        base_bottleneck * np.power(1.0 + normalized_uncertainty, PARAMS["bottleneck_uncertainty_amplification"]),
        base_bottleneck
    )
    
    # Energy term: enabled only for compute-bound tasks AND low uncertainty (dual predictability gate)
    energy_term = np.where(
        (exec_ratio >= PARAMS["compute_io_ratio_threshold"]) & 
        (normalized_uncertainty < PARAMS["low_uncertainty_gate_threshold"]),
        min_incremental_energy * (1.0 + PARAMS["energy_uncertainty_coupling"] * uncertainty),
        0.0
    )
    
    # Task-type–aware normalization: tighter clipping for compute-bound tasks
    def task_aware_normalize(x, is_compute_bound):
        x = np.copy(x)
        x_min = np.min(x) if N > 0 else 0.0
        x_max = np.max(x) if N > 0 else 1.0
        range_val = np.maximum(x_max - x_min, eps)
        # Tighter clipping for compute-bound (higher precision needed), wider for I/O-bound
        clip_factor = np.where(is_compute_bound, 0.5, 1.0) * 0.12
        clip_offset = clip_factor * range_val
        x_clipped = np.clip(x, x_min - clip_offset, x_max + clip_offset)
        return (x_clipped - x_min) / (range_val + eps)
    
    is_compute_bound = exec_ratio >= PARAMS["compute_io_ratio_threshold"]
    norm_critical = task_aware_normalize(critical_path_pressure, is_compute_bound)
    norm_bottleneck = task_aware_normalize(bottleneck_pressure, is_compute_bound)
    norm_energy = task_aware_normalize(energy_term, is_compute_bound)
    
    # Bounded power-law wait saturation: smooth, monotonic anti-starvation boost
    wait_mean = np.mean(ready_wait_time) if N > 0 else eps
    wait_scaled = np.clip(ready_wait_time / (wait_mean + eps), 0.0, 2.0)
    wait_saturation = np.power(wait_scaled + eps, PARAMS["wait_saturation_exponent"])
    norm_wait = task_aware_normalize(wait_saturation, is_compute_bound)
    
    # Final score: smaller = higher priority
    # Priority order: neg_slack >> gated bottleneck >> critical path >> energy (if safe & compute-bound) >> fairness
    score = (
        neg_slack
        + ddl_risk_gate * PARAMS["upward_rank_remaining_work_weight"] * norm_critical
        + PARAMS["upward_rank_remaining_work_weight"] * norm_bottleneck
        + 0.68 * norm_energy
        - norm_wait
    )
    
    # Enforce finiteness and avoid NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
