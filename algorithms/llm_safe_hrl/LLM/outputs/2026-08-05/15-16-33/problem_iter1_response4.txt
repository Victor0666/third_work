import numpy as np


PARAMETER_SCHEMA = {
    "schema_version": 1,
    "parameters": [
        {
            "name": "epsilon",
            "initial_value": 1e-6,
            "lower_bound": 1e-9,
            "upper_bound": 1e-3,

            "lower_bound": 1.0,

            "semantic_description": "Nonlinear amplification of negative slack (deadline violation risk).",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "criticality_boost",
            "initial_value": 1.5,
            "lower_bound": 0.5,
            "upper_bound": 5.0,
            "semantic_description": "Multiplicative gain applied to high upward-rank tasks when slack is tight.",
            "parameter_type": "float",
            "transform": "identity"

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: deadline-risk-gated criticality + energy-aware density + anti-starvation decay.
    
    Key innovations:
    - Uses piecewise slack handling: strong nonlinear penalty for negative slack (via exponent),
      relaxed duration sensitivity only when slack > 0.
    - Criticality boost activates only under tight slack: multiplies upward_rank to protect DAG bottlenecks.
    - Work-density factor combines remaining_work and duration to favor high-throughput-density tasks.
    - Wait priority uses exponential decay: long-waiting tasks get boosted, but saturation is bounded.
    - Uncertainty-slack coupling penalizes high-uncertainty tasks *only* when slack is low — avoids over-penalizing
      robust tasks with ample margin.
    - All features are robustly normalized using median absolute deviation (more outlier-resistant than mean).
    - No unbounded ops: all divisions guarded, exponents clipped, logs avoided.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Ensure finite inputs and avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust normalization: use median absolute deviation (MAD) for stability against outliers
    def robust_normalize(x):
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        center = np.median(x)
        diff = x - center
        mad = np.median(np.abs(diff))
        scale = mad if mad > eps else eps
        return diff / scale
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    
    # --- Deadline risk modeling ---
    # Strong nonlinear penalty for negative slack: (max(0, -slack))^exponent → always ≥ 0
    # Then map to priority: higher penalty ⇒ lower priority ⇒ add positive term
    slack_risk_penalty = np.power(np.maximum(0.0, -slack), PARAMS["slack_risk_exponent"])
    # Normalize the raw penalty for scale consistency
    slack_risk_penalty = robust_normalize(slack_risk_penalty)
    
    # --- Criticality boost: activate only when slack is tight (normalized slack < 0.2 threshold)
    # Threshold is fixed (not parametric) for structural simplicity and stability
    tight_slack_mask = (norm_slack < 0.2).astype(float)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_boost"] * tight_slack_mask)
    
    # --- Work density: remaining_work / (exec + comm + eps), then normalized
    duration_sum = min_exec_time + min_comm_time + eps
    work_density = remaining_work / duration_sum
    norm_work_density = robust_normalize(work_density)
    
    # --- Anti-starvation: exponential wait boost, bounded and normalized
    # exp(-rate * t) decays; we want *higher* priority for larger t → use (1 - exp(-rate*t)) ∈ [0,1)
    wait_boost_raw = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    wait_boost = robust_normalize(wait_boost_raw)
    
    # --- Uncertainty-slack coupling: only penalize uncertainty when slack is low
    # Use smooth gate: sigmoid of (-norm_slack) → ~1 when slack is very negative, ~0 when slack > 0
    slack_gate = 1.0 / (1.0 + np.exp(norm_slack))  # ≈1 for negative slack, ≈0.5 at 0, ≈0 for large positive
    coupled_uncert = norm_uncert * slack_gate * PARAMS["uncertainty_slack_coupling"]
    
    # --- Duration sensitivity only when slack > 0 (feasible regime)
    positive_slack_mask = (slack > 0.0).astype(float)
    duration_contribution = norm_duration * positive_slack_mask * PARAMS["duration_sensitivity"]
    
    # --- Energy efficiency bias applies only when slack >= 0 (no violation risk)
    energy_contribution = norm_energy * (1.0 - positive_slack_mask) * 2.0  # stronger penalty under risk
    energy_contribution += norm_energy * positive_slack_mask * PARAMS["energy_efficiency_bias"]
    
    # Final score: smaller = higher priority
    # Components ordered by lexicographic importance: deadline risk > criticality > energy > density > wait > coupling
    score = (
        slack_risk_penalty * 3.0  # dominant risk term
        + boosted_rank * (-1.2)  # promote critical paths (negative weight → higher priority)
        + energy_contribution
        + (-PARAMS["work_density_factor"]) * norm_work_density  # prefer high work density
        + (-0.8) * wait_boost  # anti-starvation: negative weight → longer wait ⇒ lower score
        + duration_contribution
        + coupled_uncert
    )
    
    # Final safeguard: ensure finite, shape-(N,) output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(N)
