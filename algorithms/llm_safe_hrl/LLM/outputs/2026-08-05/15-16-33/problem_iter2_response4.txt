def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's deadline-aware piecewise slack handling and 
    critical-path gating with Parent 1's robust quantile-based normalization and starvation mitigation.
    
    Key improvements:
      - Unified robust normalization using tunable quantiles (not percentiles) for consistency across features
      - Criticality gate now combines percentile rank *and* slack tightness (joint condition), reducing false positives
      - Added duration-normalized energy efficiency *and* uncertainty-coupled energy penalty for risk-aware efficiency
      - Replaced arctan wait saturation with exponential decay (Parent 2) but gated by slack to avoid boosting late tasks
      - All operations safeguarded with eps-safe denominators and finite clamping
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)
    N = len(slack)

    # Ensure float arrays and avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust quantile-based normalization (Parent 1 style, tunable)
    def robust_normalize(x):
        x = np.copy(x)
        q1 = np.quantile(x, PARAMS["quantile_low"])
        q3 = np.quantile(x, PARAMS["quantile_high"])
        iqr = q3 - q1
        center = np.median(x)
        scale = iqr if iqr > eps_safe else eps_safe
        return (x - center) / (scale + eps_safe)

    # Duration: exec + comm, normalized robustly
    duration = min_exec_time + min_comm_time
    norm_duration = robust_normalize(duration)

    # Slack: piecewise — strong penalty for negative, gentle reward for positive (Parent 2)
    slack_norm = robust_normalize(slack)
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_risk_penalty"] * (-slack_norm),
        -PARAMS["slack_urgency_gain"] * slack_norm
    )

    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration_safe = np.maximum(duration, eps_safe)
    energy_per_duration = min_incremental_energy / duration_safe
    norm_energy_eff = robust_normalize(energy_per_duration)

    # Uncertainty-coupled energy penalty: penalize high-energy tasks under high uncertainty
    unc_norm = robust_normalize(uncertainty)
    energy_uncertainty_penalty = unc_norm * norm_energy_eff * PARAMS["uncertainty_slack_coupling"]

    # Critical-path bonus: activated only when both rank is top-percentile AND slack is tight (< median)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps_safe)
    median_slack = np.median(slack)
    critical_gate = (
        (rank_percentile >= PARAMS["critical_rank_threshold"]) &
        (slack < median_slack)
    ).astype(float)
    norm_rank = robust_normalize(upward_rank + PARAMS["rank_normalization_offset"])
    critical_bonus = PARAMS["critical_path_bonus"] * norm_rank * critical_gate

    # Waiting-time boost: exponential decay, but only active when slack is non-negative (avoid boosting overdue tasks)
    wait_boost = np.where(
        slack >= 0,
        1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time),
        0.0
    )
    norm_wait = robust_normalize(wait_boost)

    # Uncertainty-slack coupling: amplify urgency only when slack < 0 AND uncertainty is above median
    median_unc = np.median(uncertainty)
    coupled_urgency = np.where(
        (slack < 0) & (uncertainty > median_unc),
        unc_norm * (-slack_norm) * PARAMS["uncertainty_slack_coupling"],
        0.0
    )

    # Duration-efficiency ratio: balance duration cost vs energy cost
    dur_eff_ratio = PARAMS["duration_efficiency_ratio"] * norm_duration

    # Final score: smaller = higher priority
    # Deadline safety dominates (slack_penalty), then critical path, then efficiency, then fairness
    score = (
        slack_penalty
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        + energy_uncertainty_penalty
        - critical_bonus
        - norm_wait
        + coupled_urgency
        + dur_eff_ratio
    )

    # Ensure finite output, no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
