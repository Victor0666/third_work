def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Deadline-feasibility–aware energy gating: enables energy term only when slack >= 0 AND uncertainty < median(uncertainty) * threshold — avoids brittle fixed thresholds.
      - Unified bottleneck pressure: duration × upward_rank × (1 + uncertainty), simplified and stabilized per reflection.
      - Robust wait-saturation via bounded power-law on relative wait time (vs mean), clipped to [0,2] before exponentiation.
      - All normalization uses task-agnostic median-based robust scaling instead of min/max to reduce outlier sensitivity.
      - Critical-path pressure remains raw (unnormalized) to preserve scale integrity across heterogeneous workflows.
      - Final score enforces strict priority ordering: DDL violation > bottleneck > energy (if feasible) > fairness.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust slack penalty: linear for positive slack, exponential for violations
    linear_slack_penalty = np.clip(-slack, 0.0, None)
    exp_slack_penalty = np.where(
        slack <= 0.0,
        np.power(np.abs(slack) + eps, PARAMS["slack_sensitivity_exponent"]),
        0.0
    )
    neg_slack = linear_slack_penalty + exp_slack_penalty
    
    # Deadline-feasibility gate: energy optimization only when slack >= 0 AND uncertainty is low relative to population
    unc_median = np.median(uncertainty) if N > 0 else eps
    ddl_feasibility_gate = (slack >= 0.0) & (uncertainty < unc_median * PARAMS["ddl_feasibility_gate_threshold"])
    energy_term = np.where(ddl_feasibility_gate, min_incremental_energy, 0.0)
    
    # Unified bottleneck pressure: duration × upward_rank × (1 + uncertainty)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    bottleneck_pressure = duration * upward_rank * (1.0 + uncertainty)
    
    # Critical-path pressure remains raw (no normalization) to preserve physical scale and avoid distortion
    critical_path_pressure = upward_rank * remaining_work
    
    # Anti-starvation: bounded power-law on relative wait time
    wait_mean = np.mean(ready_wait_time) if N > 0 else eps
    wait_ratio = np.clip(ready_wait_time / (wait_mean + eps), 0.0, 2.0)
    wait_saturation = np.power(wait_ratio + eps, PARAMS["wait_saturation_exponent"])
    
    # Robust normalization using median absolute deviation (MAD)-inspired scaling for stability
    def robust_normalize(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros_like(x)
        x_med = np.median(x)
        x_dev = np.abs(x - x_med)
        mad = np.median(x_dev) if np.any(x_dev) else eps
        # Scale to [0,1] via soft sigmoid-like mapping: avoids extreme outliers dominating
        scaled = (x - x_med) / (mad + eps)
        normalized = 1.0 / (1.0 + np.exp(-np.clip(scaled, -2.0, 2.0)))
        return normalized
    
    norm_bottleneck = robust_normalize(bottleneck_pressure)
    norm_energy = robust_normalize(energy_term)
    norm_wait = robust_normalize(wait_saturation)
    
    # Final score: smaller = higher priority
    # Priority hierarchy: DDL violation > bottleneck pressure > energy (if feasible) > fairness (anti-starvation)
    score = (
        neg_slack  # Hard constraint enforcement — highest priority
        + PARAMS["upward_rank_remaining_work_weight"] * critical_path_pressure  # Raw scale preserves dominance
        + PARAMS["upward_rank_remaining_work_weight"] * norm_bottleneck
        + PARAMS["energy_priority_weight"] * norm_energy
        - norm_wait
    )
    
    # Enforce finiteness and shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
