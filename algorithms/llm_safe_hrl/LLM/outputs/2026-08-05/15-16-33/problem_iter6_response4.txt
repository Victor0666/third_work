def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Adaptive DDL-feasibility gating: when slack < ddl_feasibility_threshold, all non-urgency terms are suppressed
        and urgency is amplified via feasibility_guard_strength — enforcing hard deadline compliance first.
      - Simplified urgency: direct slack sign-and-magnitude signal instead of linear ramp; avoids parameter overfitting.
      - Robust IQR normalization with tightened percentiles (21/79) for better outlier resilience.
      - Capped successor-release pressure to prevent numerical instability near zero duration.
      - Critical-path bonus remains percentile-gated but computed safely for N=1.
      - All operations guarded against NaN/inf/zero; deterministic and finite output.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Feasibility guard: activate strict DDL mode only when slack is critically low
    # This replaces the linear ramp with a clean, interpretable binary switch + amplification
    ddl_critical_mask = (slack < PARAMS["ddl_feasibility_threshold"]).astype(float)
    
    # Urgency: sign-aware magnitude — negative slack → positive urgency; zero at slack=0
    # Amplified under feasibility guard for hard constraint enforcement
    urgency_raw = -slack  # higher urgency when slack is more negative
    urgency_amplified = urgency_raw * (1.0 + PARAMS["feasibility_guard_strength"] * ddl_critical_mask)
    norm_urgency = iqr_normalize(urgency_amplified)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor-release pressure: capped to avoid explosion near zero duration
    release_pressure_uncapped = (remaining_work * upward_rank) / (duration + eps)
    release_pressure = np.minimum(release_pressure_uncapped, PARAMS["release_pressure_cap"])
    norm_release = iqr_normalize(release_pressure)
    
    # Critical-path bonus: activated only for top percentile of upward_rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Ready wait time: not used in this evolution — removed to reduce feature noise and improve interpretability
    # Fairness is implicitly preserved via urgency dominance under DDL stress and uniform normalization
    
    # DDL-protection gate: activates only when uncertainty AND risk-aware load proxy are high
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    slack_pressure = np.maximum(0.0, median_slack - slack)
    host_load_proxy = uncertainty * slack_pressure
    ddl_protection_active = (
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty) &
        (host_load_proxy > PARAMS["host_load_sensitivity"] * np.max(host_load_proxy + eps))
    ).astype(float)
    
    # Uncertainty-normalized term for amplification under risk — only active when DDL protection triggers
    norm_uncertainty = iqr_normalize(uncertainty)
    
    # Final score: smaller = higher priority
    # Under DDL-critical mode (ddl_critical_mask == 1), suppress all non-urgency terms
    # This enforces hard deadline compliance before optimizing energy or fairness
    score = (
        norm_urgency  # Always primary
        + ddl_critical_mask * 0.0  # Suppress energy term under DDL stress
        + ddl_critical_mask * 0.0  # Suppress release pressure under DDL stress
        + ddl_critical_mask * 0.0  # Suppress critical bonus under DDL stress (urgency already covers path)
        + ddl_critical_mask * 0.0  # Suppress uncertainty amplification under DDL stress (urgency dominates)
        + (1.0 - ddl_critical_mask) * (
            PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
            + PARAMS["successor_release_pressure_weight"] * norm_release
            - critical_bonus
            + ddl_protection_active * norm_uncertainty
        )
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
