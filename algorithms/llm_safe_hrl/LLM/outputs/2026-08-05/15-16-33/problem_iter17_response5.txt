def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's robust DDL-risk gate and fairness with Parent 1's additive uncertainty coupling and urgency clamping.
    
    Key structural improvements:
      - Retains Parent 2's clean separation: urgency (deadline safety) vs bottleneck (critical-path + energy) vs fairness (anti-starvation)
      - Integrates Parent 1's *additive* uncertainty coupling into bottleneck_pressure to avoid explosion under high uncertainty
      - Uses Parent 1's `urgency_clamp_lower` on raw tanh output before affine scaling to [0,1], improving stability for marginally negative slack
      - Keeps unconditional fairness term (`-wait_fairness_weight * norm_wait`) for starvation prevention across all regimes
      - Joint DDL-risk gate remains `(slack < 0) & (uncertainty > median_uncertainty * threshold)` — simple, reliable, quantile-free
      - All operations are finite, deterministic, and numerically safe; no hidden constants beyond {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization using IQR only (tuned percentiles ensure stability)
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Bounded tanh urgency: maps slack → [-1,1], then clamped and scaled to [0,1]
    # Clamp raw tanh before affine transform to prevent pathological penalties on small negative slack
    abs_slack = np.abs(slack)
    scale = np.median(abs_slack) if np.any(abs_slack > eps) else 1.0
    tanh_urgency = np.tanh(-slack / (scale * PARAMS["urgency_scale"] + eps))
    clamped_urgency = np.clip(tanh_urgency, PARAMS["urgency_clamp_lower"], 1.0)
    urgency = (clamped_urgency + 1.0) / 2.0  # map to [0,1]
    norm_urgency = adaptive_normalize(urgency)
    
    # Duration includes both compute and communication
    duration = min_exec_time + min_comm_time
    
    # Unified bottleneck pressure: critical-path + risk-adjusted energy, with *additive* uncertainty coupling
    # Avoids multiplicative explosion: `+ bottleneck_uncertainty_coupling * uncertainty * duration`
    bottleneck_pressure = (
        duration *
        upward_rank *
        (min_incremental_energy + eps) +
        PARAMS["bottleneck_uncertainty_coupling"] * uncertainty * duration
    )
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Anti-starvation: normalized ready_wait_time, subtracted (higher wait → lower score → higher priority)
    norm_wait = adaptive_normalize(ready_wait_time)
    fairness_term = -PARAMS["wait_fairness_weight"] * norm_wait  # subtractive fairness
    
    # Joint DDL-risk gate: activated only when slack < 0 AND uncertainty exceeds scaled median
    median_uncertainty = np.median(uncertainty) if N > 0 else eps
    ddl_risk_gate = (
        (slack < 0.0) &
        (uncertainty > PARAMS["ddl_risk_activation_threshold"] * median_uncertainty)
    ).astype(float)
    # Amplify urgency signal under verified stress — preserves separation of concerns
    stress_amplified_urgency = norm_urgency * (1.0 + ddl_risk_gate)
    
    # Final score: smaller = higher priority
    # Priority order: urgency (primary deadline signal, stress-amplified) → bottleneck (critical path + energy + additive uncertainty) → fairness (anti-starvation)
    score = (
        stress_amplified_urgency +
        PARAMS["successor_bottleneck_coupling"] * norm_bottleneck +
        fairness_term
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
