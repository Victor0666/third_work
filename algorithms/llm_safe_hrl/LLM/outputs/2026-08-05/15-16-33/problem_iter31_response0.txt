def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with structural improvements guided by counterfactual evidence:
      - Introduces conditional DDL protection gate: activates only when slack <= median_slack AND uncertainty > threshold.
      - Replaces generic bottleneck pressure with explicit upward_rank × remaining_work interaction, unconditionally activated (no gating).
      - Adds successor-release coupling: modulates urgency based on how much successor release time depends on current task's completion.
      - Uses bounded sigmoid wait saturation with learned exponent instead of linear scaling.
      - Removes energy-duration ratio term — inactive in elite diagnostics; replaced by direct min_incremental_energy normalization.
      - All normalization uses DDL-aware min-max (clipped range) for robustness under low-N or singleton ready sets.
      - Dominant term remains pre-normalized neg_slack to preserve hard-deadline dominance.
      - No branching beyond vectorized conditionals; all operations finite and deterministic.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Pre-normalization hard-deadline violation signal: clipped negative slack (dominant term)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # DDL protection gate: active only when task is at risk AND uncertainty is high
    median_slack = np.median(slack) if N > 0 else 0.0
    unc_normalized = (uncertainty - np.min(uncertainty + eps)) / (np.ptp(uncertainty) + eps) if N > 1 else np.zeros_like(uncertainty)
    ddl_protection_active = (slack <= median_slack) & (unc_normalized > PARAMS["ddl_protection_threshold"])
    
    # Critical-path pressure: upward_rank × remaining_work — unconditionally activated per evidence
    critical_pressure = upward_rank * remaining_work
    
    # Successor release coupling: scales urgency by how much successor release depends on this task
    # Approximated via normalized slack gradient sensitivity: higher |d(slack)/d(t)| → stronger coupling
    # Use local dispersion of slack as proxy for release sensitivity
    slack_range = np.ptp(slack) if N > 1 else eps
    successor_sensitivity = np.clip(slack_range / (np.max(np.abs(slack)) + eps), 0.0, 1.0)
    successor_release_boost = PARAMS["successor_release_coupling"] * successor_sensitivity * (1.0 + uncertainty)
    
    # Urgency: linear slack deficit, capped and scaled
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    urgency = urgency_linear + successor_release_boost
    
    # Adaptive min-max normalization (DDL-aware): preserves neg_slack dominance under low variance
    def ddl_aware_normalize(x):
        x = np.copy(x)
        x_min = np.min(x)
        x_max = np.max(x)
        range_val = x_max - x_min
        denom = np.maximum(range_val, eps)
        return (x - x_min) / (denom + eps)
    
    norm_urgency = ddl_aware_normalize(urgency)
    norm_critical = ddl_aware_normalize(critical_pressure)
    
    # Energy term: direct normalization of min_incremental_energy (no duration ratio)
    norm_energy = ddl_aware_normalize(min_incremental_energy)
    
    # Wait saturation: bounded sigmoid with learned exponent
    wait_scaled = np.clip(ready_wait_time / (np.max(ready_wait_time + eps)), 0.0, 1.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-((wait_scaled * 4.0 - 2.0) ** PARAMS["wait_saturation_exponent"])))
    norm_wait = ddl_aware_normalize(wait_saturation)
    
    # Uncertainty-amplified critical pressure (only under DDL protection)
    amplified_critical = critical_pressure * np.power(1.0 + uncertainty, PARAMS["bottleneck_uncertainty_amplification"])
    norm_amplified_critical = ddl_aware_normalize(amplified_critical)
    
    # Final score: smaller = higher priority
    # Dominant ordering: hard DDL violation > urgency > critical path > energy > fairness
    score = (
        neg_slack  # Raw penalty — strongest signal for deadline violation
        + norm_urgency
        + PARAMS["upward_rank_remaining_work_weight"] * norm_critical
        + np.where(ddl_protection_active, norm_amplified_critical, 0.0)
        + norm_energy
        - norm_wait
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
