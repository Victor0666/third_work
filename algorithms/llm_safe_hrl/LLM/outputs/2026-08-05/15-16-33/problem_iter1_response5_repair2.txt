def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Deadline-risk-gated criticality + uncertainty-aware energy efficiency.
    
    Uses exactly 12 parameters; no hidden literals beyond -2..2; all PARAMS used.
    Robust normalization via median/IQR; finite output with np.nan_to_num.
    """
    eps = PARAMS["epsilon"]
    n = len(min_exec_time)
    
    # Safe cast & copy
    exec_t = np.asarray(min_exec_time, dtype=float).copy()
    comm_t = np.asarray(min_comm_time, dtype=float).copy()
    energy = np.asarray(min_incremental_energy, dtype=float).copy()
    slk = np.asarray(slack, dtype=float).copy()
    rank = np.asarray(upward_rank, dtype=float).copy()
    work = np.asarray(remaining_work, dtype=float).copy()
    wait = np.asarray(ready_wait_time, dtype=float).copy()
    unc = np.asarray(uncertainty, dtype=float).copy()
    
    # Robust normalization: median + IQR
    def robust_normalize(x):
        x = np.where(np.isfinite(x), x, 0.0)
        med = np.median(x)
        if n > 1:
            q75, q25 = np.percentile(x, [75, 25], method='midpoint')
            iqr = q75 - q25 + eps
        else:
            iqr = eps
        return (x - med) / iqr
    
    # Normalize base features
    norm_slk = robust_normalize(slk)
    norm_energy = robust_normalize(energy)
    norm_rank = robust_normalize(rank ** PARAMS["rank_normalization_power"])
    norm_work = robust_normalize(work)
    norm_wait = robust_normalize(wait)
    norm_unc = robust_normalize(unc)
    
    # Slack risk penalty: nonlinear on negative slack
    slack_penalty_raw = np.maximum(0.0, -slk) ** PARAMS["slack_risk_exponent"]
    norm_slack_penalty = robust_normalize(slack_penalty_raw)
    
    # Coupled penalty: only when slack < 0 AND uncertainty above median
    median_unc = np.median(unc + eps) if n > 1 else 0.0
    coupled_penalty = np.where(
        (slk < 0) & (unc > median_unc),
        norm_slack_penalty * PARAMS["uncertainty_slack_coupling"],
        0.0
    )
    
    # Conditional rank boost: higher weight when slack <= 0
    rank_boost = np.where(
        slk <= 0,
        norm_rank * PARAMS["criticality_boost"],
        norm_rank * PARAMS["non_critical_rank_weight"]
    )
    
    # Energy weight: reduced under deadline pressure
    energy_weight = np.where(
        slk < 0,
        PARAMS["energy_efficiency_bias"] * 0.3,
        PARAMS["energy_efficiency_bias"]
    )
    
    # Wait-time anti-starvation: exponential saturation
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * (wait + eps))
    norm_wait_boost = robust_normalize(wait_boost)
    
    # Duration contribution only in non-critical regime
    duration = exec_t + comm_t
    norm_duration = robust_normalize(duration)
    duration_contrib = np.where(
        slk > 0,
        norm_duration * PARAMS["duration_sensitivity"],
        0.0
    )
    
    # Final score: smaller = better
    score = (
        # Primary: Slack risk dominates
        norm_slack_penalty * PARAMS["slack_penalty_weight"]
        + coupled_penalty * PARAMS["coupled_penalty_weight"]
        
        # Critical path importance
        - rank_boost
        
        # Energy efficiency
        + norm_energy * energy_weight
        
        # Duration (only when slack permits)
        + duration_contrib
        
        # Anti-starvation
        - norm_wait_boost * PARAMS["wait_boost_weight"]
        
        # Remaining work (larger → prioritize earlier)
        - norm_work
    )
    
    # Ensure finite deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
