def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL risk containment, critical-path robustness,
    and starvation-aware fairness. Uses piecewise slack gating, uncertainty-modulated
    criticality, and bounded nonlinear penalties instead of linear weighting."""
    epsilon = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps = max(finfo.tiny, epsilon)

    # Convert all inputs to float64 arrays; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # --- Robust normalization: use IQR-based scaling for outlier resilience ---
    def robust_normalize(x):
        x = np.copy(x)
        q1, q3 = np.quantile(x, [0.25, 0.75])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr

    # --- Duration & energy features ---
    total_duration = min_exec_time + min_comm_time
    duration_norm = robust_normalize(total_duration)
    energy_norm = robust_normalize(min_incremental_energy)
    rank_norm = robust_normalize(upward_rank)
    work_norm = robust_normalize(remaining_work)
    wait_norm = robust_normalize(ready_wait_time)
    uncertainty_norm = robust_normalize(uncertainty)
    slack_norm = robust_normalize(slack)

    # --- Slack-driven risk gating: strong penalty for negative slack, smooth transition ---
    # Use soft sign + power to amplify urgency without discontinuity
    slack_sign = np.sign(slack)
    slack_abs = np.abs(slack)
    # Negative slack → exponential urgency; non-negative → mild decay
    slack_risk = np.where(
        slack < 0,
        -np.power(slack_abs + eps, PARAMS["slack_risk_exponent"]),
        -np.log1p(slack_abs + eps) * 0.5
    )
    # Normalize risk signal to prevent dominance
    slack_risk = robust_normalize(slack_risk)

    # --- Critical-path boost: only activate when slack is tight (negative or near-zero) ---
    # Boost upward_rank only where deadline pressure exists
    tight_slack_mask = (slack <= eps)
    critical_boost = np.where(
        tight_slack_mask,
        rank_norm * PARAMS["critical_path_boost"],
        rank_norm * 0.3  # baseline importance
    )

    # --- Uncertainty-slack interaction: high uncertainty + tight slack → higher priority ---
    # This models risk compounding: uncertain execution near deadline is most dangerous
    slack_pressure = np.maximum(-slack_norm, 0.0)
    uncertainty_slack_coupling = (
        uncertainty_norm * slack_pressure * PARAMS["uncertainty_slack_interaction"]
    )

    # --- Anti-starvation: piecewise activation after threshold ---
    wait_boost = np.where(
        ready_wait_time >= PARAMS["wait_decay_threshold"],
        (ready_wait_time - PARAMS["wait_decay_threshold"]) * 0.3,
        0.0
    )
    wait_boost = robust_normalize(wait_boost)

    # --- Robustness penalty: penalize high uncertainty when slack is non-negative ---
    # Avoid over-prioritizing risky tasks when there's buffer
    robustness_penalty = np.where(
        slack >= 0,
        np.power(uncertainty_norm, PARAMS["robustness_penalty_power"]),
        0.0
    )

    # --- Duration-efficiency tradeoff: reward low duration *and* low energy per unit work ---
    # Compute energy-per-work ratio (avoid division by zero)
    work_safe = np.maximum(remaining_work, eps)
    energy_per_work = min_incremental_energy / work_safe
    energy_per_work_norm = robust_normalize(energy_per_work)

    # Combine duration and energy-efficiency signals
    duration_efficiency_score = (
        duration_norm
        + PARAMS["energy_efficiency_scale"] * energy_per_work_norm
    )

    # --- Final score: smaller = higher priority ---
    # Prioritize: (1) deadline risk, (2) critical path under pressure, (3) uncertainty coupling,
    #             (4) anti-starvation, (5) duration+efficiency, (6) robustness penalty
    score = (
        slack_risk * 3.0  # strongest weight for risk
        + critical_boost * 2.0
        + uncertainty_slack_coupling * 2.5
        + wait_boost * 1.2
        + duration_efficiency_score * PARAMS["duration_uncertainty_ratio"]
        + robustness_penalty * 0.8
    )

    # Ensure finite output and correct shape
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.reshape(-1)
