def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule synthesizing Parent 2's smooth urgency with Parent 1's robust feasibility gating,
    enhanced by uncertainty-aware amplification and tightened IQR normalization.
    
    Key structural improvements:
      - Replaces binary DDL-critical mask with *adaptive sigmoid urgency* AND *conditional uncertainty amplification*
        — both activated only when slack is tight relative to median AND uncertainty exceeds threshold.
      - Introduces `uncertainty_amplification_weight`: decouples uncertainty's role from urgency signal,
        allowing independent tuning of risk-aware penalty under DDL stress.
      - Tightens IQR percentiles (23/77) for improved outlier resilience in heterogeneous edge-cloud workloads.
      - Uses bottleneck pressure as `duration * (remaining_work + upward_rank + eps)` to avoid zero collapse.
      - Critical-path bonus remains percentile-gated but computed safely for N=1 via rank-indexing.
      - All operations guarded against NaN/inf/zero; deterministic and finite output.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Sigmoid urgency gate: smooth, differentiable, centered at offset, steepness tunable
    # Maps slack → [-1, 1]: negative slack → +1 (high urgency), positive slack → -1 (low urgency)
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    # Avoid overflow in exp by clamping input using declared parameter
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered, 
                            -PARAMS["sigmoid_clip_bound"], 
                            PARAMS["sigmoid_clip_bound"])
    urgency_gate = 2.0 / (1.0 + np.exp(sigmoid_input)) - 1.0  # range [-1, 1]
    norm_urgency = iqr_normalize(urgency_gate)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor bottleneck coupling: prioritize tasks whose long duration blocks high-work/high-rank successors
    # Term = duration * (remaining_work + upward_rank + eps), scaled and normalized
    bottleneck_pressure = duration * (remaining_work + upward_rank + eps)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    
    # Critical-path bonus: activated only for top percentile of upward_rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Ready wait time: smooth saturation via arctan (bounded [0,1], avoids exponential drift)
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = iqr_normalize(wait_saturation)
    
    # DDL-protection gate: activates only when slack is tight *relative to median* AND uncertainty is high
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Uncertainty-normalized term for amplification under risk — now weighted separately
    norm_uncertainty = iqr_normalize(uncertainty)
    
    # Final score: smaller = higher priority
    # Urgency dominates; bottleneck unblocks successors; energy efficiency reduces total cost;
    # critical bonus prioritizes path; wait reward ensures fairness; uncertainty amplification adds risk margin
    score = (
        norm_urgency  # Primary deadline pressure signal
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks
        - norm_wait  # Reward recently arrived tasks (fairness)
        + ddl_protection_active * PARAMS["uncertainty_amplification_weight"] * norm_uncertainty  # Risk-aware margin
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
