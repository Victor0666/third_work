def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths from both parents:
      - Uses raw neg_slack as dominant *weighted* term (Parent 2) with explicit weight tuning.
      - Introduces percentile-based critical-path gate: activates upward_rank × remaining_work only when slack <= p-th percentile (robust vs median).
      - Replaces adaptive_normalize with *risk-aligned MAD normalization*: dispersion scaled by uncertainty-aware MAD, not std(uncertainty), improving stability under low-N or skewed risk.
      - Bottleneck pressure includes uncertainty amplification via power-law on normalized uncertainty (not raw), avoiding outlier sensitivity.
      - Keeps bounded sigmoid wait saturation (Parent 2) for smooth anti-starvation.
      - All terms additive and conditionally gated — no fragile multiplicative couplings.
      - Enforces hard deadline first, then urgency, bottleneck, energy, fairness — with clear signal separation.
      - Uses only {-2,-1,0,1,2} literals; all tunables via PARAMS.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Risk-aligned MAD normalization: dispersion = MAD(x) * (1 + uncertainty_scale * median_uncertainty)
    def risk_aligned_mad_normalize(x):
        x = np.copy(x)
        center = np.median(x) if N > 0 else 0.0
        abs_dev = np.abs(x - center)
        mad = np.median(abs_dev) if N > 0 else eps
        # Scale dispersion by task-level uncertainty context
        unc_scale = np.median(uncertainty) if N > 0 else eps
        dispersion = mad * (1.0 + 0.5 * unc_scale)  # fixed 0.5 factor for bounded scaling
        # Fallback to range if dispersion too small
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = dispersion if dispersion > eps else fallback_range
        return (x - center) / (denom + eps)
    
    # Dominant hard-deadline term: raw clipped negative slack, weighted
    neg_slack = np.clip(-slack, 0.0, None)
    ddl_penalty = PARAMS["ddl_violation_weight"] * neg_slack
    
    # Urgency proxy: linear distance from median slack, capped by feasibility-preserving power law
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, None)
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_clipped = np.minimum(urgency_linear, urgency_cap)
    norm_urgency = risk_aligned_mad_normalize(urgency_clipped)
    
    # Critical-path coupling: activated only when slack <= p-th percentile (more robust than median)
    p = PARAMS["critical_path_gate_threshold"]
    percentile_slack = np.percentile(slack, 100.0 * p) if N > 0 else 0.0
    critical_path_mask = (slack <= percentile_slack).astype(float)
    critical_path_pressure = critical_path_mask * upward_rank * remaining_work
    norm_critical_path = risk_aligned_mad_normalize(critical_path_pressure)
    
    # Energy efficiency: marginal energy per unit duration (J/s), weighted
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = risk_aligned_mad_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency) × (1 + normalized_uncertainty)^exponent
    bottleneck_base = duration * upward_rank * remaining_work * (1.0 + urgency_clipped + eps)
    unc_norm = risk_aligned_mad_normalize(uncertainty)
    bottleneck_pressure = bottleneck_base * np.power(1.0 + unc_norm, 
                                                      PARAMS["uncertainty_amplification_exponent"])
    norm_bottleneck = risk_aligned_mad_normalize(bottleneck_pressure)
    
    # Bounded sigmoid wait saturation: smooth anti-starvation boost
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = risk_aligned_mad_normalize(wait_saturation)
    
    # Final score: smaller = higher priority
    # Dominance order: DDL violation > urgency > critical path > bottleneck > energy > fairness
    score = (
        ddl_penalty
        + norm_urgency
        + PARAMS["bottleneck_coupling_weight"] * norm_bottleneck
        + PARAMS["critical_path_gate_threshold"] * norm_critical_path  # reusing as weight (safe: [0.2,0.95])
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        - norm_wait
    )
    
    # Enforce finiteness and determinism
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
