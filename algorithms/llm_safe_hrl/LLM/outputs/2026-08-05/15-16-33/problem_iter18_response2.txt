def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule incorporating:
    - Deadline-urgency gating: wait-urgency coupling activates *only* when slack <= urgency_gate_threshold.
    - Critical-path–aware adaptive normalization: splits tasks by upward_rank quantile and normalizes within subgroups.
    - Risk-aware energy capping: clips min_incremental_energy using uncertainty-weighted cap to suppress over-penalization.
    - All operations remain deterministic, finite, and free of hidden state or unbounded logic.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Critical-path subgroup mask: high-upward-rank tasks form bottleneck cohort
    if N > 0:
        critical_threshold = np.quantile(upward_rank, PARAMS["critical_path_quantile"])
        is_critical = upward_rank >= critical_threshold - eps
    else:
        is_critical = np.zeros_like(upward_rank, dtype=bool)
    
    # Robust adaptive normalization per subgroup (critical/non-critical)
    def adaptive_normalize_by_group(x, mask):
        x = np.copy(x)
        if N == 0:
            return np.zeros_like(x)
        
        # Normalize critical group
        x_crit = x[mask]
        if len(x_crit) > 0:
            q_low_crit = np.percentile(x_crit, PARAMS["iqr_low_percentile"])
            q_high_crit = np.percentile(x_crit, PARAMS["iqr_high_percentile"])
            iqr_crit = q_high_crit - q_low_crit
            center_crit = np.median(x_crit)
            if iqr_crit < eps:
                x_min_crit, x_max_crit = np.min(x_crit), np.max(x_crit)
                denom_crit = x_max_crit - x_min_crit
                norm_crit = (x_crit - x_min_crit) / (denom_crit + eps) if denom_crit >= eps else np.zeros_like(x_crit)
            else:
                norm_crit = (x_crit - center_crit) / (iqr_crit + eps)
        else:
            norm_crit = np.array([])
        
        # Normalize non-critical group
        x_noncrit = x[~mask]
        if len(x_noncrit) > 0:
            q_low_non = np.percentile(x_noncrit, PARAMS["iqr_low_percentile"])
            q_high_non = np.percentile(x_noncrit, PARAMS["iqr_high_percentile"])
            iqr_non = q_high_non - q_low_non
            center_non = np.median(x_noncrit)
            if iqr_non < eps:
                x_min_non, x_max_non = np.min(x_noncrit), np.max(x_noncrit)
                denom_non = x_max_non - x_min_non
                norm_non = (x_noncrit - x_min_non) / (denom_non + eps) if denom_non >= eps else np.zeros_like(x_noncrit)
            else:
                norm_non = (x_noncrit - center_non) / (iqr_non + eps)
        else:
            norm_non = np.array([])
        
        # Reconstruct full normalized array
        result = np.zeros_like(x)
        if len(norm_crit) > 0:
            result[mask] = norm_crit
        if len(norm_non) > 0:
            result[~mask] = norm_non
        return result
    # End of adaptive_normalize_by_group
    
    # Linear slack urgency: penalize negative slack more strongly, scaled by uncertainty
    linear_slack_penalty = PARAMS["slack_linear_coeff"] * np.clip(-slack, 0.0, None)
    coupled_slack_penalty = linear_slack_penalty * (1.0 + PARAMS["uncertainty_slack_coupling"] * uncertainty)
    norm_slack_penalty = adaptive_normalize_by_group(coupled_slack_penalty, is_critical)
    
    # Upward rank × remaining work interaction: captures downstream computational load on critical path
    critical_load_pressure = upward_rank * (remaining_work + eps)
    norm_critical_load = adaptive_normalize_by_group(critical_load_pressure, is_critical)
    
    # Risk-aware energy capping: clamp before normalization using uncertainty-weighted threshold
    base_energy_cap = np.median(min_incremental_energy) if N > 0 else 0.0
    energy_cap = base_energy_cap * (1.0 + PARAMS["energy_uncertainty_cap_factor"] * np.mean(uncertainty))
    capped_energy = np.clip(min_incremental_energy, None, energy_cap + eps)
    norm_energy = adaptive_normalize_by_group(capped_energy, is_critical)
    shifted_energy = norm_energy + PARAMS["energy_normalization_offset"]
    shifted_energy = np.clip(shifted_energy, 0.0, 2.0)
    
    # Deadline-urgency gated wait coupling: only activate when slack pressure exists
    wait_normalized = adaptive_normalize_by_group(ready_wait_time, is_critical)
    # Gate activation: only apply wait-urgency boost when slack is critically tight
    urgency_mask = slack <= PARAMS["urgency_gate_threshold"]
    wait_urgency_gated = wait_normalized * (1.0 + norm_slack_penalty) * urgency_mask.astype(float)
    norm_wait_urgency = adaptive_normalize_by_group(wait_urgency_gated, is_critical)
    
    # Final score: smaller = higher priority
    # Priority hierarchy: deadline risk > critical load > gated wait-urgency > energy
    score = (
        norm_slack_penalty  # Primary DDL risk signal — linear, stable, uncertainty-amplified
        + PARAMS["upward_remaining_interaction_weight"] * norm_critical_load  # Critical-path load pressure
        + PARAMS["wait_urgency_coupling"] * norm_wait_urgency  # Starvation prevention, strictly gated by slack threshold
        + shifted_energy  # Risk-adjusted energy, capped & bias-corrected
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
