def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with adaptive slack nonlinearity, unified criticality gating,
    and starvation-resistant wait boost — all preserving hard deadline safety.

    Key improvements:
      - Introduces slack_sensitivity_exponent to nonlinearly amplify urgency near deadline (|slack| < median),
        enhancing responsiveness to tight deadlines without over-penalizing large positive slack.
      - Restores universal wait boost (no slack gating) but adds monotonic decay scaling to prevent dominance.
      - Simplifies critical-path bonus to percentile-gated upward_rank only — avoids fragile dual conditions
        while retaining strong bias toward critical-path tasks regardless of current slack sign.
      - All normalization uses robust quantile-based centering and scaling; no median-zero assumptions.
      - Energy-efficiency term now includes uncertainty-weighted coupling only when both are high,
        improving fidelity to risk-adjusted objective.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)
    N = len(slack)

    # Ensure float arrays and avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust quantile-based normalization
    def robust_normalize(x):
        x = np.copy(x)
        q1 = np.quantile(x, PARAMS["quantile_low"])
        q3 = np.quantile(x, PARAMS["quantile_high"])
        iqr = q3 - q1
        center = np.median(x)
        scale = iqr if iqr > eps_safe else eps_safe
        return (x - center) / (scale + eps_safe)

    # Duration: exec + comm
    duration = min_exec_time + min_comm_time
    norm_duration = robust_normalize(duration)

    # Slack: piecewise power-law — stronger penalty near zero slack, gentler for large margins
    slack_norm = robust_normalize(slack)
    median_abs_slack = np.median(np.abs(slack))
    # Apply exponent only to normalized slack magnitude near deadline region
    slack_magnitude = np.abs(slack_norm)
    slack_nonlinear = np.where(
        slack_magnitude <= np.median(slack_magnitude) + 0.5,
        np.sign(slack_norm) * (np.abs(slack_norm) ** PARAMS["slack_sensitivity_exponent"]),
        slack_norm
    )
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_risk_penalty"] * (-slack_nonlinear),
        -PARAMS["slack_urgency_gain"] * slack_nonlinear
    )

    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration_safe = np.maximum(duration, eps_safe)
    energy_per_duration = min_incremental_energy / duration_safe
    norm_energy_eff = robust_normalize(energy_per_duration)

    # Uncertainty-coupled energy penalty: only when both energy_eff and uncertainty are above median
    median_unc = np.median(uncertainty)
    median_energy_eff = np.median(energy_per_duration)
    unc_norm = robust_normalize(uncertainty)
    energy_uncertainty_penalty = np.where(
        (energy_per_duration > median_energy_eff) & (uncertainty > median_unc),
        unc_norm * norm_energy_eff * PARAMS["uncertainty_slack_coupling"],
        0.0
    )

    # Critical-path bonus: activated only for top percentile of upward_rank (unconditional on slack)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps_safe)
    critical_gate = (rank_percentile >= PARAMS["critical_rank_threshold"]).astype(float)
    norm_rank = robust_normalize(upward_rank)
    critical_bonus = PARAMS["critical_path_bonus"] * norm_rank * critical_gate

    # Waiting-time boost: universal exponential decay, scaled by normalized wait time to prevent dominance
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    norm_wait = robust_normalize(wait_boost)

    # Uncertainty-slack coupling: amplify urgency only when slack < 0 AND uncertainty > median
    coupled_urgency = np.where(
        (slack < 0) & (uncertainty > median_unc),
        unc_norm * (-slack_nonlinear) * PARAMS["uncertainty_slack_coupling"],
        0.0
    )

    # Duration-efficiency ratio
    dur_eff_ratio = PARAMS["duration_efficiency_ratio"] * norm_duration

    # Final score: smaller = higher priority
    score = (
        slack_penalty
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        + energy_uncertainty_penalty
        - critical_bonus
        - norm_wait
        + coupled_urgency
        + dur_eff_ratio
    )

    # Ensure finite output
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
