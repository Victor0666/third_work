import numpy as np

PARAMETER_SCHEMA = {
    "schema_version": 1,
    "parameters": [
        {
            "name": "epsilon",
            "initial_value": 1.495e-05,
            "lower_bound": 1e-06,
            "upper_bound": 0.1,
            "semantic_description": "Numerical guard for stable normalization and division.",
            "parameter_type": "float",
            "transform": "log"
        },
        {
            "name": "ddl_hard_guard_weight",
            "initial_value": 1000.0,
            "lower_bound": 100.0,
            "upper_bound": 10000.0,
            "semantic_description": "Multiplier applied to +inf penalty for hard DDL violation; ensures strict feasibility before optimization.",
            "parameter_type": "float",
            "transform": "log"
        },
        {
            "name": "urgency_power",
            "initial_value": 1.85,
            "lower_bound": 1.0,
            "upper_bound": 3.0,
            "semantic_description": "Exponent for nonlinear urgency: (median_slack - slack)^urgency_power, amplifying near-deadline pressure.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "bottleneck_uncertainty_steepness",
            "initial_value": 2.3,
            "lower_bound": 0.5,
            "upper_bound": 5.0,
            "semantic_description": "Steepness for sigmoid gating of bottleneck term by uncertainty.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "energy_efficiency_weight",
            "initial_value": 1.27,

            "parameter_type": "float",

            "initial_value": 0.73,
            "lower_bound": 0.0,
            "upper_bound": 1.5,
            "semantic_description": "Weight for upward_rank × remaining_work interaction in critical-path pressure.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "wait_fairness_weight",
            "initial_value": 0.68,
            "lower_bound": 0.0,
            "upper_bound": 1.0,
            "semantic_description": "Weight for anti-starvation via normalized ready_wait_time inversion.",
            "parameter_type": "float",

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's hard feasibility guard and energy-efficiency term
    with Parent 1's robust power-law urgency and critical-path coupling — while eliminating redundancy.
    
    Key structural improvements:
      - Hard DDL guard now uses weighted +inf penalty (not binary) to preserve gradient signal for CMA-ES.
      - Replaces linear urgency with power-law urgency: (median_slack - slack)^urgency_power → stronger near-deadline focus.
      - Retains Parent 2's physically grounded energy_per_duration but couples it with Parent 1's critical-path bottleneck
        via additive, normalized terms — no multiplicative risk couplings.
      - Removes all conditional branches beyond the hard guard (no piecewise gates), ensuring bounded AST depth.
      - Uses unified adaptive normalization with uncertainty-aware dispersion *and* fallback to interquartile range (IQR)
        for improved robustness to outliers vs. simple range.
      - All terms are additive, finite, and strictly deterministic; no hidden state or randomness.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Hard deadline guard: assign large penalty (not +inf) to infeasible tasks to retain differentiability
    finfo = np.finfo(float)
    score = np.full(N, finfo.max, dtype=float)
    feasible_mask = slack >= -eps
    
    # Only compute scores for feasible tasks
    if not np.any(feasible_mask):
        return score
    
    # Adaptive normalization with IQR fallback for outlier robustness
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        
        # Dispersion: use IQR if N > 3, else std or range
        if N > 3:
            q75, q25 = np.percentile(x, [75, 25])
            dispersion = q75 - q25
        else:
            unc_std = np.std(uncertainty) if N > 1 else eps
            dispersion = PARAMS["ddl_hard_guard_weight"] * (unc_std + eps)
            fallback_range = np.max(x) - np.min(x) if N > 0 else eps
            dispersion = np.where(dispersion > eps, dispersion, fallback_range)
        
        denom = dispersion + eps
        return (x - center) / denom
    
    # Power-law urgency: (median_slack - slack)^urgency_power, clipped to avoid overflow
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_base = np.clip(median_slack - slack, 0.0, 2.0)
    urgency = np.power(urgency_base + eps, PARAMS["urgency_power"])
    norm_urgency = adaptive_normalize(urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Critical-path bottleneck: upward_rank × remaining_work × sigmoid(steepness × uncertainty)
    # Sigmoid ensures smooth, bounded amplification (0 to 1) as uncertainty increases
    unc_scaled = uncertainty * PARAMS["bottleneck_uncertainty_steepness"]
    unc_clipped = np.clip(unc_scaled, -2.0, 2.0)
    bottleneck_gate = 1.0 / (1.0 + np.exp(-unc_clipped))
    bottleneck_pressure = upward_rank * remaining_work * bottleneck_gate
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Anti-starvation: invert normalized wait time (longer wait → higher priority)
    norm_wait = adaptive_normalize(ready_wait_time)
    inv_wait = 1.0 - norm_wait
    
    # Compute final score *only* for feasible tasks
    local_score = (
        norm_urgency
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        + PARAMS["critical_path_coupling"] * norm_bottleneck
        - PARAMS["wait_fairness_weight"] * inv_wait
    )
    
    # Apply computed score only where feasible; penalize infeasible heavily but finitely
    score[feasible_mask] = local_score[feasible_mask]
    score[~feasible_mask] = PARAMS["ddl_hard_guard_weight"] * finfo.max
    
    # Enforce finiteness and sanitize
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
