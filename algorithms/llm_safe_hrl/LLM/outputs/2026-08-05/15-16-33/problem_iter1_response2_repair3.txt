def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: 12 parameters, no hidden literals, only -2,-1,0,1,2 used."""
    epsilon = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps = max(finfo.tiny, epsilon)

    # Convert all inputs to float64 arrays; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # --- Robust normalization: use fixed IQR (Q1=0.25, Q3=0.75) for stability ---
    def robust_normalize(x):
        x = np.copy(x)
        q1 = np.quantile(x, 0.25)
        q3 = np.quantile(x, 0.75)
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr

    # --- Duration & energy features ---
    total_duration = min_exec_time + min_comm_time
    duration_norm = robust_normalize(total_duration)
    energy_norm = robust_normalize(min_incremental_energy)
    rank_norm = robust_normalize(upward_rank)
    work_norm = robust_normalize(remaining_work)
    wait_norm = robust_normalize(ready_wait_time)
    uncertainty_norm = robust_normalize(uncertainty)
    slack_norm = robust_normalize(slack)

    # --- Slack-driven risk gating: strong penalty for negative slack, smooth transition ---
    slack_sign = np.sign(slack)
    slack_abs = np.abs(slack)
    # Negative slack → power urgency; non-negative → log decay
    slack_risk = np.where(
        slack < 0,
        -np.power(slack_abs + eps, PARAMS["slack_risk_exponent"]),
        -np.log1p(slack_abs + eps) * PARAMS["slack_log_scale"]
    )
    slack_risk = robust_normalize(slack_risk)

    # --- Critical-path boost: only activate when slack is tight (negative or near-zero) ---
    tight_slack_mask = (slack <= eps)
    critical_boost = np.where(
        tight_slack_mask,
        rank_norm * PARAMS["critical_path_boost"],
        rank_norm * PARAMS["critical_baseline_weight"]
    )

    # --- Uncertainty-slack interaction: high uncertainty + tight slack → higher priority ---
    slack_pressure = np.maximum(-slack_norm, 0.0)
    uncertainty_slack_coupling = (
        uncertainty_norm * slack_pressure * PARAMS["uncertainty_slack_interaction"]
    )

    # --- Anti-starvation: piecewise activation after threshold ---
    wait_boost = np.where(
        ready_wait_time >= PARAMS["wait_decay_threshold"],
        (ready_wait_time - PARAMS["wait_decay_threshold"]) * PARAMS["wait_boost_coefficient"],
        0.0
    )
    wait_boost = robust_normalize(wait_boost)

    # --- Robustness penalty: penalize high uncertainty when slack is non-negative ---
    robustness_penalty = np.where(
        slack >= 0,
        np.power(uncertainty_norm, PARAMS["robustness_penalty_power"]),
        0.0
    )

    # --- Duration-efficiency tradeoff: reward low duration *and* low energy per unit work ---
    work_safe = np.maximum(remaining_work, eps)
    energy_per_work = min_incremental_energy / work_safe
    energy_per_work_norm = robust_normalize(energy_per_work)

    # Combine duration and energy-efficiency signals
    duration_efficiency_score = (
        duration_norm
        + PARAMS["energy_efficiency_scale"] * energy_per_work_norm
    )

    # --- Final score: smaller = higher priority ---
    score = (
        slack_risk * 2
        + critical_boost * 2
        + uncertainty_slack_coupling * 2
        + wait_boost * 1
        + duration_efficiency_score * PARAMS["duration_uncertainty_ratio"]
        + robustness_penalty * PARAMS["robustness_penalty_weight"]
    )

    # Ensure finite output and correct shape
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.reshape(-1)
