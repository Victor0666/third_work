def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating evidence-backed structural changes:
    - Replaces arctan wait saturation with bounded sigmoid (smoother, more stable under long waits)
    - Removes critical-path percentile gating (evidence shows it's inactive; replaced by direct upward_rank coupling)
    - Introduces multiplicative urgency × uncertainty coupling only under joint DDL pressure
    - Uses adaptive min-max fallback when IQR dispersion is low (improves robustness to near-constant features)
    - Eliminates redundant parameters (slack_sigmoid_* and wait_saturation_time now active per evidence)
    - Retains conditional DDL protection gate but tightens activation logic to joint condition
    - All terms normalized via tunable IQR percentiles; no unbounded growth or unstable ops
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max when dispersion is low
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR too small (low dispersion), else use IQR
        if iqr > eps:
            denom = iqr
        else:
            denom = np.max(x) - np.min(x)
            if denom < eps:
                denom = eps
        
        return (x - center) / (denom + eps)
    
    # Sigmoid urgency gate: smooth, differentiable, centered at offset, steepness tunable
    # Maps slack → [0, 1]: negative slack → ~1 (high urgency), positive slack → ~0 (low urgency)
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    # Avoid overflow in exp by clamping input using declared parameter
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered, 
                            -PARAMS["slack_sigmoid_steepness"] * 10.0,  # implicit safe bound
                            PARAMS["slack_sigmoid_steepness"] * 10.0)
    urgency_gate = 1.0 / (1.0 + np.exp(sigmoid_input))  # range [0, 1]
    
    # Normalize urgency for consistent scaling
    norm_urgency = adaptive_normalize(urgency_gate)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Successor bottleneck coupling: prioritize tasks whose long duration blocks high-work/high-rank successors
    # Term = duration * upward_rank * (1 + urgency_gate) to amplify under deadline stress
    bottleneck_pressure = duration * upward_rank * (1.0 + urgency_gate + eps)
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Ready wait time: bounded sigmoid saturation (more stable than arctan for extreme values)
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_scaled))  # [0,1], smooth, bounded
    norm_wait = adaptive_normalize(wait_saturation)
    
    # DDL-protection gate: activates only when slack is tight AND uncertainty is high
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Uncertainty-normalized term for amplification under risk — only activated when DDL at risk
    norm_uncertainty = adaptive_normalize(uncertainty)
    ddl_risk_amplifier = ddl_protection_active * norm_uncertainty
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline safety (urgency) > bottleneck release > energy > fairness (wait) > risk amplification
    score = (
        norm_urgency  # Primary deadline pressure signal
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors under stress
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - norm_wait  # Reward recently arrived tasks (anti-starvation)
        + ddl_risk_amplifier  # Boost uncertainty-aware scheduling only when DDL is threatened
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
