def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule emphasizing deadline risk mitigation via piecewise linear+convex penalty,
    gated critical-path bonus, duration-normalized energy efficiency, and successor-release delay coupling.
    
    Key mutations from prior version:
      - Replaces sigmoid urgency with robust piecewise-linear+convex slack penalty (stronger DDL violation reduction)
      - Gates critical-path bonus by percentile rank (not raw upward_rank) to prevent outlier dominance
      - Uses duration-normalized energy efficiency (energy / (exec + comm + ε)) aligned with urgency scaling
      - Adds conditional DDL-protection gate activated only when (slack < median(slack)) AND (upward_rank > percentile_threshold)
      - Introduces successor-release delay interaction: min(remaining_work / (slack + ε), successor_delay_cap) to unblock bottlenecks under tight deadlines
      - Removes wait-time saturation (evidence shows it's inactive; avoids unnecessary complexity)
      - Removes uncertainty amplification (evidence shows it's inactive; simplifies structure)
      - All normalizations use IQR with tunable percentiles; no hard thresholds or unbounded growth
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Piecewise-linear+convex slack penalty: stronger penalty for negative slack, smooth transition at zero
    # Maps slack → [0, ∞): negative slack → large positive penalty (low priority), zero → small penalty
    slack_penalty = np.where(
        slack < 0.0,
        -slack * 2.0 + (slack ** 2) * 0.5,  # convex penalty for lateness risk
        np.where(
            slack <= 1.0,
            slack * 0.5,  # gentle linear ramp for near-deadline
            0.5 + (slack - 1.0) * 0.1  # saturating tail for loose slack
        )
    )
    norm_slack_penalty = iqr_normalize(slack_penalty)
    
    # Duration-normalized energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor-release delay interaction: remaining_work / (slack + ε), capped to prevent explosion
    # Encourages early release of successors when slack is tight
    successor_release_delay = np.minimum(remaining_work / (slack + eps), PARAMS["successor_delay_cap"])
    norm_successor_delay = iqr_normalize(successor_release_delay)
    
    # Critical-path bonus: activated only for top percentile of upward_rank (percentile-ranked, not raw)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Conditional DDL-protection gate: activates only when slack is tight AND upward_rank is high
    # Uses ddl_protection_gate_threshold to gate on uncertainty-aware condition (as declared)
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Final score: smaller = higher priority
    # Order reflects objective hierarchy: deadline safety > bottleneck release > energy > critical path
    score = (
        norm_slack_penalty  # Primary DDL risk signal (piecewise convex penalty)
        + PARAMS["successor_bottleneck_coupling"] * norm_successor_delay  # Unblock successors under tight slack
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks only when ranked highly
        + ddl_protection_active * norm_rank  # Extra boost when both slack and uncertainty indicate DDL risk
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
