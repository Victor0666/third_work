def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's robust linear urgency & log wait with Parent 1's critical-path awareness,
    enhanced by performance analysis insights:
      - Replaces linear urgency with bounded sigmoid penalty on slack for smoother, more discriminative lateness response.
      - Introduces slack-modulated bottleneck coupling: (exec+comm)*upward_rank*(1 + sigmoid(slack)) — amplifies pressure only when deadlines are tight.
      - DDL-protection gate now uses strict condition (slack < 0 AND uncertainty > median_uncertainty), eliminating fragile percentile thresholds.
      - Keeps log-clipped wait term for bounded anti-starvation; removes arctan and percentile gating.
      - Retains direct upward_rank bonus (no gating) for critical path emphasis.
      - All normalization uses elite-tuned IQR percentiles from Parent 2.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Sigmoid urgency penalty: smooth, bounded, monotonic response to negative slack
    # Avoids discontinuity of linear ramp while preserving strong signal near deadline
    slack_centered = slack - np.median(slack) if N > 1 else slack
    sigmoid_input = PARAMS["slack_sigmoid_steepness"] * (-slack_centered)
    # Clamp input to avoid overflow in exp
    sigmoid_input = np.clip(sigmoid_input, -20.0, 20.0)
    urgency_penalty = 1.0 / (1.0 + np.exp(-sigmoid_input))  # [0,1]
    norm_urgency = iqr_normalize(urgency_penalty)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Slack-modulated successor bottleneck: (exec+comm) * upward_rank * (1 + sigmoid(slack))
    # Amplifies pressure on critical successors only when slack is negative (deadline stress)
    slack_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["slack_sigmoid_steepness"] * np.clip(-slack, 0.0, 20.0)))
    bottleneck_pressure = duration * (upward_rank + eps) * (1.0 + PARAMS["bottleneck_slack_modulation"] * slack_sigmoid)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    
    # Critical-path bonus: direct upward_rank contribution (no gating — proven effective)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = norm_rank
    
    # Anti-starvation wait term: log-scaled with hard clip to ensure bounded influence
    wait_clipped = np.clip(ready_wait_time, 0.0, PARAMS["wait_clip_threshold"])
    wait_log = np.log1p(wait_clipped)  # log1p avoids log(0)
    norm_wait = iqr_normalize(wait_log)
    
    # DDL-protection gate: activates only when slack < 0 AND uncertainty > median_uncertainty
    # Eliminates fragile percentile threshold; uses median for robustness
    median_uncertainty = np.median(uncertainty) if N > 0 else 0.0
    ddl_protection_active = (
        (slack < 0.0) &
        (uncertainty > median_uncertainty)
    ).astype(float)
    
    # Final score: smaller = higher priority
    # Lexicographic priority: deadline safety > bottleneck release > energy > fairness
    score = (
        norm_urgency  # Primary deadline pressure (smooth sigmoid penalty)
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors under deadline stress
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks directly
        - norm_wait  # Reward recent arrivals (anti-starvation)
        + ddl_protection_active * norm_urgency  # Amplify urgency only when both DDL risk and high uncertainty co-occur
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
