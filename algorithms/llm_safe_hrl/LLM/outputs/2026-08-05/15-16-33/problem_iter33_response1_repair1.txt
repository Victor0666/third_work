def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three structural improvements:
      - Replaces binary slack gate with smooth sigmoid gate using tunable slope for gradient-friendly CMA-ES optimization.
      - Restores core bottleneck interaction: upward_rank^p × remaining_work^p (p learned) — critical for preventing critical-path starvation.
      - Uses *structural weighting* (1.0, 0.5, 0.3) instead of parameterized weights for final sum to eliminate inactive-parameter fragility;
        all tunables now govern only gates, interactions, and normalization — improving evolvability and robustness.
      - All declared parameters are used; no unused or missing PARAMS references.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Pre-normalized hard-deadline violation signal: dominant term
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Smooth sigmoid slack feasibility gate (replaces brittle binary threshold)
    # Gate activates strongly when slack is critically tight (< 0), smoothly decays as slack increases
    slack_centered = -slack  # positive when slack negative → high urgency
    slack_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_feasibility_sigmoid_slope"] * (slack_centered + eps)))
    
    # Urgency proxy: linear deviation from zero slack, capped for stability
    urgency_linear = np.clip(-slack, 0.0, 2.0)
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.minimum(urgency_linear, urgency_cap)
    
    # Bottleneck pressure: (upward_rank * remaining_work)^p × duration × urgency × (1 + uncertainty)^amplification
    # Structural interaction: upward_rank and remaining_work coupled via learned power (not linear product)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    ur_interaction = np.power(upward_rank * remaining_work + eps, PARAMS["upward_remaining_interaction_power"])
    bottleneck_pressure = ur_interaction * duration * (1.0 + urgency_linear + eps)
    
    # Apply uncertainty amplification only where slack is tight (smoothly gated)
    unc_normalized = (uncertainty - np.min(uncertainty)) / (np.ptp(uncertainty) + eps) if N > 1 else np.zeros_like(uncertainty)
    amplified_uncertainty = np.power(1.0 + unc_normalized, PARAMS["uncertainty_amplification_exponent"])
    bottleneck_pressure = bottleneck_pressure * (slack_gate * amplified_uncertainty + (1.0 - slack_gate))
    
    # Adaptive normalization: dispersion scaled by std(uncertainty) for risk alignment
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x) if N > 0 else 0.0
        unc_std = np.std(uncertainty) if N > 1 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = np.where(dispersion > eps, dispersion, fallback_range)
        return (x - center) / (denom + eps)
    
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bounded sigmoid wait saturation: smooth anti-starvation boost
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = adaptive_normalize(wait_saturation)
    
    # Final score: smaller = higher priority
    # Structural weighting (fixed, not parameterized) eliminates fragile weight tuning:
    # 1.0 × DDL violation (dominant), 0.5 × bottleneck, 0.3 × energy, -1.0 × fairness
    score = (
        PARAMS["ddl_urgency_weight"] * neg_slack
        + 0.5 * norm_bottleneck
        + 0.3 * norm_energy_eff
        - 1.0 * norm_wait
    )
    
    # Enforce finiteness and finite numeric output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
