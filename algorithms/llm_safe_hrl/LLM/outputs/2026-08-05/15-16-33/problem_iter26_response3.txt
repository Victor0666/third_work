def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's strict feasibility enforcement and rank-based normalization
    with Parent 1's conditional DDL protection gate and power-law wait decay.
    
    Key structural improvements:
      - Replaces linear urgency ramp with sigmoid urgency gating using slack deficit relative to median,
        providing smoother, more discriminative urgency signal under tight deadlines.
      - Integrates Parent 1's power-law wait boost (with bounded exponent) instead of rank-inversion,
        preserving fairness while avoiding saturation and improving long-tail discriminability.
      - Retains Parent 2's hard deadline guard and uncertainty-gated bottleneck/energy terms for robustness.
      - Uses MAD-based robust normalization for energy and urgency terms (more stable than rank for small N).
      - All nonlinearities clipped to [-2,2] domain to prevent overflow and ensure finite gradients.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Hard deadline guard: assign large finite penalty to infeasible tasks
    finfo = np.finfo(float)
    score = np.full(N, finfo.max, dtype=float)
    feasible_mask = slack >= -eps
    
    if not np.any(feasible_mask):
        return score
    
    # Robust adaptive normalization using MAD for dispersion (more stable than std/rank under outliers & small N)
    def robust_normalize(x):
        x = np.copy(x)
        if N <= 1 or np.all(x == x[0]):
            return np.zeros_like(x, dtype=float)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        dispersion = PARAMS["mad_scale_factor"] * (mad + eps)
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = dispersion if dispersion > eps else fallback_range
        return (x - center) / (denom + eps)
    
    # Urgency: sigmoid gating on normalized slack deficit (Parent 1's idea, enhanced with Parent 2's steepness)
    median_slack = np.median(slack) if N > 0 else 0.0
    slack_deficit = np.clip(median_slack - slack, 0.0, None)  # positive when slack < median
    # Normalize deficit relative to interquartile range for stability
    q1 = np.percentile(slack, 25) if N > 0 else median_slack
    q3 = np.percentile(slack, 75) if N > 0 else median_slack
    iqr = max(q3 - q1, eps)
    norm_deficit = np.clip(slack_deficit / (iqr + eps), 0.0, 2.0)
    # Sigmoid urgency: sharp rise only near critical slack
    urgency = 1.0 / (1.0 + np.exp(-PARAMS["urgency_steepness"] * (norm_deficit - PARAMS["slack_sensitivity_threshold"])))
    norm_urgency = robust_normalize(urgency)
    
    # Raw energy efficiency: use min_incremental_energy directly, MAD-normalized
    norm_energy_eff = robust_normalize(min_incremental_energy)
    
    # Uncertainty-modulated energy term: activated only above threshold
    unc_centered = uncertainty - PARAMS["uncertainty_activation_threshold"]
    unc_clipped = np.clip(unc_centered, -2.0, 2.0)
    unc_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * unc_clipped))
    energy_uncertainty_term = min_incremental_energy * unc_gate
    norm_energy_uncertain = robust_normalize(energy_uncertainty_term)
    
    # Critical-path bottleneck: activated only above uncertainty threshold
    bottleneck_gate = np.where(uncertainty >= PARAMS["uncertainty_activation_threshold"],
                               1.0 / (1.0 + np.exp(-PARAMS["bottleneck_uncertainty_steepness"] * 
                                                  np.clip(unc_centered, 0.0, 2.0))),
                               0.0)
    bottleneck_pressure = upward_rank * remaining_work * bottleneck_gate
    norm_bottleneck = robust_normalize(bottleneck_pressure)
    
    # Anti-starvation: power-law wait boost (Parent 1's improvement over sigmoid/rank inversion)
    # Avoids saturation and preserves discriminability for long-waiting tasks
    median_wait = np.median(ready_wait_time) if N > 0 else eps
    wait_scaled = np.clip(ready_wait_time / (median_wait + eps), 0.0, 2.0)
    wait_boost = np.power(wait_scaled + eps, PARAMS["wait_fairness_weight"])  # reusing weight as exponent
    norm_wait = robust_normalize(wait_boost)
    
    # Final score: smaller = higher priority
    # Ordering: urgency > bottleneck > energy > fairness (wait boost subtracted)
    local_score = (
        norm_urgency
        + PARAMS["critical_path_coupling"] * norm_bottleneck
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        + PARAMS["energy_uncertainty_coupling"] * norm_energy_uncertain
        - norm_wait
    )
    
    # Apply computed score only where feasible; penalize infeasible heavily but finitely
    score[feasible_mask] = local_score[feasible_mask]
    score[~feasible_mask] = PARAMS["ddl_hard_guard_weight"] * finfo.max
    
    # Enforce finiteness and sanitize
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
