def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Bounded linear urgency ramp on slack (replaces sigmoid) for improved gradient signal near zero.
      - Host-load-aware DDL-protection gate: activates only when both uncertainty AND host load are high.
      - Direct successor-release pressure: (remaining_work * upward_rank) / (duration + ε), penalizing bottlenecks.
      - Tighter IQR normalization bounds (20/80) for robustness to outliers.
      - All operations guarded against NaN/inf/zero; deterministic and finite output.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Bounded linear urgency ramp: smooth, interpretable, avoids sigmoid saturation
    # Maps slack → urgency: negative slack → positive urgency; zero at slack_ramp_offset
    urgency_linear = PARAMS["slack_ramp_slope"] * (slack - PARAMS["slack_ramp_offset"])
    # Clamp to [-2, 2] to prevent unbounded growth while preserving sign and monotonicity
    urgency_clamped = np.clip(urgency_linear, -2.0, 2.0)
    norm_urgency = iqr_normalize(urgency_clamped)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor-release pressure: prioritize tasks whose long duration blocks high-work/high-rank successors
    # Term = (remaining_work * upward_rank) / (duration + ε), scaled and normalized
    release_pressure = (remaining_work * upward_rank) / (duration + eps)
    norm_release = iqr_normalize(release_pressure)
    
    # Critical-path bonus: activated only for top percentile of upward_rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Ready wait time: smooth saturation via arctan (bounded [0,1])
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = iqr_normalize(wait_saturation)
    
    # DDL-protection gate: activates only when slack is tight AND uncertainty is high AND host load is high
    # Since host_load is not provided, we proxy it with normalized uncertainty * slack pressure (risk-aware load proxy)
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    slack_pressure = np.maximum(0.0, median_slack - slack)  # higher when slack < median
    host_load_proxy = uncertainty * slack_pressure
    ddl_protection_active = (
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty) &
        (host_load_proxy > PARAMS["host_load_sensitivity"] * np.max(host_load_proxy + eps))
    ).astype(float)
    
    # Uncertainty-normalized term for amplification under risk
    norm_uncertainty = iqr_normalize(uncertainty)
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline urgency > successor release > energy > critical path > fairness > risk amplification
    score = (
        norm_urgency  # Primary deadline pressure (linear, bounded)
        + PARAMS["successor_release_pressure_weight"] * norm_release  # Unblock critical successors
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks
        - norm_wait  # Reward recently arrived tasks (fairness)
        + ddl_protection_active * norm_uncertainty  # Amplify uncertainty only when DDL & load at risk
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
