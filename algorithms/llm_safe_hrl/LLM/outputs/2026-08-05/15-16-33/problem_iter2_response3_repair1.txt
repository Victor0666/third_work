def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's superior deadline/energy structure with Parent 1's robustness and starvation control.
    
    Key structural improvements:
      - Replaces exponential wait boost with bounded arctan saturation (from Parent 1) for smoother fairness control
      - Adds remaining_work × energy coupling to penalize high-energy tasks blocking large downstream work
      - Uses robust percentile-based criticality gate (Parent 2) but adds slack-aware activation: only triggers when slack < median
      - Combines piecewise slack handling (Parent 2) with uncertainty-amplified penalty (Parent 1 style)
      - All normalizations use tunable IQR percentiles; no hardcoded quantiles
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_percentile_low"])
        q_high = np.percentile(x, PARAMS["iqr_percentile_high"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps_safe else (np.max(np.abs(x - center)) + eps_safe)
        return (x - center) / (denom + eps_safe)

    # Slack: piecewise — strong penalty for negative slack, gentle reward for positive slack
    slack_norm = iqr_normalize(slack)
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_risk_penalty"] * (-slack_norm),
        -PARAMS["slack_urgency_gain"] * slack_norm
    )

    # Energy efficiency: marginal energy per unit of total duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps_safe)
    energy_per_duration = min_incremental_energy / duration
    energy_eff_norm = iqr_normalize(energy_per_duration)

    # Critical-path bonus: activated only for top percentile AND tight slack (< median)
    rank_norm = iqr_normalize(upward_rank)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps_safe)
    median_slack = np.median(slack)
    critical_gate = np.where(
        (rank_percentile >= PARAMS["critical_rank_threshold"]) & (slack < median_slack),
        1.0,
        0.0
    )
    critical_bonus = PARAMS["critical_path_bonus"] * rank_norm * critical_gate

    # Waiting-time boost: bounded arctan saturation (Parent 1) instead of exponential (Parent 2) for smoother fairness
    wait_scaled = ready_wait_time / (PARAMS["wait_decay_rate"] + eps_safe)  # Reused parameter name for saturation scale
    wait_boost = (2.0 / np.pi) * np.arctan(wait_scaled)  # maps [0,∞) → [0,1]
    wait_norm = iqr_normalize(wait_boost)

    # Uncertainty-slack coupling: amplify urgency only under high uncertainty AND negative slack
    unc_norm = iqr_normalize(uncertainty)
    slack_pressure = np.maximum(-slack_norm, 0.0)
    coupled_urgency = PARAMS["uncertainty_slack_coupling"] * unc_norm * slack_pressure

    # Duration-efficiency ratio: balance duration cost vs energy cost
    dur_eff_ratio = PARAMS["duration_efficiency_ratio"] * iqr_normalize(duration)

    # Remaining work × energy coupling: penalize tasks with high marginal energy that block large downstream work
    # Normalized product avoids scale dominance; capped to [-2,2] for stability
    rw_norm = iqr_normalize(remaining_work)
    energy_norm = iqr_normalize(min_incremental_energy)
    rw_energy_coupling = np.clip(
        PARAMS["remaining_work_energy_coupling"] * rw_norm * energy_norm,
        -2.0, 2.0
    )

    # Final score: smaller = higher priority
    # Deadline safety dominates (slack_penalty), then efficiency (energy_eff_norm), then fairness (wait_norm), then coupling terms
    score = (
        slack_penalty
        + PARAMS["energy_efficiency_weight"] * energy_eff_norm
        - critical_bonus
        - wait_norm
        + coupled_urgency
        + dur_eff_ratio
        + rw_energy_coupling
    )

    # Ensure finite output, shape compliance, and numerical safety
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
