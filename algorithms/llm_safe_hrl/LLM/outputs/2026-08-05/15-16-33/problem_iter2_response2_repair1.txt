def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's robust structure with Parent 1's stability,
       enhanced by smooth sigmoid urgency, successor-aware delay penalty, and conditional DDL-protection.
    
    Key innovations:
      - Smooth, differentiable sigmoid urgency gate on slack (replaces piecewise), centered at configurable offset
      - Successor release delay term: `min(remaining_work / (slack + eps), successor_delay_upper_bound)` explicitly penalizes bottleneck tasks
      - DDL-protection gate: activates only when task is both high-rank *and* slack is below median (tight)
      - Critical-path coupling: multiplies normalized slack pressure with rank to amplify urgency on critical path
      - Arctan-saturated wait boost with tunable scale for fairness without starvation
      - All features normalized via configurable IQR percentiles for robustness to outliers
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float dtype and avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_percentile_low"])
        q_high = np.percentile(x, PARAMS["iqr_percentile_high"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Sigmoid urgency gate: smooth, differentiable, tunable steepness & offset
    # Maps slack → [0,1], where 0 = neutral, ~1 = high urgency (negative slack), ~0 = low urgency (positive slack)
    slack_urgency = 1.0 / (1.0 + np.exp(-PARAMS["slack_sigmoid_steepness"] * (slack - PARAMS["slack_sigmoid_offset"])))
    
    # Slack pressure: normalized negative slack magnitude, zero for positive slack
    slack_norm = iqr_normalize(slack)
    slack_pressure = np.maximum(-slack_norm, 0.0)
    
    # Successor release delay penalty: how much remaining work is blocked per unit slack margin
    # Bounded to prevent explosion when slack ≈ 0
    successor_delay = np.clip(
        remaining_work / (slack + eps),
        0.0, PARAMS["successor_delay_upper_bound"]
    )
    norm_successor_delay = iqr_normalize(successor_delay)
    
    # Critical-path coupling: amplify slack pressure only for high-rank tasks
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    ddl_protection_gate = np.where(
        (rank_percentile >= PARAMS["ddl_protection_gate_threshold"]) & (slack < np.median(slack)),
        1.0, 0.0
    )
    coupled_urgency = PARAMS["critical_path_coupling_strength"] * slack_pressure * ddl_protection_gate
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Ready wait time: smooth saturation via arctan (bounded, monotonic)
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    norm_wait = (2.0 / np.pi) * np.arctan(wait_scaled)  # maps [0,∞) → [0,1]
    
    # Final score: smaller = higher priority
    # Deadline safety dominates (urgency + successor delay + coupled urgency)
    # Efficiency follows (energy/duration), then fairness (wait boost subtracted)
    score = (
        slack_urgency * 2.0  # base urgency weight
        + PARAMS["successor_delay_penalty_weight"] * norm_successor_delay
        + coupled_urgency
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
    )
    
    # Ensure finite output and shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
