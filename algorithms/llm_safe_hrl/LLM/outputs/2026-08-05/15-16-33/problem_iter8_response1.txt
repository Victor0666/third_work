def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust urgency & host-aware energy with Parent 1's hard deadline gating.
    
    Key structural innovations:
      - Dual-mode urgency: linear urgency gate (Parent 2) *plus* hard feasibility guard (Parent 1) — 
        when slack < ddl_feasibility_threshold, urgency is amplified and all non-urgency terms suppressed.
      - Host-load–aware energy normalization preserved from Parent 2, now gated by feasibility mode to avoid over-penalizing under stress.
      - Critical-path bonus retained but computed via safe percentile ranking even for N=1.
      - IQR percentiles tightened to (20/80) for stronger outlier resilience.
      - Wait saturation uses arctan (Parent 2), but fairness term is *suppressed* under hard DDL mode to prioritize deadline compliance.
      - All operations guarded against NaN/inf/zero; deterministic and finite output.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Linear urgency gate: monotonic, bounded [-1,1], zero-crossing at slack=0
    median_abs_slack = np.maximum(np.abs(np.median(slack)), eps)
    urgency_gate = np.clip(slack / (median_abs_slack + eps), -1.0, 1.0)
    urgency_gate = -urgency_gate  # flip so negative slack → positive urgency
    
    # Feasibility guard: activate strict DDL mode only when slack is critically low
    ddl_critical_mask = (slack < PARAMS["ddl_feasibility_threshold"]).astype(float)
    
    # Amplify urgency under hard deadline stress
    urgency_amplified = urgency_gate * (1.0 + PARAMS["feasibility_guard_strength"] * ddl_critical_mask)
    norm_urgency = iqr_normalize(urgency_amplified)
    
    # Host-load–aware energy normalization: inflate marginal energy under uncertainty
    host_load_adjusted_energy = min_incremental_energy * (1.0 + PARAMS["host_load_sensitivity"] * uncertainty)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = host_load_adjusted_energy / duration
    # Normalize only outside hard DDL mode to avoid conflicting objectives
    norm_energy_eff = iqr_normalize(energy_per_duration) * (1.0 - ddl_critical_mask)
    
    # Successor bottleneck coupling: prioritize tasks whose long duration blocks high-work/high-rank successors
    bottleneck_pressure = duration * (remaining_work + upward_rank + eps)
    norm_bottleneck = iqr_normalize(bottleneck_pressure) * (1.0 - ddl_critical_mask)
    
    # Critical-path bonus: activated only for top percentile of upward_rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank * (1.0 - ddl_critical_mask)
    
    # Ready wait time: smooth saturation via arctan (bounded [0,1])
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = iqr_normalize(wait_saturation) * (1.0 - ddl_critical_mask)
    
    # Adaptive DDL-protection logic: activates only when slack is *below median* AND uncertainty is high relative to population
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_mask = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Uncertainty-normalized term for amplification under risk
    norm_uncertainty = iqr_normalize(uncertainty)
    
    # Unified risk-aware urgency amplification: multiplicative gate on norm_urgency
    # Only active when both deadline pressure (slack < 0) and risk (uncertainty high) co-occur
    risk_amplification_mask = (
        (slack < 0) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    risk_amplified_urgency = norm_urgency * (1.0 + PARAMS["uncertainty_slack_interaction_strength"] * norm_uncertainty * risk_amplification_mask)
    
    # Final score: smaller = higher priority
    # Under DDL-critical mode (ddl_critical_mask == 1), suppress all non-urgency terms
    # This enforces hard deadline compliance before optimizing energy or fairness
    score = (
        risk_amplified_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - critical_bonus
        - norm_wait
        + ddl_protection_mask * norm_uncertainty
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
