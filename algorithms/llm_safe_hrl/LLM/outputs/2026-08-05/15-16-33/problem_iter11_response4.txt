def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust joint DDL-risk gating and adaptive normalization
    with Parent 1's principled anti-starvation via bounded reciprocal scaling — but re-engineered
    to avoid over-smoothing: uses tunable saturation time instead of median-based ramp.
    Key structural improvements:
    1. Unified risk coupling: multiplicative urgency × uncertainty (Parent 2) + additive DDL-risk amplification
       only when slack < 0 (tighter than median-based condition), preserving hard deadline semantics.
    2. Bounded reciprocal anti-starvation: 1/(1 + wait/saturation)^exponent, using explicit saturation time
       (more interpretable and controllable than median-based ramp or arctan).
    3. Adaptive normalization retains IQR→min-max fallback (Parent 2) but applies it consistently across all terms.
    4. Removed critical-path bonus (evidence shows no DDL gain, harms energy); replaced with direct
       urgency-driven bottleneck coupling: duration × upward_rank × (1 + urgency) × (1 + uncertainty).
    All operations are finite, deterministic, and use only [-2,-1,0,1,2] literals.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max when dispersion is low
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR too small (low dispersion), else use IQR
        if iqr < eps:
            x_min, x_max = np.min(x), np.max(x)
            denom = x_max - x_min
            if denom < eps:
                return np.zeros_like(x)
            return (x - x_min) / (denom + eps)
        else:
            denom = iqr
            return (x - center) / (denom + eps)
    
    # Sigmoid urgency gate: maps slack → [0, 1] where negative slack → near 1.0 (high urgency)
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered,
                            -PARAMS["sigmoid_clip_bound"],
                            PARAMS["sigmoid_clip_bound"])
    urgency = 1.0 / (1.0 + np.exp(sigmoid_input))  # range [0, 1]
    
    # Normalize urgency
    norm_urgency = adaptive_normalize(urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Successor bottleneck: duration × upward_rank × (1 + urgency) × (1 + uncertainty)
    # Amplifies under both deadline pressure AND uncertainty — captures risk-aware blocking
    bottleneck_pressure = duration * upward_rank * (1.0 + urgency + eps) * (1.0 + uncertainty + eps)
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Bounded reciprocal anti-starvation: smooth, monotonic, tunable saturation
    # Avoids median dependency; uses explicit physical time scale
    wait_scaled = ready_wait_time / (PARAMS["wait_ramp_saturation_time"] + eps)
    wait_reciprocal = 1.0 / (1.0 + wait_scaled)
    norm_wait = adaptive_normalize(wait_reciprocal)
    
    # Joint DDL-risk gate: activated only when slack < 0 (hard violation risk) AND uncertainty high
    # More precise than median-based condition: directly enforces hard deadline constraint
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    ddl_risk_gate = (
        (slack < 0.0) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Risk-coupled urgency: multiplicative urgency × uncertainty (Parent 2) + additive amplification
    # Only active under hard deadline risk — preserves strict DDL semantics
    risk_coupled = urgency * uncertainty
    norm_risk = adaptive_normalize(risk_coupled)
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline safety (urgency) > risk-amplified bottleneck > DDL-risk amplification > energy > fairness
    score = (
        norm_urgency  # Primary deadline signal
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors under joint risk
        + ddl_risk_gate * norm_risk  # Amplify only under hard deadline violation risk + high uncertainty
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Energy efficiency
        - norm_wait  # Favor shorter-waiting tasks via bounded reciprocal
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
