def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's high-performance structure with Parent 1's robust regime-aware normalization.
    
    Key innovations:
      - Introduces *uncertainty-regime gating* for normalization: uses global quantile (not median) to split low/high uncertainty,
        enabling more granular dispersion adaptation than binary median split.
      - Replaces fallback min-max with *robust range-based dispersion* (max-min) only when IQR is near-zero — avoids degenerate cases
        where percentiles collapse (e.g., identical values).
      - Retains Parent 2's single-vectorized joint_ddl_pressure mask and variance-gated urgency modulation.
      - Adds regime-aware bottleneck amplification: applies uncertainty exponent only in high-uncertainty regime, not just under feasibility guard.
      - All numeric literals strictly {-2,-1,0,1,2}; no hidden constants; exactly one conditional logic path (via boolean masks).
      - Maintains deterministic, finite, shape-(N,) output with full NaN/inf protection.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Global uncertainty regime flag: high-uncertainty tasks use different normalization behavior
    uncertainty_quantile = np.quantile(uncertainty, PARAMS["regime_switch_uncertainty_quantile"]) if N > 0 else 0.0
    high_uncertainty_mask = (uncertainty > uncertainty_quantile).astype(float)
    
    # Robust adaptive normalization: IQR-based, with fallback to range (not min-max) when dispersion vanishes
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use range (max-min) instead of min-max fallback — more numerically stable for normalization
        dispersion = iqr if iqr > eps else (np.max(x) - np.min(x))
        denom = dispersion if dispersion > eps else eps
        
        return (x - center) / (denom + eps)
    
    # Linear urgency proxy: (median_slack - slack), clipped to [0, 2] for bounded influence
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    
    # Slack variance sufficiency: activate modulation only if variance is high enough relative to range
    slack_range = np.max(slack) - np.min(slack) if N > 0 else eps
    slack_var = np.var(slack) if N > 1 else 0.0
    variance_sufficient = (slack_var > PARAMS["slack_variance_activation_threshold"] * (slack_range + eps)**2).astype(float)
    
    # Joint DDL pressure: slack < median AND uncertainty > threshold → single vectorized mask
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    joint_ddl_pressure = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Modulated urgency: boost only when both variance sufficient and joint pressure holds
    modulated_urgency = urgency_linear * (1.0 + PARAMS["urgency_uncertainty_coupling_strength"] * joint_ddl_pressure * variance_sufficient)
    norm_urgency = adaptive_normalize(modulated_urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration * upward_rank * (1 + urgency), amplified only in high-uncertainty regime
    bottleneck_pressure = duration * upward_rank * (1.0 + modulated_urgency + eps)
    # Apply uncertainty exponent *only* in high-uncertainty regime (not just feasibility guard)
    bottleneck_pressure = bottleneck_pressure * np.power(1.0 + uncertainty, 
                                                         PARAMS["bottleneck_uncertainty_amplification"] * high_uncertainty_mask)
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Ready wait saturation: bounded sigmoid using only allowed literals and declared param
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)  # Safe exp bounds
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = adaptive_normalize(wait_saturation)
    
    # DDL risk amplifier: joint pressure × normalized uncertainty (enhanced by regime)
    norm_uncertainty = adaptive_normalize(uncertainty)
    ddl_risk_amplifier = joint_ddl_pressure * norm_uncertainty * (1.0 + high_uncertainty_mask)
    
    # Final score: smaller = higher priority
    score = (
        norm_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
        + ddl_risk_amplifier
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
