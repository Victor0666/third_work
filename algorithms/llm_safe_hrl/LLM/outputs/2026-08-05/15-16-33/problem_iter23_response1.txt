def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's robust deadline enforcement with Parent 1's critical-path coupling.
    
    Key structural improvements:
      - Introduces tunable `deadline_violation_penalty_weight` to scale raw neg_slack — enables CMA-ES to balance hard constraint dominance vs. other objectives.
      - Adds `critical_path_pressure_weight`-scaled normalized cp_coupling (upward_rank * remaining_work) as a dedicated structural term — preserves HEFT-like criticality without bottleneck entanglement.
      - Removes all gating logic (feasibility_mask) from bottleneck pressure: eliminates brittle conditional suppression that degraded performance under high uncertainty.
      - Retains Parent 2's pre-normalized neg_slack as dominant term and bounded sigmoid wait saturation for anti-starvation.
      - All numeric literals strictly in {-2,-1,0,1,2}; no branching beyond clipping/min/max; fully deterministic.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Uncertainty-aware adaptive normalization: dispersion scaled by std(uncertainty)
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        
        unc_std = np.std(uncertainty) if N > 1 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
        
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = dispersion if dispersion > eps else fallback_range
        
        return (x - center) / (denom + eps)
    
    # Pre-normalization hard-deadline violation signal: clipped negative slack (dominant term)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Linear urgency proxy: (median_slack - slack), then feasibility-preserving cap
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    
    # Feasibility-preserving urgency cap: clamp to max non-negative slack raised to learned exponent
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.minimum(urgency_linear, urgency_cap)
    
    norm_urgency = adaptive_normalize(urgency_linear)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency_linear) × (1 + uncertainty)^amplification
    # No gating: avoids fragile suppression under negative slack while preserving urgency coupling
    bottleneck_pressure = duration * upward_rank * remaining_work * (1.0 + urgency_linear + eps)
    
    # Uncertainty amplification: power-law boost using normalized uncertainty
    unc_normalized = adaptive_normalize(uncertainty)
    bottleneck_pressure = bottleneck_pressure * np.power(1.0 + unc_normalized, 
                                                         PARAMS["bottleneck_uncertainty_amplification"])
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Bounded sigmoid wait saturation: smooth anti-starvation boost
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = adaptive_normalize(wait_saturation)
    
    # Critical-path coupling: pure structural importance (HEFT-inspired), ungated and unamplified
    cp_coupling = upward_rank * remaining_work
    norm_cp_coupling = adaptive_normalize(cp_coupling)
    
    # Final score: smaller = higher priority
    # Dominant ordering: hard DDL violation > urgency > bottleneck > energy > fairness > critical-path
    score = (
        PARAMS["deadline_violation_penalty_weight"] * neg_slack
        + norm_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
        + PARAMS["critical_path_pressure_weight"] * norm_cp_coupling
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
