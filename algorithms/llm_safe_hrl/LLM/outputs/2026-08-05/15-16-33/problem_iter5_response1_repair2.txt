def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's smooth sigmoid urgency with Parent 1's robust convex fallback for extreme lateness,
    enhanced by adaptive bottleneck coupling, refined DDL-protection gating, and arctan-saturated wait-time fairness.
    
    Key structural improvements:
      - Hybrid slack modeling: primary sigmoid urgency for general deadline sensitivity, augmented with convex penalty *only* when slack < severe_lateness_threshold
        (prevents sigmoid saturation loss of discrimination under severe violation)
      - Adaptive bottleneck coupling: uses (duration * (remaining_work + upward_rank)) but normalized *before* multiplication with coupling weight
        to avoid scale explosion and improve gradient stability
      - Refined DDL-protection gate: now includes *relative slack deficit* (median_slack - slack) scaled by uncertainty factor, capped at relative_deficit_cap
      - Wait-time fairness: arctan saturation preserves monotonicity while bounding influence; avoids over-prioritizing long-waiting tasks at cost of deadlines
      - All normalizations use tunable IQR percentiles; no hard thresholds or unbounded growth
      - Critical-path bonus remains percentile-gated for robustness against outliers
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Hybrid slack signal: sigmoid urgency dominates, convex penalty activates only for severe lateness (slack < severe_lateness_threshold)
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered, 
                            -PARAMS["sigmoid_clip_bound"], 
                            PARAMS["sigmoid_clip_bound"])
    urgency_gate = 2.0 / (1.0 + np.exp(sigmoid_input)) - 1.0  # range [-1, 1]
    
    # Convex fallback for severe violations: only active when slack < severe_lateness_threshold
    convex_penalty = np.where(
        slack < PARAMS["severe_lateness_threshold"],
        (-slack) * 2.0 + (slack ** 2) * PARAMS["slack_convex_coeff"],
        0.0
    )
    
    # Combine: sigmoid provides fine-grained urgency; convex adds discriminative penalty for hard violations
    hybrid_slack_signal = urgency_gate + convex_penalty
    norm_slack_signal = iqr_normalize(hybrid_slack_signal)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor bottleneck coupling: normalize components *before* interaction to stabilize scale
    norm_duration = iqr_normalize(duration)
    norm_remaining_work = iqr_normalize(remaining_work)
    norm_upward_rank = iqr_normalize(upward_rank)
    # Interaction: weighted sum of normalized features, then coupled
    bottleneck_pressure = (
        norm_duration + norm_remaining_work + norm_upward_rank
    )
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    
    # Critical-path bonus: activated only for top percentile of upward_rank (percentile-ranked)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    critical_bonus = critical_gate * norm_upward_rank
    
    # Ready wait time: smooth saturation via arctan (bounded [0,1])
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = iqr_normalize(wait_saturation)
    
    # Refined DDL-protection term: continuous activation based on relative slack deficit × uncertainty factor
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    # Relative deficit: how far below median slack; capped to avoid extreme values
    relative_deficit = np.clip(median_slack - slack, 0.0, PARAMS["relative_deficit_cap"])
    # Activation strength: scales with both deficit magnitude and uncertainty level (using declared factor)
    ddl_protection_strength = relative_deficit * (uncertainty / (max_uncertainty + eps)) * PARAMS["ddl_protection_uncertainty_factor"]
    # Normalize to align with other terms
    norm_ddl_protection = iqr_normalize(ddl_protection_strength)
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline safety (hybrid slack) > bottleneck release > energy > critical path > fairness > risk amplification
    score = (
        norm_slack_signal  # Primary deadline signal (smooth + convex fallback)
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors robustly
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks only when ranked highly
        - norm_wait  # Reward recently arrived tasks (fairness)
        + norm_ddl_protection  # Amplify protection where slack deficit and uncertainty jointly indicate risk
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
