def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces hard feasibility gating with soft exponential slack decay on energy,
    reintroduces conditional slack_sigmoid_offset under high uncertainty, and upgrades bottleneck coupling to
    additive uncertainty interaction (not multiplicative) for monotonic, well-bounded risk amplification.
    
    Key structural improvements:
      - Soft exponential decay on energy term: weight = exp(-decay_rate * max(0, slack)), smooth & differentiable
      - Conditional slack offset: applied only when uncertainty exceeds threshold, improving urgency calibration in noisy regimes
      - Additive uncertainty interaction: bottleneck_pressure + weight * norm_uncertainty * bottleneck_pressure,
        preserving sign and avoiding zero-crossing artifacts
      - Tunable IQR percentiles (22/78) further improve outlier robustness
      - All operations guarded; deterministic; finite output guaranteed
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using tunable percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Sigmoid urgency gate: smooth, differentiable
    # Offset applied conditionally only under high uncertainty to avoid overfitting noise
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    high_uncertainty_mask = (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty).astype(float)
    effective_offset = high_uncertainty_mask * PARAMS["slack_sigmoid_offset"]
    slack_centered = slack - effective_offset
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered, 
                            -PARAMS["sigmoid_clip_bound"], 
                            PARAMS["sigmoid_clip_bound"])
    urgency_gate = 2.0 / (1.0 + np.exp(sigmoid_input)) - 1.0  # range [-1, 1]
    norm_urgency = iqr_normalize(urgency_gate)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor bottleneck coupling: duration * (remaining_work + upward_rank + eps)
    # Additive uncertainty interaction: preserves monotonicity and avoids pathological scaling
    bottleneck_pressure = duration * (remaining_work + upward_rank + eps)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    norm_uncertainty = iqr_normalize(uncertainty)
    bottleneck_with_risk = norm_bottleneck + PARAMS["uncertainty_interaction_weight"] * norm_uncertainty * norm_bottleneck
    
    # Critical-path bonus: activated only for top percentile of upward_rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Ready wait time: smooth saturation via arctan (bounded [0,1])
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = iqr_normalize(wait_saturation)
    
    # Soft exponential slack decay for energy term weight: decays smoothly from 1.0 → 0 as slack increases
    slack_positive = np.maximum(slack, 0.0)
    energy_weight = np.exp(-PARAMS["soft_slack_energy_decay_rate"] * slack_positive)
    
    # DDL-protection gate: activates only when slack is tight relative to median AND uncertainty is high
    median_slack = np.median(slack)
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Final score: smaller = higher priority
    # Urgency dominates; bottleneck unblocks successors (additively risk-modulated); energy softly gated;
    # critical bonus prioritizes path; wait reward ensures fairness; DDL protection adds margin under stress
    score = (
        norm_urgency  # Primary deadline pressure
        + PARAMS["successor_bottleneck_coupling"] * bottleneck_with_risk  # Monotonic risk-aware blocking
        + energy_weight * PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Smooth energy penalty decay
        - critical_bonus  # Boost critical path
        - norm_wait  # Reward recent arrivals
        + ddl_protection_active * norm_uncertainty  # Risk margin under stress
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
