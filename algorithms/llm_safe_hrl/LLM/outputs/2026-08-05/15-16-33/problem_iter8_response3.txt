def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robustness with Parent 1's hard-deadline enforcement.
    
    Key innovations:
      - Dual-mode urgency: smooth sigmoid baseline + multiplicative amplification *only* under joint stress (slack<0 ∧ high uncertainty)
      - Capped bottleneck pressure to prevent numerical instability while preserving structural coupling
      - Adaptive normalization with tighter IQR percentiles (22/78) and refined fallback logic
      - Critical-path bonus remains percentile-gated but computed safely for N=1
      - All operations guarded against NaN/inf/zero; deterministic and finite output
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: use IQR unless dispersion is too low → fallback to min-max
    def robust_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        x_range = np.max(x) - np.min(x)
        # If IQR is negligible relative to range, use min-max to avoid division-by-near-zero
        use_minmax = (iqr < PARAMS["minmax_fallback_threshold"] * (x_range + eps))
        if use_minmax:
            x_min, x_max = np.min(x), np.max(x)
            denom = x_max - x_min
            norm_x = (x - x_min) / (denom + eps)
        else:
            center = np.median(x)
            denom = iqr if iqr > eps else np.max(np.abs(x - center)) + eps
            norm_x = (x - center) / (denom + eps)
        return norm_x
    
    # Base sigmoid urgency: smooth, differentiable, centered at offset, steepness tunable
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered, 
                            -PARAMS["sigmoid_clip_bound"], 
                            PARAMS["sigmoid_clip_bound"])
    urgency_base = 2.0 / (1.0 + np.exp(sigmoid_input)) - 1.0  # range [-1, 1]
    
    # Multiplicative amplification only under joint DDL stress: slack < 0 AND uncertainty above threshold
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_stress_mask = (
        (slack < 0.0) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    urgency_amplified = urgency_base * (1.0 + PARAMS["urgency_amplification_under_ddl_stress"] * ddl_stress_mask)
    norm_urgency = robust_normalize(urgency_amplified)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = robust_normalize(energy_per_duration)
    
    # Successor bottleneck coupling: capped to prevent explosion near zero duration or extreme work
    bottleneck_pressure_uncapped = duration * (remaining_work + upward_rank + eps)
    bottleneck_pressure = np.minimum(bottleneck_pressure_uncapped, PARAMS["bottleneck_pressure_cap"])
    norm_bottleneck = robust_normalize(bottleneck_pressure)
    
    # Critical-path bonus: activated only for top percentile of upward_rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = robust_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Ready wait time: smooth saturation via arctan (bounded [0,1])
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = robust_normalize(wait_saturation)
    
    # Final score: smaller = higher priority
    # Deadline safety dominates via amplified urgency + multiplicative protection
    score = (
        norm_urgency  # Primary deadline pressure signal, amplified under joint stress
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks
        - norm_wait  # Reward recently arrived tasks (fairness)
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
