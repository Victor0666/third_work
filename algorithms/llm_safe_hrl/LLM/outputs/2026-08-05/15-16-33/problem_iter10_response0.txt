def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating counterfactual evidence:
    - Removes inactive parameters (wait_saturation_time, critical_rank_percentile) per inactivity analysis
    - Replaces arctan wait saturation with bounded linear ramp + soft clamp to reduce complexity
    - Introduces multiplicative urgency × uncertainty coupling only under joint DDL pressure (tight slack AND high uncertainty)
    - Uses adaptive min-max fallback when IQR dispersion is low (more robust than pure IQR)
    - Eliminates critical-path bonus (no cross-scenario evidence supporting it) and replaces with bottleneck-coupled urgency
    - Prioritizes tasks whose execution unblocks successors *and* reduces deadline risk simultaneously
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max if dispersion too low
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR is negligible relative to range
        x_min, x_max = np.min(x), np.max(x)
        range_val = x_max - x_min
        if iqr < eps * (range_val + eps):
            denom = range_val + eps
            return (x - x_min) / denom
        else:
            denom = iqr + eps
            return (x - center) / denom
    
    # Sigmoid urgency gate: smooth, centered at offset, clamped to avoid overflow
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered,
                            -PARAMS["sigmoid_clip_bound"],
                            PARAMS["sigmoid_clip_bound"])
    urgency_gate = 2.0 / (1.0 + np.exp(sigmoid_input)) - 1.0  # [-1, 1]
    
    # Normalize urgency
    norm_urgency = adaptive_normalize(urgency_gate)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Successor bottleneck pressure: duration × upward_rank × (1 + urgency) to amplify under deadline stress
    # This couples critical path importance with urgency — high rank matters most when slack is tight
    bottleneck_pressure = duration * upward_rank * (1.0 + np.clip(urgency_gate, 0.0, 1.0) + eps)
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Joint DDL-protection gate: activates only when slack is tight AND uncertainty is high
    median_slack = np.median(slack) if N > 0 else 0.0
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Multiplicative urgency × uncertainty coupling under joint DDL pressure
    # Only amplifies risk-aware prioritization where both signals are strong
    urgency_uncertainty_coupling = ddl_protection_active * urgency_gate * uncertainty
    norm_coupling = adaptive_normalize(urgency_uncertainty_coupling)
    
    # Ready wait time: bounded linear ramp with soft clamp (simpler, more stable than arctan)
    # Saturates at 1.0 after wait exceeds threshold, avoids unbounded growth
    wait_normalized = np.clip(ready_wait_time / (np.max(ready_wait_time + eps)), 0.0, 1.0)
    norm_wait = adaptive_normalize(wait_normalized)
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline safety (urgency + coupling) > bottleneck release > energy > fairness (wait)
    score = (
        norm_urgency  # Primary deadline signal
        + norm_coupling  # Risk-amplified urgency only under joint pressure
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors under urgency
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Energy efficiency
        - norm_wait  # Anti-starvation: reward long-waiting tasks (smaller score = higher priority)
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
