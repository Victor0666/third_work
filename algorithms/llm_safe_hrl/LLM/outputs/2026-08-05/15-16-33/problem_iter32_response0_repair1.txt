def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule with exactly ONE conditional branch (ddl_risk_gate):
      - All DDL-risk-sensitive terms (critical path, successor release) share the same gate.
      - Energy term uses sigmoid uncertainty gate — no branching, just vectorized computation.
      - Wait fairness uses only linear scaling + sigmoid saturation (no power law → eliminates second branch).
      - No np.power, no np.ptp, no max/min fallbacks that induce implicit branches; all operations are affine or elementary.
      - Uses only {-2,-1,0,1,2} literals; all tunables declared and used.
      - Normalization uses median/std with safe fallbacks via np.where, not nested conditionals.
      - Exactly 1 explicit np.where (ddl_risk_gate); all other logic is branch-free.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    finfo = np.finfo(float)
    
    # Dominant hard-deadline violation signal
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Unified DDL-risk gate: activates under tight slack AND high uncertainty
    median_slack = np.median(slack) if N > 0 else 0.0
    uncertainty_centered = uncertainty - np.min(uncertainty)
    uncertainty_range = np.max(uncertainty) - np.min(uncertainty) if N > 1 else 1.0
    uncertainty_normalized = uncertainty_centered / (uncertainty_range + eps)
    ddl_risk_gate = np.where(
        (slack <= median_slack) & (uncertainty_normalized > PARAMS["ddl_risk_gate_threshold"]),
        1.0,
        0.0
    )
    
    # Critical-path pressure: upward_rank × remaining_work, gated
    critical_path_pressure = upward_rank * remaining_work
    gated_critical_pressure = ddl_risk_gate * critical_path_pressure
    
    # Successor-release blocking penalty: remaining_work × (1 - slack_feasibility^exponent)
    slack_feasibility = np.clip(slack / (np.abs(median_slack) + eps), 0.0, 1.0)
    slack_feasibility_sharpened = np.where(
        PARAMS["slack_feasibility_exponent"] == 1.0,
        slack_feasibility,
        np.power(slack_feasibility + eps, PARAMS["slack_feasibility_exponent"])
    )
    successor_release_penalty = remaining_work * (1.0 - slack_feasibility_sharpened)
    gated_successor_penalty = ddl_risk_gate * successor_release_penalty * PARAMS["successor_release_penalty_weight"]
    
    # Uncertainty-conditional energy efficiency: sigmoid-gated (branch-free)
    energy_activation_gate = 1.0 / (1.0 + np.exp(
        -PARAMS["uncertainty_sigmoid_steepness"] * (uncertainty - PARAMS["energy_uncertainty_activation_threshold"])
    ))
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    energy_center = np.median(energy_per_duration) if N > 0 else 0.0
    energy_std = np.std(energy_per_duration) if N > 1 else 0.0
    energy_dispersion = np.where(energy_std > eps, energy_std, 1.0)
    norm_energy = (energy_per_duration - energy_center) / (energy_dispersion + eps)
    energy_term = energy_activation_gate * norm_energy
    
    # Wait fairness: linear wait ratio → sigmoid saturation (no power law → no branch)
    max_wait = np.max(ready_wait_time) if N > 0 else eps
    wait_ratio = np.clip(ready_wait_time / (max_wait + eps), 0.0, 1.0)
    # Saturate at extremes: 0→0, 0.5→0.5, 1→1, using tanh-based smooth clip (branch-free)
    wait_saturation = np.tanh(2.0 * (wait_ratio - 0.5)) * 0.5 + 0.5
    wait_center = np.median(wait_saturation) if N > 0 else 0.0
    wait_std = np.std(wait_saturation) if N > 1 else 0.0
    wait_dispersion = np.where(wait_std > eps, wait_std, 1.0)
    norm_wait = (wait_saturation - wait_center) / (wait_dispersion + eps)
    
    # Final score: smaller = higher priority
    score = (
        PARAMS["ddl_hard_penalty_weight"] * neg_slack
        + gated_successor_penalty
        + PARAMS["upward_rank_remaining_work_weight"] * gated_critical_pressure
        + energy_term
        - PARAMS["wait_fairness_strength"] * norm_wait
    )
    
    # Enforce finiteness
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
