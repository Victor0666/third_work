def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule addressing reflection insights:
      - Replaces power-law DDL penalty with bounded linear urgency (slope param) for balanced hard-constraint fidelity.
      - Restores raw `neg_slack` as dominant additive term — preserves monotonic, interpretable deadline violation signal.
      - Introduces *host-load-aware energy gating*: energy term is suppressed when `min_incremental_energy` is high *and*
        `ready_wait_time` is low (indicating low contention), preventing energy-driven scheduling that risks DDL violation.
      - Keeps all robust MAD normalizations, bottleneck-aware duration scaling, and power-law wait decay.
      - Adds explicit feasibility-aware coupling: energy term scaled by `(1 - gate)` where gate activates under low-load + high-energy conditions.
      - All operations finite, deterministic, and use only {-2,-1,0,1,2} literals.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust MAD-based normalization
    def mad_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        abs_dev = np.abs(x - center)
        mad = np.median(abs_dev) if N > 0 else eps
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = mad if mad > eps else fallback_range
        return (x - center) / (denom + eps)
    
    # Median slack for relative urgency scaling
    median_slack = np.median(slack) if N > 0 else 0.0
    
    # Conditional DDL risk gate: activate only when slack < threshold * median_slack
    ddl_risk_mask = (slack < PARAMS["ddl_risk_gate_threshold"] * (median_slack + eps)).astype(float)
    
    # Raw negative slack — dominant, unnormalized hard-deadline violation signal
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Linear urgency penalty under DDL risk (replaces power-law; avoids over-penalization)
    urgency_score = neg_slack * PARAMS["ddl_urgency_linear_slope"] * ddl_risk_mask
    
    # Critical-path pressure: upward_rank × remaining_work, gated and scaled
    critical_pressure = upward_rank * remaining_work
    critical_pressure = critical_pressure * (1.0 + ddl_risk_mask * (PARAMS["critical_path_coupling"] - 1.0))
    norm_critical_pressure = mad_normalize(critical_pressure)
    
    # Bottleneck-aware duration scaling: duration × critical_pressure, only under DDL risk
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    bottleneck_pressure = duration * critical_pressure * ddl_risk_mask
    norm_bottleneck = mad_normalize(bottleneck_pressure)
    
    # Host-load-aware energy gating: suppress energy optimization when load is low *and* energy cost is high
    # Normalized energy and wait — high energy + low wait → high gate → suppress energy term
    norm_energy_cost = mad_normalize(min_incremental_energy)
    norm_wait_level = mad_normalize(ready_wait_time)
    # Gate activates when energy cost is above median AND wait time is below median (low contention)
    energy_feasibility_gate = (
        (norm_energy_cost > 0.0).astype(float) *
        (norm_wait_level < 0.0).astype(float) *
        (np.abs(norm_energy_cost) > PARAMS["energy_feasibility_gate_threshold"]).astype(float)
    )
    
    # Energy term: marginal energy per effective duration, robustly normalized and gated
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy = mad_normalize(energy_per_duration) * PARAMS["energy_normalization_scale"]
    # Apply feasibility gate: reduce energy priority when scheduling under low load but high energy cost
    norm_energy = norm_energy * (1.0 - energy_feasibility_gate)
    
    # Uncertainty-coupled slack deficit: only active under DDL risk
    norm_slack_deficit = mad_normalize(neg_slack)
    unc_coupled_deficit = norm_slack_deficit * np.power(1.0 + uncertainty, PARAMS["uncertainty_slack_coupling"])
    unc_coupled_deficit = unc_coupled_deficit * ddl_risk_mask
    
    # Bounded power-law wait decay: smooth anti-starvation
    max_wait = np.max(ready_wait_time) if N > 0 else eps
    wait_ratio = np.clip(ready_wait_time / (max_wait + eps), 0.0, 1.0)
    wait_decay = np.power(wait_ratio + eps, PARAMS["wait_decay_exponent"])
    wait_priority = 1.0 - wait_decay
    norm_wait = mad_normalize(wait_priority)
    
    # Final score: smaller = higher priority
    # Dominance order: raw DDL violation > linear urgency > bottleneck > critical path > gated energy > uncertainty deficit > fairness
    score = (
        neg_slack  # Raw violation — strongest, monotonic, unnormalized
        + urgency_score
        + norm_bottleneck
        + norm_critical_pressure
        + norm_energy
        + mad_normalize(unc_coupled_deficit)
        - norm_wait
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
