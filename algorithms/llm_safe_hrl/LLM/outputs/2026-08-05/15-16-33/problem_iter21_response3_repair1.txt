def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Rank-based normalization (replacing percentile/IQR) for cross-scenario stability;
      - Hard deadline-aware criticality gating: suppresses non-critical-path tasks when slack is tight;
      - Decoupled energy term using only min_incremental_energy, normalized via MAD for outlier resilience;
      - All numeric literals strictly in {-2,-1,0,1,2}; no ListComp or unbounded loops.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Hard deadline guard: assign large finite penalty to infeasible tasks
    finfo = np.finfo(float)
    score = np.full(N, finfo.max, dtype=float)
    feasible_mask = slack >= -eps
    
    if not np.any(feasible_mask):
        return score
    
    # Robust rank-based normalization: maps values to [-1,1] via sorting-based rank approximation
    def rank_normalize(x):
        x = np.copy(x)
        if N <= 1 or np.all(x == x[0]):
            return np.zeros_like(x, dtype=float)
        # Compute rank via argsort: lower value → lower rank index → lower percentile
        sorted_idx = np.argsort(x)
        ranks = np.empty_like(sorted_idx, dtype=float)
        ranks[sorted_idx] = np.arange(N, dtype=float)
        # Normalize to [-1, 1]: (2 * rank / (N-1)) - 1
        norm_ranks = (2.0 * ranks / (N - 1.0)) - 1.0
        return np.clip(norm_ranks, -1.0, 1.0)
    
    # Criticality gating: zero out non-critical-path contribution when slack is tight
    median_slack = np.median(slack) if N > 0 else 0.0
    tight_slack_mask = (slack < (median_slack * (1.0 - PARAMS["slack_criticality_threshold"]))) & (slack >= -eps)
    # Only apply gating to non-critical-path tasks: those with low upward_rank relative to max
    max_upward = np.max(upward_rank) if N > 0 else 1.0
    non_critical_mask = (upward_rank < (max_upward * 0.2)) & tight_slack_mask
    critical_gate = np.where(non_critical_mask, 0.0, 1.0)
    
    # Power-law urgency: (median_slack - slack)^urgency_power, clipped for stability
    urgency_base = np.clip(median_slack - slack, 0.0, 2.0)
    urgency = np.power(urgency_base + eps, PARAMS["urgency_power"])
    norm_urgency = rank_normalize(urgency)
    
    # Energy efficiency: use raw min_incremental_energy, normalized via MAD for robustness
    med_energy = np.median(min_incremental_energy) if N > 0 else 0.0
    abs_devs = np.abs(min_incremental_energy - med_energy)
    mad_energy = np.median(abs_devs) if N > 0 else eps
    denom_energy = PARAMS["mad_scale_factor"] * (mad_energy + eps)
    norm_energy_eff = (min_incremental_energy - med_energy) / (denom_energy + eps)
    norm_energy_eff = np.clip(norm_energy_eff, -1.0, 1.0)
    
    # Critical-path bottleneck: upward_rank × remaining_work × sigmoid(steepness × uncertainty), gated
    unc_scaled = uncertainty * PARAMS["bottleneck_uncertainty_steepness"]
    unc_clipped = np.clip(unc_scaled, -2.0, 2.0)
    bottleneck_gate = 1.0 / (1.0 + np.exp(-unc_clipped))
    bottleneck_pressure = upward_rank * remaining_work * bottleneck_gate
    norm_bottleneck = rank_normalize(bottleneck_pressure)
    
    # Anti-starvation: invert rank-normalized wait time (longer wait → higher priority)
    norm_wait = rank_normalize(ready_wait_time)
    inv_wait = 1.0 - norm_wait
    
    # Compute final score *only* for feasible tasks, applying criticality gate to bottleneck & urgency
    local_score = (
        norm_urgency * critical_gate
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        + PARAMS["critical_path_coupling"] * norm_bottleneck * critical_gate
        - PARAMS["wait_fairness_weight"] * inv_wait
    )
    
    # Apply computed score only where feasible; penalize infeasible heavily but finitely
    score[feasible_mask] = local_score[feasible_mask]
    score[~feasible_mask] = PARAMS["ddl_hard_guard_weight"] * finfo.max
    
    # Enforce finiteness and sanitize
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
