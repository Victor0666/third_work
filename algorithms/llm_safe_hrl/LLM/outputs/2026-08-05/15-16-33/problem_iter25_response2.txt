def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with structural improvements guided by counterfactual evidence:
      - Introduces conditional DDL protection gate: amplifies negative slack penalty only when slack <= median_slack * threshold AND uncertainty > median_uncertainty.
      - Replaces sigmoid wait saturation with bounded power-law decay: avoids asymptotic flatness and improves discriminability for long-waiting tasks.
      - Uses MAD (median absolute deviation) instead of std for robust energy normalization — less sensitive to outlier VMs.
      - Couples critical path importance (upward_rank × remaining_work) only under feasible slack conditions, avoiding over-prioritization when deadlines are violated.
      - Removes all multiplicative risk couplings; uses additive, conditionally gated terms for clarity and stability.
      - All operations are finite, deterministic, and use only {-2,-1,0,1,2} literals.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization using MAD for dispersion (more stable than std under outliers)
    def robust_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        
        # MAD-based dispersion: more robust to extreme values in energy/uncertainty
        mad = np.median(np.abs(x - center)) if N > 0 else eps
        dispersion = PARAMS["energy_normalization_scale"] * (mad + eps)
        
        # Fallback to range if dispersion too small
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = dispersion if dispersion > eps else fallback_range
        
        return (x - center) / (denom + eps)
    
    # Pre-normalized hard-deadline violation signal: raw negative slack (dominant term)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Conditional DDL protection gate: activate only under high risk (tight slack AND high uncertainty)
    median_slack = np.median(slack) if N > 0 else 0.0
    median_uncertainty = np.median(uncertainty) if N > 0 else 0.0
    ddl_risk_condition = (
        (slack <= median_slack * PARAMS["uncertainty_slack_gate_threshold"]) & 
        (uncertainty > median_uncertainty)
    )
    ddl_protection = neg_slack * PARAMS["ddl_protection_strength"] * ddl_risk_condition.astype(float)
    
    # Critical path coupling: upward_rank × remaining_work, but only when slack >= 0 (feasible sub-deadlines)
    feasible_mask = (slack >= 0.0).astype(float)
    critical_path_pressure = upward_rank * remaining_work * feasible_mask
    norm_critical_path = robust_normalize(critical_path_pressure) * PARAMS["critical_path_coupling"]
    
    # Energy term: risk-adjusted marginal energy, normalized robustly
    norm_energy = robust_normalize(min_incremental_energy)
    
    # Wait-time boost: power-law decay instead of sigmoid — preserves discriminability
    # Scaled and clipped to avoid numerical instability
    wait_scaled = np.clip(ready_wait_time / (np.maximum(np.median(ready_wait_time), eps) + eps), 0.0, 2.0)
    wait_boost = np.power(wait_scaled + eps, PARAMS["wait_decay_exponent"])
    norm_wait = robust_normalize(wait_boost)
    
    # Final score: smaller = higher priority
    # Ordering: conditional DDL penalty > critical path > energy > fairness (wait boost subtracted)
    score = (
        ddl_protection  # Dominant, gated penalty for imminent deadline violations
        + norm_critical_path
        + norm_energy
        - norm_wait
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
