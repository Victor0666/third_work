def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL risk containment, critical-path awareness,
    and robust anti-starvation with bounded nonlinear interactions.
    
    Key innovations:
      - Uses *signed* slack with asymmetric penalty: strong convex penalty for negative slack,
        gentle linear reward for positive slack → prioritizes rescue of at-risk tasks.
      - Introduces a criticality gate: multiplies priority boost only when both high upward_rank
        AND tight slack co-occur — avoids over-prioritizing critical tasks in safe regimes.
      - Normalizes features via robust IQR-based scaling (not mean), resistant to outliers.
      - Couples duration and uncertainty multiplicatively to penalize high-uncertainty long tasks.
      - Applies smooth saturation to ready_wait_time to avoid unbounded starvation correction.
      - Uses rank-normalized upward_rank with small offset to ensure numerical stability.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Robust normalization: use IQR-based scaling (Q3-Q1) with epsilon guard
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q1, q3 = np.quantile(x, [0.25, 0.75])
        iqr = q3 - q1
        center = np.median(x)
        scale = iqr if iqr > eps_safe else eps_safe
        return (x - center) / scale

    # Duration: exec + comm, normalized robustly
    duration = min_exec_time + min_comm_time
    norm_duration = robust_normalize(duration)

    # Slack: signed, asymmetric penalty — convex for negative, linear for positive
    # Negative slack → large penalty (smaller score = higher priority ⇒ we *add* penalty)
    slack_penalty = np.where(
        slack < 0,
        -np.abs(slack) ** PARAMS["slack_penalty_exponent"],
        slack  # positive slack gives mild reward (reduces score)
    )
    norm_slack = robust_normalize(slack_penalty)

    # Upward rank: normalized with small offset to avoid division-by-zero or dominance collapse
    norm_rank = robust_normalize(upward_rank + PARAMS["rank_normalization_offset"])

    # Criticality gate: only activates boost when both rank is high *and* slack is tight (< median slack)
    median_slack = np.median(slack)
    is_critical_and_at_risk = (
        (upward_rank > np.median(upward_rank)) &
        (slack < median_slack)
    ).astype(float)
    criticality_boost = is_critical_and_at_risk * PARAMS["criticality_boost"]

    # Energy: normalized marginal fuzzy energy — lower energy → better → subtract
    norm_energy = robust_normalize(min_incremental_energy)

    # Uncertainty-duration coupling: penalize long + uncertain tasks
    # Bounded by [0, 2] to avoid explosion
    coupled_uncertainty = np.clip(
        norm_duration * uncertainty * PARAMS["duration_uncertainty_coupling"],
        -2.0, 2.0
    )

    # Ready wait time: smooth saturation via arctan (bounded, monotonic, differentiable)
    # Saturates near wait_saturation_threshold; prevents runaway priority for old tasks
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_threshold"] + eps_safe)
    norm_wait = (2.0 / np.pi) * np.arctan(wait_scaled)  # maps [0,∞) → [0,1]

    # Uncertainty-slack interaction: amplify slack penalty under high uncertainty
    # Only active when slack is negative
    slack_uncertainty_amplifier = np.where(
        slack < 0,
        uncertainty * PARAMS["uncertainty_slack_gate"],
        0.0
    )
    norm_slack_uncertainty = robust_normalize(slack_uncertainty_amplifier)

    # Final score: smaller = higher priority
    # Components ordered by lexicographic importance: DDL risk > critical path > energy > fairness
    score = (
        # Primary: Deadline risk dominates — strong penalty for negative slack
        norm_slack
        # Secondary: Amplify risk under uncertainty
        + norm_slack_uncertainty
        # Tertiary: Boost critical-at-risk tasks
        + criticality_boost * norm_rank
        # Quaternary: Favor energy-efficient assignments (subtract energy benefit)
        - PARAMS["energy_efficiency_scale"] * norm_energy
        # Quinary: Penalize long+uncertain tasks
        + coupled_uncertainty
        # Senary: Anti-starvation — add mild priority for long-waiting tasks
        - norm_wait  # subtract → increases priority for larger wait
    )

    # Ensure finite output and shape compliance
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
