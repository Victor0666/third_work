def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining strengths:
      - Pre-normalized negative-slack penalty (Parent 2) for hard-deadline dominance
      - Adaptive uncertainty-aware normalization (Parent 2) using std(uncertainty)
      - Tight-slack urgency gating with linear ramp (Parent 1) for calibrated urgency
      - Unified uncertainty gating via sigmoid (Parent 1) applied to both energy and bottleneck terms
      - Bounded sigmoid wait saturation (Parent 2) for robust anti-starvation
      - All nonlinearities clipped to [-2,2] or [0,2]; no unbounded ops; literals only {-2,-1,0,1,2}
      - Feasibility mask ensures infeasible tasks receive finite but dominant penalty
      - Explicit finite-domain clipping for all sigmoids and powers
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Feasibility mask: tasks violating deadline by more than epsilon are infeasible
    finfo = np.finfo(float)
    feasible_mask = slack >= -eps
    score = np.full(N, finfo.max, dtype=float)
    
    if not np.any(feasible_mask):
        return score
    
    # Adaptive uncertainty-aware normalization: dispersion scaled by std(uncertainty)
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x) if N > 0 else 0.0
        
        # Base dispersion: use std(uncertainty) as risk-aligned scaling factor
        unc_std = np.std(uncertainty) if N > 1 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
        
        # Fallback to range if dispersion too small
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = dispersion if dispersion > eps else fallback_range
        
        return (x - center) / (denom + eps)
    
    # Pre-normalized hard-deadline violation signal: clipped negative slack (dominant term)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Tight-slack urgency: linear ramp only in region defined by slack_sensitivity_threshold
    median_slack = np.median(slack) if N > 0 else 0.0
    tight_slack_boundary = median_slack * (1.0 - PARAMS["slack_sensitivity_threshold"])
    urgency_linear = np.clip(tight_slack_boundary - slack, 0.0, 2.0)
    urgency_mask = (slack <= tight_slack_boundary) & (slack >= -eps)
    urgency = np.where(urgency_mask, urgency_linear, 0.0)
    
    # Urgency capping via learned exponent on max non-negative slack (Parent 2 improvement)
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency = np.minimum(urgency, urgency_cap)
    
    norm_urgency = adaptive_normalize(urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s), uncertainty-gated
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    
    # Uncertainty gating: same steepness for energy and bottleneck (Parent 1 consistency)
    unc_centered = uncertainty - 0.5  # centered at 0.5 for balanced activation
    unc_clipped = np.clip(unc_centered, -2.0, 2.0)
    unc_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * unc_clipped))
    
    # Uncertainty-modulated energy term
    energy_uncertain = energy_per_duration * unc_gate
    norm_energy_eff = adaptive_normalize(energy_uncertain)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency) × (1 + normalized_uncertainty)^amplification
    bottleneck_pressure = duration * upward_rank * remaining_work * (1.0 + urgency + eps)
    
    # Uncertainty amplification: power-law boost using normalized uncertainty
    unc_normalized = adaptive_normalize(uncertainty)
    bottleneck_pressure = bottleneck_pressure * np.power(1.0 + unc_normalized, 
                                                         PARAMS["bottleneck_uncertainty_amplification"])
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Bounded sigmoid wait saturation: smooth anti-starvation boost
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = adaptive_normalize(wait_saturation)
    
    # Final score: smaller = higher priority
    # Dominant ordering: hard DDL violation > urgency > bottleneck > energy > fairness
    local_score = (
        PARAMS["ddl_hard_penalty_weight"] * neg_slack
        + norm_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
    )
    
    # Apply computed score only where feasible; penalize infeasible heavily but finitely
    score[feasible_mask] = local_score[feasible_mask]
    score[~feasible_mask] = PARAMS["ddl_hard_penalty_weight"] * finfo.max
    
    # Enforce finiteness and sanitize
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
