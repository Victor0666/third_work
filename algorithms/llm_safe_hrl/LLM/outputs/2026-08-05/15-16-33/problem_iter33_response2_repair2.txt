def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Piecewise slack sensitivity: linear penalty for slack > 0, exponential (slack^exponent) for slack <= 0 — preserves convexity while sharply penalizing violations.
      - Task-type–aware normalization: compute-bound (exec_ratio >= threshold) tasks use tighter outlier clipping; I/O-bound use wider range to preserve communication urgency.
      - Energy optimization gated by *both* slack >= 0 AND uncertainty below tunable low_uncertainty_gate_threshold, ensuring energy savings only under predictable conditions.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Compute-bound vs I/O-bound ratio: robust against zero denominator
    exec_comm_sum = min_exec_time + min_comm_time
    exec_comm_sum_safe = np.maximum(exec_comm_sum, eps)
    exec_ratio = min_exec_time / exec_comm_sum_safe
    
    # Dominant hard-constraint signal: piecewise slack penalty
    # For slack > 0: linear penalty (encourages early completion margin)
    # For slack <= 0: exponential penalty (sharp convex enforcement of hard DDL)
    linear_slack_penalty = np.clip(-slack, 0.0, None)  # same as before for clarity
    exp_slack_penalty = np.where(
        slack <= 0.0,
        np.power(np.abs(slack) + eps, PARAMS["slack_sensitivity_exponent"]),
        0.0
    )
    neg_slack = linear_slack_penalty + exp_slack_penalty
    
    # Median-based gating reference
    median_slack = np.median(slack) if N > 0 else 0.0
    
    # Robust uncertainty normalization for gating: [0,1] scaled
    unc_min, unc_max = np.min(uncertainty), np.max(uncertainty)
    unc_range = np.maximum(unc_max - unc_min, eps)
    unc_normalized = (uncertainty - unc_min) / (unc_range + eps)
    
    # DDL-protection activation: tight slack AND high normalized uncertainty
    ddl_protection_active = (slack <= median_slack) & (unc_normalized > PARAMS["ddl_protection_threshold"])
    
    # Unconditional critical-path pressure: upward_rank × remaining_work
    critical_pressure = upward_rank * remaining_work
    
    # Successor-release coupling: (1 + slack/median_slack) * (1 + uncertainty), bounded and safe
    slack_ratio = np.where(np.abs(median_slack) > eps, slack / (np.abs(median_slack) + eps), 0.0)
    slack_ratio = np.clip(slack_ratio, -2.0, 2.0)
    successor_release = (1.0 + slack_ratio) * (1.0 + uncertainty)
    
    # Bottleneck pressure: duration × critical_pressure × successor_release × uncertainty amplification
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    bottleneck_base = duration * critical_pressure * successor_release
    bottleneck_pressure = np.where(
        ddl_protection_active,
        bottleneck_base * np.power(1.0 + unc_normalized, PARAMS["bottleneck_uncertainty_amplification"]),
        bottleneck_base
    )
    
    # Energy term: prioritized only when slack >= 0 AND uncertainty is low (predictable execution)
    energy_term = np.where(
        (slack >= 0.0) & (unc_normalized < PARAMS["low_uncertainty_gate_threshold"]),
        min_incremental_energy,
        0.0
    )
    
    # Task-type–aware normalization: tighter clipping for compute-bound tasks
    def task_aware_normalize(x, is_compute_bound):
        x = np.copy(x)
        x_min = np.min(x) if N > 0 else 0.0
        x_max = np.max(x) if N > 0 else 1.0
        range_val = np.maximum(x_max - x_min, eps)
        # Tighter clipping for compute-bound (higher precision needed), wider for I/O-bound
        clip_factor = np.where(is_compute_bound, PARAMS["compute_bound_clipping_scale"], 1.0) * PARAMS["wait_saturation_clipping_factor"]
        clip_offset = clip_factor * range_val
        x_clipped = np.clip(x, x_min - clip_offset, x_max + clip_offset)
        return (x_clipped - x_min) / (range_val + eps)
    
    is_compute_bound = exec_ratio >= PARAMS["compute_io_ratio_threshold"]
    norm_critical = task_aware_normalize(critical_pressure, is_compute_bound)
    norm_bottleneck = task_aware_normalize(bottleneck_pressure, is_compute_bound)
    norm_energy = task_aware_normalize(energy_term, is_compute_bound)
    
    # Anti-starvation: bounded sigmoid on ready_wait_time
    wait_scaled = ready_wait_time / (1.0 + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = task_aware_normalize(wait_saturation, is_compute_bound)
    
    # Final score: smaller = higher priority
    # Priority order: DDL violation > critical path > bottleneck > energy (if safe & predictable) > fairness
    score = (
        neg_slack  # Hard constraint enforcement — highest priority, now exponentially sharpened for violations
        + PARAMS["upward_rank_remaining_work_weight"] * norm_critical
        + PARAMS["successor_release_coupling"] * norm_bottleneck
        + PARAMS["energy_priority_weight"] * norm_energy
        - norm_wait
    )
    
    # Enforce finiteness and shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
