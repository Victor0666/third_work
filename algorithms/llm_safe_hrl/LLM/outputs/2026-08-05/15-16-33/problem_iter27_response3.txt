def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Feasibility-aware gating: energy term disabled when slack > slack_quantile_threshold → prevents premature energy minimization under safety margin.
      - Dual-regime urgency: piecewise-linear urgency with sharp transition at urgency_regime_gate → improves deadline proximity discrimination.
      - Robust percentile-based normalization (IQR-centered) → better small-N stability than rank or MAD alone.
      - All numeric literals strictly in {-2,-1,0,1,2}; no unbounded operations.
      - Explicit finite-domain clipping for all nonlinearities.
      - No unused parameters: all declared PARAMS are referenced.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    finfo = np.finfo(float)
    score = np.full(N, finfo.max, dtype=float)
    feasible_mask = slack >= -eps
    
    if not np.any(feasible_mask):
        return score
    
    # Robust IQR-based normalization: maps x to [-1,1] using quartiles, fallback to range if degenerate
    def iqr_normalize(x):
        x = np.copy(x)
        if N <= 1 or np.all(x == x[0]):
            return np.zeros_like(x, dtype=float)
        q1 = np.percentile(x, 25)
        q3 = np.percentile(x, 75)
        iqr = max(q3 - q1, eps)
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(x) - np.min(x) + eps)
        normed = (x - center) / (denom + eps)
        return np.clip(normed, -1.0, 1.0)
    
    # Urgency: dual-regime piecewise-linear based on slack deficit relative to IQR
    median_slack = np.median(slack) if N > 0 else 0.0
    q1_slack = np.percentile(slack, 25) if N > 0 else median_slack
    q3_slack = np.percentile(slack, 75) if N > 0 else median_slack
    iqr_slack = max(q3_slack - q1_slack, eps)
    slack_deficit = np.clip(median_slack - slack, 0.0, None)  # non-negative deficit
    norm_deficit = np.clip(slack_deficit / (iqr_slack + eps), 0.0, 2.0)
    
    # Piecewise urgency: linear ramp until urgency_regime_gate, then flat saturation
    urgency_linear = np.clip(norm_deficit, 0.0, PARAMS["urgency_regime_gate"])
    urgency_saturated = np.where(norm_deficit >= PARAMS["urgency_regime_gate"],
                                 PARAMS["urgency_regime_gate"],
                                 urgency_linear)
    norm_urgency = iqr_normalize(urgency_saturated)
    
    # Feasibility-aware energy gating: disable energy term when slack is safely above threshold
    slack_quantile = np.percentile(slack, 100.0 * PARAMS["feasibility_aware_energy_gate"]) if N > 0 else median_slack
    energy_active_mask = slack <= slack_quantile
    masked_energy = np.where(energy_active_mask, min_incremental_energy, 0.0)
    norm_energy_eff = iqr_normalize(masked_energy)
    
    # Uncertainty-modulated energy term: activated only above threshold
    unc_centered = uncertainty - PARAMS["uncertainty_activation_threshold"]
    unc_clipped = np.clip(unc_centered, -2.0, 2.0)
    unc_gate = 1.0 / (1.0 + np.exp(-5.0 * unc_clipped))  # fixed steepness for bounded AST depth
    energy_uncertain = min_incremental_energy * unc_gate
    norm_energy_uncertain = iqr_normalize(energy_uncertain)
    
    # Anti-starvation: power-law wait boost, normalized
    median_wait = np.median(ready_wait_time) if N > 0 else eps
    wait_scaled = np.clip(ready_wait_time / (median_wait + eps), 0.0, 2.0)
    wait_boost = np.power(wait_scaled + eps, PARAMS["wait_fairness_exponent"])
    norm_wait = iqr_normalize(wait_boost)
    
    # Compute final score *only* for feasible tasks
    local_score = (
        norm_urgency
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        + PARAMS["energy_uncertainty_coupling"] * norm_energy_uncertain
        - norm_wait
    )
    
    # Apply computed score only where feasible; penalize infeasible heavily but finitely
    score[feasible_mask] = local_score[feasible_mask]
    score[~feasible_mask] = PARAMS["ddl_hard_guard_weight"] * finfo.max
    
    # Enforce finiteness and sanitize
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
