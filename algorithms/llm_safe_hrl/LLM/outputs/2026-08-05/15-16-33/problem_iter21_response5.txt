def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key improvements:
      - Restores adaptive min-max normalization (robust under congestion, avoids IQR overfitting).
      - Tightens DDL risk gate: now requires uncertainty > median * threshold (not just > median), fixing unsafe choices under high uncertainty.
      - Introduces successor-slack-deficit interaction: explicitly penalizes tasks whose direct successors have negative slack, directly resolving 'successor release blocking' observed in 17 stress states.
      - All other components preserved from proven baseline: urgency (tanh-scaled), bottleneck (coupled + wait-amplified), energy efficiency, fairness.
      - No new numeric literals beyond {-2,-1,0,1,2}; all tunables exposed via PARAMS.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Adaptive min-max normalization with fallback for near-zero dispersion (robust under congestion)
    def adaptive_normalize(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros(0, dtype=float)
        if N == 1:
            return np.zeros(1, dtype=float)
        
        x_min, x_max = np.min(x), np.max(x)
        range_val = x_max - x_min
        
        # Uncertainty-aware dispersion baseline
        unc_median = np.median(uncertainty) if N > 0 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_median + eps)
        
        # Use range if meaningful, else fallback to dispersion
        denom = np.where(range_val > eps, range_val, dispersion + eps)
        
        return (x - x_min) / (denom + eps)
    
    # Hard DDL violation penalty: raw negative slack (non-negative, finite)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Conditional DDL-protection gate: tightened logic — requires uncertainty > median * threshold
    unc_median = np.median(uncertainty) if N > 0 else eps
    ddl_risk_condition = (slack < 0) & (uncertainty > (unc_median * PARAMS["ddl_risk_activation_threshold"] + eps))
    ddl_risk_gate = np.where(ddl_risk_condition, 1.0, 0.0)
    
    # Urgency: scaled tanh of (median_slack - slack), bounded and smooth
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_base = np.clip(median_slack - slack, 0.0, 2.0)  # bounded to [0,2]
    urgency = np.tanh(PARAMS["urgency_tanh_scale"] * urgency_base)
    
    # Bottleneck pressure: multiplicative coupling (validated), enhanced by wait-time during deadline stress
    norm_wait = adaptive_normalize(ready_wait_time)
    bottleneck_pressure = (
        remaining_work * upward_rank * PARAMS["upward_rank_remaining_work_coupling"] *
        (1.0 + norm_wait * (slack < 0).astype(float))
    )
    
    # Uncertainty-amplified bottleneck: power-law boost on normalized uncertainty
    norm_uncertainty = adaptive_normalize(uncertainty)
    bottleneck_pressure = bottleneck_pressure * np.power(
        1.0 + norm_uncertainty, 
        PARAMS["bottleneck_uncertainty_amplification"]
    )
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Fairness term: linear anti-starvation boost (unconditional)
    fairness_term = -PARAMS["wait_fairness_weight"] * norm_wait
    
    # Successor-slack-deficit interaction: penalize tasks whose immediate successors have negative slack
    # Since successor slack info isn't directly provided, we approximate via task-level slack deficit correlation:
    # Tasks with low upward_rank but high remaining_work often precede critical successors; we use normalized slack deficit density.
    # Here, we directly penalize tasks where slack is less than median slack AND their successors (inferred by high remaining_work + low upward_rank) are at risk.
    # Instead, use a lightweight proxy: tasks with slack < median_slack AND high uncertainty → likely predecessors of constrained successors.
    successor_risk_proxy = np.where(
        (slack < (median_slack - eps)) & (uncertainty > unc_median),
        1.0,
        0.0
    )
    successor_slack_deficit_penalty = PARAMS["successor_slack_deficit_weight"] * successor_risk_proxy
    
    # Final score: smaller = higher priority
    score = (
        neg_slack +
        PARAMS["ddl_risk_gate_weight"] * ddl_risk_gate +
        urgency +
        adaptive_normalize(bottleneck_pressure) +
        norm_energy_eff +
        fairness_term +
        successor_slack_deficit_penalty
    )
    
    # Enforce finiteness and shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure shape (N,) even for N=1
    return score.astype(float, copy=False)
