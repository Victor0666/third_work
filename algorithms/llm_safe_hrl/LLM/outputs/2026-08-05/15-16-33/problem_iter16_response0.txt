def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating:
    - Explicit negative-slack penalty (not sigmoid) for stronger hard-DDL enforcement
    - Upward-rank × remaining-work interaction as primary bottleneck signal (validated in replay failures)
    - Uncertainty-coupled energy: min_incremental_energy * (1 + uncertainty) to penalize high-risk energy choices
    - Removed urgency sigmoid and wait ramp; replaced with direct anti-starvation via normalized ready_wait_time
    - All normalizations use adaptive IQR→min-max fallback with shared epsilon
    - No inactive parameters; all declared parameters are used
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max when dispersion is low
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR too small (low dispersion), else use IQR
        if iqr < eps:
            x_min, x_max = np.min(x), np.max(x)
            denom = x_max - x_min
            if denom < eps:
                return np.zeros_like(x)
            return (x - x_min) / (denom + eps)
        else:
            denom = iqr
            return (x - center) / (denom + eps)
    
    # Duration = exec + comm, clipped against epsilon
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    
    # Critical-path bottleneck: upward_rank × remaining_work (validated in replay for starvation)
    # Weighted by parameter to control amplification strength
    critical_bottleneck = upward_rank * remaining_work
    norm_critical_bottleneck = adaptive_normalize(critical_bottleneck)
    
    # Negative-slack penalty: binary activation + linear penalty magnitude scaled by weight
    # Ensures hard-DDL tasks dominate priority regardless of other signals
    negative_slack_mask = (slack < 0.0).astype(float)
    slack_magnitude = np.abs(slack) * negative_slack_mask
    norm_slack_magnitude = adaptive_normalize(slack_magnitude)
    negative_slack_penalty = PARAMS["negative_slack_penalty_weight"] * norm_slack_magnitude
    
    # Uncertainty-coupled energy: penalize high-energy choices more under high uncertainty
    coupled_energy = min_incremental_energy * (1.0 + uncertainty + eps)
    norm_coupled_energy = adaptive_normalize(coupled_energy)
    
    # Anti-starvation: directly normalize ready_wait_time (no ramp/saturation — simpler, deterministic)
    norm_wait = adaptive_normalize(ready_wait_time)
    
    # Final score: smaller = higher priority
    # Priority hierarchy: hard-DDL safety > critical-path bottleneck > uncertainty-energy > fairness
    score = (
        negative_slack_penalty  # Highest priority: enforce hard deadlines first
        + PARAMS["upward_rank_remaining_work_interaction_weight"] * norm_critical_bottleneck  # Unblock successors on critical path
        + PARAMS["uncertainty_coupled_energy_weight"] * norm_coupled_energy  # Energy cost grows with risk
        - norm_wait  # Favor longer-waiting tasks (anti-starvation)
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
