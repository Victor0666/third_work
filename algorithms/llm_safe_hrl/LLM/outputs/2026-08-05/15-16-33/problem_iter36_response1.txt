def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with slack-gradient modulation:
      - Replaces binary DDL-protection gating with *continuous slack-gradient sensitivity*:
        bottleneck and energy weights scaled by normalized local slack slope (median - slack) / (std(slack)+eps),
        preserving signal strength under skewed but non-critical slack distributions.
      - Retains pre-normalized `neg_slack` as dominant hard-DDL penalty — no distortion, no amplification.
      - Keeps bounded sigmoid wait saturation for robust anti-starvation.
      - All normalization uses uncertainty-aware `adaptive_normalize()` with fallback to range.
      - Final composition: hard-DDL > urgency > modulated bottleneck > modulated energy > fairness.
      - No branching; all operations vectorized and finite-valued.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Uncertainty-aware adaptive normalization: dispersion scaled by std(uncertainty)
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x) if N > 0 else 0.0
        
        unc_std = np.std(uncertainty) if N > 1 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
        
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = np.where(dispersion > eps, dispersion, fallback_range)
        
        return (x - center) / (denom + eps)
    
    # Pre-normalized hard-deadline violation signal: raw negative slack (dominant term)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Linear urgency proxy: (median_slack - slack), capped for feasibility preservation
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    
    # Feasibility-preserving urgency cap: max non-negative slack raised to learned exponent
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.minimum(urgency_linear, urgency_cap)
    
    norm_urgency = adaptive_normalize(urgency_linear)
    
    # Slack-gradient sensitivity: continuous modulation factor for bottleneck & energy terms
    # Uses normalized urgency gradient: higher when slack is low relative to population spread
    slack_std = np.std(slack) if N > 1 else eps
    slack_gradient = (median_slack - slack) / (slack_std + eps)
    modulation_factor = np.clip(slack_gradient, 0.0, 2.0) ** PARAMS["slack_gradient_sensitivity"]
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency_linear)
    bottleneck_base = duration * upward_rank * remaining_work * (1.0 + urgency_linear + eps)
    
    # Uncertainty amplification: power-law boost using normalized uncertainty
    unc_normalized = adaptive_normalize(uncertainty)
    bottleneck_pressure = bottleneck_base * np.power(1.0 + unc_normalized, 
                                                      PARAMS["bottleneck_uncertainty_amplification"])
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Bounded sigmoid wait saturation: smooth anti-starvation boost
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = adaptive_normalize(wait_saturation)
    
    # Final score: smaller = higher priority
    # Dominant ordering: hard DDL violation > urgency > modulated bottleneck > modulated energy > fairness
    score = (
        neg_slack  # Raw penalty — strongest signal for deadline violation
        + norm_urgency
        + modulation_factor * PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + modulation_factor * PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
    )
    
    # Enforce finiteness and shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
