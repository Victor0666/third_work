def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's smooth urgency and bottleneck coupling
    with Parent 1's robust DDL-protection semantics and enhanced uncertainty-slack interaction.
    
    Key structural improvements:
      - Replaces Parent 2's binary DDL-protection gate with *graded* uncertainty-slack coupling:
        multiplies normalized uncertainty with sigmoid urgency (not just thresholded activation),
        enabling fine-grained risk amplification without hard discontinuities.
      - Uses asymmetric IQR percentiles (20/80) for tighter normalization robustness against outliers.
      - Retains smooth arctan wait saturation and successor bottleneck term for fairness & blocking mitigation.
      - Critical-path bonus now uses *normalized* upward_rank scaled by percentile gate — avoids rank-scale bias.
      - All terms are IQR-normalized and bounded; no piecewise discontinuities or unbounded growth.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps_safe else (np.max(np.abs(x - center)) + eps_safe)
        return (x - center) / (denom + eps_safe)

    # Sigmoid urgency gate: smooth, differentiable, centered at offset, steepness tunable
    # Maps slack → [-1, 1]: negative slack → +1 (high urgency), positive slack → -1 (low urgency)
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    # Avoid overflow in exp by clamping input using declared parameter
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered, 
                            -PARAMS["sigmoid_clip_bound"], 
                            PARAMS["sigmoid_clip_bound"])
    urgency_gate = 2.0 / (1.0 + np.exp(sigmoid_input)) - 1.0  # range [-1, 1]
    norm_urgency = iqr_normalize(urgency_gate)

    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps_safe)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)

    # Successor bottleneck coupling: prioritize tasks whose long duration blocks high-work/high-rank successors
    # Term = duration * (remaining_work + upward_rank), scaled and normalized
    bottleneck_pressure = duration * (remaining_work + upward_rank + eps_safe)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)

    # Critical-path bonus: activated only for top percentile of upward_rank, applied to normalized rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps_safe)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank

    # Ready wait time: smooth saturation via arctan (bounded [0,1], avoids exponential drift)
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps_safe)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = iqr_normalize(wait_saturation)

    # Graded uncertainty-slack coupling: multiply normalized uncertainty with urgency (not binary gate)
    # Amplifies urgency proportionally to risk level — stronger when both tight slack AND high uncertainty co-occur
    norm_uncertainty = iqr_normalize(uncertainty)
    graded_coupling = PARAMS["uncertainty_slack_coupling_weight"] * norm_uncertainty * norm_urgency

    # Final score: smaller = higher priority
    # Hierarchy: urgency dominates → bottleneck release → energy → critical path → fairness → coupling
    score = (
        norm_urgency  # Primary deadline pressure signal
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks
        - norm_wait  # Reward recently arrived tasks (fairness)
        + graded_coupling  # Fine-grained risk-aware urgency amplification
    )

    # Ensure finite output, no NaN/inf, correct shape
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
