def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: sigmoid urgency + critical-path bonus + anti-starvation + DDL-protection.
    
    Structural improvements:
      - Hybrid urgency: sigmoid for smooth gradient near deadline, plus bounded linear fallback for extreme negative slack
      - Critical-path bonus reintroduced from Parent 1 (top-percentile upward_rank boost) to strengthen DAG-aware scheduling
      - Anti-starvation via inverse wait-time saturation (Parent 2) but with robust clipping to avoid zero-division
      - Unified bottleneck term: (exec+comm) * upward_rank * remaining_work * (1 + uncertainty) — couples risk into blocking pressure
      - All normalizations use IQR with elite-tuned percentiles; no hardcoded constants beyond [-2, -1, 0, 1, 2]
      - Final score preserves strict hierarchy: deadline > critical path > bottleneck > energy > fairness
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = (q_low + q_high) / 2.0
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Hybrid urgency: sigmoid for moderate slack, bounded linear for extreme lateness
    # Sigmoid dominates near zero; linear kicks in below slack < -2*std to prevent flatness
    finfo = np.finfo(float)
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    sigmoid_input = -PARAMS["slack_sigmoid_steepness"] * slack_centered
    sigmoid_input = np.clip(sigmoid_input, -finfo.max, finfo.max)
    sigmoid_urgency = 1.0 / (1.0 + np.exp(-sigmoid_input))
    
    # Linear urgency for severely negative slack (stronger signal when deadline violation imminent)
    linear_urgency_raw = np.where(
        slack <= 0,
        PARAMS["urgency_linear_slope"] * (slack - PARAMS["urgency_linear_offset"]),
        0.0
    )
    linear_urgency = np.clip(linear_urgency_raw, -2.0, 2.0)
    
    # Blend: full sigmoid above slack=-1, full linear below slack=-3, linear interpolation in between
    blend_weight = np.clip((slack + 2.0) / 2.0, 0.0, 1.0)  # 0 at slack=-2, 1 at slack=0
    urgency_gate = blend_weight * sigmoid_urgency + (1.0 - blend_weight) * linear_urgency
    
    norm_urgency = iqr_normalize(urgency_gate)
    
    # Energy efficiency: log-normalized marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    log_energy_eff = np.log1p(energy_per_duration + eps)
    norm_energy_eff = iqr_normalize(log_energy_eff)
    
    # Successor bottleneck coupling: triple interaction + uncertainty amplification
    # Captures both computational pressure and risk-aware blocking
    bottleneck_pressure = duration * (upward_rank + eps) * (remaining_work + eps) * (1.0 + uncertainty)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    
    # Critical-path bonus: activate only for top percentile of upward_rank (Parent 1)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Ready wait time: bounded linear ramp with soft cap + anti-starvation
    wait_normalized = np.clip(ready_wait_time / (PARAMS["wait_saturation_time"] + eps), 0.0, 1.0)
    wait_priority = 1.0 - wait_normalized  # smaller wait → higher priority
    norm_wait = iqr_normalize(wait_priority)
    
    # DDL-protection gate: activates when uncertainty exceeds threshold fraction of max uncertainty
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty
    ).astype(float)
    
    # Uncertainty-normalized term for amplification under risk, only when DDL at risk
    norm_uncertainty = iqr_normalize(uncertainty)
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline safety > critical path > bottleneck release > energy efficiency > fairness
    score = (
        norm_urgency  # Primary deadline pressure signal (hybrid)
        + critical_bonus  # Boost critical path nodes (Parent 1)
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Risk-aware blocking pressure
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - norm_wait  # Reward shorter-waiting tasks (anti-starvation)
        + ddl_protection_active * norm_uncertainty  # Amplify uncertainty only when DDL at risk
    )
    
    # Ensure finite output, no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=np.finfo(float).min)
    
    return score.astype(float, copy=False)
