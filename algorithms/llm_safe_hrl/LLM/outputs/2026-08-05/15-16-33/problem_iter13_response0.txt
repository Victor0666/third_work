def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with:
    - Simplified urgency: direct sigmoid on slack (no offset/steepness parameters) → removes inactive params
    - Unified risk-aware bottleneck: (duration + comm_time) * upward_rank * (1 + sigmoid(-slack)) * (1 + uncertainty)
    - Anti-starvation via normalized wait time (no saturation ramp → simpler, more robust)
    - DDL-protection gate now uses joint condition: tight slack AND high uncertainty, activated only when both hold
    - Removed redundant terms (wait_ramp_saturation_factor, sigmoid_clip_bound, etc.) per inactivity evidence
    - All normalizations use adaptive IQR; fallback to min-max if dispersion too low.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max when dispersion is low
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR too small (low dispersion), else use IQR
        if iqr < eps:
            x_min, x_max = np.min(x), np.max(x)
            denom = x_max - x_min
            if denom < eps:
                return np.zeros_like(x)
            return (x - x_min) / (denom + eps)
        else:
            denom = iqr
            return (x - center) / (denom + eps)
    
    # Direct sigmoid urgency: maps slack → [0,1], negative slack → near 1.0 (high urgency)
    # Avoids inactive parameters (offset, steepness, clip) — uses bounded exp via np.clip
    sigmoid_input = np.clip(-slack, -20.0, 20.0)  # Prevent overflow, no tunable steepness
    urgency = 1.0 / (1.0 + np.exp(sigmoid_input))
    
    # Normalize urgency
    norm_urgency = adaptive_normalize(urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Unified bottleneck pressure: duration × upward_rank × (1 + urgency) × (1 + uncertainty)
    # Removes remaining_work coupling (evidence shows weak cross-generation impact) and simplifies structure
    bottleneck_pressure = duration * upward_rank * (1.0 + urgency) * (1.0 + uncertainty + eps)
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Anti-starvation: normalized ready_wait_time (linear fairness, no saturation)
    norm_wait = adaptive_normalize(ready_wait_time)
    
    # Joint DDL-protection gate: activated only when slack is tight AND uncertainty is high
    # Uses max_uncertainty to scale threshold — avoids absolute thresholds that fail across scenarios
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    ddl_gate = (
        (slack < 0.0) & 
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Final score: smaller = higher priority
    # Prioritizes deadline safety first, then bottleneck release, then energy, then fairness
    score = (
        norm_urgency  # Primary deadline signal (strongest weight)
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock critical successors under joint risk
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Energy efficiency
        - norm_wait  # Favor fairness: longer wait → lower score (higher priority)
        + ddl_gate * upward_rank  # Amplify priority for high-upward-rank tasks under hard violation risk
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
