def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
    - Deadline-urgency gating (no branching: uses vectorized comparison + multiplication)
    - Critical-path–aware normalization (no loops: uses boolean indexing & np.where)
    - Risk-aware energy capping (no branching: uses np.clip)
    - All conditionals replaced by vectorized arithmetic; branch count = 0.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Critical-path subgroup mask: high-upward-rank tasks form bottleneck cohort
    if N == 0:
        is_critical = np.zeros(0, dtype=bool)
    else:
        critical_threshold = np.quantile(upward_rank, PARAMS["critical_path_quantile"])
        is_critical = upward_rank >= (critical_threshold - eps)
    
    # Robust adaptive normalization — unified per-group via np.where, no explicit branches
    def adaptive_normalize_by_group(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros_like(x)
        
        # Compute stats for critical group
        x_crit = x[is_critical]
        q_low_crit = np.percentile(x_crit, PARAMS["iqr_low_percentile"]) if len(x_crit) > 0 else 0.0
        q_high_crit = np.percentile(x_crit, PARAMS["iqr_high_percentile"]) if len(x_crit) > 0 else 0.0
        iqr_crit = q_high_crit - q_low_crit
        center_crit = np.median(x_crit) if len(x_crit) > 0 else 0.0
        
        # Compute stats for non-critical group
        x_noncrit = x[~is_critical]
        q_low_non = np.percentile(x_noncrit, PARAMS["iqr_low_percentile"]) if len(x_noncrit) > 0 else 0.0
        q_high_non = np.percentile(x_noncrit, PARAMS["iqr_high_percentile"]) if len(x_noncrit) > 0 else 0.0
        iqr_non = q_high_non - q_low_non
        center_non = np.median(x_noncrit) if len(x_noncrit) > 0 else 0.0
        
        # Normalize each group using vectorized selection
        norm_x = np.zeros_like(x)
        
        # Critical group normalization
        if len(x_crit) > 0:
            denom_crit = np.where(iqr_crit < eps, np.max(x_crit) - np.min(x_crit), iqr_crit)
            center_use_crit = np.where(iqr_crit < eps, np.min(x_crit), center_crit)
            norm_crit = np.where(denom_crit < eps, 0.0, (x_crit - center_use_crit) / (denom_crit + eps))
            norm_x[is_critical] = norm_crit
        
        # Non-critical group normalization
        if len(x_noncrit) > 0:
            denom_non = np.where(iqr_non < eps, np.max(x_noncrit) - np.min(x_noncrit), iqr_non)
            center_use_non = np.where(iqr_non < eps, np.min(x_noncrit), center_non)
            norm_non = np.where(denom_non < eps, 0.0, (x_noncrit - center_use_non) / (denom_non + eps))
            norm_x[~is_critical] = norm_non
        
        return norm_x
    
    # Linear slack urgency: penalize negative slack more strongly, scaled by uncertainty
    linear_slack_penalty = PARAMS["slack_linear_coeff"] * np.clip(-slack, 0.0, None)
    coupled_slack_penalty = linear_slack_penalty * (1.0 + PARAMS["uncertainty_slack_coupling"] * uncertainty)
    norm_slack_penalty = adaptive_normalize_by_group(coupled_slack_penalty)
    
    # Upward rank × remaining work interaction
    critical_load_pressure = upward_rank * (remaining_work + eps)
    norm_critical_load = adaptive_normalize_by_group(critical_load_pressure)
    
    # Risk-aware energy capping: clamp before normalization
    base_energy_cap = np.median(min_incremental_energy) if N > 0 else 0.0
    energy_cap = base_energy_cap * (1.0 + PARAMS["energy_uncertainty_cap_factor"] * np.mean(uncertainty))
    capped_energy = np.clip(min_incremental_energy, None, energy_cap + eps)
    norm_energy = adaptive_normalize_by_group(capped_energy)
    shifted_energy = norm_energy + PARAMS["energy_normalization_offset"]
    shifted_energy = np.clip(shifted_energy, 0.0, 2.0)
    
    # Deadline-urgency gated wait coupling: vectorized activation (no if/else)
    wait_normalized = adaptive_normalize_by_group(ready_wait_time)
    urgency_gate = (slack <= PARAMS["urgency_gate_threshold"]).astype(float)
    wait_urgency_gated = wait_normalized * (1.0 + norm_slack_penalty) * urgency_gate
    norm_wait_urgency = adaptive_normalize_by_group(wait_urgency_gated)
    
    # Final score: smaller = higher priority
    score = (
        norm_slack_penalty
        + PARAMS["upward_remaining_interaction_weight"] * norm_critical_load
        + PARAMS["wait_urgency_coupling"] * norm_wait_urgency
        + shifted_energy
    )
    
    # Ensure finite output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
