def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with adaptive deadline-criticality detection and MAD-based robust scaling.
    
    Key structural improvements:
      - Replaces fixed `ddl_feasibility_threshold` with percentile-based adaptive criticality detection:
        tasks in bottom `slack_critical_percentile` of slack distribution enter DDL-stress mode.
      - Introduces multiplicative `upward_rank * remaining_work` coupling *only* under DDL stress,
        replacing linear bottleneck sum — better captures joint critical-path + work-pressure amplification.
      - Switches `slack` and `uncertainty` normalization from IQR to MAD scaling (with consistent 1.4826 factor)
        for superior outlier resilience and stability across heterogeneous workflows.
      - Removes redundant `ddl_protection_mask` duplication; unifies risk logic into single adaptive urgency path.
      - All non-urgency terms are suppressed *only* under adaptive DDL stress (not hard threshold), improving robustness.
      - Retains arctan wait saturation, host-load–aware energy, and safe percentile ranking for N=1.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust MAD-based normalization (replaces IQR per reflection)
    def mad_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        abs_devs = np.abs(x - center)
        mad = np.median(abs_devs)
        scale = PARAMS["mad_scale_factor"] * mad
        denom = scale if scale > eps else (np.max(abs_devs) + eps)
        return (x - center) / (denom + eps)
    
    # Adaptive deadline-criticality detection: bottom percentile of slack → DDL stress mode
    if N == 1:
        slack_percentile = np.array([0.0])
    else:
        sorted_slack = np.sort(slack)
        slack_idx = np.searchsorted(sorted_slack, slack, side='left')
        slack_percentile = slack_idx / (N + eps)
    ddl_stress_mask = (slack_percentile < PARAMS["slack_critical_percentile"]).astype(float)
    
    # Linear urgency gate: monotonic, bounded [-1,1], zero-crossing at slack=0
    median_abs_slack = np.maximum(np.abs(np.median(slack)), eps)
    urgency_gate = np.clip(slack / (median_abs_slack + eps), -1.0, 1.0)
    urgency_gate = -urgency_gate  # flip so negative slack → positive urgency
    
    # Amplify urgency under adaptive DDL stress
    urgency_amplified = urgency_gate * (1.0 + PARAMS["feasibility_guard_strength"] * ddl_stress_mask)
    norm_urgency = mad_normalize(urgency_amplified)
    
    # Host-load–aware energy normalization
    host_load_adjusted_energy = min_incremental_energy * (1.0 + PARAMS["host_load_sensitivity"] * uncertainty)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = host_load_adjusted_energy / duration
    # Normalize only outside DDL stress mode
    norm_energy_eff = mad_normalize(energy_per_duration) * (1.0 - ddl_stress_mask)
    
    # Multiplicative bottleneck coupling *only under DDL stress*: rank × work pressure
    # More expressive than linear sum for capturing joint critical-path + load pressure
    rank_work_pressure = (upward_rank + eps) * (remaining_work + eps)
    norm_rank_work = mad_normalize(rank_work_pressure) * ddl_stress_mask
    
    # Critical-path bonus: activated only for top percentile of upward_rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = mad_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank * (1.0 - ddl_stress_mask)
    
    # Ready wait time: smooth saturation via arctan (bounded [0,1])
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = mad_normalize(wait_saturation) * (1.0 - ddl_stress_mask)
    
    # Unified risk-aware urgency amplification: uses MAD-normalized inputs
    norm_uncertainty = mad_normalize(uncertainty)
    risk_amplification_mask = (
        (slack < 0) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * np.max(uncertainty + eps))
    ).astype(float)
    risk_amplified_urgency = norm_urgency * (1.0 + PARAMS["uncertainty_slack_interaction_strength"] * norm_uncertainty * risk_amplification_mask)
    
    # Final score: smaller = higher priority
    # Under DDL stress: prioritize amplified urgency + rank×work pressure; suppress fairness & energy
    # Outside stress: full multi-objective balance
    score = (
        risk_amplified_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_rank_work
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - critical_bonus
        - norm_wait
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
