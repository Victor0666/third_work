def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: deadline-risk-gated criticality + uncertainty-aware energy efficiency.
    
    Key innovations:
    - Negative slack triggers a smooth, bounded power-law penalty (not linear)
    - Upward rank boosts only when slack is tight (conditional coupling), avoiding blind criticality bias
    - Ready wait time uses exponential decay to prevent late-burst starvation without dominating
    - Energy term is down-weighted when slack < 0 (DDL violation imminent), prioritizing feasibility first
    - All features normalized via median-IQR (robust to outliers) with epsilon safeguards
    - Uncertainty modulates slack penalty only when both slack < 0 AND uncertainty > median
    - Duration included only in slack-sufficient regime to avoid short-task bias under pressure
    """
    eps = PARAMS["epsilon"]
    n = len(min_exec_time)
    
    # Safe cast & copy to avoid mutation
    exec_t = np.asarray(min_exec_time, dtype=float).copy()
    comm_t = np.asarray(min_comm_time, dtype=float).copy()
    energy = np.asarray(min_incremental_energy, dtype=float).copy()
    slk = np.asarray(slack, dtype=float).copy()
    rank = np.asarray(upward_rank, dtype=float).copy()
    work = np.asarray(remaining_work, dtype=float).copy()
    wait = np.asarray(ready_wait_time, dtype=float).copy()
    unc = np.asarray(uncertainty, dtype=float).copy()
    
    # Robust normalization: median + IQR (more stable than mean/std for skewed distributions)
    def robust_normalize(x):
        x = np.where(np.isfinite(x), x, 0.0)
        med = np.median(x)
        q75, q25 = np.percentile(x, [75, 25], method='midpoint') if n > 1 else (med, med)
        iqr = q75 - q25 + eps
        return (x - med) / iqr
    
    # Normalize all base features
    norm_slk = robust_normalize(slk)
    norm_energy = robust_normalize(energy)
    norm_rank = robust_normalize(rank ** PARAMS["rank_normalization_power"])
    norm_work = robust_normalize(work)
    norm_wait = robust_normalize(wait)
    norm_unc = robust_normalize(unc)
    
    # Duration: exec + comm, normalized robustly
    duration = exec_t + comm_t
    norm_duration = robust_normalize(duration)
    
    # === SLACK RISK PENALTY ===
    # Strong nonlinear penalty for negative slack: (max(0, -slack))^exponent → always ≥ 0
    # Then map to priority score: higher penalty → higher score → lower priority → so invert logic later
    slack_penalty_raw = np.maximum(0.0, -slk) ** PARAMS["slack_risk_exponent"]
    norm_slack_penalty = robust_normalize(slack_penalty_raw)
    
    # === UNCERTAINTY-MODULATED SLACK PENALTY ===
    # Only activate extra penalty when slack is negative AND uncertainty is above median
    high_unc_mask = (unc > np.median(unc + eps)) if n > 1 else np.full(n, True)
    coupled_penalty = np.where(
        (slk < 0) & high_unc_mask,
        norm_slack_penalty * PARAMS["uncertainty_slack_coupling"],
        0.0
    )
    
    # === CONDITIONAL CRITICALITY BOOST ===
    # Boost rank importance only when slack is tight (≤ 0), otherwise neutral
    rank_boost = np.where(
        slk <= 0,
        norm_rank * PARAMS["criticality_boost"],
        norm_rank * 0.5  # mild baseline weight when slack > 0
    )
    
    # === ENERGY EFFICIENCY BIAS ===
    # Down-weight energy when slack < 0 (feasibility first); full weight when slack ≥ 0
    energy_weight = np.where(
        slk < 0,
        PARAMS["energy_efficiency_bias"] * 0.3,
        PARAMS["energy_efficiency_bias"]
    )
    
    # === WAIT-TIME ANTI-STARVATION ===
    # Exponential decay: longer wait → higher priority, but saturates smoothly
    # Use normalized wait time scaled by decay rate; clamp to avoid overflow
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * (wait + eps))
    norm_wait_boost = robust_normalize(wait_boost)
    
    # === DURATION IN NON-CRITICAL REGIME ONLY ===
    # Only consider duration when slack > 0 (sufficient margin), else ignore to avoid biasing away from urgent tasks
    duration_contrib = np.where(
        slk > 0,
        norm_duration * PARAMS["duration_sensitivity"],
        0.0
    )
    
    # === FINAL SCORE: smaller = better ===
    # Components ordered by lexicographic priority: DDL risk > critical path > energy > duration > wait > work
    score = (
        # Primary: Slack risk dominates — negative slack must be penalized strongly
        norm_slack_penalty * 5.0
        + coupled_penalty * 3.0
        
        # Secondary: Critical path importance, gated by slack
        - rank_boost * 2.0
        
        # Tertiary: Energy efficiency — scaled by slack condition
        + norm_energy * energy_weight
        
        # Quaternary: Duration only when slack permits
        + duration_contrib
        
        # Anti-starvation: wait boost (positive contribution → lowers score)
        - norm_wait_boost * 1.2
        
        # Work and remaining load: weak negative influence (larger work → slightly higher priority)
        - norm_work * 0.4
    )
    
    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
