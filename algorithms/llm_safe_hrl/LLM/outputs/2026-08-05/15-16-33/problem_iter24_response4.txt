def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three structural improvements:
      - Hard DDL protection gate: zero-weights non-critical-path tasks (upward_rank < median) when slack < threshold, preserving dominant neg_slack while preventing misprioritization under deadline pressure.
      - Conditional multiplicative activation: successor-release and bottleneck terms only activate when slack < median_slack, avoiding spurious pressure in slack-rich regimes.
      - Simplified adaptive normalization: uses fixed-range fallback only (no unc_std dependency), improving generalization and eliminating fragile risk-context coupling.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Simplified adaptive normalization: fixed-range fallback only (no uncertainty std dependency)
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x) if N > 0 else 0.0
        range_val = np.max(x) - np.min(x) if N > 0 else eps
        denom = np.maximum(range_val, eps)
        return (x - center) / (denom + eps)
    
    # Dominant hard-deadline violation signal: raw negative slack (pre-normalized, unbounded penalty)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Linear urgency proxy with feasibility-preserving cap
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.minimum(urgency_linear, urgency_cap)
    norm_urgency = adaptive_normalize(urgency_linear)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency_linear) × (1 + normalized_uncertainty)^amplification
    bottleneck_pressure = duration * upward_rank * remaining_work * (1.0 + urgency_linear + eps)
    unc_normalized = adaptive_normalize(uncertainty)
    bottleneck_pressure = bottleneck_pressure * np.power(1.0 + unc_normalized, 
                                                         PARAMS["bottleneck_uncertainty_amplification"])
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Successor-release pressure: upward_rank × remaining_work × decay factor based on |slack|
    slack_abs = np.abs(slack) + eps
    release_decay = np.power(1.0 + slack_abs, -PARAMS["successor_release_pressure_exponent"])
    successor_release_pressure = upward_rank * remaining_work * release_decay
    norm_successor_release = adaptive_normalize(successor_release_pressure)
    
    # Bounded sigmoid wait saturation: smooth anti-starvation boost
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_saturation = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait = adaptive_normalize(wait_saturation)
    
    # Critical-path coupling: pure structural importance, normalized and scaled to be secondary
    cp_coupling = upward_rank * remaining_work
    norm_cp_coupling = adaptive_normalize(cp_coupling)
    
    # Hard DDL protection gate: suppress non-critical-path tasks when slack is critically negative
    median_upward_rank = np.median(upward_rank) if N > 0 else 0.0
    is_critical_path = (upward_rank >= median_upward_rank).astype(float)
    ddl_gate = (slack >= PARAMS["ddl_protection_gate_threshold"]).astype(float)
    # Only apply gate when slack is dangerously negative AND task is not on critical path
    critical_path_gate = np.where(
        (slack < PARAMS["ddl_protection_gate_threshold"]) & (upward_rank < median_upward_rank),
        0.0,
        1.0
    )
    
    # Conditional activation: only activate bottleneck and successor-release pressure when under deadline pressure
    pressure_activation = (slack < median_slack).astype(float)
    
    # Final score: smaller = higher priority
    # Ordering: hard DDL violation > urgency > (conditional) successor-release > (conditional) bottleneck > energy > fairness > CP coupling
    score = (
        neg_slack  # Strongest signal: direct penalty for lateness risk
        + norm_urgency
        + PARAMS["successor_bottleneck_coupling"] * pressure_activation * norm_successor_release
        + PARAMS["successor_bottleneck_coupling"] * pressure_activation * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
        + PARAMS["upward_rank_remaining_work_coupling"] * norm_cp_coupling
        + (1.0 - critical_path_gate) * 1e6  # Hard penalty for non-critical-path tasks under critical slack
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
