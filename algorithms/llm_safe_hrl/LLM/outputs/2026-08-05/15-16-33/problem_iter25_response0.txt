def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with structural improvements guided by counterfactual evidence:
      - Introduces conditional DDL risk protection gate: only amplifies urgency when slack is below threshold * median_slack.
      - Replaces adaptive dispersion with robust MAD-based normalization for energy and bottleneck terms.
      - Adds explicit upward_rank × remaining_work interaction to strengthen critical path pressure (per high-confidence 'add_upward_rank_remaining_work_interaction').
      - Uses uncertainty-slack coupling only under DDL risk, avoiding unconditional risk inflation.
      - Removes wait saturation (deemed inactive per diagnostics) and replaces with bounded linear wait boost.
      - All operations are finite, deterministic, and use only {-2,-1,0,1,2} literals.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust MAD-based normalization (more stable than std for small N or outliers)
    def mad_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        # Compute MAD: median absolute deviation
        mad = np.median(np.abs(x - center)) if N > 0 else eps
        # Fallback to range if MAD near zero
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = mad if mad > eps else fallback_range
        return (x - center) / (denom + eps)
    
    # Pre-normalized hard deadline violation signal: dominant term
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Median slack for relative urgency scaling
    median_slack = np.median(slack) if N > 0 else 0.0
    
    # Conditional DDL risk gate: activate only when slack < threshold * median_slack
    # Avoids over-amplification in safe regions; aligns with 'add_conditional_ddl_protection_gate'
    ddl_risk_mask = (slack < PARAMS["ddl_risk_gate_threshold"] * (median_slack + eps)).astype(float)
    
    # Urgency: linear deficit scaled by risk gate and amplified under risk
    urgency_linear = np.clip(median_slack - slack, 0.0, None)
    urgency_score = urgency_linear * (1.0 + ddl_risk_mask * (PARAMS["ddl_risk_amplification"] - 1.0))
    
    # Critical path pressure: upward_rank × remaining_work, gated by DDL risk
    critical_pressure = upward_rank * remaining_work
    critical_pressure = critical_pressure * (1.0 + ddl_risk_mask * (PARAMS["upward_rank_remaining_work_coupling"] - 1.0))
    
    # Energy term: marginal energy normalized robustly
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy = mad_normalize(energy_per_duration) * PARAMS["energy_normalization_scale"]
    
    # Uncertainty-coupled slack deficit: only active under DDL risk
    norm_slack_deficit = mad_normalize(neg_slack)
    unc_coupled_deficit = norm_slack_deficit * np.power(1.0 + uncertainty, PARAMS["uncertainty_slack_coupling"])
    unc_coupled_deficit = unc_coupled_deficit * ddl_risk_mask
    
    # Bounded linear wait boost (anti-starvation): clipped and scaled
    wait_boost = np.clip(ready_wait_time, 0.0, 2.0 * (np.median(ready_wait_time) + eps) if N > 0 else 2.0 * eps)
    wait_boost = wait_boost / (np.median(ready_wait_time) + eps) if N > 0 else wait_boost / eps
    wait_boost = np.clip(wait_boost, 0.0, 2.0)
    
    # Final score: smaller = higher priority
    # Ordering: hard violation > conditional urgency > critical path > energy > uncertainty-coupled deficit > fairness (wait)
    score = (
        neg_slack  # Raw violation penalty — strongest signal
        + mad_normalize(urgency_score)
        + mad_normalize(critical_pressure)
        + norm_energy
        + mad_normalize(unc_coupled_deficit)
        - wait_boost  # Boost priority for long-waiting tasks
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
