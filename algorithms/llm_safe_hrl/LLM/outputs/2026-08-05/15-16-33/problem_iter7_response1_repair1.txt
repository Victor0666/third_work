def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid urgency with robust slack-based linear penalty,
    removes inactive parameters (slack_sigmoid_* and wait_saturation_time), simplifies critical bonus,
    introduces normalized bottleneck coupling via (exec+comm) * upward_rank only (no remaining_work),
    and adds anti-starvation wait-term using clipped log scaling.
    
    Key structural changes:
      - Replaces fragile sigmoid urgency with bounded linear slack penalty: max(0, -slack) + eps
      - Removes all inactive parameters (slack_sigmoid_offset/steepness, sigmoid_clip_bound, 
        critical_rank_percentile, wait_saturation_time) per counterfactual evidence
      - Simplified bottleneck: uses (exec+comm) * upward_rank only — avoids over-coupling with remaining_work
      - Anti-starvation: log-scaled ready_wait_time with hard clip to prevent unbounded growth
      - DDL-protection gate now multiplies urgency penalty (not uncertainty) when both conditions hold
      - All features normalized via IQR with elite-tuned percentiles
      - Final score prioritizes deadline safety > bottleneck release > energy > fairness
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Linear urgency penalty: zero for positive slack, linear penalty for negative slack
    # Bounded to avoid extreme values; ensures monotonic priority increase with lateness risk
    urgency_penalty = np.maximum(-slack, 0.0) + eps
    norm_urgency = iqr_normalize(urgency_penalty)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Simplified successor bottleneck: duration * upward_rank only — avoids redundancy with remaining_work
    # Focuses on critical-path blocking without double-counting work volume
    bottleneck_pressure = duration * (upward_rank + eps)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    
    # Critical-path bonus: direct upward_rank contribution (no percentile gating — removed per inactivity evidence)
    norm_rank = iqr_normalize(upward_rank)
    critical_bonus = norm_rank
    
    # Anti-starvation wait term: log-scaled with upper clip to prevent dominance under long queues
    # Uses natural log for smooth growth; clip at configured threshold ensures bounded influence
    wait_clipped = np.clip(ready_wait_time, 0.0, PARAMS["wait_clip_threshold"])
    wait_log = np.log1p(wait_clipped)  # log1p avoids log(0)
    norm_wait = iqr_normalize(wait_log)
    
    # DDL-protection gate: activates only when slack is tight AND uncertainty is high
    # Now amplifies urgency penalty (not uncertainty) — aligns with DDL-first objective hierarchy
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Final score: smaller = higher priority
    # Order reflects lexicographic priority: deadline safety > bottleneck release > energy > fairness
    score = (
        norm_urgency  # Primary deadline pressure signal (linear penalty)
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock critical successors
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Favor energy-efficient durations
        - critical_bonus  # Boost critical path tasks (direct rank contribution)
        - norm_wait  # Reward recently arrived tasks (anti-starvation)
        + ddl_protection_active * norm_urgency  # Amplify urgency only when DDL at risk
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
