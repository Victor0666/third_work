import numpy as np

PARAMETER_SCHEMA = {
    "schema_version": 1,
    "parameters": [
        {
            "name": "epsilon",
            "initial_value": 0.000119826464751383,
            "lower_bound": 1e-6,
            "upper_bound": 0.1,
            "semantic_description": "Numerical guard for stable normalization and division.",
            "parameter_type": "float",
            "transform": "log"
        },
        {
            "name": "ddl_protection_gate_threshold",
            "initial_value": 0.590663072072125,
            "lower_bound": 0.5,
            "upper_bound": 0.95,
            "semantic_description": "Fraction of max uncertainty required to activate DDL-protection gate.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "energy_duration_ratio_weight",
            "initial_value": 1.8745925620127373,
            "lower_bound": 0.1,
            "upper_bound": 2.0,
            "semantic_description": "Weight for energy-per-duration efficiency term.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "successor_bottleneck_coupling",
            "initial_value": 0.15698643006544633,
            "lower_bound": 0.0,
            "upper_bound": 1.5,
            "semantic_description": "Strength of interaction between duration, remaining_work, and upward_rank for successor-release pressure.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "tanh_clip_bound",
            "initial_value": 10.0,
            "lower_bound": 1.0,
            "upper_bound": 20.0,
            "semantic_description": "Absolute bound on tanh input to prevent numerical instability.",
            "parameter_type": "float",
            "transform": "identity"
        }
    ]
}

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with:
    - Smooth tanh-based urgency mapping for stable gradient near zero slack.
    - Joint DDL-uncertainty gating: only activates when slack is tight AND uncertainty is high.
    - Successor bottleneck couples upward_rank * remaining_work * (1 + smoothed inverse slack penalty).
    - Energy term uses robust min-max normalization.
    - Anti-starvation via linear ready_wait_time scaled by median_wait.
    - All parameters declared in schema are used; no unused or missing entries.
    - No numeric literals beyond -2, -1, 0, 1, 2; all thresholds/weights exposed via PARAMS.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust min-max normalization
    def minmax_normalize(x):
        x_min, x_max = np.min(x), np.max(x)
        denom = x_max - x_min
        if denom < eps:
            return np.zeros_like(x)
        return (x - x_min) / (denom + eps)
    
    # Smooth tanh-based urgency: maps slack → [0, 1], high urgency for negative slack
    median_abs_slack = np.abs(np.median(slack)) if N > 0 else eps
    tanh_input = np.clip(slack / (median_abs_slack + eps), -PARAMS["tanh_clip_bound"], PARAMS["tanh_clip_bound"])
    urgency_raw = 0.5 * (1.0 + np.tanh(-tanh_input))
    norm_urgency = minmax_normalize(urgency_raw)
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = minmax_normalize(energy_per_duration)
    
    # Successor bottleneck: upward_rank * remaining_work * (1 + smoothed inverse slack penalty)
    inv_slack_penalty = 1.0 / (np.abs(slack) + eps)
    inv_slack_penalty = np.clip(inv_slack_penalty, 0.0, 100.0)
    bottleneck_pressure = upward_rank * remaining_work * (1.0 + inv_slack_penalty) * (1.0 + uncertainty)
    norm_bottleneck = minmax_normalize(bottleneck_pressure)
    
    # Joint DDL-uncertainty protection gate
    median_slack = np.median(slack) if N > 0 else 0.0
    tight_slack_condition = (slack <= median_slack + eps).astype(float)
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    high_uncertainty_condition = (uncertainty >= PARAMS["ddl_protection_gate_threshold"] * max_uncertainty).astype(float)
    ddl_uncertainty_gate = tight_slack_condition * high_uncertainty_condition
    
    # Anti-starvation: linear ready_wait_time term, scaled by median_wait
    if N == 1:
        median_wait = ready_wait_time[0]
    else:
        median_wait = np.median(ready_wait_time)
    norm_wait = minmax_normalize(ready_wait_time / (median_wait + eps))
    
    # Final score: smaller = higher priority
    score = (
        norm_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + ddl_uncertainty_gate * upward_rank
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
    )
    
    # Ensure finite output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
