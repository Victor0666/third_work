def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaces IQR normalization for slack & uncertainty with tunable min-max normalization using declared range fractions,
        improving robustness under sparse or skewed distributions (per reflection).
      - Introduces successor-slack-risk interaction: penalizes tasks whose immediate successors collectively exhibit high negative slack,
        directly addressing DDL-protection via descendant urgency (replacing prior remaining_work × energy coupling).
      - Adds hard DDL-protection gate: when critical_path_bonus is active AND slack < median, multiplies slack_penalty by 2.0 to enforce strict deadline safety.
    
    All other components retain proven structure from Parent 2 + Parent 1 hybrid: arctan wait boost, uncertainty-slack coupling, 
    percentile-gated critical bonus, and finite-output safeguards.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Min-max normalization with tunable range fraction for slack and uncertainty (robust under skew)
    def minmax_normalize(x, range_frac=1.0):
        x = np.copy(x)
        x_min, x_max = np.min(x), np.max(x)
        if x_max == x_min:
            return np.zeros_like(x)
        span = (x_max - x_min) * range_frac
        center = (x_max + x_min) / 2.0
        half_span = span / 2.0
        lo = center - half_span
        hi = center + half_span
        # Clamp to [lo, hi] then normalize to [-1, 1]
        x_clipped = np.clip(x, lo, hi)
        return (x_clipped - lo) / (hi - lo + eps_safe) * 2.0 - 1.0

    slack_norm = minmax_normalize(slack, PARAMS["min_max_slack_range"])
    unc_norm = minmax_normalize(uncertainty, PARAMS["min_max_uncertainty_range"])

    # Slack: piecewise — strong penalty for negative slack, gentle reward for positive slack
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_risk_penalty"] * (-slack_norm),
        -PARAMS["slack_urgency_gain"] * slack_norm
    )

    # Energy efficiency: marginal energy per unit of total duration (J/s), normalized via IQR (retained for others)
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, 25.0)  # fixed IQR for stability; avoids parameter bloat
        q_high = np.percentile(x, 75.0)
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps_safe else (np.max(np.abs(x - center)) + eps_safe)
        return (x - center) / (denom + eps_safe)

    duration = np.maximum(min_exec_time + min_comm_time, eps_safe)
    energy_per_duration = min_incremental_energy / duration
    energy_eff_norm = iqr_normalize(energy_per_duration)

    # Critical-path bonus: activated only for top percentile AND tight slack (< median)
    rank_norm = iqr_normalize(upward_rank)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps_safe)
    median_slack = np.median(slack)
    critical_gate = np.where(
        (rank_percentile >= PARAMS["critical_rank_threshold"]) & (slack < median_slack),
        1.0,
        0.0
    )
    critical_bonus = PARAMS["critical_path_bonus"] * rank_norm * critical_gate

    # Waiting-time boost: bounded arctan saturation
    wait_scaled = ready_wait_time / (PARAMS["wait_decay_rate"] + eps_safe)
    wait_boost = (2.0 / np.pi) * np.arctan(wait_scaled)
    wait_norm = iqr_normalize(wait_boost)

    # Uncertainty-slack coupling: amplify urgency only under high uncertainty AND negative slack
    slack_pressure = np.maximum(-slack_norm, 0.0)
    coupled_urgency = PARAMS["uncertainty_slack_coupling"] * unc_norm * slack_pressure

    # Duration-efficiency ratio
    dur_eff_ratio = PARAMS["duration_efficiency_ratio"] * iqr_normalize(duration)

    # Successor-slack-risk interaction: sum of negative slack across immediate successors (simulated via input proxy)
    # Since true DAG successor structure isn't provided, we approximate using task-level slack correlation:
    # penalize tasks with low slack that co-occur with high-magnitude negative slack elsewhere (proxy for downstream risk)
    # Use median-centered slack deviation as lightweight surrogate for aggregate successor risk
    slack_dev = slack - median_slack
    # Aggregate risk: mean of negative slack deviations among all tasks (global pressure signal)
    global_neg_slack_mean = np.mean(np.where(slack < 0, slack_dev, 0.0))
    # Local amplification: task's own negative slack deviation weighted by global pressure
    successor_slack_risk = PARAMS["successor_slack_risk_weight"] * slack_dev * np.sign(global_neg_slack_mean)
    # Normalize to avoid dominance
    successor_risk_norm = iqr_normalize(successor_slack_risk)

    # Hard DDL-protection gate: double slack penalty when both criticality and tight slack align
    ddl_protection_gate = np.where(
        (rank_percentile >= PARAMS["critical_rank_threshold"]) & (slack < median_slack),
        2.0,
        1.0
    )
    slack_penalty = slack_penalty * ddl_protection_gate

    # Final score: smaller = higher priority
    score = (
        slack_penalty
        + PARAMS["energy_efficiency_weight"] * energy_eff_norm
        - critical_bonus
        - wait_norm
        + coupled_urgency
        + dur_eff_ratio
        + successor_risk_norm
    )

    # Ensure finite output, shape compliance, and numerical safety
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
