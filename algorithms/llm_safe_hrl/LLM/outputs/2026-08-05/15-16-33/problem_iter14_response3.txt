def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths from both parents:
    - Retains Parent 2's adaptive linear urgency with variance-gated modulation and joint DDL pressure.
    - Adds Parent 1's critical-path bonus (but reparameterized and gated by percentile).
    - Replaces sigmoid wait saturation with arctan-based bounded [0,1] saturation (more numerically stable than exp).
    - Uses unified adaptive_normalize with IQR→min-max fallback for all features.
    - Introduces novel *critical-path amplification*: only top-percentile upward_rank tasks receive multiplicative
      urgency boost (not additive bonus), improving deadline safety for truly critical nodes.
    - All operations are vectorized; zero Python conditionals; exactly one boolean mask (joint_ddl_pressure) reused.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max when dispersion is low
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR too small (low dispersion), else use IQR
        if iqr > eps:
            denom = iqr
        else:
            denom = np.max(x) - np.min(x)
            if denom < eps:
                denom = eps
        
        return (x - center) / (denom + eps)
    
    # Linear urgency proxy: (median_slack - slack), clipped to [0, 2]
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    
    # Slack variance sufficiency: activate modulation only if variance is high enough
    slack_range = np.max(slack) - np.min(slack) if N > 0 else eps
    slack_var = np.var(slack) if N > 1 else 0.0
    variance_sufficient = (slack_var > PARAMS["slack_variance_activation_threshold"] * (slack_range + eps)**2).astype(float)
    
    # Joint DDL pressure: slack < median AND uncertainty > threshold → single boolean mask
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    joint_ddl_pressure = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Modulated urgency: boost only when both variance sufficient and joint pressure holds
    modulated_urgency = urgency_linear * (1.0 + PARAMS["urgency_uncertainty_coupling_strength"] * joint_ddl_pressure * variance_sufficient)
    norm_urgency = adaptive_normalize(modulated_urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration * upward_rank * (1 + urgency) — captures criticality & urgency coupling
    bottleneck_pressure = duration * upward_rank * (1.0 + modulated_urgency + eps)
    
    # Hard feasibility guard for uncertainty amplification (separate from joint_ddl_pressure)
    bottleneck_guard = (
        (slack < median_slack) &
        (uncertainty > PARAMS["bottleneck_feasibility_guard"] * max_uncertainty)
    ).astype(float)
    bottleneck_pressure = bottleneck_pressure * np.power(1.0 + uncertainty, 
                                                         PARAMS["bottleneck_uncertainty_amplification"] * bottleneck_guard)
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Ready wait time: smooth saturation via arctan (bounded [0,1], avoids exp overflow)
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = adaptive_normalize(wait_saturation)
    
    # Critical-path amplification: multiplicative urgency boost only for top percentile of upward_rank
    norm_rank = adaptive_normalize(upward_rank)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_norm_ranks = np.sort(norm_rank)
        rank_idx = np.searchsorted(sorted_norm_ranks, norm_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_activation_threshold"], 1.0, 0.0)
    critical_urgency_boost = critical_gate * modulated_urgency
    
    # DDL risk amplifier: joint pressure × normalized uncertainty (for risk-aware prioritization)
    norm_uncertainty = adaptive_normalize(uncertainty)
    ddl_risk_amplifier = joint_ddl_pressure * norm_uncertainty
    
    # Final score: smaller = higher priority
    # Lexicographic emphasis: deadline safety (urgency + critical boost) > bottleneck release > energy > fairness
    score = (
        norm_urgency
        + critical_urgency_boost  # Novel multiplicative critical-path amplification
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
        + ddl_risk_amplifier
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
