def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's stability with Parent 1's hard-DDL enforcement.
    
    Key structural improvements:
      - Introduces *adaptive urgency amplification*: when slack < urgency_amplification_threshold,
        urgency is multiplied by urgency_amplification_factor — a clean, interpretable, bounded
        mechanism to enforce hard deadlines without fragile sigmoid or binary gating.
      - Retains Parent 2's robust bottleneck term: duration × (upward_rank × remaining_work),
        confirmed superior in unblocking critical successors.
      - Keeps linear urgency mapping (sliding [-1,1] via median-normalized slack) for monotonicity,
        but adds amplification only where needed — avoids over-suppression of non-critical tasks.
      - Drops all inactive features (critical-rank gate, release cap, feasibility guard mask) per analysis.
      - Uses elite-tuned IQR percentiles (35/77) for outlier-resilient normalization.
      - DDL-protection gate now uses *median-relative slack* AND *scaled uncertainty*, matching Parent 2's high-confidence logic.
      - Ready wait time remains linear ramp for fairness, anti-starvation, and simplicity.
      - All operations guarded against NaN/inf/zero; deterministic and finite output.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using hybrid-tuned percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Linear urgency: maps slack → [-1,1] relative to median_slack; negative slack → positive urgency
    median_slack = np.median(slack)
    urgency_base = np.clip(-slack / (np.abs(median_slack) + eps), -1.0, 1.0)
    
    # Adaptive amplification: only when slack is critically negative (hard-DDL zone)
    amplification_mask = (slack < PARAMS["urgency_amplification_threshold"]).astype(float)
    urgency_amplified = urgency_base * (1.0 + PARAMS["urgency_amplification_factor"] * amplification_mask)
    norm_urgency = iqr_normalize(urgency_amplified)
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor bottleneck coupling: joint criticality & load pressure — proven most effective
    bottleneck_pressure = duration * (upward_rank * remaining_work + eps)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    
    # Ready wait time: bounded linear ramp [0,1] — anti-starvation, monotonic, simple
    wait_ramp = np.clip(ready_wait_time / PARAMS["wait_ramp_scale"], 0.0, 1.0)
    norm_wait = iqr_normalize(wait_ramp)
    
    # DDL-protection gate: activates only under high-confidence risk (tight slack AND high uncertainty)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Uncertainty-normalized amplification term — only active when DDL is truly at risk
    norm_uncertainty = iqr_normalize(uncertainty)
    
    # Final score: smaller = higher priority
    # Hierarchy: amplified urgency (hard-DDL safety) > bottleneck release > energy > fairness > risk amplification
    score = (
        norm_urgency  # Primary, adaptive deadline signal
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock critical paths
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Penalize inefficient energy use
        - norm_wait  # Favor fresher tasks (anti-starvation)
        + ddl_protection_active * norm_uncertainty  # Risk-aware boost only when both signals align
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
