def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
    - Hard DDL-first gating: tasks with slack <= 0 receive top priority via additive penalty.
    - Rank-based robust normalization using quantile mapping with alpha-tail control.
    - Unified uncertainty-slack interaction exponent for risk amplifier.
    - Bottleneck pressure raised to tunable exponent before normalization.
    - Exactly 12 parameters; all used once.
    - All numeric literals in {-2,-1,0,1,2}.
    - No branching — fully vectorized.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Rank-based normalization: map x to [-1,1] via quantile rank + alpha stretch
    def rank_normalize(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros(0, dtype=float)
        if N == 1:
            return np.array([0.0], dtype=float)
        ranks = np.argsort(np.argsort(x))  # dense rank
        percentile = (ranks + 1.0) / (len(x) + 1.0)  # avoid 0/1 edge issues
        stretched = np.power(percentile, PARAMS["rank_based_normalization_alpha"])
        return stretched * 2.0 - 1.0  # [0,1] → [-1,1]
    
    # Hard DDL priority gate: assign massive penalty to overdue tasks (slack <= 0)
    ddl_violation_mask = (slack <= 0.0).astype(float)
    ddl_penalty = ddl_violation_mask * PARAMS["ddl_hard_priority_weight"]
    
    # Urgency: linear proxy (median_slack - slack), clipped to [0,2]
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    
    # Slack variance sufficiency for modulation
    slack_range = np.max(slack) - np.min(slack) if N > 0 else eps
    slack_var = np.var(slack) if N > 1 else 0.0
    variance_sufficient = (slack_var > PARAMS["slack_variance_activation_threshold"] * (slack_range + eps)**2).astype(float)
    
    # Joint DDL pressure: slack < median AND uncertainty high relative to max
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    joint_ddl_pressure = (
        (slack < median_slack) &
        (uncertainty > PARAMS["joint_ddl_uncertainty_threshold_factor"] * max_uncertainty)
    ).astype(float)
    
    # Modulated urgency
    modulated_urgency = urgency_linear * (1.0 + PARAMS["urgency_uncertainty_coupling_strength"] * joint_ddl_pressure * variance_sufficient)
    norm_urgency = rank_normalize(modulated_urgency)
    
    # Energy efficiency: marginal energy per unit duration
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = rank_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration * upward_rank * (1 + urgency), raised to exponent
    bottleneck_pressure = duration * upward_rank * (1.0 + modulated_urgency + eps)
    bottleneck_pressure = np.power(bottleneck_pressure + eps, PARAMS["bottleneck_pressure_exponent"])
    norm_bottleneck = rank_normalize(bottleneck_pressure)
    
    # Ready wait saturation: arctan-based bounded growth
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # maps R+ → [0,1)
    norm_wait = rank_normalize(wait_saturation)
    
    # Critical path bonus: top percentile of upward_rank
    rank_percentile_threshold = PARAMS["critical_rank_activation_threshold"] * 100.0  # fixed API arg for np.percentile
    rank_threshold = np.percentile(upward_rank, rank_percentile_threshold)
    critical_gate = (upward_rank >= rank_threshold).astype(float)
    norm_rank = rank_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Risk amplifier: unified slack-uncertainty interaction with exponent
    # Use absolute slack to avoid sign inversion; add eps for safety
    risk_input = np.abs(slack) + eps
    risk_amplifier = np.power(uncertainty / (risk_input + eps), PARAMS["uncertainty_slack_interaction_exponent"])
    norm_risk = rank_normalize(risk_amplifier)
    
    # Final score: smaller = higher priority
    score = (
        ddl_penalty +
        norm_urgency +
        critical_bonus +
        PARAMS["successor_bottleneck_coupling"] * norm_bottleneck +
        PARAMS["energy_duration_ratio_weight"] * norm_energy_eff -
        norm_wait +
        norm_risk
    )
    
    # Ensure finite output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
