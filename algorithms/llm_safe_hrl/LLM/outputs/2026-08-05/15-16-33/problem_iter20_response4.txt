def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths from both parents:
    - Uses Parent 2's slack-feasibility gating on bottleneck pressure (critical for DDL safety)
    - Adds Parent 1's DDL-risk conditional urgency amplification (improves responsiveness under lateness)
    - Replaces artificial urgency clipping with slack-feasibility-gated urgency: zero when slack < 0
    - Introduces novel dual-gating: bottleneck suppression AND urgency amplification both conditioned on verified DDL risk
    - Keeps blended fairness and uncertainty-aware normalization for robustness
    - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants beyond that set
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Uncertainty-aware adaptive normalization: dispersion scaled by std(uncertainty)
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        
        unc_std = np.std(uncertainty) if N > 1 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
        
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = dispersion if dispersion > eps else fallback_range
        
        return (x - center) / (denom + eps)
    
    # Slack-feasibility-gated urgency: zero when slack < 0, else (median_slack - slack), capped
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.where(slack >= 0.0, median_slack - slack, 0.0)
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.minimum(urgency_linear, urgency_cap)
    norm_urgency = adaptive_normalize(urgency_linear)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency)
    # Gated by slack feasibility: suppress amplification when slack is critically negative
    base_bottleneck = duration * upward_rank * remaining_work * (1.0 + urgency_linear + eps)
    
    # Feasibility gate: binary mask (0 or 1) based on slack threshold
    feasibility_mask = (slack >= PARAMS["slack_feasibility_gate_threshold"]).astype(float)
    
    # Uncertainty amplification applied only when feasible
    unc_normalized = adaptive_normalize(uncertainty)
    amp_factor = np.power(1.0 + unc_normalized, PARAMS["bottleneck_uncertainty_amplification"])
    bottleneck_pressure = base_bottleneck * amp_factor * feasibility_mask
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Anti-starvation: blend sigmoid and linear inverted components using tunable weight
    # Sigmoid component
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
    wait_sigmoid = 1.0 / (1.0 + np.exp(-wait_clipped))
    norm_wait_sigmoid = adaptive_normalize(wait_sigmoid)
    # Linear inverted component
    norm_wait_linear = 1.0 - adaptive_normalize(ready_wait_time)
    # Weighted blend
    norm_wait = (
        PARAMS["wait_fairness_blend_weight"] * norm_wait_sigmoid
        + (1.0 - PARAMS["wait_fairness_blend_weight"]) * norm_wait_linear
    )
    
    # Critical-path coupling term (pure structural importance, ungated)
    cp_coupling = upward_rank * remaining_work
    norm_cp_coupling = adaptive_normalize(cp_coupling)
    
    # DDL-risk activation gate: activated only when slack < 0 AND uncertainty exceeds scaled median
    median_uncertainty = np.median(uncertainty) if N > 0 else eps
    ddl_risk_gate = (
        (slack < 0.0) &
        (uncertainty > PARAMS["ddl_risk_activation_threshold"] * median_uncertainty)
    ).astype(float)
    
    # Risk-modulated urgency: amplify urgency signal only under verified DDL risk
    risk_amplified_urgency = norm_urgency * (1.0 + PARAMS["risk_modulated_urgency_weight"] * ddl_risk_gate)
    
    # Final score: smaller = higher priority
    # Hierarchy: risk-amplified urgency > bottleneck > energy > fairness > critical-path coupling
    score = (
        risk_amplified_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
        + PARAMS["upward_rank_remaining_work_coupling"] * norm_cp_coupling
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
