def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with median-based robust normalization, linear urgency suppression,
       hard DDL violation penalty, and unified bottleneck-pressure formulation.
    
    Key structural improvements:
      - Replaces percentile-based normalization with median + robust range (Q90-Q10) to avoid NaN when N<=2
        and improve stability under sparse/low-variance distributions.
      - Uses linear urgency suppression (not sigmoid) for monotonic, interpretable deadline pressure.
      - Hard DDL violation penalty applied *before* normalization to prevent score distortion.
      - Bottleneck pressure now combines duration, upward_rank, remaining_work, and urgency into one
        coherent term — no redundant coupling or double weighting.
      - All normalization uses same robust center/dispersion logic for consistency and reduced AST depth.
      - Removed inactive parameters from previous versions; retained only those actively used.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust normalization: median center + inter-decile range (Q90 - Q10) dispersion
    def robust_normalize(x):
        x = np.copy(x)
        if N > 2:
            q10 = np.percentile(x, 10.0)
            q50 = np.median(x)
            q90 = np.percentile(x, 90.0)
            dispersion = q90 - q10 + eps
        else:
            # Fallback: use max-min if too few samples, but clamp to avoid zero dispersion
            q50 = np.median(x) if N > 0 else 0.0
            dispersion = np.max(x) - np.min(x) if N > 0 else eps
            dispersion = max(dispersion, eps)
        return (x - q50) / (dispersion + eps)
    
    # Hard DDL violation penalty: assign maximum finite score before normalization
    finfo = np.finfo(float)
    is_hard_violated = slack < PARAMS["urgency_suppression_threshold"]  # reuse as hard gate
    raw_score = np.full(N, finfo.max, dtype=float)
    
    # Compute all components only for non-violated tasks
    valid_mask = ~is_hard_violated
    if np.any(valid_mask):
        # Linear urgency: zero when slack < threshold, else (median_slack - slack), capped
        median_slack = np.median(slack[valid_mask]) if np.sum(valid_mask) > 0 else 0.0
        urgency_linear = np.where(
            slack >= PARAMS["urgency_suppression_threshold"],
            np.clip(median_slack - slack, 0.0, 2.0),
            0.0
        )
        non_neg_slack = np.clip(slack, 0.0, None)
        max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
        urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
        urgency = np.minimum(urgency_linear, urgency_cap)
        norm_urgency = robust_normalize(urgency)
        
        # Energy efficiency: marginal energy per unit duration (J/s)
        duration = np.maximum(min_exec_time + min_comm_time, eps)
        energy_per_duration = min_incremental_energy / (duration + eps)
        norm_energy_eff = robust_normalize(energy_per_duration)
        
        # Successor-release pressure: structural importance × slack decay
        slack_abs = np.abs(slack) + eps
        release_decay = np.power(1.0 + slack_abs, -PARAMS["successor_release_pressure_exponent"])
        successor_release_pressure = upward_rank * remaining_work * release_decay
        norm_successor_release = robust_normalize(successor_release_pressure)
        
        # Unified bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency)
        # Feasibility-gated and uncertainty-amplified only when slack permits
        base_bottleneck = duration * upward_rank * remaining_work * (1.0 + urgency + eps)
        feasibility_mask = (slack >= PARAMS["slack_feasibility_gate_threshold"]).astype(float)
        unc_norm = robust_normalize(uncertainty)
        amp_factor = np.power(1.0 + np.abs(unc_norm), PARAMS["bottleneck_uncertainty_amplification"])
        bottleneck_pressure = base_bottleneck * amp_factor * feasibility_mask
        norm_bottleneck = robust_normalize(bottleneck_pressure)
        
        # Anti-starvation: blend sigmoid and linear inverted components
        wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
        wait_clipped = np.clip(wait_scaled, -2.0, 2.0)
        wait_sigmoid = 1.0 / (1.0 + np.exp(-wait_clipped))
        norm_wait_sigmoid = robust_normalize(wait_sigmoid)
        norm_wait_linear = 1.0 - robust_normalize(ready_wait_time)
        norm_wait = (
            PARAMS["wait_fairness_blend_weight"] * norm_wait_sigmoid
            + (1.0 - PARAMS["wait_fairness_blend_weight"]) * norm_wait_linear
        )
        
        # Critical-path coupling (pure structural importance)
        cp_coupling = upward_rank * remaining_work
        norm_cp_coupling = robust_normalize(cp_coupling)
        
        # Final score: smaller = higher priority
        score_valid = (
            norm_urgency
            + PARAMS["successor_bottleneck_coupling"] * norm_successor_release
            + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
            + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
            - norm_wait
            + PARAMS["upward_rank_remaining_work_coupling"] * norm_cp_coupling
        )
        raw_score[valid_mask] = score_valid[valid_mask]
    
    # Enforce finiteness and return
    raw_score = np.nan_to_num(raw_score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return raw_score.astype(float, copy=False)
