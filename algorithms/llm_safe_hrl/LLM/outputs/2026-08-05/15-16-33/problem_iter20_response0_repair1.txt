def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths from both parents with novel structural enhancements:
    - Replaces adaptive_normalize with robust uncertainty-aware IQR normalization using declared percentiles.
    - Introduces slack-feasibility gate on urgency (not just bottleneck) to suppress urgency when slack is critically negative.
    - Replaces blended fairness with saturating inverse wait-time: 1.0 - tanh(wait / scale), improving starvation resistance.
    - Uses multiplicative bottleneck pressure with uncertainty amplification gated by feasibility.
    - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants beyond that set.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.quantile(x, PARAMS["iqr_percentile_low"] / 100.0)
        q_high = np.quantile(x, PARAMS["iqr_percentile_high"] / 100.0)
        iqr = q_high - q_low + eps
        center = np.median(x)
        return (x - center) / (iqr + eps)
    
    # Feasibility-preserving urgency: linear urgency capped by non-negative slack exponent
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.minimum(urgency_linear, urgency_cap)
    
    # Urgency slack gate: disable urgency signal under critical deadline risk
    urgency_gate = (slack >= PARAMS["urgency_slack_gate_threshold"]).astype(float)
    gated_urgency = urgency_linear * urgency_gate
    norm_urgency = iqr_normalize(gated_urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency)
    base_bottleneck = duration * upward_rank * remaining_work * (1.0 + gated_urgency + eps)
    
    # Feasibility gate for bottleneck: suppress amplification when slack is critically negative
    feasibility_mask = (slack >= PARAMS["slack_feasibility_gate_threshold"]).astype(float)
    
    # Uncertainty amplification applied only when feasible
    unc_normalized = iqr_normalize(uncertainty)
    amp_factor = np.power(1.0 + unc_normalized, PARAMS["bottleneck_uncertainty_amplification"])
    bottleneck_pressure = base_bottleneck * amp_factor * feasibility_mask
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    
    # Anti-starvation: saturating inverse wait-time via tanh
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_tanh = np.tanh(wait_scaled)
    fairness_term = 1.0 - wait_tanh  # smaller wait → larger fairness_term → lower score → higher priority
    norm_fairness = iqr_normalize(fairness_term)
    
    # Critical-path coupling term (pure structural importance, ungated)
    cp_coupling = upward_rank * remaining_work
    norm_cp_coupling = iqr_normalize(cp_coupling)
    
    # Final score: smaller = higher priority
    # Hierarchy: urgency > bottleneck > energy > fairness > critical-path coupling
    score = (
        norm_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_fairness
        + PARAMS["upward_rank_remaining_work_coupling"] * norm_cp_coupling
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
