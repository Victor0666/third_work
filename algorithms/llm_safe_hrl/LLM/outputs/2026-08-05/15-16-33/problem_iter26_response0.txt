def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths from both parents:
      - Robust rank-based normalization (from Parent 2) for skewed distributions.
      - Hard deadline guard with finite penalty (Parent 2).
      - Scenario-aware activation gates for energy & bottleneck (Parent 2).
      - Novel urgency formulation: direct negative-slack power law (replacing linear ramp) for stronger lateness signal.
      - Unified uncertainty gating using same steepness for consistency.
      - Removed redundant parameters (ddl_protection_gate_threshold, critical_path_violation_penalty, urgency_cap_exponent) per analysis.
      - All nonlinearities clipped to [-2,2] or bounded domains to ensure stability.
      - No unbounded operations; all literals in {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Hard deadline guard: assign large finite penalty to infeasible tasks
    finfo = np.finfo(float)
    score = np.full(N, finfo.max, dtype=float)
    feasible_mask = slack >= -eps
    
    if not np.any(feasible_mask):
        return score
    
    # Robust rank-based normalization: maps values to [-1,1] via sorting-based rank approximation
    def rank_normalize(x):
        x = np.copy(x)
        if N <= 1 or np.all(x == x[0]):
            return np.zeros_like(x, dtype=float)
        sorted_idx = np.argsort(x)
        ranks = np.empty_like(sorted_idx, dtype=float)
        ranks[sorted_idx] = np.arange(N, dtype=float)
        norm_ranks = (2.0 * ranks / (N - 1.0)) - 1.0
        return np.clip(norm_ranks, -1.0, 1.0)
    
    # Novel urgency: direct negative-slack power law — stronger signal for critical lateness risk
    neg_slack = np.clip(-slack, 0.0, None)
    urgency = np.power(neg_slack + eps, PARAMS["urgency_decay_exponent"])
    norm_urgency = rank_normalize(urgency)
    
    # Raw energy efficiency: use min_incremental_energy directly (no duration coupling), MAD-normalized
    med_energy = np.median(min_incremental_energy) if N > 0 else 0.0
    abs_devs = np.abs(min_incremental_energy - med_energy)
    mad_energy = np.median(abs_devs) if N > 0 else eps
    denom_energy = PARAMS["mad_scale_factor"] * (mad_energy + eps)
    norm_energy_eff = (min_incremental_energy - med_energy) / (denom_energy + eps)
    norm_energy_eff = np.clip(norm_energy_eff, -1.0, 1.0)
    
    # Uncertainty-modulated energy term: activated only above threshold, using declared steepness
    unc_centered = uncertainty - PARAMS["uncertainty_activation_threshold"]
    unc_clipped = np.clip(unc_centered, -2.0, 2.0)
    unc_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * unc_clipped))
    energy_uncertainty_term = min_incremental_energy * unc_gate
    norm_energy_uncertain = rank_normalize(energy_uncertainty_term)
    
    # Critical-path bottleneck: activated only above uncertainty threshold, using clipped steepness
    bottleneck_gate = np.where(uncertainty >= PARAMS["uncertainty_activation_threshold"],
                               1.0 / (1.0 + np.exp(-PARAMS["bottleneck_uncertainty_steepness"] * 
                                                  np.clip(unc_centered, 0.0, 2.0))),
                               0.0)
    bottleneck_pressure = upward_rank * remaining_work * bottleneck_gate
    norm_bottleneck = rank_normalize(bottleneck_pressure)
    
    # Anti-starvation: rank-normalized wait time inversion (restored from v0 for stronger fairness signal)
    norm_wait = rank_normalize(ready_wait_time)
    inv_wait = 1.0 - norm_wait
    
    # Compute final score *only* for feasible tasks
    local_score = (
        norm_urgency
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        + PARAMS["energy_uncertainty_coupling"] * norm_energy_uncertain
        + PARAMS["critical_path_coupling"] * norm_bottleneck
        - PARAMS["wait_fairness_weight"] * inv_wait
    )
    
    # Apply computed score only where feasible; penalize infeasible heavily but finitely
    score[feasible_mask] = local_score[feasible_mask]
    score[~feasible_mask] = PARAMS["ddl_hard_guard_weight"] * finfo.max
    
    # Enforce finiteness and sanitize
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
