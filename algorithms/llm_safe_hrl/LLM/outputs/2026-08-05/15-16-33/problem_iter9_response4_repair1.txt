def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with feasibility-aware gating and local robust scaling.
    
    Key structural improvements:
      - Replaces global IQR normalization with *task-local median absolute deviation (MAD)* scaling
        for outlier resilience in heterogeneous ready sets — avoids distortion from extreme outliers.
      - Introduces *feasibility-aware gating*: nullifies energy & uncertainty contributions when slack > threshold,
        preventing over-penalization of low-risk tasks and aligning with hard deadline constraint priority.
      - Decouples uncertainty: now *multiplicatively modulates bottleneck pressure* (not additive),
        reflecting that risk amplifies blocking impact — not independent penalty.
      - All normalization uses local center/scale per feature; no cross-task contamination.
      - Maintains smooth sigmoid urgency, arctan wait saturation, and percentile-gated critical bonus.
      - Deterministic, finite, and fully guarded against NaN/inf/zero.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Local robust normalization: median + MAD (more outlier-resilient than IQR)
    def mad_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        abs_devs = np.abs(x - center)
        mad = np.median(abs_devs)
        denom = mad if mad > eps else (np.max(abs_devs) + eps)
        return (x - center) / (denom + eps)
    
    # Sigmoid urgency gate: smooth, differentiable, centered at offset
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered, 
                            -PARAMS["sigmoid_clip_bound"], 
                            PARAMS["sigmoid_clip_bound"])
    urgency_gate = 2.0 / (1.0 + np.exp(sigmoid_input)) - 1.0  # [-1, 1]
    norm_urgency = mad_normalize(urgency_gate)
    
    # Energy efficiency: marginal energy per unit duration (J/s), locally normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = mad_normalize(energy_per_duration)
    
    # Successor bottleneck coupling: duration * (remaining_work + upward_rank + eps)
    bottleneck_pressure = duration * (remaining_work + upward_rank + eps)
    norm_bottleneck = mad_normalize(bottleneck_pressure)
    
    # Critical-path bonus: activated only for top percentile of upward_rank
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_percentile"], 1.0, 0.0)
    norm_rank = mad_normalize(upward_rank)
    critical_bonus = critical_gate * norm_rank
    
    # Ready wait time: smooth saturation via arctan
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_time"] + eps)
    wait_saturation = (2.0 / np.pi) * np.arctan(wait_scaled)  # [0,1]
    norm_wait = mad_normalize(wait_saturation)
    
    # Feasibility-aware gating: deactivate energy & uncertainty terms when slack is safely positive
    # Use fixed threshold parameter instead of adding new one
    feasibility_gate = np.where(slack > 5.0, 0.0, 1.0)
    
    # DDL-protection gate: activates only when slack is tight relative to median AND uncertainty is high
    median_slack = np.median(slack)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Uncertainty-modulated bottleneck: multiplicative coupling reflects risk-amplified blocking
    norm_uncertainty = mad_normalize(uncertainty)
    bottleneck_with_risk = norm_bottleneck * (1.0 + 0.35 * norm_uncertainty)
    
    # Final score: smaller = higher priority
    # Urgency dominates; bottleneck unblocks successors (risk-amplified); energy only when needed;
    # critical bonus prioritizes path; wait reward ensures fairness; feasibility gating prevents over-penalization
    score = (
        norm_urgency  # Primary deadline pressure
        + PARAMS["successor_bottleneck_coupling"] * bottleneck_with_risk  # Risk-aware blocking pressure
        + feasibility_gate * PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Energy only when needed
        - critical_bonus  # Boost critical path
        - norm_wait  # Reward recent arrivals
        + ddl_protection_active * PARAMS["uncertainty_amplification_weight"] * norm_uncertainty  # Risk margin under stress
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
