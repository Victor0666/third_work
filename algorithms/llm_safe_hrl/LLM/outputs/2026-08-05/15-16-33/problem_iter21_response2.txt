def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
    - Hard urgency gating: when slack < urgency_hard_gate_threshold, urgency term overrides all others (score = norm_urgency)
    - Bottleneck softening: power-law compression (exponent < 1) prevents domination by extremely large upward_rank/duration products
    - Risk-gated energy coupling: multiplicative (not additive) penalty only when uncertainty exceeds median → preserves urgency dominance under safety
    - Unified adaptive normalization with fallback to std-based scaling when IQR collapses
    - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants beyond that set
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Adaptive normalization using declared IQR percentiles, with std fallback if IQR ≈ 0
    def adaptive_normalize(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros_like(x)
        q_low = np.percentile(x, PARAMS["iqr_percentile_low"])
        q_high = np.percentile(x, PARAMS["iqr_percentile_high"])
        iqr = q_high - q_low + eps
        center = np.median(x)
        # Fallback to std if IQR is degenerate
        std_val = np.std(x) + eps
        dispersion = np.where(iqr > eps * 10.0, iqr, std_val)
        return (x - center) / dispersion
    
    # Bounded tanh urgency: maps slack → [-1,1], then clamped and scaled to [0,1]
    abs_slack = np.abs(slack)
    scale = np.median(abs_slack) if np.any(abs_slack > eps) else 1.0
    tanh_urgency = np.tanh(-slack / (scale * PARAMS["urgency_scale"] + eps))
    clamped_urgency = np.clip(tanh_urgency, PARAMS["urgency_clamp_lower"], 1.0)
    urgency = (clamped_urgency + 1.0) / 2.0  # map to [0,1]
    norm_urgency = adaptive_normalize(urgency)
    
    # Duration includes both compute and communication
    duration = min_exec_time + min_comm_time
    
    # Softened multiplicative bottleneck pressure: critical-path × duration × energy, raised to sublinear power
    bottleneck_pressure = (
        duration *
        upward_rank *
        (min_incremental_energy + eps)
    )
    # Apply softening exponent to prevent outlier domination
    softened_bottleneck = np.power(bottleneck_pressure + eps, PARAMS["bottleneck_softening_factor"])
    norm_bottleneck = adaptive_normalize(softened_bottleneck)
    
    # Risk-gated energy coupling: multiplicative penalty only under verified uncertainty risk
    median_uncertainty = np.median(uncertainty) if N > 0 else eps
    risk_gate = (uncertainty > median_uncertainty).astype(float)
    risk_adjusted_energy = min_incremental_energy * (1.0 + PARAMS["energy_uncertainty_coupling"] * risk_gate)
    norm_energy = adaptive_normalize(risk_adjusted_energy)
    
    # Anti-starvation: normalized ready_wait_time, subtracted (higher wait → lower score → higher priority)
    norm_wait = adaptive_normalize(ready_wait_time)
    fairness_term = -PARAMS["wait_fairness_weight"] * norm_wait  # subtractive fairness
    
    # Joint DDL-risk gate: activated only when slack < 0 AND uncertainty exceeds scaled median
    ddl_risk_gate = (
        (slack < 0.0) &
        (uncertainty > PARAMS["ddl_risk_activation_threshold"] * median_uncertainty)
    ).astype(float)
    
    # Conditional bottleneck amplification: only under verified DDL risk
    risk_amplified_bottleneck = norm_bottleneck * (1.0 + PARAMS["risk_modulated_bottleneck_weight"] * ddl_risk_gate)
    
    # Amplify urgency signal under verified stress
    stress_amplified_urgency = norm_urgency * (1.0 + ddl_risk_gate)
    
    # Hard urgency gating: when slack is critically negative, override all other terms
    hard_urgency_mask = (slack < PARAMS["urgency_hard_gate_threshold"]).astype(float)
    
    # Final score: smaller = higher priority
    # Under hard urgency condition: use only normalized urgency
    # Otherwise: combine urgency, bottleneck, energy, fairness
    base_score = (
        stress_amplified_urgency +
        PARAMS["successor_bottleneck_coupling"] * risk_amplified_bottleneck +
        norm_energy +
        fairness_term
    )
    score = np.where(hard_urgency_mask == 1.0, norm_urgency, base_score)
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
