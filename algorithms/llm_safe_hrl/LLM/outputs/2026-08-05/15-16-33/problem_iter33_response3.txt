def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with deadline proximity–uncertainty joint thresholding and adaptive urgency:
      - Replaces median-based DDL gating with robust quartile threshold: activates bottleneck/energy only when
        (slack <= median_slack) AND (uncertainty >= percentile(uncertainty, 75)), eliminating outlier sensitivity.
      - Removes all inactive parameters (bottleneck_uncertainty_amplification, uncertainty_dispersion_scale,
        wait_saturation_exponent) and replaces them with single adaptive_urgency_scale modulating fuzzy urgency.
      - Uses smooth fuzzy deadline violation score: sigmoid((median_slack - slack) / (eps + std_slack)),
        preserving gradient near hard deadlines without discontinuities or piecewise clipping.
      - Introduces wait_fairness_weight to control anti-starvation strength independently from urgency scaling.
      - All normalizations use epsilon-guarded DDL-aware min-max; no rank-based nor std-only normalization on small-N sets.
      - Energy term now uses sigmoid(uncertainty) instead of linear unc_normalized for bounded, interpretable coupling.
      - Final composition: fuzzy_urgency > critical_path > energy_uncertainty > fairness, all normalized and gated.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    finfo = np.finfo(float)
    
    # Compute robust statistics — handle edge cases
    median_slack = np.median(slack) if N > 0 else 0.0
    std_slack = np.std(slack) if N > 1 else eps
    
    # Fuzzy deadline violation score: smooth, differentiable, gradient-preserving
    # Uses std_slack in denominator for adaptive scaling near deadline boundary
    urgency_input = (median_slack - slack) / (std_slack + eps)
    # Clip input to avoid overflow in sigmoid
    urgency_input_clipped = np.clip(urgency_input, -2.0, 2.0)
    fuzzy_urgency = 1.0 / (1.0 + np.exp(-urgency_input_clipped))
    
    # Adaptive urgency scaling: (1 + std_uncertainty) * fuzzy_urgency
    std_uncertainty = np.std(uncertainty) if N > 1 else eps
    adaptive_urgency = fuzzy_urgency * (1.0 + PARAMS["adaptive_urgency_scale"] * std_uncertainty)
    
    # Normalize urgency using DDL-aware min-max (robust to small-N)
    u_min, u_max = np.min(adaptive_urgency), np.max(adaptive_urgency)
    u_range = np.maximum(u_max - u_min, eps)
    norm_urgency = (adaptive_urgency - u_min) / (u_range + eps)
    
    # Quartile-based DDL-protection activation: tight slack AND high uncertainty (75th percentile)
    q75_unc = np.percentile(uncertainty, 75) if N > 0 else np.max(uncertainty)
    ddl_protection_active = (slack <= median_slack) & (uncertainty >= q75_unc)
    
    # Unconditional critical-path pressure: upward_rank × remaining_work
    critical_path_pressure = upward_rank * remaining_work
    cp_min, cp_max = np.min(critical_path_pressure), np.max(critical_path_pressure)
    cp_range = np.maximum(cp_max - cp_min, eps)
    norm_critical_path = (critical_path_pressure - cp_min) / (cp_range + eps)
    
    # Uncertainty-modulated energy term: only active under DDL protection, using sigmoid for bounded coupling
    unc_sigmoid = 1.0 / (1.0 + np.exp(-(uncertainty - np.median(uncertainty)) / (eps + std_uncertainty)))
    energy_uncertainty_term = np.where(
        ddl_protection_active,
        min_incremental_energy * unc_sigmoid,
        0.0
    )
    e_min, e_max = np.min(energy_uncertainty_term), np.max(energy_uncertainty_term)
    e_range = np.maximum(e_max - e_min, eps)
    norm_energy_uncertain = np.where(
        ddl_protection_active,
        (energy_uncertainty_term - e_min) / (e_range + eps),
        0.0
    )
    
    # Fairness term: normalized wait time → inverted sigmoid → weighted
    w_min, w_max = np.min(ready_wait_time), np.max(ready_wait_time)
    w_range = np.maximum(w_max - w_min, eps)
    norm_wait_raw = (ready_wait_time - w_min) / (w_range + eps)
    # Apply bounded sigmoid and invert for priority (smaller = higher priority)
    wait_clipped = np.clip(norm_wait_raw, -2.0, 2.0)
    wait_sigmoid = 1.0 / (1.0 + np.exp(-wait_clipped))
    inv_wait = 1.0 - wait_sigmoid
    
    # Final score: smaller = higher priority
    # Ordering: fuzzy_urgency > critical_path > energy_uncertainty > fairness
    score = (
        norm_urgency
        + PARAMS["upward_rank_remaining_work_weight"] * norm_critical_path
        + PARAMS["energy_uncertainty_coupling"] * norm_energy_uncertain
        - PARAMS["wait_fairness_weight"] * inv_wait
    )
    
    # Enforce finiteness and shape compliance
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure output is (N,) float64 array
    return score.astype(float, copy=False)
