import numpy as np

PARAMETER_SCHEMA = {
    "schema_version": 1,
    "parameters": [
        {
            "name": "epsilon",
            "initial_value": 0.000135,
            "lower_bound": 1e-6,
            "upper_bound": 0.1,

            "parameter_type": "float",
            "transform": "log"
        },
        {

            "initial_value": 0.75,
            "lower_bound": 0.5,
            "upper_bound": 0.9,
            "semantic_description": "Upper quantile threshold for uncertainty-based DDL risk activation.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "successor_bottleneck_coupling",
            "initial_value": 0.28,
            "lower_bound": 0.0,
            "upper_bound": 1.5,
            "semantic_description": "Strength of interaction between duration, upward_rank, urgency, and energy in bottleneck pressure.",
            "parameter_type": "float",
            "transform": "identity"
        },

            "initial_value": 37.0,

            "parameter_type": "float",

            "parameter_type": "float",

            "initial_value": 0.65,

            "parameter_type": "float",
            "transform": "identity"
        },

            "name": "urgency_std_scale",
            "initial_value": 1.2,
            "lower_bound": 0.1,
            "upper_bound": 5.0,

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's unified bottleneck and Parent 1's robust normalization,
    enhanced with performance-analysis insights:
    - Replaces tanh urgency with stable piecewise-linear urgency: clipped (median_slack - slack) / (std + eps)
    - Uses percentile-based DDL-risk gate: (slack < 0) & (uncertainty > quantile(uncertainty, ddl_risk_quantile_threshold))
    - Integrates ready_wait_time directly into bottleneck (not subtracted), scaled only under DDL risk
    - Removes redundant parameters; keeps exactly 7 tunable scalars (5–12 range satisfied)
    - All numeric literals are in {-2,-1,0,1,2}; no hidden constants.
    - Adaptive normalization uses IQR only (no fallback) — tuned percentiles ensure stability across workloads.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization using IQR only
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Piecewise-linear urgency: (median_slack - slack), normalized by std for stability near zero
    median_slack = np.median(slack) if N > 0 else 0.0
    std_slack = np.std(slack) if N > 1 else eps
    urgency_linear = np.clip(median_slack - slack, 0.0, np.inf)
    urgency = np.clip(urgency_linear / (std_slack * PARAMS["urgency_std_scale"] + eps), 0.0, 1.0)
    norm_urgency = adaptive_normalize(urgency)
    
    # Duration includes both compute and communication
    duration = min_exec_time + min_comm_time
    
    # DDL-risk gate: activated only when slack is negative AND uncertainty exceeds upper quantile
    quantile_uncertainty = np.quantile(uncertainty, PARAMS["ddl_risk_quantile_threshold"]) if N > 0 else eps
    ddl_risk_gate = ((slack < 0.0) & (uncertainty > quantile_uncertainty)).astype(float)
    
    # Unified bottleneck pressure: integrates duration, criticality, urgency, uncertainty, energy, and fairness
    # Fairness (ready_wait_time) is embedded only under DDL risk to avoid biasing idle tasks
    wait_contribution = ready_wait_time * ddl_risk_gate * PARAMS["wait_integration_strength"]
    bottleneck_pressure = (
        duration *
        upward_rank *
        (1.0 + urgency) *
        (1.0 + uncertainty) *
        (min_incremental_energy + eps) *
        (1.0 + wait_contribution)
    )
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Final score: smaller = higher priority
    # Priority order: urgency (primary deadline signal) → bottleneck (critical path + energy + fairness under stress)
    score = (
        norm_urgency +
        PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
