def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with structural improvements guided by counterfactual evidence:
      - Replaces pre-normalized neg_slack with *risk-scaled DDL violation gate*: applies strong penalty only when slack < -eps, avoiding soft degradation.
      - Introduces explicit upward_rank × remaining_work interaction (validated by 'CRITICAL_PATH_STARVATION' replay failures).
      - Uses sigmoid-based urgency modulation instead of linear clipping: smoother transition near deadline boundary.
      - Removes bottleneck_uncertainty_amplification exponent; replaces with bounded power-law coupling using bottleneck_coupling_exponent.
      - Adds conditional DDL protection gate: if any task has slack < -eps, all non-violating tasks receive +inf priority boost to force violation resolution first.
      - All normalizations use robust median/std fallbacks and epsilon safeguards.
      - No unbounded ops, no branching beyond vectorized conditionals, no hidden constants beyond {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Adaptive normalization with robust dispersion estimation
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x) if N > 0 else 0.0
        
        # Use std(uncertainty) as risk-aligned dispersion base
        unc_std = np.std(uncertainty) if N > 1 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
        
        # Fallback to IQR or range if dispersion too small
        q75, q25 = np.percentile(x, [75, 25]) if N > 1 else (x[0], x[0]) if N == 1 else (0.0, 0.0)
        iqr = q75 - q25
        fallback_dispersion = iqr if iqr > eps else (np.max(x) - np.min(x) if N > 0 else eps)
        denom = dispersion if dispersion > eps else fallback_dispersion
        
        return (x - center) / (denom + eps)
    
    # Hard DDL violation detection: binary flag for any negative slack (structural safety net)
    has_violation = np.any(slack < -eps)
    
    # DDL risk penalty: infinite priority for violating tasks, zero otherwise — enforced before normalization
    ddl_violation_penalty = np.where(slack < -eps, np.inf, 0.0)
    
    # Conditional DDL protection gate: if any violation exists, non-violating tasks get massive penalty boost
    # This forces scheduler to resolve violations first — lexicographic priority over energy
    ddl_protection_boost = np.where(has_violation & (slack >= -eps), np.inf, 0.0)
    
    # Sigmoid urgency: smooth mapping from slack to urgency (higher when slack is low/negative)
    # Centered at median slack, scaled by uncertainty-aware dispersion
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_input = (median_slack - slack) / (np.std(slack) + eps) if N > 1 else 0.0
    urgency_sigmoid = 1.0 / (1.0 + np.exp(-np.clip(urgency_input, -2.0, 2.0)))
    
    # Bottleneck pressure: duration × (upward_rank × remaining_work) × (1 + uncertainty)^exponent
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    critical_bottleneck = upward_rank * remaining_work
    bottleneck_pressure = (
        duration * 
        critical_bottleneck * 
        np.power(1.0 + uncertainty, PARAMS["bottleneck_coupling_exponent"])
    )
    
    # Anti-starvation: saturating inverse wait-time via tanh (replaces sigmoid for better gradient behavior)
    wait_normalized = ready_wait_time / (PARAMS["wait_saturation_scale"] + eps)
    wait_boost = 1.0 - np.tanh(np.clip(wait_normalized, 0.0, 2.0))
    
    # Energy efficiency: marginal energy per unit duration, normalized
    energy_per_duration = min_incremental_energy / (duration + eps)
    
    # Normalize components
    norm_urgency = adaptive_normalize(urgency_sigmoid)
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    norm_energy = adaptive_normalize(energy_per_duration)
    norm_wait = adaptive_normalize(wait_boost)
    
    # Final score: smaller = higher priority
    # Lexicographic ordering enforced: DDL violation > DDL protection > urgency > bottleneck > energy > fairness
    score = (
        PARAMS["ddl_risk_penalty_weight"] * ddl_violation_penalty +
        ddl_protection_boost +
        norm_urgency +
        PARAMS["upward_rank_remaining_work_interaction"] * norm_bottleneck +
        PARAMS["energy_duration_ratio_weight"] * norm_energy -
        norm_wait
    )
    
    # Enforce finiteness and determinism
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
