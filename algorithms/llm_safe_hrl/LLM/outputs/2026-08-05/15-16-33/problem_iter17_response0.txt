def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining best elements from both parents:
    - Keeps Parent 2's robust IQR→min-max adaptive normalization and DDL-protection gate.
    - Integrates Parent 1's feasibility-preserving urgency cap to prevent inversion under negative slack.
    - Adds uncertainty-amplified bottleneck pressure using power-law (1+unc)^exponent instead of linear coupling.
    - Uses unified bottleneck: duration × upward_rank × remaining_work × (1 + urgency) × (1 + unc)^amplification.
    - All numeric literals strictly in {-2,-1,0,1,2}; no hidden state or unbounded ops.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization: IQR fallback to min-max when dispersion is low
    def adaptive_normalize(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros_like(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low
        center = np.median(x)
        
        # Use min-max if IQR too small (low dispersion), else use IQR
        if iqr < eps:
            x_min, x_max = np.min(x), np.max(x)
            denom = x_max - x_min
            if denom < eps:
                return np.zeros_like(x)
            return (x - x_min) / (denom + eps)
        else:
            denom = iqr
            return (x - center) / (denom + eps)
    
    # Sigmoid urgency gate: maps slack → [0, 1] where negative slack → near 1.0 (high urgency)
    slack_centered = slack - PARAMS["slack_sigmoid_offset"]
    sigmoid_input = np.clip(-PARAMS["slack_sigmoid_steepness"] * slack_centered,
                            -PARAMS["sigmoid_clip_bound"],
                            PARAMS["sigmoid_clip_bound"])
    urgency_raw = 1.0 / (1.0 + np.exp(sigmoid_input))  # range [0, 1]
    
    # Feasibility-preserving urgency cap: clamp raw urgency to avoid inversion under extreme negative slack
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.clip(PARAMS["slack_sigmoid_steepness"] * (max_non_neg_slack - slack + eps), 0.0, urgency_cap)
    urgency = np.minimum(urgency_raw, np.clip(urgency_linear, 0.0, 1.0))
    
    norm_urgency = adaptive_normalize(urgency)
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency) × (1 + uncertainty)^amplification
    # Power-law uncertainty amplification improves sensitivity to high-risk tasks without over-saturation
    unc_power = np.power(1.0 + uncertainty + eps, PARAMS["bottleneck_uncertainty_amplification"])
    bottleneck_pressure = duration * upward_rank * remaining_work * (1.0 + urgency + eps) * unc_power
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Adaptive linear ramp anti-starvation: saturates at median_wait * factor
    if N == 1:
        median_wait = ready_wait_time[0]
    else:
        median_wait = np.median(ready_wait_time)
    saturation_wait = median_wait * PARAMS["wait_ramp_saturation_factor"]
    wait_ramp = np.clip(ready_wait_time / (saturation_wait + eps), 0.0, 1.0)
    norm_wait = adaptive_normalize(wait_ramp)
    
    # Slack-conditioned DDL-protection gate: activated only when slack < 0 AND uncertainty high, weighted by upward_rank
    max_uncertainty = np.max(uncertainty) if N > 0 else eps
    ddl_risk_condition = (
        (slack < 0.0) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    ddl_risk_amplification = ddl_risk_condition * upward_rank
    
    # Final score: smaller = higher priority
    # Hierarchy: deadline safety > risk-amplified bottleneck > DDL-risk amplification > energy > fairness
    score = (
        norm_urgency  # Primary deadline signal
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock successors under joint risk & load
        + ddl_risk_amplification  # Directly amplify priority for high-upward-rank tasks under hard DDL violation
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Energy efficiency
        - norm_wait  # Favor recently arrived tasks (adaptive fairness)
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
