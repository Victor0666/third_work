def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with scenario-aware normalization and bounded piecewise urgency gating.
    
    Key improvements:
      - Replaces adaptive amplification with a *bounded piecewise urgency gate*: activates only when
        slack < median_slack * deadline_risk_gate_factor — avoids over-amplification in mild-latency regimes
        while preserving hard-DDL enforcement where truly needed.
      - Introduces *scenario-aware IQR normalization*: uses distinct percentile bounds for high- vs low-uncertainty
        tasks, gated by uncertainty quantile (median), improving robustness across heterogeneous workload regimes.
      - Removes inactive parameters (urgency_amplification_threshold/factor) and consolidates deadline risk
        into one interpretable, bounded conditional term.
      - Retains proven bottleneck coupling and DDL-protection gate logic; all operations remain finite, deterministic,
        and compliant with interface contract.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Scenario-aware IQR normalization: select percentiles based on uncertainty regime
    uncertainty_median = np.median(uncertainty) if N > 0 else 0.0
    high_uncertainty_mask = (uncertainty > uncertainty_median).astype(float)
    
    # Precompute both IQR configurations
    q_low_high = PARAMS["iqr_low_percentile_high_uncertainty"]
    q_high_high = PARAMS["iqr_high_percentile_high_uncertainty"]
    q_low_low = PARAMS["iqr_low_percentile_low_uncertainty"]
    q_high_low = PARAMS["iqr_high_percentile_low_uncertainty"]
    
    def iqr_normalize(x):
        x = np.copy(x)
        # Choose percentiles based on uncertainty regime
        q_low = np.where(high_uncertainty_mask == 1.0, q_low_high, q_low_low)
        q_high = np.where(high_uncertainty_mask == 1.0, q_high_high, q_high_low)
        # Compute per-task IQR using broadcasted percentiles — but np.percentile doesn't support per-element,
        # so we use global thresholds per regime: compute two normalizations and blend
        center = np.median(x)
        
        # Compute IQR for high-uncertainty subset
        x_high = x[high_uncertainty_mask == 1.0] if np.any(high_uncertainty_mask) else np.array([center])
        q_low_val_high = np.percentile(x_high, q_low_high) if len(x_high) > 0 else center
        q_high_val_high = np.percentile(x_high, q_high_high) if len(x_high) > 0 else center
        iqr_high = q_high_val_high - q_low_val_high
        
        # Compute IQR for low-uncertainty subset
        x_low = x[high_uncertainty_mask == 0.0] if np.any(high_uncertainty_mask == 0.0) else np.array([center])
        q_low_val_low = np.percentile(x_low, q_low_low) if len(x_low) > 0 else center
        q_high_val_low = np.percentile(x_low, q_high_low) if len(x_low) > 0 else center
        iqr_low = q_high_val_low - q_low_val_low
        
        # Blend denominators
        denom = np.where(high_uncertainty_mask == 1.0,
                        iqr_high if iqr_high > eps else (np.max(np.abs(x - center)) + eps),
                        iqr_low if iqr_low > eps else (np.max(np.abs(x - center)) + eps))
        denom = np.where(denom == 0, eps, denom)
        return (x - center) / (denom + eps)
    
    # Bounded piecewise urgency: activate only under severe deadline risk
    median_slack = np.median(slack)
    deadline_risk_threshold = PARAMS["deadline_risk_gate_factor"] * median_slack
    urgency_base = np.clip(-slack / (np.abs(median_slack) + eps), -1.0, 1.0)
    # Apply bounded gate: only amplify urgency when slack is critically below threshold
    urgency_gate = (slack < deadline_risk_threshold).astype(float)
    urgency_final = urgency_base * (1.0 + urgency_gate)  # simple doubling under critical risk
    norm_urgency = iqr_normalize(urgency_final)
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor bottleneck coupling: joint criticality & load pressure
    bottleneck_pressure = duration * (upward_rank * remaining_work + eps)
    norm_bottleneck = iqr_normalize(bottleneck_pressure)
    
    # Ready wait time: bounded linear ramp [0,1] — anti-starvation
    wait_ramp = np.clip(ready_wait_time / 9.5, 0.0, 1.0)  # fixed scale for determinism; matches prior elite
    norm_wait = iqr_normalize(wait_ramp)
    
    # DDL-protection gate: high-confidence risk (tight slack AND high uncertainty)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Uncertainty-normalized amplification term — only active when DDL is truly at risk
    norm_uncertainty = iqr_normalize(uncertainty)
    
    # Final score: smaller = higher priority
    # Hierarchy: bounded urgency (hard-DDL safety) > bottleneck release > energy > fairness > risk amplification
    score = (
        norm_urgency  # Primary, bounded deadline signal
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Unblock critical paths
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff  # Penalize inefficient energy use
        - norm_wait  # Favor fresher tasks (anti-starvation)
        + ddl_protection_active * norm_uncertainty  # Risk-aware boost only when both signals align
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
