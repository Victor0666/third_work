def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with soft criticality attenuation and sharpened successor-risk modeling:
      - Replaces hard zero-out `critical_gate` with smooth attenuation: preserves gradient flow and avoids disabling successor mitigation.
      - Sharpens successor-slack-deficit penalty using power-law sensitivity to slack deficit magnitude and tunable uncertainty correlation bias.
      - Maintains all robustness features: adaptive normalization, tightened DDL gate, urgency tanh, fairness, energy efficiency.
      - No numeric literals beyond {-2,-1,0,1,2}; all tunables exposed via PARAMS.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Adaptive min-max normalization with fallback for near-zero dispersion
    def adaptive_normalize(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros(0, dtype=float)
        if N == 1:
            return np.zeros(1, dtype=float)
        
        x_min, x_max = np.min(x), np.max(x)
        range_val = x_max - x_min
        
        # Uncertainty-aware dispersion baseline
        unc_median = np.median(uncertainty) if N > 0 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_median + eps)
        
        # Use range if meaningful, else fallback to dispersion
        denom = np.where(range_val > eps, range_val, dispersion + eps)
        
        return (x - x_min) / (denom + eps)
    
    # Hard DDL violation penalty: raw negative slack (non-negative, finite)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Conditional DDL-protection gate: tightened logic — requires uncertainty > median * threshold
    unc_median = np.median(uncertainty) if N > 0 else eps
    ddl_risk_condition = (slack < 0) & (uncertainty > (unc_median * PARAMS["ddl_risk_activation_threshold"] + eps))
    ddl_risk_gate = np.where(ddl_risk_condition, 1.0, 0.0)
    
    # Criticality-aware soft attenuation: reduces (not zeros) bottleneck pressure for non-critical tasks under tight slack
    median_slack = np.median(slack) if N > 0 else 0.0
    tight_slack_mask = (slack < (median_slack * (1.0 - 0.05))) & (slack >= -eps)  # fixed small margin for stability
    max_upward = np.max(upward_rank) if N > 0 else 1.0
    non_critical_mask = (upward_rank < (max_upward * 0.25)) & tight_slack_mask  # fixed structural threshold for clarity
    # Soft attenuation: scale down bottleneck pressure by (1 - strength * mask), preserving gradient
    attenuation_factor = 1.0 - PARAMS["criticality_attenuation_strength"] * non_critical_mask.astype(float)
    
    # Urgency: scaled tanh of (median_slack - slack), bounded and smooth
    urgency_base = np.clip(median_slack - slack, 0.0, 2.0)
    urgency = np.tanh(PARAMS["urgency_tanh_scale"] * urgency_base)
    
    # Bottleneck pressure: multiplicative coupling, softly attenuated
    bottleneck_pressure = (
        remaining_work * upward_rank * PARAMS["upward_rank_remaining_work_coupling"]
    )
    bottleneck_pressure = bottleneck_pressure * attenuation_factor
    
    # Uncertainty-amplified bottleneck: power-law boost on normalized uncertainty
    norm_uncertainty = adaptive_normalize(uncertainty)
    bottleneck_pressure = bottleneck_pressure * np.power(
        1.0 + norm_uncertainty, 
        PARAMS["bottleneck_uncertainty_amplification"]
    )
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Fairness term: linear anti-starvation boost (unconditional)
    norm_wait = adaptive_normalize(ready_wait_time)
    fairness_term = -PARAMS["wait_fairness_weight"] * norm_wait
    
    # Sharpened successor-slack-deficit penalty: power-law response to slack deficit magnitude
    # Uses actual slack deficit (not binary proxy) scaled by uncertainty correlation with tunable bias
    slack_deficit = np.clip(median_slack - slack, 0.0, None)
    uncertainty_correlation = np.where(uncertainty > unc_median, 1.0, PARAMS["uncertainty_correlation_bias"])
    successor_slack_deficit_penalty = (
        PARAMS["successor_slack_deficit_weight"] *
        np.power(slack_deficit + eps, PARAMS["slack_deficit_sensitivity"]) *
        uncertainty_correlation
    )
    
    # Final score: smaller = higher priority
    score = (
        neg_slack +
        PARAMS["ddl_risk_gate_weight"] * ddl_risk_gate +
        urgency +
        adaptive_normalize(bottleneck_pressure) +
        norm_energy_eff +
        fairness_term +
        successor_slack_deficit_penalty
    )
    
    # Enforce finiteness and shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure shape (N,) even for N=1
    return score.astype(float, copy=False)
