def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with IQR-based robust normalization and refined urgency modeling.
    
    Key features:
      - Uses configurable quantile-based dispersion (Q_{center+upper} - Q_{center-lower}) for outlier-robust normalization.
      - Conditional DDL gate remains: activates only when slack < 0 AND uncertainty > median_uncertainty.
      - Bottleneck pressure multiplies upward_rank and remaining_work (validated coupling), enhanced by wait during deadline stress.
      - Uncertainty amplification via power-law on normalized uncertainty.
      - Urgency uses tanh scaled by tunable parameter — no hardcoded constants beyond {-2,-1,0,1,2}.
      - All numeric literals strictly in {-2,-1,0,1,2}; no try/except, no loops, no I/O, no randomness.
      - Fully deterministic, shape-compliant, and finite-valued.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust quantile-based adaptive normalization
    def adaptive_normalize(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros(0, dtype=float)
        if N == 1:
            return np.zeros(1, dtype=float)
        
        # Compute configurable quantiles for dispersion: Q_upper - Q_lower
        q_lower = np.clip(PARAMS["quantile_center"] - PARAMS["quantile_lower_offset"], 0.0, 1.0)
        q_upper = np.clip(PARAMS["quantile_center"] + PARAMS["quantile_upper_offset"], 0.0, 1.0)
        q_lo = np.quantile(x, q_lower, method='midpoint')
        q_hi = np.quantile(x, q_upper, method='midpoint')
        range_val = q_hi - q_lo
        
        # Uncertainty-aware dispersion baseline
        unc_median = np.median(uncertainty) if N > 0 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_median + eps)
        
        # Use quantile range if meaningful, else fallback to dispersion
        denom = np.where(range_val > eps, range_val, dispersion + eps)
        
        center = np.quantile(x, PARAMS["quantile_center"], method='midpoint')
        return (x - center) / (denom + eps)
    
    # Hard DDL violation penalty: raw negative slack (non-negative, finite)
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Conditional DDL-protection gate: active only when slack < 0 AND uncertainty > median_uncertainty
    unc_median = np.median(uncertainty) if N > 0 else eps
    ddl_risk_condition = (slack < 0) & (uncertainty > (unc_median + eps))
    ddl_risk_gate = np.where(ddl_risk_condition, 1.0, 0.0)
    
    # Urgency: scaled tanh of (median_slack - slack), bounded and smooth
    median_slack = np.quantile(slack, PARAMS["quantile_center"], method='midpoint') if N > 0 else 0.0
    urgency_base = np.clip(median_slack - slack, 0.0, 2.0)  # bounded to [0,2]
    urgency = np.tanh(PARAMS["urgency_tanh_scale"] * urgency_base)
    
    # Bottleneck pressure: multiplicative coupling (validated), enhanced by wait-time during deadline stress
    norm_wait = adaptive_normalize(ready_wait_time)
    bottleneck_pressure = (
        remaining_work * upward_rank * PARAMS["upward_rank_remaining_work_coupling"] *
        (1.0 + norm_wait * (slack < 0).astype(float))
    )
    
    # Uncertainty-amplified bottleneck: power-law boost on normalized uncertainty
    norm_uncertainty = adaptive_normalize(uncertainty)
    bottleneck_pressure = bottleneck_pressure * np.power(
        1.0 + norm_uncertainty, 
        PARAMS["bottleneck_uncertainty_amplification"]
    )
    
    # Energy efficiency: marginal energy per unit duration (J/s), normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Fairness term: linear anti-starvation boost (unconditional)
    fairness_term = -PARAMS["wait_fairness_weight"] * norm_wait
    
    # Final score: smaller = higher priority
    score = (
        neg_slack +
        PARAMS["ddl_risk_gate_weight"] * ddl_risk_gate +
        urgency +
        adaptive_normalize(bottleneck_pressure) +
        norm_energy_eff +
        fairness_term
    )
    
    # Enforce finiteness and shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure shape (N,) even for N=1
    return score.astype(float, copy=False)
