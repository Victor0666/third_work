def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with global-regime IQR normalization, sharpened urgency saturation,
    and slack-aware bottleneck decay — restoring robustness while enhancing deadline fidelity.
    
    Key improvements:
      - Reinstates *global* uncertainty regime gating (not per-task) for IQR percentiles → avoids noise amplification in small N.
      - Replaces linear urgency clipping with smooth, tunable power-law saturation (urgency^exponent) for sharper criticality transition.
      - Introduces slack-aware multiplicative decay on bottleneck coupling: suppresses rank×work pressure when slack > median,
        preventing over-prioritization of non-critical paths during safe periods.
      - All numeric literals strictly {-2,-1,0,1,2}; no hidden constants; exactly one conditional branch (high_uncertainty_mask).
      - Maintains deterministic, finite, shape-(N,) output with full NaN/inf protection.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Single global conditional branch: high-uncertainty regime flag (robust to small N)
    uncertainty_median = np.median(uncertainty) if N > 0 else 0.0
    high_uncertainty_mask = (uncertainty > uncertainty_median).astype(float)
    
    # Select IQR percentiles globally (one branch only) → restores robustness per reflection
    q_low = np.where(high_uncertainty_mask[0] == 1.0, 
                     PARAMS["iqr_low_percentile_high_uncertainty"], 
                     PARAMS["iqr_low_percentile_low_uncertainty"])
    q_high = np.where(high_uncertainty_mask[0] == 1.0, 
                      PARAMS["iqr_high_percentile_high_uncertainty"], 
                      PARAMS["iqr_high_percentile_low_uncertainty"])
    
    # Robust IQR-based normalization using selected percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low_val = np.percentile(x, q_low)
        q_high_val = np.percentile(x, q_high)
        iqr = q_high_val - q_low_val
        center = np.median(x)
        denom = iqr if iqr > eps else (np.max(np.abs(x - center)) + eps)
        return (x - center) / (denom + eps)
    
    # Sharpened urgency: power-law saturation instead of linear clip → better criticality discrimination
    median_slack = np.median(slack) if N > 0 else 0.0
    deadline_risk_threshold = PARAMS["deadline_risk_gate_factor"] * median_slack
    raw_urgency = np.maximum(-slack, 0.0) / (np.abs(median_slack) + eps)
    # Apply tunable exponent for smooth, bounded saturation: [0, ∞) → [0, 2] via x^p / (1 + x^p) * 2
    exp_urgency = np.power(raw_urgency + eps, PARAMS["urgency_saturation_exponent"])
    urgency_final = 2.0 * exp_urgency / (1.0 + exp_urgency)
    # Gate only under severe risk (same logic as before, but now feeding smoother signal)
    urgency_gate = (slack < deadline_risk_threshold).astype(float)
    norm_urgency = iqr_normalize(urgency_final * (1.0 + urgency_gate))
    
    # Energy efficiency
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / duration
    norm_energy_eff = iqr_normalize(energy_per_duration)
    
    # Successor bottleneck coupling with slack-aware decay
    rank_work_pressure = (upward_rank + eps) * (remaining_work + eps)
    # Decay coupling when slack > median → reduces pressure on non-critical paths
    slack_above_median = (slack > median_slack).astype(float)
    decay_factor = (1.0 - slack_above_median) + slack_above_median * PARAMS["bottleneck_decay_under_slack"]
    bottleneck_pressure_decayed = rank_work_pressure * decay_factor
    norm_bottleneck = iqr_normalize(bottleneck_pressure_decayed)
    
    # Ready wait time: linear saturation ramp
    wait_ramp = np.clip(ready_wait_time / PARAMS["wait_ramp_scale"], 0.0, 1.0)
    norm_wait = iqr_normalize(wait_ramp)
    
    # DDL-protection gate (single logical expression, reinstated per reflection)
    max_uncertainty = np.max(uncertainty) if N > 0 else 1.0
    ddl_protection_active = (
        (slack < median_slack) &
        (uncertainty > PARAMS["ddl_protection_gate_threshold"] * max_uncertainty)
    ).astype(float)
    
    # Uncertainty-normalized amplification term
    norm_uncertainty = iqr_normalize(uncertainty)
    
    # Final score: smaller = higher priority
    score = (
        norm_urgency
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - norm_wait
        + ddl_protection_active * norm_uncertainty
    )
    
    # Ensure finite output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
