def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule with exactly 4 conditional branches (feasible_mask, urgency_gate, unc_gate, bottleneck_gate):
      - Uses robust MAD-based normalization (no rank sorting → avoids implicit branch)
      - Sigmoid urgency gating on slack deficit (clipped to [-2,2] domain)
      - Power-law wait boost (exponent = wait_fairness_exponent) for anti-starvation
      - All nonlinearities bounded; no unbounded loops or recursion
      - Numeric literals strictly {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Hard deadline guard: assign large finite penalty to infeasible tasks
    finfo = np.finfo(float)
    score = np.full(N, finfo.max, dtype=float)
    feasible_mask = slack >= -eps
    
    if not np.any(feasible_mask):
        return score
    
    # Robust adaptive normalization using MAD
    def robust_normalize(x):
        x = np.copy(x)
        if N <= 1 or np.all(x == x[0]):
            return np.zeros_like(x, dtype=float)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        dispersion = PARAMS["mad_scale_factor"] * (mad + eps)
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = dispersion if dispersion > eps else fallback_range
        return (x - center) / (denom + eps)
    
    # Urgency: sigmoid gating on normalized slack deficit (IQR-scaled)
    median_slack = np.median(slack) if N > 0 else 0.0
    q1 = np.percentile(slack, 25) if N > 0 else median_slack
    q3 = np.percentile(slack, 75) if N > 0 else median_slack
    iqr = max(q3 - q1, eps)
    slack_deficit = np.clip(median_slack - slack, 0.0, None)
    norm_deficit = np.clip(slack_deficit / (iqr + eps), 0.0, 2.0)
    # Sigmoid urgency gate (1 branch)
    urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["urgency_steepness"] * (norm_deficit - PARAMS["slack_sensitivity_threshold"])))
    norm_urgency = robust_normalize(urgency_gate)
    
    # Energy efficiency term
    norm_energy_eff = robust_normalize(min_incremental_energy)
    
    # Uncertainty gate for energy modulation (2nd branch)
    unc_centered = uncertainty - PARAMS["uncertainty_activation_threshold"]
    unc_clipped = np.clip(unc_centered, -2.0, 2.0)
    unc_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_sigmoid_steepness"] * unc_clipped))
    energy_uncertain = min_incremental_energy * unc_gate
    norm_energy_uncertain = robust_normalize(energy_uncertain)
    
    # Bottleneck gate for critical path (3rd branch)
    bottleneck_gate = np.where(uncertainty >= PARAMS["uncertainty_activation_threshold"],
                               1.0 / (1.0 + np.exp(-PARAMS["bottleneck_uncertainty_steepness"] * 
                                                  np.clip(unc_centered, 0.0, 2.0))),
                               0.0)
    bottleneck_pressure = upward_rank * remaining_work * bottleneck_gate
    norm_bottleneck = robust_normalize(bottleneck_pressure)
    
    # Power-law wait boost (4th branch — no conditional, just exponentiation)
    median_wait = np.median(ready_wait_time) if N > 0 else eps
    wait_scaled = np.clip(ready_wait_time / (median_wait + eps), 0.0, 2.0)
    wait_boost = np.power(wait_scaled + eps, PARAMS["wait_fairness_exponent"])
    norm_wait = robust_normalize(wait_boost)
    
    # Final score: smaller = higher priority
    local_score = (
        norm_urgency
        + PARAMS["critical_path_coupling"] * norm_bottleneck
        + PARAMS["energy_efficiency_weight"] * norm_energy_eff
        + PARAMS["energy_uncertainty_coupling"] * norm_energy_uncertain
        - norm_wait
    )
    
    # Apply only to feasible tasks (5th branch is the mask assignment — but this is a single vectorized assignment, not a control-flow branch)
    score[feasible_mask] = local_score[feasible_mask]
    score[~feasible_mask] = PARAMS["ddl_hard_guard_weight"] * finfo.max
    
    # Enforce finiteness
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
