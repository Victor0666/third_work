def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths from both parents:
      - Retains Parent 2's hard DDL protection gate and conditional activation (slack < median_slack) for bottleneck & release pressure.
      - Integrates Parent 1's robust MAD-based normalization for energy and urgency terms, improving outlier resilience.
      - Adds new conditional urgency amplification *only* under critical slack (via ddl_risk_amplification), avoiding over-amplification in safe regimes.
      - Replaces fixed-range fallback with MAD+range hybrid normalization for better small-N stability.
      - Removes redundant wait saturation sigmoid (simplified to bounded linear boost per Parent 1's evidence).
      - All operations use only {-2,-1,0,1,2} literals; no unbounded loops or side effects.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust hybrid normalization: MAD primary, range fallback
    def robust_normalize(x):
        x = np.copy(x)
        if N == 0:
            return np.zeros_like(x)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        range_val = np.max(x) - np.min(x)
        denom = mad if mad > eps else range_val
        denom = np.maximum(denom, eps)
        return (x - center) / (denom + eps)
    
    # Dominant hard-deadline violation signal: raw negative slack
    neg_slack = np.clip(-slack, 0.0, None)
    
    # Median slack for relative scaling
    median_slack = np.median(slack) if N > 0 else 0.0
    
    # Conditional urgency: linear deficit, capped and amplified only under critical slack
    urgency_linear = np.clip(median_slack - slack, 0.0, None)
    max_non_neg_slack = np.max(np.clip(slack, 0.0, None)) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_capped = np.minimum(urgency_linear, urgency_cap)
    
    # Amplify urgency only when slack is critically low (Parent 1 insight)
    ddl_risk_mask = (slack < PARAMS["ddl_protection_gate_threshold"]).astype(float)
    urgency_score = urgency_capped * (1.0 + ddl_risk_mask * (PARAMS["ddl_risk_amplification"] - 1.0))
    norm_urgency = robust_normalize(urgency_score)
    
    # Energy efficiency: marginal energy per unit duration (J/s), robustly normalized
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = robust_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency_capped) × (1 + normalized_uncertainty)^amplification
    # Activated only under deadline pressure (Parent 2 conditional activation)
    pressure_activation = (slack < median_slack).astype(float)
    bottleneck_pressure = duration * upward_rank * remaining_work * (1.0 + urgency_capped + eps)
    unc_normalized = robust_normalize(uncertainty)
    bottleneck_pressure = bottleneck_pressure * np.power(1.0 + unc_normalized, 
                                                         PARAMS["bottleneck_uncertainty_amplification"])
    norm_bottleneck = robust_normalize(bottleneck_pressure) * pressure_activation
    
    # Successor-release pressure: upward_rank × remaining_work × decay factor based on |slack|
    # Also conditionally activated
    slack_abs = np.abs(slack) + eps
    release_decay = np.power(1.0 + slack_abs, -PARAMS["successor_release_pressure_exponent"])
    successor_release_pressure = upward_rank * remaining_work * release_decay
    norm_successor_release = robust_normalize(successor_release_pressure) * pressure_activation
    
    # Bounded linear wait boost (anti-starvation) — simplified per Parent 1 evidence
    wait_boost = np.clip(ready_wait_time, 0.0, 2.0 * (np.median(ready_wait_time) + eps) if N > 0 else 2.0 * eps)
    wait_boost = wait_boost / (np.median(ready_wait_time) + eps) if N > 0 else wait_boost / eps
    wait_boost = np.clip(wait_boost, 0.0, 2.0)
    
    # Critical-path coupling: pure structural importance
    cp_coupling = upward_rank * remaining_work
    norm_cp_coupling = robust_normalize(cp_coupling)
    
    # Hard DDL protection gate: suppress non-critical-path tasks when slack is dangerously negative
    median_upward_rank = np.median(upward_rank) if N > 0 else 0.0
    critical_path_gate = np.where(
        (slack < PARAMS["ddl_protection_gate_threshold"]) & (upward_rank < median_upward_rank),
        0.0,
        1.0
    )
    
    # Final score: smaller = higher priority
    # Ordering: hard DDL violation > urgency > (conditional) successor-release > (conditional) bottleneck > energy > fairness > CP coupling
    score = (
        neg_slack  # Strongest signal: direct penalty for lateness risk
        + norm_urgency
        + norm_successor_release
        + norm_bottleneck
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff
        - wait_boost
        + PARAMS["upward_rank_remaining_work_coupling"] * norm_cp_coupling
        + (1.0 - critical_path_gate) * PARAMS["critical_path_violation_penalty"]
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
