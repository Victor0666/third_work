def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating evidence-backed structural changes:
      - Replaces linear neg_slack with robust power-law DDL risk amplification (ddl_risk_amplification)
      - Introduces conditional critical-path gate: activates upward_rank × remaining_work only when slack is tight (<= median_slack) AND uncertainty is high
      - Uses bounded power-law wait decay (ready_wait_time^p) instead of sigmoid saturation for anti-starvation — simpler, monotonic, and numerically stable
      - Removes all multiplicative joint-risk couplings; replaces with additive, conditionally gated terms
      - Normalizes each feature independently using adaptive dispersion scaled by uncertainty std, then applies learned scaling
      - Enforces strict finiteness and avoids NaN/inf via np.finfo safeguards and clipping
      - All parameters exposed in PARAMETER_SCHEMA; no hidden constants beyond {-2,-1,0,1,2}
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Adaptive dispersion base: use uncertainty std to scale normalization sensitivity
    unc_std = np.std(uncertainty) if N > 1 else eps
    
    def adaptive_normalize(x, scale_factor=1.0):
        x = np.copy(x)
        center = np.median(x) if N > 0 else 0.0
        # Dispersion: uncertainty-aware + fallback to range
        dispersion = scale_factor * (unc_std + eps)
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = np.where(dispersion > eps, dispersion, fallback_range)
        return (x - center) / (denom + eps)
    
    # === DDL Risk Term ===
    # Power-law amplification of negative slack: strong penalty for violation, smooth near zero
    neg_slack = np.clip(-slack, 0.0, None)
    ddl_risk = np.power(neg_slack + eps, PARAMS["ddl_risk_amplification"])
    
    # === Critical-Path Pressure (Conditional Gate) ===
    # Activate only when slack is tight AND uncertainty is above threshold
    median_slack = np.median(slack) if N > 0 else 0.0
    slack_tight = (slack <= median_slack)
    unc_high = (uncertainty > np.quantile(uncertainty, 0.75) if N > 1 else 0.5)
    critical_gate = np.where(slack_tight & unc_high, 1.0, 0.0)
    
    # Bottleneck pressure: upward_rank × remaining_work, normalized and scaled
    raw_bottleneck = upward_rank * remaining_work
    norm_bottleneck = adaptive_normalize(raw_bottleneck)
    bottleneck_pressure = critical_gate * PARAMS["uncertainty_coupling_strength"] * norm_bottleneck
    
    # === Energy Efficiency Term ===
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy = adaptive_normalize(energy_per_duration, PARAMS["energy_normalization_scale"])
    
    # === Anti-Starvation Term ===
    # Power-law wait decay: longer wait → higher priority, but diminishing returns
    wait_ratio = np.where(N > 0, ready_wait_time / (np.maximum(np.median(ready_wait_time), eps) + eps), 0.0)
    wait_ratio = np.clip(wait_ratio, 0.0, 2.0)  # prevent overflow in pow
    wait_priority = np.power(wait_ratio + eps, PARAMS["wait_decay_exponent"])
    # Normalize to avoid dominance; invert so larger wait → smaller score (higher priority)
    norm_wait = -adaptive_normalize(wait_priority)
    
    # === Slack-Based Urgency Term (Fallback for non-violating tasks) ===
    # Linear urgency for positive slack: median_slack - slack, capped at median_slack
    urgency_linear = np.clip(median_slack - slack, 0.0, median_slack if N > 0 else eps)
    norm_urgency = adaptive_normalize(urgency_linear)
    
    # === Final Score: smaller = higher priority ===
    # Dominance order: DDL risk > critical path > urgency > energy > fairness
    score = (
        ddl_risk
        + bottleneck_pressure
        + norm_urgency
        + norm_energy
        + norm_wait
    )
    
    # Enforce finiteness and finite numeric output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
