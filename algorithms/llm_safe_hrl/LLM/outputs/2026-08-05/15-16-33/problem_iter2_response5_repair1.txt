def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining deadline safety, critical-path awareness, 
    starvation prevention, and work-normalized energy efficiency.
    
    Key features:
      - Piecewise slack handling: strong convex penalty for negative slack, gentle linear reward for positive
      - Work-normalized energy: min_incremental_energy / (remaining_work + epsilon) → prioritizes J/MI efficiency
      - Percentile-gated critical-path bonus to avoid outlier dominance
      - Exponential wait decay with saturation guard
      - Uncertainty-slack coupling only under tight slack
      - All normalization uses tunable IQR percentiles
      - No numeric literals beyond {-2,-1,0,1,2}; all thresholds/weights parameterized
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    # Robust IQR-based normalization using declared percentiles
    def iqr_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_percentile_low"])
        q_high = np.percentile(x, PARAMS["iqr_percentile_high"])
        iqr = q_high - q_low
        center = np.median(x)
        denom = iqr if iqr > eps_safe else (np.max(np.abs(x - center)) + eps_safe)
        return (x - center) / (denom + eps_safe)

    # Slack: piecewise — strong convex penalty for negative slack, gentle linear reward for positive
    slack_norm = iqr_normalize(slack)
    slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_risk_penalty"] * (-slack_norm),
        -PARAMS["slack_urgency_gain"] * slack_norm
    )

    # Duration: exec + comm, used for normalization and coupling
    duration = np.maximum(min_exec_time + min_comm_time, eps_safe)

    # Work-normalized energy efficiency: J per MI → lower is better → subtract in score
    work_efficient_energy = min_incremental_energy / np.maximum(remaining_work, eps_safe)
    energy_norm = iqr_normalize(work_efficient_energy)

    # Critical-path bonus: gated by upward_rank percentile (not raw value)
    if N == 1:
        rank_percentile = np.array([1.0])
    else:
        sorted_ranks = np.sort(upward_rank)
        rank_idx = np.searchsorted(sorted_ranks, upward_rank, side='right')
        rank_percentile = rank_idx / (N + eps_safe)
    critical_gate = np.where(rank_percentile >= PARAMS["critical_rank_threshold"], 1.0, 0.0)
    rank_norm = iqr_normalize(upward_rank)
    critical_bonus = PARAMS["critical_path_bonus"] * rank_norm * critical_gate

    # Waiting-time boost: exponential decay with saturation guard
    wait_boost = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    wait_norm = iqr_normalize(wait_boost)

    # Uncertainty-slack coupling: only amplify urgency when both slack is tight AND uncertainty is high
    unc_norm = iqr_normalize(uncertainty)
    slack_pressure = np.maximum(-slack_norm, 0.0)  # pressure from negative slack only
    coupled_urgency = PARAMS["uncertainty_slack_coupling"] * unc_norm * slack_pressure

    # Duration-efficiency ratio: balance duration cost vs energy cost
    dur_eff_ratio = PARAMS["duration_efficiency_ratio"] * iqr_normalize(duration)

    # Final score: smaller = higher priority
    # Deadline safety dominates; then energy/work efficiency; then fairness; then urgency coupling
    score = (
        slack_penalty
        + PARAMS["energy_efficiency_weight"] * energy_norm
        - critical_bonus
        - wait_norm
        + coupled_urgency
        + dur_eff_ratio
    )

    # Ensure finite output, no NaN/inf, correct shape
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
