def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's uncertainty-aware normalization and feasibility-preserving urgency cap with Parent 1's explicit negative-slack penalty and joint risk coupling.
    
    Key structural improvements:
      - Retains uncertainty-weighted dispersion scaling (Parent 2) for risk-contextual normalization.
      - Keeps feasibility-preserving urgency cap: linear urgency capped by max_non_neg_slack^exponent to avoid inversion under negative slack.
      - Adds explicit clipped negative slack penalty (Parent 1) *before* normalization to preserve hard-deadline signal integrity.
      - Introduces joint_risk_coupling: normalized negative slack × normalized uncertainty → direct amplification of priority when both deadline risk and execution uncertainty are high.
      - Removes redundant wait_saturation sigmoid; replaces with linear anti-starvation via normalized and inverted ready_wait_time (simpler, more robust, fewer parameters).
      - Bottleneck pressure now includes upward_rank × remaining_work × (1 + urgency_linear) × (1 + uncertainty)^amplification — unifying critical-path load and risk-aware release pressure.
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants; zero branching; deterministic.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Uncertainty-aware adaptive normalization: dispersion scaled by std(uncertainty) to reflect risk context
    def adaptive_normalize(x):
        x = np.copy(x)
        center = np.median(x)
        
        # Base dispersion: use std(uncertainty) as risk-aligned scaling factor
        unc_std = np.std(uncertainty) if N > 1 else eps
        dispersion = PARAMS["uncertainty_dispersion_scale"] * (unc_std + eps)
        
        # Fallback to range if dispersion too small
        fallback_range = np.max(x) - np.min(x) if N > 0 else eps
        denom = dispersion if dispersion > eps else fallback_range
        
        return (x - center) / (denom + eps)
    
    # Explicit negative-slack penalty (Parent 1): raw clipped penalty before normalization for strong DDL enforcement
    neg_slack = np.clip(-slack, 0.0, None)
    norm_neg_slack = adaptive_normalize(neg_slack)
    
    # Linear urgency proxy: (median_slack - slack), then feasibility-preserving cap (Parent 2)
    median_slack = np.median(slack) if N > 0 else 0.0
    urgency_linear = np.clip(median_slack - slack, 0.0, 2.0)
    
    # Feasibility-preserving urgency cap: clamp to max non-negative slack raised to learned exponent
    non_neg_slack = np.clip(slack, 0.0, None)
    max_non_neg_slack = np.max(non_neg_slack) if N > 0 else eps
    urgency_cap = np.power(max_non_neg_slack + eps, PARAMS["urgency_cap_exponent"])
    urgency_linear = np.minimum(urgency_linear, urgency_cap)
    
    norm_urgency = adaptive_normalize(urgency_linear)
    
    # Energy efficiency: marginal energy per unit duration (J/s)
    duration = np.maximum(min_exec_time + min_comm_time, eps)
    energy_per_duration = min_incremental_energy / (duration + eps)
    norm_energy_eff = adaptive_normalize(energy_per_duration)
    
    # Bottleneck pressure: duration × upward_rank × remaining_work × (1 + urgency_linear) × (1 + uncertainty)^amplification
    bottleneck_pressure = duration * upward_rank * remaining_work * (1.0 + urgency_linear + eps)
    
    # Uncertainty amplification: power-law boost using normalized uncertainty
    unc_normalized = adaptive_normalize(uncertainty)
    bottleneck_pressure = bottleneck_pressure * np.power(1.0 + unc_normalized, 
                                                         PARAMS["bottleneck_uncertainty_amplification"])
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Joint risk coupling: normalized negative slack × normalized uncertainty → direct amplification under dual risk
    norm_uncertainty = adaptive_normalize(uncertainty)
    joint_risk = norm_neg_slack * norm_uncertainty
    norm_joint_risk = adaptive_normalize(joint_risk)
    
    # Anti-starvation: linear scaling of ready_wait_time, normalized and inverted (longer wait → higher priority)
    norm_wait = adaptive_normalize(ready_wait_time)
    inv_wait = 1.0 - norm_wait  # invert so longer wait increases priority
    
    # Final score: smaller = higher priority
    # Hierarchy: explicit DDL violation > joint risk > urgency > bottleneck > energy > fairness
    score = (
        PARAMS["slack_risk_penalty_weight"] * norm_neg_slack  # Primary hard-deadline violation signal (Parent 1)
        + PARAMS["joint_risk_coupling"] * norm_joint_risk     # Amplify when both slack deficit and uncertainty are high (Parent 1 + Parent 2 synthesis)
        + norm_urgency                                          # Feasibility-capped urgency (Parent 2)
        + PARAMS["successor_bottleneck_coupling"] * norm_bottleneck  # Critical-path release pressure (Parent 2)
        + PARAMS["energy_duration_ratio_weight"] * norm_energy_eff     # Risk-adjusted energy efficiency (Parent 2)
        + inv_wait                                                  # Anti-starvation (simplified from Parent 1)
    )
    
    # Enforce finiteness
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
