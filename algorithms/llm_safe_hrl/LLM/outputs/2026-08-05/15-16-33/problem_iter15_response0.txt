import numpy as np

PARAMETER_SCHEMA = {
    "schema_version": 1,
    "parameters": [
        {
            "name": "epsilon",
            "initial_value": 0.000127,
            "lower_bound": 1e-6,
            "upper_bound": 0.1,

            "parameter_type": "float",
            "transform": "log"
        },

            "initial_value": 0.62,

            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "successor_bottleneck_coupling",
            "initial_value": 0.31,
            "lower_bound": 0.0,
            "upper_bound": 1.5,

            "parameter_type": "float",

            "parameter_type": "float",

            "parameter_type": "float",

        {
            "name": "urgency_scale",
            "initial_value": 1.15,
            "lower_bound": 0.1,

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
    - Restored bounded tanh urgency (superior monotonicity & stability vs piecewise-linear)
    - Anti-starvation reinstated as subtracted fairness term (not embedded), ensuring starvation prevention even in feasible regimes
    - Joint DDL-risk gate simplified to `(slack < 0) & (uncertainty > median_uncertainty * threshold)` — robust, interpretable, no quantile estimation noise
    - Bottleneck now cleanly separates deadline-driven urgency (tanh) from critical-path + energy pressure, avoiding coupling artifacts
    - Fairness term uses adaptive-normalized wait time scaled by `wait_fairness_weight`, applied unconditionally (not gated) for consistent fairness
    - All numeric literals are in {-2,-1,0,1,2}; no hidden constants; strict adherence to interface contract.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure float arrays without mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust adaptive normalization using IQR only (no fallback — tuned percentiles ensure stability)
    def adaptive_normalize(x):
        x = np.copy(x)
        q_low = np.percentile(x, PARAMS["iqr_low_percentile"])
        q_high = np.percentile(x, PARAMS["iqr_high_percentile"])
        iqr = q_high - q_low + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Bounded tanh urgency: maps slack → [0,1], negative slack → high urgency, stable near zero
    # Scale avoids overflow: use median absolute slack or eps if degenerate
    abs_slack = np.abs(slack)
    scale = np.median(abs_slack) if np.any(abs_slack > eps) else 1.0
    tanh_urgency = np.tanh(-slack / (scale * PARAMS["urgency_scale"] + eps))
    urgency = np.clip((tanh_urgency + 1.0) / 2.0, 0.0, 1.0)
    norm_urgency = adaptive_normalize(urgency)
    
    # Duration includes both compute and communication
    duration = min_exec_time + min_comm_time
    
    # Unified bottleneck pressure: critical-path + risk-adjusted energy, decoupled from urgency
    # (duration * upward_rank) captures structural importance
    # (1 + uncertainty) weights volatility exposure
    # (min_incremental_energy + eps) embeds marginal energy cost
    bottleneck_pressure = (
        duration *
        upward_rank *
        (1.0 + uncertainty) *
        (min_incremental_energy + eps)
    )
    norm_bottleneck = adaptive_normalize(bottleneck_pressure)
    
    # Anti-starvation: normalized ready_wait_time, subtracted (higher wait → lower score → higher priority)
    norm_wait = adaptive_normalize(ready_wait_time)
    fairness_term = -PARAMS["wait_fairness_weight"] * norm_wait  # subtractive fairness
    
    # Joint DDL-risk gate: activated only when slack < 0 AND uncertainty exceeds scaled median
    median_uncertainty = np.median(uncertainty) if N > 0 else eps
    ddl_risk_gate = (
        (slack < 0.0) &
        (uncertainty > PARAMS["ddl_risk_activation_threshold"] * median_uncertainty)
    ).astype(float)
    # Amplify urgency signal under verified stress — not bottleneck, preserving separation of concerns
    stress_amplified_urgency = norm_urgency * (1.0 + ddl_risk_gate)
    
    # Final score: smaller = higher priority
    # Priority order: urgency (primary deadline signal, stress-amplified) → bottleneck (critical path + energy) → fairness (anti-starvation)
    score = (
        stress_amplified_urgency +
        PARAMS["successor_bottleneck_coupling"] * norm_bottleneck +
        fairness_term
    )
    
    # Ensure finite output, no NaN/inf
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score.astype(float, copy=False)
