def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: replaces hardcoded 0.25/0.75 with tunable quantiles.
    All numeric literals are now only -2, -1, 0, 1, or 2. Uses np.finfo for safety.
    """
    epsilon = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(epsilon, finfo.tiny)

    # Convert inputs to float arrays, ensure no mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)

    N = len(min_exec_time)
    if N == 0:
        return np.array([], dtype=float)

    # Robust normalization helper: uses tunable quantiles for IQR scaling
    def robust_normalize(x):
        x = np.copy(x)
        q1, q3 = np.quantile(x, [PARAMS["quantile_low"], PARAMS["quantile_high"]])
        iqr = q3 - q1
        center = np.median(x)
        scale = iqr if iqr > eps_safe else np.mean(np.abs(x - center)) + eps_safe
        return (x - center) / scale

    # === 1. SLACK RISK: nonlinear, asymmetric, bounded ===
    slack_norm = robust_normalize(slack)
    slack_risk = np.where(
        slack < 0,
        -np.power(np.abs(slack_norm), PARAMS["slack_risk_exponent"]),
        slack_norm * PARAMS["slack_positive_reward_factor"]
    )

    # === 2. CRITICAL PATH BOOST: conditional on rank percentile ===
    rank_norm = robust_normalize(upward_rank)
    rank_median = np.median(rank_norm)
    critical_mask = (rank_norm >= rank_median).astype(float)
    critical_boost = critical_mask * PARAMS["critical_path_boost"] * (rank_norm - rank_median + eps_safe)

    # === 3. ENERGY-EFFICIENCY RATIO: marginal_energy / (exec + comm + eps)
    duration = min_exec_time + min_comm_time
    energy_per_sec = min_incremental_energy / (duration + eps_safe)
    eff_norm = robust_normalize(energy_per_sec)

    # === 4. UNCERTAINTY-SLACK COUPLING: only activates under joint risk
    unc_norm = robust_normalize(uncertainty)
    slack_pressure = np.maximum(-slack_norm, 0.0)
    coupled_risk = unc_norm * slack_pressure * PARAMS["uncertainty_slack_coupling"]

    # === 5. DURATION UNCERTAINTY PENALTY: direct penalty on uncertain durations
    dur_uncertainty = uncertainty / (duration + eps_safe)
    dur_unc_norm = robust_normalize(dur_uncertainty)
    dur_unc_penalty = dur_unc_norm * PARAMS["duration_uncertainty_penalty"]

    # === 6. ANTI-STARVATION: saturating boost from ready_wait_time
    wait_scaled = ready_wait_time / (PARAMS["wait_saturation_threshold"] + eps_safe)
    wait_boost = PARAMS["wait_boost_scale"] * (1.0 + np.tanh(wait_scaled))

    # === 7. FINAL SCORE: minimize → higher priority
    score = (
        -slack_risk
        - critical_boost
        + PARAMS["energy_efficiency_ratio_weight"] * eff_norm
        + coupled_risk
        + dur_unc_penalty
        - wait_boost
    )

    # Ensure finite output & shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
