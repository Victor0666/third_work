def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing risk-aware urgency, critical-path coupling, and bounded anti-starvation.
    
    Key innovations:
    - Slack is transformed via smooth, bounded risk penalty: negative slack triggers quadratic penalty;
      zero/near-zero slack receives mild penalty; positive slack yields diminishing returns.
    - Upward rank is gated: only activated when slack pressure exceeds threshold, avoiding over-prioritization
      of critical tasks when deadlines are relaxed.
    - Energy term is dynamically down-weighted under high deadline pressure (via slack-based gating).
    - Uncertainty interacts multiplicatively with both slack pressure and duration, not additively.
    - Ready wait time uses saturating ramp (not linear) to prevent runaway priority inflation.
    - All features normalized robustly using median absolute deviation (MAD) for outlier resilience.
    """
    epsilon = PARAMS["epsilon"]
    finfo = np.finfo(float)
    
    # Robust normalization: use MAD instead of mean/std to resist outliers
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        dev = np.median(np.abs(x - center))
        scale = np.maximum(dev, epsilon)
        return (x - center) / scale
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_uncert = robust_normalize(uncertainty)
    norm_wait = robust_normalize(ready_wait_time)
    
    # --- Risk-aware slack transformation: smooth, bounded, asymmetric ---
    # Negative slack → strong penalty (quadratic), positive slack → soft reward (saturating)
    slack_penalty = np.where(
        norm_slack < 0,
        -np.power(-norm_slack, PARAMS["slack_risk_exponent"]),
        np.minimum(norm_slack * 0.5, 1.0)  # cap positive benefit
    )
    
    # --- Criticality gating: only boost rank when deadline pressure is high ---
    slack_pressure_indicator = np.clip(-norm_slack, 0.0, np.inf)
    rank_gate = np.where(
        slack_pressure_indicator > PARAMS["rank_slack_gate"],
        PARAMS["criticality_boost"],
        1.0
    )
    boosted_rank_score = -rank_gate * norm_rank  # negative because higher rank = higher priority
    
    # --- Dynamic energy weighting: reduce energy focus when slack is critically low ---
    energy_weight = PARAMS["energy_efficiency_scale"] * np.clip(
        1.0 - 0.5 * slack_pressure_indicator, 0.2, 1.0
    )
    
    # --- Uncertainty-duration coupling: penalize high-uncertainty tasks *only* when duration is large AND slack is tight ---
    duration_uncert_interaction = (
        norm_duration * norm_uncert * 
        np.clip(slack_pressure_indicator, 0.0, 2.0) * 
        PARAMS["duration_uncertainty_coupling"]
    )
    
    # --- Saturating anti-starvation term for ready_wait_time ---
    # Cap effective wait impact beyond threshold (in normalized space)
    wait_effect = np.tanh(norm_wait / PARAMS["wait_decay_threshold"])
    
    # --- Uncertainty penalty: power-law penalty on normalized uncertainty ---
    uncert_penalty = np.power(np.abs(norm_uncert), PARAMS["uncertainty_penalty_power"])
    
    # Final score: smaller = better
    score = (
        slack_penalty  # dominant risk term: negative slack → large positive penalty → low priority (bad)
        + energy_weight * norm_energy
        + boosted_rank_score
        + 0.3 * robust_normalize(remaining_work)  # mild work-awareness (larger work → lower priority if not critical)
        - 0.8 * wait_effect  # anti-starvation: larger wait → more negative → higher priority
        + duration_uncert_interaction
        + 0.5 * uncert_penalty
    )
    
    # Ensure finite output and shape compliance
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score.astype(float, copy=False)
