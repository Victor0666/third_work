def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: adds DDL-protection override and successor-criticality boost; all literals declared."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Slack-aware pre-normalization: preserve urgency signal integrity under tight deadlines ===
    slack_abs_max = np.maximum(np.abs(np.min(slack)), np.abs(np.max(slack))) + eps
    slack_clamped = np.clip(slack, -PARAMS["ddl_protection_threshold"] * slack_abs_max, PARAMS["ddl_protection_threshold"] * slack_abs_max)
    norm_slack = slack_clamped / (slack_abs_max + eps)
    
    # === Robust normalization for other features (MAD-based, safe for N=1) ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.zeros(1, dtype=float) if np.abs(x[0]) < eps else np.sign(x[0]) * np.log1p(np.abs(x[0]))
        x_abs = np.abs(x)
        center = np.median(x_abs)
        spread = np.median(np.abs(x_abs - center)) + eps
        return (x - center) / spread
    
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack penalty: strong nonlinearity for negative slack only ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    penalty_abs_max = np.maximum(np.abs(np.min(raw_slack_penalty)), np.abs(np.max(raw_slack_penalty))) + eps
    norm_slack_penalty = np.clip(raw_slack_penalty, 0.0, PARAMS["ddl_protection_threshold"] * penalty_abs_max) / (penalty_abs_max + eps)
    
    # === Criticality acceleration gate: sharp sigmoid activation based on slack pressure ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Successor-release interaction: amplify priority using local criticality proxy ===
    # Approximate successor pressure as weighted sum of current rank and urgency (no graph access)
    successor_boost = (
        PARAMS["remaining_work_weight"] *
        (PARAMS["successor_rank_weight"] * np.maximum(norm_rank, 0.0) +
         PARAMS["successor_slack_weight"] * np.maximum(-norm_slack, 0.0))
    )
    
    # === Uncertainty-duration interaction: activate only under high uncertainty AND low slack ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = norm_duration * uncert_gate * slack_pressure
    
    # === Starvation relief: saturating anti-starvation term with numerical safeguard ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Novel risk-coupled energy penalty: penalize high-energy tasks more when uncertainty is high ===
    energy_uncertainty_penalty = norm_energy * norm_uncert * PARAMS["energy_uncertainty_coupling"]
    
    # === DDL-protection gate: hard override when slack is critically low ===
    ddl_protection_mask = np.where(norm_slack < -PARAMS["ddl_protection_threshold"], 1.0, 0.0)
    protected_score = (
        + PARAMS["ddl_protection_slack_penalty_weight"] * norm_slack_penalty
        - PARAMS["ddl_protection_rank_boost_weight"] * boosted_rank
        - successor_boost
        + PARAMS["ddl_protection_wait_benefit_weight"] * wait_benefit
    )
    normal_score = (
        + norm_slack_penalty
        - boosted_rank
        - PARAMS["energy_sensitivity"] * norm_energy
        - norm_duration
        - wait_benefit
        + PARAMS["duration_robustness"] * duration_risk_score
        + PARAMS["remaining_work_weight"] * norm_work
        + energy_uncertainty_penalty
    )
    score = np.where(ddl_protection_mask == 1.0, protected_score, normal_score)
    
    # Final safeguard: ensure finite output, deterministic shape
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — critical for N=1 case
    return score.reshape(-1)
