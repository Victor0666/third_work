def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: combines Parent 2's robust quantile normalization and uncertainty gating with Parent 1's clean congestion logic;
       introduces wait_benefit_cap to prevent starvation overcompensation;
       replaces raw slack pressure with sign-preserving MAD-normalized slack for deadline-boundary sensitivity;
       uses *multiplicative* criticality amplification: (1 + criticality_scale * |norm_slack|) * norm_rank * ddl_breach for sharper urgency coupling;
       enforces strict [-2,2] clipping per term to guarantee bounded AST depth and stability;
       all numeric literals are -2,-1,0,1,2 or np.finfo safeguards; no unbounded operations."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust quantile-based normalization: (x - Q1) / (Q3 - Q1 + eps), preserves sign and resists outliers ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])
            q3 = np.quantile(x, PARAMS["quantile_q3"])
        spread = q3 - q1 + eps
        return (x - q1) / spread
    
    # Normalize features except slack (uses sign-preserving MAD for deadline-boundary sensitivity)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_duration = robust_range_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # Sign-preserving MAD normalization for slack: sensitive to small negative values near deadline boundary
    if N == 1:
        slack_med = slack[0]
        slack_mad = eps
    else:
        slack_med = np.median(slack)
        slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = slack_mad if slack_mad > eps else eps
    norm_slack = (slack - slack_med) / slack_spread
    
    # === DDL-protection gate: enables energy/uncertainty terms only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Deadline breach indicator: activates critical-path and risk penalties ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Critical path leverage: multiplicative urgency amplification — sharpens priority under pressure ===
    # Uses absolute normalized slack to scale rank boost proportionally to violation severity
    critical_path_leverage = norm_rank * (1.0 + PARAMS["criticality_scale"] * np.abs(norm_slack)) * ddl_breach
    
    # === Raw slack penalty: power-law on negative slack, then robust-normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_range_normalize(raw_slack_penalty)
    
    # === Unified congestion signal: ready_wait_time + uncertainty, gated by DDL risk (1 - ddl_gate) ===
    congestion_signal = norm_wait + norm_uncert
    congestion_score = congestion_signal * (1.0 - ddl_gate)
    
    # === Starvation relief: linear-in-wait, capped to prevent overcompensation ===
    wait_mean = np.mean(ready_wait_time)
    wait_denom = np.maximum(wait_mean, eps)
    wait_benefit = np.clip(ready_wait_time / wait_denom, 0.0, PARAMS["wait_benefit_cap"])
    
    # === Energy-uncertainty coupling: active only under both deadline breach AND high uncertainty ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_breach
    
    # === Composite score: smaller = higher priority; each term clipped to [-2,2] for stability and bounded AST depth ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Sharp critical-path boost under breach
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy minimization only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Capped anti-starvation
        + PARAMS["congestion_weight"] * np.clip(congestion_score, -2.0, 2.0)  # Load-aware congestion relief under DDL risk
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * ddl_gate, -2.0, 2.0)  # Downstream work preference only when feasible
        + np.clip(norm_rank * (1.0 - ddl_gate), -2.0, 2.0)                # Mild rank boost when slack > 0
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
