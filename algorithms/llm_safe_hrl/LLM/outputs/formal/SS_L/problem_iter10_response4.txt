def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces linear starve-relief with bounded waiting bonus;
       adds successor-release-aware term via remaining_work * upward_rank interaction;
       replaces global robust_normalize with median-MAD per-feature; uses conditional DDL protection gate.
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — preserves sign, outlier-resilient ===
    def median_mad_normalize(x):
        x = np.copy(x)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Conditional DDL protection gate: binary activation only when slack <= 0 ===
    # Stronger than sigmoid: ensures immediate priority shift under hard deadline breach
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Critical path amplification: upward_rank * remaining_work interaction (validated by DDL diagnostics) ===
    # Captures both importance and impact — tasks unblocking large work on critical paths get boosted
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate (sigmoid) for pre-breach urgency ramp-up ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Uncertainty-duration interaction: activated only under dual pressure (high uncertainty AND low slack) ===
    uncert_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    duration_risk_score = norm_duration * uncert_gate * ddl_breach
    
    # === Bounded waiting bonus: saturating linear ramp up to max benefit, prevents over-prioritization ===
    # Avoids exponential saturation artifacts; uses wait_decay and small offset for stability
    wait_bonus = np.clip(PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]), 0.0, 1.0)
    
    # === Energy-uncertainty coupling: multiplicative penalty only when both high ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate
    
    # === Composite score: smaller = higher priority ===
    # Prioritize: (1) deadline breach response, (2) critical path leverage, (3) slack penalty,
    #             (4) energy efficiency, (5) starvation relief, (6) risk mitigation
    score = (
        + norm_slack_penalty                           # Primary lateness penalty
        - critical_path_leverage                       # DDL-diagnostic validated successor-release boost
        - boosted_rank                                 # HEFT-based urgency under pressure
        - PARAMS["energy_sensitivity"] * norm_energy  # Energy minimization under feasibility
        - wait_bonus                                   # Anti-starvation (bounded linear)
        + PARAMS["duration_robustness"] * duration_risk_score  # Slow + uncertain → penalized
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Joint risk penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Downstream work preference (moderate)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
