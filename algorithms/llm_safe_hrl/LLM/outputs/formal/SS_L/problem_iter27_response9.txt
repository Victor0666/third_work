def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule:
       - Restores strict binary deadline-pressure activation for criticality coupling (slack <= 0) — eliminates sigmoid dilution.
       - Replaces unstable pre-clamping with *adaptive outlier suppression via median-based IQR inflation*, preserving rank-order fidelity for N=1.
       - Introduces `urgency_sharpening_factor`: scales MAD-normalized slack to strengthen primary urgency driver without clipping distortion.
       - Removes redundant `successor_release_smooth` term; all urgency is channeled through sharpened slack + conditional critical coupling.
       - All normalization uses safe fallbacks for N=1; no unbounded ops; final score finite, deterministic, shape-(N,)."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust quantile normalization with adaptive IQR inflation for small-N stability ===
    # Instead of pre-clamping (which breaks rank order), we inflate IQR when quantiles collapse
    def quantile_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])
            q3 = np.quantile(x, PARAMS["quantile_q3"])
        iqr = q3 - q1
        # Inflate IQR minimally for stability without distorting scale
        iqr_safe = np.maximum(iqr, eps * (np.abs(q1) + np.abs(q3) + eps))
        return (x - q1) / iqr_safe
    
    norm_slack = quantile_normalize(slack)
    norm_energy = quantile_normalize(min_incremental_energy)
    norm_duration = quantile_normalize(min_exec_time + min_comm_time)
    norm_rank = quantile_normalize(upward_rank)
    norm_work = quantile_normalize(remaining_work)
    norm_wait = quantile_normalize(ready_wait_time)
    norm_uncert = quantile_normalize(uncertainty)
    
    # === Sign-preserving MAD normalization for slack — sharpened with tunable factor ===
    slack_med = np.median(slack)
    slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = np.maximum(slack_mad, eps)
    norm_slack_mad = (slack - slack_med) / slack_spread
    # Sharpen urgency near deadline violation without clipping distortion
    sharpened_urgency = PARAMS["urgency_sharpening_factor"] * norm_slack_mad
    
    # === Binary deadline pressure gate: strict (slack <= 0) for critical coupling ===
    ddl_pressure = (slack <= 0.0).astype(float)
    
    # === DDL-feasibility gate: smooth sigmoid for slack >= 0 ===
    ddl_feasible_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (slack - 0.0)))
    
    # === Slack-aware uncertainty gating: active only when feasible AND uncertainty exceeds adaptive threshold ===
    adaptive_uncert_threshold = PARAMS["slack_aware_uncert_threshold"] * (1.0 - 1.0 / (1.0 + np.exp(-slack)))
    slack_aware_uncert_gate = (uncertainty > adaptive_uncert_threshold).astype(float) * ddl_feasible_gate
    
    # === Criticality coupling: strictly binary-activated (no sigmoid smoothing) to preserve urgency fidelity ===
    critical_coupling = norm_rank * norm_work * ddl_pressure * PARAMS["criticality_coupling_strength"]
    
    # === Starvation relief: linear benefit only under deadline pressure ===
    wait_benefit_pressure = np.clip(PARAMS["wait_benefit_under_pressure"] * ready_wait_time, 0.0, 2.0) * ddl_pressure
    
    # === Energy-uncertainty coupling: enabled only under DDL feasibility and adaptive uncertainty condition ===
    energy_uncert_coupling = norm_energy * norm_uncert * slack_aware_uncert_gate
    
    # === Slack-driven energy amplification: only for feasible tasks, using sharpened urgency ===
    slack_pressure = np.clip(-norm_slack_mad, 0.0, 2.0)
    energy_slack_amplifier = 1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure
    
    # === Direct risk signal: normalized uncertainty weighted by feasibility ===
    risk_signal = norm_uncert * ddl_feasible_gate
    
    # === Composite score: smaller = higher priority; all terms bounded to [-2,2] for stability ===
    score = (
        + np.clip(sharpened_urgency, -2.0, 2.0)                           # Primary urgency driver, sharpened
        - np.clip(critical_coupling, -2.0, 2.0)                          # Strictly pressure-activated critical path leverage
        - np.clip(norm_rank, -2.0, 2.0)                                   # Baseline importance
        - np.clip(wait_benefit_pressure, -2.0, 2.0)                      # Pressure-specific anti-starvation
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_feasible_gate, -2.0, 2.0)  # Feasibility-gated energy preference
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_coupling, -2.0, 2.0)
        + np.clip(norm_energy * energy_slack_amplifier * ddl_feasible_gate, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * ddl_pressure, -2.0, 2.0)
        + np.clip(risk_signal, -2.0, 2.0)                                 # Feasibility-weighted risk
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
