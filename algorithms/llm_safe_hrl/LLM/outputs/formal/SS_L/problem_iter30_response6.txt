def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule:
       - Replaces unstable median-MAD with robust min-max normalization using explicit zero-slack guard.
       - Replaces power-law successor-release with bounded linear interaction: min_exec_time * norm_rank * min(slack_urgency_cap, max(0, -slack + eps)).
       - Removes inactive ddl_pressure_gate_threshold and redundant epsilon-guarded divisions.
       - Introduces slack_urgency_cap to prevent urgency overamplification near deadline breach.
       - All features normalized via [x - min(x)] / (max(x) - min(x) + eps) with clipping to [-2,2] for stability.
       - Preserves starvation relief, host-load surrogate, and energy-uncertainty coupling under unified feasibility gating.
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust min-max normalization with zero-slack guard — stable near critical slack ===
    def robust_minmax_normalize(x):
        x = np.copy(x)
        if N == 1:
            return np.zeros_like(x)
        x_min = np.min(x)
        x_max = np.max(x)
        spread = x_max - x_min
        denom = spread if spread > eps else eps
        return (x - x_min) / denom
    
    # Normalize all features — preserves sign via offset-free scaling; avoids MAD instability near zero slack
    norm_slack = robust_minmax_normalize(slack)
    norm_energy = robust_minmax_normalize(min_incremental_energy)
    norm_duration = robust_minmax_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_minmax_normalize(upward_rank)
    norm_work = robust_minmax_normalize(remaining_work)
    norm_wait = robust_minmax_normalize(ready_wait_time)
    norm_uncert = robust_minmax_normalize(uncertainty)
    
    # === Bounded slack urgency: linear penalty capped at slack_urgency_cap, active only when slack <= 0 ===
    # Uses -slack (positive urgency magnitude) but clamped to prevent dominance; adds eps to avoid zero-spike
    slack_urgency_magnitude = np.maximum(0.0, -slack + eps)
    bounded_slack_urgency = np.clip(slack_urgency_magnitude, 0.0, PARAMS["slack_urgency_cap"])
    # Normalize urgency to [0,1] for consistent weighting
    norm_slack_urgency = robust_minmax_normalize(bounded_slack_urgency)
    
    # === Critical-path urgency: rank × work × breach indicator — unchanged but now uses robust norm ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_urgency = norm_rank * norm_work * ddl_breach
    
    # === Bounded successor-release: linear (not power-law), gated by breach and capped urgency ===
    # Avoids overamplification while retaining fast-blocker prioritization
    successor_release = min_exec_time * norm_rank * ddl_breach * norm_slack_urgency
    
    # === Smooth ddl_pressure derived from normalized urgency (replaces fragile 1/(1+|slack|+eps)) ===
    # Now directly uses bounded, normalized urgency signal for consistency and stability
    ddl_pressure = norm_slack_urgency
    
    # === Host-load surrogate: duration × uncertainty, gated by both uncertainty threshold and ddl_pressure ===
    load_gate = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float) * ddl_pressure
    host_load_surrogate = norm_duration * norm_uncert * load_gate
    
    # === Energy-uncertainty coupling: active only when both uncertainty high AND ddl pressure active ===
    energy_uncert_gate = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float) * ddl_pressure
    energy_uncert_penalty = norm_energy * norm_uncert * energy_uncert_gate
    
    # === Starvation relief: exponential decay scaled by (1 - ddl_pressure) — anti-starvation only under safety margin ===
    wait_benefit = np.exp(-PARAMS["wait_decay"] * ready_wait_time) * (1.0 - ddl_pressure)
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for bounded AST depth and stability ===
    score = (
        + np.clip(norm_slack_urgency, -2.0, 2.0)                          # Primary urgency anchor: bounded, normalized, zero-guarded
        - np.clip(critical_path_urgency, -2.0, 2.0)                      # Critical path leverage under breach
        - np.clip(successor_release, -2.0, 2.0)                          # Linear, capped successor release
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_pressure, -2.0, 2.0)  # Energy preference gated by urgency
        - np.clip(wait_benefit, -2.0, 2.0)                               # Starvation relief under safety margin
        + np.clip(host_load_surrogate, -2.0, 2.0)                        # Load-aware congestion penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + PARAMS["criticality_scale"] * np.clip(norm_rank * ddl_pressure, -2.0, 2.0)  # HEFT rank scaled by bounded urgency
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
