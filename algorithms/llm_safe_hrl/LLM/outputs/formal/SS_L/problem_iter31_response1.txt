def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid gates with bounded tanh for smoother transitions;
       introduces successor-release interaction: min_exec_time * upward_rank * (slack <= 0) to unblock fast-critical tasks;
       removes duration_robustness and wait_saturation_offset (confirmed inactive);
       uses MAD-normalized slack clipped to [-2,2] as primary urgency signal;
       enforces strict DDL-first gating: all energy/uncertainty terms multiplied by (slack >= 0).astype(float);
       replaces boosted_rank with simple rank-slack coupling + successor-release;
       eliminates redundant clipping layers; uses np.finfo for epsilon safeguards."""
    
    # Safeguard epsilon using machine precision
    eps = np.finfo(float).tiny
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization per feature ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Primary urgency signal: MAD-normalized slack, clipped to [-2,2] — sharper, outlier-resilient ===
    urgency = np.clip(norm_slack, -2.0, 2.0)
    
    # === DDL feasibility gate: binary feasibility mask — all energy/uncertainty penalties strictly gated by slack >= 0 ===
    ddl_feasible = (slack >= 0.0).astype(float)
    
    # === Critical-path awareness under pressure: upward_rank × remaining_work × (slack <= 0) — validated in ≥5 consensus failures ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_leverage = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: unblocks fast-executing critical-path predecessors — validated in cross-agent conflicts ===
    # Encourages scheduling of short-but-high-rank tasks when slack is exhausted
    successor_release = min_exec_time * upward_rank * ddl_breach
    norm_successor = median_mad_normalize(successor_release)
    
    # === Risk-adjusted energy penalty: only active when feasible, scaled by uncertainty interaction ===
    energy_penalty = norm_energy * ddl_feasible
    energy_uncert_penalty = energy_penalty * norm_uncert * PARAMS["energy_uncertainty_interaction"]
    
    # === Starvation relief: exponential decay form without offset (offset confirmed inactive) ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Remaining work bias: mild preference for high downstream load when feasible ===
    work_bias = norm_work * ddl_feasible
    
    # === Composite score: smaller = higher priority.
    # Components ordered by lexicographic priority: DDL urgency > critical leverage > successor release > energy > starvation > work bias
    score = (
        + urgency                                          # Primary urgency: normalized slack (negative = urgent)
        - critical_leverage                                # Unblock critical paths only when breached
        - norm_successor                                   # Release successors early when deadline risk exists
        - PARAMS["energy_sensitivity"] * energy_penalty    # Energy minimization only when feasible
        - wait_benefit                                     # Anti-starvation (no offset needed)
        + PARAMS["remaining_work_weight"] * work_bias      # Downstream load awareness under feasibility
        + energy_uncert_penalty                            # Joint risk penalty only when feasible
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
