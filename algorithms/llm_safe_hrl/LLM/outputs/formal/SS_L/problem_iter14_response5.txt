def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: sign-preserving MAD normalization; strict DDL-protection gating;
       successor-release coupling enabled only under feasibility (not urgency); unified duration robustness;
       clipped linear slack mapping replaces power-law for stability under near-zero slack;
       all parameters used, no literals beyond [-2,-1,0,1,2]."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Sign-preserving MAD normalization: median center + MAD spread, retain raw sign ===
    def sign_preserving_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = sign_preserving_mad_normalize(slack)
    norm_energy = sign_preserving_mad_normalize(min_incremental_energy)
    norm_duration = sign_preserving_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = sign_preserving_mad_normalize(upward_rank)
    norm_work = sign_preserving_mad_normalize(remaining_work)
    norm_wait = sign_preserving_mad_normalize(ready_wait_time)
    norm_uncert = sign_preserving_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside (sigmoid) enabling energy/uncertainty ONLY when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Slack pressure gate: bounded linear interpolation for monotonic urgency ramp — avoids instability near zero ===
    # Maps slack ∈ [-∞, ∞] → pressure ∈ [0,1] via clipped normalized slack
    slack_pressure = np.clip(norm_slack, -1.0, 1.0)
    pressure_gate = 0.5 * (1.0 + slack_pressure)  # linear map [-1,1] → [0,1]
    
    # === Critical-path boost: upward_rank scaled by pressure_gate only (no exponentiation) ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * pressure_gate)
    
    # === Successor-release coupling: norm_work * norm_rank activated ONLY under DDL feasibility (not urgency) ===
    # Addresses counterfactual failures where high-downstream-work tasks were blocked despite feasibility
    successor_coupling = norm_work * norm_rank * ddl_gate
    
    # === Starvation relief: logistic decay with explicit clipping to prevent overflow ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    wait_benefit = np.clip(wait_benefit, 0.0, 0.999999)
    
    # === Uncertainty-duration interaction: gated by both DDL feasibility and uncertainty threshold ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * uncert_gate * ddl_gate
    
    # === Energy-uncertainty interaction: multiplicative penalty, gated only under DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_gate
    
    # === Slack-driven energy sensitivity: increases energy penalty linearly as slack tightens (positive but small) ===
    # Uses clipped linear slack instead of power-law for numerical stability under tight deadlines
    tight_slack_ratio = np.clip((np.maximum(0.0, slack) / (np.max(np.abs(slack)) + eps)), 0.0, 1.0)
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * (1.0 - tight_slack_ratio)) * ddl_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for bounded AST depth ===
    score = (
        # Primary urgency: negative slack dominates (penalty for lateness)
        + np.clip(-norm_slack, -2.0, 2.0)
        # Critical path boost: HEFT rank amplified under pressure
        - np.clip(boosted_rank, -2.0, 2.0)
        # Successor release: prioritize unblocking large downstream work only when DDL feasible
        - PARAMS["successor_coupling_strength"] * np.clip(successor_coupling, -2.0, 2.0)
        # Energy preference only when feasible, scaled by slack tightness
        - PARAMS["energy_sensitivity"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        # Anti-starvation bonus (reduces score → increases chance)
        - np.clip(wait_benefit, -2.0, 2.0)
        # Duration robustness: penalty for uncertain durations only when feasible
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        # Risk penalty: energy-uncertainty interaction only under feasibility
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        # Remaining work bias: mild preference for high-impact nodes
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
