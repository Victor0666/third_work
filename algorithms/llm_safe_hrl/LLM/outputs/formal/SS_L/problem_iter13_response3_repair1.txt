def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces unstable sigmoid gates with bounded linear interpolation;
       introduces successor-release interaction via norm_work * norm_rank under DDL pressure;
       removes duration_robustness (inactive per diagnostics) and wait_saturation_offset (redundant);
       uses clipped slack-based boosting instead of rank_slack_coupling + rank_gate;
       enforces strict monotonicity via clip(norm_slack, -1, 1); eliminates unbounded exponentials."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL feasibility gate: binary-like but smooth (tanh-based) for slack <= 0 → 1.0, slack > 0 → 0.0
    # Replaces sigmoid with numerically stable tanh(-slack / eps), then clamped to [0,1]
    ddl_feasible = np.clip(0.5 * (1.0 - np.tanh(slack / (eps + np.finfo(float).tiny))), 0.0, 1.0)
    
    # === Slack pressure indicator: bounded linear interpolation from -1 to 1 → [0,1] for urgency scaling
    # More stable than sigmoid; monotonic and numerically robust
    slack_pressure = np.clip(norm_slack, -1.0, 1.0)
    
    # === Critical path leverage: successor-release interaction — norm_work * norm_rank under DDL pressure
    # Directly implements 'add_successor_release_interaction' and 'add_upward_rank_remaining_work_interaction'
    # Activated only when slack is tight (negative or near-zero), per high-confidence structural feedback
    ddl_urgent = (slack <= eps).astype(float)
    successor_release = norm_work * norm_rank * ddl_urgent
    
    # === Negative slack penalty: power law on raw negative slack, normalized and clipped
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Energy penalty: only applied when feasible AND under slack pressure
    energy_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * (1.0 - ddl_feasible)
    
    # === Uncertainty-energy interaction: gated by both uncertainty threshold and DDL feasibility
    uncert_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * (1.0 - ddl_feasible)
    
    # === Starvation relief: exponential decay with wait time, no saturation offset (inactive per diagnostics)
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability
    # Removed: duration_robustness (inactive), wait_saturation_offset (inactive), redundant rank coupling
    # Added: direct successor_release boost (validated by DDL diagnostics), simplified slack pressure
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                          # Lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                           # Critical-path release boost (new core interaction)
        - np.clip(norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_pressure), -2.0, 2.0)  # Urgency amplification
        - PARAMS["energy_sensitivity"] * np.clip(energy_penalty, -2.0, 2.0)  # Feasible-region energy penalty
        - np.clip(wait_benefit, -2.0, 2.0)                                # Anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
