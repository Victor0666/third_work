def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: removes unused 'rank_slack_coupling'; retains robust starvation relief, adaptive gating, and novel energy-slack interaction."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median-centered MAD + epsilon → resilient to outliers and zero-scales ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        center = np.median(x_abs)
        scale = np.median(np.abs(x_abs - center)) + eps
        return (x_abs - center) / scale
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack pressure: clipped power penalty for overdue tasks only ===
    slack_pressure = np.power(np.clip(-norm_slack, 0.0, None), PARAMS["slack_penalty_exponent"])
    
    # === Critical-path acceleration gate: smooth linear ramp down from 1→0 over [0, slack_gate_width] ===
    rank_gate = np.clip(1.0 - (norm_slack / (PARAMS["slack_gate_width"] + eps)), 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Uncertainty-duration interaction: binary AND-gate — activates only when both high uncertainty and slack pressure exist ===
    uncert_gate = (
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) &
        (slack_pressure > 0)
    ).astype(float)
    duration_risk_interaction = norm_duration * uncert_gate * PARAMS["duration_robustness"]
    
    # === Starvation relief: saturating exponential form — superior gradient behavior vs tanh ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + eps))
    
    # === Energy-slack interaction: penalizes high marginal energy more under tight slack ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["energy_slack_interaction"] * slack_pressure)
    
    # === Final score: smaller = higher priority ===
    score = (
        + slack_pressure                                  # Urgency driver
        + energy_slack_penalty                            # Deadline-amplified energy cost
        + PARAMS["energy_sensitivity"] * norm_energy     # Base energy penalty
        - boosted_rank                                   # Critical path reward
        - wait_benefit                                   # Anti-starvation reward
        + duration_risk_interaction                      # Conditional risk-duration penalty
    )
    
    # Final safeguard: ensure finite, deterministic output with strict shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(-1)
