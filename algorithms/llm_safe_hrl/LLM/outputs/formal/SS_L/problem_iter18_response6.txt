def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces deprecated duration_robustness with active comm-energy coupling;
       introduces successor-release gate to prioritize unblocking of critical downstream work;
       removes inactive wait_saturation_offset and uses robust logistic starvation relief;
       retains median-MAD normalization and strict DDL-first semantics via dual-gated terms."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — outlier-resilient and preserves sign ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    # NEW: comm-energy coupling — directly penalizes high-comm & high-energy jointly
    norm_comm_energy = median_mad_normalize(min_comm_time * min_incremental_energy)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: upward_rank * remaining_work interaction, activated only under deadline breach ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded slack-pressure activation: clipped negative slack for pre-breach urgency ramp-up ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    
    # === Sharpened rank amplification: sigmoid-gated coupling of slack pressure and rank ===
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Successor-release gate: prioritizes tasks whose execution unblocks critical downstream work ===
    # Uses slack vs. normalized release horizon (remaining_work / (upward_rank + eps)) to identify bottleneck releases
    release_horizon = remaining_work / (upward_rank + eps)
    successor_release_gate = 1.0 / (1.0 + np.exp(-PARAMS["successor_release_steepness"] * (slack - release_horizon)))
    
    # === Starvation relief: saturating logistic benefit (avoids linear bias, ensures bounded influence) ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Energy-uncertainty coupling: gated by both uncertainty and DDL feasibility ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Comm-energy coupling: penalizes joint high communication and high marginal energy under DDL pressure ===
    comm_energy_penalty = norm_comm_energy * ddl_gate * slack_pressure
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Unblock critical paths when breaching
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Amplified urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy minimization only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation (saturating)
        + PARAMS["comm_energy_coupling_weight"] * np.clip(comm_energy_penalty, -2.0, 2.0)  # NEW: direct comm+energy penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)  # Prefer large downstream work when safe
        - np.clip(successor_release_gate, -2.0, 2.0)                       # NEW: prioritize successor-releasing tasks
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
