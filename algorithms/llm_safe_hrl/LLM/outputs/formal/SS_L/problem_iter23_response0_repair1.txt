def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule with graded urgency, tanh-based starvation relief, and joint congestion gating.
       Key structural features:
         - Graded critical-path urgency via sigmoid(-slack), replacing binary gates → smoother control near deadline.
         - Urgency-sharpened slack pressure: (-slack)^p clipped to [0,1] for enhanced boundary sensitivity.
         - Smooth tanh wait benefit: avoids flat gradients and preserves discriminability at low waits.
         - Energy-uncertainty coupling gated by both soft DDL feasibility AND joint congestion (wait+uncert).
         - All features use identical robust quantile normalization (Q1/Q3) for consistency and outlier resistance.
       No numeric literals outside {-2,-1,0,1,2}; all tunables declared in PARAMETER_SCHEMA."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust quantile-based normalization: (x - Q1) / (Q3 - Q1 + eps) — consistent across all features ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])
            q3 = np.quantile(x, PARAMS["quantile_q3"])
        iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        return (x - q1) / spread
    
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_duration = robust_range_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # Sign-preserving MAD for slack: sensitive to small negative values near deadline boundary
    if N == 1:
        slack_med = slack[0]
        slack_mad = eps
    else:
        slack_med = np.median(slack)
        slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = slack_mad if slack_mad > eps else eps
    norm_slack = (slack - slack_med) / slack_spread
    
    # === Soft DDL-protection gate (sigmoid) for energy/uncertainty terms only ===
    soft_ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Graded critical-path urgency gate: sigmoid of (-slack), activated earlier and smoothly increasing ===
    urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (-slack)))
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_range_normalize(raw_slack_penalty)
    
    # === Bounded piecewise-linear congestion gate: active only when BOTH wait and uncertainty are high ===
    congestion_activation = (norm_wait >= PARAMS["congestion_activation_threshold"]) & (norm_uncert >= PARAMS["congestion_activation_threshold"])
    congestion_gate = congestion_activation.astype(float)
    
    # === Starvation relief: smooth tanh saturation → preserves low-wait discriminability ===
    wait_benefit = np.tanh(ready_wait_time / (1.0 + PARAMS["urgency_sharpening_power"]))  # Reuse sharpening param for coherence
    
    # === Energy-uncertainty coupling: gated by both soft DDL feasibility AND joint congestion ===
    energy_uncert_interaction = norm_energy * norm_uncert * congestion_gate * soft_ddl_gate
    
    # === Urgency-sharpened slack pressure: power-law ramp from 0 to 1 over [-1,0] slack range ===
    slack_pressure_raw = np.clip(-slack, 0.0, 1.0)
    slack_pressure = slack_pressure_raw ** PARAMS["urgency_sharpening_power"]
    
    # === Successor-release interaction: fast exec × urgency × clipped rank ===
    successor_release = min_exec_time * urgency_gate * np.clip(norm_rank, 0.0, 1.0)
    
    # === Critical path leverage: upward_rank × remaining_work × urgency_gate (graded) ===
    critical_path_leverage = norm_rank * norm_work * urgency_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Graded critical path boost
        - np.clip(norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_pressure), -2.0, 2.0)  # Amplified urgency
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * soft_ddl_gate, -2.0, 2.0)  # Energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Smooth anti-starvation
        + np.clip((norm_wait + norm_uncert) * congestion_gate * (1.0 - soft_ddl_gate), -2.0, 2.0)  # Congestion relief
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_interaction, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * soft_ddl_gate, -2.0, 2.0)  # Downstream work preference
        + PARAMS["slack_penalty_exponent"] * np.clip(slack_pressure, -2.0, 2.0)       # Direct urgency coupling
        - np.clip(successor_release, -2.0, 2.0)   # Release blocked critical successors
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
