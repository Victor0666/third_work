def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid gates with bounded linear interpolation for monotonicity and overflow safety;
       replaces median-MAD with sign-preserving robust range normalization (Q3-Q1) for stronger urgency directionality;
       removes duration_robustness and wait_saturation_offset (inactive per diagnostics);
       introduces direct upward_rank * remaining_work interaction under DDL breach, validated by counterfactuals;
       enforces strict priority ordering: slack penalty dominates, then critical path leverage, then energy/uncertainty only if feasible."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust range normalization (Q3 - Q1) — preserves sign, resists outliers better than MAD for urgency ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1, q3 = x[0], x[0]
            iqr = eps
        else:
            q1 = np.quantile(x, 0.25)
            q3 = np.quantile(x, 0.75)
            iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        return (x - (q1 + q3) / 2.0) / spread
    
    norm_slack = robust_range_normalize(slack)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_duration = robust_range_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside using slope from PARAMS — now used ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: direct upward_rank * remaining_work interaction, activated only under slack <= 0 ===
    # Strongly validated by counterfactuals: 'critical_path_or_successor_blocked+higher_ddl_risk_choice'
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Slack penalty: power-law on raw negative slack, normalized robustly, then clipped ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_range_normalize(raw_slack_penalty)
    
    # === Slack-pressure coupling: sigmoid gate using slack_pressure_gate_steepness — now used ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    rank_coupling = 1.0 + PARAMS["criticality_scale"] * rank_gate
    
    # === Uncertainty gating: linear threshold gate centered at uncertainty_gate_threshold ===
    uncert_centered = norm_uncert - PARAMS["uncertainty_gate_threshold"]
    uncert_gate = np.clip(uncert_centered / (0.5 + eps), 0.0, 1.0)
    
    # === Starvation relief: linear benefit capped at 1.0, no log/exp → avoids saturation artifacts ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Energy-uncertainty coupling: only active when both uncertainty is high AND slack permits energy consideration ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Composite score: strictly ordered priority hierarchy — smaller = higher priority ===
    # Order: (1) lateness penalty dominates, (2) critical path rescue boosts, (3) urgency coupling,
    # (4) energy only if feasible, (5) anti-starvation, (6) uncertainty penalty, (7) remaining work boost
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary deadline violation penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Rescues blocked critical paths (DDL-diagnostic validated)
        - np.clip(norm_rank * rank_coupling, -2.0, 2.0)                   # Urgency amplification only in safe slack zone
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference gated by feasibility
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Linear anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguards: finite values, deterministic shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(-1)
