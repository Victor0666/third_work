def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces raw_slack_pressure with differentiable tanh urgency;
       enforces strict critical-path suppression when slack > 0 via (1 - ddl_gate) gating;
       normalizes starvation relief by workflow-age scale instead of quantiles;
       uses tunable tanh_offset (0.5) instead of hardcoded 0.5."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely without in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust range-based normalization: (x - Q1) / (Q3 - Q1 + eps) — preserves sign, resists outliers ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])
            q3 = np.quantile(x, PARAMS["quantile_q3"])
        iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        return (x - q1) / spread
    
    # Normalize all features except slack (which gets sign-preserving MAD for urgency sensitivity near zero)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_exec = robust_range_normalize(min_exec_time)
    norm_comm = robust_range_normalize(min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # Sign-preserving MAD for slack: sensitive to small negative values near deadline boundary
    if N == 1:
        slack_med = slack[0]
        slack_mad = eps
    else:
        slack_med = np.median(slack)
        slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = slack_mad if slack_mad > eps else eps
    norm_slack = (slack - slack_med) / slack_spread
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty terms only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Bounded differentiable urgency: tanh-based sharp transition near slack=0, scaled by urgency_sharpness ===
    # tanh(k * -slack) → [-1,1]; maps negative slack to positive urgency signal, smoothly decays to 0 as slack increases
    # Use tunable tanh_offset to replace hardcoded 0.5
    urgency_signal = PARAMS["tanh_offset"] * (1.0 - np.tanh(PARAMS["urgency_sharpness"] * slack))
    
    # === Critical path leverage: activated ONLY when slack <= 0 (feasible-first enforcement) ===
    # Now strictly gated by (1 - ddl_gate) to suppress when slack > 0
    critical_path_leverage = norm_rank * norm_work * (1.0 - ddl_gate)
    
    # === Unified urgency surface: bounded product, now strictly feasibility-gated ===
    urgency_surface = np.clip(norm_rank * norm_work * urgency_signal * (1.0 - ddl_gate), -2.0, 2.0)
    
    # === Separated execution & communication penalties: individually gated and scaled ===
    exec_penalty = PARAMS["exec_comm_separation_factor"] * norm_exec * urgency_signal * ddl_gate
    comm_penalty = (1.0 - PARAMS["exec_comm_separation_factor"]) * norm_comm * urgency_signal * ddl_gate
    
    # === Uncertainty-weighted risk gating: activates only when uncertainty exceeds threshold and slack is tight ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: normalized relative to workflow age (not quantiles) for generalization ===
    # Estimate workflow age as max(ready_wait_time) across ready tasks → proxy for current workflow time horizon
    workflow_age = np.max(ready_wait_time) if N > 0 else eps
    wait_normalized = ready_wait_time / (workflow_age * PARAMS["wait_saturation_point"] + eps)
    wait_benefit = np.clip(wait_normalized, 0.0, 1.0)
    
    # === Slack-aware energy scaling: direct amplification proportional to urgency_signal ===
    energy_slack_penalty = norm_energy * urgency_signal * ddl_gate
    
    # === Composite score: smaller = higher priority; simplified and stabilized ===
    score = (
        + np.clip(urgency_signal ** PARAMS["slack_penalty_exponent"], -2.0, 2.0)  # Primary nonlinear lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                              # Unblock critical paths only when needed
        - np.clip(urgency_surface, -2.0, 2.0)                                     # Feasibility-gated critical activation
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only in safe zone
        - np.clip(wait_benefit, -2.0, 2.0)                                        # Temporally anchored anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + np.clip(exec_penalty, -2.0, 2.0)                                       # Compute-bound urgency
        + np.clip(comm_penalty, -2.0, 2.0)                                       # I/O-bound deferral control
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * (1.0 - ddl_gate), -2.0, 2.0)  # Downstream work in safe regime
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
