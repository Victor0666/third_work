def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
       - Replaced brittle multiplicative criticality scaling with numerically stable `norm_rank * (1 + PARAMS["tanh_slack_coupling_scale"] * np.tanh(slack))`.
       - Introduced host-load conditional gate: suppresses energy/uncertainty penalties when congestion detected via `(ready_wait_time * uncertainty) > threshold`.
       - Removed `duration_criticality` term to reduce feature redundancy and improve interpretability.
       - All intermediate terms clipped to [-2,2]; no numeric literals beyond {-2,-1,0,1,2}; deterministic and finite output."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Deadline pressure gate: binary for slack <= 0 (activates urgency & starvation relief) ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Piecewise-linear slack-pressure gate: bounded ramp from 0 to 1 over [-offset, +offset] around zero slack ===
    slack_ramp_center = PARAMS["piecewise_slack_offset"]
    slack_pressure_ramp = np.clip(
        (norm_slack + slack_ramp_center) * PARAMS["slack_pressure_gate_steepness"],
        0.0, 1.0
    )
    
    # === Host-load congestion gate: detects queue pressure via product of wait time and uncertainty ===
    load_metric = ready_wait_time * uncertainty
    host_load_gate = (load_metric <= PARAMS["host_load_gate_threshold"]).astype(float)
    
    # === Successor-release interaction: upward_rank × remaining_work × uncertainty-gated × deadline-pressure-activated ===
    successor_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    successor_release = norm_rank * norm_work * successor_gate * ddl_breach * PARAMS["successor_release_coupling"]
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Smooth tanh-based criticality coupling: replaces multiplicative amplification with bounded, differentiable slack-aware rank boost ===
    # More stable near slack=0 than linear ramps or sigmoids; avoids flat regions and discontinuities
    tanh_slack_coupling = 1.0 + PARAMS["tanh_slack_coupling_scale"] * np.tanh(slack)
    boosted_rank = norm_rank * tanh_slack_coupling
    
    # === Starvation relief: logistic-shaped anti-starvation bonus gated by deadline pressure AND load congestion ===
    wait_benefit_pressure = (1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)) * ddl_breach * host_load_gate
    
    # === Energy-uncertainty coupling: now suppressed under host congestion to avoid over-penalizing during queue pressure ===
    energy_uncert_penalty = norm_energy * norm_uncert * successor_gate * ddl_gate * host_load_gate
    
    # === Slack-aware energy scaling: remains active only when feasible AND not congested ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure_ramp) * ddl_gate * host_load_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Critical-path-aware successor unblocking
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Smooth tanh-coupled HEFT urgency
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate * host_load_gate, -2.0, 2.0)  # Energy preference only when feasible AND uncongested
        - np.clip(wait_benefit_pressure, -2.0, 2.0)                        # Congestion-gated anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
