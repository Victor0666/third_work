def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: removes unused slack_penalty_exponent; all declared parameters used.
    Adds conditional DDL protection gate, successor-release interaction, and median-MAD normalization."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization: preserves sign and handles outliers robustly ===
    def median_mad_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        dev = np.abs(x - center)
        spread = np.median(dev) if N > 1 else dev[0] + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features — preserve original sign semantics
    norm_slack = median_mad_normalize(slack)  # negative slack → low value → high priority
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Binary DDL urgency gate: hard prioritization for tasks violating slack threshold ===
    ddl_urgent = (slack <= 0.0).astype(float)  # 1.0 if at or past deadline, else 0.0
    
    # === Conditional DDL protection gate: boosts upward_rank *only* when slack is tight or violated ===
    slack_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (-norm_slack)))
    protected_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * (ddl_urgent + (1.0 - ddl_urgent) * slack_pressure_gate))
    
    # === Successor-release interaction: amplifies priority for tasks unblocking many successors ===
    successor_release_score = norm_work * norm_rank
    
    # === Energy-uncertainty interaction: penalizes high-energy tasks *only* when uncertainty exceeds threshold ===
    energy_uncert_penalty = norm_energy * np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    
    # === Duration-robustness coupling: multiplies duration by uncertainty only under DDL pressure ===
    duration_risk_score = norm_duration * norm_uncert * np.clip(-norm_slack, 0.0, np.inf)
    
    # === Starvation relief: bounded logistic benefit with saturating offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Composite score: smaller = higher priority ===
    score = (
        - ddl_urgent * 100.0                        # Hard priority boost for urgent tasks
        - protected_rank                           # Critical path awareness under pressure
        - successor_release_score                  # Prefer tasks releasing large downstream work
        - PARAMS["energy_sensitivity"] * norm_energy  # Favor low marginal energy
        - norm_duration                            # Favor short execution+comm time
        - wait_benefit                             # Anti-starvation bonus
        + PARAMS["duration_robustness"] * duration_risk_score  # Penalize risky-slow under pressure
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned energy penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
