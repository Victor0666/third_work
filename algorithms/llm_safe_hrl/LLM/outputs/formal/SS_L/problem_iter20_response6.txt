def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's robust DDL-gating and saturation-safe starvation relief with Parent 1's validated critical-path leverage;
       introduces *power-law rank-slack coupling* (replacing linear) for smoother, more discriminative urgency scaling;
       refines *slack alignment* using adaptive clip threshold to balance sensitivity and stability;
       unifies all risk terms (energy, duration, uncertainty) under identical ddl_gate and slack_pressure gating;
       removes redundant 'critical_path_leverage_weight' and replaces with direct norm_rank × norm_work × ddl_breach — preserving diagnostic validity;
       clips all intermediate terms to [-2,2] for AST depth control and numerical stability."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: upward_rank × remaining_work × ddl_breach — validated and retained ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Adaptive slack alignment: signed, clipped deviation from zero slack with tunable bound ===
    # Improves monotonicity and avoids saturation at extremes vs fixed [-1,1]
    slack_alignment = np.clip(norm_slack, -PARAMS["slack_alignment_clip"], PARAMS["slack_alignment_clip"])
    
    # === Power-law rank-slack coupling: sub/super-linear response under pressure ===
    # Replaces linear coupling with more expressive, bounded interaction
    coupled_slack_power = np.sign(slack_alignment) * np.abs(slack_alignment) ** PARAMS["rank_coupling_power"]
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack_power
    
    # === Sharpened slack-pressure gate: sigmoid activation centered at pressure=1.0 ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Unified risk interaction block: duration-uncertainty-energy all share same gating logic ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * norm_uncert * uncert_gate * slack_pressure * ddl_gate
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus with saturation-safe offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]))
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Validated breach-specific leverage
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Power-coupled, pressure-gated criticality
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # DDL-feasible energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Saturation-safe starvation relief
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
