def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces dual-gated urgency with *relative-slack* tanh urgency
       (tanh(-slack / (mean|slack| + eps))) for improved gradient sensitivity near deadlines;
       reverts successor-release to additive local-gated term (no coupling); removes global ddl_protection_boost
       to avoid diluting per-task urgency; retains robust mean-abs normalization and bounded tanh anti-starvation.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # === Robust mean-abs feature scaling: x / (mean(|x|) + eps) — low-cost, outlier-resilient ===
    def robust_norm(x):
        x_abs = np.abs(x)
        denom = np.mean(x_abs) + eps
        return x / denom
    
    norm_slack = robust_norm(slack)
    norm_energy = robust_norm(min_incremental_energy)
    norm_duration = robust_norm(min_exec_time + min_comm_time)
    norm_rank = robust_norm(upward_rank)
    norm_work = robust_norm(remaining_work)
    norm_wait = robust_norm(ready_wait_time)
    norm_uncert = robust_norm(uncertainty)
    
    # === Relative-slack urgency: tanh(-slack / (mean|slack| + eps)) → preserves fine-grained gradient
    # near zero slack while remaining bounded, finite, and monotonic — addresses reflection's core concern
    slack_abs_mean = np.mean(np.abs(slack)) + eps
    relative_urgency = -slack / slack_abs_mean
    slack_urgency = np.tanh(PARAMS["slack_urgency_scale"] * relative_urgency)
    norm_urgency = robust_norm(slack_urgency)
    
    # === Conditional DDL protection gate (local): safe+urgent → low uncertainty AND tight slack ===
    slack_tight = (slack < 0.0) | (norm_slack < PARAMS["slack_tightness_threshold"])
    uncert_low = norm_uncert < PARAMS["uncertainty_gate_threshold"]
    local_ddl_gate = np.where(slack_tight & uncert_low, 1.0, 0.0)
    
    # === Critical path boost: upward_rank amplified only under local DDL gate ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * local_ddl_gate)
    
    # === Local successor-release bonus: additive remaining_work * upward_rank under local DDL gate
    # (reverts coupling; avoids noise amplification while preserving critical-unblocking intent)
    release_bonus = (
        PARAMS["local_release_bonus_weight"] *
        norm_work * norm_rank * local_ddl_gate
    )
    
    # === Bounded anti-starvation: tanh-based benefit, offset for numerical stability ===
    wait_benefit = np.tanh(PARAMS["wait_benefit_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Joint risk-energy penalty: only active when both slack < 0 AND uncertainty high ===
    risk_energy_gate = np.where((slack < 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]), 1.0, 0.0)
    energy_uncert_penalty = norm_energy * risk_energy_gate * PARAMS["energy_uncertainty_interaction"]
    
    # === Composite score: smaller = higher priority ===
    # Prioritization order: (1) relative-slack urgency (fine-grained near deadline), (2) critical path boost,
    #                       (3) local unblocking bonus, (4) anti-starvation, (5) joint risk-energy penalty,
    #                       (6) mild downstream work weight
    score = (
        + norm_urgency                                     # Fine-grained, bounded urgency near deadline
        - boosted_rank                                     # Local critical path amplification
        - PARAMS["energy_sensitivity"] * norm_energy      # Energy preference
        - wait_benefit                                       # Stable starvation relief
        - release_bonus                                      # Local unblocking of critical work
        + energy_uncert_penalty                              # Risk-energy penalty
        + PARAMS["remaining_work_weight"] * norm_work       # Downstream work weight
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    return score.reshape(-1)
