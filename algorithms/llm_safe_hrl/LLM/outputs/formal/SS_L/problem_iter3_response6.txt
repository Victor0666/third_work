def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with hard-deadline protection and load-aware proxy.
    
    Key structural improvements:
      - DDL-protection gate: when normalized slack falls below dynamic threshold,
        switches to pure urgency/criticality ranking (ignores energy/duration) to
        enforce hard deadline feasibility first.
      - Host-load proxy: uses ready_wait_time × uncertainty as bounded surrogate
        for VM congestion risk — penalizes tasks likely to worsen overloaded hosts.
      - Task-type–aware normalization: splits normalization into critical-path-
        dominant (upward_rank > median) vs. non-critical paths to preserve
        discriminative signal under high variance.
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Task-type–aware normalization: split by upward_rank percentile ===
    # Critical-path tasks (top 50% rank) use separate MAD scaling for sharper discrimination
    rank_median = np.median(upward_rank)
    is_critical = (upward_rank >= rank_median)
    
    def robust_normalize(x, mask=None):
        x = np.abs(x)
        center = np.median(x) if mask is None else np.median(x[mask])
        # Use masked scale only if mask has sufficient support
        if mask is not None and np.sum(mask) > 1:
            scale = np.median(np.abs(x[mask] - center)) + eps
        else:
            scale = np.median(np.abs(x - center)) + eps
        return (x - center) / scale
    
    # Normalize base features with task-type awareness
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # Critical-path-specific normalization for rank & slack to enhance sensitivity
    norm_rank_crit = robust_normalize(upward_rank, is_critical)
    norm_slack_crit = robust_normalize(slack, is_critical)
    
    # === Raw slack penalty: exponentiated overdue magnitude ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Slack pressure signal: clamped & centered ===
    slack_pressure = np.clip(-norm_slack, 0.0, PARAMS["slack_pressure_clamp"])
    
    # === Criticality gate: sigmoid-shaped coupling using clamped pressure ===
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["rank_slack_coupling"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank_crit * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Uncertainty-duration interaction: active only under high uncertainty AND pressure ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = norm_duration * uncert_gate * slack_pressure
    
    # === Energy-uncertainty risk amplification ===
    energy_uncert_risk = norm_energy * norm_uncert * uncert_gate
    
    # === Host-load proxy: ready_wait_time × uncertainty → captures starvation-induced load pressure ===
    load_proxy = np.clip(norm_wait * norm_uncert, 0.0, 2.0)
    
    # === Starvation relief: logistic with explicit offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === DDL-protection gate: activate strict urgency mode when slack critically low ===
    # Uses normalized slack (not raw) to be scale-invariant across workloads
    ddl_protect_mask = (norm_slack < -PARAMS["ddl_protection_threshold"])
    
    # In protection mode: prioritize only urgency + criticality; suppress energy/duration preferences
    score_base = (
        + norm_slack_penalty
        - boosted_rank
        - wait_benefit
        + PARAMS["duration_robustness"] * duration_risk_score
        + PARAMS["load_proxy_weight"] * load_proxy
    )
    
    score_relaxed = (
        + norm_slack_penalty
        - boosted_rank
        - PARAMS["energy_sensitivity"] * norm_energy
        - norm_duration
        - wait_benefit
        + PARAMS["duration_robustness"] * duration_risk_score
        + PARAMS["remaining_work_weight"] * norm_work
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_risk
        + PARAMS["load_proxy_weight"] * load_proxy
    )
    
    # Blend: full protection when mask true, else relaxed
    score = np.where(ddl_protect_mask, score_base, score_relaxed)
    
    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required even for N=1
    return score.reshape(-1)
