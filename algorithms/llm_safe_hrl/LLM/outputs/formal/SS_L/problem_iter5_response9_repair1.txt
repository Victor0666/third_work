def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with all declared parameters used and no unused/missing references.
       Key properties:
         - Uses only declared PARAMS keys; no slack_penalty_exponent or slack_tightness_threshold.
         - All numeric literals are in {-2,-1,0,1,2}; epsilon comes from PARAMS["epsilon"].
         - Robust mean-abs scaling without centering.
         - Smooth tanh-based urgency and gating (no boolean branches).
         - Successor-release coupling via norm_work * norm_rank * ddl_protection_gate.
         - Finite, shape-(N,) output with full NaN/inf protection.
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust feature scaling: mean(|x|) + eps as scale (no centering) ===
    def scale_feature(x):
        x_abs = np.abs(x)
        spread = np.mean(x_abs) + eps
        return x / (spread + eps)
    
    # Normalize base features
    norm_slack = scale_feature(slack)
    norm_energy = scale_feature(min_incremental_energy)
    norm_duration = scale_feature(min_exec_time + min_comm_time)
    norm_rank = scale_feature(upward_rank)
    norm_work = scale_feature(remaining_work)
    norm_wait = scale_feature(ready_wait_time)
    norm_uncert = scale_feature(uncertainty)
    
    # === Smooth urgency signal: tanh(-norm_slack * urgency_smoothing_factor) → [-1,1] ===
    urgency_signal = np.tanh(-norm_slack * PARAMS["urgency_smoothing_factor"])
    
    # === Smooth DDL-protection gate: activated when urgency is high AND uncertainty is low ===
    # Uses tanh to smoothly suppress gate when uncertainty exceeds threshold
    uncert_gate = np.tanh((PARAMS["uncertainty_gate_threshold"] - norm_uncert) * 2.0)
    ddl_protection_gate = np.clip(urgency_signal * uncert_gate, 0.0, 1.0)
    
    # === Boosted rank: critical-path importance amplified only under safe+urgent conditions ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_protection_gate)
    
    # === Successor-release coupling: unblocking large critical work under DDL pressure ===
    successor_release_score = (
        norm_work * norm_rank * PARAMS["successor_release_coupling_strength"] * ddl_protection_gate
    )
    
    # === Bounded anti-starvation: tanh-based, stable, saturates at ±1 ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Joint risk-energy penalty: active only when both urgency and uncertainty are high ===
    # Uses complementary gate: high uncertainty * high urgency → penalty
    risk_energy_penalty = (
        norm_energy * norm_uncert * ddl_protection_gate * (1.0 - uncert_gate)
    )
    
    # === Composite score: smaller = higher priority ===
    # Prioritizes: (1) Urgency compliance, (2) Critical path under safety, (3) Unblocking successors,
    #              (4) Starvation relief, (5) Risk-energy penalty, (6) Energy efficiency, (7) Downstream load
    score = (
        + urgency_signal                               # Primary DDL urgency (bounded, smooth)
        - boosted_rank                                 # Critical path boost under safe+urgent
        - successor_release_score                      # Unblock large critical downstream work
        - wait_benefit                                 # Anti-starvation (tanh-scaled)
        + PARAMS["energy_uncertainty_interaction"] * risk_energy_penalty  # Risk-aligned energy penalty
        - PARAMS["energy_sensitivity"] * norm_energy  # Energy efficiency preference
        + PARAMS["remaining_work_weight"] * norm_work  # Mild downstream work weight
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
