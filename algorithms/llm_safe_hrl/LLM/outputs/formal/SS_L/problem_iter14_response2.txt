def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's numerical stability with Parent 1's structured slack-pressure gating;
       introduces novel sigmoid-based slack-pressure gate for critical-path coupling — smoother and more discriminative than linear ramp;
       retains Parent 2's strict DDL-protection gate to disable all risk-sensitive penalties when slack < 0;
       replaces raw duration penalty with robustified duration-uncertainty product under DDL pressure;
       adds explicit successor-release weight parameter for calibrated downstream impact;
       uses median-MAD normalization throughout for outlier resilience and sign preservation."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: binary-like smooth step (clipped linear) — disables risk terms when slack < 0 ===
    ddl_gate = np.clip(slack, 0.0, 1.0)
    ddl_gate = np.where(slack < 0.0, 0.0, ddl_gate)  # Strict zero for negative slack → no energy/uncertainty penalties
    
    # === Slack-pressure signal: sigmoid over normalized slack for stable, bounded critical-path activation ===
    # Maps norm_slack ∈ [-∞, ∞] → [0,1], centered at norm_slack = 0 (tight deadline), steepness controlled by param
    slack_pressure = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_steepness"] * (-norm_slack)))
    
    # === Successor-release interaction: rank * work, scaled by pressure and new tunable weight ===
    successor_release = PARAMS["successor_release_weight"] * norm_rank * norm_work * slack_pressure
    
    # === Critical-path coupling: upward_rank scaled by slack pressure, not exponentiated or clipped ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_pressure)
    
    # === Uncertainty-aware energy penalty: gated by both DDL feasibility AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: linear ramp-up with clean saturation ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Robustified duration-uncertainty penalty: only active under DDL pressure and feasibility ===
    duration_uncert_penalty = norm_duration * norm_uncert * slack_pressure * ddl_gate
    
    # === Linear slack penalty: bounded linear penalty for negative slack — monotonic and overflow-safe ===
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Composite score: smaller = higher priority; all components clipped to [-2,2] for bounded AST depth ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Calibrated successor-release boost
        - np.clip(coupled_rank, -2.0, 2.0)                                 # Smoothly amplified critical path
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Feasibility-gated energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Bounded anti-starvation
        + np.clip(duration_uncert_penalty, -2.0, 2.0)                      # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(slack_pressure * PARAMS["ddl_protection_gate_slope"], -2.0, 2.0)  # Direct pressure signal
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
