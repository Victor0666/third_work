def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's robustness with Parent 1's bounded anti-starvation,
       plus novel slack-margin sigmoid gate and starvation saturation clamp.
       Key improvements:
         - Replaces binary ddl_urgent with smooth sigmoid slack-margin gate
         - Adds starvation_saturation_limit to prevent anti-starvation from overriding deadline urgency
         - Uses normalized successor-release interaction (norm_rank * norm_work) for scale invariance
         - Retains load-aware energy modulation gated by both uncertainty and slack-margin urgency
         - Keeps median-MAD normalization for outlier resilience and sign preservation."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median + MAD per feature (preserves sign semantics and outlier resilience) ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Smooth slack-margin gate: sigmoid over slack_margin_ratio for early DDL protection ===
    # Activates gradually as slack becomes small relative to execution+comm time
    slack_margin_ratio = np.abs(slack) / (min_exec_time + min_comm_time + eps)
    slack_margin_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_margin_gate_steepness"] * 
                                              (slack_margin_ratio - PARAMS["slack_margin_ratio_threshold"])))
    
    # === Successor-release interaction: normalized product for scale-invariant impact ===
    successor_release_impact = norm_rank * norm_work
    
    # === Load-aware energy modulation: only penalize energy under high uncertainty AND slack pressure ===
    energy_modulation_gate = np.where(
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) & (slack_margin_gate > 0.5),
        1.0,
        0.0
    )
    
    # === Sharpened slack-pressure gate: sigmoid over normalized slack for smoother transition ===
    slack_pressure_norm = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_norm - 1.0)))
    
    # Boost rank under both margin pressure and slack pressure, capped for stability
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate * slack_margin_gate)
    
    # === Bounded anti-starvation: logistic with explicit saturation limit ===
    wait_benefit = (1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))) * PARAMS["starvation_saturation_limit"]
    
    # === Duration robustness: penalize long durations only under high uncertainty AND slack pressure ===
    duration_risk_score = norm_duration * norm_uncert * slack_margin_gate
    
    # === Energy-uncertainty interaction: multiplicative penalty, gated by slack-margin urgency ===
    energy_uncert_penalty = norm_energy * norm_uncert * energy_modulation_gate
    
    # === Composite score: smaller = higher priority ===
    # Prioritize DDL feasibility first (via slack/urgency terms), then energy, then starvation/work
    score = (
        + norm_slack                                  # Raw slack bias: lower slack → lower score (higher priority)
        + (norm_slack ** PARAMS["slack_penalty_exponent"]) * slack_margin_gate  # Nonlinear penalty only when margin-tight
        - boosted_rank                              # Critical path boost under pressure
        - successor_release_impact                  # Prefer tasks that unblock large downstream work
        - PARAMS["energy_sensitivity"] * norm_energy * energy_modulation_gate  # Energy only matters under risk
        - norm_duration * (1.0 - slack_margin_gate)  # Prefer short duration when margin is safe
        - wait_benefit                              # Anti-starvation bonus (saturated)
        + PARAMS["duration_robustness"] * duration_risk_score  # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned marginal energy penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
