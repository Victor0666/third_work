def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's robust median-MAD normalization and logistic starvation relief with Parent 1's explicit comm+uncert penalty;
       enforces strict DDL-first via hard-gated energy/uncert terms; all numeric literals are -2, -1, 0, 1, or 2."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    norm_comm = median_mad_normalize(min_comm_time)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: upward_rank * remaining_work interaction, activated only under deadline breach ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded slack-pressure gate: sigmoid over clipped normalized slack pressure ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    slack_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    
    # === Rank-slack coupling: single monotonic amplifier using slack_pressure_gate, bounded to [1.0, 2.5] ===
    rank_amplifier = 1.0 + PARAMS["criticality_scale"] * slack_pressure_gate
    rank_amplifier = np.clip(rank_amplifier, 1.0, 2.5)
    boosted_rank = norm_rank * rank_amplifier
    
    # === Duration risk score: exec+comm × uncertainty × slack pressure × DDL gate ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * uncert_gate * slack_pressure * ddl_gate
    
    # === Communication-slack urgency: low-comm tasks prioritized under pressure, gated by DDL feasibility ===
    comm_slack_urgency = norm_comm * slack_pressure * ddl_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus with saturation offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty coupling: gated by both uncertainty and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: linearly scaled penalty under pressure, gated by feasibility ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Unblock critical paths on breach
        - np.clip(boosted_rank, -2.0, 2.0)                                # Amplified HEFT urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy minimization only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation (logistic benefit)
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + np.clip(comm_slack_urgency, -2.0, 2.0)                          # Direct comm-slack urgency
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
