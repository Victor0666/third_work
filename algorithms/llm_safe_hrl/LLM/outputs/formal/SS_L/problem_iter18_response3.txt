def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces median-MAD with slack-aware min-max normalization to sharpen deadline-driven discrimination;
       introduces piecewise ddl_gate active only on [-δ, 0] to enable gradient-aware repair of near-violation states while preserving hard feasibility;
       adds explicit infeasible-mode successor-release boost (rank × work × (1−ddl_gate)) to combat critical-path starvation under violation;
       retains power-law slack pressure and robustness-adjusted duration penalty;
       all terms clipped to [-2,2] for bounded AST depth, CMA-ES stability, and deterministic output."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Slack-aware min-max normalization: tightens dynamic range for tasks near deadline violation ===
    # Normalizes each feature over [min, max], but shifts and scales relative to slack-feasibility boundaries
    def slack_aware_minmax_normalize(x):
        x = np.copy(x)
        if N == 1:
            return np.zeros_like(x)
        # Use global bounds but bias toward critical region: clamp outliers to [q1, q99] for stability
        q1, q99 = np.quantile(x, [0.01, 0.99])
        x_clipped = np.clip(x, q1, q99)
        x_min, x_max = np.min(x_clipped), np.max(x_clipped)
        spread = x_max - x_min if x_max > x_min + eps else eps
        return (x - x_min) / spread
    
    norm_slack = slack_aware_minmax_normalize(slack)
    norm_energy = slack_aware_minmax_normalize(min_incremental_energy)
    total_duration = min_exec_time + min_comm_time
    norm_duration = slack_aware_minmax_normalize(total_duration)
    norm_rank = slack_aware_minmax_normalize(upward_rank)
    norm_work = slack_aware_minmax_normalize(remaining_work)
    norm_wait = slack_aware_minmax_normalize(ready_wait_time)
    norm_uncert = slack_aware_minmax_normalize(uncertainty)
    
    # === Piecewise ddl_gate: 0 for slack < -δ, linear ramp from 0→1 over [-δ, 0], 1 for slack ≥ 0 ===
    # Enables gradient flow *only* in the critical repair band, preserving hard feasibility outside
    delta = PARAMS["feasibility_margin_delta"]
    ddl_gate = np.where(
        slack < -delta,
        0.0,
        np.where(
            slack <= 0.0,
            (slack + delta) / delta,  # linear ramp: 0 at -δ, 1 at 0
            1.0
        )
    )
    
    # === Power-law slack pressure: same as before, but now defined on raw slack for consistency with gate ===
    raw_slack_pressure = np.maximum(-slack, 0.0)
    slack_pressure = np.power(raw_slack_pressure + eps, PARAMS["slack_nonlinearity_exponent"])
    max_pressure = np.max(slack_pressure) if N > 1 else slack_pressure[0]
    ddl_pressure = np.where(max_pressure > eps, slack_pressure / (max_pressure + eps), 0.0)
    
    # === Successor-release interaction: split into two modes — feasible & infeasible ===
    # Feasible mode: rank × work × ddl_pressure (as before)
    # Infeasible mode: rank × work × (1 − ddl_gate) → boosts successors *only* when slack < -δ (i.e., fully infeasible)
    infeasible_mask = (slack < -delta).astype(float)
    feasible_successor_release = norm_rank * norm_work * ddl_pressure
    infeasible_successor_boost = norm_rank * norm_work * infeasible_mask * PARAMS["infeasible_successor_boost"]
    successor_release = feasible_successor_release + infeasible_successor_boost
    
    # === Linear slack penalty: bounded, monotonic, numerically stable ===
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = slack_aware_minmax_normalize(raw_slack_penalty)
    
    # === Critical-path coupling: upward_rank scaled by nonlinear pressure ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_pressure)
    
    # === Uncertainty-aware energy penalty: gated by both DDL feasibility band AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: linear ramp-up with explicit saturation ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Robustness-adjusted duration penalty: penalizes high-uncertainty *and* long durations under pressure ===
    robust_duration_penalty = norm_duration * norm_uncert * ddl_pressure * PARAMS["duration_robustness_weight"]
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for bounded AST depth and stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Dual-mode successor-release boost
        - np.clip(coupled_rank, -2.0, 2.0)                                 # Critical-path urgency with nonlinear coupling
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy only in repair band & feasible region
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + np.clip(robust_duration_penalty, -2.0, 2.0)                      # Uncertainty-weighted duration robustness
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(ddl_pressure * PARAMS["ddl_protection_gate_slope"], -2.0, 2.0)  # Direct nonlinear pressure signal
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
