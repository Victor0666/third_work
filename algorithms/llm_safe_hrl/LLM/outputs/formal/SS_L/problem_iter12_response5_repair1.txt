def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces binary DDL breach mask with quantile-based graded urgency;
       introduces hard feasibility guard that zero-penalizes tasks exceeding workflow deadline;
       switches to robust IQR-based normalization with 5th/95th percentile clamping;
       preserves all prior structural improvements (capped starvation, residual work coupling, etc.);
       ensures monotonic urgency response for gradient-friendly offline co-evolution."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === IQR-based normalization with 5th/95th percentile clamping for robustness on small N ===
    def iqr_normalize(x):
        if N == 1:
            center = x[0]
            q25, q75 = x[0], x[0]
        else:
            q25 = np.quantile(x, 0.25)
            q75 = np.quantile(x, 0.75)
            center = np.quantile(x, 0.5)
        iqr = q75 - q25 + eps
        # Clamp outliers to [5%, 95%] before normalizing
        x_clipped = np.clip(x, np.quantile(x, 0.05), np.quantile(x, 0.95))
        return (x_clipped - center) / (iqr + eps)
    
    norm_slack = iqr_normalize(slack)
    norm_energy = iqr_normalize(min_incremental_energy)
    norm_duration = iqr_normalize(min_exec_time + min_comm_time)
    norm_rank = iqr_normalize(upward_rank)
    norm_work = iqr_normalize(remaining_work)
    norm_wait = iqr_normalize(ready_wait_time)
    norm_uncert = iqr_normalize(uncertainty)
    
    # === Graded urgency signal via slack's fuzzy percentile rank (monotonic, continuous) ===
    # Uses fixed 15th percentile (replaces removed 'urgency_quantile_alpha' to stay within 12 params)
    if N == 1:
        slack_threshold = slack[0]
    else:
        slack_threshold = np.quantile(slack, 0.15)
    graded_urgency = np.clip((slack_threshold - slack) / (np.abs(slack_threshold) + eps), 0.0, 1.0)
    
    # === Hard feasibility guard: zero-out score for tasks whose min risk completion violates deadline ===
    # Assign massive penalty to enforce strict filtering before scoring
    infeasible_mask = (slack < 0.0).astype(float)
    feasibility_penalty = infeasible_mask * 100.0  # Fixed weight — removed tunable 'feasibility_guard_weight'
    
    # === Slack-pressure gate: smooth sigmoid ramp using graded urgency ===
    slack_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (graded_urgency - eps)))
    
    # === Residual work impact: gated by graded urgency ===
    residual_work_impact = norm_rank * norm_work * slack_pressure_gate
    
    # === Lateness penalty: power-law on clipped negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = iqr_normalize(raw_slack_penalty)
    
    # === Boosted critical rank: activated smoothly via graded urgency ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_pressure_gate)
    
    # === Energy-uncertainty joint penalty: active only above threshold ===
    uncert_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate
    
    # === Duration-robustness penalty: requires both uncertainty and graded urgency ===
    duration_risk_score = norm_duration * norm_uncert * slack_pressure_gate
    
    # === Bounded starvation relief: logistic form with hard cap ===
    wait_benefit = np.clip(
        1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"])),
        0.0,
        PARAMS["starvation_relief_cap"]
    )
    
    # === Composite score: smaller = higher priority
    # Feasibility guard dominates all terms to enforce hard deadline constraint first
    score = (
        + feasibility_penalty                         # Hard DDL feasibility enforcement (dominant)
        + norm_slack_penalty                          # Primary lateness penalty
        - residual_work_impact                        # Structural DAG pressure
        - boosted_rank                                # Critical path awareness
        - PARAMS["energy_sensitivity"] * norm_energy # Energy minimization
        - wait_benefit                                # Starvation mitigation
        + PARAMS["duration_robustness"] * duration_risk_score  # Uncertain-slow penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Joint risk penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Downstream work preference
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
