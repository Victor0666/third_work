def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: deadline-risk-gated criticality + starvation-aware efficiency.
    
    Fixes: uses all declared parameters, removes unused literals, ensures epsilon safety via np.finfo.
    Key logic:
    - Slack split: linear penalty for negative slack; sublinear urgency for positive slack.
    - Upward rank amplified only under deadline pressure OR high uncertainty (gated by uncertainty_adaptation).
    - Starvation relief uses tanh-saturated wait time, scaled by starvation_relief and bounded by max_wait_threshold.
    - All features robustly normalized using median/IQR with epsilon guard.
    - Energy term inverted: lower energy → higher priority → subtracted.
    - Uncertainty_adaptation now used in pressure gate and uncertainty scaling.
    """
    epsilon = PARAMS["epsilon"]
    N = len(slack)

    # Robust normalization: median + IQR, with epsilon guard
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(x)
        q75 = np.percentile(x, 75)
        q25 = np.percentile(x, 25)
        iqr = q75 - q25
        scale = np.where(iqr == 0, np.mean(np.abs(x - med)) + epsilon, iqr)
        return (x - med) / (scale + epsilon)

    # Normalize all inputs
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # --- Slack risk & urgency modeling ---
    # Negative slack: linear penalty (stronger the more negative)
    slack_penalty = np.where(slack < 0, 
                            PARAMS["slack_risk_penalty"] * (-norm_slack), 
                            0.0)
    # Positive slack: diminishing urgency (power law, capped at 1.0 contribution)
    slack_urgency = np.where(slack >= 0,
                            np.minimum(1.0, np.power(norm_slack + 1, PARAMS["slack_urgency_scale"])),
                            0.0)

    # --- Critical path leverage: amplify rank only under pressure ---
    # Gate: activated when slack is tight (norm_slack < 0) OR uncertainty is high (norm_uncert > 0.5)
    # Scaled by uncertainty_adaptation to tune sensitivity
    pressure_gate = np.where((norm_slack < 0) | (norm_uncert > 0.5),
                           PARAMS["uncertainty_adaptation"],
                           np.maximum(0.0, 1.0 - norm_uncert * PARAMS["uncertainty_adaptation"]))
    leveraged_rank = PARAMS["critical_path_leverage"] * norm_rank * pressure_gate

    # --- Starvation relief: saturating tanh boost, scaled and bounded ---
    # Scale wait time relative to threshold, then apply smooth saturation
    wait_ratio = np.clip(ready_wait_time / (PARAMS["max_wait_threshold"] + epsilon), 0, 5)
    starvation_boost = PARAMS["starvation_relief"] * np.tanh(wait_ratio)

    # --- Duration balance: prevent ultra-short tasks from dominating unless efficient ---
    duration_balance = PARAMS["duration_balance"] * norm_duration

    # --- Energy efficiency: lower energy → higher priority → subtract from score ---
    energy_benefit = -PARAMS["energy_efficiency_gain"] * norm_energy

    # --- Assemble final score: smaller = better ---
    # Core terms: deadline risk dominates, then criticality, then efficiency & fairness
    score = (
        slack_penalty                    # [↑ priority for negative slack]
        + (1.0 - slack_urgency)          # [↑ priority for small positive slack; 0 when slack large]
        + leveraged_rank                 # [↑ priority for critical paths under pressure]
        + duration_balance               # [moderate bias toward balanced duration]
        + energy_benefit                 # [↑ priority for low incremental energy]
        - starvation_boost               # [↑ priority for long-waiting tasks]
    )

    # Ensure finite output, shape (N,), deterministic
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    return score
