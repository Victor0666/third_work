def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: uses robust percentile-rank normalization (clipped to [low, high]);
       introduces successor-release interaction via norm_work * (1 - norm_slack) proxy;
       replaces global Q1/Q3 with rank-based scaling → stable under N=1/skew;
       uses task-specific feasibility mask based on slack + upward_rank margin;
       all numeric literals are -2,-1,0,1,2 or np.finfo safeguards."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust percentile-rank normalization: map x_i → percentile(x_i), clip to [low, high] ===
    def percentile_rank_normalize(x):
        x = np.copy(x)
        if N == 1:
            return np.array([0.5])
        sorted_x = np.sort(x)
        ranks = np.searchsorted(sorted_x, x, side='right')
        pct = ranks / N
        return np.clip(pct, PARAMS["rank_percentile_clip_low"], PARAMS["rank_percentile_clip_high"])
    
    # Normalize all features using percentile ranks
    norm_energy = percentile_rank_normalize(min_incremental_energy)
    norm_duration = percentile_rank_normalize(min_exec_time + min_comm_time)
    norm_rank = percentile_rank_normalize(upward_rank)
    norm_work = percentile_rank_normalize(remaining_work)
    norm_wait = percentile_rank_normalize(ready_wait_time)
    norm_uncert = percentile_rank_normalize(uncertainty)
    
    # Sign-preserving MAD normalization for slack
    if N == 1:
        slack_med = slack[0]
        slack_mad = eps
    else:
        slack_med = np.median(slack)
        slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = slack_mad if slack_mad > eps else eps
    norm_slack = (slack - slack_med) / slack_spread
    
    # === Task-specific feasibility mask: per-task slack relative to critical-path margin ===
    feasibility_mask = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (slack + eps * upward_rank)))
    
    # === Deadline breach indicator ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Critical path leverage: multiplicative urgency amplification ===
    critical_path_leverage = norm_rank * (1.0 + PARAMS["criticality_scale"] * np.abs(norm_slack)) * ddl_breach
    
    # === Raw slack penalty: power-law on negative slack, then percentile-normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = percentile_rank_normalize(raw_slack_penalty)
    
    # === Unified congestion signal: ready_wait_time + uncertainty, gated by DDL risk ===
    congestion_signal = norm_wait + norm_uncert
    congestion_score = congestion_signal * (1.0 - feasibility_mask)
    
    # === Starvation relief: linear-in-wait, capped ===
    wait_mean = np.mean(ready_wait_time)
    wait_denom = np.maximum(wait_mean, eps)
    wait_benefit = np.clip(ready_wait_time / wait_denom, 0.0, PARAMS["wait_benefit_cap"])
    
    # === Energy-uncertainty coupling: active only under both deadline breach AND high uncertainty ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_breach
    
    # === Successor-release interaction: penalize current task if its successors likely block progress ===
    # Proxy: high remaining_work * and * low slack → use norm_work * clipped(1 - norm_slack)
    successor_pressure_proxy = norm_work * np.clip(1.0 - norm_slack, 0.0, 1.0)
    successor_pressure_penalty = PARAMS["successor_pressure_weight"] * successor_pressure_proxy * ddl_breach
    
    # === Composite score: smaller = higher priority; each term clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Critical-path boost under breach
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * feasibility_mask, -2.0, 2.0)  # Energy minimization only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Capped anti-starvation
        + PARAMS["congestion_weight"] * np.clip(congestion_score, -2.0, 2.0)  # Congestion relief under DDL risk
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * feasibility_mask, -2.0, 2.0)  # Downstream work preference only when feasible
        + np.clip(norm_rank * (1.0 - feasibility_mask), -2.0, 2.0)        # Mild rank boost when slack > 0
        + np.clip(successor_pressure_penalty, -2.0, 2.0)                  # Successor-release pressure
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly
    return score.reshape(-1)
