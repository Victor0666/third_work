def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's robust hinge coupling and joint gate with Parent 1's congestion-aware load signal;
       replaces softplus starvation relief with *steepened softplus* to sharpen low-wait sensitivity while preserving saturation;
       introduces *congestion_coupling_strength* to explicitly weight duration-uncertainty co-penalty only under joint feasibility;
       retains median-MAD normalization for outlier resistance and sign stability;
       removes redundant rank_slack_coupling amplification (redundant with slack_pressure_gate) to reduce AST depth;
       enforces strict clipping at [-2,2] on all composite terms to bound gradient magnitude and prevent dominance collapse."""
    eps = PARAMS["epsilon"]
    
    # Safe cast; no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — outlier-resistant and sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Unified joint gate: active only when slack >= 0 AND normalized uncertainty <= threshold ===
    # Combines feasibility and risk into one clean switch
    joint_gate = (
        1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack)) *
        1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    )
    
    # === DDL breach indicator for critical path leverage (validated diagnostic) ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Robust slack penalty: capped power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate for rank boosting: centered at pressure=1.0 for better dynamic range ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Steepened softplus starvation relief: sharper low-wait discrimination, same saturation ===
    # log1p(exp(k*x)) → k controls steepness; avoids overflow, preserves monotonicity
    starvation_signal = PARAMS["wait_decay"] * (ready_wait_time + eps)
    wait_benefit = np.log1p(np.exp(PARAMS["starvation_relief_steepness"] * starvation_signal))
    
    # === Congestion-aware load signal: duration × uncertainty interaction, gated jointly ===
    # From Parent 1, refined with new coupling strength and joint gating
    congestion_coupling = PARAMS["congestion_coupling_strength"] * norm_duration * norm_uncert * joint_gate
    
    # === Energy-uncertainty penalty: pure interaction under joint safety ===
    energy_uncert_penalty = norm_energy * norm_uncert * joint_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] to enforce bounded influence ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Critical-path unblocking boost
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency amplification
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * joint_gate, -2.0, 2.0)  # Energy preference only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief (steepened softplus)
        + np.clip(congestion_coupling, -2.0, 2.0)                          # Load-congestion signal
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * joint_gate, -2.0, 2.0)  # Downstream work bonus only when safe
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
