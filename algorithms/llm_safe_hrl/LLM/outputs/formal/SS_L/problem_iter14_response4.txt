def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: hybridizes Parent 2's stability with Parent 1's safe-slack gating.
       Novel structural change: dual-gate energy suppression — (i) hard DDL-protection gate (slack < 0 → disable), 
       and (ii) soft safe-slack gate (norm_slack > threshold → suppress), enabling robust feasibility-first behavior
       across all slack regimes. Introduces sigmoid-coupled successor-release interaction for sharper critical-path activation.
       Replaces linear ddl_gate with clipped sigmoid on normalized slack to improve gradient continuity while avoiding overflow."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Dual DDL gating: hard protection (slack < 0) + soft suppression (safe slack) ===
    # Hard gate: zero energy/uncertainty impact when slack < 0 (strict feasibility first)
    hard_ddl_gate = np.where(slack < 0.0, 0.0, 1.0)
    # Soft gate: suppress energy terms when normalized slack exceeds safe threshold
    safe_slack_mask = (norm_slack > PARAMS["safe_slack_energy_disable_threshold"])
    soft_energy_suppress = np.where(safe_slack_mask, 0.0, 1.0)
    # Combined energy enable mask: active only under deadline pressure AND not in safe regime
    energy_enabled = hard_ddl_gate * soft_energy_suppress
    
    # === Sigmoid-based slack pressure signal: smoother than linear ramp, bounded, monotonic ===
    # Uses normalized slack for consistent scaling across workloads
    slack_pressure_norm = np.clip(-norm_slack, 0.0, 2.0)
    slack_pressure = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_steepness"] * (slack_pressure_norm - 1.0)))
    
    # === Successor-release interaction: sigmoid-coupled rank × work to sharpen activation near deadlines ===
    # Captures release-criticality of high-work successors only under tightening slack
    successor_release = norm_rank * norm_work * slack_pressure
    
    # === Linear slack penalty: bounded linear penalty for negative slack → numerically stable ===
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Critical-path coupling: upward_rank scaled by sigmoid slack pressure (more stable than power law) ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_pressure)
    
    # === Uncertainty-aware energy penalty: gated by both DDL feasibility AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * energy_enabled
    
    # === Starvation relief: linear ramp-up with explicit saturation at 1.0 ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Duration penalty: only under deadline pressure and feasibility ===
    exec_penalty = norm_duration * slack_pressure * hard_ddl_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for bounded AST depth ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Sharpened successor-release boost
        - np.clip(coupled_rank, -2.0, 2.0)                                 # Critical-path urgency with sigmoid coupling
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * energy_enabled, -2.0, 2.0)  # Dual-gated energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + np.clip(exec_penalty, -2.0, 2.0)                                 # Duration penalty under pressure
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(slack_pressure * PARAMS["ddl_protection_gate_slope"], -2.0, 2.0)  # Direct pressure signal
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
