def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid-based gates with robust clipped-MAD slack gating;
       integrates successor-release interaction via min_exec_time × upward_rank × (slack <= 0);
       removes inactive 'duration_robustness' and 'wait_saturation_offset';
       uses MAD-normalized slack [-2,2] as primary urgency signal;
       applies strict feasibility-gating (slack >= 0) to all energy/uncertainty terms;
       leverages validated 'add_conditional_ddl_protection_gate' and 'add_upward_rank_remaining_work_interaction';
       avoids unbounded ops, ensures finite output and shape (N,)."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — sign-preserving, outlier-resistant ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    # Normalize key features
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_exec = median_mad_normalize(min_exec_time)
    norm_comm = median_mad_normalize(min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Primary urgency signal: MAD-normalized slack clipped to [-2, 2] — sharper & feasibility-first ===
    urgency = np.clip(norm_slack, -2.0, 2.0)
    
    # === DDL-protection gate: enables energy/uncertainty only when slack >= 0 (hard constraint alignment) ===
    ddl_feasible = (slack >= 0.0).astype(float)
    
    # === Critical-path leverage: upward_rank × remaining_work × (slack <= 0) — validated by DDL diagnostics ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_leverage = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: unblocks fast-executing critical-path blockers — validated across ≥5 consensus failures ===
    successor_release = norm_exec * norm_rank * ddl_breach
    
    # === Energy penalty: only active under feasibility (slack >= 0), scaled by sensitivity ===
    energy_penalty = PARAMS["energy_sensitivity"] * norm_energy * ddl_feasible
    
    # === Uncertainty-energy coupling: gated by both feasibility and uncertainty threshold ===
    uncert_gate = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_penalty = PARAMS["energy_uncertainty_interaction"] * norm_energy * norm_uncert * ddl_feasible * uncert_gate
    
    # === Starvation relief: exponential decay on normalized wait time — no saturation offset (inactive per diagnostics) ===
    wait_relief = 1.0 - np.exp(-PARAMS["wait_decay"] * norm_wait)
    
    # === Slack pressure modulation: steepened linear ramp from -2→0, then flat — avoids sigmoid instability ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    pressure_mod = 1.0 + PARAMS["slack_pressure_gate_steepness"] * slack_pressure
    
    # === Feasibility-weighted rank amplification: upward_rank × pressure_mod × ddl_breach ===
    rank_boost = norm_rank * pressure_mod * ddl_breach
    
    # === Composite score: smaller = higher priority; all terms clipped for stability ===
    score = (
        + urgency                                        # Primary urgency (MAD-slack, clipped)
        - np.clip(critical_leverage, -2.0, 2.0)        # Critical path leverage under breach
        - np.clip(successor_release, -2.0, 2.0)        # Successor-release interaction
        + np.clip(rank_boost, -2.0, 2.0)               # Pressure-modulated rank boost under breach
        + energy_penalty                               # Energy penalty only when feasible
        + energy_uncert_penalty                        # Risk-aware energy-uncertainty coupling
        - np.clip(wait_relief, -2.0, 2.0)             # Anti-starvation relief
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)  # Downstream work preference
    )
    
    # Final safeguards: replace NaN/inf, ensure finite values and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce explicit (N,) shape — required for N=1 case
    return score.reshape(-1)
