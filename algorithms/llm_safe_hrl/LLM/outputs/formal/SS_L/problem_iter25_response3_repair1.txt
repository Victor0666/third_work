def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: uses robust range-based normalization;
       introduces successor-release interaction: min_exec_time × clipped_upward_rank × (slack <= 0);
       replaces sigmoid DDL gate with hard binary gating for strict feasibility-first enforcement;
       removes unused parameters (ddl_protection_gate_slope, slack_pressure_gate_steepness);
       adds sign-preserving MAD-normalized slack for sharper urgency near zero;
       applies joint congestion gating only under both high ready_wait_time AND high uncertainty;
       enforces energy/uncertainty penalties exclusively when slack >= 0 (feasible regime)."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust range-based normalization: uses interquartile range + epsilon for stability ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q25, q75 = x[0], x[0]
            rng = eps
        else:
            q25 = np.quantile(x, 0.25)
            q75 = np.quantile(x, 0.75)
            rng = q75 - q25
        spread = rng if rng > eps else eps
        center = (q25 + q75) / 2.0
        return (x - center) / spread
    
    norm_slack = robust_range_normalize(slack)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_exec = robust_range_normalize(min_exec_time)
    norm_comm = robust_range_normalize(min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # === Hard DDL feasibility gate: binary activation — energy/uncertainty terms only when slack >= 0 ===
    ddl_feasible = (slack >= 0.0).astype(float)
    
    # === Sign-preserving MAD-normalized slack for sharper urgency near slack ≈ 0 ===
    # Uses absolute deviation from median to preserve sign while enhancing sensitivity
    if N == 1:
        med_slack = slack[0]
        mad_slack = eps
    else:
        med_slack = np.median(slack)
        mad_slack = np.median(np.abs(slack - med_slack))
    spread_slack = mad_slack if mad_slack > eps else eps
    signed_slack_norm = (slack - med_slack) / spread_slack
    
    # === Critical path leverage: upward_rank × remaining_work × (slack <= 0) — validated DDL pressure term ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_leverage = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: fast-executing blockers of critical descendants ===
    # min_exec_time × clipped_upward_rank × (slack <= 0) — targets release of critical successors
    clipped_rank = np.clip(norm_rank, 0.0, 2.0)
    successor_release = norm_exec * clipped_rank * ddl_breach
    
    # === Joint congestion gating: active only under both high wait AND high uncertainty ===
    wait_high = (norm_wait > PARAMS["uncertainty_gate_threshold"]).astype(float)
    uncert_high = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    congestion_gate = wait_high * uncert_high
    
    # === Starvation relief: logistic decay with saturation offset, scaled by congestion_gate to avoid biasing under load ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]))
    wait_benefit = wait_benefit * (1.0 - congestion_gate)  # suppress anti-starvation when congested
    
    # === Energy-uncertainty penalty: only applied in DDL-feasible region and gated by uncertainty threshold ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_feasible
    
    # === Slack-aware marginal energy penalty: amplifies energy cost when slack is tight but still feasible ===
    # Uses signed_slack_norm to smoothly increase weight as slack approaches zero from positive side
    slack_tightness = np.clip(1.0 - signed_slack_norm, 0.0, 2.0)
    energy_slack_penalty = norm_energy * slack_tightness * ddl_feasible
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for numerical stability ===
    score = (
        + np.clip(signed_slack_norm, -2.0, 2.0)                           # Primary urgency signal, sharpened near zero
        - np.clip(critical_leverage, -2.0, 2.0)                          # Critical path blocking under breach
        - np.clip(successor_release, -2.0, 2.0)                          # Fast blocker of critical successors
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_feasible, -2.0, 2.0)  # Feasible-region energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                               # Anti-starvation, suppressed under congestion
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + PARAMS["criticality_scale"] * np.clip(norm_rank * ddl_breach, -2.0, 2.0)  # Direct criticality boost under breach
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
