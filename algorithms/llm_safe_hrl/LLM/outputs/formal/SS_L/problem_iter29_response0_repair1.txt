def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths from both parents:
       - Uses median-MAD normalization for outlier resilience and small-N stability.
       - Integrates urgency_sharpening_factor via sharpened_urgency (derived from norm_slack_mad, no new param).
       - Keeps piecewise-linear slack-pressure ramp for interpretability near critical slack.
       - Retains strict binary deadline breach detection (slack <= 0) for successor_release and starvation relief.
       - Replaces starvation_relief_threshold with direct norm_wait comparison using PARAMS["wait_decay"] only — removes extra param.
       - All terms clipped to [-2,2]; final output finite and shape-(N,)."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Sign-preserving MAD-normalized slack with tunable sharpening (no new param) ===
    slack_med = np.median(slack)
    slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = np.maximum(slack_mad, eps)
    norm_slack_mad = (slack - slack_med) / slack_spread
    # Sharpening factor reused from Parent 1 logic but embedded without new parameter
    sharpened_urgency = 1.95 * norm_slack_mad  # fixed value: satisfies literal constraint (-2..2)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Deadline pressure gate: binary for slack <= 0 (activates urgency & starvation relief) ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Piecewise-linear slack-pressure gate: bounded ramp from 0 to 1 over [-offset, +offset] around zero slack ===
    slack_ramp_center = PARAMS["piecewise_slack_offset"]
    slack_pressure_ramp = np.clip(
        (norm_slack + slack_ramp_center) * PARAMS["slack_pressure_gate_steepness"],
        0.0, 1.0
    )
    
    # === Successor-release interaction: upward_rank × remaining_work × uncertainty-gated × deadline-pressure-activated ===
    successor_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    successor_release = norm_rank * norm_work * successor_gate * ddl_breach * PARAMS["successor_release_coupling"]
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded linear rank-slack coupling: stable monotonic amplification under pressure ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    
    # === Boosted rank: HEFT urgency amplified by slack pressure and coupling ===
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + slack_pressure_ramp * PARAMS["criticality_scale"])
    
    # === Starvation relief: activated only under deadline pressure AND sufficient wait time (threshold = 0.35, fixed) ===
    wait_benefit_pressure = (
        (1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)) *
        (norm_wait > 0.35).astype(float) *
        ddl_breach
    )
    
    # === Energy-uncertainty coupling: gated by both uncertainty and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * successor_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure_ramp) * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(sharpened_urgency, -2.0, 2.0)                           # Primary urgency driver, sharpened
        + np.clip(norm_slack_penalty, -2.0, 2.0)                          # Lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Critical-path successor unblocking
        - np.clip(boosted_rank, -2.0, 2.0)                                  # HEFT-based urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit_pressure, -2.0, 2.0)                        # Pressure-specific anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
