def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: integrates communication-awareness, refines slack-pressure coupling via dual-gated rank modulation,
       replaces MAD with quantile-based robust normalization for better small-N stability, and adds explicit slack-feasibility gating
       to all non-urgency terms to enforce hard deadline compliance while preserving energy optimization within feasible region."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Quantile-based robust normalization: more stable than MAD for small N ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        if N == 1:
            center = x_abs[0]
            spread = eps
        else:
            q25, q50, q75 = np.quantile(x_abs, [0.25, 0.5, 0.75])
            iqr = q75 - q25 + eps
            center = q50
            spread = iqr
        return (x_abs - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_comm = robust_normalize(min_comm_time)  # Explicit comm-time feature
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack penalty: normalized power penalty only on overdue tasks ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Dual-gated rank modulation: combines linear slack coupling AND sigmoid pressure gate ===
    # Ensures monotonic rank boost under pressure without over-amplification
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    linear_boost = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    sigmoid_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    rank_modulation = linear_boost * (1.0 + sigmoid_gate)
    boosted_rank = norm_rank * rank_modulation
    
    # === DDL-protection gate: smooth Heaviside enabling *all non-urgency* terms only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Uncertainty-duration-comm interaction: activated only under dual pressure AND slack feasibility ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * uncert_gate * slack_pressure * ddl_gate
    comm_risk_score = norm_comm * uncert_gate * slack_pressure * ddl_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus with saturation offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty interaction: gated by DDL feasibility → only active when slack >= 0 ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Communication-aware penalty: explicitly penalizes high-communication tasks under uncertainty & pressure ===
    comm_penalty = PARAMS["communication_penalty_weight"] * comm_risk_score
    
    # === Decoupled duration weight: restores flexibility lost in fixed 0.5-weight average ===
    duration_preference = PARAMS["duration_robustness"] * norm_duration * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] to prevent explosion ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness risk penalty
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Critical-path boost under dual-gated pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - duration_preference                                                # Decoupled, tunable duration preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Always active anti-starvation
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + comm_penalty                                                     # Novel communication-aware penalty
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
