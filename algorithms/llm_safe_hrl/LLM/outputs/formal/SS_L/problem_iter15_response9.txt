def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces fragile median-MAD normalization of slack with bounded slack-specific normalization
       using percentile-based centering and clipped MAD, then applies additive feasibility bias ('slack_normalized_mad_shift').
       Removes all redundant rank interactions (e.g., un-gated norm_work*norm_rank) to prevent signal cancellation.
       Introduces strict slack-monotonic base term: lower raw slack always yields lower score — preserving hard-DDL dominance.
       All energy terms remain fully gated by safe_slack_mask and ddl_urgent, ensuring zero energy influence when slack > threshold."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Bounded slack-specific normalization: percentile center + clipped MAD → preserves urgency ordering ===
    # Uses 25th percentile (not median) for left-skewed slack distributions; clips spread to avoid collapse
    slack_center = np.percentile(slack, 25) if N > 1 else slack[0]
    slack_abs_dev = np.abs(slack - slack_center)
    slack_spread = np.clip(np.percentile(slack_abs_dev, 75) if N > 1 else slack_abs_dev[0], eps, np.inf)
    norm_slack = (slack - slack_center) / (slack_spread + eps)
    
    # Apply additive feasibility bias: pushes positive-slack tasks *up* (worse score), negative ones *down* (better score)
    norm_slack_biased = norm_slack + PARAMS["slack_normalized_mad_shift"] * np.sign(norm_slack)
    
    # === Other features retain robust median-MAD normalization ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Hard feasibility gating: binary urgency flag based on raw slack and margin ratio ===
    slack_margin_ratio = np.abs(slack) / (min_exec_time + min_comm_time + eps)
    ddl_urgent = np.where((slack <= 0.0) | (slack_margin_ratio < PARAMS["safe_slack_energy_disable_threshold"]), 1.0, 0.0)
    
    # === Strict energy disable mask: energy terms fully zeroed when slack is sufficiently safe ===
    safe_slack_mask = (norm_slack > PARAMS["safe_slack_energy_disable_threshold"])
    energy_enabled = np.where(safe_slack_mask, 0.0, 1.0)
    
    # === Sharpened slack-pressure gate: sigmoid over *biased* normalized slack for smoother, more reliable activation ===
    slack_pressure_norm = np.clip(-norm_slack_biased, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_norm - 1.0)))
    
    # Boost rank only under urgency and pressure — no unconditional coupling
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate * ddl_urgent)
    
    # === Starvation relief: logistic with explicit saturation, using biased norm_wait ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + eps))
    
    # === Energy modulation: active only under both high uncertainty AND urgency ===
    energy_modulation_gate = np.where(
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) & (ddl_urgent == 1.0),
        1.0,
        0.0
    )
    
    # === Duration robustness: penalize long durations only under uncertainty + urgency ===
    duration_risk_score = norm_duration * norm_uncert * ddl_urgent
    
    # === Energy-uncertainty interaction: gated by both safety and urgency ===
    energy_uncert_penalty = norm_energy * norm_uncert * energy_modulation_gate * energy_enabled
    
    # === Composite score: smaller = higher priority ===
    # Primary term: raw slack bias ensures monotonic deadline adherence
    # All secondary terms strictly gated — no unbounded or un-gated contributions
    score = (
        + np.clip(norm_slack_biased, -2.0, 2.0)  # Feasibility-first anchor: lower slack → lower score
        + np.clip(norm_slack ** PARAMS["slack_penalty_exponent"], -2.0, 2.0) * ddl_urgent  # Nonlinear lateness penalty only when urgent
        - np.clip(boosted_rank, -2.0, 2.0)  # Critical path boost, sharply gated
        - np.clip(norm_work * ddl_urgent, -2.0, 2.0)  # Downstream work preference only under urgency (no rank coupling)
        - np.clip(norm_energy * PARAMS["energy_sensitivity"] * energy_modulation_gate * energy_enabled, -2.0, 2.0)  # Hard-gated energy penalty
        - np.clip(norm_duration * (1.0 - ddl_urgent), -2.0, 2.0)  # Soft short-duration preference when not urgent
        - np.clip(wait_benefit, -2.0, 2.0)  # Anti-starvation bonus
        + np.clip(PARAMS["duration_robustness"] * duration_risk_score, -2.0, 2.0)  # Risk-aware duration penalty
        + np.clip(PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty, -2.0, 2.0)  # Risk-aligned marginal energy penalty
        + np.clip(PARAMS["remaining_work_weight"] * norm_work, -2.0, 2.0)  # Mild preference for large downstream work, ungated but bounded
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
