def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: restored overdue-nonlinear urgency; strict DDL-gated successor coupling;
       sign-preserving MAD normalization; rank-slack multiplicative coupling; all 13 params used;
       no literals beyond [-2,-1,0,1,2]."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Sign-preserving MAD normalization (robust to outliers & preserves direction) ===
    def sign_preserving_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = np.maximum(mad, eps)
        return (x - med) / spread
    
    norm_slack = sign_preserving_mad_normalize(slack)
    norm_energy = sign_preserving_mad_normalize(min_incremental_energy)
    norm_duration = sign_preserving_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = sign_preserving_mad_normalize(upward_rank)
    norm_work = sign_preserving_mad_normalize(remaining_work)
    norm_wait = sign_preserving_mad_normalize(ready_wait_time)
    norm_uncert = sign_preserving_mad_normalize(uncertainty)
    
    # === DDL-protection gate: only enable energy/uncertainty terms when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Overdue-aware urgency gate: sigmoid on raw negative slack — preserves scale fidelity ===
    overdue_slack = np.clip(-slack, 0.0, np.inf)
    if N == 1:
        urgency_center = overdue_slack[0]
        urgency_spread = eps
    else:
        urgency_center = np.median(overdue_slack)
        urgency_spread = np.median(np.abs(overdue_slack - urgency_center)) + eps
    norm_overdue = (overdue_slack - urgency_center) / (urgency_spread + eps)
    urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (norm_overdue - 0.5)))
    
    # === Multiplicative rank-slack coupling: replaces additive boosting → tighter urgency alignment ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * urgency_gate)
    
    # === Strict DDL-gated successor coupling: only active when slack >= 0 (feasibility-first) ===
    successor_coupling = norm_work * norm_rank * ddl_gate
    
    # === Starvation relief: logistic anti-starvation bonus, bounded to [0, upper_bound] ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    wait_benefit = np.clip(wait_benefit, 0.0, PARAMS["wait_benefit_upper_bound"])
    
    # === Uncertainty gating: activates only above threshold, and only under feasibility ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * uncert_gate * ddl_gate
    
    # === Energy-uncertainty interaction: only penalized when feasible AND uncertain ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Tight-slack energy amplification: penalizes energy more when slack is small-but-positive ===
    max_abs_slack = np.max(np.abs(slack)) + eps
    tight_slack_ratio = np.clip(np.maximum(0.0, slack) / max_abs_slack, 0.0, 1.0)
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * (1.0 - tight_slack_ratio)) * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"], -2.0, 2.0)  # Raw overdue penalty (restored power-law)
        - np.clip(coupled_rank, -2.0, 2.0)                                                 # Urgency-coupled rank
        - PARAMS["remaining_work_weight"] * np.clip(successor_coupling, -2.0, 2.0)         # Feasibility-gated coupling
        - PARAMS["energy_sensitivity"] * np.clip(energy_slack_penalty, -2.0, 2.0)          # Slack-aware energy
        - np.clip(wait_benefit, -2.0, 2.0)                                                  # Anti-starvation
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)         # Robust duration risk
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)  # Risk-aware energy
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)                 # Downstream work preference
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
