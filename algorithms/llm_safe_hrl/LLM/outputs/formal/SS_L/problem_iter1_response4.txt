def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing risk-aware urgency, critical-path acceleration under deadline pressure,
    and robust starvation control. Uses piecewise-normalized, bounded interactions — no unbounded ops."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: use median absolute deviation (MAD) + epsilon for outlier resilience ===
    def robust_normalize(x):
        x = np.abs(x)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return x / (spread + eps)
    
    # Normalize base features — all non-negative after abs()
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack penalty: strong nonlinear push for negative slack ===
    # Use smooth, bounded penalty: (max(-slack, 0))^p → penalizes lateness aggressively but finitely
    slack_penalty = np.power(np.maximum(-slack, 0) + eps, PARAMS["slack_penalty_exponent"])
    # Normalize penalty to match scale of other terms
    slack_penalty = robust_normalize(slack_penalty)
    
    # === Critical-path acceleration gate: amplify upward_rank *only* when slack is low or negative ===
    # Gate: soft step using sigmoid-like bounded activation (no exp overflow)
    slack_pressure_level = np.clip(-norm_slack, 0, 2)  # maps negative slack → [0,2], positive → 0
    rank_amplifier = 1.0 + PARAMS["criticality_scale"] * np.tanh(slack_pressure_level)
    
    # === Uncertainty-aware duration weighting: only activate if uncertainty exceeds threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"],
                          norm_uncert,
                          np.zeros_like(norm_uncert))
    duration_risk_adjusted = norm_duration * (1.0 + PARAMS["duration_robustness"] * uncert_gate)
    
    # === Starvation control: saturating anti-starvation bonus (logistic-like), decays smoothly ===
    # Ensures long-waiting tasks get bounded boost, avoids unbounded growth
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * norm_wait)
    
    # === Coupled criticality term: upward_rank * slack_pressure, weighted and clipped ===
    coupled_criticality = np.clip(norm_rank * slack_pressure_level,
                                0, 2) * PARAMS["rank_slack_coupling"]
    
    # === Final score: smaller = higher priority ===
    # Dominant terms: slack penalty (urgent), coupled criticality (DDL-constrained acceleration),
    #                 energy (secondary objective), duration risk, starvation relief (bonus → subtract)
    score = (
        1.0 * slack_penalty                    # High priority for at-risk tasks
        + 0.8 * coupled_criticality           # Accelerate critical paths *only* when DDL pressure exists
        + PARAMS["energy_sensitivity"] * norm_energy
        + 0.5 * duration_risk_adjusted
        - 0.3 * wait_benefit                  # Bonus → lower score for waiting tasks
        - 0.2 * norm_work                     # Prefer tasks with less downstream load (reduces cascade risk)
    )
    
    # Ensure finite output: replace NaN/inf with safe large value (low priority)
    score = np.nan_to_num(score, nan=np.finfo(float).max, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Final safeguard: ensure shape is (N,) and all values finite
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values in priority score"
    
    return score
