def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's sharper sigmoid gate and energy-uncertainty interaction with Parent 1's smooth uncertainty activation and energy-slack coupling. Replaces brittle binary gates with continuous, differentiable responses for better CMA-ES optimization. Adds robust slack-aware energy scaling and unified risk-duration interaction."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median-centered MAD + epsilon → resilient to outliers and zero-scales ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        if N == 1:
            center = x_abs[0]
            spread = eps
        else:
            center = np.median(x_abs)
            spread = np.median(np.abs(x_abs - center))
        return (x_abs - center) / (spread + eps)
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack pressure: clipped power penalty for overdue tasks only ===
    slack_pressure_raw = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_pressure = robust_normalize(slack_pressure_raw)
    
    # === Sharpened slack-pressure gate: sigmoid with tunable steepness for precise critical-path activation ===
    slack_pressure_bounded = np.clip(-norm_slack, 0.0, 2.0)  # bounded [0,2]
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_bounded - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Soft uncertainty activation: graded sigmoidal response instead of binary AND-gate ===
    # Enables partial activation under moderate uncertainty + slack pressure → improves gradient continuity
    uncertainty_activation = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_smoothness"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    
    # === Duration-risk interaction: soft coupling of duration, uncertainty, and slack pressure ===
    duration_risk_interaction = norm_duration * uncertainty_activation * norm_slack_pressure * PARAMS["duration_robustness"]
    
    # === Starvation relief: saturating exponential form — superior gradient behavior vs tanh ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Energy-slack interaction: penalizes high marginal energy more under tight slack ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["energy_slack_interaction"] * norm_slack_pressure)
    
    # === Novel energy-uncertainty interaction: multiplicative penalty for high-risk-high-energy tasks ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncertainty_activation * PARAMS["energy_uncertainty_interaction"]
    
    # === Composite score: smaller = higher priority ===
    # Slack pressure dominates; energy cost amplified by slack and uncertainty; criticality rewarded adaptively;
    # starvation mitigated; risk-duration penalized softly; downstream work gently encouraged
    score = (
        + norm_slack_pressure                          # Urgency driver (primary)
        + energy_slack_penalty                         # Deadline-amplified energy cost
        + PARAMS["energy_sensitivity"] * norm_energy  # Base energy penalty
        - boosted_rank                                 # Critical-path reward under pressure
        - wait_benefit                                 # Anti-starvation reward
        + duration_risk_interaction                    # Graded risk-duration penalty
        + energy_uncert_penalty                        # Risk-aligned energy penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output with strict shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(-1)
