def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid gates with bounded linear interpolation for numerical stability;
       introduces successor-release interaction via norm_work * norm_rank under DDL pressure;
       removes redundant duration_robustness and wait_saturation_offset per evidence of inactivity;
       uses clipped linear slack scaling instead of power law to avoid overflow and preserve monotonicity;
       applies strict DDL-protection gating to all energy/uncertainty terms, validated by counterfactuals."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: binary-like smooth step (clipped linear) — avoids sigmoid overflow, preserves monotonicity ===
    # Gate = 0 when slack < 0, ramps linearly to 1 over [0, 1] sec slack margin
    ddl_gate = np.clip(slack, 0.0, 1.0)
    ddl_gate = np.where(slack < 0.0, 0.0, ddl_gate)  # Explicit zero for negative slack
    
    # === Successor-release interaction: product of normalized rank and work, activated only under deadline pressure ===
    # Validated by DDL diagnostics as 'successor_release_interaction'; captures release-criticality of large downstream impact
    ddl_pressure = np.clip(-norm_slack, 0.0, 1.0)  # 0→1 as slack tightens
    successor_release = norm_rank * norm_work * ddl_pressure
    
    # === Linear slack penalty: bounded linear penalty for negative slack, no exponentiation → numerically stable ===
    # Replaces unstable power law; maintains monotonic priority boost for tighter slack
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Critical-path coupling: upward_rank scaled by slack pressure, not exponentiated or sigmoid-gated ===
    # Simpler, more stable than prior rank_slack_coupling + rank_gate combo; matches 'criticality_scale' semantics
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_pressure)
    
    # === Uncertainty-aware energy penalty: gated by both DDL feasibility AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: linear ramp-up with saturation at max benefit = 1.0 ===
    # Replaces exp-based form; avoids log/exp near-zero issues; simpler and empirically sufficient
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Slack-aware execution time penalty: only when slack is tight AND feasible ===
    exec_penalty = norm_duration * ddl_pressure * ddl_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for bounded AST depth ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # DDL-diagnostic validated successor-release boost
        - np.clip(coupled_rank, -2.0, 2.0)                                 # Critical-path urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + np.clip(exec_penalty, -2.0, 2.0)                                 # Duration penalty under DDL pressure
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(ddl_pressure * PARAMS["ddl_protection_gate_slope"], -2.0, 2.0)  # Direct pressure signal
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
