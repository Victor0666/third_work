def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: removes unused ddl_protection_gate_slope;
       retains successor-release interaction, clipped-MAD slack urgency, DDL-feasibility gating;
       uses only declared parameters; all numeric literals are -2, -1, 0, 1, or 2."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    # Normalize core signals
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_exec = median_mad_normalize(min_exec_time)
    norm_comm = median_mad_normalize(min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Primary urgency signal: clipped-MAD slack, bounded to [-2, 2] — sharp, feasibility-first ===
    urgency_slack = np.clip(norm_slack, -2.0, 2.0)
    
    # === DDL-feasibility gate: binary but numerically stable (1 where slack >= 0, 0 otherwise) ===
    ddl_feasible = (slack >= 0.0).astype(float)
    
    # === Critical path release interaction: unblocks fast-executing critical successors ===
    successor_release = norm_exec * norm_rank * (slack <= 0.0).astype(float)
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then robustly normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Upward-rank × remaining-work interaction under deadline breach ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_leverage = norm_rank * norm_work * ddl_breach
    
    # === Starvation relief: linear wait benefit scaled by decay ===
    wait_benefit = np.clip(ready_wait_time * PARAMS["wait_decay"], 0.0, 2.0)
    
    # === Energy penalty: only applied when DDL feasible, scaled by sensitivity ===
    energy_penalty = PARAMS["energy_sensitivity"] * norm_energy * ddl_feasible
    
    # === Uncertainty-energy coupling: active only under both uncertainty threshold AND DDL feasibility ===
    uncert_active = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_penalty = PARAMS["energy_uncertainty_interaction"] * norm_energy * norm_uncert * ddl_feasible * uncert_active
    
    # === Slack-pressure modulation: soft gate for rank amplification when slack tightens ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_amplifier = 1.0 + PARAMS["criticality_scale"] * (1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0))))
    
    # === Composite score: smaller = higher priority ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                    # Lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                   # Unblock fast critical successors
        - np.clip(critical_leverage, -2.0, 2.0)                   # Critical path work leverage
        - np.clip(urgency_slack, -2.0, 2.0)                       # Direct urgency signal
        - np.clip(wait_benefit, -2.0, 2.0)                        # Anti-starvation
        + energy_penalty                                          # Energy minimization only when feasible
        + energy_uncert_penalty                                   # Joint risk-energy penalty
        - np.clip(norm_rank * rank_amplifier, -2.0, 2.0)         # Amplified rank under pressure
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)  # Downstream work preference
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
