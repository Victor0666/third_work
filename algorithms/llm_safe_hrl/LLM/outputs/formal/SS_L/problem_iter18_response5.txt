def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural advances:
       1. Contextual robust scaling: uses IQR (not MAD) for uncertainty & energy, scaled by iqr_scale_factor for statistical equivalence.
       2. Bounded min-max duration sensitivity gate: replaces linear duration penalty with saturating sigmoid of (exec+comm)/(|slack|+eps), 
          suppressing duration influence near tight deadlines — directly addresses reflection's weakness.
       3. Feasibility-preserving slack-aware normalization: upward_rank and remaining_work are rescaled *only* when slack > 0, 
          preserving critical-path importance under deadline pressure without distortion.
       All normalizations are sign-preserving; all outputs clipped to [-2,2] for AST depth control and stability."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust contextual normalization using IQR for uncertainty and energy ===
    # IQR provides better outlier isolation than MAD for risk-dominant features
    def iqr_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1, q3 = x[0], x[0]
        else:
            q1 = np.percentile(x, 25)
            q3 = np.percentile(x, 75)
        iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        # Scale to Gaussian-equivalent: IQR ≈ 1.349 * sigma → divide by 1.349 for sigma match
        return (x - np.median(x)) / (spread * PARAMS["iqr_scale_factor"])
    
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    # Apply IQR normalization only to uncertainty and energy (risk-sensitive features)
    norm_uncert = iqr_normalize(uncertainty)
    norm_energy = iqr_normalize(min_incremental_energy)
    # Use MAD for others (slower-changing, less outlier-prone)
    norm_slack = median_mad_normalize(slack)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    
    # === Feasibility-preserving slack-aware normalization: rescale rank & work only when slack > 0 ===
    # Avoids distorting critical-path importance under deadline pressure (self-reflection improvement)
    positive_slack_mask = (slack > eps).astype(float)
    norm_rank_feasible = norm_rank * positive_slack_mask + norm_rank * (1.0 - positive_slack_mask)
    norm_work_feasible = norm_work * positive_slack_mask + norm_work * (1.0 - positive_slack_mask)
    
    # === Composite DDL gate: combines tanh feasibility (soft) with hard zero for negative slack (strict)
    tanh_feasible = np.clip(PARAMS["tanh_half_scale"] * (1.0 - np.tanh(slack / (eps + np.finfo(float).tiny))), 0.0, 1.0)
    hard_deadline_violated = (slack < 0.0).astype(float)
    ddl_gate = (1.0 - hard_deadline_violated) * tanh_feasible
    
    # === Slack pressure indicator: bounded linear interpolation from -1 to 1 → [0,1] for urgency scaling
    slack_pressure = np.clip(norm_slack, -1.0, 1.0)
    
    # === Dual-gated bottleneck release: only activate successor coupling when slack is critical AND uncertainty is low
    ddl_urgent = (slack <= eps).astype(float)
    low_uncert = (norm_uncert <= PARAMS["bottleneck_activation_gate"]).astype(float)
    successor_release = norm_work_feasible * norm_rank_feasible * ddl_urgent * low_uncert
    
    # === Piecewise slack penalty: linear for mild lateness, quadratic beyond threshold
    abs_norm_slack = np.abs(norm_slack)
    linear_penalty = abs_norm_slack
    quadratic_penalty = abs_norm_slack ** 2
    piecewise_slack_penalty = np.where(
        abs_norm_slack <= PARAMS["slack_linear_threshold"],
        linear_penalty,
        quadratic_penalty
    )
    
    # === Energy penalty: only applied when DDL is violated or marginally feasible, scaled by slack pressure
    energy_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * (1.0 - ddl_gate)
    
    # === Uncertainty-energy interaction: gated by both uncertainty threshold and DDL violation
    uncert_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * (1.0 - ddl_gate)
    
    # === Bounded min-max duration sensitivity gate: saturating sigmoid over (duration / (|slack| + eps)) ===
    # Suppresses duration influence near tight deadlines — directly implements self-reflection requirement
    duration_ratio = (min_exec_time + min_comm_time) / (np.abs(slack) + eps)
    duration_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["duration_sigmoid_steepness"] * (duration_ratio - 1.0)))
    duration_penalty = PARAMS["duration_robustness_factor"] * norm_duration * slack_pressure * (1.0 - ddl_gate) * duration_sigmoid
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for bounded AST depth
    score = (
        + np.clip(piecewise_slack_penalty, -2.0, 2.0)                     # Gravity-aligned lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                          # Dual-gated bottleneck release
        - np.clip(norm_rank_feasible * (1.0 + PARAMS["criticality_scale"] * slack_pressure), -2.0, 2.0)  # Urgency amplification, slack-aware
        - PARAMS["energy_sensitivity"] * np.clip(energy_penalty, -2.0, 2.0)  # Feasible-region energy penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work_feasible, -2.0, 2.0)
        + np.clip(duration_penalty, -2.0, 2.0)                           # Saturating duration sensitivity gate
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
