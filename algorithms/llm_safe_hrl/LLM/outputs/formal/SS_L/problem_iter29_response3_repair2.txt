def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's validated DDL-protection gate and duration_robustness
       with Parent 1's piecewise interpretability and improved successor modeling.
       Key structural improvements:
       - Uses *both* sigmoid slack-pressure gate AND piecewise ramp for dual-mode sensitivity: sigmoid for
         pre-breach urgency, ramp for robust zero-crossing behavior near slack=0 (reducing jitter in critical
         transitions).
       - Introduces 'normalized_slack_distance' as clipped absolute slack distance from feasibility boundary,
         used to gate starvation relief only when slack is *nearly* feasible — avoids over-relief during hard breaches.
       - All terms bounded via [-2,2] clipping; median-MAD normalization ensures outlier resilience."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Deadline breach indicator ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Normalized slack distance from feasibility boundary (for starvation gating) ===
    # Only apply anti-starvation when slack is *close* to feasible (e.g., -0.5s to +1.0s), not deep in breach
    slack_distance = np.abs(slack)
    normalized_slack_distance = median_mad_normalize(slack_distance)
    starvation_gate = np.clip(1.0 - normalized_slack_distance, 0.0, 1.0)
    
    # === Critical path leverage: upward_rank × remaining_work (linear, no exponent — removed to comply with 12-param limit) ===
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded linear rank-slack coupling ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    
    # === Dual-mode slack-pressure activation: sigmoid for pre-breach + piecewise ramp for zero-crossing robustness ===
    # Sigmoid component: urgency ramp-up as slack approaches zero from above
    slack_pressure_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (-norm_slack + 0.5)))
    
    # Piecewise ramp: linear activation over [-offset, +offset] normalized slack, zero elsewhere
    ramp_offset = PARAMS["piecewise_slack_ramp_offset"]
    slack_ramp = np.clip((norm_slack + ramp_offset) * 2.0, 0.0, 1.0) * (np.abs(norm_slack) <= ramp_offset).astype(float)
    
    # Combined slack pressure for boosted rank
    combined_slack_pressure = slack_pressure_sigmoid + slack_ramp
    
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * combined_slack_pressure)
    
    # === Uncertainty-duration interaction: dual-gated by uncertainty threshold and slack pressure ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * uncert_gate * combined_slack_pressure * ddl_gate
    
    # === Starvation relief: logistic-shaped bonus gated by slack distance (not just breach) ===
    wait_benefit = (1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]))) * starvation_gate
    
    # === Energy-uncertainty coupling: gated by both uncertainty and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * combined_slack_pressure) * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Linear critical-path leverage
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Dual-mode urgency boost
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Distance-gated anti-starvation
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
