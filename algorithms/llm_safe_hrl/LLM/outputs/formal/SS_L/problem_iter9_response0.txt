def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces fragile dual-gated criticality with hard-DDL-triggered rank scaling.
       Introduces host-load–conditional energy penalty: penalizes high marginal energy *only* when task is 'ready but unassigned'
       (low wait time) AND has high downstream load (high remaining_work), targeting avoidable energy waste.
       Uses min-max normalization for slack and upward_rank to preserve deadline urgency and critical-path signal integrity
       under sparse or skewed distributions — avoids median-MAD distortion of extreme negative slack.
       All operations are bounded, deterministic, and shape-safe."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Min-max normalization for slack and upward_rank: preserves urgency & criticality signal integrity ===
    # Avoids median-MAD suppression of extreme negative slack (e.g., -10s vs -0.1s both become large negatives)
    def minmax_normalize(x, span_param):
        x = np.asarray(x, dtype=float)
        if N == 0:
            return np.zeros(0, dtype=float)
        x_min = np.min(x)
        x_max = np.max(x)
        range_val = x_max - x_min
        # Use parameterized span to avoid degenerate zero-range + ensure consistent scaling across runs
        scaled_range = np.maximum(range_val, span_param * eps)
        return (x - x_min) / scaled_range - 0.5  # center at 0; [-0.5, +0.5] range
    
    norm_slack = minmax_normalize(slack, PARAMS["slack_normalization_span"])
    norm_rank = minmax_normalize(upward_rank, PARAMS["rank_normalization_span"])
    
    # Normalize other features with robust median-MAD (less sensitive to outliers than minmax for non-deadline features)
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 0:
            return np.zeros(0, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        dev = x - center
        spread = np.median(np.abs(dev)) if N > 1 else np.abs(dev[0]) + eps
        return dev / (spread + eps)
    
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Hard DDL urgency gate: binary trigger for absolute deadline enforcement ===
    ddl_urgent = (slack <= 0.0).astype(float)
    
    # === Hard-DDL-triggered upward_rank scaling: simple, bounded, and interpretable ===
    # Only amplifies rank when slack <= 0 — no sigmoid fragility, no uncertainty modulation
    protected_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_urgent)
    
    # === Successor-release interaction: unblocks downstream parallelism ===
    successor_release_score = norm_work * norm_rank
    
    # === Host-load–conditional energy penalty: targets 'avoidable marginal energy' failure mode ===
    # Active only when task is freshly ready (low wait) AND high downstream load exists (high work) → implies schedulable now without delay
    # This prevents selecting high-energy VMs when a low-energy one is available *and* assignment is urgent
    load_conditional_mask = np.where(
        (norm_wait < 0.0) & (norm_work > 0.0),  # low wait time AND high remaining work
        1.0,
        0.0
    )
    conditional_energy_penalty = norm_energy * load_conditional_mask * PARAMS["energy_load_conditional_penalty"]
    
    # === Energy-uncertainty interaction: penalize high-energy only when uncertainty exceeds threshold ===
    energy_uncert_penalty = norm_energy * np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    
    # === Duration-robustness coupling: only under hard DDL violation (slack <= 0) ===
    duration_risk_score = norm_duration * norm_uncert * ddl_urgent
    
    # === Starvation relief: saturating exponential with safe offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Composite score: smaller = higher priority ===
    # Hard DDL enforcement dominates; then critical path; then anti-starvation; then conditional energy control
    score = (
        - ddl_urgent * PARAMS["ddl_urgency_boost"]           # Absolute deadline enforcement (dominant)
        - protected_rank                                     # Hard-DDL-triggered criticality (simple & robust)
        - successor_release_score                            # Unblocks downstream work
        - PARAMS["energy_sensitivity"] * norm_energy       # Base energy efficiency
        - wait_benefit                                       # Anti-starvation reward
        + PARAMS["duration_robustness"] * duration_risk_score  # Risk-duration penalty *only* when overdue
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned energy penalty
        + conditional_energy_penalty                         # Host-load–conditional energy penalty (new core improvement)
        + PARAMS["remaining_work_weight"] * norm_work        # Mild preference for large downstream load
    )
    
    # Final safeguard: ensure finite, deterministic output with strict shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(-1)
