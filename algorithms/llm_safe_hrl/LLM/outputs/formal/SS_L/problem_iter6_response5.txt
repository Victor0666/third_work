def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: adds hard-deadline enforcement via lateness penalty multiplier.
    Key improvements:
      - Introduces `lateness_penalty_strength`: multiplies *all* penalty terms (not rewards) when slack is critically negative,
        ensuring strict adherence to hard deadlines without distorting intra-feasible prioritization.
      - Reverts to original robust_normalize (MAD-only, no clipping) per reflection — preserves rank fidelity under stress.
      - Restores `slack_pressure` (not raw) in energy/rank terms for stronger nonlinear urgency amplification.
      - Removes successor pressure term as instructed; avoids unbounded upstream propagation.
      - All numeric literals remain in {-2,-1,0,1,2}; uses np.finfo for safe finite bounds.
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median-centered MAD + epsilon → resilient to outliers and zero-scales ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        center = np.median(x_abs)
        scale = np.median(np.abs(x_abs - center)) + eps
        return (x_abs - center) / scale
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack pressure: clipped power penalty for overdue tasks only ===
    slack_pressure_raw = np.clip(-norm_slack, 0.0, None)
    slack_pressure = np.power(slack_pressure_raw + eps, PARAMS["slack_penalty_exponent"])
    
    # === Lateness penalty gate: multiplicative hard-deadline enforcement ===
    # Activates only when slack < -eps (i.e., truly overdue), boosting penalties to dominate rewards
    is_late = (slack < -eps).astype(float)
    lateness_multiplier = 1.0 + PARAMS["lateness_penalty_strength"] * is_late
    
    # === Critical-path acceleration gate: smooth linear ramp down from 1→0 over [0, slack_gate_width] ===
    rank_gate = np.clip(1.0 - (norm_slack / (PARAMS["slack_gate_width"] + eps)), 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Soft uncertainty activation: graded sigmoidal response ===
    uncertainty_activation = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_smoothness"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_interaction = norm_duration * uncertainty_activation * PARAMS["duration_robustness"]
    
    # === Starvation relief: saturating exponential form ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + eps))
    
    # === Energy-slack interaction: penalizes high marginal energy more under tight slack ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["energy_slack_interaction"] * slack_pressure)
    
    # === Restored rank-slack coupling: strengthens critical path reward exactly when slack is constrained ===
    coupled_rank_reward = norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure)
    
    # === Final score: smaller = higher priority ===
    # Apply lateness multiplier only to penalties (not rewards) to preserve reward integrity while forcing deadline compliance
    score = (
        + lateness_multiplier * slack_pressure                # Urgency driver (scaled when late)
        + lateness_multiplier * energy_slack_penalty          # Deadline-amplified energy cost (scaled when late)
        + lateness_multiplier * PARAMS["energy_sensitivity"] * norm_energy  # Base energy penalty (scaled when late)
        - coupled_rank_reward                                # Critical path reward (untouched by lateness)
        - wait_benefit                                       # Anti-starvation relief (untouched)
        + lateness_multiplier * duration_risk_interaction    # Graded risk-duration penalty (scaled when late)
    )
    
    # Final safeguard: ensure finite, deterministic output with strict shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    return score
