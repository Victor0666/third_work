def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's higher performance with Parent 1's robustness:
       - Adaptive outlier clamping before quantile normalization using MAD-based bounds.
       - Smooth successor-release coupling: norm_rank * norm_work * sigmoid(slack) with tunable steepness.
       - Sharpened urgency via MAD-normalized slack (sign-preserving) as primary driver.
       - Energy terms remain feasibility-gated and uncertainty-modulated; no redundant parameters."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust pre-clamping: clip outliers to ±k*MAD before quantile normalization ===
    def robust_preclamp(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        clamp_bound = PARAMS["outlier_clamp_mad_factor"] * (mad + eps)
        return np.clip(x, med - clamp_bound, med + clamp_bound)
    
    # === Quantile-based robust normalization with pre-clamping ===
    def quantile_normalize(x):
        x_clamped = robust_preclamp(x)
        if N == 1:
            q1 = q3 = x_clamped[0]
        else:
            q1 = np.quantile(x_clamped, PARAMS["quantile_q1"])
            q3 = np.quantile(x_clamped, PARAMS["quantile_q3"])
        iqr = q3 - q1 if q3 > q1 else eps
        return (x_clamped - q1) / iqr
    
    norm_slack = quantile_normalize(slack)
    norm_energy = quantile_normalize(min_incremental_energy)
    norm_duration = quantile_normalize(min_exec_time + min_comm_time)
    norm_rank = quantile_normalize(upward_rank)
    norm_work = quantile_normalize(remaining_work)
    norm_wait = quantile_normalize(ready_wait_time)
    norm_uncert = quantile_normalize(uncertainty)
    
    # === Sign-preserving MAD normalization *only for slack* to sharpen urgency near deadline violation ===
    slack_med = np.median(slack)
    slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = slack_mad if slack_mad > eps else eps
    norm_slack_mad = (slack - slack_med) / slack_spread
    
    # === DDL-feasibility gate: sigmoid activated only for slack >= 0 ===
    ddl_feasible_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (slack - 0.0)))
    
    # === Smooth successor-release coupling: norm_rank * norm_work * sigmoid(slack) — monotonic ramp-up near deadline ===
    successor_release_smooth = norm_rank * norm_work * (
        1.0 / (1.0 + np.exp(-PARAMS["successor_release_smoothness"] * slack))
    )
    
    # === Deadline pressure gate: sigmoid for continuous pressure signal (replaces binary) ===
    ddl_pressure_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["successor_release_smoothness"] * (-slack)))
    
    # === Slack-aware uncertainty activation: enables risk gating only when feasible AND uncertainty exceeds adaptive threshold ===
    adaptive_uncert_threshold = PARAMS["slack_aware_uncert_threshold"] * (1.0 - 1.0 / (1.0 + np.exp(-slack)))
    slack_aware_uncert_gate = (uncertainty > adaptive_uncert_threshold).astype(float) * ddl_feasible_gate
    
    # === Criticality coupling: upward_rank × remaining_work, activated smoothly under deadline pressure ===
    critical_coupling = norm_rank * norm_work * ddl_pressure_sigmoid * PARAMS["criticality_coupling_strength"]
    
    # === Starvation relief: dedicated linear benefit *only* under deadline pressure (smoothly activated) ===
    wait_benefit_pressure = np.clip(PARAMS["wait_benefit_under_pressure"] * ready_wait_time, 0.0, 2.0) * ddl_pressure_sigmoid
    
    # === Energy-uncertainty coupling: enabled only when DDL-feasible AND slack-aware uncertainty condition met ===
    energy_uncert_coupling = norm_energy * norm_uncert * slack_aware_uncert_gate
    
    # === Slack-driven energy amplification: energy penalty scaled by normalized slack pressure ===
    slack_pressure = np.clip(-norm_slack_mad, 0.0, 2.0)
    energy_slack_amplifier = 1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure
    
    # === Direct risk signal: normalized uncertainty weighted by DDL feasibility ===
    risk_signal = norm_uncert * ddl_feasible_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_mad, -2.0, 2.0)                              # Primary urgency driver
        - np.clip(successor_release_smooth, -2.0, 2.0)                    # Smooth critical-path release
        - np.clip(critical_coupling, -2.0, 2.0)                           # Smooth urgency-coupled importance
        - np.clip(wait_benefit_pressure, -2.0, 2.0)                       # Smooth anti-starvation
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_feasible_gate, -2.0, 2.0)  # Feasibility-gated energy preference
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_coupling, -2.0, 2.0)
        + np.clip(norm_energy * energy_slack_amplifier * ddl_feasible_gate, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * ddl_pressure_sigmoid, -2.0, 2.0)
        + np.clip(risk_signal, -2.0, 2.0)                                 # Direct feasibility-weighted risk
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
