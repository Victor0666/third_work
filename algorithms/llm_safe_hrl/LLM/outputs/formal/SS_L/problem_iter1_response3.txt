def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing risk-aware urgency, critical-path acceleration under deadline pressure,
    and robust starvation control. Uses piecewise-normalized, bounded interactions — no unbounded ops."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: use median absolute deviation (MAD) + epsilon for outlier resilience ===
    def robust_normalize(x):
        x = np.abs(x)
        center = np.median(x)
        scale = np.median(np.abs(x - center)) + eps
        return (x - center) / scale
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack pressure: strong nonlinear penalty for negative slack ===
    # Use clipped exponent to avoid overflow: max(0, -norm_slack)^p → penalizes lateness risk aggressively
    slack_pressure = np.power(np.clip(-norm_slack, 0.0, None), PARAMS["slack_penalty_exponent"])
    
    # === Critical-path acceleration gate: boost upward_rank only when slack is low (tight deadlines) ===
    # Gate: 1 if norm_slack < 0, else smoothly decays to 0 at norm_slack = 0.5
    rank_gate = np.clip(1.0 - 2.0 * np.clip(norm_slack, 0.0, 0.5), 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Uncertainty-duration interaction: only activate under high uncertainty AND tight slack ===
    # Bounded gate: activates only when both normalized uncertainty > threshold AND slack_pressure > 0
    uncert_gate = (
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) &
        (slack_pressure > 0)
    ).astype(float)
    duration_risk_interaction = norm_duration * uncert_gate * PARAMS["duration_robustness"]
    
    # === Starvation relief: saturating benefit from wait time, not linearly growing ===
    # tanh-based soft saturation: prevents runaway priority from long waits
    wait_benefit = np.tanh(PARAMS["wait_decay"] * norm_wait)
    
    # === Rank-slack coupling: enhance critical tasks *only* when slack is constrained ===
    coupled_rank = norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure)
    
    # === Final score: smaller = higher priority ===
    # Prioritize: (1) deadline risk (penalty), (2) energy efficiency (reward), (3) critical path (reward under pressure),
    # (4) starvation relief (reward), (5) uncertainty-aware duration (penalty when risky+urgent)
    score = (
        + slack_pressure                                  # Urgency penalty (dominant for late tasks)
        + PARAMS["energy_sensitivity"] * norm_energy     # Energy cost (penalty)
        - coupled_rank                                   # Critical path reward (amplified by slack pressure)
        - wait_benefit                                   # Starvation relief (reward)
        + duration_risk_interaction                      # Risk-duration penalty (only active when needed)
    )
    
    # Ensure finite output, shape (N,), deterministic
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(N)
