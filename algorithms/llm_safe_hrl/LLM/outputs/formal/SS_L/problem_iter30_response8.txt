def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces duration_robustness with host-load-aware marginal-energy gating;
       eliminates wait_saturation_offset and uses bounded linear starvation relief conditioned on slack;
       introduces additive critical-path release priority (not multiplicative) gated by breach AND successor criticality;
       preserves median-MAD normalization, DDL-protection gate, and all stability safeguards."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Deadline breach indicator: binary activation for urgency and coupling ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Uncertainty activation gate: sigmoid threshold on normalized uncertainty ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    
    # === Host-load-aware marginal-energy gating: replaces duration_robustness with VM load proxy
    # Uses norm_duration as surrogate for host load (higher duration → higher load pressure)
    # Activated only when both load pressure and slack pressure are high
    host_load_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_duration - PARAMS["host_load_gate_threshold"])))
    load_pressure = np.clip(norm_duration, 0.0, 2.0)
    marginal_energy_penalty = norm_energy * host_load_gate * load_pressure * ddl_gate
    
    # === Critical-path release priority: additive term, not multiplicative
    # Gated by deadline breach AND high successor criticality (upward_rank > median)
    successor_critical = (upward_rank > np.median(upward_rank)).astype(float)
    critical_path_release = PARAMS["critical_path_release_weight"] * norm_rank * ddl_breach * successor_critical
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded linear rank-slack coupling: stable monotonic amplification under pressure ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    
    # === Sharpened slack-pressure gate: sigmoid-based ramp for pre-breach urgency ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Linear starvation relief: bounded, slack-conditioned, no offset or exp fragility
    # Benefit increases linearly with wait time but saturates at 1.0 and is suppressed when slack > 0
    wait_benefit = np.clip(PARAMS["wait_linear_coeff"] * ready_wait_time, 0.0, 1.0) * (1.0 - ddl_gate)
    
    # === Energy-uncertainty coupling: gated by uncertainty and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        + critical_path_release                                             # Additive critical-path release (no clipping needed: bounded by design)
        - np.clip(boosted_rank, -2.0, 2.0)                                  # HEFT-based urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation (linear, bounded, slack-gated)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(marginal_energy_penalty, -2.0, 2.0)                      # Host-load-aware marginal energy penalty
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
