def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's smooth urgency gate and rank floor with Parent 1's successor-release interaction.
       Uses robust median-MAD normalization; all operations safeguarded against NaN/inf/zero.
       Prioritizes hard DDL feasibility first, then critical path, then energy efficiency, then starvation relief — with strict ordering via gated terms."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization: sign-preserving, outlier-resilient scaling ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Smooth DDL urgency gate: sigmoid transition — avoids optimization cliffs ===
    # Uses raw slack scaled by median absolute slack for numerical stability
    median_abs_slack = np.median(np.abs(slack)) if N > 1 else np.abs(slack[0])
    urgency_input = -slack / (median_abs_slack + eps)
    ddl_urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_urgency_steepness"] * urgency_input))
    
    # === Pure raw-slack-penalty dominance for hard deadline enforcement ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    
    # === Critical-path boost with floor: prevents rank suppression under all conditions ===
    rank_boost = PARAMS["rank_boost_floor"] + (1.0 - PARAMS["rank_boost_floor"]) * ddl_urgency_gate
    protected_rank = norm_rank * rank_boost
    
    # === Successor-release interaction: prioritizes tasks that unblock both critical and voluminous successors ===
    # Product of normalized rank and work captures 'high-impact release' pattern
    successor_release_impact = norm_rank * norm_work
    
    # === Starvation relief: capped logistic benefit to prevent dominance over DDL enforcement ===
    wait_raw = PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"])
    wait_benefit = np.clip(1.0 - np.exp(-wait_raw), 0.0, 1.0)
    
    # === Energy-uncertainty interaction: penalize only when both exceed thresholds ===
    energy_uncert_penalty = (
        norm_energy * norm_uncert *
        np.where(
            (norm_energy > 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]),
            1.0,
            0.0
        )
    )
    
    # === Duration robustness: penalize long durations only under combined risk (tight slack + high uncertainty) ===
    duration_risk_score = (
        norm_duration *
        np.where(
            (slack <= eps) & (uncertainty >= PARAMS["uncertainty_gate_threshold"]),
            1.0,
            0.0
        )
    )
    
    # === Composite score: smaller = higher priority ===
    # Ordering reflects objective hierarchy: DDL feasibility > critical path > energy > starvation > work
    score = (
        + raw_slack_penalty                           # Hard DDL enforcement dominates
        - PARAMS["criticality_scale"] * protected_rank  # Critical path amplified with floor
        - successor_release_impact                    # Prefer unblocking high-impact successors
        - PARAMS["energy_sensitivity"] * norm_energy   # Low marginal energy preferred
        - norm_duration                                # Short duration preferred
        - wait_benefit                                 # Anti-starvation bonus, strictly capped
        + PARAMS["duration_robustness"] * duration_risk_score  # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
