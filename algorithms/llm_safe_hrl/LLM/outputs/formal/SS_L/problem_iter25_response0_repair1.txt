def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: removes unused parameters; retains successor-release interaction;
       uses joint congestion gating (high ready_wait_time AND high uncertainty);
       applies energy/uncertainty penalties only under feasibility (slack >= 0);
       replaces sigmoid gates with direct threshold-based logic for stability and evidence alignment."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: MAD-based, sign-preserving, epsilon-guarded ===
    def robust_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = np.maximum(mad, eps)
        return (x - med) / spread
    
    norm_slack = robust_mad_normalize(slack)
    norm_energy = robust_mad_normalize(min_incremental_energy)
    norm_rank = robust_mad_normalize(upward_rank)
    norm_work = robust_mad_normalize(remaining_work)
    norm_wait = robust_mad_normalize(ready_wait_time)
    norm_uncert = robust_mad_normalize(uncertainty)
    
    # === Feasibility mask: only apply energy/uncertainty penalties when slack >= 0 (DDL-feasible region) ===
    feasible_mask = (slack >= 0.0).astype(float)
    
    # === Deadline pressure indicator: binary flag for immediate risk (slack <= 0) ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Critical path leverage: upward_rank * remaining_work, activated only under breach ===
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: fast-executing blockers of critical descendants ===
    # Validated in cross-agent consensus failures
    successor_release = np.clip(min_exec_time, eps, None) * np.clip(norm_rank, -1.0, 1.0) * ddl_breach
    
    # === Sign-preserving slack urgency: use raw signed slack normalized by its own MAD (sharper near zero) ===
    slack_mad = np.maximum(np.median(np.abs(slack - np.median(slack))), eps)
    norm_slack_urgency = (slack - np.median(slack)) / (slack_mad + eps)
    
    # === Risk-adjusted slack penalty: power law on raw negative slack only ===
    raw_slack_penalty = np.where(slack < 0.0, (-slack) ** PARAMS["slack_penalty_exponent"], 0.0)
    norm_slack_penalty = robust_mad_normalize(raw_slack_penalty)
    
    # === Joint congestion gating: activates only when BOTH ready_wait_time AND uncertainty are high ===
    # Uses median/std for robustness; avoids fragile z-score assumptions
    wait_centered = ready_wait_time - np.median(ready_wait_time)
    wait_std = np.maximum(np.std(ready_wait_time), eps)
    wait_z = wait_centered / (wait_std + eps)
    
    uncert_centered = uncertainty - np.median(uncertainty)
    uncert_std = np.maximum(np.std(uncertainty), eps)
    uncert_z = uncert_centered / (uncert_std + eps)
    
    congestion_gate = (wait_z > PARAMS["uncertainty_gate_threshold"]) & (uncert_z > PARAMS["uncertainty_gate_threshold"])
    congestion_gate = congestion_gate.astype(float)
    
    # === Starvation relief: exponential decay scaled by infeasibility to avoid interfering with DDL pressure ===
    wait_benefit = np.exp(-PARAMS["wait_decay"] * ready_wait_time) * (1.0 - feasible_mask)
    
    # === Energy-uncertainty coupling: gated by both feasibility AND joint congestion ===
    energy_uncert_penalty = norm_energy * norm_uncert * feasible_mask * congestion_gate
    
    # === Composite score: smaller = higher priority; all terms clipped for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Critical path importance under breach
        - np.clip(successor_release, -2.0, 2.0)                           # Fast blocker of critical successors
        - np.clip(norm_slack_urgency, -2.0, 2.0)                          # Direct urgency signal
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * feasible_mask, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation only when DDL violated
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + PARAMS["criticality_scale"] * np.clip(norm_rank * feasible_mask, -2.0, 2.0)  # Mild rank boost when feasible
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
