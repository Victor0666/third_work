def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule:
       - Replaces hard threshold with smooth, differentiable slack-sensitivity gate (centered at PARAMS["slack_sensitivity_center"])
         for critical-path leverage — improves gradient-based optimization and avoids brittle boundaries.
       - Introduces successor-release interaction: penalizes tasks whose successors are not yet ready using
         `upward_rank * (1 - normalized_successor_release_prob)` where release probability is estimated via
         robustly normalized `ready_wait_time` as proxy for upstream completion readiness.
       - Adds host load proxy gating: unifies `ready_wait_time + uncertainty` under a sigmoid gated by itself,
         decoupling starvation relief from deadline pressure and enabling adaptive load-awareness.
       - All numeric literals are restricted to -2, -1, 0, 1, 2; uses np.finfo for epsilon safeguards."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely without in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust range-based normalization: (x - Q1) / (Q3 - Q1 + eps) — preserves sign, resists outliers ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])
            q3 = np.quantile(x, PARAMS["quantile_q3"])
        iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        return (x - q1) / spread
    
    # Normalize all features except slack (which gets sign-preserving MAD for urgency sensitivity near zero)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_exec = robust_range_normalize(min_exec_time)
    norm_comm = robust_range_normalize(min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # Sign-preserving MAD for slack: sensitive to small negative values near deadline boundary
    if N == 1:
        slack_med = slack[0]
        slack_mad = eps
    else:
        slack_med = np.median(slack)
        slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = slack_mad if slack_mad > eps else eps
    norm_slack = (slack - slack_med) / slack_spread
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty terms only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Smooth slack-sensitivity gate for critical-path leverage (replaces hard threshold) ===
    # Activates near deadline boundary with tunable center and slope — differentiable & robust
    slack_sensitivity_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (slack - PARAMS["slack_sensitivity_center"])))
    critical_path_score = norm_rank * norm_work * slack_sensitivity_gate
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized via robust range ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_range_normalize(raw_slack_penalty)
    
    # === Successor-release interaction: estimate successor readiness via wait-time proxy ===
    # Higher ready_wait_time → lower perceived successor release prob → stronger blocking penalty
    # Uses same robust normalization for consistency and boundedness
    successor_release_prob = 1.0 - np.clip(norm_wait, 0.0, 1.0)  # monotonic decreasing proxy
    successor_blocking_penalty = norm_rank * (1.0 - successor_release_prob) * PARAMS["successor_release_interaction"]
    
    # === Host load proxy gating: adaptive congestion signal based on self-measured wait+uncertainty ===
    # Enables dynamic load-awareness independent of deadline status
    load_proxy = norm_wait + norm_uncert
    load_gate = 1.0 / (1.0 + np.exp(-PARAMS["host_load_proxy_gate_slope"] * load_proxy))
    congestion_score = load_proxy * load_gate * PARAMS["congestion_weight"]
    
    # === Separated execution & communication penalties: individually gated and scaled under DDL breach ===
    ddl_breach = (slack <= PARAMS["slack_sensitivity_center"]).astype(float)
    exec_penalty = PARAMS["exec_comm_separation_factor"] * norm_exec * ddl_breach
    comm_penalty = (1.0 - PARAMS["exec_comm_separation_factor"]) * norm_comm * ddl_breach
    
    # === Energy-uncertainty coupling: only active under DDL breach and high uncertainty ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - 0.25)))  # fixed center for stability
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_breach
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability and scale balance ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_score, -2.0, 2.0)                          # Smooth critical-path leverage
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy minimization only when feasible
        + np.clip(congestion_score, -2.0, 2.0)                             # Adaptive load-aware congestion relief
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * ddl_gate, -2.0, 2.0)  # Prefer large downstream work only when feasible
        + np.clip(exec_penalty, -2.0, 2.0)                                 # Compute-bound urgency under breach
        + np.clip(comm_penalty, -2.0, 2.0)                                 # I/O-bound deferral unless bandwidth-resilient
        + np.clip(successor_blocking_penalty, -2.0, 2.0)                   # Blocking mitigation via successor readiness proxy
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
