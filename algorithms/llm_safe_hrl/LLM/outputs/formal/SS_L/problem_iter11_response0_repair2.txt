def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's robust DDL-protection gating and clipping with Parent 1's successor-release coupling;
       replaces median-MAD with trimmed-mean+IQR normalization using tunable trim_fraction;
       introduces explicit successor-release term gated by both DDL urgency AND upstream dependency count proxy (via slack sign);
       enforces strict monotonicity via bounded linear coupling and term-wise [-2,2] clipping."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Trimmed-mean + IQR normalization: tunable trim fraction for outlier resilience ===
    def robust_normalize(x):
        if N == 1:
            center = x[0]
            spread = eps
        else:
            # Trim PARAMS["trim_fraction"] from each tail
            k = max(1, int(PARAMS["trim_fraction"] * N))
            x_sorted = np.sort(x)
            trimmed = x_sorted[k:-k] if len(x_sorted) > 2*k else x_sorted
            center = np.mean(trimmed)
            q1 = np.percentile(trimmed, 25)
            q3 = np.percentile(trimmed, 75)
            spread = q3 - q1 + eps
        return (x - center) / spread
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack penalty: normalized power penalty on overdue tasks ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Bounded linear rank-slack coupling: stable, monotonic, sign-preserving ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    boosted_rank = norm_rank * rank_slack_coupling
    
    # === Sharpened slack-pressure gate: sigmoid over bounded slack pressure ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = boosted_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Successor-release interaction: activated only under DDL urgency (slack <= 0) ===
    # Encodes structural pressure: unblocking high `remaining_work` under deadline risk is urgent
    ddl_urgent_mask = (slack <= 0.0).astype(float)
    successor_release_score = norm_rank * norm_work * ddl_urgent_mask
    
    # === Uncertainty-duration interaction: gated by both uncertainty threshold AND DDL feasibility ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = norm_duration * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty interaction: gated by DDL feasibility → only active when slack >= 0 ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Decoupled duration weight: tunable preference for low-duration tasks when feasible ===
    duration_preference = PARAMS["duration_robustness"] * norm_duration * ddl_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness risk dominates
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Critical path under pressure
        - np.clip(successor_release_score, -2.0, 2.0)                      # Unblock large workloads first when overdue
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy minimization only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Always active starvation mitigation
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
