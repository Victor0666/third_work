def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces raw slack in rank coupling with normalized gradient-based urgency;
       tightens deadline-breach activation to slack <= -eps to suppress noise near zero;
       uses adaptive relative criticality (upward_rank / (median + offset)) for topology-agnostic sensitivity;
       unifies all duration-related signals into a single robust duration-uncertainty composite;
       clips all intermediate terms to [-2,2] and enforces strict finite output shape (N,)."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = np.maximum(mad, eps)
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    # Unified duration-uncertainty exposure: sum of min times weighted by uncertainty
    duration_exposure = (min_exec_time + min_comm_time) * (1.0 + uncertainty)
    norm_duration_exposure = median_mad_normalize(duration_exposure)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Sigmoid DDL-protection gate: sharp, stable boundary at slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: activated only under *strict* deadline breach (slack <= -eps) AND non-trivial relative criticality ===
    rank_median = np.median(upward_rank) if N > 1 else upward_rank[0]
    # Relative criticality with additive offset for stability and soft thresholding
    rel_criticality = upward_rank / (rank_median + PARAMS["relative_criticality_offset"])
    ddl_breach_mask = (slack <= -eps).astype(float)
    critical_rank_mask = (rel_criticality > 1.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach_mask * critical_rank_mask * PARAMS["rank_work_coupling_weight"]
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    
    # === Smooth normalized slack gradient for urgency coupling — more robust to scale shifts than raw slack ===
    # Gradient approximated via clipped, smoothed derivative of sigmoid gate
    slack_gradient = PARAMS["ddl_protection_gate_slope"] * ddl_gate * (1.0 - ddl_gate)
    # Apply exponent smoothing for monotonic robustness
    smoothed_urgency = np.clip(slack_gradient, eps, 1.0) ** PARAMS["slack_gradient_smoothing"]
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * smoothed_urgency
    
    # === Starvation relief: linear, capped, and numerically bounded ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Unified duration-uncertainty risk score: replaces separate duration/uncert couplings ===
    tight_slack_mask = (slack <= eps).astype(float)
    high_uncert_mask = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float)
    duration_uncert_risk = norm_duration_exposure * tight_slack_mask * high_uncert_mask * PARAMS["duration_slack_coupling_weight"]
    
    # === Energy-uncertainty coupling: gated by both uncertainty threshold and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Host-load–conditional suppression: robustified via relative criticality mask ===
    low_rel_rank_mask = (rel_criticality < 1.0).astype(float) * (slack > 0.0).astype(float)
    host_load_penalty = norm_energy * norm_wait * low_rel_rank_mask
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + raw_slack_penalty                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0) # Critical-path blocking relief (tightened activation)
        - np.clip(norm_rank * rank_slack_coupling, -2.0, 2.0)  # Robust urgency coupling via gradient
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)           # Anti-starvation (capped)
        + PARAMS["duration_robustness"] * np.clip(duration_uncert_risk, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(host_load_penalty, -2.0, 2.0)      # Suppression now uses relative criticality
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
