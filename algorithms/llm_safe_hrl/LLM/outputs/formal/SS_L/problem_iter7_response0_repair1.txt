def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: removes unused 'slack_pressure_gate_steepness'; uses binary DDL protection gate,
       successor-release interaction, and simplified robust normalization. All declared parameters are used."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization: sign-preserving, outlier-resilient scaling ===
    def robust_normalize(x):
        center = np.median(x) if N > 1 else x[0]
        # Use MAD (median absolute deviation) for spread estimation
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        # Avoid division by zero; ensure finite output
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Binary DDL protection gate: hard urgency signal for overdue or critically tight tasks ===
    # Activates when slack <= 0 (guaranteed violation risk) OR slack is near zero with high uncertainty
    ddl_urgent = np.where(
        (slack <= 0.0) | 
        ((slack <= PARAMS["epsilon"]) & (uncertainty >= PARAMS["uncertainty_gate_threshold"])),
        1.0,
        0.0
    )
    
    # === Conditional DDL protection: boost rank *only* under urgent conditions ===
    protected_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_urgent)
    
    # === Successor-release interaction: amplify priority for tasks whose execution unblocks many successors ===
    # Approximated via remaining_work * upward_rank — reflects both critical path importance and descendant load
    successor_release_score = norm_work * norm_rank
    
    # === Energy-uncertainty interaction: only penalize high-energy *and* high-uncertainty tasks ===
    energy_uncert_penalty = norm_energy * norm_uncert * np.where(
        (norm_energy > 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]),
        1.0,
        0.0
    )
    
    # === Starvation relief: linear-in-wait bonus capped by saturation offset ===
    wait_benefit = np.clip(
        PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]),
        0.0,
        1.0
    )
    
    # === Duration robustness: penalize long durations *only* when uncertainty is high and slack is tight ===
    duration_risk_score = norm_duration * np.where(
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) & (slack <= 0.0),
        1.0,
        0.0
    )
    
    # === Slack penalty: power-based penalty for negative slack, normalized for stability ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Composite score: smaller = higher priority ===
    score = (
        + norm_slack_penalty                           # Strong penalty for lateness risk
        - protected_rank                             # Critical-path boost *only* when urgent
        - successor_release_score                    # Prefer tasks releasing many successors
        - PARAMS["energy_sensitivity"] * norm_energy  # Prefer low marginal energy
        - norm_duration                                # Prefer short duration
        + PARAMS["duration_robustness"] * duration_risk_score  # Penalize risky+slow under violation risk
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # High-risk-high-energy penalty
        - wait_benefit                                 # Anti-starvation bonus (reduces score → increases priority)
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
