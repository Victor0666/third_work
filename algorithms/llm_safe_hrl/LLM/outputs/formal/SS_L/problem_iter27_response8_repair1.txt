def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces group-aware quantile normalization with safe median-MAD per feature
       using task-type proxy derived solely from upward_rank magnitude (no loops); introduces conditional
       DDL-protection gate; adds exponentiated joint penalty for low-criticality + high-risk-energy;
       preserves all robustness safeguards, bounded clipping, and deterministic structure."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Task-type proxy: entry (high rank), interior (medium), exit (low rank) — computed vectorized ===
    # Use fixed thresholds instead of quantiles to avoid loops/conditionals on variable-length arrays
    rank_mean = np.mean(upward_rank) if N > 1 else upward_rank[0]
    rank_std = np.std(upward_rank, ddof=1) if N > 1 else 1.0
    # Normalize rank to [0,1] for grouping
    rank_norm = np.clip((upward_rank - (rank_mean - rank_std)) / (2.0 * rank_std + eps), 0.0, 1.0)
    # Assign groups: 0=exit (low rank), 1=interior, 2=entry (high rank)
    rank_groups = np.where(rank_norm < 0.33, 0, np.where(rank_norm < 0.66, 1, 2))
    
    # === Group-aware robust normalization: compute median/MAD per group for key features ===
    def group_aware_normalize(x, groups):
        x = np.asarray(x, dtype=float)
        norm_x = np.zeros_like(x)
        # Process each group separately using boolean indexing (no loops)
        for g in (0, 1, 2):
            mask = (groups == g)
            if np.any(mask):
                x_g = x[mask]
                if len(x_g) == 1:
                    med = x_g[0]
                    mad = eps
                else:
                    med = np.median(x_g)
                    mad = np.median(np.abs(x_g - med))
                spread = mad + eps
                norm_x[mask] = (x_g - med) / spread
            else:
                # Fallback to global normalization for empty group
                if N == 1:
                    med = x[0]
                    mad = eps
                else:
                    med = np.median(x)
                    mad = np.median(np.abs(x - med))
                spread = mad + eps
                norm_x[mask] = (x[mask] - med) / spread
        return norm_x
    
    norm_slack = group_aware_normalize(slack, rank_groups)
    norm_energy = group_aware_normalize(min_incremental_energy, rank_groups)
    norm_duration = group_aware_normalize(min_exec_time + min_comm_time, rank_groups)
    norm_rank = group_aware_normalize(upward_rank, rank_groups)
    norm_work = group_aware_normalize(remaining_work, rank_groups)
    norm_wait = group_aware_normalize(ready_wait_time, rank_groups)
    norm_uncert = group_aware_normalize(uncertainty, rank_groups)
    
    # === Hard DDL-feasibility gate: binary suppression — zero energy/uncert penalties if slack < 0 ===
    hard_feasibility_gate = np.where(slack < 0.0, 0.0, 1.0)
    
    # === Conditional DDL-protection gate: activates only when slack < threshold (decouples urgency enforcement) ===
    ddl_protection_gate = np.where(slack < PARAMS["ddl_protection_threshold"], 1.0, 0.0)
    
    # === Smooth, bounded ddl-pressure: finite-floor inverse mapping → stable monotonic urgency signal ===
    slack_abs = np.abs(slack) + eps
    ddl_pressure = np.clip(1.0 / (1.0 + slack_abs), 0.0, 1.0)
    
    # === Linear slack penalty with tunable coefficient ===
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = group_aware_normalize(raw_slack_penalty, rank_groups)
    slack_penalty = PARAMS["slack_penalty_linear_coeff"] * norm_slack_penalty
    
    # === Critical-path coupling: additive rank × work × pressure — stable & interpretable ===
    successor_release = norm_rank * norm_work * ddl_pressure
    
    # === Critical-path coupling: upward_rank scaled linearly by ddl_pressure ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_pressure)
    
    # === Uncertainty-aware energy penalty: gated by hard_feasibility_gate AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * hard_feasibility_gate
    
    # === Starvation relief: enhanced coupling — modulated by both feasibility AND pressure ===
    wait_benefit = np.clip(
        PARAMS["wait_decay"] * ready_wait_time * 
        PARAMS["starvation_feasibility_coupling"] * 
        hard_feasibility_gate * (1.0 - ddl_pressure),
        0.0, 2.0
    )
    
    # === Robustified duration penalty: damped and fully gated by hard_feasibility_gate and ddl_pressure ===
    duration_penalty = PARAMS["duration_robustness_factor"] * norm_duration * ddl_pressure * hard_feasibility_gate
    
    # === Hard slack-driven energy suppression: tunable strength replaces binary zeroing ===
    slack_energy_suppress = np.where(
        slack < 0.0,
        1.0 - PARAMS["slack_energy_suppression_strength"],
        1.0
    )
    suppressed_norm_energy = norm_energy * slack_energy_suppress
    
    # === Host-load surrogate gating: uses norm_duration * norm_uncert as congestion proxy;
    #     activated only when DDL-feasible and uncertainty above threshold ===
    load_surrogate = norm_duration * norm_uncert
    load_gate = np.where(
        (hard_feasibility_gate > 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]),
        np.clip(load_surrogate, 0.0, 2.0),
        0.0
    )
    host_load_penalty = PARAMS["host_load_sensitivity"] * load_gate * suppressed_norm_energy
    
    # === Joint critical-path & risk penalty: exponentiated coupling to sharpen discrimination ===
    # Compute (1 - clipped successor_release)^exponent * host_load_penalty
    clipped_sr = np.clip(successor_release, 0.0, 1.0)
    sr_complement = 1.0 - clipped_sr
    # Use np.power for safe exponentiation (avoids loops)
    joint_penalty = np.power(sr_complement, PARAMS["critical_path_coupling_exponent"]) * host_load_penalty
    
    # === Composite score: smaller = higher priority; all terms clipped individually to [-2,2] ===
    score = (
        + np.clip(slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                      # Additive release-driven criticality boost
        - np.clip(coupled_rank, -2.0, 2.0)                           # Pressure-scaled critical path
        - PARAMS["energy_sensitivity"] * np.clip(suppressed_norm_energy, -2.0, 2.0)  # Tunably suppressed energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                           # Feasibility-and-pressure-coupled anti-starvation
        + np.clip(duration_penalty, -2.0, 2.0)                       # Damped, feasibility-gated duration penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)   # Downstream work bias
        + np.clip(joint_penalty, -2.0, 2.0)                          # Exponentiated joint criticality-risk penalty
        + np.clip(ddl_protection_gate * (-2.0 * norm_rank), -2.0, 2.0)  # DDL-protection gate boosts critical tasks when urgent
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
