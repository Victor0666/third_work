def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid DDL gates with bounded smooth pressure function;
       introduces successor-release interaction (min_exec_time * upward_rank * (slack <= 0));
       removes redundant duration_robustness and wait_saturation_offset (validated inactive);
       uses sign-preserving MAD-normalized slack for sharper urgency near zero;
       applies joint congestion gating only when both ready_wait_time and uncertainty are high;
       enforces strict feasibility-first energy penalization via hard binary gate (slack >= 0);
       adds host-load surrogate (norm_duration * norm_uncert) gated by feasibility and uncertainty threshold."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    # Normalize all features robustly
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Hard binary feasibility gate: enables energy/uncertainty terms only if slack >= 0 ===
    # Aligns with strict deadline-first policy: no energy trade-off under violation risk
    feasible_gate = (slack >= 0.0).astype(float)
    
    # === Smooth, bounded ddl_pressure: 1/(1+|slack|+eps) — graded urgency without instability near zero ===
    ddl_pressure = 1.0 / (1.0 + np.abs(slack) + eps)
    
    # === Critical-path urgency: upward_rank × remaining_work × (slack <= 0) ===
    # Targets tasks blocking critical descendants *only* when DDL breach is imminent
    critical_breach = (slack <= 0.0).astype(float)
    critical_path_impact = norm_rank * norm_work * critical_breach
    
    # === Successor-release interaction: fast-executing blockers of critical paths ===
    # min_exec_time × clipped_upward_rank × (slack <= 0) — validated for cross-agent consensus failures
    clipped_rank = np.clip(norm_rank, 0.0, 2.0)
    successor_release = min_exec_time * clipped_rank * critical_breach
    
    # === Joint congestion gating: active only when *both* high wait time AND high uncertainty ===
    # Prevents spurious load penalties while preserving anti-starvation
    wait_high = (ready_wait_time > np.median(ready_wait_time) + 0.5 * np.std(ready_wait_time)).astype(float)
    uncert_high = (uncertainty > PARAMS["uncertainty_gate_threshold"]).astype(float)
    congestion_gate = wait_high * uncert_high
    
    # === Host-load surrogate: duration × uncertainty, gated by feasibility & uncertainty threshold ===
    # Enables congestion-aware energy penalization without VM-level visibility
    host_load_surrogate = norm_duration * norm_uncert * feasible_gate * uncert_high
    
    # === Starvation relief: bounded exponential decay with linear scaling ===
    # Uses raw wait time (not normalized) for intuitive anti-starvation behavior
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Energy penalty: only applied when feasible (slack >= 0), scaled by ddl_pressure ===
    energy_penalty = norm_energy * feasible_gate * (1.0 + PARAMS["slack_penalty_exponent"] * ddl_pressure)
    
    # === Uncertainty-energy coupling: penalizes high-risk + high-energy only under feasibility ===
    energy_uncert_coupling = norm_energy * norm_uncert * feasible_gate * uncert_high
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack, -2.0, 2.0)                                  # Direct slack urgency (sign-preserving)
        + np.clip(ddl_pressure, 0.0, 2.0)                                 # Graded pressure near deadline
        - np.clip(critical_path_impact, -2.0, 2.0)                        # Critical path leverage under breach
        - np.clip(successor_release, -2.0, 2.0)                           # Fast blockers of critical successors
        - PARAMS["energy_sensitivity"] * np.clip(energy_penalty, -2.0, 2.0)  # Feasibility-gated energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                # Anti-starvation relief
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_coupling, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + PARAMS["slack_pressure_gate_steepness"] * np.clip(host_load_surrogate, -2.0, 2.0)
        + PARAMS["criticality_scale"] * np.clip(norm_rank * ddl_pressure, -2.0, 2.0)
        + congestion_gate * np.clip(norm_wait + norm_uncert, -2.0, 2.0)   # Congestion-aware load signal
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
