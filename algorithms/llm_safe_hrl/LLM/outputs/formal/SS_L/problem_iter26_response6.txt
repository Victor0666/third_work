def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust median-MAD normalization and sharpened slack gates 
       with Parent 1's validated critical-path coupling and slack-aware uncertainty gating.
       Key improvements:
         - Replaces fixed uncertainty_gate_threshold with adaptive version: threshold rises as slack approaches zero
         - Introduces dedicated starvation relief *only* under deadline pressure (slack <= 0), decoupled from congestion
         - Combines both critical-path signals: (1) boosted_rank (Parent 2) + (2) critical_coupling_under_pressure (Parent 1)
         - Uses clipped MAD-normalized slack for both penalty and gating to enhance sensitivity near deadlines
         - All terms bounded via [-2,2] clipping; final score deterministic, finite, shape-(N,)."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Adaptive uncertainty gating: threshold increases smoothly as slack approaches 0 from above ===
    # Enables risk-aware gating only when DDL-feasible AND uncertainty exceeds context-sensitive bound
    adaptive_uncert_thresh = PARAMS["uncertainty_gate_threshold"] * (1.0 - 1.0 / (1.0 + np.exp(-2.0 * slack)))
    uncert_gate_adaptive = (norm_uncert > adaptive_uncert_thresh).astype(float) * (slack >= 0.0).astype(float)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Deadline pressure gate: binary for slack <= 0 (activates urgency & starvation relief) ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Critical path leverage: upward_rank × remaining_work interaction, activated *only* under deadline breach ===
    critical_coupling = norm_rank * norm_work * ddl_breach * PARAMS["critical_coupling_under_pressure"]
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded linear rank-slack coupling: stable monotonic amplification under pressure ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    
    # === Sharpened slack-pressure gate: sigmoid activation for pre-breach urgency ramp-up ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Uncertainty-duration interaction: dual-gated (adaptive uncertainty + slack pressure + DDL feasibility) ===
    duration_risk_score = norm_duration * uncert_gate_adaptive * slack_pressure * ddl_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus *only* under deadline pressure ===
    wait_benefit_pressure = (1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]))) * ddl_breach
    
    # === Energy-uncertainty coupling: gated by both adaptive uncertainty and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate_adaptive * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_coupling, -2.0, 2.0)                           # Deadline-pressure-only critical-path coupling
        - np.clip(boosted_rank, -2.0, 2.0)                                  # HEFT-based urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit_pressure, -2.0, 2.0)                        # Pressure-specific anti-starvation
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
