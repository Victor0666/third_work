def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces robust_normalize with mean-abs scaling for stability;
       introduces conditional DDL-protection gate; replaces exponential wait relief with tanh-based anti-starvation;
       adds successor-release interaction via remaining_work * slack coupling under deadline pressure."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Stable feature normalization: mean(|x|) + eps scaling (not median/MAD) for bounded sensitivity ===
    def stable_normalize(x):
        x_abs = np.abs(x)
        scale = np.mean(x_abs) + eps
        return x / (scale + eps)
    
    # Normalize base features
    norm_slack = stable_normalize(slack)
    norm_energy = stable_normalize(min_incremental_energy)
    norm_duration = stable_normalize(min_exec_time + min_comm_time)
    norm_rank = stable_normalize(upward_rank)
    norm_work = stable_normalize(remaining_work)
    norm_wait = stable_normalize(ready_wait_time)
    norm_uncert = stable_normalize(uncertainty)
    
    # === DDL-protection gate: activated only when slack is critically low (negative or near-zero) ===
    # Uses smooth tanh to avoid step discontinuities; centered at slack=0 with tunable width
    slack_pressure = np.tanh(-norm_slack * PARAMS["slack_pressure_gate_steepness"])
    ddl_protection_gate = np.clip(slack_pressure, 0.0, 1.0)
    
    # === Boosted rank: only active under DDL pressure, prevents over-amplification in safe regions ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_protection_gate)
    
    # === Critical-path successor release interaction: reward tasks that unblock large downstream work *only* under DDL pressure ===
    # This directly addresses 'critical_path_or_successor_blocked' counterfactual evidence
    successor_release_bonus = norm_work * ddl_protection_gate
    
    # === Starvation relief: bounded tanh-based, avoids saturation artifacts of exp decay ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Uncertainty-duration risk score: only activated when both uncertainty and slack pressure are high ===
    uncert_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]) & (ddl_protection_gate > PARAMS["ddl_protection_gate_threshold"])
    duration_risk_score = norm_duration * uncert_gate.astype(float) * ddl_protection_gate
    
    # === Energy-uncertainty interaction: multiplicative penalty, scaled by DDL pressure to focus on risky critical tasks ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_protection_gate * uncert_gate.astype(float)
    
    # === Slack penalty: power law on raw negative slack, then stabilized normalization ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = stable_normalize(raw_slack_penalty)
    
    # === Composite score: smaller = higher priority ===
    # Prioritizes DDL safety first (penalty + protection), then energy/efficiency, then anti-starvation and successor release
    score = (
        + norm_slack_penalty                           # Primary lateness risk penalty
        - boosted_rank                                 # Critical-path boost under pressure
        - PARAMS["energy_sensitivity"] * norm_energy  # Energy efficiency preference
        - norm_duration                                # Duration preference
        - wait_benefit                                 # Anti-starvation (tanh-scaled)
        + PARAMS["duration_robustness"] * duration_risk_score  # Risk-aware slowdown penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned energy penalty
        - PARAMS["remaining_work_weight"] * successor_release_bonus  # Release large downstream work under DDL pressure
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
