def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid gates with bounded linear interpolation;
       uses sign-preserving MAD normalization; removes inactive parameters (duration_robustness, wait_saturation_offset);
       introduces critical successor release interaction via upward_rank * remaining_work under ddl_gate;
       applies clipped linear DDL-protection gate instead of soft sigmoid for stronger hard-constraint enforcement;
       prioritizes tasks with smallest slack first, then critical path impact, then starvation relief."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Sign-preserving MAD normalization — robust to outliers, preserves directionality ===
    def mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = np.maximum(mad, eps)
        return (x - med) / spread
    
    norm_slack = mad_normalize(slack)
    norm_energy = mad_normalize(min_incremental_energy)
    norm_duration = mad_normalize(min_exec_time + min_comm_time)
    norm_rank = mad_normalize(upward_rank)
    norm_work = mad_normalize(remaining_work)
    norm_wait = mad_normalize(ready_wait_time)
    norm_uncert = mad_normalize(uncertainty)
    
    # === Hard DDL-protection gate: linear ramp from 0→1 over [0, 1] slack units (clipped), zero for slack < 0 ===
    # Stronger enforcement than sigmoid: ensures no energy penalty under breach
    ddl_gate = np.clip(slack / (np.abs(np.median(slack)) + eps), 0.0, 1.0)
    
    # === Critical-path coupling: upward_rank * remaining_work interaction, only active when ddl_gate > 0 ===
    # Validated by cross-seed replay: captures blocking effect on successor release
    critical_coupling = norm_rank * norm_work * ddl_gate
    
    # === Negative-slack penalty: clipped linear penalty scaled by exponent, normalized ===
    raw_neg_slack = np.maximum(-slack, 0.0)
    slack_penalty = (raw_neg_slack ** PARAMS["slack_penalty_exponent"]) * (1.0 + PARAMS["criticality_scale"] * norm_rank)
    norm_slack_penalty = mad_normalize(slack_penalty)
    
    # === Slack pressure: monotonic urgency ramp using bounded linear interpolation over [-2,2] range ===
    # Replaces unstable sigmoid; avoids overflow and preserves ordering
    slack_pressure = np.clip(norm_slack, -2.0, 2.0)
    pressure_ramp = np.clip((slack_pressure + 2.0) / 4.0, 0.0, 1.0)  # maps [-2,2] → [0,1]
    
    # === Uncertainty gating: step-like activation above threshold, smoothed with linear ramp ===
    uncert_norm = np.clip(norm_uncert, 0.0, 2.0)
    uncert_gate = np.clip((uncert_norm - PARAMS["uncertainty_gate_threshold"]) / (eps + 0.5), 0.0, 1.0)
    
    # === Starvation relief: bounded exponential decay with explicit saturation ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    wait_benefit = np.clip(wait_benefit, 0.0, 0.99)
    
    # === Energy-uncertainty penalty: only applied when both uncertainty is high AND slack permits ===
    energy_uncert_penalty = norm_energy * uncert_gate * ddl_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness risk
        - np.clip(critical_coupling, -2.0, 2.0)                            # Unblock critical successors
        - np.clip(norm_rank * (1.0 + PARAMS["criticality_scale"] * pressure_ramp), -2.0, 2.0)  # Pressure-amplified rank
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Feasible-region energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation boost
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * ddl_gate, -2.0, 2.0)  # Work-aware only when feasible
        + np.clip(pressure_ramp * norm_slack, -2.0, 2.0)                   # Direct slack-pressure term
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
