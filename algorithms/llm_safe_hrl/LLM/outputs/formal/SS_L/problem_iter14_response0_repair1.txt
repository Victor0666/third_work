def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's strict DDL-dominant energy disable gate and logistic starvation relief
       with Parent 1's robust slack-margin-ratio-based urgency detection and successor-release interaction.
       Novel improvement: dual-gated energy modulation — combines both safe_slack_mask AND uncertainty-aware activation
       to avoid over-suppression under high uncertainty even with moderate slack; ensures risk-aware energy tradeoffs
       only when both feasibility and uncertainty conditions are jointly satisfied."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median + MAD per feature (preserves sign semantics and outlier resilience) ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Dual-trigger DDL protection gate: uses both raw slack sign AND normalized margin ratio ===
    # More resilient than single-threshold logic — captures both hard violations and relative tightness
    slack_margin_ratio = np.abs(slack) / (min_exec_time + min_comm_time + eps)
    ddl_urgent = np.where((slack <= 0.0) | (slack_margin_ratio < PARAMS["safe_slack_energy_disable_threshold"]), 1.0, 0.0)
    
    # === Hard DDL-dominant energy disable gate: fully suppress energy terms when slack is safe ===
    # Enforces strict feasibility-first behavior: energy only matters when slack is risky
    safe_slack_mask = (norm_slack > PARAMS["safe_slack_energy_disable_threshold"])
    energy_enabled = np.where(safe_slack_mask, 0.0, 1.0)
    
    # === Dual-gated energy modulation: requires BOTH urgency AND high uncertainty to activate energy penalty ===
    # Prevents unnecessary energy suppression under low uncertainty, even when slack is tight — enables informed trade-offs
    energy_modulation_gate = np.where(
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) & (ddl_urgent == 1.0),
        1.0,
        0.0
    )
    
    # === Sharpened slack-pressure gate: sigmoid over normalized slack for smooth, bounded transition ===
    slack_pressure_norm = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_norm - 1.0)))
    
    # === Successor-release interaction: amplify impact of unblocking high-work successors (Parent 1 strength) ===
    successor_release_impact = norm_work * norm_rank
    
    # === Starvation relief: logistic with explicit saturation (Parent 2 strength) ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Duration robustness: risk-weighted penalty only under deadline pressure ===
    duration_risk_score = norm_duration * norm_uncert * ddl_urgent
    
    # === Energy-uncertainty interaction: multiplicative penalty, gated by both safety and urgency ===
    energy_uncert_penalty = norm_energy * norm_uncert * energy_modulation_gate * energy_enabled
    
    # === Composite score: smaller = higher priority ===
    # Strict hierarchy: (1) deadline bias dominates, (2) energy penalties conditional on dual risk triggers,
    # (3) anti-starvation and successor release provide secondary balancing signals
    score = (
        + np.clip(norm_slack, -2.0, 2.0)                                  # Primary DDL signal: lower slack → lower score
        + np.clip(norm_slack ** PARAMS["slack_penalty_exponent"], -2.0, 2.0) * ddl_urgent  # Strong nonlinear penalty only when urgent
        - np.clip(norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate * ddl_urgent), -2.0, 2.0)  # Urgency-gated criticality
        - np.clip(successor_release_impact, -2.0, 2.0)                    # Prefer tasks unlocking large downstream work
        - np.clip(PARAMS["energy_sensitivity"] * norm_energy * energy_modulation_gate * energy_enabled, -2.0, 2.0)  # Dual-gated energy penalty
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation bonus
        + np.clip(PARAMS["duration_robustness"] * duration_risk_score, -2.0, 2.0)  # Risk-aware duration penalty
        + np.clip(PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty, -2.0, 2.0)  # Risk-aligned marginal energy penalty
        + np.clip(PARAMS["remaining_work_weight"] * norm_work, -2.0, 2.0)  # Mild preference for large downstream scope
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
