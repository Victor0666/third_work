def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining best elements from both parents with novel smooth modulation.
    
    Key improvements:
    - Replaces hard boolean gates with smooth tanh-based slack pressure: avoids priority jumps at threshold crossings.
    - Introduces uncertainty-coupled energy penalty: `energy * (1 + uncertainty * energy_uncertainty_coupling)` for joint risk amplification.
    - Adds successor-release signal: `remaining_work * (1 - clip(slack/median_slack, 0, 1))` to unblock critical paths under deadline pressure.
    - Preserves robust normalization, finite safeguards, and deterministic structure.
    - All numeric literals are only -2, -1, 0, 1, or 2; all tunables are declared in PARAMETER_SCHEMA.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Sanitize inputs: ensure float, replace NaN/inf with safe bounded values
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust normalization: x / (mean(|x|) + eps), handles zeros & negatives safely
    def robust_norm(x):
        x = np.asarray(x)
        mean_abs = np.mean(np.abs(x))
        scale = mean_abs + eps
        return x / scale
    
    # 1. Smooth slack pressure using tanh — avoids discontinuities at slack=0
    median_slack = np.median(np.abs(slack)) + eps
    slack_pressure = np.tanh(slack / (eps + PARAMS["slack_pressure_smoothing"] * median_slack))
    
    # 2. Negative slack penalty: sharp, superlinear penalty only for lateness risk
    neg_slack = np.maximum(0.0, -slack)
    slack_penalty = np.power(neg_slack + eps, PARAMS["slack_penalty_exponent"])
    
    # 3. Risk-adjusted duration: convex blend of duration and uncertainty
    duration = min_exec_time + min_comm_time
    duration_norm = robust_norm(duration)
    uncertainty_norm = robust_norm(uncertainty)
    risk_adjusted_duration = (
        PARAMS["duration_risk_ratio"] * duration_norm +
        (1.0 - PARAMS["duration_risk_ratio"]) * uncertainty_norm
    )
    
    # 4. Energy: apply uncertainty-coupled penalty — higher uncertainty amplifies energy cost
    energy_norm = robust_norm(min_incremental_energy)
    energy_uncertainty_factor = 1.0 + PARAMS["energy_uncertainty_coupling"] * uncertainty_norm
    coupled_energy_penalty = PARAMS["energy_efficiency_scale"] * energy_norm * energy_uncertainty_factor
    
    # 5. Upward rank: modulated by smooth slack pressure (not binary gate)
    rank_base = robust_norm(upward_rank)
    rank_weighted = rank_base * (1.0 + PARAMS["rank_slack_coupling"] * np.maximum(0.0, -slack_pressure))
    
    # 6. Criticality boost: activate only when rank ≥ median AND slack ≤ ratio*median_slack
    median_rank = np.median(upward_rank)
    is_critical = (
        (upward_rank >= median_rank) &
        (slack <= np.maximum(eps, PARAMS["slack_median_ratio_threshold"] * median_slack))
    ).astype(float)
    criticality_bonus = is_critical * PARAMS["criticality_boost"]
    
    # 7. Successor-release signal: prioritize unblocking high remaining_work when slack is tight
    work_norm = robust_norm(remaining_work)
    slack_tightness = np.clip(slack / (eps + median_slack), 0.0, 1.0)
    successor_release = PARAMS["successor_release_weight"] * work_norm * (1.0 - slack_tightness)
    
    # 8. Anti-starvation: linear fairness bonus (reduces score for long-waiting tasks)
    wait_norm = robust_norm(ready_wait_time)
    wait_bonus = -PARAMS["wait_fairness_gain"] * wait_norm  # negative → improves priority
    
    # 9. Uncertainty-slack gate: tighten effective penalty when uncertainty is high and slack is low
    unc_gate_active = (
        (uncertainty_norm > PARAMS["uncertainty_slack_gate"]) &
        (slack < eps)
    ).astype(float)
    tightened_penalty = slack_penalty + unc_gate_active * PARAMS["uncertainty_slack_tightening_weight"] * uncertainty_norm
    
    # Composite score: smaller = higher priority
    score = (
        tightened_penalty
        + coupled_energy_penalty
        + 1.0 * risk_adjusted_duration
        - 1.0 * rank_weighted
        + 0.0 * criticality_bonus  # neutralized: not dominant in smoothed regime
        - 1.0 * successor_release
        + wait_bonus
    )
    
    # Final safeguard: replace NaN/inf with finite machine bounds
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values in priority score"
    
    return score
