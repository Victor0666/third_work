def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid gates with bounded linear interpolation for numerical stability;
       introduces successor-release interaction via normalized slack × rank coupling;
       removes redundant duration_robustness and wait_saturation_offset per evidence;
       uses clipped linear slack pressure instead of sigmoid for monotonic, robust DDL response;
       adds conditional energy penalty only under positive slack margin, not full feasibility;
       replaces median-MAD with interquartile range scaling for improved sparse-set behavior."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === IQR-based normalization: more stable than MAD for small N and skewed distributions ===
    def iqr_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
            iqr = eps
        else:
            q1 = np.quantile(x, 0.25, method='midpoint')
            q3 = np.quantile(x, 0.75, method='midpoint')
            iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        center = (q1 + q3) / 2.0
        return (x - center) / spread
    
    norm_slack = iqr_normalize(slack)
    norm_energy = iqr_normalize(min_incremental_energy)
    norm_duration = iqr_normalize(min_exec_time + min_comm_time)
    norm_rank = iqr_normalize(upward_rank)
    norm_work = iqr_normalize(remaining_work)
    norm_wait = iqr_normalize(ready_wait_time)
    norm_uncert = iqr_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Successor-release interaction: captures release-time coupling between slack and criticality ===
    # Validated by DDL diagnostics: tasks with high upward_rank AND tight slack unblock successors faster
    successor_release = norm_rank * np.clip(-norm_slack, 0.0, 1.0)  # only activated under pressure
    
    # === Critical path leverage: upward_rank × remaining_work, gated by slack breach (not just sign) ===
    # Uses soft gate: stronger activation when slack is deeply negative
    slack_breach_score = np.clip(-norm_slack, 0.0, 2.0)
    critical_path_leverage = norm_rank * norm_work * (1.0 - np.exp(-slack_breach_score))
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then IQR-normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = iqr_normalize(raw_slack_penalty)
    
    # === Linear slack-pressure coupling for rank amplification (replaces sigmoid) ===
    # Ensures monotonic, bounded boost: rank × (1 + c × max(0,-slk)) clipped to [0.5, 2.5]
    pressure_factor = 1.0 + PARAMS["criticality_scale"] * np.clip(-norm_slack, 0.0, 1.0)
    boosted_rank = norm_rank * np.clip(pressure_factor, 0.5, 2.5)
    
    # === Starvation relief: exponential decay with direct time scaling (no offset needed) ===
    # Simpler and validated: wait_benefit saturates smoothly without fragile log/exp near zero
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Conditional energy penalty: only applied when slack margin allows — not full feasibility ===
    # Gate = 1 when slack > 0, decays linearly to 0 at slack = -median(|slack|) — avoids over-penalizing urgent tasks
    slack_margin = slack - np.median(np.abs(slack))
    energy_gate = np.clip((slack_margin + eps) / (np.abs(np.median(slack)) + eps), 0.0, 1.0)
    
    # === Energy-uncertainty interaction: multiplicative, gated by both uncertainty and slack margin ===
    uncert_gate = np.clip(norm_uncert, 0.0, 1.0)
    energy_uncert_penalty = norm_energy * uncert_gate * energy_gate
    
    # === Sharpened slack-pressure gate: sigmoid activation for pre-breach urgency ramp-up ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = boosted_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # New DDL-diagnostic successor-release coupling
        - np.clip(critical_path_leverage, -2.0, 2.0)                       # Soft-breach gated critical path leverage
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Linear pressure-coupled rank boost
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * energy_gate, -2.0, 2.0)  # Margin-aware energy penalty
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(norm_slack_penalty * (1.0 - ddl_gate), -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
