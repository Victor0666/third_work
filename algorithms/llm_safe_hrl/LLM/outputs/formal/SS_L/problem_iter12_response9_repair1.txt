def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: introduces conditional successor-release coupling with dual gating (slack_pressure AND rank threshold);
       replaces raw_duration with norm_duration in risk score for scale consistency; enforces hard DDL protection via +inf clamping;
       uses median-MAD normalization throughout; all parameters declared and used.
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, preserves sign ===
    def median_mad_normalize(x):
        x = np.copy(x)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    
    # === Risk-aware slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate: sigmoid with tunable steepness ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    
    # === Dynamic criticality threshold for successor-release coupling: use fixed median (no extra param) ===
    rank_threshold = np.median(upward_rank)
    rank_above_thresh = (upward_rank > rank_threshold).astype(float)
    
    # === Conditional successor-release coupling: activated only under dual pressure ===
    # Ensures amplification occurs *only* when both slack pressure is high AND task is above median importance
    successor_release_active = np.where(
        (slack_pressure > PARAMS["successor_release_gate_threshold"]) & (rank_above_thresh > 0.5),
        1.0, 0.0
    )
    critical_path_leverage = norm_rank * norm_work * successor_release_active
    
    # === Boosted rank: HEFT urgency under pressure, decoupled from leverage logic ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Uncertainty-duration interaction: now uses norm_duration for consistent scaling ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = norm_duration * uncert_gate * slack_pressure
    
    # === Starvation relief: logistic-shaped anti-starvation bonus using raw wait time for physical grounding ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty coupling: multiplicative penalty ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate
    
    # === Composite score: smaller = higher priority ===
    score = (
        + norm_slack_penalty                           # Lateness penalty
        - boosted_rank                                 # Critical-path urgency
        - PARAMS["criticality_scale"] * critical_path_leverage  # Conditional successor-release boost
        - PARAMS["energy_sensitivity"] * norm_energy  # Energy minimization
        - wait_benefit                                 # Starvation relief
        + PARAMS["duration_robustness"] * duration_risk_score  # Robustness penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Joint risk penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Downstream work preference
    )
    
    # === Hard DDL protection: clamp score to +inf for any task violating deadline constraint ===
    # Enforces strict feasibility before energy optimization — satisfies hard deadline requirement
    deadline_violated = (slack < -eps)
    score = np.where(deadline_violated, np.finfo(float).max, score)
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
