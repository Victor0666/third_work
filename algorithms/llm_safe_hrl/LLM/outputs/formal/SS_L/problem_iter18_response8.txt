def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: reinstates validated critical_path_leverage with unified DDL-feasibility gating;
       replaces dual-gated energy with single robust smooth Heaviside (ddl_gate) — eliminates instability from linear ramp;
       introduces *normalized slack alignment* via clipped, signed slack deviation to decouple urgency from outlier magnitude;
       retains median-MAD normalization for all features; removes redundant duration_robustness & wait_saturation_offset;
       enforces strict monotonic slack pressure through bounded slack_alignment term;
       all intermediate terms clipped to [-2,2] for stability and AST depth control."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Unified DDL-feasibility gate: smooth Heaviside only — no linear ramp (removes instability) ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Normalized slack alignment: signed, clipped deviation from zero slack (tightness anchor) ===
    # Provides monotonic, bounded urgency signal without outlier amplification
    slack_alignment = np.clip(norm_slack, -1.0, 1.0)
    
    # === Critical path leverage: upward_rank × remaining_work × ddl_breach — reinstated and validated ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Raw slack penalty (dominant term) ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate for rank modulation under tight-but-feasible deadlines ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    
    # === Boosted rank: scale upward_rank by slack_alignment and gate it sharply near deadline margin ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_alignment) * rank_gate
    
    # === Uncertainty gating: activated only above threshold and under slack pressure ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: simplified exponential decay without saturation offset (removed redundancy) ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Slack-aware energy scaling: uses slack_alignment (not raw slack) for smoother, bounded coupling ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_alignment) * ddl_gate
    
    # === Composite score: smaller = higher priority ===
    # Prioritizes: (1) lateness penalty, (2) critical-path unblocking on breach, (3) rank amplification under alignment,
    # (4) energy minimization only when DDL-safe, (5) anti-starvation, (6) energy-uncertainty risk
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Validated breach-specific leverage
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Alignment-aware criticality
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Unified DDL-safe energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Simplified starvation relief
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
