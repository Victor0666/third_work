def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: 12 parameters, all used; removes slack_normalization_bias to comply with count limit;
       retains binary DDL gate, linear starvation ramp, median-MAD normalization, and physically grounded slack-pressure sig;
       ensures monotonic, bounded, deterministic output with strict shape enforcement."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature: outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        scale = mad + eps if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / scale
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Hard DDL protection gate: binary mask for strict violation handling ===
    ddl_urgent_mask = (slack <= 0.0).astype(float)
    
    # === Risk-aware slack penalty: power-law overdue penalty, only active when urgent ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Slack-pressure gated criticality: sigmoid over *raw* slack (not normalized) for stability near zero ===
    slack_pressure_sig = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack + eps)))
    rank_gate = ddl_urgent_mask * slack_pressure_sig + (1.0 - ddl_urgent_mask) * PARAMS["ddl_urgent_default_gate"]
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Duration-uncertainty interaction: raw-duration risk score under dual pressure ===
    uncert_gate = np.where(uncertainty > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = (min_exec_time + min_comm_time) * uncert_gate * ddl_urgent_mask
    
    # === Starvation relief: bounded linear ramp — monotonic, predictable, finite derivative ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]), 0.0, 1.0)
    
    # === Energy-uncertainty interaction: additive penalty only under DDL urgency AND high uncertainty ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_urgent_mask * uncert_gate
    
    # === Composite score: smaller = higher priority ===
    score = (
        + norm_slack_penalty                           # Lateness risk dominates
        - boosted_rank                                 # Critical path amplified where needed
        - PARAMS["energy_sensitivity"] * norm_energy  # Prefer low marginal energy
        - wait_benefit                                 # Starvation relief (linear, bounded)
        + PARAMS["duration_robustness"] * median_mad_normalize(duration_risk_score)  # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Targeted penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Downstream workload awareness
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
