def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: replaces hardcoded 0.25/0.75 with tunable quantiles;
       uses sign-preserving MAD-normalized slack for sharp deadline violation sensitivity;
       applies DDL-protection gate only for slack >= 0 to enable energy/uncertainty penalties exclusively when feasible;
       includes upward_rank × remaining_work interaction under deadline pressure;
       introduces congestion signal (ready_wait_time + uncertainty) gated by DDL feasibility and uncertainty threshold;
       uses clipped linear wait_benefit for low-wait discriminability."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Quantile-based robust normalization: tunable Q1/Q3 range, outlier-resilient ===
    def quantile_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])
            q3 = np.quantile(x, PARAMS["quantile_q3"])
        iqr = q3 - q1 if q3 > q1 else eps
        return (x - q1) / iqr
    
    norm_slack = quantile_normalize(slack)
    norm_energy = quantile_normalize(min_incremental_energy)
    norm_duration = quantile_normalize(min_exec_time + min_comm_time)
    norm_rank = quantile_normalize(upward_rank)
    norm_work = quantile_normalize(remaining_work)
    norm_wait = quantile_normalize(ready_wait_time)
    norm_uncert = quantile_normalize(uncertainty)
    
    # === Sign-preserving MAD normalization *only for slack* to sharpen urgency near deadline violation ===
    slack_med = np.median(slack)
    slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = slack_mad if slack_mad > eps else eps
    norm_slack_mad = (slack - slack_med) / slack_spread
    
    # === DDL-protection gate: sigmoid activated only for slack >= 0 to enable energy/uncertainty penalties exclusively when DDL-feasible ===
    ddl_feasible = (slack >= 0.0).astype(float)
    ddl_protection_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (slack - 0.0)))
    
    # === Critical-path leverage: upward_rank × remaining_work interaction, activated only under deadline pressure (slack <= 0) ===
    ddl_pressure = (slack <= 0.0).astype(float)
    critical_leverage = norm_rank * norm_work * ddl_pressure
    
    # === Congestion signal: ready_wait_time + uncertainty, gated by both DDL feasibility AND high-congestion threshold ===
    congestion = ready_wait_time + uncertainty
    congestion_norm = quantile_normalize(congestion)
    congestion_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (congestion_norm - PARAMS["uncertainty_gate_threshold"])))
    congestion_signal = congestion_norm * congestion_gate * ddl_feasible
    
    # === Starvation relief: clipped linear wait_benefit (not logistic) for improved low-wait discriminability ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 2.0)
    
    # === Energy-uncertainty coupling: enabled only when DDL-feasible AND uncertainty exceeds threshold ===
    energy_uncert_coupling = norm_energy * norm_uncert * congestion_gate * ddl_feasible
    
    # === Slack-driven energy amplification: energy penalty scaled by normalized slack pressure (negative slack → higher weight) ===
    slack_pressure = np.clip(-norm_slack_mad, 0.0, 2.0)  # 0–2 range for monotonic pressure ramp
    energy_slack_amplifier = 1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure
    
    # === Composite score: smaller = higher priority; all terms bounded and stabilized ===
    score = (
        + np.clip(norm_slack_mad, -2.0, 2.0)                              # Sharpened urgency via MAD-normalized slack
        - np.clip(critical_leverage, -2.0, 2.0)                           # DDL-validated critical path boost
        - np.clip(norm_rank, -2.0, 2.0)                                   # Baseline critical-path importance
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_protection_gate, -2.0, 2.0)  # Feasibility-gated energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                # Linear anti-starvation
        + np.clip(congestion_signal, -2.0, 2.0)                           # Jointly gated congestion relief
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_coupling, -2.0, 2.0)
        + np.clip(norm_energy * energy_slack_amplifier * ddl_feasible, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
