def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robustness from Parent 2 and structured successor-release coupling from Parent 1.
       Key improvements:
         - Replaces brittle boolean gates with smooth, differentiable tanh-based urgency and gating;
         - Introduces explicit successor-release coupling: norm_work * norm_rank * tanh(-norm_slack) to reward unblocking critical paths;
         - Uses unified risk-aware scaling (mean-abs + eps) for all features — no median or std-based normalization;
         - Removes all redundant parameters (duration_robustness, gate_steepness) and fragile conditional branches;
         - Enforces strict monotonic urgency via tanh(-slack * urgency_smoothing_factor);
         - Maintains deterministic, finite, shape-(N,) output with full NaN/inf protection.
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust feature scaling: mean(|x|) + eps as scale (no centering to avoid sign inversion artifacts) ===
    def scale_feature(x):
        x_abs = np.abs(x)
        spread = np.mean(x_abs) + eps
        return x / (spread + eps)
    
    # Normalize base features using robust absolute-mean scaling
    norm_slack = scale_feature(slack)
    norm_energy = scale_feature(min_incremental_energy)
    norm_duration = scale_feature(min_exec_time + min_comm_time)
    norm_rank = scale_feature(upward_rank)
    norm_work = scale_feature(remaining_work)
    norm_wait = scale_feature(ready_wait_time)
    norm_uncert = scale_feature(uncertainty)
    
    # === Smooth urgency signal: tanh-based, monotonic, bounded [-1, 1], centered at slack=0 ===
    # Stronger response to negative slack, gentle for positive slack — avoids discontinuities
    urgency_signal = np.tanh(-norm_slack * PARAMS["urgency_smoothing_factor"])
    
    # === DDL-protection gate: smooth activation (not binary) based on urgency and low uncertainty ===
    # Enables gradient-based optimization while preserving physical meaning
    uncert_gate = np.tanh((PARAMS["uncertainty_gate_threshold"] - norm_uncert) * 2.0)
    ddl_protection_gate = np.clip(urgency_signal * uncert_gate, 0.0, 1.0)
    
    # === Boosted rank: only activated under safe+urgent conditions, smoothly scaled ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_protection_gate)
    
    # === Successor-release coupling: product of work and rank, gated by urgency — captures 'unblock critical mass' ===
    # More physically grounded than raw work or rank alone; reflects counterfactual evidence
    successor_release_score = (
        norm_work * norm_rank * PARAMS["successor_release_coupling_strength"] * ddl_protection_gate
    )
    
    # === Bounded anti-starvation: tanh-based, stable, saturates at ±1, avoids exponential drift ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Joint risk-energy penalty: only active when both urgency and uncertainty are high ===
    risk_energy_penalty = (
        norm_energy * norm_uncert * ddl_protection_gate * (1.0 - uncert_gate)
    )
    
    # === Slack penalty: bounded, smooth, monotonic — replaces power law with tanh-derived urgency ===
    # Eliminates numerical instability of exponentiation on negative values and ensures finite gradient
    norm_slack_penalty = urgency_signal
    
    # === Composite score: smaller = higher priority ===
    # Ordering: (1) Urgency (primary constraint), (2) Critical path boost, (3) Successor release,
    #           (4) Anti-starvation, (5) Risk-energy penalty, (6) Energy efficiency, (7) Downstream weight
    score = (
        + norm_slack_penalty                           # Primary DDL compliance signal
        - boosted_rank                                 # Critical-path amplification under safety+urgency
        - successor_release_score                      # Unblock large critical downstream work
        - wait_benefit                                 # Prevent starvation, bounded
        + PARAMS["energy_uncertainty_interaction"] * risk_energy_penalty  # Penalize risky high-energy choices
        - PARAMS["energy_sensitivity"] * norm_energy  # Prefer low marginal energy
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for high downstream load
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
