def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with strict hard-deadline enforcement:
       - Introduces *finite but dominant* hard_ddl_violation_penalty applied only when slack < slack_sensitivity_center,
         replacing infinite scores (which break gradient-based tuning) while preserving DDL-first dominance.
       - Restores slack_pressure_gate_steepness (dropped in v1) for sharper, more responsive critical-path activation.
       - Removes exec_comm_balance and energy_uncertainty_interaction per reflection — both were overfitting noise;
         instead uses unified slack-conditioned penalty structure with single sensitivity center.
       - Keeps robust median-MAD normalization and bounded linear starvation from v2 baseline.
       - All intermediate terms clipped to [-2,2] and final score safeguarded with nan_to_num using machine precision bounds."""
    
    eps = PARAMS["epsilon"]
    
    # Safe cast; no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — applied only to core decision features ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    
    # === Hard deadline violation flag: triggers dominant finite penalty ===
    ddl_violated = (slack < PARAMS["slack_sensitivity_center"]).astype(float)
    hard_ddl_penalty = ddl_violated * PARAMS["hard_ddl_violation_penalty"]
    
    # === Slack feasibility gate: enables energy preference only when safe ===
    slack_feasibility = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Robust slack penalty: capped power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Adaptive clipped linear hinge for slack-driven urgency coupling ===
    slack_hinge = np.clip(-norm_slack, 0.0, PARAMS["rank_slack_hinge_width"])
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * (slack_hinge / (PARAMS["rank_slack_hinge_width"] + eps))
    
    # === Sharpened urgency gate centered at sensitivity boundary (restored from v0) ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * urgency_gate)
    
    # === Bounded linear starvation relief ===
    wait_benefit = np.clip(1.0 - PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + hard_ddl_penalty                                                 # Strict DDL-first enforcement
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency-amplified critical path
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * slack_feasibility, -2.0, 2.0)  # Energy minimization only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief
        + PARAMS["slack_penalty_exponent"] * np.clip(norm_work * ddl_violated, -2.0, 2.0)  # Downstream work prioritization under breach
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
