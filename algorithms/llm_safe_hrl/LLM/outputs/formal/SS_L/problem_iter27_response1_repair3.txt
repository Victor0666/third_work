def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: adaptive piecewise-linear urgency; additive critical-path coupling;
       continuous feasibility-aware energy suppression (0→1 over slack); robust normalization; strict [-2,2] clipping."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization: sign-preserving, outlier-resilient scaling ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad + eps
        return (x - med) / spread
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Hard DDL-feasibility gate: binary suppression — zero energy/uncert penalties if slack < 0 ===
    hard_feasibility_gate = np.where(slack < 0.0, 0.0, 1.0)
    
    # === Smooth, bounded ddl-pressure: finite-floor inverse mapping ===
    slack_abs = np.abs(slack) + eps
    ddl_pressure = np.clip(1.0 / (1.0 + slack_abs), 0.0, 1.0)
    
    # === Adaptive piecewise-linear urgency: linear near deadline, saturated beyond transition point
    # Ensures monotonicity, avoids over-penalization for moderate lateness, and handles large positive slack smoothly
    urgency_transition = PARAMS["urgency_transition_slack"]
    urgency_sat_slope = PARAMS["urgency_saturation_slope"]
    raw_slack_penalty = np.maximum(-slack, 0.0)
    # Linear branch: urgent when slack <= urgency_transition (i.e., more negative or equal)
    linear_urgency = raw_slack_penalty
    # Saturated branch: small constant slope for slack > urgency_transition → prevents artificial priority inversion
    saturated_urgency = urgency_sat_slope * (slack - urgency_transition)
    piecewise_urgency = np.where(
        slack <= urgency_transition,
        linear_urgency,
        np.maximum(saturated_urgency, 0.0)  # ensure non-negative
    )
    norm_slack_penalty = robust_normalize(piecewise_urgency + eps)
    slack_penalty = PARAMS["slack_penalty_linear_coeff"] * norm_slack_penalty
    
    # === Additive critical-path coupling: rank × work × pressure — interpretable, no collapse under low pressure ===
    successor_release = norm_rank * norm_work * ddl_pressure
    
    # === Uncertainty-aware energy penalty: gated by feasibility AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * hard_feasibility_gate
    
    # === Starvation relief: clipped linear ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 2.0)
    
    # === Robustified duration penalty: damped and fully gated ===
    duration_penalty = PARAMS["duration_robustness_factor"] * norm_duration * ddl_pressure * hard_feasibility_gate
    
    # === Continuous feasibility-aware energy suppression: smoothly interpolates from 0 (slack << 0) to 1 (slack >> 0)
    # Uses linear ramp over normalized slack range [-1, 1], centered at energy_suppression_midpoint with slope
    slack_norm_for_suppress = np.clip(norm_slack, -1.0, 1.0)
    continuous_suppress = np.clip(
        PARAMS["energy_suppression_midpoint"] + PARAMS["energy_suppression_slope"] * slack_norm_for_suppress,
        0.0, 1.0
    )
    suppressed_norm_energy = norm_energy * continuous_suppress
    
    # === Host-load surrogate gating: uses norm_duration * norm_uncert as congestion proxy ===
    load_surrogate = norm_duration * norm_uncert
    load_gate = np.where(
        (hard_feasibility_gate > 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]),
        np.clip(load_surrogate, 0.0, 2.0),
        0.0
    )
    host_load_penalty = PARAMS["host_load_sensitivity"] * load_gate * suppressed_norm_energy
    
    # === Composite score: smaller = higher priority; all components clipped individually to [-2,2] ===
    score = (
        + np.clip(slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                      # Additive release-driven criticality boost
        - PARAMS["energy_sensitivity"] * np.clip(suppressed_norm_energy, -2.0, 2.0)  # Continuously suppressed energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                           # Linear anti-starvation
        + np.clip(duration_penalty, -2.0, 2.0)                       # Damped, feasibility-gated duration penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)   # Downstream work bias
        + host_load_penalty                                          # Congestion-aware energy penalty
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
