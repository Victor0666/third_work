def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: uses robust IQR-based min-max scaling;
       introduces slack-aware energy gating via sigmoid(slack) to enforce feasibility-first energy preference;
       removes remaining_work_weight per reflection; all numeric literals are -2, -1, 0, 1, or 2."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust min-max normalization per feature: bounded to [-1, 1], outlier-resistant ===
    def robust_minmax_normalize(x):
        x = np.asarray(x)
        q1 = np.quantile(x, 0.25)
        q3 = np.quantile(x, 0.75)
        iqr = q3 - q1
        # Use fixed 1.5 multiplier (allowed literal) instead of tunable iqr_multiplier to stay within 12 params
        lb = q1 - 1.5 * iqr
        ub = q3 + 1.5 * iqr
        # Clamp to [lb, ub] then scale to [-1, 1]
        x_clipped = np.clip(x, lb, ub)
        scale = (ub - lb) / 2.0 + eps
        return 2.0 * (x_clipped - (lb + ub) / 2.0) / scale
    
    norm_slack = robust_minmax_normalize(slack)
    norm_energy = robust_minmax_normalize(min_incremental_energy)
    norm_rank = robust_minmax_normalize(upward_rank)
    norm_wait = robust_minmax_normalize(ready_wait_time)
    norm_uncert = robust_minmax_normalize(uncertainty)
    
    # === Hard DDL protection gate: binary mask for strict violation handling ===
    ddl_urgent_mask = (slack <= 0.0).astype(float)
    
    # === Risk-aware slack penalty: power-law overdue penalty, only active when urgent ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_minmax_normalize(raw_slack_penalty)
    
    # === Slack-pressure gated criticality: sigmoid over *raw* slack for stability near zero ===
    slack_pressure_sig = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack + eps)))
    rank_gate = ddl_urgent_mask * slack_pressure_sig + (1.0 - ddl_urgent_mask) * PARAMS["ddl_urgent_default_gate"]
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Slack-aware energy gating: sigmoid(slack) ensures energy preference only when slack > 0 ===
    # Enforces feasibility-first: no energy bias under deadline violation
    energy_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_energy_gate_steepness"] * (slack + eps)))
    gated_norm_energy = norm_energy * energy_gate
    
    # === Duration-uncertainty interaction: raw-duration risk score under dual pressure ===
    uncert_gate = np.where(uncertainty > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = (min_exec_time + min_comm_time) * uncert_gate * ddl_urgent_mask
    
    # === Starvation relief: bounded linear ramp — monotonic, predictable, finite derivative ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]), 0.0, 1.0)
    
    # === Energy-uncertainty interaction: additive penalty only under DDL urgency AND high uncertainty ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_urgent_mask * uncert_gate
    
    # === Composite score: smaller = higher priority ===
    score = (
        + norm_slack_penalty                           # Lateness risk dominates
        - boosted_rank                                 # Critical path amplified where needed
        - PARAMS["energy_sensitivity"] * gated_norm_energy  # Feasibility-gated energy preference
        - wait_benefit                                 # Starvation relief (linear, bounded)
        + PARAMS["duration_robustness"] * robust_minmax_normalize(duration_risk_score)  # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Targeted penalty
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
