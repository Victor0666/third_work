def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: restores monotonicity and smoothness via soft uncertainty coupling,
    applies uniform robust normalization to all time-domain features, and replaces brittle binary gates
    with continuous sigmoid-modulated interactions."""
    eps = PARAMS["epsilon"]
    
    # Convert inputs to float arrays without modifying in-place
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust MAD-based normalization — applied uniformly to *all* time/energy/work features
    # Uses signed values (not abs) to preserve directional semantics (e.g., slack sign matters)
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        scale = np.median(np.abs(x - center)) + eps
        return (x - center) / (scale + eps)
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)  # ✅ Fixed: now normalized consistently
    norm_uncert = robust_normalize(uncertainty)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    
    # Slack urgency: smooth tanh-based penalty/gain with declared clip bounds
    slack_urgency = (
        PARAMS["slack_risk_penalty"] * np.tanh(np.clip(-norm_slack, 0, PARAMS["slack_clip_upper"]))
        + PARAMS["slack_urgency_gain"] * np.tanh(np.clip(norm_slack, 0, PARAMS["slack_clip_upper_pos"]))
    )
    
    # Criticality gate: sigmoid-controlled attenuation of upward_rank under deadline pressure
    slack_pressure = np.clip(-norm_slack, 0, PARAMS["slack_clip_upper"])
    rank_gate = 1.0 / (1.0 + np.exp(PARAMS["rank_slack_gate"] * (slack_pressure - 2)))
    
    # ✅ Self-evolution: Replaces binary `np.where` with smooth, uncertainty-coupled criticality boost
    # Uses sigmoid of uncertainty × pressure to softly amplify rank importance *only* when both are high
    # Removed redundant parameter; reuses existing `uncertainty_slack_coupling` for continuity & count compliance
    unc_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_slack_coupling"] * (norm_uncert + slack_pressure)))
    criticality_term = PARAMS["criticality_weight"] * rank_gate * norm_rank * (1.0 + 0.5 * unc_pressure_gate)
    
    # Uncertainty-slack coupling: continuous tanh of product — preserves gradation & avoids discontinuities
    coupling = np.tanh(PARAMS["uncertainty_slack_coupling"] * norm_uncert * slack_pressure)
    
    # Duration fairness: penalizes deviation from median duration magnitude
    duration_fairness = PARAMS["duration_fairness"] * np.abs(norm_duration)
    
    # Anti-starvation: normalized wait boost — scale-invariant and bounded
    wait_boost = PARAMS["wait_bias"] * norm_wait
    
    # Work burden: de-prioritizes tasks with large downstream computational load
    work_term = -PARAMS["work_weight"] * norm_work
    
    # Total score: smaller = higher priority
    score = (
        slack_urgency
        + coupling
        + PARAMS["energy_scale"] * norm_energy
        + criticality_term
        + duration_fairness
        - wait_boost
        + work_term
    )
    
    # Final safeguard: ensure finite output, preserve shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
