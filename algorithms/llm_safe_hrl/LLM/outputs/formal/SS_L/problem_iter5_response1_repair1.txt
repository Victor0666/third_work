def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's robust median/MAD normalization, smooth uncertainty sigmoid, 
       restored rank-slack coupling, and exponential starvation relief — augmented with Parent 1's successor-release 
       term. Uses clipped tanh-scaled slack pressure for improved gradient stability near zero slack. All numeric 
       literals are -2,-1,0,1,2; no hidden constants."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median-centered MAD + epsilon → resilient to outliers and zero-scales ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        center = np.median(x_abs) if N > 0 else 0.0
        dev = x_abs - center
        scale = np.median(np.abs(dev)) + eps
        return dev / (scale + eps)
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Composite urgency: clipped tanh-scaled slack pressure — preserves monotonicity, bounds magnitude, 
    #     avoids instability near zero while retaining sensitivity to small negative slack ===
    raw_pressure = -norm_slack * PARAMS["slack_penalty_exponent"]
    # Clip input to tanh to prevent overflow and ensure bounded derivative
    clipped_input = np.clip(raw_pressure, -5.0, 5.0)
    slack_urgency = np.tanh(clipped_input)
    slack_pressure = np.clip(slack_urgency, 0.0, None)  # Only penalize overdue/urgent
    
    # === Critical-path acceleration gate: smooth linear ramp down from 1→0 over [0, slack_gate_width] ===
    rank_gate = np.clip(1.0 - (norm_slack / (PARAMS["slack_gate_width"] + eps)), 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Soft uncertainty activation: graded sigmoidal response instead of binary AND-gate ===
    uncertainty_activation = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_smoothness"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_interaction = norm_duration * uncertainty_activation * PARAMS["duration_robustness"]
    
    # === Starvation relief: saturating exponential form with explicit offset for numerical safety ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + eps))
    
    # === Energy-slack interaction: penalizes high marginal energy more under tight slack ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["energy_slack_interaction"] * slack_pressure)
    
    # === Restored rank-slack coupling: strengthens critical path reward exactly when slack is constrained ===
    coupled_rank_reward = norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure)
    
    # === Successor-release interaction: reward tasks whose execution unblocks many high-rank successors ===
    # Uses normalized work * rank, gated by urgency to prioritize release under deadline pressure
    successor_release_score = norm_work * norm_rank * slack_pressure * PARAMS["successor_release_weight"]
    
    # === Final score: smaller = higher priority ===
    # Primary drivers: urgency, coupled rank reward, and energy-slack penalty.
    # Secondary: anti-starvation (benefit), risk-duration penalty, and successor-release (benefit).
    score = (
        + slack_pressure                                  # Urgency driver (primary)
        + energy_slack_penalty                            # Deadline-amplified energy cost
        + PARAMS["energy_sensitivity"] * norm_energy     # Base energy penalty
        - coupled_rank_reward                            # Restored deadline-coupled critical path reward
        - wait_benefit                                   # Anti-starvation reward
        + duration_risk_interaction                      # Graded risk-duration penalty
        - successor_release_score                        # Prefer tasks unblocking large critical work
    )
    
    # Final safeguard: ensure finite, deterministic output with strict shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(-1)
