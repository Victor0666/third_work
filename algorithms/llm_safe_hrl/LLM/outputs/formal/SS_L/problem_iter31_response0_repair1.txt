def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all declared parameters are used; no unused or missing references;
       uses median-MAD normalization; clips all intermediate terms to [-2,2];
       applies DDL-protection gate via sigmoid using ddl_protection_gate_slope;
       includes slack_penalty_exponent for negative-slack power penalty;
       uses slack_pressure_gate_steepness for urgency ramp-up;
       retains successor-release interaction (min_exec_time × upward_rank × (slack <= 0));
       gates energy/uncertainty strictly by feasibility_gate = sigmoid(slack) from ddl_protection_gate_slope;
       removes inactive parameters (duration_robustness, wait_saturation_offset) per evidence."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_exec = median_mad_normalize(min_exec_time)
    norm_comm = median_mad_normalize(min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical-path release boost: min_exec_time × upward_rank × (slack <= 0) — validated ===
    ddl_breach = (slack <= 0.0).astype(float)
    successor_release = norm_exec * norm_rank * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate: sigmoid activation for pre-breach urgency ramp-up ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Energy preference: only active under feasibility gate ===
    energy_term = PARAMS["energy_sensitivity"] * norm_energy * ddl_gate
    
    # === Uncertainty-energy coupling: gated by both uncertainty and DDL feasibility ===
    energy_uncert_penalty = PARAMS["energy_uncertainty_interaction"] * norm_energy * norm_uncert * ddl_gate
    
    # === Starvation relief: exponential decay with wait_decay ===
    wait_term = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Composite score: smaller = higher priority ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Unblock critical path
        - np.clip(boosted_rank, -2.0, 2.0)                                 # HEFT urgency under pressure
        + energy_term                                                       # Energy cost adds to score only if feasible
        + energy_uncert_penalty                                             # Risk-energy penalty only if feasible
        - np.clip(wait_term, -2.0, 2.0)                                    # Anti-starvation
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)  # Downstream work bias
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
