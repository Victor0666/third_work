def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces median-MAD with quantile-based robust normalization;
       introduces congestion-aware gating via (ready_wait_time + uncertainty) threshold;
       replaces boosted_rank with conditional critical-path coupling (upward_rank × remaining_work) activated only under ddl pressure;
       removes duration_robustness and wait_saturation_offset (deemed inactive per diagnostics);
       enforces sign-preserving slack normalization via MAD;
       adds load-successor-release interaction: penalizes high-uncertainty tasks with large remaining_work when slack is tight."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Quantile-based robust normalization (Q1/Q3 range) — preserves outlier sensitivity near boundaries ===
    def quantile_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1, q3 = x[0], x[0]
        else:
            q1 = np.quantile(x, 0.25)
            q3 = np.quantile(x, 0.75)
        iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        return (x - q1) / spread
    
    # === Sign-preserving MAD normalization *only for slack* — sharpens urgency near violation boundary ===
    def mad_normalize_slack(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = mad_normalize_slack(slack)
    norm_energy = quantile_normalize(min_incremental_energy)
    norm_duration = quantile_normalize(min_exec_time + min_comm_time)
    norm_rank = quantile_normalize(upward_rank)
    norm_work = quantile_normalize(remaining_work)
    norm_wait = quantile_normalize(ready_wait_time)
    norm_uncert = quantile_normalize(uncertainty)
    
    # === DDL feasibility gate: binary indicator for slack <= 0 — eliminates penalty leakage ===
    ddl_feasible = (slack > 0.0).astype(float)
    ddl_breach = 1.0 - ddl_feasible
    
    # === Critical-path coupling: upward_rank × remaining_work interaction, gated by ddl_breach only ===
    # Validated by counterfactual evidence: avoids over-amplifying non-critical paths
    critical_path_coupling = norm_rank * norm_work * ddl_breach
    
    # === Slack pressure gate: sigmoid activated only when slack >= 0 — enables energy/uncertainty penalties exclusively in feasible region ===
    ddl_protection_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Congestion signal: unified load indicator from waiting + uncertainty, gated by both ddl_feasible and high-congestion threshold ===
    congestion = ready_wait_time + uncertainty
    congestion_norm = quantile_normalize(congestion)
    congestion_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (congestion_norm - PARAMS["uncertainty_gate_threshold"])))
    congestion_penalty = norm_duration * congestion_gate * ddl_feasible
    
    # === Load-successor-release interaction: penalizes high-uncertainty + large-work tasks *only when slack is tight* (i.e., slack <= 0) ===
    # Addresses 'avoidable_marginal_energy_or_load_cost' failures per diagnostic feedback
    successor_release_penalty = norm_uncert * norm_work * ddl_breach
    
    # === Starvation relief: clipped linear (not logistic) to improve low-wait discriminability without flat gradients ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 2.0)
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then quantile-normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = quantile_normalize(raw_slack_penalty)
    
    # === Composite score: smaller = higher priority; all terms bounded and stabilized ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        + np.clip(critical_path_coupling, -2.0, 2.0)                       # Critical-path unblocking under breach
        - np.clip(norm_rank, -2.0, 2.0)                                     # Baseline criticality (no amplification outside breach)
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_protection_gate, -2.0, 2.0)  # Energy preference only when ddl-feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                  # Anti-starvation (linear)
        + np.clip(congestion_penalty, -2.0, 2.0)                            # Host-load congestion signal
        + PARAMS["energy_uncertainty_interaction"] * np.clip(successor_release_penalty, -2.0, 2.0)  # Load-successor release penalty
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * ddl_feasible, -2.0, 2.0)  # Downstream work bonus only when feasible
        + PARAMS["slack_pressure_gate_steepness"] * np.clip(norm_slack * ddl_feasible, -2.0, 2.0)  # Slack-margin bonus in feasible region
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
