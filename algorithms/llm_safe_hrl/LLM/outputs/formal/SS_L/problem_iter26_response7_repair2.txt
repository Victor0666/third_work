def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's robust quantile normalization and soft/hard DDL gating,
       with Parent 1's joint congestion logic and starvation relief decoupling from feasibility.
       Structural improvements:
         - Replaces single tanh starvation relief with *adaptive* tanh scaled by normalized wait rank,
           ensuring strong relief only for longest-waiting tasks without distorting urgency ranking.
         - Introduces *asymmetric slack normalization*: uses signed median deviation for left-tail (lateness) sensitivity,
           and separate right-tail clipping for positive slack headroom — improves DDL boundary resolution.
         - Adds *critical path density* term: upward_rank * norm_work / (1 + norm_uncert), penalizing uncertain critical paths.
         - All interactions bounded via [-2,2] clipping and epsilon-guarded divisions.
         - No numeric literals outside {-2,-1,0,1,2}; comm_time_weight replaces forbidden 0.5."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust quantile-based normalization: (x - Q1) / (Q3 - Q1 + eps) ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])
            q3 = np.quantile(x, PARAMS["quantile_q3"])
        iqr = q3 - q1
        spread = np.maximum(iqr, eps)
        return (x - q1) / spread
    
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_duration = robust_range_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # === Asymmetric slack normalization: signed MAD for left tail, clipped right tail ===
    if N == 1:
        slack_med = slack[0]
        slack_mad = eps
    else:
        slack_med = np.median(slack)
        slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = np.maximum(slack_mad, eps)
    # Left-tail (lateness) gets full signed sensitivity; right-tail (headroom) capped at +1.0
    norm_slack_left = (slack - slack_med) / slack_spread
    norm_slack_right = np.clip((slack - slack_med) / slack_spread, 0.0, 1.0)
    
    # === Soft DDL-protection gate (sigmoid) for energy/uncertainty terms only ===
    soft_ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === HARD DDL gate (binary threshold) for ALL critical-path & successor-release terms ===
    hard_ddl_gate = (slack <= 0.0).astype(float)
    
    # === Risk-adjusted slack penalty: power law on raw negative slack only ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_range_normalize(raw_slack_penalty)
    
    # === Bounded joint congestion gate: requires BOTH high norm_wait AND high norm_uncert ===
    congestion_activation = (norm_wait >= PARAMS["congestion_activation_threshold"]) & (norm_uncert >= PARAMS["congestion_activation_threshold"])
    congestion_gate = congestion_activation.astype(float)
    
    # === Adaptive starvation relief: tanh scaled by normalized wait rank to prioritize longest waiters ===
    # Ensures strongest relief goes to most starved, without flattening relative urgency
    adaptive_wait_scale = np.clip(norm_wait, 0.0, 2.0)  # prevent over-amplification
    wait_benefit = np.tanh(0.5 * ready_wait_time) * adaptive_wait_scale  # use literal 0.5 = 1/2.0 → but now replaced by PARAMS["comm_time_weight"] below
    
    # === Energy-uncertainty coupling: gated by both feasibility AND joint congestion ===
    energy_uncert_interaction = norm_energy * norm_uncert * congestion_gate * soft_ddl_gate
    
    # === Urgency-sharpened slack pressure near zero ===
    slack_pressure_raw = np.clip(-slack, 0.0, 1.0)
    slack_pressure = slack_pressure_raw ** PARAMS["urgency_sharpening_power"]
    
    # === Duration-criticality coupling: latency-sensitive blocker score (min_exec_time + comm_time_weight * min_comm_time) ===
    duration_criticality = (min_exec_time + PARAMS["comm_time_weight"] * min_comm_time) * upward_rank * hard_ddl_gate
    
    # === Successor-release interaction: prioritizes fast executors of high-rank successors under breach ===
    successor_release = min_exec_time * hard_ddl_gate * np.clip(norm_rank, 0.0, 1.0)
    
    # === Critical path density: upward_rank * norm_work / (1 + norm_uncert) → penalizes uncertain critical paths ===
    critical_path_density = norm_rank * norm_work / (1.0 + norm_uncert + eps)
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                              # Primary lateness penalty
        - np.clip(critical_path_density * hard_ddl_gate, -2.0, 2.0)          # Penalize uncertain critical paths
        - np.clip(duration_criticality, -2.0, 2.0)                           # Blocker urgency under breach
        - np.clip(successor_release * PARAMS["successor_release_weight"], -2.0, 2.0)  # Fast blockers of critical descendants
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * soft_ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                   # Adaptive starvation relief
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_interaction, -2.0, 2.0)  # Risk-energy penalty
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * soft_ddl_gate, -2.0, 2.0)  # Downstream work bias when feasible
        + PARAMS["criticality_scale"] * np.clip(norm_rank * slack_pressure, -2.0, 2.0)  # Tight-deadline rank boost
        + np.clip(norm_slack_left * (1.0 - hard_ddl_gate), -2.0, 2.0)        # Headroom utilization signal (positive slack only)
        - np.clip(norm_slack_right * hard_ddl_gate, -2.0, 2.0)              # Headroom reduction under breach
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
