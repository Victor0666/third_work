def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces quantile-based normalization with MAD-based for outlier resilience;
       simplifies rank modulation to single robust slack-pressure gate; removes redundant min_comm_time coupling;
       introduces explicit *sign-aware* slack normalization to preserve directional urgency semantics."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === MAD-based robust normalization: median absolute deviation — resistant to outliers in sparse-ready sets ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        if N == 1:
            center = x_abs[0]
            spread = eps
        else:
            center = np.median(x_abs)
            spread = np.median(np.abs(x_abs - center))
        return (x_abs - center) / (spread + eps)
    
    # Normalize base features (using MAD, not quantiles)
    norm_slack = robust_normalize(slack)  # retains magnitude but loses sign → fix below
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)  # no redundant comm coupling
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Sign-aware slack normalization: preserves directionality of urgency (negative slack = urgent) ===
    # Reconstruct signed slack score: scale sign * |normalized|, bounded to [-2,2] to prevent inversion
    slack_sign = np.sign(slack)
    norm_slack_signed = slack_sign * np.clip(np.abs(norm_slack), 0.0, 2.0)
    
    # === Risk-aware slack penalty: normalized power penalty on overdue tasks ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Simplified slack-pressure gate: single sigmoid activation centered at zero slack ===
    # Uses *signed* slack directly for clean monotonic gating — no overlapping linear/sigmoid couplings
    slack_pressure = np.clip(-slack, 0.0, np.inf)  # non-negative urgency signal
    # Normalize urgency for stable gating input
    if N == 1:
        urgency_center = slack_pressure[0]
        urgency_spread = eps
    else:
        urgency_center = np.median(slack_pressure)
        urgency_spread = np.median(np.abs(slack_pressure - urgency_center)) + eps
    norm_urgency = (slack_pressure - urgency_center) / (urgency_spread + eps)
    # Gate: sharp activation only when overdue or critically tight
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (norm_urgency - 0.5)))
    
    # === Boosted rank: upward_rank scaled by urgency gate only — clean, monotonic, no inversion risk ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Uncertainty-duration interaction: gated by both urgency and feasibility ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * uncert_gate * rank_gate * ddl_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus with saturation offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty interaction: gated by DDL feasibility AND urgency threshold ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only under positive-but-tight slack ===
    # Use *signed* slack to distinguish safe slack (>0) from risky slack (~0)
    tight_slack_mask = (slack > 0) & (slack < PARAMS["uncertainty_gate_threshold"] * np.max(np.abs(slack) + eps))
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * np.where(tight_slack_mask, 1.0, 0.0)) * ddl_gate
    
    # === Decoupled duration weight: restores flexibility lost in fixed 0.5-weight average ===
    duration_preference = PARAMS["duration_robustness"] * norm_duration * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty dominates
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Critical-path boost under urgency
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - duration_preference                                                # Tunable duration preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation always active
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
