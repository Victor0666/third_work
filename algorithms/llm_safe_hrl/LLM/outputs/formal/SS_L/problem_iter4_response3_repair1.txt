def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: removes unused parameters; replaces tanh(-slack) with bounded urgency;
       uses only declared PARAMS; all numeric literals are -2,-1,0,1,2; uses np.finfo for epsilon safety."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust feature scaling using mean(abs(x)) + eps — avoids median computation cost and outlier sensitivity ===
    def scale_feature(x):
        x_abs = np.abs(x)
        center = np.mean(x_abs) if N > 0 else 0.0
        spread = np.mean(x_abs) + eps  # use mean abs as robust scale proxy
        return (x - center) / (spread + eps)
    
    # Normalize base features with mean-abs scaling
    norm_slack = scale_feature(slack)
    norm_energy = scale_feature(min_incremental_energy)
    norm_duration = scale_feature(min_exec_time + min_comm_time)
    norm_rank = scale_feature(upward_rank)
    norm_work = scale_feature(remaining_work)
    norm_wait = scale_feature(ready_wait_time)
    norm_uncert = scale_feature(uncertainty)
    
    # === Risk-aware slack penalty: bounded linear-tanh hybrid urgency — smooth, finite, monotonic ===
    # Use tanh to bound penalty magnitude while preserving sign and gradient near zero
    raw_urgency = -slack * PARAMS["slack_penalty_exponent"]
    slack_urgency = np.tanh(raw_urgency)
    norm_slack_penalty = scale_feature(slack_urgency)
    
    # === Conditional DDL protection gate: activate criticality boost only under safe+urgent conditions ===
    # Safe: low uncertainty; Urgent: slack < 0 or norm_slack < -0.5 (tuned threshold)
    slack_tight = (slack < 0.0) | (norm_slack < -0.5)
    uncert_low = norm_uncert < PARAMS["uncertainty_gate_threshold"]
    ddl_protection_gate = np.where(slack_tight & uncert_low, 1.0, 0.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_protection_gate)
    
    # === Successor-release interaction: reward tasks whose execution unblocks many high-rank successors ===
    successor_release_score = norm_work * norm_rank * ddl_protection_gate
    
    # === Bounded anti-starvation: tanh-based instead of exponential — stable, discriminative, saturates at ±1 ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Unified risk-energy interaction: penalize high energy only when both slack is tight AND uncertainty is high ===
    risk_energy_gate = np.where((slack < 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]), 1.0, 0.0)
    energy_uncert_penalty = norm_energy * risk_energy_gate
    
    # === Composite score: smaller = higher priority ===
    # Prioritize: (1) DDL compliance via bounded urgency, (2) critical path under safe+urgent conditions,
    #             (3) unblocking large critical work, (4) anti-starvation, (5) joint risk-energy penalty,
    #             (6) mild preference for large downstream work
    score = (
        + norm_slack_penalty                           # Bounded urgency for overdue tasks
        - boosted_rank                                 # Critical path boost only when safe+urgent
        - PARAMS["energy_sensitivity"] * norm_energy  # Energy preference, gated by joint risk
        - wait_benefit                                 # Stable, bounded starvation relief
        - successor_release_score                      # Prefer tasks releasing large critical work
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Joint risk-energy penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Downstream work weight
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
