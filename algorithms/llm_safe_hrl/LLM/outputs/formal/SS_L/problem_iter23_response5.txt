def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 1's robust MAD-slack urgency and Parent 2's decoupled gating & successor-release.
       Key novel improvements:
       1. Dual-gated starvation relief: wait_benefit activated *only* under DDL pressure (hard gate) to prevent premature aging bias;
       2. Urgency-coupled criticality: norm_rank amplified by *both* hard_ddl_gate *and* continuous slack_pressure, modulated by tunable strength;
       3. Unified congestion signal using *normalized sum* (not product) of wait+uncert, gated jointly by congestion threshold and soft_ddl_gate;
       4. Reintroduced sign-preserving MAD normalization for slack (from Parent 1) to sharpen sensitivity near zero deadline margin.
       All terms clipped to [-2,2], no literals beyond -2,-1,0,1,2, and shape (N,) enforced."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust quantile-based normalization: (x - Q1) / (Q3 - Q1 + eps) — preserves sign, resists outliers ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])
            q3 = np.quantile(x, PARAMS["quantile_q3"])
        iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        return (x - q1) / spread
    
    # Normalize all features except slack (which gets sign-preserving MAD for urgency sensitivity near zero)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_duration = robust_range_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # Sign-preserving MAD for slack: sensitive to small negative values near deadline boundary (Parent 1 strength)
    if N == 1:
        slack_med = slack[0]
        slack_mad = eps
    else:
        slack_med = np.median(slack)
        slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = slack_mad if slack_mad > eps else eps
    norm_slack_mad = (slack - slack_med) / slack_spread
    
    # === Soft DDL-protection gate (sigmoid) for energy/uncertainty terms only ===
    soft_ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Hard DDL gate (binary threshold) for urgency, critical-path & successor-release terms ===
    hard_ddl_gate = (slack <= 0.0).astype(float)
    
    # === Critical path leverage: upward_rank * remaining_work interaction, activated only under hard deadline breach ===
    critical_path_leverage = norm_rank * norm_work * hard_ddl_gate
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized via robust range ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_range_normalize(raw_slack_penalty)
    
    # === Bounded piecewise-linear congestion gate: active only when BOTH wait and uncertainty are high ===
    congestion_activation = (norm_wait >= PARAMS["congestion_activation_threshold"]) & (norm_uncert >= PARAMS["congestion_activation_threshold"])
    congestion_gate = congestion_activation.astype(float)
    congestion_score = (norm_wait + norm_uncert) * congestion_gate * (1.0 - soft_ddl_gate)
    
    # === Dual-gated starvation relief: linear wait benefit *only* under DDL pressure (avoids biasing idle periods) ===
    wait_mean = np.mean(ready_wait_time)
    wait_denom = np.maximum(wait_mean, eps)
    raw_wait_benefit = np.clip(ready_wait_time / wait_denom, 0.0, PARAMS["wait_benefit_saturation"])
    wait_benefit = raw_wait_benefit * hard_ddl_gate
    
    # === Energy-uncertainty coupling: only active under DDL breach and high uncertainty ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * hard_ddl_gate
    
    # === Slack pressure near boundary: clipped linear ramp from 0 to 1 over [-0.5, 0] slack range ===
    slack_pressure = np.clip(-slack, 0.0, 1.0)
    
    # === Successor-release interaction: prioritize fast-executing tasks blocking high-rank successors under deadline pressure ===
    successor_release = min_exec_time * hard_ddl_gate * np.clip(norm_rank, 0.0, 1.0)
    
    # === Urgency-coupled criticality: norm_rank amplified by both hard_ddl_gate AND continuous slack_pressure ===
    coupled_criticality = norm_rank * (1.0 + PARAMS["urgency_coupling_strength"] * slack_pressure) * hard_ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Critical path boost under hard breach
        - np.clip(coupled_criticality, -2.0, 2.0)                         # Novel: urgency-modulated criticality
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * soft_ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Dual-gated anti-starvation
        + np.clip(congestion_score, -2.0, 2.0)                             # Unified load-aware congestion relief
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * soft_ddl_gate, -2.0, 2.0)  # Prefer large downstream work only when feasible
        + PARAMS["slack_penalty_exponent"] * np.clip(slack_pressure, -2.0, 2.0)       # Direct urgency coupling
        - PARAMS["successor_release_weight"] * np.clip(successor_release, -2.0, 2.0)   # Release blocked critical successors
        + np.clip(norm_slack_mad, -2.0, 2.0)                               # Sharpened urgency via MAD-normalized slack (Parent 1)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
