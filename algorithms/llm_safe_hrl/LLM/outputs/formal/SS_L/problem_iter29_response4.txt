def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's robust ddl_pressure and successor-release with Parent 1's slack_penalty_exponent;
       replaces static host_load_surrogate with sigmoid-gated duration-uncertainty coupling for sharper congestion response;
       introduces slack-aware anti-starvation: wait_benefit scaled by (1 - ddl_pressure) AND activated only when slack > 0;
       uses clipped MAD-normalized slack_penalty as primary urgency signal, enhanced by exponent for asymmetric lateness risk;
       eliminates redundant rank_slack_coupling and rank_gate — replaced by unified criticality_scale × ddl_pressure;
       all intermediate terms clipped to [-2,2] to ensure bounded AST depth and stability."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Smooth ddl_pressure: 1/(1+|slack|+eps) — graded urgency without numerical instability near zero ===
    ddl_pressure = 1.0 / (1.0 + np.abs(slack) + eps)
    
    # === Critical-path urgency: upward_rank × remaining_work × (slack <= 0) — validated DDL diagnostic action ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_urgency = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: fast-executing blockers of critical descendants — prioritizes low min_exec_time tasks with high upward_rank under breach ===
    successor_release = min_exec_time * norm_rank * ddl_breach
    
    # === Asymmetric slack penalty: power law on raw negative slack, then MAD-normalized and clipped ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = np.clip(median_mad_normalize(raw_slack_penalty), -2.0, 2.0)
    
    # === Sigmoid-gated host-load surrogate: sharp activation of duration-uncertainty coupling under congestion + deadline pressure ===
    load_activation = norm_duration * norm_uncert
    host_load_gate = 1.0 / (1.0 + np.exp(-PARAMS["host_load_steepness"] * (load_activation - 0.5)))
    host_load_surrogate = load_activation * host_load_gate * ddl_pressure
    
    # === Energy-uncertainty coupling: active only when both ddl_pressure > threshold AND uncertainty above threshold ===
    energy_uncert_gate = (ddl_pressure > PARAMS["ddl_pressure_gate_threshold"]).astype(float) * (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_penalty = norm_energy * norm_uncert * energy_uncert_gate
    
    # === Slack-conditioned starvation relief: only granted when slack > 0 (feasibility margin exists), scaled by (1 - ddl_pressure) ===
    wait_eligible = (slack > 0.0).astype(float)
    wait_benefit = np.exp(-PARAMS["wait_decay"] * ready_wait_time) * (1.0 - ddl_pressure) * wait_eligible
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability and bounded AST depth ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty: asymmetric, exponentiated, normalized
        - np.clip(critical_path_urgency, -2.0, 2.0)                      # Critical path leverage under breach
        - np.clip(successor_release, -2.0, 2.0)                          # Fast-executing blocker release boost
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_pressure, -2.0, 2.0)  # Energy preference gated by feasibility margin
        - np.clip(wait_benefit, -2.0, 2.0)                               # Starvation relief strictly conditional on slack safety
        + np.clip(host_load_surrogate, -2.0, 2.0)                        # Sharper congestion penalty via sigmoid gating
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + PARAMS["criticality_scale"] * np.clip(norm_rank * ddl_pressure, -2.0, 2.0)  # HEFT rank scaled by graded deadline pressure
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
