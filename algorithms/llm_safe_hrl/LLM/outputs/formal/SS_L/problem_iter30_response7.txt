def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with tightened deadline focus and robust fallback:
       - Removes redundant urgency_sharpening_factor and binary ddl_pressure per reflection.
       - Introduces *upward_rank_fallback_weight*: uses raw upward_rank as stable urgency anchor
         when median-MAD normalization degrades (e.g., N=1 or low-variance slack), preventing rank collapse.
       - Replaces all norm-based urgency terms with a unified, smooth slack-gated criticality signal:
         combines normalized upward_rank with raw slack sensitivity via sigmoid centered at slack_sensitivity_center.
       - Tightens hinge width and steepness parameters to concentrate priority shifts within high-risk slack band.
       - Keeps hard_ddl_violation_penalty + slack_sensitivity_center as non-negotiable DDL enforcement core.
       - All intermediate terms clipped to [-2,2]; final score safeguarded with machine-precision bounds."""
    
    eps = PARAMS["epsilon"]
    
    # Safe cast; no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — applied only to features where variance matters ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = np.maximum(mad, eps)
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    
    # === Hard deadline violation flag: finite dominant penalty ===
    ddl_violated = (slack < PARAMS["slack_sensitivity_center"]).astype(float)
    hard_ddl_penalty = ddl_violated * PARAMS["hard_ddl_violation_penalty"]
    
    # === Slack feasibility gate: enables energy preference only when safe ===
    slack_feasibility = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Robust slack penalty: capped power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Unified slack-gated criticality: smooth sigmoid coupling between raw slack and upward_rank ===
    # Uses normalized slack for shape, raw upward_rank as stable urgency anchor when normalization unstable
    slack_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * 
                                         (slack - PARAMS["slack_sensitivity_center"])))
    # Fallback: blend normalized rank with raw rank weighted by instability proxy (MAD < eps)
    rank_stability = np.where(N == 1, 0.0, np.median(np.abs(upward_rank - np.median(upward_rank))))
    use_raw_rank = (rank_stability < eps).astype(float)
    stable_rank = (1.0 - use_raw_rank) * norm_rank + use_raw_rank * (upward_rank / (np.max(np.abs(upward_rank)) + eps))
    criticality_signal = PARAMS["criticality_scale"] * slack_sigmoid * stable_rank
    
    # === Adaptive clipped linear hinge now replaced by direct sigmoid-coupled criticality ===
    # Eliminates hinge complexity while preserving sharp, interpretable urgency activation
    
    # === Bounded linear starvation relief ===
    wait_benefit = np.clip(1.0 - PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + hard_ddl_penalty                                                 # Strict DDL-first enforcement
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty
        - np.clip(criticality_signal, -2.0, 2.0)                            # Unified slack-gated urgency
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * slack_feasibility, -2.0, 2.0)  # Energy minimization only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief
        + PARAMS["upward_rank_fallback_weight"] * np.clip(norm_work * ddl_violated, -2.0, 2.0)  # Minimal downstream work under breach (reduced weight, focused role)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
