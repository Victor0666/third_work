def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: restores `ddl_protection_gate_slope` and introduces `load_aware_suppression_strength`
       to dynamically suppress energy preference on high-duration tasks — directly targeting 'avoidable_marginal_energy_or_load_cost'.
       Removes unstable `slack_alignment_exponent` per reflection, reverting to robust linear slack penalty + gated critical-path amplification.
       Introduces duration-load gate: `np.clip((min_exec_time + min_comm_time) / (median_duration + eps), 0.0, 2.0)` to identify congested candidates.
       All energy-related contributions now modulated by both hard feasibility AND duration-load gate — preventing over-aggressive low-energy scheduling on already-loaded VMs.
       Maintains hard slack-driven energy suppression and unified robust normalization."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization: sign-preserving, outlier-resilient scaling ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad + eps
        return (x - med) / spread
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # Compute duration load signal: normalized total task duration (exec + comm)
    duration = min_exec_time + min_comm_time
    if N == 1:
        median_duration = duration[0]
    else:
        median_duration = np.median(duration)
    duration_load_signal = np.clip(duration / (median_duration + eps), 0.0, 2.0)
    
    # === Hard DDL-feasibility gate: binary suppression — zero energy/uncert penalties if slack < 0 ===
    hard_feasibility_gate = np.where(slack < 0.0, 0.0, 1.0)
    
    # === DDL-pressure signal: monotonic, normalized urgency proxy — negative slack mapped to [0,1] ===
    ddl_pressure = np.clip(-norm_slack, 0.0, 1.0)
    
    # === Linear slack penalty with tunable coefficient ===
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    slack_penalty = PARAMS["slack_penalty_linear_coeff"] * norm_slack_penalty
    
    # === Successor-release interaction: rank × work × pressure ===
    successor_release = norm_rank * norm_work * ddl_pressure
    
    # === Critical-path coupling: upward_rank scaled linearly by ddl_pressure, amplified by slope gate ===
    # Smooth gate: sigmoid-like activation near zero slack using slope parameter
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * ddl_pressure))
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_gate)
    
    # === Uncertainty-aware energy penalty: gated by hard_feasibility_gate AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * hard_feasibility_gate
    
    # === Starvation relief: clipped linear — preserves monotonicity and low-wait sensitivity ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 2.0)
    
    # === Load-aware energy suppression: reduces energy preference on high-duration tasks to mitigate marginal congestion cost ===
    # Applies smooth suppression: strength increases with duration_load_signal
    load_suppress = 1.0 - PARAMS["load_aware_suppression_strength"] * (duration_load_signal - 1.0)
    load_suppress = np.clip(load_suppress, 0.0, 1.0)
    
    # === Hard slack-driven energy suppression: tunable strength replaces binary zeroing → enables soft DDL tradeoff ===
    slack_energy_suppress = np.where(
        slack < 0.0,
        1.0 - PARAMS["slack_energy_suppression_strength"],
        1.0
    )
    suppressed_norm_energy = norm_energy * slack_energy_suppress * load_suppress * hard_feasibility_gate
    
    # === Composite score: smaller = higher priority; all terms clipped individually to [-2,2] ===
    score = (
        + np.clip(slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                      # Release-driven criticality boost
        - np.clip(coupled_rank, -2.0, 2.0)                           # Gate-amplified critical path
        - PARAMS["energy_sensitivity"] * np.clip(suppressed_norm_energy, -2.0, 2.0)  # Dual-gated energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                           # Linear anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)   # Downstream work bias
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
