def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces brittle sigmoid gates with robust bounded pressure functions;
       introduces successor-release interaction via min_exec_time × upward_rank × (slack <= 0);
       replaces DDL-protection Heaviside with smooth 1/(1+|slack|+eps) ddl_pressure;
       removes redundant duration_robustness and wait_saturation_offset per inactivity analysis;
       uses MAD-normalized slack for sharper urgency near zero; enforces strict feasibility-first ordering;
       adds host-load surrogate via norm_duration × norm_uncert gated by both ddl_pressure and uncertainty threshold."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Smooth ddl_pressure: 1/(1+|slack|+eps) — graded urgency without numerical instability near zero ===
    ddl_pressure = 1.0 / (1.0 + np.abs(slack) + eps)
    
    # === Critical-path urgency: upward_rank × remaining_work × (slack <= 0) — validated DDL diagnostic action ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_urgency = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: fast-executing blockers of critical descendants — validated across ≥5 consensus failures ===
    successor_release = min_exec_time * norm_rank * ddl_breach
    
    # === Bounded slack penalty: sign-preserving MAD-normalized slack, clipped to [-2,2] for monotonic sensitivity ===
    norm_slack_penalty = np.clip(norm_slack, -2.0, 2.0)
    
    # === Host-load surrogate: norm_duration × norm_uncert, gated by ddl_pressure AND uncertainty threshold ===
    load_gate = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float) * ddl_pressure
    host_load_surrogate = norm_duration * norm_uncert * load_gate
    
    # === Energy-uncertainty coupling: only active when ddl_pressure > 0.5 AND uncertainty above threshold ===
    energy_uncert_gate = (ddl_pressure > 0.5).astype(float) * (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_penalty = norm_energy * norm_uncert * energy_uncert_gate
    
    # === Starvation relief: exponential decay scaled by ddl_pressure — anti-starvation only under feasibility safety margin ===
    wait_benefit = np.exp(-PARAMS["wait_decay"] * ready_wait_time) * (1.0 - ddl_pressure)
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability and bounded AST depth ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary urgency signal: sharp near slack=0
        - np.clip(critical_path_urgency, -2.0, 2.0)                      # Critical path leverage under breach
        - np.clip(successor_release, -2.0, 2.0)                          # Fast-executing blocker release boost
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_pressure, -2.0, 2.0)  # Energy preference scaled by feasibility margin
        - np.clip(wait_benefit, -2.0, 2.0)                               # Starvation relief gated by ddl safety margin
        + np.clip(host_load_surrogate, -2.0, 2.0)                        # Load-aware congestion penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + PARAMS["criticality_scale"] * np.clip(norm_rank * ddl_pressure, -2.0, 2.0)  # HEFT rank scaled by graded deadline pressure
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
