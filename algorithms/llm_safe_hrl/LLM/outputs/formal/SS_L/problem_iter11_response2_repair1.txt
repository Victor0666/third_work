def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's stability and gating with Parent 1's successor-release insight;
       uses power-law slack-rank coupling and saturating anti-starvation;
       replaces per-term clipping with global score bounding;
       uses median-MAD normalization with sign-preserving centering."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust per-feature normalization: median centering + MAD scaling, preserves sign ===
    def robust_normalize(x):
        if N == 1:
            center = x[0]
            spread = eps
        else:
            center = np.median(x)
            spread = np.median(np.abs(x - center))
        return (x - center) / (spread + eps)
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware lateness penalty: power-law penalty on overdue slack only ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Power-law slack-rank coupling: monotonic amplification under tight slack ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)  # [0,2] bounded pressure signal
    rank_coupling_factor = (1.0 + slack_pressure) ** PARAMS["slack_penalty_exponent"]
    boosted_rank = norm_rank * rank_coupling_factor
    
    # === DDL-protection gate: smooth Heaviside on slack to disable energy/uncertainty penalties when violating ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Uncertainty-duration interaction: gated by both uncertainty threshold and DDL feasibility ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = norm_duration * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: saturating logistic with tunable power law for sharper or gentler decay ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ((norm_wait + PARAMS["wait_saturation_offset"]) ** 1.0))
    
    # === Energy-uncertainty interaction: only active when DDL-feasible AND uncertainty exceeds threshold ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: increases energy penalty under tightening slack (but only if feasible) ===
    energy_slack_penalty = norm_energy * (1.0 + slack_pressure * PARAMS["slack_penalty_exponent"]) * ddl_gate
    
    # === Successor-release term from Parent 1: prioritizes unblocking high-downstream-work tasks under urgency ===
    successor_release_score = norm_rank * norm_work * (slack <= 0.0).astype(float)
    
    # === Composite score: smaller = higher priority
    score = (
        + norm_slack_penalty                                     # Primary violation penalty
        - successor_release_score                                # Unblock large workloads first when urgent
        - boosted_rank                                           # Critical path amplified by slack pressure
        - PARAMS["energy_sensitivity"] * norm_energy * ddl_gate  # Energy minimization only when safe
        - wait_benefit                                           # Anti-starvation always active
        + PARAMS["duration_robustness"] * duration_risk_score   # Risk-weighted duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty
        + PARAMS["slack_penalty_exponent"] * energy_slack_penalty
        + PARAMS["remaining_work_weight"] * norm_work           # Mild preference for large downstream work
    )
    
    # Global safeguard: bound final score to [-10, 10] to prevent explosion while preserving ordinal ranking
    finfo = np.finfo(float)
    score = np.clip(score, -10.0, 10.0)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
