def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's robust DDL pressure handling with Parent 1's successor-release insight.
       Key innovations: (1) Urgency saturation offset prevents gradient vanishing at slack=0; (2) Duration penalty now uses
       physical-scale sum (exec+comm) instead of normalized version, preserving latency sensitivity; (3) Critical-path
       coupling extended to include *normalized* remaining_work (not just binary), enabling graded boosting;
       (4) All numeric literals are -2,-1,0,1,2; no hidden constants."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Stable normalization: mean(abs(x)) + eps scaling — robust for small N and zeros ===
    def stable_normalize(x):
        x_abs = np.abs(x)
        center = np.mean(x_abs) if N > 0 else 0.0
        spread = np.mean(x_abs) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features (except slack & duration for physical-scale urgency/duration terms)
    norm_energy = stable_normalize(min_incremental_energy)
    norm_rank = stable_normalize(upward_rank)
    norm_work = stable_normalize(remaining_work)
    norm_wait = stable_normalize(ready_wait_time)
    norm_uncert = stable_normalize(uncertainty)
    
    # === Physical-scale urgency: tanh(-slack + offset) preserves interpretability and avoids gradient singularity at slack=0 ===
    # Offset ensures nonzero derivative even when slack exactly equals zero — critical for CMA-ES convergence
    raw_urgency_input = -slack + PARAMS["urgency_saturation_offset"]
    slack_urgency = np.tanh(raw_urgency_input * PARAMS["slack_pressure_gate_steepness"])
    
    # === Graded critical-path coupling: continuous boost based on both slack pressure and normalized work magnitude ===
    # Replaces brittle binary gates with smooth, differentiable product: stronger boost for overdue + high-work tasks
    slack_pressure_score = np.clip(-slack, 0.0, np.inf)  # only positive when overdue
    work_pressure_score = np.clip(norm_work, 0.0, np.inf)  # only positive when above-mean work
    coupling_strength = slack_pressure_score * work_pressure_score
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * coupling_strength)
    
    # === Conditional energy-risk coupling: active only under joint stress (overdue + high uncertainty) ===
    risk_active = (slack < 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"])
    energy_uncert_penalty = np.where(risk_active, 
                                     norm_energy * norm_uncert, 
                                     np.zeros_like(norm_energy))
    
    # === Bounded anti-starvation: tanh on normalized wait time — stable, discriminative, saturates at ±1 ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * norm_wait)
    
    # === Feasibility-aware energy balancing: reduce energy sensitivity when slack <= 0 ===
    energy_weight = np.where(slack > 0.0, 
                             PARAMS["energy_sensitivity"], 
                             PARAMS["energy_sensitivity"] * PARAMS["energy_relaxation_under_ddl_stress"])
    
    # === Physical-scale duration penalty: use unnormalized sum (sec) to preserve latency sensitivity across VM types ===
    # Avoids distortion from normalization — e.g., a 10ms task remains meaningfully smaller than a 10s task
    duration_penalty = (min_exec_time + min_comm_time) * PARAMS["duration_penalty_weight"]
    
    # === Composite score: smaller = higher priority ===
    # Components ordered by priority hierarchy: (1) Urgency dominates, (2) Critical path under dual pressure,
    # (3) Energy minimization relaxed under DDL stress, (4) Latency penalty, (5) Starvation relief,
    # (6) Risk-aligned penalty, (7) Downstream work preference
    score = (
        + (1.0 - slack_urgency)                           # Dominant urgency: lower slack → lower score
        - boosted_rank                                    # Graded critical-path boost
        - energy_weight * norm_energy                    # Energy preference, relaxed under DDL stress
        - duration_penalty                               # Physical-scale latency penalty
        - wait_benefit                                   # Anti-starvation (bounded tanh)
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Joint risk-energy penalty
        + PARAMS["remaining_work_weight"] * norm_work    # Preference for unblocking large work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
