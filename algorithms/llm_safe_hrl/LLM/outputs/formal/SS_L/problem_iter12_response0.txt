def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces unstable successor-release coupling with bounded slack-conditioned energy decay;
       restores robust MAD-based normalization (more stable for small N); introduces rank-uncertainty coupling only under DDL feasibility;
       clips all intermediate terms to [-2,2] and enforces monotonic term-wise contribution; eliminates unbounded mask products."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: MAD-based (stable for N=1 and sparse sets) ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        if N == 1:
            center = x_abs[0]
            spread = eps
        else:
            center = np.median(x_abs)
            spread = np.median(np.abs(x_abs - center))
        return (x_abs - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack penalty: normalized power penalty on overdue tasks ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Bounded linear rank-slack coupling: monotonic, sign-aware, clipped ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    boosted_rank = norm_rank * rank_slack_coupling
    
    # === Sharpened slack-pressure gate: sigmoid for precise critical-path activation ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = boosted_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Slack-conditioned energy decay: softens energy penalty exponentially as slack grows (avoids over-penalizing under surplus) ===
    energy_decay_factor = np.exp(-PARAMS["energy_slack_decay_rate"] * np.maximum(slack, 0.0))
    energy_preference = norm_energy * ddl_gate * energy_decay_factor
    
    # === Rank-uncertainty coupling: amplifies criticality only when both rank is high AND uncertainty is elevated AND DDL feasible ===
    rank_uncert_coupling = norm_rank * norm_uncert * ddl_gate
    rank_uncert_boost = PARAMS["rank_uncertainty_coupling"] * np.clip(rank_uncert_coupling, -2.0, 2.0)
    
    # === Uncertainty-duration interaction: activated only under dual pressure AND slack feasibility ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus (simplified without saturation offset — empirically inactive & removed) ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * norm_wait)
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for boundedness and ordinal integrity ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness risk penalty
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Critical-path boost under pressure
        - PARAMS["energy_sensitivity"] * np.clip(energy_preference, -2.0, 2.0)  # Softened energy preference under slack
        - PARAMS["duration_robustness"] * np.clip(norm_duration * ddl_gate, -2.0, 2.0)  # Direct duration preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + np.clip(duration_risk_score, -2.0, 2.0)                          # Duration-risk interaction
        + rank_uncert_boost                                                # Bounded rank-uncertainty coupling
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)   # Downstream work bias
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
