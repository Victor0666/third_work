def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's robust slack penalty normalization with Parent 1's explicit starvation relief and adds novel latency-risk coupling.
       Key improvements:
         - Introduces `latency_risk_coupling`: multiplies (exec+comm) * uncertainty to directly model latency risk exposure
         - Uses raw-slack penalty *before* normalization (Parent 1 insight) but applies MAD normalization *only* to the penalty itself (Parent 2 refinement)
         - Retains sharp sigmoid slack-pressure gating on norm_slack for stable critical-path activation
         - Preserves logistic starvation relief with saturation offset for fairness
         - All 12 parameters used; no numeric literals except -2,-1,0,1,2."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: MAD-based, centered and scaled, avoids outlier skew ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        center = np.median(x_abs) if N > 1 else x_abs[0]
        spread = np.median(np.abs(x_abs - center)) if N > 1 else np.abs(x_abs[0] - center) + eps
        return (x_abs - center) / (spread + eps)
    
    # Normalize base features
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    norm_slack = robust_normalize(slack)
    
    # === Risk-aware slack penalty: raw negative slack raised to power, then robust-normalized (Parent 2 best practice) ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate: driven by normalized slack for stable sigmoid activation ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)  # bounded [0,2]
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Latency risk coupling: novel interaction modeling joint impact of duration and uncertainty ===
    # Directly captures how uncertain long tasks amplify deadline violation risk
    latency_risk_score = (min_exec_time + min_comm_time) * uncertainty
    norm_latency_risk = robust_normalize(latency_risk_score)
    
    # === Uncertainty-duration interaction: activated only under dual pressure (low slack AND high uncertainty) ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = norm_duration * uncert_gate * slack_pressure
    
    # === Starvation relief: logistic-shaped anti-starvation bonus with saturation offset (Parent 1 strength) ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty interaction: multiplicative penalty for high-risk-high-energy tasks ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate
    
    # === Composite score: smaller = higher priority ===
    score = (
        + norm_slack_penalty                           # Primary urgency: normalized lateness penalty
        - boosted_rank                                 # Boost critical-path under pressure
        - PARAMS["energy_sensitivity"] * norm_energy  # Prefer low marginal energy
        - norm_duration                                # Prefer short duration
        - wait_benefit                                 # Anti-starvation bonus
        + PARAMS["duration_robustness"] * duration_risk_score  # Penalty for risky+slow
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
        + PARAMS["latency_risk_coupling"] * norm_latency_risk  # Novel joint latency risk term
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
