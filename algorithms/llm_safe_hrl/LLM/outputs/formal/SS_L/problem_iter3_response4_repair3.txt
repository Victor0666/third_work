def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with adaptive slack clipping (using fixed quantiles),
    conditional fairness gate, and tanh-saturated energy-uncertainty interaction."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization using configurable quantiles (IQR) ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q_low = np.quantile(x, PARAMS["quantile_low"])
        q_high = np.quantile(x, PARAMS["quantile_high"])
        iqr = q_high - q_low + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    
    # === Adaptive slack clipping bounds derived from slack distribution ===
    # Use fixed quantiles (0.1 and 0.9) — no new parameters; literals -2,-1,0,1,2 only
    slack_abs = np.abs(slack)
    slack_q90 = np.quantile(slack_abs, 0.9) + eps
    slack_q10 = np.quantile(slack_abs, 0.1) + eps
    # Clamp to safe finite range using only allowed literals
    neg_slack_clip = np.clip(slack_q10, eps, 2.0)  # literal 2 only
    pos_slack_clip = np.clip(slack_q90, eps, 2.0)  # literal 2 only
    
    # === Piecewise slack urgency: tanh with adaptive clipping ===
    slack_urgency = (
        PARAMS["slack_risk_penalty"] * np.tanh(np.clip(-norm_slack, 0, neg_slack_clip))
        + PARAMS["slack_urgency_gain"] * np.tanh(np.clip(norm_slack, 0, pos_slack_clip))
    )
    
    # === Sigmoid gate for criticality: attenuate upward rank under severe deadline pressure ===
    slack_pressure = np.clip(-norm_slack, 0, 2)  # literal 2 only
    rank_gate = 1.0 / (1.0 + np.exp(PARAMS["rank_slack_gate"] * (slack_pressure - 2)))
    criticality_term = PARAMS["criticality_weight"] * rank_gate * norm_rank
    
    # === Conditional fairness gate: activate duration fairness only when duration CV exceeds threshold ===
    duration = min_exec_time + min_comm_time
    duration_mean = np.mean(duration) + eps
    duration_std = np.std(duration)
    duration_cv = duration_std / duration_mean
    fairness_gate = 1.0 if duration_cv > PARAMS["cv_duration_threshold"] else 0.0
    duration_fairness = fairness_gate * PARAMS["duration_fairness"] * np.abs(norm_duration)
    
    # === Anti-starvation: linear boost per waiting second, normalized ===
    wait_boost = PARAMS["wait_bias"] * norm_wait
    
    # === Work burden term: reduce priority for large downstream computational burden ===
    work_term = -PARAMS["work_weight"] * norm_work
    
    # === Energy-uncertainty interaction: product + tanh saturation ===
    energy_uncert_prod = norm_energy * norm_uncert
    energy_uncert_interaction = np.tanh(energy_uncert_prod)  # literal 1 implied in tanh
    energy_term = PARAMS["energy_scale"] * norm_energy + energy_uncert_interaction
    
    # === Composite score: smaller = higher priority ===
    score = (
        slack_urgency                    # Urgency penalty
        + energy_term                    # Energy + saturated interaction penalty
        + criticality_term               # Critical path weight
        + duration_fairness              # Conditionally activated duration fairness penalty
        - wait_boost                     # Starvation relief (subtract → improves priority)
        + work_term                      # Work burden penalty
    )
    
    # Final safeguard: ensure finite, deterministic output with correct shape
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — critical for N=1 case
    return score.reshape(-1)
