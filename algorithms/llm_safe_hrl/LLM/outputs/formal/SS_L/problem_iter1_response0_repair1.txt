def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule with three-mode logic: critical (negative slack), urgent (tight slack), and normal.
    
    Key innovations:
    - Critical mode: triggers hard penalty for slack < critical_slack_threshold, scaled by slack_penalty_factor
      and amplified by uncertainty_slack_coupling → ensures immediate attention to at-risk tasks.
    - Urgent mode: applies rank_leverage to upward_rank only when slack is small but non-negative, 
      balancing critical-path progress and deadline headroom.
    - Normal mode: emphasizes energy efficiency via nonlinear_energy_exponent (superlinear penalty) and 
      bounded fairness via wait_fairness_gain.
    - All features are robustly normalized using mean-abs + epsilon; no unbounded ops.
    - No division by raw inputs; all denominators guarded.
    - Returns strictly (N,) array of finite floats.
    """
    eps = PARAMS["epsilon"]
    nan_sub = PARAMS["nan_substitute"]
    inf_clip = PARAMS["inf_clip_bound"]
    N = len(slack)
    
    # Ensure finite, safe arrays — copy and sanitize using declared parameters only
    min_exec_time = np.nan_to_num(np.asarray(min_exec_time, dtype=float), nan=nan_sub, posinf=inf_clip, neginf=-inf_clip)
    min_comm_time = np.nan_to_num(np.asarray(min_comm_time, dtype=float), nan=nan_sub, posinf=inf_clip, neginf=-inf_clip)
    min_incremental_energy = np.nan_to_num(np.asarray(min_incremental_energy, dtype=float), nan=nan_sub, posinf=inf_clip, neginf=-inf_clip)
    slack = np.nan_to_num(np.asarray(slack, dtype=float), nan=0.0, posinf=inf_clip, neginf=-inf_clip)
    upward_rank = np.nan_to_num(np.asarray(upward_rank, dtype=float), nan=nan_sub, posinf=inf_clip, neginf=-inf_clip)
    remaining_work = np.nan_to_num(np.asarray(remaining_work, dtype=float), nan=nan_sub, posinf=inf_clip, neginf=-inf_clip)
    ready_wait_time = np.nan_to_num(np.asarray(ready_wait_time, dtype=float), nan=0.0, posinf=inf_clip, neginf=0.0)
    uncertainty = np.nan_to_num(np.asarray(uncertainty, dtype=float), nan=nan_sub, posinf=inf_clip, neginf=-inf_clip)

    # Robust normalization: use mean absolute value + epsilon to avoid division by zero or collapse
    def robust_norm(x):
        x_abs = np.abs(x)
        denom = np.mean(x_abs) + eps
        return x / denom

    # Normalize base features
    norm_exec_comm = robust_norm(min_exec_time + min_comm_time)
    norm_energy = robust_norm(min_incremental_energy)
    norm_rank = robust_norm(upward_rank)
    norm_work = robust_norm(remaining_work)
    norm_wait = robust_norm(ready_wait_time)
    norm_uncert = robust_norm(uncertainty)
    
    # Slacks: compute signed pressure (more negative → higher urgency)
    slack_pressure = robust_norm(slack)  # baseline normalized slack
    
    # Critical mode indicator: smoothed step from 0→1 as slack drops below threshold
    is_critical = np.clip((PARAMS["critical_slack_threshold"] - slack) / (eps + 0.1), 0.0, 1.0)
    
    # Uncertainty-amplified slack penalty: only active when slack is negative or near-threshold
    slack_risk_amplifier = np.maximum(0.0, -slack_pressure) * (1.0 + PARAMS["uncertainty_slack_coupling"] * norm_uncert)
    
    # Critical penalty term: dominates score when triggered
    critical_penalty = PARAMS["slack_penalty_factor"] * is_critical * slack_risk_amplifier
    
    # Urgent mode: apply rank leverage when slack is low but not yet critical (0 <= slack < threshold)
    # Use piecewise linear gate: 1.0 when slack ∈ [0, critical_slack_threshold), else 0
    is_urgent = np.where((slack >= 0.0) & (slack < PARAMS["critical_slack_threshold"]), 1.0, 0.0)
    urgent_rank_boost = PARAMS["rank_leverage"] * is_urgent * norm_rank
    
    # Normal mode: energy penalty with superlinear exponent + fairness term
    energy_penalty = PARAMS["energy_sensitivity"] * (np.abs(norm_energy) ** PARAMS["nonlinear_energy_exponent"])
    fairness_bonus = -PARAMS["wait_fairness_gain"] * norm_wait  # smaller score = higher priority
    
    # Duration urgency: favor fast-executing/comms tasks outside critical mode
    duration_urgency = PARAMS["duration_urgency_weight"] * norm_exec_comm * (1.0 - is_critical)
    
    # Assemble final score: lower = better
    score = (
        critical_penalty
        + energy_penalty
        + duration_urgency
        - urgent_rank_boost
        + fairness_bonus
    )
    
    # Final safeguard: ensure finite output using machine-precision bounds
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,)
    assert score.shape == (N,), f"Expected (N,) but got {score.shape}"
    
    return score
