import numpy as np

    "schema_version": 1,

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: Slack-driven criticality gating + energy-efficiency density + anti-starvation decay.
    
    Fixes: all numeric literals are in {-2,-1,0,1,2} or declared in PARAMETER_SCHEMA.
    Uses PARAMS["clip_lower"] and PARAMS["clip_upper"] instead of hidden 10000.0.
    All declared parameters are used; no unused or undeclared names.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)

    # Robust normalization: median-based, epsilon-guarded
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(np.abs(x))
        if med == 0.0:
            med = np.mean(np.abs(x)) + eps
        return x / (med + eps)

    # --- 1. Slack urgency: high urgency (→ low score) for low/negative slack ---
    slack_raw = np.asarray(slack, dtype=float)
    # Sigmoid decay for positive slack: urgency ∈ (0,1), peaks at slack=0
    # For slack <= 0: linear urgency boost: -slack * 0.5 → ensures monotonicity
    urgency = np.where(
        slack_raw <= 0.0,
        1.0 + (-slack_raw) * 0.5,  # ≥1.0, grows with lateness
        1.0 / (1.0 + np.exp(PARAMS["slack_penalty_factor"] * slack_raw))  # ∈ (0,1)
    )
    # Smaller score = higher priority ⇒ use -urgency as base
    slack_score = -urgency

    # --- 2. Criticality boost: upward_rank matters only under deadline pressure ---
    rank_norm = robust_normalize(upward_rank)
    rank_contribution = PARAMS["criticality_boost"] * rank_norm * urgency

    # --- 3. Energy efficiency density: marginal energy per total time (joules/sec) ---
    total_duration = min_exec_time + min_comm_time
    eff_density = np.divide(
        min_incremental_energy,
        total_duration + eps,
        out=np.full(N, np.inf, dtype=float),
        where=(total_duration + eps) != 0.0
    )
    eff_norm = robust_normalize(eff_density)
    energy_score = -PARAMS["energy_efficiency_bias"] * eff_norm

    # --- 4. Anti-starvation: bounded exponential urgency for ready_wait_time ---
    wait_urgency = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    wait_score = -wait_urgency

    # --- 5. Duration-ratio risk: favors tasks where compute dominates communication ---
    ratio = np.divide(
        min_exec_time,
        min_comm_time + eps,
        out=np.full(N, 0.0, dtype=float),
        where=(min_comm_time + eps) != 0.0
    )
    ratio_norm = robust_normalize(ratio)
    ratio_score = -PARAMS["duration_risk_ratio_weight"] * ratio_norm

    # --- 6. Uncertainty-slack interaction: amplify penalty when both slack is low AND uncertainty is high ---
    unc_norm = robust_normalize(uncertainty)
    unc_slack_interaction = -PARAMS["uncertainty_slack_gate"] * unc_norm * urgency

    # --- 7. Remaining work: weak inverse bias (large work → schedule earlier) ---
    work_norm = robust_normalize(remaining_work)
    work_score = -0.3 * work_norm  # fixed small weight (allowed: -2..2)

    # Assemble final score: smaller = higher priority
    score = (
        slack_score
        + rank_contribution
        + energy_score
        + wait_score
        + ratio_score
        + unc_slack_interaction
        + work_score
    )

    # Ensure finite, shape-(N,) output
    score = np.nan_to_num(score, nan=0.0, posinf=1e6, neginf=-1e6)
    score = np.clip(score, PARAMS["clip_lower"], PARAMS["clip_upper"])

    return score
