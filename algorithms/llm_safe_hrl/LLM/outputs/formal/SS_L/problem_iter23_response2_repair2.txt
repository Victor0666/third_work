def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's superior robust median-MAD normalization and synthetic host-load surrogate
       with soft sigmoidal ddl_feasibility_gate for differentiable deadline tradeoffs;
       uses quantile-based normalization *only for slack* (to sharpen boundary sensitivity), while all other features use robust MAD;
       introduces deadline-pressure-modulated starvation relief: wait_benefit scaled by (1 - ddl_pressure) to prioritize waiting tasks more when slack is ample;
       eliminates redundant congestion_penalty and successor_release_penalty — unified into critical-path coupling and host-load surrogate."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Quantile normalization for slack only (boundary-sensitive) ===
    def quantile_normalize_slack(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            q1 = x[0]
            q3 = x[0]
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])  # Now parameterized
            q3 = np.quantile(x, PARAMS["quantile_q3"])  # Now parameterized
        iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        return (x - q1) / spread
    
    # === Robust median-MAD normalization for all other features ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad + eps
        return (x - med) / spread
    
    # Slack: quantile-normalized for sharp deadline boundary handling
    norm_slack = quantile_normalize_slack(slack)
    
    # All others: robust median-MAD normalized
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Soft DDL-feasibility gate: sigmoid activated around slack=0 — enables smooth transition ===
    soft_feasibility_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === DDL-pressure signal: normalized urgency proxy — bounded [0,1] for differentiable scaling ===
    # Maps negative slack → high pressure, positive slack → low pressure
    ddl_pressure = np.clip((-norm_slack + 1.0) / 2.0, 0.0, 1.0)
    
    # === Linear slack penalty with tunable coefficient ===
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = quantile_normalize_slack(raw_slack_penalty)
    slack_penalty = PARAMS["slack_penalty_linear_coeff"] * norm_slack_penalty
    
    # === Critical-path coupling: additive rank × work × pressure ===
    successor_release = norm_rank * norm_work * ddl_pressure
    
    # === Criticality amplification: upward_rank scaled by pressure ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_pressure)
    
    # === Uncertainty-aware energy penalty: gated by soft_feasibility_gate AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * soft_feasibility_gate
    
    # === Starvation relief: clipped linear, modulated by (1 - ddl_pressure) to boost waiting priority when slack is ample ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time * (1.0 - ddl_pressure), 0.0, 2.0)
    
    # === Robustified duration penalty: damped and gated by soft_feasibility_gate and ddl_pressure ===
    duration_penalty = PARAMS["duration_robustness_factor"] * norm_duration * ddl_pressure * soft_feasibility_gate
    
    # === Hard slack-driven energy suppression: tunable strength ===
    slack_energy_suppress = np.where(
        slack < 0.0,
        1.0 - PARAMS["slack_energy_suppression_strength"],
        1.0
    )
    suppressed_norm_energy = norm_energy * slack_energy_suppress
    
    # === Host-load surrogate gating: uses norm_duration * norm_uncert as congestion proxy;
    #     activated only when soft_feasibility_gate > 0.5 and uncertainty above threshold ===
    load_surrogate = norm_duration * norm_uncert
    load_gate = np.where(
        (soft_feasibility_gate > 0.5) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]),
        np.clip(load_surrogate, 0.0, 2.0),
        0.0
    )
    host_load_penalty = PARAMS["host_load_sensitivity"] * load_gate * suppressed_norm_energy
    
    # === Composite score: smaller = higher priority; all terms clipped individually to [-2,2] ===
    score = (
        + np.clip(slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                      # Additive release-driven criticality boost
        - np.clip(coupled_rank, -2.0, 2.0)                           # Pressure-scaled critical path
        - PARAMS["energy_sensitivity"] * np.clip(suppressed_norm_energy, -2.0, 2.0)  # Tunably suppressed energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                           # Deadline-modulated anti-starvation
        + np.clip(duration_penalty, -2.0, 2.0)                       # Damped, feasibility-gated duration penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)   # Downstream work bias
        + np.clip(PARAMS["ddl_protection_gate_slope"] * ddl_pressure, -2.0, 2.0)  # Bounded DDL-urgency amplification
        + host_load_penalty                                          # Congestion-aware energy penalty (synthetic load proxy)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
