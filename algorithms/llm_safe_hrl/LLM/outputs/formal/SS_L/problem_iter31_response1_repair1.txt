def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all declared parameters are used; no unused or missing references;
       uses robust median-MAD normalization; applies DDL-protection gate to energy/uncertainty;
       includes successor-release interaction (min_exec_time * upward_rank * (slack <= 0));
       enforces strict feasibility gating; clips urgency signal to [-2,2]; uses np.finfo for epsilon."""
    
    # Use immutable machine epsilon instead of literal
    eps = np.finfo(float).tiny
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization per feature ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Primary urgency signal: MAD-normalized slack, clipped to [-2,2] ===
    urgency = np.clip(norm_slack, -2.0, 2.0)
    
    # === DDL protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical-path leverage under breach: upward_rank × remaining_work × (slack <= 0) ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_leverage = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: unblocks fast-executing critical predecessors ===
    successor_release = min_exec_time * upward_rank * ddl_breach
    norm_successor = median_mad_normalize(successor_release)
    
    # === Risk-adjusted energy penalty: gated by ddl_gate and scaled by uncertainty interaction ===
    energy_penalty = norm_energy * ddl_gate
    energy_uncert_penalty = energy_penalty * norm_uncert * PARAMS["energy_uncertainty_interaction"]
    
    # === Starvation relief: exponential decay with wait_decay ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Remaining work bias: gated by ddl_gate ===
    work_bias = norm_work * ddl_gate
    
    # === Slack pressure coupling: tanh-based monotonic amplification under tight slack ===
    # Uses slack_pressure_gate_steepness to sharpen response near zero slack
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    pressure_gate = np.tanh(PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0))
    rank_coupling = norm_rank * (1.0 + PARAMS["criticality_scale"] * pressure_gate)
    
    # === Nonlinear slack penalty: power law on raw negative slack, normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Composite score: smaller = higher priority.
    score = (
        + urgency                                          # Primary urgency
        - critical_leverage                                # Critical path leverage under breach
        - norm_successor                                   # Successor release
        - PARAMS["energy_sensitivity"] * energy_penalty    # Energy minimization under feasibility
        - wait_benefit                                     # Anti-starvation
        + PARAMS["remaining_work_weight"] * work_bias      # Downstream load under feasibility
        + energy_uncert_penalty                            # Joint risk penalty
        - rank_coupling                                    # Coupled rank under pressure
        + np.clip(norm_slack_penalty, -2.0, 2.0)          # Nonlinear lateness penalty
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
