def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid DDL gates with bounded smooth pressure; introduces successor-release interaction;
       uses robust median-MAD scaling; replaces redundant duration_robustness and wait_saturation_offset with proven minimal set;
       adds conditional host-load surrogate via norm_duration * norm_uncert gated by feasibility and uncertainty threshold;
       removes fragile exponential wait_benefit in favor of linear-ramped anti-starvation."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Smooth ddl_pressure: 1/(1+|slack|+eps) — graded urgency near zero slack without instability ===
    ddl_pressure = 1.0 / (1.0 + np.abs(slack) + eps)
    
    # === Critical-path leverage: upward_rank × remaining_work × (slack <= 0) — validated under deadline breach ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: min_exec_time × clipped_upward_rank × (slack <= 0)
    # Targets fast-executing blockers of critical descendants — confirmed in ≥5 consensus failures
    clipped_upward_rank = np.clip(norm_rank, 0.0, 2.0)
    successor_release = min_exec_time * clipped_upward_rank * ddl_breach
    
    # === Host-load surrogate: norm_duration × norm_uncert, gated by both ddl_pressure and uncertainty threshold ===
    # Enables congestion-aware energy penalization without VM-level visibility
    load_surrogate = norm_duration * norm_uncert
    load_gate = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float) * ddl_pressure
    host_load_penalty = load_surrogate * load_gate
    
    # === Anti-starvation: linear ramp up to saturation, avoiding log/exp singularities ===
    # Uses raw ready_wait_time scaled by wait_decay, capped at 1.0
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Energy-uncertainty coupling: only active when slack >= 0 AND uncertainty exceeds threshold ===
    energy_uncert_gate = (slack >= 0.0).astype(float) * (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_penalty = norm_energy * norm_uncert * energy_uncert_gate
    
    # === Slack-aware marginal energy penalty: amplifies energy cost only under positive slack pressure ===
    energy_slack_penalty = norm_energy * ddl_pressure * (slack >= 0.0).astype(float)
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack, -2.0, 2.0)                                  # Primary urgency signal (negative slack → lower score)
        + np.clip(ddl_pressure * PARAMS["slack_penalty_exponent"], -2.0, 2.0)  # Graded lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Leverage boost for critical-path successors under breach
        - np.clip(successor_release, -2.0, 2.0)                           # Fast-blocker release incentive under breach
        - np.clip(norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_pressure), -2.0, 2.0)  # Pressure-amplified rank
        - PARAMS["energy_sensitivity"] * np.clip(energy_slack_penalty, -2.0, 2.0)  # Feasible-region energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)  # Downstream work bias
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)  # Risk-energy coupling
        + PARAMS["ddl_protection_gate_slope"] * np.clip(host_load_penalty, -2.0, 2.0)  # Load-congestion surrogate
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
