def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with unified slack-driven gating:
       - Replaces *all* ad-hoc gates (feasibility, pressure, load) with a single, smooth,
         exponentiated slack mask: `mask = sigmoid(-slack)^unified_slack_mask_exponent`.
       - This mask jointly governs energy preference, starvation relief, and uncertainty coupling —
         eliminating uncontrolled interactions while preserving monotonic urgency response.
       - Introduces urgency_coupling_strength to fuse remaining_work and upward_rank *only under slack pressure*,
         promoting high-work descendants when deadlines tighten — improving end-to-end latency control.
       - Removes successor_release and host_load_surrogate: their empirical degradation confirmed structural over-complexity.
       - Keeps robust median-MAD normalization on core features only; all intermediates clipped to [-2,2].
       - Hard DDL enforcement remains dominant and finite via hard_ddl_violation_penalty.
       - Final score is fully deterministic, shape-(N,), and machine-precision safe."""
    
    eps = PARAMS["epsilon"]
    
    # Safe cast; no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — only on core decision features ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    
    # === Hard deadline violation flag: triggers dominant finite penalty ===
    ddl_violated = (slack < PARAMS["slack_sensitivity_center"]).astype(float)
    hard_ddl_penalty = ddl_violated * PARAMS["hard_ddl_violation_penalty"]
    
    # === Unified slack-driven mask: smooth, monotonic, exponentiated suppression ===
    # mask ∈ (0,1], decreases as slack worsens → suppresses non-urgency terms progressively
    slack_mask_base = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (-slack)))
    unified_slack_mask = np.clip(slack_mask_base ** PARAMS["unified_slack_mask_exponent"], 0.0, 1.0)
    
    # === Robust slack penalty: capped power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Adaptive clipped linear hinge for slack-driven urgency coupling ===
    slack_hinge = np.clip(-norm_slack, 0.0, PARAMS["rank_slack_hinge_width"])
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * (slack_hinge / (PARAMS["rank_slack_hinge_width"] + eps))
    
    # === Sharpened urgency gate centered at sensitivity boundary ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * urgency_gate)
    
    # === Urgency-coupled downstream work: promotes tasks whose descendants dominate critical path under pressure ===
    coupled_work_urgency = PARAMS["urgency_coupling_strength"] * norm_work * (1.0 - unified_slack_mask)
    
    # === Bounded linear starvation relief — now gated by unified slack mask ===
    wait_benefit = np.clip(1.0 - PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0) * unified_slack_mask
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + hard_ddl_penalty                                                 # Strict DDL-first enforcement
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency-amplified critical path
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * unified_slack_mask, -2.0, 2.0)  # Energy minimization only when slack allows
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief (slacked)
        + np.clip(PARAMS["slack_penalty_exponent"] * coupled_work_urgency, -2.0, 2.0)  # Downstream work prioritization under breach
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
