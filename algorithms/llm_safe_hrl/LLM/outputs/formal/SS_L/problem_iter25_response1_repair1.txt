def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: uses robust range-based normalization;
       introduces successor-release interaction (min_exec_time × clipped_upward_rank × (slack <= 0));
       enforces hard DDL feasibility via binary gating;
       removes all unused parameters; retains only validated structural terms;
       uses sign-preserving linear slack urgency and bounded wait bonus."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust range-based normalization: (x - q1) / (q3 - q1 + eps), sign-preserving ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
        else:
            q1 = np.quantile(x, 0.25)
            q3 = np.quantile(x, 0.75)
        iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        return (x - q1) / spread
    
    norm_slack = robust_range_normalize(slack)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_exec = robust_range_normalize(min_exec_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # === Hard DDL feasibility gating: binary indicator for slack <= 0 ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Successor-release interaction: fast-executing blockers of critical descendants ===
    clipped_rank = np.clip(norm_rank, 0.0, 2.0)
    successor_release = norm_exec * clipped_rank * ddl_breach
    
    # === Upward-rank × remaining-work interaction: critical-path importance weighted by downstream workload ===
    critical_path_importance = norm_rank * norm_work * ddl_breach
    
    # === Sign-preserving slack urgency: linear penalty on negative slack only ===
    raw_slack_urgency = -np.clip(slack, -np.inf, 0.0)
    norm_slack_urgency = robust_range_normalize(raw_slack_urgency)
    slack_penalty = np.clip(norm_slack_urgency, -2.0, 2.0)
    
    # === Joint congestion gating: high wait time AND high uncertainty ===
    wait_high = (ready_wait_time >= np.quantile(ready_wait_time, 0.75)).astype(float)
    uncert_high = (uncertainty >= np.quantile(uncertainty, 0.75)).astype(float)
    congestion_gate = wait_high * uncert_high
    
    # === Energy penalty only when DDL-feasible (slack >= 0) ===
    energy_feasible = (slack >= 0.0).astype(float)
    energy_penalty = norm_energy * energy_feasible
    
    # === Uncertainty-energy coupling only under congestion and feasibility ===
    energy_uncert_coupling = norm_energy * norm_uncert * congestion_gate * energy_feasible
    
    # === Starvation relief: linear wait bonus with saturation offset ===
    wait_bonus = np.clip(ready_wait_time / (ready_wait_time + PARAMS["wait_saturation_offset"]), 0.0, 1.0)
    
    # === Composite score: smaller = higher priority ===
    score = (
        + slack_penalty                                      # Primary urgency signal
        - PARAMS["criticality_scale"] * critical_path_importance
        - PARAMS["criticality_scale"] * successor_release
        - PARAMS["energy_sensitivity"] * energy_penalty
        - PARAMS["energy_uncertainty_interaction"] * energy_uncert_coupling
        - wait_bonus
        + PARAMS["remaining_work_weight"] * norm_work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(-1)
