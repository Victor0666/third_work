def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces robust_normalize with mean-abs scaling for stability;
       introduces joint successor-release gating using slack & remaining_work;
       replaces exponential wait_benefit with bounded tanh-based anti-starvation;
       adds conditional DDL-protection gate that activates only under combined slack pressure and uncertainty;
       removes unused 'duration_risk_score' in favor of direct slack-uncertainty coupling."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust feature scaling: use mean(abs(x)) + eps instead of MAD to avoid median instability on small N ===
    def scale_feature(x):
        x_abs = np.abs(x)
        center = np.mean(x_abs) if N > 0 else 0.0
        spread = np.mean(x_abs) + eps  # stable, avoids zero & outlier sensitivity
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = scale_feature(slack)
    norm_energy = scale_feature(min_incremental_energy)
    norm_duration = scale_feature(min_exec_time + min_comm_time)
    norm_rank = scale_feature(upward_rank)
    norm_work = scale_feature(remaining_work)
    norm_wait = scale_feature(ready_wait_time)
    norm_uncert = scale_feature(uncertainty)
    
    # === Risk-aware slack penalty: clipped power penalty on overdue tasks, normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = scale_feature(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate: sigmoid with tunable steepness, centered at slack=0 ===
    # Directly uses raw slack (seconds) for physical interpretability, then clips to [-2,2] before sigmoid
    slack_deviation = np.clip(-slack, -2.0, 2.0)  # maps negative slack → positive pressure
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_deviation - 0.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Joint successor-release gating: activates only when both slack is tight AND remaining_work is large ===
    # Encourages releasing bottlenecks early by boosting rank when downstream impact is high *and* deadline risk exists
    work_pressure = np.where(norm_work > PARAMS["successor_release_work_threshold"], 1.0, 0.0)
    slack_pressure_binary = np.where(norm_slack < PARAMS["ddl_protection_slack_threshold"], 1.0, 0.0)  # tight slack indicator
    successor_release_gate = slack_pressure_binary * work_pressure
    boosted_rank = boosted_rank + PARAMS["remaining_work_weight"] * norm_work * successor_release_gate
    
    # === Conditional DDL-protection gate: activated only under dual pressure (tight slack AND high uncertainty) ===
    # Prevents high-risk assignments when both DDL and uncertainty stress are present
    ddl_protection_gate = np.where(
        (norm_slack < PARAMS["ddl_protection_slack_threshold"]) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]),
        1.0,
        0.0
    )
    
    # === Bounded tanh-based anti-starvation: smooth, discriminative, saturates at ±1 ===
    # Avoids unbounded growth and provides stable gradient; scaled by wait_decay
    wait_benefit = np.tanh(PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty coupling: multiplicative penalty only under DDL-protection condition ===
    # Ensures energy-risk tradeoff is considered *only* where DDL violation risk is active
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_protection_gate
    
    # === Composite score: smaller = higher priority ===
    # Prioritizes: (1) lateness penalty, (2) critical-path boost under pressure, (3) low energy, (4) short duration,
    #              (5) anti-starvation, (6) DDL-protection coupling, (7) successor release.
    score = (
        + norm_slack_penalty                           # Lateness risk penalty
        - boosted_rank                                 # Critical path importance amplified under pressure
        - PARAMS["energy_sensitivity"] * norm_energy  # Energy minimization
        - norm_duration                                # Duration minimization
        - wait_benefit                                 # Anti-starvation (tanh-scaled)
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned penalty under DDL threat
        + PARAMS["duration_robustness"] * norm_duration * ddl_protection_gate  # Slow tasks penalized only under DDL+uncert
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
