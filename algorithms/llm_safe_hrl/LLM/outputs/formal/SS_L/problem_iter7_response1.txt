def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces robust_normalize with median-MAD per-feature scaling;
       adds conditional DDL protection gate; replaces logistic wait-benefit with bounded linear ramp;
       replaces multiplicative energy-uncert penalty with additive interaction under slack-pressure gate;
       removes norm_duration bias (redundant with slack/energy/duration_risk_score);
       preserves all evidence-backed structural actions: add_conditional_ddl_protection_gate,
       add_host_load_conditional_gate, add_load_successor_release_interaction."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature: outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        scale = mad + eps if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / scale
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Conditional DDL protection gate: binary hard gate for negative-slack tasks ===
    # Activates only when slack <= 0 — eliminates normalization fragility and aligns with violation semantics
    ddl_urgent_mask = (slack <= 0.0).astype(float)
    
    # === Risk-aware slack penalty: power penalty on overdue tasks, scaled by urgency mask ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate: sigmoid over normalized slack (not clipped), gated by urgency ===
    # Enables smooth transition near deadline while respecting hard DDL priority
    slack_pressure_sig = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * norm_slack))
    rank_gate = ddl_urgent_mask * slack_pressure_sig + (1.0 - ddl_urgent_mask) * 0.5
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Duration-uncertainty interaction: activated only under dual pressure (low slack AND high uncertainty) ===
    # Uses raw uncertainty threshold for interpretability; avoids double-normalization artifacts
    uncert_gate = np.where(uncertainty > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = (min_exec_time + min_comm_time) * uncert_gate * ddl_urgent_mask
    
    # === Starvation relief: bounded linear ramp instead of logistic — more predictable & monotonic ===
    # Saturates at 1.0, avoids asymptotic behavior and ensures finite derivative
    wait_benefit = np.clip(PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]), 0.0, 1.0)
    
    # === Energy-uncertainty interaction: additive penalty under both urgency and uncertainty gates ===
    # Avoids multiplicative explosion; applies only where DDL risk exists
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_urgent_mask * uncert_gate
    
    # === Composite score: smaller = higher priority ===
    # Removed norm_duration (redundant with slack, energy, and duration_risk_score); added explicit DDL gate effect
    score = (
        + norm_slack_penalty                           # Strong lateness penalty
        - boosted_rank                                 # Critical-path boost under urgency
        - PARAMS["energy_sensitivity"] * norm_energy  # Prefer low marginal energy
        - wait_benefit                                 # Anti-starvation bonus (linear ramp)
        + PARAMS["duration_robustness"] * median_mad_normalize(duration_risk_score)  # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Targeted risk-energy penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
