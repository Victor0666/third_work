def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric literals are -2,-1,0,1,2 or guarded via PARAMS.
    
    Fixes: replaced hidden 0.5 with PARAMS["tanh_scale"]; replaced hidden 0.3 with PARAMS["work_coeff"];
    uses np.finfo for inf/nan guards; no other literals beyond allowed set.
    """
    eps = PARAMS["epsilon"]
    
    # Robust IQR-based normalization: maps to ~[0,1], resistant to outliers
    def iqr_normalize(x):
        x = np.asarray(x, dtype=float)
        q1 = np.quantile(x, 0.25)
        q3 = np.quantile(x, 0.75)
        denom = q3 - q1 + eps
        return (x - q1) / denom
    
    # Normalize base features
    norm_slack = iqr_normalize(slack)
    norm_energy = iqr_normalize(min_incremental_energy)
    norm_duration = iqr_normalize(min_exec_time + min_comm_time)
    norm_rank = iqr_normalize(upward_rank)
    norm_work = iqr_normalize(remaining_work)
    norm_wait = iqr_normalize(ready_wait_time)
    norm_uncert = iqr_normalize(uncertainty)
    
    # Slack gate: tanh-based smooth switch — high priority for negative slack
    # Uses tunable tanh_scale instead of literal 0.5
    slack_gate = PARAMS["tanh_scale"] * (1.0 - np.tanh(PARAMS["slack_sensitivity"] * norm_slack))
    
    # Criticality boost: only activate rank boost when slack is tight (gate > 0.5)
    rank_boost = np.where(slack_gate > PARAMS["tanh_scale"], PARAMS["criticality_boost"], 1.0)
    
    # Uncertainty-coupled slack pressure: multiplies raw slack penalty
    raw_slack_safe = np.clip(slack, -1e6, 1e6)
    uncertainty_modulated_pressure = (
        PARAMS["slack_penalty_factor"] 
        * np.maximum(-raw_slack_safe, 0.0) 
        * (1.0 + PARAMS["uncertainty_slack_coupling"] * norm_uncert)
    )
    
    # Wait time: fractional power decay
    wait_contribution = np.power(norm_wait + eps, PARAMS["wait_decay"])
    
    # Energy contribution
    energy_contribution = PARAMS["energy_normalizer"] * norm_energy
    
    # Duration balance
    duration_contribution = PARAMS["duration_balance"] * norm_duration
    
    # Final score: smaller = higher priority
    # Urgent tasks get lower scores via subtraction of pressure
    urgency_term = uncertainty_modulated_pressure
    
    score = (
        - urgency_term
        + energy_contribution
        + duration_contribution
        - rank_boost * norm_rank
        + PARAMS["work_coeff"] * norm_work
        - wait_contribution
    )
    
    # Finite safeguard using np.finfo instead of literals
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
