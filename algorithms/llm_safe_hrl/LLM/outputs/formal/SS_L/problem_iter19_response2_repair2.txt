def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: replaces hidden 0.5 with PARAMS["rank_boost_lower_bound"];
       uses robust range-based normalization;
       introduces unified congestion signal (ready_wait_time + uncertainty) gated by DDL feasibility;
       enforces critical path leverage only under strict slack <= 0;
       replaces multiplicative ddl_gate with additive binary gate for numerical stability;
       all terms clipped to [-2,2] and fused with bounded coefficients."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust range-based normalization: handles outliers better than MAD for small N ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            xmin = xmax = x[0]
        else:
            xmin = np.min(x)
            xmax = np.max(x)
        spread = xmax - xmin if (xmax - xmin) > eps else eps
        return (x - xmin) / spread
    
    norm_slack = robust_range_normalize(slack)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # === Unified congestion signal: waiting time + uncertainty — direct proxy for resource contention ===
    congestion = ready_wait_time + uncertainty
    norm_congestion = robust_range_normalize(congestion)
    
    # === DDL protection gate: binary activation only when slack <= 0 → ensures energy/uncertainty vanish outside violation zone ===
    ddl_gate = (slack <= 0.0).astype(float)
    
    # === Critical path leverage: upward_rank * remaining_work, activated *only* under hard deadline breach ===
    critical_path_leverage = norm_rank * norm_work * ddl_gate
    
    # === Slack pressure: linear ramp from 0 to 1 over [0, 1] in normalized slack, flipped for urgency ===
    slack_pressure = np.clip(1.0 - norm_slack, 0.0, 1.0)
    
    # === Starvation relief: simple exponential saturation ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Energy-uncertainty interaction: gated by both DDL breach and congestion threshold ===
    congestion_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_congestion - PARAMS["uncertainty_gate_threshold"])))
    energy_uncert_penalty = norm_energy * norm_uncert * congestion_gate * ddl_gate
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_range_normalize(raw_slack_penalty)
    
    # === Bounded rank amplification: linear coupling under pressure, capped at [lower, upper] ===
    rank_boost = 1.0 + PARAMS["criticality_scale"] * slack_pressure
    rank_boost = np.clip(rank_boost, PARAMS["rank_boost_lower_bound"], PARAMS["rank_boost_upper_bound"])
    boosted_rank = norm_rank * rank_boost
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty dominates
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Unblock critical paths only when breached
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency scaling under pressure
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy only under breach
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(norm_congestion * ddl_gate, -2.0, 2.0)                   # Direct congestion penalty under breach
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)

    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
