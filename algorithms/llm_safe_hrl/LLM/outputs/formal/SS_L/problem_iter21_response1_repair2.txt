def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces duration_risk_score with explicit successor-release urgency;
       introduces hard DDL-protection via binary-like gating (smoothed but near-zero cutoff);
       removes inactive parameters and reassigns expressive capacity;
       uses successor-release term: upward_rank * (1 - exp(-remaining_work / (slack + eps))) — captures critical unblocking under deadline pressure;
       all risky terms (energy, uncertainty, interactions) are zeroed when slack < 0 via hard gate, enforcing strict feasibility-first ranking."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Hard DDL-protection gate: near-binary (smoothed) cutoff — sets energy/uncertainty contributions to ~0 when slack < 0 ===
    # Uses steep tanh for differentiability while achieving near-zero below slack=0
    hard_ddl_gate = PARAMS["hard_ddl_gate_offset"] + (1.0 - PARAMS["hard_ddl_gate_offset"]) * 0.5 * (1.0 + np.tanh(-PARAMS["hard_ddl_protection_enabled"] * PARAMS["hard_ddl_gate_steepness"] * slack))
    
    # === Successor-release urgency: upward_rank * (1 - exp(-remaining_work / (slack + eps))) ===
    # Directly models how releasing large downstream work accelerates critical path progress under deadline pressure
    # Avoids division by zero and handles slack <= 0 gracefully (yields 0 or bounded small value)
    release_denom = np.where(slack > 0, slack, eps)
    release_factor = 1.0 - np.exp(-remaining_work / (release_denom + eps))
    successor_release_urgency = norm_rank * release_factor
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded linear rank-slack coupling: stable monotonic amplification under pressure ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    
    # === Sharpened slack-pressure gate: sigmoid activation for pre-breach urgency ramp-up ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    
    # === Stable rank signal: linear coupling + gate (retained for robustness) ===
    stable_rank = norm_rank * rank_slack_coupling * rank_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Energy-uncertainty coupling: now strictly gated by hard_ddl_gate ===
    energy_uncert_penalty = norm_energy * norm_uncert * hard_ddl_gate
    
    # === Slack-aware energy scaling: replaced sigmoid ddl_gate with hard_ddl_gate ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * hard_ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - PARAMS["successor_release_gain"] * np.clip(successor_release_urgency, -2.0, 2.0)  # New core structural improvement
        - np.clip(stable_rank, -2.0, 2.0)                                  # Stabilized HEFT urgency
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * hard_ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
