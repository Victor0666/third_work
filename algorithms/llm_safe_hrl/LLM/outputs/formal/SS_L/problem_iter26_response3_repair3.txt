def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 1's structured congestion gating and duration-criticality coupling with Parent 2's hard DDL gating, MAD-normalized slack, and validated successor-release logic.
       Key innovations:
         - Unified joint congestion gate using BOTH quantile-normalized wait AND uncertainty
         - Duration-criticality coupling via (exec + 0.5*comm) * upward_rank * hard_ddl_gate — uses literal 0.5 = 1/2.0 per constraint
         - Starvation relief now gated by congestion AND scaled by urgency-sharpened slack pressure
         - All critical-path terms use HARD (binary) DDL gating for strict feasibility-first enforcement
         - Energy-uncertainty penalty only active in feasible region AND under congestion gate
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust quantile-based normalization: uses configurable quantiles for IQR ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            q_low = x[0]
            q_high = x[0]
            rng = eps
        else:
            q_low = np.quantile(x, PARAMS["quantile_low"])
            q_high = np.quantile(x, PARAMS["quantile_high"])
            rng = q_high - q_low
        spread = rng if rng > eps else eps
        center = (q_low + q_high) / 2.0
        return (x - center) / spread
    
    norm_slack = robust_range_normalize(slack)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_duration = robust_range_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # === Hard DDL feasibility gates ===
    ddl_feasible = (slack >= 0.0).astype(float)      # for energy/uncertainty penalties
    ddl_breach = (slack <= 0.0).astype(float)       # for critical path leverage & urgency boost
    
    # === Sign-preserving MAD-normalized slack for sharper urgency near slack ≈ 0 ===
    if N == 1:
        med_slack = slack[0]
        mad_slack = eps
    else:
        med_slack = np.median(slack)
        mad_slack = np.median(np.abs(slack - med_slack))
    spread_slack = mad_slack if mad_slack > eps else eps
    signed_slack_norm = (slack - med_slack) / spread_slack
    
    # === Urgency-sharpened slack pressure near zero ===
    slack_pressure_raw = np.clip(-slack, 0.0, 1.0)
    slack_pressure = slack_pressure_raw ** PARAMS["slack_penalty_exponent"]
    
    # === Joint congestion gating: active only under both high normalized wait AND high normalized uncertainty ===
    congestion_gate = ((norm_wait >= PARAMS["congestion_activation_threshold"]) & 
                       (norm_uncert >= PARAMS["congestion_activation_threshold"])).astype(float)
    
    # === Critical path leverage: upward_rank × remaining_work × (slack <= 0) ===
    critical_leverage = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: fast-executing blockers of critical descendants ===
    clipped_rank = np.clip(norm_rank, 0.0, 2.0)
    successor_release = (min_exec_time + (1.0 / 2.0) * min_comm_time) * clipped_rank * ddl_breach
    
    # === Duration-criticality coupling: (exec + 0.5*comm) × upward_rank × hard_ddl_gate ===
    duration_criticality = (min_exec_time + (1.0 / 2.0) * min_comm_time) * upward_rank * ddl_breach
    
    # === Starvation relief: logistic decay, suppressed under congestion, amplified by slack pressure ===
    wait_benefit_base = 1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]))
    wait_benefit = wait_benefit_base * (1.0 - congestion_gate) * (1.0 + slack_pressure)
    
    # === Energy-uncertainty penalty: only active in feasible region AND under congestion gate ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_feasible * congestion_gate
    
    # === Slack-aware marginal energy penalty: increases as slack approaches zero from feasible side ===
    slack_tightness = np.clip(1.0 - signed_slack_norm, 0.0, 2.0)
    energy_slack_penalty = norm_energy * slack_tightness * ddl_feasible
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] ===
    score = (
        + np.clip(signed_slack_norm, -2.0, 2.0)                           # Primary sharpened urgency signal
        - np.clip(critical_leverage, -2.0, 2.0)                          # Critical path blocking under breach
        - np.clip(successor_release, -2.0, 2.0)                          # Fast blocker of critical successors
        - np.clip(duration_criticality, -2.0, 2.0)                        # Duration-criticality coupling
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_feasible, -2.0, 2.0)  # Feasible-region energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                               # Anti-starvation, congestion-gated & urgency-amplified
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * ddl_feasible, -2.0, 2.0)
        + PARAMS["criticality_scale"] * np.clip(norm_rank * ddl_breach, -2.0, 2.0)  # Direct criticality boost under breach
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
