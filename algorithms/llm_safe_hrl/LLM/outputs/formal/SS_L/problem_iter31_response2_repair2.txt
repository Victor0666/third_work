def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: removes unused parameters; replaces sigmoid with tanh gates;
       introduces successor-release interaction via min_exec_time × upward_rank × (slack <= 0);
       uses MAD-clipped slack [-2,2] as primary urgency signal;
       gates all energy/uncertainty terms strictly by slack >= 0;
       eliminates inactive duration_robustness and wait_saturation_offset;
       retains median-MAD normalization and anti-starvation wait_benefit."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Bounded MAD-slack urgency: clipped to [-2,2] — sharper, feasibility-first aligned ===
    urgency_slack = np.clip(norm_slack, -2.0, 2.0)
    
    # === DDL-feasibility gate: strict activation (tanh-squashed) for energy/uncertainty terms ===
    ddl_feasible = 0.5 * (1.0 + np.tanh(PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical-path successor release: unblocks fast-executing critical predecessors ===
    ddl_breach = (slack <= 0.0).astype(float)
    successor_release = min_exec_time * upward_rank * ddl_breach
    norm_successor_release = median_mad_normalize(successor_release)
    
    # === Starvation relief: logistic-shaped anti-starvation bonus ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Energy-uncertainty coupling: gated only when feasible AND uncertainty is high ===
    uncert_gate = 0.5 * (1.0 + np.tanh(2.0 * (norm_uncert - PARAMS["tanh_shift"])))
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_feasible
    
    # === Feasibility-gated energy term ===
    energy_term = norm_energy * ddl_feasible
    
    # === Work-driven criticality boost: upward_rank × remaining_work interaction under breach ===
    critical_work_boost = norm_rank * norm_work * ddl_breach
    
    # === Composite score: smaller = higher priority; all terms clipped for stability ===
    score = (
        + urgency_slack                                 # Primary urgency signal
        - np.clip(critical_work_boost, -2.0, 2.0)     # Boost for blocking critical-path work under breach
        - np.clip(norm_successor_release, -2.0, 2.0)  # Unblock fast successors early
        - PARAMS["energy_sensitivity"] * np.clip(energy_term, -2.0, 2.0)  # Energy minimization only when safe
        - np.clip(wait_benefit, -2.0, 2.0)            # Anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        - PARAMS["criticality_scale"] * np.clip(norm_rank * (1.0 + 0.5 * np.tanh(-urgency_slack)), -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
