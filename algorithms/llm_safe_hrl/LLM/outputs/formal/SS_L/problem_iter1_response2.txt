import numpy as np


PARAMETER_SCHEMA = {
    "schema_version": 1,
    "parameters": [
        {
            "name": "epsilon",

        {
            "name": "criticality_boost",
            "initial_value": 1.5,

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: risk-gated slack penalty + critical-path-aware energy fairness.
    
    Key innovations:
      - Nonlinear slack penalty: (max(0, -slack))^p penalizes lateness risk sharply
      - Criticality boost: multiplies priority only when both high upward_rank AND tight slack
      - Risk-adjusted duration: combines duration & uncertainty via convex interpolation
      - Rank-slack coupling: modulates upward_rank importance based on slack pressure
      - Wait fairness: linear anti-starvation term with bounded gain
      - All features robustly normalized using mean-abs + epsilon; no division by std or var
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Ensure finite inputs and avoid in-place modification
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust normalization: x / (mean(|x|) + eps), handles zeros & negatives safely
    def robust_norm(x):
        x = np.asarray(x)
        mean_abs = np.mean(np.abs(x))
        scale = mean_abs + eps
        return x / scale
    
    # 1. Slack risk: sharp penalty for negative slack, zero otherwise
    # Use exponent >1 for stronger differentiation among late tasks
    neg_slack = np.maximum(0.0, -slack)
    slack_penalty = np.power(neg_slack + eps, PARAMS["slack_penalty_exponent"])
    
    # 2. Risk-adjusted duration: convex blend of duration and uncertainty
    duration = min_exec_time + min_comm_time
    duration_norm = robust_norm(duration)
    uncertainty_norm = robust_norm(uncertainty)
    risk_adjusted_duration = (
        PARAMS["duration_risk_ratio"] * duration_norm +
        (1.0 - PARAMS["duration_risk_ratio"]) * uncertainty_norm
    )
    
    # 3. Energy: normalize marginal fuzzy energy — lower is better → higher priority → smaller score
    energy_norm = robust_norm(min_incremental_energy)
    
    # 4. Upward rank: scale by slack pressure (tighter slack → higher rank importance)
    # Avoid amplifying low-rank tasks; use coupling coefficient <1 for soft gating
    rank_base = robust_norm(upward_rank)
    slack_pressure = robust_norm(slack)  # negative = urgent
    # Modulate rank impact: more important when slack is tight (i.e., slack_pressure is small/negative)
    rank_weighted = rank_base * (1.0 + PARAMS["rank_slack_coupling"] * np.maximum(0.0, -slack_pressure))
    
    # 5. Criticality boost: activate only when rank is above median AND slack is tight (< 0.5*median slack)
    median_rank = np.median(upward_rank)
    median_slack = np.median(slack)
    is_critical = (
        (upward_rank >= median_rank) &
        (slack <= np.maximum(eps, 0.5 * median_slack))
    ).astype(float)
    criticality_bonus = is_critical * PARAMS["criticality_boost"]
    
    # 6. Anti-starvation: linear wait boost, bounded to prevent dominance
    wait_norm = robust_norm(ready_wait_time)
    wait_bonus = PARAMS["wait_fairness_gain"] * wait_norm
    
    # 7. Uncertainty-slack gate: tighten effective slack when uncertainty is high
    # Only apply if uncertainty exceeds threshold AND slack is already tight
    unc_gate_active = (
        (uncertainty_norm > PARAMS["uncertainty_slack_gate"]) &
        (slack < eps)
    ).astype(float)
    # Tighten slack penalty further: add scaled uncertainty contribution
    tightened_penalty = slack_penalty + unc_gate_active * 0.5 * uncertainty_norm
    
    # Composite score: smaller = higher priority
    # Prioritize: (1) deadline risk (penalty), (2) energy efficiency, (3) duration, (4) critical path,
    #             (5) fairness, (6) anti-starvation — all with robust scaling
    score = (
        tightened_penalty  # primary driver: punish lateness aggressively
        + PARAMS["energy_efficiency_scale"] * energy_norm
        + 0.8 * risk_adjusted_duration
        - 0.9 * rank_weighted  # subtract because higher rank = higher priority → lower score
        + 0.1 * criticality_bonus  # small additive boost for truly critical cases
        - wait_bonus  # subtract wait bonus → longer wait → lower score → higher priority
    )
    
    # Final safeguard: replace NaN/inf with finite values, preserve shape
    score = np.nan_to_num(score, nan=np.finfo(float).max, posinf=np.finfo(float).max, neginf=np.finfo(float).min)
    
    # Ensure output is finite and shape-(N,)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values in priority score"
    
    return score
