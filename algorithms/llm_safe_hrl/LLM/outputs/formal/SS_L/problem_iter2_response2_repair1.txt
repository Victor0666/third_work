def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robust normalization (MAD-based), tight-slack criticality boost,
    uncertainty-slack tightening, and energy-slack interaction — all parameterized for CMA-ES tuning.
    
    Key structural improvements:
      - Reintroduces MAD-based robust_normalize with tunable robustness factor.
      - Adds explicit energy-slack interaction: penalizes high incremental energy *more* when slack is negative.
      - Preserves Parent 2's median-based criticality gate and uncertainty-slack tightening.
      - All numeric literals are -2, -1, 0, 1, or 2; no hidden thresholds; all parameters declared and used.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # === Robust normalization: MAD-based with tunable scaling factor ===
    def robust_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        scale = mad * PARAMS["robustness_mad_factor"] + eps
        return (x - center) / scale
    
    # Normalize base features (finite, deterministic, scale-invariant)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Slack risk: sharp nonlinear penalty for negative slack ===
    neg_slack = np.maximum(0.0, -slack)  # raw seconds → preserve magnitude scale
    slack_penalty = np.power(neg_slack + eps, PARAMS["slack_penalty_exponent"])
    
    # === Risk-adjusted duration: convex blend of normalized duration and uncertainty ===
    risk_adjusted_duration = (
        PARAMS["duration_risk_ratio"] * norm_duration +
        (1.0 - PARAMS["duration_risk_ratio"]) * norm_uncert
    )
    
    # === Energy-slack interaction: amplify energy penalty when urgency is high ===
    # Uses raw slack (not norm) for physical interpretability of thresholding
    slack_pressure_raw = np.maximum(0.0, -slack)
    energy_slack_interaction = norm_energy * (1.0 + PARAMS["energy_slack_interaction_weight"] * slack_pressure_raw)
    
    # === Upward rank: weighted by slack pressure (tighter slack → higher importance) ===
    slack_pressure_norm = np.maximum(0.0, -norm_slack)
    rank_weighted = norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure_norm)
    
    # === Criticality boost: activate only when rank ≥ median AND slack ≤ threshold × median_slack ===
    median_rank = np.median(upward_rank)
    median_slack = np.median(slack)
    is_critical = (
        (upward_rank >= median_rank) &
        (slack <= np.maximum(eps, PARAMS["slack_median_ratio_threshold"] * median_slack))
    ).astype(float)
    criticality_bonus = is_critical * PARAMS["criticality_boost"]
    
    # === Anti-starvation: linear wait benefit, bounded and normalized ===
    wait_bonus = PARAMS["wait_fairness_gain"] * norm_wait
    
    # === Uncertainty-slack tightening: add uncertainty-driven penalty only when slack < ε AND uncertainty high ===
    unc_gate_active = (
        (norm_uncert > PARAMS["uncertainty_slack_gate"]) &
        (slack < eps)
    ).astype(float)
    tightened_penalty = slack_penalty + unc_gate_active * PARAMS["uncertainty_slack_tightening_weight"] * norm_uncert
    
    # === Final score: smaller = higher priority ===
    # Components:
    #   + tightened_penalty           → dominant urgency signal
    #   + energy_slack_interaction  → energy cost scaled by urgency (novel interaction)
    #   + risk_adjusted_duration    → duration + uncertainty risk
    #   - rank_weighted             → reward critical-path progress
    #   - wait_bonus                → fairness relief
    #   + 0 * criticality_bonus     → preserved for semantic clarity (not active in base score;
    #                                 may be activated via future param tuning)
    score = (
        tightened_penalty
        + energy_slack_interaction
        + 1 * risk_adjusted_duration
        - 1 * rank_weighted
        - wait_bonus
    )
    
    # Final safeguard: replace NaN/inf with finite values, preserve shape
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure output is finite and shape-(N,)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values in priority score"
    
    return score
