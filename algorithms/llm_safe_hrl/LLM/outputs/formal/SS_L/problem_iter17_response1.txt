def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: merges best practices from both parents.
       - Retains Parent 2's robust median-MAD normalization, linear slack penalty, unified ddl_gate, and duration_robustness.
       - Adopts Parent 1's crisp *dual-gated* rank-uncertainty coupling but replaces binary ddl_infeasible with graded ddl_pressure (smooth monotonic activation).
       - Introduces novel starvation relief using clipped *logarithmic* form with saturating offset — preserves discriminability at low wait times while preventing singularity.
       - Removes redundant terms (e.g., separate exec_penalty) and unifies all feasibility-sensitive terms under ddl_gate + ddl_pressure for tighter DDL alignment.
       - All interactions bounded and clipped to [-2,2] for stability and ranking integrity."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization: sign-preserving, outlier-resilient scaling ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad + eps
        return (x - med) / spread
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === DDL-protection gate: smooth step (0→1) over [0,1] slack range → fully suppresses non-feasibility terms when slack < 0 ===
    ddl_gate = np.where(slack < 0.0, 0.0, np.clip(slack, 0.0, 1.0))
    
    # === DDL-pressure signal: monotonic, normalized urgency proxy — negative slack mapped to [0,1] ===
    ddl_pressure = np.clip(-norm_slack, 0.0, 1.0)
    
    # === Linear slack penalty with tunable coefficient ===
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    slack_penalty = PARAMS["slack_penalty_linear_coeff"] * norm_slack_penalty
    
    # === Successor-release interaction: multiplicative coupling of rank and downstream work, activated by ddl_pressure ===
    successor_release = norm_rank * norm_work * ddl_pressure
    
    # === Critical-path coupling: upward_rank scaled linearly by ddl_pressure ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_pressure)
    
    # === Uncertainty-aware energy penalty: gated by ddl_gate AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: logarithmic saturation with safe offset — preserves sensitivity at low wait times ===
    wait_benefit = np.log1p(PARAMS["wait_decay"] * (ready_wait_time + PARAMS["starvation_saturation_offset"]))
    wait_benefit = np.clip(wait_benefit, 0.0, 2.0)  # Bound for stability
    
    # === Robustified duration penalty: damped and fully gated by ddl_gate and ddl_pressure ===
    duration_penalty = PARAMS["duration_robustness_factor"] * norm_duration * ddl_pressure * ddl_gate
    
    # === Graded rank-uncertainty coupling: replaces binary infeasibility gate with smooth ddl_pressure activation ===
    # Captures joint criticality-risk exposure without hard thresholds
    rank_uncert_coupling = norm_rank * norm_uncert * ddl_pressure * PARAMS["rank_uncert_coupling_strength"]
    
    # === Composite score: smaller = higher priority; all terms clipped individually to [-2,2] ===
    score = (
        + np.clip(slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                      # Release-driven criticality boost
        - np.clip(coupled_rank, -2.0, 2.0)                           # Pressure-scaled critical path
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                           # Log-saturated anti-starvation
        + np.clip(duration_penalty, -2.0, 2.0)                       # Damped, pressure-gated duration penalty
        + np.clip(rank_uncert_coupling, -2.0, 2.0)                   # Graded criticality-risk coupling
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)   # Downstream work bias
        + np.clip(ddl_pressure * PARAMS["ddl_protection_gate_slope"], -2.0, 2.0)  # Direct pressure signal
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
