def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces hard DDL gate with adaptive sigmoid gate;
       introduces deadline-aware relative normalization (x / (|slack| + ε)) for duration & energy;
       adds critical-path starvation relief: wait_benefit × indicator(upward_rank > median ∧ slack < 0);
       retains power-law duration penalty, load-proxy synthesis, and bounded [-2,2] composition.
       All gates and normalizations preserve feasibility-first ordering while enabling smoother gradients
       for offline CMA-ES optimization."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Deadline-aware relative normalization: amplifies sensitivity near deadlines
    # Uses |slack| + eps as dynamic scale — smaller slack → larger normalized values → stronger impact
    slack_abs = np.abs(slack) + eps
    norm_duration_rel = (min_exec_time + min_comm_time) / slack_abs
    norm_energy_rel = min_incremental_energy / slack_abs
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Host-load proxy: synthesizes congestion signal from wait time & uncertainty
    load_proxy = 1.0 / (1.0 + np.exp(-2.0 * (norm_wait + norm_uncert)))
    
    # === Soft DDL gate: sigmoid-smoothed transition instead of hard step; preserves gradient
    soft_ddl_gate = 1.0 / (1.0 + np.exp(5.0 * (-slack)))  # Fixed steepness (5.0) to stay within literal limits
    
    # === Tunable linear slack pressure: ramps from 0→1 over [0, -width] slack
    slack_pressure = np.clip(-slack / (PARAMS["slack_pressure_width"] + eps), 0.0, 1.0)
    
    # === Successor-release interaction: linear-coupled rank × work, activated only under deadline pressure
    successor_release = norm_rank * norm_work * slack_pressure
    
    # === Linear slack penalty: bounded linear penalty for negative slack
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Critical-path coupling: upward_rank scaled by linear slack pressure
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_pressure)
    
    # === Uncertainty-aware energy penalty: gated by soft DDL gate AND uncertainty threshold
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * soft_ddl_gate
    
    # === Starvation relief: linear ramp-up with explicit saturation at 1.0
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Critical-path starvation relief: activates only for high-rank tasks under lateness risk
    # Use fixed median threshold logic without tunable parameter to reduce count
    rank_median = np.median(upward_rank) if N > 1 else upward_rank[0]
    critical_starvation_gate = np.where((upward_rank > rank_median) & (slack < 0.0), 1.0, 0.0)
    critical_wait_benefit = wait_benefit * critical_starvation_gate
    
    # === Power-law duration penalty using deadline-aware normalized duration
    duration_base = np.abs(norm_duration_rel) * slack_pressure * soft_ddl_gate
    exec_penalty = np.power(np.abs(duration_base) + eps, PARAMS["duration_penalty_exponent"])
    
    # === Host-load–aware energy modulation: scales energy penalty *only* when load proxy exceeds threshold
    load_gate = np.where(load_proxy > PARAMS["host_load_gate_threshold"], 1.0, 0.0)
    load_scaled_energy = norm_energy * (1.0 + PARAMS["host_load_sensitivity"] * load_gate)
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2]
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Successor-release boost
        - np.clip(coupled_rank, -2.0, 2.0)                                 # Critical-path urgency
        - PARAMS["energy_sensitivity"] * np.clip(load_scaled_energy * soft_ddl_gate, -2.0, 2.0)  # Load-modulated energy
        - np.clip(critical_wait_benefit, -2.0, 2.0)                        # Critical-path starvation relief
        + np.clip(exec_penalty, -2.0, 2.0)                                 # Power-law duration penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(slack_pressure * PARAMS["ddl_protection_gate_slope"], -2.0, 2.0)  # Direct pressure signal
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
