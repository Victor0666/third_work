def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: merges Parent 2's robustness-driven duration penalty and clipped linear DDL gate with Parent 1's validated successor-release coupling;
       introduces power-law slack nonlinearity (exponent >1) to sharpen ranking among critically tight tasks without brittle sigmoid artifacts;
       retains median-MAD normalization for outlier resilience;
       replaces synthetic host-load proxy with direct, interpretable robustness logic;
       enforces strict feasibility-first via ddl_gate (0 when slack<0), while allowing smooth gradient flow for optimization;
       all terms bounded in [-2,2] to ensure finite AST depth and CMA-ES stability."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    total_duration = min_exec_time + min_comm_time
    norm_duration = median_mad_normalize(total_duration)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Clipped linear DDL gate: 0 for slack < 0, ramps linearly to 1 over [0,1] sec → feasibility-first + smooth gradient ===
    ddl_gate = np.clip(slack, 0.0, 1.0)
    ddl_gate = np.where(slack < 0.0, 0.0, ddl_gate)
    
    # === Power-law slack pressure: enhances discrimination among critical tasks (exponent >1), bounded and stable ===
    # Replaces linear ramp to better separate high-priority candidates near deadline violation
    raw_slack_pressure = np.maximum(-slack, 0.0)
    slack_pressure = np.power(raw_slack_pressure + eps, PARAMS["slack_nonlinearity_exponent"])
    # Normalize pressure range to [0,1] for composability
    max_pressure = np.max(slack_pressure) if N > 1 else slack_pressure[0]
    ddl_pressure = np.where(max_pressure > eps, slack_pressure / (max_pressure + eps), 0.0)
    
    # === Successor-release interaction: linear-coupled rank × work, activated only under deadline pressure (validated) ===
    successor_release = norm_rank * norm_work * ddl_pressure
    
    # === Linear slack penalty: bounded, monotonic, numerically stable ===
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Critical-path coupling: upward_rank scaled by nonlinear pressure ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_pressure)
    
    # === Uncertainty-aware energy penalty: gated by both DDL feasibility AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Starvation relief: linear ramp-up with explicit saturation ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Robustness-adjusted duration penalty: penalizes high-uncertainty *and* long durations under pressure ===
    robust_duration_penalty = norm_duration * norm_uncert * ddl_pressure * PARAMS["duration_robustness_weight"]
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for bounded AST depth and stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Validated linear successor-release boost
        - np.clip(coupled_rank, -2.0, 2.0)                                 # Critical-path urgency with nonlinear coupling
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + np.clip(robust_duration_penalty, -2.0, 2.0)                      # Uncertainty-weighted duration robustness
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(ddl_pressure * PARAMS["ddl_protection_gate_slope"], -2.0, 2.0)  # Direct nonlinear pressure signal
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
