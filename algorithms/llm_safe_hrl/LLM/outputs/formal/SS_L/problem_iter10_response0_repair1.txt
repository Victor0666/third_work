def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: adds conditional DDL protection gate, successor-release interaction, 
    uses clipped tanh for bounded anti-starvation, and declares all numeric coefficients as parameters."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization per feature ===
    def robust_normalize(x):
        x = np.abs(x)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) + eps if N > 1 else eps
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Conditional DDL protection gate: binary urgency for tasks with slack <= 0 ===
    # Strongly prioritizes overdue or at-deadline tasks regardless of other factors
    ddl_urgent_mask = (slack <= eps).astype(float)
    
    # === Risk-aware slack penalty: power-based penalty only for negative slack ===
    raw_slack_penalty = np.where(slack < 0, (-slack) ** PARAMS["slack_penalty_exponent"], 0.0)
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate: sigmoid activation for critical-path boosting ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Successor-release interaction: reward tasks whose completion unblocks many successors ===
    # Approximated by upward_rank * remaining_work — captures both criticality and release impact
    successor_release_score = norm_rank * norm_work
    
    # === Duration-uncertainty interaction: penalize slow execution under high uncertainty ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = norm_duration * uncert_gate * slack_pressure
    
    # === Bounded anti-starvation: tanh-based benefit avoids saturation artifacts ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * norm_wait)
    
    # === Energy-uncertainty interaction: multiplicative penalty for risky+energy-intensive tasks ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate
    
    # === Composite score: smaller = higher priority ===
    # DDL-urgent tasks receive strongest priority boost (negative contribution)
    score = (
        + norm_slack_penalty                           # Lateness penalty (positive → lower priority)
        - ddl_urgent_mask * PARAMS["ddl_urgency_boost"]  # Hard DDL protection: immediate priority for overdue
        - boosted_rank                                 # Critical path boost under pressure
        - PARAMS["energy_sensitivity"] * norm_energy  # Energy minimization
        - norm_duration                                # Duration minimization
        - wait_benefit                                 # Anti-starvation (tanh ensures bounded [-1,1])
        + PARAMS["duration_robustness"] * duration_risk_score  # Risk-duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned penalty
        + PARAMS["remaining_work_weight"] * successor_release_score  # Prefer tasks releasing large work
    )
    
    # Final safeguard: ensure finite output and enforce shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Ensure shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
