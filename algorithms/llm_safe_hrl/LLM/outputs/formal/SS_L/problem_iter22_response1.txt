def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces median-MAD with quantile-based robust normalization;
       introduces congestion-aware gate using `ready_wait_time + uncertainty`;
       removes redundant `duration_robustness` and `wait_saturation_offset` per evidence;
       enforces sign-preserving MAD only on slack; uses clipped linear `wait_benefit`;
       adds conditional `ddl_protection_gate` activated only for slack >= 0;
       combines critical path leverage via `upward_rank * remaining_work` gated by DDL feasibility."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Quantile-based robust normalization (Q1/Q3 range) for all features except slack ===
    def quantile_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1, q3 = x[0], x[0]
        else:
            q1 = np.quantile(x, 0.25)
            q3 = np.quantile(x, 0.75)
        iqr = q3 - q1 if (q3 - q1) > eps else eps
        return (x - (q1 + q3) / 2.0) / iqr
    
    # === Sign-preserving MAD normalization *only* for slack (sharpened urgency near deadline violation) ===
    def mad_normalize_slack(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = mad_normalize_slack(slack)
    norm_energy = quantile_normalize(min_incremental_energy)
    norm_duration = quantile_normalize(min_exec_time + min_comm_time)
    norm_rank = quantile_normalize(upward_rank)
    norm_work = quantile_normalize(remaining_work)
    norm_wait = quantile_normalize(ready_wait_time)
    norm_uncert = quantile_normalize(uncertainty)
    
    # === Binary DDL feasibility gate: 1 if slack >= 0, else 0 — eliminates penalty leakage ===
    ddl_feasible = (slack >= 0.0).astype(float)
    
    # === Conditional DDL protection gate: sigmoid activated *only* when slack >= 0 ===
    # Enables energy/uncertainty penalties exclusively in DDL-feasible region
    ddl_protection_gate = ddl_feasible * (
        1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (slack - 0.0)))
    )
    
    # === Critical path leverage: upward_rank × remaining_work, gated by DDL feasibility ===
    # Validated for 'higher_ddl_risk_choice' and 'critical_path_or_successor_blocked'
    critical_leverage = norm_rank * norm_work * ddl_feasible
    
    # === Risk penalty: power-law on raw negative slack, then quantile-normalized ===
    raw_slack_risk = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_risk = quantile_normalize(raw_slack_risk)
    
    # === Congestion signal: unified load proxy = ready_wait_time + uncertainty, gated jointly ===
    congestion = ready_wait_time + uncertainty
    congestion_gate = ddl_feasible * (
        1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * 
                           (congestion - PARAMS["uncertainty_gate_threshold"] * np.quantile(congestion, 0.75))))
    )
    
    # === Starvation relief: clipped linear wait_benefit (improves low-wait discriminability) ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Energy-uncertainty interaction: only active when both DDL-feasible and uncertain ===
    energy_uncert_interaction = norm_energy * norm_uncert * ddl_protection_gate * congestion_gate
    
    # === Composite score: smaller = higher priority; all terms bounded and deterministic ===
    score = (
        + norm_slack_risk                                    # Lateness risk penalty (strongest signal)
        - critical_leverage                                # Boost for critical downstream work under feasibility
        - norm_rank * (1.0 + PARAMS["criticality_scale"] * np.clip(-norm_slack, 0.0, 1.0))  # Urgency amplification near deadline
        - PARAMS["energy_sensitivity"] * norm_energy * ddl_protection_gate  # Energy preference only when safe
        - wait_benefit                                     # Anti-starvation (linear, no saturation offset)
        + congestion_gate * norm_duration                  # Load-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_interaction
        + PARAMS["remaining_work_weight"] * norm_work * ddl_feasible  # Downstream work bonus only when feasible
    )
    
    # Final safeguards: ensure finite output, shape (N,), no NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
