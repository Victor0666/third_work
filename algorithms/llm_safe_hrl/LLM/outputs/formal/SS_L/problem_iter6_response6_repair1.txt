def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: adds DDL-protection gate (criticality boost only when rank > median AND slack < 0);
       introduces congestion_penalty using ready_wait_time as load proxy under uncertainty;
       replaces linear slack_reward with piecewise-linear ramp: zero for slack < 0, linear from 0 to max over [0, median_slack], then saturates;
       all literals are -2,-1,0,1,2; epsilon via PARAMS; no side effects."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Sign-preserving robust normalization: median-center + MAD scale + epsilon ===
    def robust_normalize(x):
        center = np.median(x)
        scale = np.median(np.abs(x - center)) + eps
        return (x - center) / scale
    
    # Normalize base features — preserves physical sign semantics
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # Compute median_slack for piecewise slack reward
    median_slack = np.median(slack)
    
    # === Unified slack pressure: power-law penalty only for negative slack ===
    slack_pressure = np.power(np.clip(-norm_slack, 0.0, None), PARAMS["slack_penalty_exponent"])
    
    # === DDL-protection critical-path gate: activate boost only if slack < 0 AND norm_rank > median_norm_rank ===
    median_norm_rank = np.median(norm_rank)
    ddl_protection_gate = (
        (norm_slack < 0.0) &
        (norm_rank > median_norm_rank)
    ).astype(float)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_protection_gate)
    
    # === Uncertainty-duration interaction: activate only under both high uncertainty AND deadline pressure ===
    uncert_gate = (
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) &
        (slack_pressure > 0)
    ).astype(float)
    duration_risk_interaction = norm_duration * uncert_gate * PARAMS["duration_robustness"]
    
    # === Energy-uncertainty interaction: penalize high-energy tasks when uncertainty is high and slack is tight ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * PARAMS["energy_uncertainty_interaction"]
    
    # === Starvation relief: bounded tanh using sign-preserving norm_wait ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * norm_wait)
    
    # === Piecewise-linear slack reward: zero for slack < 0; linear ramp from 0 to PARAMS["slack_sensitivity"] over [0, median_slack]; saturate beyond ===
    # Ensures early scheduling bias without over-rewarding extreme slack
    slack_reward = np.zeros_like(norm_slack)
    positive_slack_mask = (slack >= 0.0)
    if N > 0 and np.any(positive_slack_mask):
        ramp_region = (slack > 0.0) & (slack <= median_slack)
        saturation_region = slack > median_slack
        slack_reward[ramp_region] = PARAMS["slack_sensitivity"] * (slack[ramp_region] / (median_slack + eps))
        slack_reward[saturation_region] = PARAMS["slack_sensitivity"]
    
    # === Slack-gated work leverage: prefer unblocking large downstream work only when slack is non-negative ===
    work_leverage_gate = (slack >= 0.0).astype(float)
    work_leverage = norm_work * work_leverage_gate * PARAMS["work_leverage_factor"]
    
    # === Congestion penalty: penalize long-waiting tasks under high uncertainty (host-load proxy) ===
    # Uses ready_wait_time as observable congestion signal when VM load not available
    congestion_penalty = norm_wait * uncert_gate * PARAMS["congestion_penalty_weight"]
    
    # === Final score: smaller = higher priority ===
    score = (
        + slack_pressure                                  # Urgency penalty (dominant for late tasks)
        + PARAMS["energy_sensitivity"] * norm_energy     # Energy cost (penalty)
        - boosted_rank                                   # Critical path reward (DDL-protected)
        + duration_risk_interaction                      # Risk-duration penalty (conditional)
        - wait_benefit                                   # Starvation relief (reward)
        - slack_reward                                   # Piecewise-positive-slack reward
        + energy_uncert_penalty                          # High-risk-high-energy penalty (conditional)
        - work_leverage                                  # Large-work unblocking benefit (safe context only)
        + congestion_penalty                           # Load-proxy penalty under uncertainty
    )
    
    # Final safeguard: ensure finite, deterministic, shape-(N,) output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly
    return score.reshape(-1)
