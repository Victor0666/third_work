def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: combines Parent 2's lateness multiplier with Parent 1's robustness logic;
       adds novel work-slack coupling to prioritize high-computation tasks under deadline pressure;
       replaces fragile norm_slack clipping with monotonic slack-pressure-based interactions;
       uses MAD-only normalization without absolute-value clipping to preserve rank ordering fidelity;
       enforces hard-deadline compliance via selective lateness scaling of penalties only;
       eliminates redundant gates while retaining smooth uncertainty activation and saturating wait relief."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median-centered MAD only (no abs clipping) → preserves sign & rank under stress ===
    def robust_normalize(x):
        if N == 0:
            return np.zeros(0, dtype=float)
        center = np.median(x)
        scale = np.median(np.abs(x - center)) + eps
        return (x - center) / (scale + eps)
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack pressure: monotonic clipped power (avoids zero-division & preserves order) ===
    slack_pressure_raw = np.clip(-norm_slack, 0.0, None)
    slack_pressure = np.power(slack_pressure_raw + eps, PARAMS["slack_penalty_exponent"])
    
    # === Lateness penalty gate: multiplicative enforcement only on penalties when critically overdue ===
    is_late = (slack < -eps).astype(float)
    lateness_multiplier = 1.0 + PARAMS["lateness_penalty_strength"] * is_late
    
    # === Critical-path acceleration gate: smooth linear ramp down from 1→0 over [0, slack_gate_width] ===
    rank_gate = np.clip(1.0 - (norm_slack / (PARAMS["slack_gate_width"] + eps)), 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Soft uncertainty activation: graded sigmoidal response (no threshold gate needed) ===
    uncertainty_activation = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_smoothness"] * (norm_uncert - 0.0)))
    duration_risk_interaction = norm_duration * uncertainty_activation * PARAMS["duration_robustness"]
    
    # === Starvation relief: saturating exponential form ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + eps))
    
    # === Energy-slack interaction: penalizes high marginal energy more under tight slack ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["energy_slack_interaction"] * slack_pressure)
    
    # === Restored rank-slack coupling: strengthens critical path reward exactly when slack is constrained ===
    coupled_rank_reward = norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure)
    
    # === Novel work-slack coupling: prioritizes high remaining_work tasks when slack is negative ===
    # Enables large-task acceleration under urgency without violating precedence or resource constraints
    work_slack_penalty = norm_work * (1.0 + PARAMS["work_slack_coupling"] * slack_pressure)
    
    # === Final score: smaller = higher priority ===
    # Apply lateness multiplier only to penalty terms (not rewards) to preserve reward integrity
    # Slack pressure remains primary driver; work coupling adds structural novelty for compute-heavy workflows
    score = (
        + lateness_multiplier * slack_pressure                # Primary urgency driver
        + lateness_multiplier * energy_slack_penalty          # Deadline-amplified energy cost
        + lateness_multiplier * PARAMS["energy_sensitivity"] * norm_energy  # Base energy penalty
        + lateness_multiplier * work_slack_penalty             # Novel compute-load acceleration under urgency
        - coupled_rank_reward                                # Critical path reward (untouched by lateness)
        - wait_benefit                                       # Anti-starvation relief (untouched)
        + lateness_multiplier * duration_risk_interaction    # Graded risk-duration penalty
    )
    
    # Final safeguard: ensure finite, deterministic output with strict shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    return score
