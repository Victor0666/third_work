def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's superior median-MAD normalization and conditional anti-starvation with Parent 1's quantile-inspired slack-sharpening insight;
       introduces novel exec-comm coupling term under slack pressure to model I/O-bound compute bottlenecks;
       replaces linear rank amplification with power-law coupling (upward_rank ^ rank_slack_nonlinear_coupling) for stronger critical-path differentiation near deadlines;
       retains all proven gates: ddl_protection_gate (enables penalties only when slack >= 0), uncertainty_gate, and conditional wait_activation;
       removes redundant terms to keep AST depth bounded and feature interactions ≤ 8."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving (Parent 2 strength) ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_exec = median_mad_normalize(min_exec_time)
    norm_comm = median_mad_normalize(min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 (Parent 2) ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Slack pressure signal: sigmoid over clipped negative slack → sharp urgency transition (Parent 2) ===
    raw_slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    slack_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (raw_slack_pressure - 1.0)))
    
    # === Tight-coupled critical-path booster: feasibility-aware (Parent 2) ===
    critical_path_booster = norm_rank * norm_work * ddl_gate
    
    # === Risk-adjusted slack penalty: power law on raw negative slack (Parent 2) ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Nonlinear rank-slack coupling: upward_rank raised to slack-pressure-modulated exponent (novel) ===
    # Stronger differentiation than linear amplification near violation boundary
    coupled_rank_exponent = 1.0 + PARAMS["rank_slack_nonlinear_coupling"] * raw_slack_pressure
    coupled_rank = np.power(np.abs(norm_rank) + eps, coupled_rank_exponent) * np.sign(norm_rank)
    
    # === Uncertainty gating: threshold-triggered for coupling terms only (Parent 2) ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    
    # === Conditional anti-starvation gate: activated only when wait exceeds adaptive threshold (Parent 2) ===
    wait_median = np.median(ready_wait_time) if N > 1 else ready_wait_time[0]
    wait_threshold = PARAMS["wait_activation_threshold_ratio"] * (wait_median + PARAMS["wait_saturation_offset"])
    wait_activation = 1.0 / (1.0 + np.exp(-PARAMS["wait_benefit_steepness"] * (ready_wait_time - wait_threshold)))
    wait_benefit = wait_activation * (1.0 - np.exp(-1.0 * (ready_wait_time - wait_threshold + PARAMS["wait_saturation_offset"])))
    
    # === Energy-uncertainty coupling: gated by both uncertainty and DDL feasibility (Parent 2) ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Exec-comm coupling under pressure: models bottleneck synergy when both compute & I/O are constrained (novel) ===
    exec_comm_coupling = (norm_exec + norm_comm) * PARAMS["exec_comm_coupling_weight"] * slack_pressure_gate * ddl_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability and scale balance ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_booster, -2.0, 2.0)                       # Feasibility-gated critical path unblocking
        - np.clip(coupled_rank, -2.0, 2.0)                                  # Nonlinear rank-slack criticality (novel)
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Conditional anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + np.clip(exec_comm_coupling, -2.0, 2.0)                           # Novel bottleneck coupling
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
