def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces linear rank coupling with successor-release interaction;
       adds conditional DDL protection gate; uses median-MAD per-feature normalization;
       enforces hard deadline feasibility via binary urgency gating before composite scoring."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust per-feature normalization using median & MAD (preserves sign semantics) ===
    def robust_normalize(x):
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Hard deadline protection gate: binary urgency activation for slack <= 0 ===
    # Activates strict prioritization of overdue or critically tight tasks (empirically validated)
    ddl_urgent_mask = (slack <= 0.0).astype(float)
    
    # === Successor-release interaction: amplifies priority when task unblocks many successors ===
    # Captures structural DAG pressure: tasks with high remaining_work AND high upward_rank under urgency
    successor_release_score = norm_rank * norm_work * ddl_urgent_mask
    
    # === Conditional DDL protection gate: sharp sigmoid activation only when slack is negative ===
    # Ensures critical-path tasks dominate *only* when DDL violation risk exists
    slack_gap = np.maximum(-slack, 0.0)  # non-negative gap magnitude
    ddl_protection_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_gap - eps)))
    
    # === Upward-rank boosting only under DDL pressure — avoids over-prioritizing critical path unnecessarily ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_protection_gate)
    
    # === Energy-uncertainty interaction: only penalized under high uncertainty AND non-negligible energy ===
    energy_uncert_penalty = norm_energy * norm_uncert * np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    
    # === Duration-robustness penalty: active only when both uncertainty and slack pressure coexist ===
    duration_risk_score = norm_duration * norm_uncert * ddl_protection_gate
    
    # === Anti-starvation: soft logistic bonus with fine-grained decay and saturation offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Composite score: smaller = higher priority
    # Priority hierarchy: DDL urgency > successor release > critical path > energy > starvation relief
    score = (
        + robust_normalize(np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"])  # Lateness penalty
        - successor_release_score                                                         # Unblock large workloads first under urgency
        - boosted_rank                                                                    # Critical path awareness only when needed
        - PARAMS["energy_sensitivity"] * norm_energy                                     # Energy minimization
        - wait_benefit                                                                    # Starvation mitigation
        + PARAMS["duration_robustness"] * duration_risk_score                           # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty              # Joint risk-energy penalty
        + PARAMS["remaining_work_weight"] * norm_work                                   # Downstream work preference
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
