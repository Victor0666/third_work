def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining best elements from both parents:
      - Retains Parent 2's sharper sigmoid slack-pressure gate and energy-uncertainty interaction
      - Restores Parent 1's explicit rank-slack coupling (non-redundant with boosted_rank)
      - Uses robust_normalize with MAD+median, resilient to outliers and zero-scales
      - Introduces novel 'dual-pressure' gating: only activate uncertainty-duration interaction when BOTH slack is tight AND uncertainty is high
      - Removes brittle lateness multiplier; replaces with unified slack-pressure-driven terms for cleaner optimization surface
      - All numeric literals are in {-2,-1,0,1,2}; uses np.finfo for safe bounds
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median-centered MAD + epsilon → resilient to outliers and zero-scales ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        center = np.median(x_abs) if N > 1 else x_abs[0]
        scale = np.median(np.abs(x_abs - center)) if N > 1 else np.abs(x_abs[0] - center) + eps
        return (x_abs - center) / (scale + eps)
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack pressure: bounded [0,2] clipped power penalty for overdue tasks ===
    slack_pressure_raw = np.clip(-norm_slack, 0.0, 2.0)
    slack_pressure = np.power(slack_pressure_raw + eps, PARAMS["slack_penalty_exponent"])
    
    # === Sharpened slack-pressure gate: sigmoid with tunable steepness for precise critical-path activation ===
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_raw - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Dual-pressure uncertainty-duration interaction: activated only under joint pressure (tight slack AND high uncertainty) ===
    dual_pressure_mask = (slack_pressure_raw > PARAMS["dual_pressure_slack_threshold"]) & (norm_uncert > PARAMS["uncertainty_gate_threshold"])
    duration_risk_score = norm_duration * dual_pressure_mask.astype(float) * slack_pressure
    
    # === Starvation relief: logistic-shaped anti-starvation bonus with saturation offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty interaction: multiplicative penalty for high-risk-high-energy tasks ===
    energy_uncert_penalty = norm_energy * norm_uncert * dual_pressure_mask.astype(float)
    
    # === Rank-slack coupling: strengthens critical path reward exactly when slack is constrained (complementary to boosted_rank) ===
    coupled_rank_reward = norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure)
    
    # === Composite score: smaller = higher priority ===
    # Note: all penalties scaled by slack_pressure or dual-pressure mask; no brittle multipliers
    score = (
        + slack_pressure                           # Primary urgency driver
        - boosted_rank                           # Critical-path acceleration under pressure
        - coupled_rank_reward                      # Additional slack-coupled criticality boost
        - PARAMS["energy_sensitivity"] * norm_energy  # Base energy preference
        - norm_duration                            # Preference for short duration
        - wait_benefit                             # Anti-starvation relief
        + PARAMS["duration_robustness"] * duration_risk_score  # Penalty for risky+slow under dual pressure
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Joint risk-energy penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output with strict shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    return score
