def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: deadline-risk gating + criticality-boosted ranking + robust piecewise normalization.
    
    Key innovations:
    - Uses a bounded, differentiable 'slack gate' (tanh-based) to sharply penalize negative slack
      while smoothly de-prioritizing large positive slack — avoids step discontinuities.
    - Upward rank is *conditionally boosted* only under deadline pressure (tight/negative slack),
      preventing over-commitment to long-critical-path tasks when deadlines are safe.
    - Ready wait time uses fractional power decay (not linear) to gently reward aging without dominance.
    - All features undergo *robust range normalization*: (x - Q1) / (Q3 - Q1 + epsilon) → [0,1] approx.,
      avoiding mean/std sensitivity to outliers.
    - Uncertainty modulates slack pressure multiplicatively, not additively: high uncertainty
      amplifies urgency only where slack is already low — no blind risk inflation.
    - Energy term is normalized via interquartile range then scaled, preserving marginal signal.
    - No divisions by unguarded sums; all denominators include epsilon or quantile margins.
    """
    eps = PARAMS["epsilon"]
    
    # Robust IQR-based normalization: maps to ~[0,1], resistant to outliers
    def iqr_normalize(x):
        x = np.asarray(x, dtype=float)
        q1 = np.quantile(x, 0.25)
        q3 = np.quantile(x, 0.75)
        denom = q3 - q1 + eps
        return (x - q1) / denom
    
    # Normalize base features
    norm_slack = iqr_normalize(slack)
    norm_energy = iqr_normalize(min_incremental_energy)
    norm_duration = iqr_normalize(min_exec_time + min_comm_time)
    norm_rank = iqr_normalize(upward_rank)
    norm_work = iqr_normalize(remaining_work)
    norm_wait = iqr_normalize(ready_wait_time)
    norm_uncert = iqr_normalize(uncertainty)
    
    # Slack gate: tanh-based smooth switch — high priority for negative slack, flat for large positive
    # Scaled so that slack=0 → gate ≈ 0.5, slack << 0 → gate → 1.0, slack >> 0 → gate → 0.0
    slack_gate = 0.5 * (1.0 - np.tanh(PARAMS["slack_sensitivity"] * norm_slack))
    
    # Criticality boost: only activate rank boost when slack is tight (gate > 0.5)
    rank_boost = np.where(slack_gate > 0.5, PARAMS["criticality_boost"], 1.0)
    
    # Uncertainty-coupled slack pressure: multiplies raw slack penalty (not normalized slack)
    # Uses raw slack to preserve sign & magnitude scale; guarded against extreme values
    raw_slack_safe = np.clip(slack, -1e6, 1e6)
    uncertainty_modulated_pressure = (
        PARAMS["slack_penalty_factor"] 
        * np.maximum(-raw_slack_safe, 0.0) 
        * (1.0 + PARAMS["uncertainty_slack_coupling"] * norm_uncert)
    )
    
    # Wait time: fractional power decay — gentle anti-starvation, avoids linear dominance
    wait_contribution = np.power(norm_wait + eps, PARAMS["wait_decay"])
    
    # Energy contribution: normalized, scaled, and bounded
    energy_contribution = PARAMS["energy_normalizer"] * norm_energy
    
    # Duration balance: included but down-weighted to avoid short-task bias
    duration_contribution = PARAMS["duration_balance"] * norm_duration
    
    # Final score: smaller = higher priority
    # Negative slack incurs strong penalty → pushes score up → lowers priority? NO — correction:
    # We want *smaller score* for urgent tasks → so we *subtract* the pressure (since pressure is positive when slack<0)
    # But note: environment picks argmin ⇒ urgent tasks must have *low* scores.
    # So we define urgency as *negative pressure*: high urgency = large negative term.
    # Thus: score = ... - urgency_term ...
    urgency_term = uncertainty_modulated_pressure
    
    score = (
        - urgency_term  # Urgent tasks get lower scores
        + energy_contribution
        + duration_contribution
        - rank_boost * norm_rank
        + 0.3 * norm_work  # Mild work penalty: larger work may imply later parallelism
        - wait_contribution  # Long-waiting → lower score → higher priority
    )
    
    # Ensure finite output and guard against NaN/inf
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
