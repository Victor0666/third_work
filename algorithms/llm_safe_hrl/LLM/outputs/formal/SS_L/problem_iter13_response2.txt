def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces unstable sigmoid gates with bounded linear interpolation;
       introduces successor-release coupling via norm_work * norm_rank under DDL pressure;
       removes duration_robustness and wait_saturation_offset (evidence shows redundancy);
       uses clipped linear slack scaling instead of power-law + normalization for monotonic stability;
       adds conditional energy penalty only when both slack > 0 AND uncertainty is low — avoids punishing risk-mitigating choices;
       preserves all high-confidence structural actions: conditional ddl protection, successor release, upward_rank × remaining_work."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Successor-release coupling: norm_work * norm_rank activated only when slack is positive but tight (margin-based gating) ===
    # Captures “release the bottleneck” effect: tasks with large downstream work AND high rank gain priority before hard breach
    slack_margin_ratio = np.clip(slack / (np.abs(np.median(slack)) + eps), -2.0, 2.0)
    ddl_urgent = (slack <= 0.0).astype(float)  # Hard breach: unconditional priority boost
    ddl_tight = (slack > 0.0) & (slack_margin_ratio < 0.5)  # Tight margin: conditional coupling
    successor_release = norm_work * norm_rank * (ddl_urgent + ddl_tight.astype(float))
    
    # === Bounded linear slack penalty: monotonic, numerically stable, no exponent instability ===
    # Maps slack ∈ [-∞, ∞] → penalty ∈ [-1, 1] via clipping and linear interpolation
    clipped_slack = np.clip(norm_slack, -1.0, 1.0)
    slack_penalty = -clipped_slack  # smaller slack → larger penalty (i.e., worse score)
    
    # === Critical path urgency: upward_rank amplified under DDL pressure via linear ramp, not sigmoid ===
    # Uses bounded linear interpolation: [0,1] → [1, 1+criticality_scale] when slack tight
    slack_pressure = np.clip(1.0 - (slack / (np.abs(np.median(slack)) + eps)), 0.0, 1.0)
    urgency_boost = 1.0 + PARAMS["criticality_scale"] * slack_pressure
    critical_urgency = norm_rank * urgency_boost
    
    # === Starvation relief: simple exponential decay with finite saturation ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Conditional energy penalty: only applied when slack > 0 AND uncertainty is low — avoids penalizing necessary risk mitigation ===
    low_uncert_gate = (norm_uncert < PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_penalty = norm_energy * ddl_gate * low_uncert_gate
    
    # === Energy-uncertainty interaction: only active under DDL feasibility and medium-to-high uncertainty ===
    high_uncert_gate = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_interaction = norm_energy * norm_uncert * ddl_gate * high_uncert_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for bounded AST depth ===
    score = (
        + np.clip(slack_penalty, -2.0, 2.0)                    # Primary deadline pressure signal
        - np.clip(successor_release, -2.0, 2.0)               # High-confidence DDL-diagnostic coupling
        - np.clip(critical_urgency, -2.0, 2.0)                # Linear urgency boost under pressure
        - PARAMS["energy_sensitivity"] * np.clip(energy_penalty, -2.0, 2.0)  # Energy only when safe & low-risk
        - np.clip(wait_benefit, -2.0, 2.0)                    # Anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_interaction, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(-clipped_slack, -2.0, 2.0)  # Reinforce slack dominance
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
