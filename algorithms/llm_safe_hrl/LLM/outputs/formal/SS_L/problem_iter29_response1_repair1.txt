def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule unifying best practices:
       - Preserves Parent 2's dominant finite hard_ddl_violation_penalty and adaptive hinge.
       - Integrates Parent 1's validated successor_release interaction (min_exec_time × upward_rank × ddl_breach).
       - Adds host_load_surrogate from Parent 1, now gated by both ddl_pressure and tunable uncertainty_gate_threshold.
       - Uses median-MAD normalization only on core features (slack, energy, rank, work) to avoid over-normalization noise.
       - Replaces raw_slack_penalty with robust MAD-normalized slack + clipped power law for smoother gradient behavior.
       - All intermediate terms clipped to [-2,2] for bounded AST depth and stability; final output safeguarded with machine-precision nan_to_num.
       - No infinite values, no in-place mutation, fully deterministic and shape-compliant."""
    
    eps = PARAMS["epsilon"]
    
    # Safe cast to float; no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — applied only to decision-critical features ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Hard deadline violation flag: triggers dominant finite penalty ===
    ddl_violated = (slack < PARAMS["slack_sensitivity_center"]).astype(float)
    hard_ddl_penalty = ddl_violated * PARAMS["hard_ddl_violation_penalty"]
    
    # === Slack feasibility gate: enables energy preference only when safe ===
    slack_feasibility = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Robust slack penalty: normalized clipped power law on negative slack ===
    neg_slack = np.maximum(-slack, 0.0)
    raw_slack_penalty = neg_slack ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Adaptive clipped linear hinge for slack-driven urgency coupling ===
    slack_hinge = np.clip(-norm_slack, 0.0, PARAMS["rank_slack_hinge_width"])
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * (slack_hinge / (PARAMS["rank_slack_hinge_width"] + eps))
    
    # === Sharpened urgency gate centered at sensitivity boundary ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * urgency_gate)
    
    # === Successor-release interaction: fast-executing blockers of critical descendants ===
    ddl_breach = (slack <= 0.0).astype(float)
    successor_release = min_exec_time * norm_rank * ddl_breach
    successor_release_score = PARAMS["successor_release_weight"] * np.clip(successor_release, -2.0, 2.0)
    
    # === Host-load surrogate: duration-uncertainty congestion proxy, gated by ddl_pressure and tunable uncertainty threshold ===
    load_gate = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float) * slack_feasibility
    host_load_surrogate = norm_duration * norm_uncert * load_gate
    host_load_score = PARAMS["host_load_surrogate_weight"] * np.clip(host_load_surrogate, -2.0, 2.0)
    
    # === Bounded linear starvation relief ===
    wait_benefit = np.clip(1.0 - PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability ===
    score = (
        + hard_ddl_penalty                                                 # Strict DDL-first enforcement
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency-amplified critical path
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * slack_feasibility, -2.0, 2.0)  # Energy minimization only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief
        + np.clip(PARAMS["slack_penalty_exponent"] * norm_work * ddl_violated, -2.0, 2.0)  # Downstream work under breach
        - successor_release_score                                          # Fast-release boost for critical successors
        + host_load_score                                                  # Congestion-aware load penalty
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
