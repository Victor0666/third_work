def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: integrates Parent 2's DDL protection and successor-release logic with Parent 1's bounded rank-slack coupling and explicit starvation relief.
       Novel additions: (1) additive bias in rank-slack coupling replaced by unified sigmoid coupling, (2) linear energy modulation by slack pressure improves gradient continuity,
       (3) unified robust normalization with sign-preserving MAD scaling, (4) clipped composite terms to [-2,2] ensure ordinal stability under perturbation."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median + MAD per feature (preserves sign semantics and outlier resilience) ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Conditional DDL protection gate: binary urgency boost for overdue or near-deadline tasks ===
    slack_margin_ratio = np.abs(slack) / (min_exec_time + min_comm_time + eps)
    ddl_urgent = np.where((slack <= 0.0) | (slack_margin_ratio < PARAMS["slack_margin_ratio_threshold"]), 1.0, 0.0)
    
    # === Successor-release interaction: amplify rank when task unblocks high-work successors ===
    successor_release_impact = norm_work * norm_rank
    
    # === Energy modulation under slack pressure: linear interpolation of sensitivity based on normalized slack pressure ===
    # Ensures smooth transition from full energy consideration (tight slack) to reduced emphasis (safe slack)
    slack_pressure_norm = np.clip(-norm_slack, 0.0, 2.0)
    energy_sensitivity_modulated = PARAMS["energy_sensitivity"] * (
        1.0 + PARAMS["energy_slack_modulation_slope"] * slack_pressure_norm
    )
    
    # === Sharpened slack-pressure gate: sigmoid over normalized slack for smoother transition ===
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_norm - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate) * ddl_urgent
    
    # === Starvation relief: logistic with explicit saturation (no infinite growth) ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Duration robustness: penalize long durations only under high uncertainty AND tight slack ===
    duration_risk_score = norm_duration * norm_uncert * ddl_urgent
    
    # === Energy-uncertainty interaction: multiplicative penalty, gated by urgency ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_urgent
    
    # === Composite score: smaller = higher priority ===
    # Prioritize DDL feasibility first (via slack/urgency terms), then energy, then starvation/work
    score = (
        + np.clip(norm_slack, -2.0, 2.0)                                  # Primary deadline bias: lower slack → lower score
        + np.clip(norm_slack ** PARAMS["slack_penalty_exponent"], -2.0, 2.0) * ddl_urgent  # Nonlinear lateness penalty only when urgent
        - np.clip(boosted_rank, -2.0, 2.0)                                 # Critical path boost with sigmoid coupling
        - np.clip(successor_release_impact, -2.0, 2.0)                    # Unblock high-work successors
        - np.clip(norm_energy * energy_sensitivity_modulated * ddl_urgent, -2.0, 2.0)  # Modulated energy penalty
        - np.clip(norm_duration * (1.0 - ddl_urgent), -2.0, 2.0)          # Soft short-duration preference when not urgent
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation bonus
        + np.clip(PARAMS["duration_robustness"] * duration_risk_score, -2.0, 2.0)  # Risk-aware duration penalty
        + np.clip(PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty, -2.0, 2.0)  # Risk-aligned marginal energy penalty
        + np.clip(PARAMS["remaining_work_weight"] * norm_work, -2.0, 2.0)  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
