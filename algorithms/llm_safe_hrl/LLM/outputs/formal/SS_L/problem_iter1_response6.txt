def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing risk-aware urgency, critical-path acceleration under deadline pressure,
    and robust starvation control. Uses piecewise-normalized, bounded interactions — no unbounded ops."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: use median absolute deviation (MAD) + epsilon for outlier resilience ===
    def robust_normalize(x):
        x = np.abs(x)
        center = np.median(x)
        scale = np.median(np.abs(x - center)) + eps
        return (x - center) / scale
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack pressure: strong nonlinear penalty for negative slack ===
    # Use clipped exponent to avoid overflow; shift to ensure monotonicity
    slack_penalty_base = np.where(
        norm_slack < 0,
        -np.power(np.clip(-norm_slack, 0.0, 10.0), PARAMS["slack_penalty_exponent"]),
        0.0
    )
    
    # === Critical-path acceleration gate: boost rank priority only when slack is low ===
    # Bounded sigmoid-like coupling: 0 when slack > 0, rises smoothly as slack → -∞
    slack_pressure_ratio = np.clip(-norm_slack, 0.0, 2.0)  # [0,2] bounded pressure
    rank_boost = PARAMS["criticality_scale"] * norm_rank * (
        1.0 - np.exp(-PARAMS["rank_slack_coupling"] * slack_pressure_ratio)
    )
    
    # === Duration-uncertainty interaction: penalize high-uncertainty tasks *only* if they're also tight on slack ===
    # Gate: activate only when uncertainty > threshold AND slack is negative
    unc_gate_active = (norm_uncert > PARAMS["uncertainty_gate_threshold"]) & (norm_slack < 0)
    duration_uncert_penalty = np.where(
        unc_gate_active,
        PARAMS["duration_robustness"] * norm_duration * np.abs(norm_slack),
        0.0
    )
    
    # === Starvation relief: saturating benefit from waiting time (logistic shape) ===
    # Prevents runaway priority from long waits while remaining smooth and bounded
    wait_benefit = PARAMS["wait_decay"] * (1.0 - np.exp(-norm_wait / (PARAMS["wait_decay"] + eps)))
    
    # === Composite score: smaller = higher priority ===
    # Core objective: minimize risk-adjusted energy *subject to DDL feasibility*
    # So: strongly penalize deadline risk (slack_penalty_base), moderately reward criticality (rank_boost),
    # mildly penalize energy & duration-uncertainty, and gently reward wait (anti-starvation).
    score = (
        slack_penalty_base                  # dominant urgency term: pushes negative-slack tasks first
        - rank_boost                        # promotes critical path *only when needed*
        + PARAMS["energy_sensitivity"] * norm_energy
        + duration_uncert_penalty
        - wait_benefit                      # anti-starvation: subtracts score → increases priority
    )
    
    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required even for N=1
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    
    return score
