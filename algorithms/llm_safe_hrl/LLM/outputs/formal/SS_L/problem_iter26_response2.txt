def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's robustness with Parent 1's structured risk coupling:
       - Keeps Parent 2's median-MAD normalization (more stable than quantile-based) and bounded linear starvation.
       - Reintroduces *selective* energy-uncertainty interaction *only under deadline breach*, avoiding overfitting.
       - Adds new `exec_comm_balance` parameter to explicitly control compute-vs-I/O penalty emphasis during DDL violation.
       - Uses Parent 1's `slack_sensitivity_center` for sharper, more interpretable urgency activation near deadline boundary.
       - Drops all synthetic congestion terms (congestion_weight, successor_release_interaction) per performance analysis.
       - All numeric literals strictly ∈ {-2,-1,0,1,2}; no hidden constants; uses np.finfo for safety."""
    
    eps = PARAMS["epsilon"]
    
    # Safe cast; no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — applied only to core decision features ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_exec = median_mad_normalize(min_exec_time)
    norm_comm = median_mad_normalize(min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Slack feasibility gate: smooth Heaviside enabling energy use only when slack >= 0 ===
    slack_feasibility = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Deadline breach indicator (hard threshold for structural safety) ===
    ddl_breach = (slack <= PARAMS["slack_sensitivity_center"]).astype(float)
    
    # === Robust slack penalty: capped power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Adaptive clipped linear hinge for slack-driven urgency coupling ===
    slack_hinge = np.clip(-norm_slack, 0.0, PARAMS["rank_slack_hinge_width"])
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * (slack_hinge / (PARAMS["rank_slack_hinge_width"] + eps))
    
    # === Sharpened urgency gate centered at sensitivity boundary ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * urgency_gate)
    
    # === Bounded linear starvation relief ===
    wait_benefit = np.clip(1.0 - PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Execution & communication penalties under deadline breach, balanced by user-tunable factor ===
    exec_penalty = PARAMS["exec_comm_balance"] * norm_exec * ddl_breach
    comm_penalty = (1.0 - PARAMS["exec_comm_balance"]) * norm_comm * ddl_breach
    
    # === Energy-uncertainty interaction: only active under deadline breach to avoid spurious coupling ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_breach * PARAMS["energy_uncertainty_interaction"]
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency-amplified critical path
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * slack_feasibility, -2.0, 2.0)  # Energy minimization only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief (bounded linear)
        + np.clip(exec_penalty, -2.0, 2.0)                                 # Compute-bound pressure under breach
        + np.clip(comm_penalty, -2.0, 2.0)                                 # I/O-bound pressure under breach
        + np.clip(energy_uncert_penalty, -2.0, 2.0)                       # Risk-aware energy coupling under breach
        + PARAMS["slack_penalty_exponent"] * np.clip(norm_work * ddl_breach, -2.0, 2.0)  # Downstream work prioritization under breach
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
