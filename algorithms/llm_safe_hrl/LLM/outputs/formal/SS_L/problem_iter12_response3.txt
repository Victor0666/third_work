def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces fragile piecewise starvation with bounded arctan relief,
       removes inactive parameters, simplifies rank-slack coupling, and enforces strict DDL-first logic.
       Critical path amplification now exclusively gated by slack urgency — no double-counting.
       All normalization uses median-MAD; all operations guarded against NaN/inf/zero."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median + MAD per feature (preserves sign semantics and outlier resilience) ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Conditional DDL protection gate: binary urgency boost for overdue or near-deadline tasks ===
    slack_margin_ratio = np.abs(slack) / (min_exec_time + min_comm_time + eps)
    ddl_urgent = np.where((slack <= 0.0) | (slack_margin_ratio < PARAMS["slack_margin_ratio_threshold"]), 1.0, 0.0)
    
    # === Successor-release interaction: amplify impact of unblocking high-work successors ===
    successor_release_impact = norm_work * norm_rank
    
    # === Energy modulation under risk: only penalize energy when both uncertainty & urgency are high ===
    energy_modulation_gate = np.where(
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) & (ddl_urgent == 1.0),
        1.0,
        0.0
    )
    
    # === Sharpened slack-pressure gate: sigmoid over normalized slack for smooth, bounded transition ===
    slack_pressure_norm = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_norm - 1.0)))
    
    # === Bounded anti-starvation relief: arctan-based monotonic benefit — no saturation, no linear artifacts ===
    # arctan(x) ∈ (-π/2, π/2); scaled & shifted to [0, 1) for clean additive benefit
    wait_benefit = (2.0 / np.pi) * np.arctan(PARAMS["wait_arctan_scale"] * norm_wait)
    
    # === Duration robustness: risk-weighted penalty only under deadline pressure ===
    duration_risk_score = norm_duration * norm_uncert * ddl_urgent
    
    # === Energy-uncertainty interaction: multiplicative penalty, strictly gated by urgency ===
    energy_uncert_penalty = norm_energy * norm_uncert * energy_modulation_gate
    
    # === Composite score: smaller = higher priority ===
    # DDL feasibility dominates: slack bias + nonlinear penalty dominate baseline
    # Critical path boosted only under urgency, avoiding double-counting with slack itself
    # Energy penalties are conditional, not unconditional — preserving feasibility first
    score = (
        + norm_slack                                  # Primary DDL signal: lower slack → lower score
        + norm_slack ** PARAMS["slack_penalty_exponent"] * ddl_urgent  # Strong nonlinear penalty only when urgent
        - norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate * ddl_urgent)  # Urgency-gated criticality
        - successor_release_impact                  # Prefer tasks unlocking large downstream work
        - PARAMS["energy_sensitivity"] * norm_energy * energy_modulation_gate  # Energy matters only under risk
        - wait_benefit                              # Smooth, bounded anti-starvation boost
        + PARAMS["duration_robustness"] * duration_risk_score  # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned marginal energy penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream scope
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
