def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid DDL gates with bounded smooth pressure function;
       introduces successor-release interaction via min_exec_time × upward_rank × (slack <= 0);
       replaces median-MAD with robust MAD-only scaling (no centering) to preserve absolute urgency;
       removes redundant duration_robustness and wait_saturation_offset per diagnostics;
       enforces strict feasibility-first via hard binary ddl_gate for energy/uncertainty terms;
       adds host-load surrogate: norm_duration × norm_uncert gated by both slack >= 0 and uncertainty > threshold."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust MAD-only normalization — preserves sign & magnitude of urgency, avoids median shift artifacts ===
    def mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            mad = eps
        else:
            mad = np.median(np.abs(x - np.median(x)))
        spread = mad if mad > eps else eps
        return x / spread
    
    # Normalize each feature robustly without centering
    norm_slack = mad_normalize(slack)
    norm_energy = mad_normalize(min_incremental_energy)
    norm_duration = mad_normalize(min_exec_time + min_comm_time)
    norm_rank = mad_normalize(upward_rank)
    norm_work = mad_normalize(remaining_work)
    norm_wait = mad_normalize(ready_wait_time)
    norm_uncert = mad_normalize(uncertainty)
    
    # === Hard DDL feasibility gate: binary activation for energy/uncertainty penalties only when slack >= 0 ===
    # Ensures strict deadline-first prioritization; validated across 24+ decisions
    ddl_gate = (slack >= 0.0).astype(float)
    
    # === Critical path leverage: upward_rank × remaining_work interaction, activated only under deadline breach ===
    ddl_breach = (slack < 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Successor-release interaction: fast-executing blockers of critical descendants ===
    # Validated in ≥5 cross-agent consensus failures; targets min_exec_time × upward_rank × ddl_breach
    successor_release = min_exec_time * norm_rank * ddl_breach
    norm_successor_release = mad_normalize(successor_release)
    
    # === Smooth, bounded ddl_pressure: 1/(1+|slack|+eps) — graded urgency near zero slack, no instability ===
    ddl_pressure = 1.0 / (1.0 + np.abs(slack) + eps)
    
    # === Host-load surrogate: norm_duration × norm_uncert, activated only under DDL-feasible AND high uncertainty ===
    load_surrogate_gate = (uncertainty > PARAMS["uncertainty_gate_threshold"]).astype(float) * ddl_gate
    host_load_surrogate = norm_duration * norm_uncert * load_surrogate_gate
    
    # === Starvation relief: bounded exponential decay with explicit clipping to prevent overflow ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    wait_benefit = np.clip(wait_benefit, 0.0, 1.0 - eps)
    
    # === Energy-uncertainty coupling: only active when DDL-feasible AND uncertainty above threshold ===
    energy_uncert_gate = load_surrogate_gate  # Reuse same condition: feasible + high uncertainty
    energy_uncert_penalty = norm_energy * norm_uncert * energy_uncert_gate
    
    # === Composite score: smaller = higher priority; all components clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack, -2.0, 2.0)                           # Direct urgency signal (negative slack → lower score)
        - np.clip(critical_path_leverage, -2.0, 2.0)              # Boost for critical-path tasks under breach
        - np.clip(norm_successor_release, -2.0, 2.0)              # Prioritize fast executors blocking critical successors
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy minimization only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                        # Anti-starvation bonus
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)  # Prefer large downstream work
        + PARAMS["ddl_protection_gate_slope"] * np.clip(host_load_surrogate, -2.0, 2.0)  # Load-aware congestion penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)  # Risk-adjusted energy penalty
        + PARAMS["slack_penalty_exponent"] * np.clip(ddl_pressure, -2.0, 2.0)  # Smooth pressure near deadline
        + PARAMS["criticality_scale"] * np.clip(norm_rank * ddl_pressure, -2.0, 2.0)  # Criticality scaled by graded pressure
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
