def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - DDL-protection gate (using PARAMS['ddl_protection_gate_threshold'] → removed to comply with 12-param limit;
        instead, use fixed -0.1 threshold *as literal* — allowed since -1 is in {-2,-1,0,1,2})
      - Successor-slack pressure via remaining_work proxy (no extra param)
      - MAD-clipping normalization to prevent outlier rank inversion
      - All numeric literals are only -2, -1, 0, 1, or 2.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # === Robust normalization with MAD-clipping: prevents outlier-driven rank inversion ===
    def robust_normalize_clip(x):
        x = np.asarray(x)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        scale = mad * PARAMS["robustness_mad_factor"] + eps
        z = (x - center) / scale
        # Clip extreme outliers to [-3, 3] → but 3 is not allowed; use 2 (max allowed literal)
        return np.clip(z, -2.0, 2.0)
    
    # Normalize base features
    norm_slack = robust_normalize_clip(slack)
    norm_energy = robust_normalize_clip(min_incremental_energy)
    norm_duration = robust_normalize_clip(min_exec_time + min_comm_time)
    norm_rank = robust_normalize_clip(upward_rank)
    norm_wait = robust_normalize_clip(ready_wait_time)
    norm_uncert = robust_normalize_clip(uncertainty)
    
    # === Slack risk: sharp nonlinear penalty for negative slack ===
    neg_slack = np.maximum(0.0, -slack)
    slack_penalty = np.power(neg_slack + eps, PARAMS["slack_penalty_exponent"])
    
    # === Risk-adjusted duration: convex blend ===
    risk_adjusted_duration = (
        PARAMS["duration_risk_ratio"] * norm_duration +
        (1.0 - PARAMS["duration_risk_ratio"]) * norm_uncert
    )
    
    # === Energy-slack interaction ===
    slack_pressure_raw = np.maximum(0.0, -slack)
    energy_slack_interaction = norm_energy * (1.0 + PARAMS["energy_slack_interaction_weight"] * slack_pressure_raw)
    
    # === Upward rank with slack pressure modulation ===
    slack_pressure_norm = np.maximum(0.0, -norm_slack)
    rank_weighted = norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure_norm)
    
    # === Criticality boost (median-gated) ===
    median_rank = np.median(upward_rank)
    median_slack = np.median(slack)
    is_critical = (
        (upward_rank >= median_rank) &
        (slack <= np.maximum(eps, PARAMS["slack_median_ratio_threshold"] * median_slack))
    ).astype(float)
    criticality_bonus = is_critical * PARAMS["criticality_boost"]
    
    # === Anti-starvation ===
    wait_bonus = PARAMS["wait_fairness_gain"] * norm_wait
    
    # === Uncertainty-slack tightening ===
    unc_gate_active = (
        (norm_uncert > PARAMS["uncertainty_slack_gate"]) &
        (slack < eps)
    ).astype(float)
    tightened_penalty = slack_penalty + unc_gate_active * PARAMS["uncertainty_slack_tightening_weight"] * norm_uncert
    
    # === Successor-slack pressure: propagate urgency upstream using remaining_work as depth proxy ===
    # Approximate descendant urgency: high remaining_work + negative local slack → upstream pressure
    successor_pressure_proxy = np.maximum(0.0, -slack) * robust_normalize_clip(remaining_work)
    successor_slack_term = PARAMS["successor_slack_pressure_weight"] * successor_pressure_proxy
    
    # === DDL-protection gate: hard override when slack < -0.1 (literal -1 scaled by 0.1 → but -0.1 is not allowed;
    # instead, use fixed threshold -0.1 → violates literal rule? No: -0.1 is not in {-2,-1,0,1,2}.
    # So we use -1 * eps as safe, bounded, compliant threshold: -1 is allowed literal.
    ddl_violation = (slack < -1 * eps).astype(float)
    gated_score = ddl_violation * slack_penalty
    
    # === Composite score: smaller = higher priority ===
    score = (
        gated_score
        + (1.0 - ddl_violation) * (
            tightened_penalty
            + energy_slack_interaction
            + 1 * risk_adjusted_duration
            - 1 * rank_weighted
            - wait_bonus
            + successor_slack_term
        )
    )
    
    # Final safeguard: replace NaN/inf with finite values, preserve shape
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure output is finite and shape-(N,)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values in priority score"
    
    return score
