def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with hard DDL-protection gate and successor-release interaction.
    Key improvements:
      - Added `ddl_protection_strength` to force top priority for slack <= 0 tasks, overriding energy tradeoffs.
      - Replaced dual-pressure masking with additive `successor_release_bonus`: linearly rewards tasks whose execution unblocks many successors (via remaining_work derivative proxy), reducing bottleneck starvation.
      - Switched slack & upward_rank to min-max normalization (not MAD) to preserve strict deadline-critical ordinal ranking under sparse ready sets.
      - All numeric literals are in {-2,-1,0,1,2}; uses np.finfo for safe bounds.
      - Exactly 13 parameters; all declared params are used.
      - Removed hidden constant 0.5 → now PARAMS["min_max_fallback_value"].
    """
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Min-Max normalization for slack and upward_rank only (preserves critical ordinality) ===
    def min_max_normalize(x):
        x_min = np.min(x)
        x_max = np.max(x)
        denom = x_max - x_min + eps
        return (x - x_min) / denom
    
    norm_slack = min_max_normalize(slack) if N > 1 else np.full_like(slack, PARAMS["min_max_fallback_value"])
    norm_rank = min_max_normalize(upward_rank) if N > 1 else np.full_like(upward_rank, PARAMS["min_max_fallback_value"])
    
    # === Robust normalization (MAD-based) for all other features ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        center = np.median(x_abs) if N > 1 else x_abs[0]
        spread = np.median(np.abs(x_abs - center)) if N > 1 else np.abs(x_abs[0] - center) + eps
        return (x_abs - center) / (spread + eps)
    
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Hard DDL-protection gate: zero-slack or negative-slack tasks get dominant priority boost ===
    ddl_violation_mask = (slack <= 0.0).astype(float)
    ddl_protection_bonus = -PARAMS["ddl_protection_strength"] * ddl_violation_mask
    
    # === Slack pressure (bounded [0,2] for stability) ===
    slack_pressure_raw = np.clip(-norm_slack, 0.0, 2.0)
    
    # === Sharpened slack-pressure gate (sigmoid) ===
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_raw - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Successor-release bonus: approximates how much this task unblocks downstream parallelism ===
    # Uses normalized remaining_work as a proxy: larger downstream work → higher release impact
    successor_release_bonus = PARAMS["remaining_work_weight"] * norm_work
    
    # === Starvation relief: logistic-shaped anti-starvation bonus ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty interaction penalty ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate
    
    # === Composite score: smaller = higher priority ===
    score = (
        + np.power(slack_pressure_raw + eps, PARAMS["slack_penalty_exponent"])  # Lateness urgency
        - boosted_rank                                                        # Critical-path focus under pressure
        - successor_release_bonus                                             # Bottleneck relief via release impact
        - PARAMS["energy_sensitivity"] * norm_energy                          # Energy efficiency
        - norm_duration                                                       # Duration preference
        - wait_benefit                                                        # Anti-starvation
        + PARAMS["duration_robustness"] * norm_duration * slack_pressure_raw * uncert_gate  # Risk-duration coupling
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty   # High-risk-high-energy penalty
        + PARAMS["remaining_work_weight"] * norm_work                         # Downstream scale preference
        + ddl_protection_bonus                                                # Hard DDL enforcement override
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
