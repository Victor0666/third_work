def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: Slack-driven criticality gating + energy-efficiency density + anti-starvation decay.
    
    Key innovations:
    - Uses piecewise slack penalty: strong linear penalty for slack <= 0, soft sigmoid ramp for slack > 0
    - Upward rank is boosted *only* when slack is tight (enforcing DDL-first feasibility)
    - Energy priority is computed as 'risk-adjusted energy per effective duration' — a proxy for efficiency
    - Ready wait time uses bounded exponential urgency: prioritizes long-waiting tasks but saturates smoothly
    - Introduces duration-ratio risk term: favors tasks where exec dominates comm (more predictable under bandwidth uncertainty)
    - All normalizations use median-based robust scaling to resist outliers
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)

    # Robust normalization: use median absolute deviation scaled by median, fallback to mean if degenerate
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        med = np.median(np.abs(x))
        if med == 0.0:
            med = np.mean(np.abs(x)) + eps
        return x / (med + eps)

    # --- 1. Slack penalty: hard penalty for negative slack, smooth ramp above zero ---
    # Use clipped linear + bounded sigmoid to avoid explosion; smaller score = higher priority
    slack_raw = np.asarray(slack, dtype=float)
    slack_penalty = np.where(
        slack_raw <= 0.0,
        -PARAMS["slack_penalty_factor"] * slack_raw,  # positive penalty → increases score → lowers priority? NO: we want *smaller* score for urgent tasks
        # So instead: map slack>0 to *reduced* priority weight → lower score means higher priority
        # Thus: for slack>0, apply diminishing urgency: 1/(1+exp(k*slack)) decays from 1→0
        -1.0 + 1.0 / (1.0 + np.exp(PARAMS["slack_penalty_factor"] * (slack_raw - 0.1)))
    )
    # But final score must be *smaller* for urgent → so we negate the urgency signal:
    # Let urgency = 1 for slack<=0, decaying to 0 for large slack → then use -urgency as base?
    # Instead: define base urgency: high when slack low → then subtract from baseline
    # Better: construct base score where low slack → low score
    urgency_base = np.where(
        slack_raw <= 0.0,
        slack_raw * 0.0,  # dummy; will override
        0.0
    )
    urgency = np.where(
        slack_raw <= 0.0,
        1.0 + (-slack_raw) * 0.5,  # linear boost: more negative → higher urgency → lower score
        1.0 / (1.0 + np.exp(PARAMS["slack_penalty_factor"] * (slack_raw)))  # decays from 1→0
    )
    # So urgency ∈ (0, ~1.5]; we want high urgency → low score ⇒ use -urgency as core term
    slack_score = -urgency

    # --- 2. Criticality boost: upward_rank matters *only* under deadline pressure ---
    # Gate rank contribution using slack urgency itself
    rank_contribution = robust_normalize(upward_rank) * (
        PARAMS["criticality_boost"] * urgency
        + (1.0 - urgency) * 0.1  # residual minimal weight
    )

    # --- 3. Energy efficiency density: marginal energy per total time (joules/sec), risk-adjusted ---
    # Avoid division by zero; cap extreme ratios via robust normalization
    total_duration = min_exec_time + min_comm_time
    # Efficiency = energy / duration → lower is better → contributes negatively to score
    eff_density = np.divide(
        min_incremental_energy,
        total_duration + eps,
        out=np.full(N, np.inf, dtype=float),
        where=(total_duration + eps) != 0.0
    )
    # Normalize efficiency density → lower value = more efficient → assign negative weight
    eff_norm = robust_normalize(eff_density)
    energy_score = -PARAMS["energy_efficiency_bias"] * eff_norm

    # --- 4. Anti-starvation: bounded exponential urgency for ready_wait_time ---
    # Saturating urgency: u(t) = 1 - exp(-k*t), bounded in [0,1)
    wait_urgency = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * ready_wait_time)
    wait_score = -wait_urgency  # smaller score = higher priority

    # --- 5. Duration-ratio risk: favors tasks where compute dominates communication ---
    # High exec/comm ratio → less sensitive to bandwidth uncertainty → more predictable
    ratio = np.divide(
        min_exec_time,
        min_comm_time + eps,
        out=np.full(N, 0.0, dtype=float),
        where=(min_comm_time + eps) != 0.0
    )
    # Normalize ratio; higher ratio → lower score (more predictable → prefer earlier)
    ratio_norm = robust_normalize(ratio)
    ratio_score = -PARAMS["duration_risk_ratio_weight"] * ratio_norm

    # --- 6. Uncertainty-slack interaction: amplify penalty when both slack is low AND uncertainty is high ---
    # Use product of normalized uncertainty and slack urgency
    unc_norm = robust_normalize(uncertainty)
    unc_slack_interaction = -PARAMS["uncertainty_slack_gate"] * unc_norm * urgency

    # --- 7. Remaining work: weak inverse bias (large work → schedule earlier to avoid bottleneck) ---
    # But not dominant: avoid starving small tasks
    work_norm = robust_normalize(remaining_work)
    work_score = -0.3 * work_norm  # fixed small weight; tunable via new param would overcomplicate

    # Assemble final score: all components designed so smaller = more urgent/efficient/preferred
    score = (
        slack_score
        + rank_contribution
        + energy_score
        + wait_score
        + ratio_score
        + unc_slack_interaction
        + work_score
    )

    # Ensure finite output & shape compliance
    score = np.nan_to_num(score, nan=0.0, posinf=1e6, neginf=-1e6)
    # Clamp extreme values to prevent numerical dominance
    score = np.clip(score, -1e4, 1e4)

    return score
