def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: hybridizes Parent 2's median-MAD normalization and hard ddl_urgent_mask
       with Parent 1's local release bonus and refined anti-starvation logic;
       replaces linear wait ramp with smooth clipped tanh for better gradient behavior near saturation;
       introduces adaptive slack-pressure gate blending using both raw slack and normalized urgency;
       retains all evidence-backed structural actions while adding one novel interaction: 
       local unblocking bonus gated by both slack and uncertainty for risk-aware successor release."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature: outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        scale = mad + eps if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / scale
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Hard DDL urgency mask: binary gate aligned with violation semantics ===
    ddl_urgent_mask = (slack <= 0.0).astype(float)
    
    # === Risk-aware slack penalty: power penalty on overdue tasks, scaled by urgency mask ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Adaptive slack-pressure gate: sigmoid over normalized slack, blended with raw urgency signal ===
    # Improves robustness across scales by combining normalized shape with absolute urgency
    slack_pressure_sig = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * norm_slack))
    raw_urgency_sig = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (-slack + eps)))
    rank_gate = ddl_urgent_mask * (
        PARAMS["slack_pressure_sig_weight"] * slack_pressure_sig + 
        PARAMS["raw_urgency_sig_weight"] * raw_urgency_sig
    ) + (1.0 - ddl_urgent_mask) * PARAMS["ddl_urgent_default_gate"]
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Duration-uncertainty interaction: activated only under dual pressure (low slack AND high uncertainty) ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = (min_exec_time + min_comm_time) * uncert_gate * ddl_urgent_mask
    
    # === Smooth anti-starvation: clipped tanh provides bounded monotonicity with better gradient near saturation ===
    # Avoids linear ramp’s abrupt clip and preserves differentiability
    wait_benefit = np.clip(np.tanh(PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"])), 0.0, 1.0)
    
    # === Energy-uncertainty interaction: additive penalty under both urgency and uncertainty gates ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_urgent_mask * uncert_gate
    
    # === Local unblocking bonus: remaining_work * upward_rank under joint slack+uncertainty gate ===
    # Novel risk-aware coupling — prevents unblocking high-downstream work only when both deadline pressure AND uncertainty are high
    joint_gate = ddl_urgent_mask * uncert_gate
    release_bonus = PARAMS["remaining_work_weight"] * norm_work * norm_rank * joint_gate
    
    # === Composite score: smaller = higher priority ===
    # Prioritization order: (1) lateness penalty, (2) critical path boost, (3) energy preference,
    #                       (4) starvation relief, (5) duration risk penalty, (6) risk-energy penalty,
    #                       (7) risk-aware unblocking bonus
    score = (
        + norm_slack_penalty                           # Strong lateness penalty
        - boosted_rank                                 # Critical-path boost under urgency
        - PARAMS["energy_sensitivity"] * norm_energy  # Prefer low marginal energy
        - wait_benefit                                 # Anti-starvation bonus (smooth tanh ramp)
        + PARAMS["duration_robustness"] * median_mad_normalize(duration_risk_score)  # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Targeted risk-energy penalty
        - release_bonus                                # Risk-aware unblocking bonus
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
