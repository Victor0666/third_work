def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: adopts quantile-based (IQR) normalization for slack and remaining_work
       using tunable quantiles; introduces dynamic DDL-protection gate that activates
       critical-path boosting only when slack falls below adaptive safety margin (median_slack * factor);
       retains all prior robust interactions (lateness multiplier, work-slack coupling, soft uncertainty);
       ensures monotonic priority ordering under deadline stress while maintaining deterministic structure."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Quantile-based normalization for slack and remaining_work (IQR + median) ===
    # Uses tunable quantiles to replace hidden constants 0.25 and 0.75
    def quantile_normalize(x):
        if N == 0:
            return np.zeros(0, dtype=float)
        q_low = np.quantile(x, PARAMS["quantile_alpha"])
        q_high = np.quantile(x, PARAMS["quantile_beta"])
        iqr = q_high - q_low + eps
        center = np.median(x)
        return (x - center) / (iqr + eps)
    
    # === Robust MAD-based normalization for other features ===
    def robust_normalize_mad(x):
        if N == 0:
            return np.zeros(0, dtype=float)
        center = np.median(x)
        scale = np.median(np.abs(x - center)) + eps
        return (x - center) / (scale + eps)
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = quantile_normalize(slack)
    norm_energy = robust_normalize_mad(min_incremental_energy)
    norm_duration = robust_normalize_mad(min_exec_time + min_comm_time)
    norm_rank = robust_normalize_mad(upward_rank)
    norm_work = quantile_normalize(remaining_work)
    norm_wait = robust_normalize_mad(ready_wait_time)
    norm_uncert = robust_normalize_mad(uncertainty)
    
    # === Dynamic DDL-protection gate: activates criticality boost only when slack < median_slack * factor ===
    median_slack = np.median(slack)
    dynamic_margin = median_slack * PARAMS["ddl_protection_factor"]
    ddl_protection_gate = (slack < dynamic_margin).astype(float)
    
    # === Risk-aware slack pressure: monotonic clipped power ===
    slack_pressure_raw = np.clip(-norm_slack, 0.0, None)
    slack_pressure = np.power(slack_pressure_raw + eps, PARAMS["slack_penalty_exponent"])
    
    # === Lateness penalty gate: multiplicative enforcement only on penalties when critically overdue ===
    is_late = (slack < -eps).astype(float)
    lateness_multiplier = 1.0 + PARAMS["lateness_penalty_strength"] * is_late
    
    # === Critical-path acceleration gate: driven by dynamic DDL-protection ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_protection_gate)
    
    # === Soft uncertainty activation: graded sigmoidal response ===
    uncertainty_activation = 1.0 / (1.0 + np.exp(-PARAMS["uncertainty_smoothness"] * (norm_uncert - 0.0)))
    duration_risk_interaction = norm_duration * uncertainty_activation * PARAMS["duration_robustness"]
    
    # === Starvation relief: saturating exponential form ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + eps))
    
    # === Energy-slack interaction ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["energy_slack_interaction"] * slack_pressure)
    
    # === Rank-slack coupling ===
    coupled_rank_reward = norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure)
    
    # === Work-slack coupling ===
    work_slack_penalty = norm_work * (1.0 + PARAMS["work_slack_coupling"] * slack_pressure)
    
    # === Final score: smaller = higher priority ===
    score = (
        + lateness_multiplier * slack_pressure                # Primary urgency driver
        + lateness_multiplier * energy_slack_penalty          # Deadline-amplified energy cost
        + lateness_multiplier * PARAMS["energy_sensitivity"] * norm_energy  # Base energy penalty
        + lateness_multiplier * work_slack_penalty             # Compute-load acceleration under urgency
        - coupled_rank_reward                                # Critical path reward
        - wait_benefit                                       # Anti-starvation relief
        + lateness_multiplier * duration_risk_interaction    # Graded risk-duration penalty
        - boosted_rank * ddl_protection_gate                 # Dynamic DDL-protection: critical-path acceleration only when needed
    )
    
    # Final safeguard: ensure finite, deterministic output with strict shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    return score
