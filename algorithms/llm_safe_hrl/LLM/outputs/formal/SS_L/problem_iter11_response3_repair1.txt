def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: merges Parent 2's evidence-backed DDL protection and successor-release logic
       with Parent 1's bounded rank-slack coupling and explicit starvation-linearization.
       Introduces novel piecewise starvation relief: logistic below threshold, linear above — prevents over-saturation
       while ensuring monotonic anti-starvation boost. All normalization uses median-MAD for robustness.
       Critical path amplification is now explicitly decoupled from urgency gates to avoid double-counting."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median + MAD per feature (preserves sign semantics and outlier resilience) ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Conditional DDL protection gate: binary urgency boost for overdue or near-deadline tasks ===
    slack_margin_ratio = np.abs(slack) / (min_exec_time + min_comm_time + eps)
    ddl_urgent = np.where((slack <= 0.0) | (slack_margin_ratio < PARAMS["slack_margin_ratio_threshold"]), 1.0, 0.0)
    
    # === Successor-release interaction: amplify rank when task unblocks high-work successors ===
    successor_release_impact = norm_work * norm_rank
    
    # === Energy modulation under risk: penalize energy only when uncertainty is high AND slack is tight ===
    energy_modulation_gate = np.where(
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) & (ddl_urgent == 1.0),
        1.0,
        0.0
    )
    
    # === Bounded linear rank-slack coupling: replaces unstable tanh and sigmoid coupling with interpretable,
    # sign-aware, and numerically stable interaction — ensures monotonic criticality increase as slack degrades
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)  # bounded [-1,1]
    rank_slack_coupling = 1.0 + 0.85 * (-coupled_slack)  # fixed strength to remove extra param; preserves structure intent
    boosted_rank = norm_rank * rank_slack_coupling
    
    # === Sharpened slack-pressure gate: sigmoid over normalized slack-pressure for smooth transition ===
    slack_pressure_norm = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_norm - 1.0)))
    
    # Apply urgency-gated rank amplification *only once*, avoiding cascaded multiplication with ddl_urgent
    final_rank_score = boosted_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Piecewise starvation relief: logistic below threshold, linear above — avoids asymptotic saturation
    # while guaranteeing monotonic, bounded, and differentiable anti-starvation boost
    starvation_threshold = 0.2  # fixed threshold to remove extra param; keeps logic intact
    wait_linear_region = np.where(norm_wait > starvation_threshold,
                                  norm_wait - starvation_threshold,
                                  0.0)
    wait_logistic_region = 1.0 - np.exp(-PARAMS["wait_decay"] * (np.clip(norm_wait, 0.0, starvation_threshold) + PARAMS["wait_saturation_offset"]))
    wait_benefit = wait_logistic_region + wait_linear_region
    
    # === Duration robustness: penalize long durations only under high uncertainty AND tight slack ===
    duration_risk_score = norm_duration * norm_uncert * ddl_urgent
    
    # === Energy-uncertainty interaction: multiplicative penalty, gated by urgency ===
    energy_uncert_penalty = norm_energy * norm_uncert * energy_modulation_gate
    
    # === Composite score: smaller = higher priority ===
    # Prioritize DDL feasibility first (via slack/urgency), then critical path, then energy under risk,
    # then starvation, then risk-aware duration, then successor release
    score = (
        + norm_slack                                  # Primary deadline bias: lower slack → lower score
        + norm_slack ** PARAMS["slack_penalty_exponent"] * ddl_urgent  # Nonlinear lateness penalty only when urgent
        - final_rank_score                          # Adaptive critical path boost (bounded, sign-aware)
        - successor_release_impact                  # Prefer tasks unblocking large downstream work
        - PARAMS["energy_sensitivity"] * norm_energy * energy_modulation_gate  # Energy only matters under joint risk
        - wait_benefit                              # Anti-starvation: piecewise (logistic+linear) for monotonicity & saturation control
        + PARAMS["duration_robustness"] * duration_risk_score  # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Joint risk penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
