def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: removes aggregated duration penalty to restore discriminative power;
       introduces *exec_comm_separation_factor* to independently gate and scale execution vs communication penalties;
       preserves all successful normalization, gating, and slack-coupling from Parent 2;
       ensures compute-bound tasks get urgent scheduling under tight slack while I/O-bound tasks are deprioritized only if bandwidth-limited;
       avoids structural overloading by decoupling exec/comm signals instead of summing them."""
    
    # Immutable machine precision safeguard
    eps = np.finfo(float).tiny
    
    # Cast inputs safely without in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust sign-preserving normalization: mean ± std with eps fallback ===
    def robust_normalize(x):
        x = np.copy(x)
        if N == 1:
            center = x[0]
            spread = eps
        else:
            center = np.mean(x)
            spread = np.std(x, ddof=0)
        spread = np.where(spread > eps, spread, eps)
        return (x - center) / spread
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_exec = robust_normalize(min_exec_time)
    norm_comm = robust_normalize(min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside — suppresses risk terms when slack < 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Slack pressure signal: sigmoid over clipped negative slack → sharp urgency transition ===
    raw_slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    slack_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (raw_slack_pressure - 1.0)))
    
    # === Critical-path urgency: only activated under DDL breach ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_score = norm_rank * norm_work * ddl_breach
    
    # === Upward-rank amplification: linear coupling with slack pressure, bounded ===
    rank_amplifier = 1.0 + PARAMS["criticality_scale"] * slack_pressure_gate
    rank_amplifier = np.clip(rank_amplifier, PARAMS["rank_amplifier_lower_bound"], PARAMS["rank_amplifier_upper_bound"])
    boosted_rank = norm_rank * rank_amplifier
    
    # === Uncertainty-weighted communication penalty: uses norm_comm & norm_uncert separately ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    comm_uncert_penalty = norm_comm * norm_uncert * uncert_gate * ddl_breach
    
    # === Anti-starvation: additive linear bonus — preserves monotonicity and avoids saturation ===
    wait_bonus = PARAMS["wait_decay"] * norm_wait
    
    # === Energy-uncertainty interaction: double-gated by ddl_gate and uncertainty threshold ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: linear boost proportional to slack pressure ===
    energy_slack_penalty = norm_energy * raw_slack_pressure * ddl_gate
    
    # === Separated execution & communication penalties: individually gated and scaled ===
    # Enables precise control: e.g., prioritize fast-compute tasks on edge under tight slack, defer large-comm tasks to cloud
    exec_penalty = PARAMS["exec_comm_separation_factor"] * norm_exec * slack_pressure_gate * ddl_gate
    comm_penalty = (1.0 - PARAMS["exec_comm_separation_factor"]) * norm_comm * slack_pressure_gate * ddl_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability and scale balance ===
    score = (
        + np.clip(norm_slack, -2.0, 2.0)                                    # Base urgency: lower slack → lower score
        + np.clip(raw_slack_pressure * PARAMS["slack_penalty_exponent"], -2.0, 2.0)  # Nonlinear lateness risk penalty
        - np.clip(critical_path_score, -2.0, 2.0)                          # Unblock critical paths first when breaching
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Amplified rank under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy minimization only when feasible
        + np.clip(wait_bonus, -2.0, 2.0)                                   # Starvation relief: larger wait → higher priority
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + np.clip(comm_uncert_penalty, -2.0, 2.0)                          # Explicit comms+uncertainty penalty
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)  # Prefer large downstream work when safe
        + np.clip(exec_penalty, -2.0, 2.0)                                 # Compute-bound urgency under deadline pressure
        + np.clip(comm_penalty, -2.0, 2.0)                                 # I/O-bound deferral unless bandwidth-resilient
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
