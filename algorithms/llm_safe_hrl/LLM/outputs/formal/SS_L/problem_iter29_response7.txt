def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining robustness from Parent 1 and strict DDL enforcement from Parent 2:
       - Uses median-MAD normalization (Parent 2) for stability, but adds *urgency_sharpening_factor* (from Parent 1)
         applied to MAD-normalized slack for stronger primary urgency signal.
       - Keeps hard_ddl_violation_penalty + slack_sensitivity_center for strict DDL-first behavior.
       - Introduces *remaining_work_weight_under_pressure*: activates downstream work preference *only* under slack<=0,
         replacing generic work weighting — avoids diluting urgency with irrelevant large-work tasks.
       - Removes all quantile-based gating and uncertainty interaction terms (overfitting per analysis);
         instead uses unified slack-conditioned logic with sharpened urgency and binary pressure activation.
       - All intermediate terms clipped to [-2,2] for bounded AST depth and gradient-friendly tuning."""
    
    eps = PARAMS["epsilon"]
    
    # Safe cast; no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — applied to core features only ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = np.maximum(mad, eps)
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    
    # === Sign-preserving MAD-normalized slack with tunable sharpening ===
    slack_med = np.median(slack)
    slack_mad = np.median(np.abs(slack - slack_med))
    slack_spread = np.maximum(slack_mad, eps)
    norm_slack_mad = (slack - slack_med) / slack_spread
    sharpened_urgency = PARAMS["urgency_sharpening_factor"] * norm_slack_mad
    
    # === Hard deadline violation flag: finite dominant penalty ===
    ddl_violated = (slack < PARAMS["slack_sensitivity_center"]).astype(float)
    hard_ddl_penalty = ddl_violated * PARAMS["hard_ddl_violation_penalty"]
    
    # === Slack feasibility gate: enables energy preference only when safe ===
    slack_feasibility = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Robust slack penalty: capped power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Adaptive clipped linear hinge for slack-driven urgency coupling ===
    slack_hinge = np.clip(-norm_slack, 0.0, PARAMS["rank_slack_hinge_width"])
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * (slack_hinge / (PARAMS["rank_slack_hinge_width"] + eps))
    
    # === Sharpened urgency gate centered at sensitivity boundary ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * urgency_gate)
    
    # === Bounded linear starvation relief ===
    wait_benefit = np.clip(1.0 - PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Remaining work weight activated *only* under deadline pressure (binary gate) ===
    ddl_pressure = (slack <= 0.0).astype(float)
    work_under_pressure = PARAMS["remaining_work_weight_under_pressure"] * norm_work * ddl_pressure
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + hard_ddl_penalty                                                 # Strict DDL-first enforcement
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty
        + np.clip(sharpened_urgency, -2.0, 2.0)                            # Primary sharpened urgency driver
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency-amplified critical path
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * slack_feasibility, -2.0, 2.0)  # Energy minimization only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief
        - np.clip(work_under_pressure, -2.0, 2.0)                          # Downstream work prioritization *only* under breach
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
