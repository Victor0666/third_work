def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's numerical stability with Parent 1's robust successor-release modeling;
       introduces exponentiated successor-release interaction (validated by DDL diagnostics) for stronger critical-path unblocking;
       replaces linear DDL gate with centered linear ramp for sharper, tunable feasibility enforcement;
       retains clipped linear slack penalty and median-MAD normalization for monotonicity and outlier resilience."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Centered linear slack-pressure gate: ramps from 0→1 over [center-halfwidth, center+halfwidth] ===
    gate_lower = PARAMS["slack_pressure_gate_center"] - PARAMS["slack_pressure_gate_halfwidth"]
    gate_upper = PARAMS["slack_pressure_gate_center"] + PARAMS["slack_pressure_gate_halfwidth"]
    ddl_pressure = np.clip((norm_slack - gate_lower) / (gate_upper - gate_lower + eps), 0.0, 1.0)
    ddl_pressure = np.where(norm_slack <= gate_lower, 0.0, ddl_pressure)
    ddl_pressure = np.where(norm_slack >= gate_upper, 1.0, ddl_pressure)
    
    # === Successor-release interaction: exponentiated product under pressure for stronger unblocking bias ===
    successor_release = (norm_rank * norm_work) ** PARAMS["successor_release_exponent"] * ddl_pressure
    
    # === Linear slack penalty: bounded linear penalty for negative slack, numerically stable and monotonic ===
    raw_slack_penalty = np.maximum(-slack, 0.0)
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Critical-path coupling: upward_rank scaled by pressure ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_pressure)
    
    # === Uncertainty-aware energy penalty: gated by both DDL feasibility AND uncertainty threshold ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_pressure
    
    # === Starvation relief: linear ramp-up with explicit saturation at 1.0 ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Slack-aware execution time penalty: only when pressure is active ===
    exec_penalty = norm_duration * ddl_pressure
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for bounded AST depth and stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Exponentiated successor-release boost
        - np.clip(coupled_rank, -2.0, 2.0)                                 # Critical-path urgency
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_pressure, -2.0, 2.0)  # Energy only under pressure
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + np.clip(exec_penalty, -2.0, 2.0)                                 # Duration penalty under pressure
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(ddl_pressure * PARAMS["ddl_protection_gate_slope"], -2.0, 2.0)  # Direct pressure signal
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
