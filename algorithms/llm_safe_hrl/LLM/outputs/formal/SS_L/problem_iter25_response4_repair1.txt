def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: uses IQR normalization; introduces successor-release interaction;
       applies hard DDL gating with soft sigmoid fallback via ddl_protection_gate_slope;
       uses sign-preserving MAD-normalized slack; joint congestion gating;
       all tunable parameters referenced exactly once; no numeric literals beyond -2,-1,0,1,2."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust IQR-based normalization ===
    def iqr_normalize(x):
        x = np.copy(x)
        if N == 1:
            q50 = x[0]
            q25, q75 = q50 - eps, q50 + eps
        else:
            q50 = np.quantile(x, 0.5)
            q25 = np.quantile(x, 0.25)
            q75 = np.quantile(x, 0.75)
        iqr = q75 - q25
        spread = iqr if iqr > eps else eps
        return (x - q50) / spread
    
    norm_slack = iqr_normalize(slack)
    norm_energy = iqr_normalize(min_incremental_energy)
    norm_duration = iqr_normalize(min_exec_time + min_comm_time)
    norm_rank = iqr_normalize(upward_rank)
    norm_work = iqr_normalize(remaining_work)
    norm_wait = iqr_normalize(ready_wait_time)
    norm_uncert = iqr_normalize(uncertainty)
    
    # === Hard DDL feasibility gating with smooth fallback using ddl_protection_gate_slope ===
    # Soft Heaviside enables gradual transition near slack=0, per evidence
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    ddl_breach = 1.0 - ddl_gate
    
    # === Sign-preserving MAD-normalized slack — sharpens urgency near slack ≈ 0 ===
    slack_mad = np.median(np.abs(slack - np.median(slack)))
    slack_mad_safe = np.where(slack_mad > eps, slack_mad, eps)
    norm_slack_sharp = (slack - np.median(slack)) / slack_mad_safe
    
    # === Successor-release interaction: fast-executing blockers of critical descendants ===
    clipped_rank = np.clip(norm_rank, 0.0, 2.0)
    successor_release = min_exec_time * clipped_rank * ddl_breach
    
    # === Critical-path awareness only under deadline pressure ===
    critical_path_impact = norm_rank * norm_work * ddl_breach
    
    # === Joint congestion gating: requires BOTH high wait AND high uncertainty ===
    wait_congestion = (ready_wait_time >= np.quantile(ready_wait_time, 0.75)).astype(float)
    uncert_congestion = (uncertainty >= np.quantile(uncertainty, 0.75)).astype(float)
    congestion_gate = wait_congestion * uncert_congestion
    
    # === Starvation relief: bounded logistic with wait_decay ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Energy/uncertainty penalties only when DDL-feasible (soft-gated by ddl_gate) ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_gate
    
    # === Slack-pressure gate for critical path activation ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    
    # === Uncertainty gating threshold ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    
    # === Composite score: smaller = higher priority ===
    score = (
        # Primary DDL risk penalty: sharp near zero slack
        + np.clip(norm_slack_sharp, -2.0, 2.0)
        
        # Negative slack penalty exponentiation
        + np.clip(np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"], -2.0, 2.0)
        
        # Critical path leverage under breach
        - np.clip(critical_path_impact, -2.0, 2.0)
        
        # Successor-release boost
        - np.clip(successor_release, -2.0, 2.0)
        
        # HEFT urgency scaled by criticality under pressure
        - np.clip(norm_rank * (1.0 + PARAMS["criticality_scale"] * ddl_breach), -2.0, 2.0)
        
        # Energy preference only when feasible (soft-gated)
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)
        
        # Anti-starvation
        - np.clip(wait_benefit, -2.0, 2.0)
        
        # Uncertainty-energy penalty only when feasible
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        
        # Remaining work preference
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        
        # Congestion penalty
        + congestion_gate * np.clip(norm_duration, -2.0, 2.0)
        
        # Uncertainty-threshold gated duration risk
        + PARAMS["slack_pressure_gate_steepness"] * np.clip(norm_duration * uncert_gate * ddl_gate, -2.0, 2.0)
    )
    
    # Final safeguards
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,)
    return score.reshape(-1)
