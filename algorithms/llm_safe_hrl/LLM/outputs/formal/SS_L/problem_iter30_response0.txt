def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
       - Bounded urgency ramp: replaces unbounded linear `sharpened_urgency` with a symmetric, finite clip-based ramp
         centered at `urgency_bounded_ramp_center`, width-controlled by `urgency_bounded_ramp_width`,
         ensuring robust monotonic urgency without explosion near zero slack.
       - Unified slack-conditioning: all DDL-sensitive terms now use identical `ddl_gate` (sigmoid) for consistency —
         fixes prior misalignment between piecewise ramp and sigmoid gating that caused feasibility violations.
       - Strictly bounded intermediates: all feature interactions clipped to [-2,2]; final score finite and shape-(N,).
       - No unbounded operators: removed `1.95 * norm_slack_mad` and all non-clipped linear amplifications."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Unified DDL-protection gate: smooth Heaviside enabling energy/uncertainty ONLY when slack >= 0 ===
    # Used consistently across ALL DDL-sensitive terms — fixes prior gate misalignment
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Deadline breach flag: binary for slack <= 0 (activates urgency & starvation relief) ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Bounded urgency ramp: symmetric finite ramp centered at urgency_bounded_ramp_center, width-controlled ===
    # Replaces unstable linear `1.95 * norm_slack_mad`; ensures monotonic, bounded, and interpretable urgency
    urgency_ramp = np.clip(
        (norm_slack - PARAMS["urgency_bounded_ramp_center"]) / PARAMS["urgency_bounded_ramp_width"],
        -1.0, 1.0
    )
    
    # === Piecewise-linear slack-pressure gate: retained but now used only for rank coupling (not gating) ===
    slack_ramp_center = PARAMS["piecewise_slack_offset"]
    slack_pressure_ramp = np.clip(
        (norm_slack + slack_ramp_center) * PARAMS["slack_pressure_gate_steepness"],
        0.0, 1.0
    )
    
    # === Successor-release interaction: gated by uncertainty AND unified ddl_gate (not ddl_breach) for safety ===
    successor_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    successor_release = (
        norm_rank * norm_work * successor_gate * ddl_gate * PARAMS["successor_release_coupling"]
    )
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded linear rank-slack coupling: stable monotonic amplification under pressure ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    
    # === Boosted rank: HEFT urgency amplified by slack pressure and coupling, under unified ddl_gate ===
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + slack_pressure_ramp * PARAMS["criticality_scale"]) * ddl_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus *only* under deadline pressure ===
    wait_benefit_pressure = (1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)) * ddl_breach
    
    # === Energy-uncertainty coupling: now uses unified ddl_gate (not successor_gate) for correctness ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_gate
    
    # === Slack-aware energy scaling: applies only when ddl_gate is active ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure_ramp) * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(urgency_ramp, -2.0, 2.0)                                  # Bounded urgency ramp (replaces unbounded linear)
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(successor_release, -2.0, 2.0)                            # Critical-path-aware successor unblocking
        - np.clip(boosted_rank, -2.0, 2.0)                                  # HEFT-based urgency under pressure & feasibility
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit_pressure, -2.0, 2.0)                        # Pressure-specific anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
