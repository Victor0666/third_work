def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: hard-deadline protection gate, monotonic work/slack coupling, and urgency-preserving piecewise slack penalty.
    
    Key improvements:
    - Replaces tanh with binary DDL-protection gate: when median_slack <= 0, amplify both rank and slack penalties → enforces feasibility-first hierarchy.
    - Replaces clipped successor_release with monotonic `remaining_work / (|slack| + eps)` — preserves critical-path signal even for small positive slack.
    - Uses piecewise `np.where(slack < 0, ..., 0)` for slack penalty instead of smoothed tanh → retains sharp urgency discrimination for lateness risk.
    - All numeric literals are -2,-1,0,1,2; all parameters declared and used exactly once.
    - Robust normalization and finite safeguards via np.finfo.
    """
    eps = PARAMS["epsilon"]
    N = len(slack)
    
    # Sanitize inputs — no in-place modification
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    # Robust normalization: x / (mean(|x|) + eps)
    def robust_norm(x):
        x_abs = np.abs(x)
        denom = np.mean(x_abs) + eps
        return x / denom
    
    # 1. Hard deadline pressure gate: activate only when median slack is exhausted
    median_slack = np.median(slack)
    ddl_protection_active = (median_slack <= eps).astype(float)
    ddl_boost = 1.0 + PARAMS["ddl_protection_boost"] * ddl_protection_active
    
    # 2. Piecewise slack penalty: sharp activation only for negative slack (preserves urgency)
    neg_slack = np.where(slack < 0, -slack, 0.0)
    slack_penalty = np.power(neg_slack + eps, PARAMS["slack_penalty_exponent"])
    
    # 3. Risk-adjusted duration: convex blend
    duration = min_exec_time + min_comm_time
    duration_norm = robust_norm(duration)
    uncertainty_norm = robust_norm(uncertainty)
    risk_adjusted_duration = (
        PARAMS["duration_risk_ratio"] * duration_norm +
        (1 - PARAMS["duration_risk_ratio"]) * uncertainty_norm
    )
    
    # 4. Uncertainty-coupled energy penalty
    energy_norm = robust_norm(min_incremental_energy)
    energy_uncertainty_factor = 1 + PARAMS["energy_uncertainty_coupling"] * uncertainty_norm
    coupled_energy_penalty = PARAMS["energy_efficiency_scale"] * energy_norm * energy_uncertainty_factor
    
    # 5. Upward rank modulated by slack pressure — amplified under DDL protection
    rank_base = robust_norm(upward_rank)
    rank_weighted = rank_base * (1 + PARAMS["rank_slack_coupling"] * np.where(slack < 0, 1.0, 0.0))
    rank_weighted = rank_weighted * ddl_boost
    
    # 6. Criticality boost: only when rank ≥ median AND slack ≤ threshold × |median_slack|
    median_rank = np.median(upward_rank)
    abs_median_slack = np.abs(median_slack) + eps
    is_critical = (
        (upward_rank >= median_rank) &
        (slack <= PARAMS["slack_median_ratio_threshold"] * abs_median_slack)
    ).astype(float)
    criticality_bonus = is_critical * PARAMS["criticality_boost"]
    
    # 7. Monotonic successor-release: remaining_work / (|slack| + eps) — avoids clipping, preserves signal for small positive slack
    work_slack_ratio = remaining_work / (np.abs(slack) + eps)
    work_slack_norm = robust_norm(work_slack_ratio)
    successor_release = PARAMS["work_slack_ratio_weight"] * work_slack_norm
    
    # 8. Anti-starvation fairness bonus (reduces score for long-waiting tasks)
    wait_norm = robust_norm(ready_wait_time)
    wait_bonus = -PARAMS["wait_fairness_gain"] * wait_norm
    
    # 9. Uncertainty-slack gate: tighten penalty when uncertainty is high and slack is low
    unc_gate_active = (
        (uncertainty_norm > PARAMS["uncertainty_slack_gate"]) &
        (slack < eps)
    ).astype(float)
    tightened_penalty = slack_penalty + unc_gate_active * PARAMS["uncertainty_slack_tightening_weight"] * uncertainty_norm
    tightened_penalty = tightened_penalty * ddl_boost  # Amplify under DDL pressure
    
    # Final score: smaller = higher priority
    score = (
        tightened_penalty
        + coupled_energy_penalty
        + 1 * risk_adjusted_duration
        - 1 * rank_weighted
        + 0 * criticality_bonus
        - 1 * successor_release
        + wait_bonus
    )
    
    # Safeguard: ensure finite output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape {(N,)}, got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values in priority score"
    
    return score
