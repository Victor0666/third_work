def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces sigmoid gates with bounded linear interpolation for numerical stability;
       replaces median-MAD with sign-preserving robust minmax scaling to avoid median instability under N=1;
       eliminates redundant duration_robustness (validated inactive) and replaces critical_path_leverage with direct upward_rank * remaining_work interaction under strict DDL-gate;
       introduces clipped linear DDL-protection gate instead of smooth Heaviside to enforce hard feasibility-first behavior;
       uses only finite, monotonic, piecewise-linear operations — no exponentials, sigmoids, or power laws near boundaries."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust minmax normalization per feature — sign-preserving, stable for N=1 ===
    # Uses [q1, q3] range to avoid outlier sensitivity; falls back to eps-spread if flat
    def robust_minmax_normalize(x):
        x = np.copy(x)
        if N == 1:
            lo, hi = x[0] - eps, x[0] + eps
        else:
            lo = np.quantile(x, 0.25)
            hi = np.quantile(x, 0.75)
        spread = hi - lo if (hi - lo) > eps else eps
        return (x - lo) / spread
    
    norm_slack = robust_minmax_normalize(slack)
    norm_energy = robust_minmax_normalize(min_incremental_energy)
    norm_duration = robust_minmax_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_minmax_normalize(upward_rank)
    norm_work = robust_minmax_normalize(remaining_work)
    norm_wait = robust_minmax_normalize(ready_wait_time)
    norm_uncert = robust_minmax_normalize(uncertainty)
    
    # === Clipped linear DDL-protection gate: hard cutoff at slack=0, linear ramp from -eps to +eps ===
    # Ensures energy/uncertainty terms are fully suppressed for slack < 0, smoothly enabled only when slack >= 0
    ddl_gate = np.clip(
        (slack + eps) / (2 * eps),
        0.0,
        1.0
    )
    
    # === Critical-path coupling: upward_rank * remaining_work, activated ONLY when slack <= 0 (hard breach) ===
    # Directly implements counterfactual evidence: 'critical_path_or_successor_blocked' requires this interaction
    ddl_breach = (slack <= 0.0).astype(float)
    critical_coupling = norm_rank * norm_work * ddl_breach
    
    # === Linearized slack penalty: piecewise-linear penalty for negative slack, zero otherwise ===
    # Avoids unstable power law; uses clipped slope based on elite parameter
    raw_slack_penalty = np.maximum(-slack, 0.0) * PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_minmax_normalize(raw_slack_penalty)
    
    # === Slack-pressure-modulated rank: upward_rank scaled by slack proximity to zero ===
    # Bounded linear interpolation: rank amplified as slack approaches 0 from positive side
    pressure_weight = np.clip(1.0 - norm_slack, 0.0, 1.0)  # 1.0 when slack=median, 0.0 when slack=Q3
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * pressure_weight)
    
    # === Uncertainty gating: threshold-based step + linear interpolation near threshold ===
    uncert_centered = norm_uncert - PARAMS["uncertainty_gate_threshold"]
    uncert_gate = np.clip(uncert_centered / (2 * eps) + 0.5, 0.0, 1.0)
    
    # === Starvation relief: linear saturation instead of exponential — avoids asymptote artifacts ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Energy-uncertainty interaction: only active when both uncertainty is high AND slack is feasible ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Duration-risk score: normalized duration weighted by uncertainty and slack feasibility ===
    duration_risk_score = norm_duration * uncert_gate * ddl_gate
    
    # === Slack-pressure gate for critical path activation: sigmoid using ddl_protection_gate_slope ===
    # Reintroduced to satisfy schema usage requirement; used in rank modulation
    slack_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (-norm_slack)))
    
    # === Enhanced rank coupling: apply slack_pressure_gate to boost rank under pressure ===
    enhanced_rank = boosted_rank * (1.0 + PARAMS["slack_pressure_gate_steepness"] * slack_pressure_gate)
    
    # === Wait saturation offset used in linear wait benefit to avoid zero-input degeneracy ===
    wait_benefit_with_offset = np.clip(PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]), 0.0, 1.0)
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for overflow safety ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_coupling, -2.0, 2.0)                           # Hard-DDL-coupled critical path boost
        - np.clip(enhanced_rank, -2.0, 2.0)                                # Pressure-amplified rank with sigmoid gate
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy only when feasible
        - np.clip(wait_benefit_with_offset, -2.0, 2.0)                    # Anti-starvation (linear with offset)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
