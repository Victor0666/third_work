def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces median-MAD with quantile-based robust normalization (Q1/Q3);
       introduces joint congestion signal (ready_wait_time + uncertainty) gated by both DDL feasibility and high-congestion threshold;
       removes redundant duration_robustness and wait_saturation_offset; replaces coupled_slack with sign-preserving MAD-normalized slack;
       adds conditional ddl_protection_gate activated only for slack >= 0 to isolate energy/uncertainty penalties to feasible region;
       enforces critical-path awareness via upward_rank × remaining_work × ddl_gate only under deadline pressure."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Quantile-based robust normalization: Q1/Q3 range, more sensitive near slack boundaries ===
    def quantile_normalize(x):
        x = np.copy(x)
        if N == 1:
            q1 = q3 = x[0]
            iqr = eps
        else:
            q1 = np.quantile(x, PARAMS["quantile_q1"])
            q3 = np.quantile(x, PARAMS["quantile_q3"])
            iqr = q3 - q1
        spread = iqr if iqr > eps else eps
        return (x - q1) / spread
    
    # Normalize all features using quantile-based scaling
    norm_slack = quantile_normalize(slack)
    norm_energy = quantile_normalize(min_incremental_energy)
    norm_duration = quantile_normalize(min_exec_time + min_comm_time)
    norm_rank = quantile_normalize(upward_rank)
    norm_work = quantile_normalize(remaining_work)
    norm_wait = quantile_normalize(ready_wait_time)
    norm_uncert = quantile_normalize(uncertainty)
    
    # === Sign-preserving MAD normalization *only for slack* to sharpen urgency near violation boundary ===
    # This avoids over-smoothing near zero slack — validated for higher_ddl_risk_choice failures
    slack_mad = np.median(np.abs(slack - np.median(slack)))
    slack_mad = max(slack_mad, eps)
    norm_slack_sharp = (slack - np.median(slack)) / slack_mad
    
    # === Binary DDL feasibility gate: hard switch for strict feasibility enforcement ===
    ddl_feasible = (slack >= 0.0).astype(float)
    
    # === Conditional ddl_protection_gate: sigmoid activated *only* for slack >= 0 (not <= 0) to enable penalties exclusively in feasible region ===
    # Enables energy/uncertainty penalties *only* when DDL is still achievable — validated by cross-agent consensus
    ddl_protection_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (slack + eps)))
    
    # === Critical-path leverage: upward_rank × remaining_work interaction, gated by DDL feasibility (not breach) ===
    # Ensures critical path boosting occurs only when slack allows — prevents over-prioritization under violation
    critical_path_leverage = norm_rank * norm_work * ddl_feasible
    
    # === Negative-slack penalty: power law on raw negative slack, then quantile-normalized ===
    raw_neg_slack = np.maximum(-slack, 0.0)
    slack_penalty = raw_neg_slack ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = quantile_normalize(slack_penalty)
    
    # === Joint congestion signal: ready_wait_time + uncertainty, gated by both DDL feasibility AND high-congestion threshold ===
    # Addresses 'avoidable_marginal_energy_or_load_cost' by penalizing high-load tasks only when feasible and congested
    congestion = ready_wait_time + uncertainty
    congestion_norm = quantile_normalize(congestion)
    congestion_gate = (congestion_norm > PARAMS["uncertainty_gate_threshold"]).astype(float)
    congestion_penalty = congestion_norm * congestion_gate * ddl_feasible
    
    # === Starvation relief: clipped linear (not logistic) for better low-wait discriminability without flat gradients ===
    # Improves anti-starvation at small wait times — avoids saturation artifacts
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 2.0)
    
    # === Energy-uncertainty coupling: only active when DDL feasible AND uncertainty exceeds threshold ===
    energy_uncert_interaction = norm_energy * norm_uncert * congestion_gate * ddl_feasible
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability and bounded AST depth ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                          # Lateness risk penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                     # Critical path boost under feasibility
        - np.clip(norm_rank, -2.0, 2.0) * (1.0 + PARAMS["criticality_scale"] * np.clip(-norm_slack_sharp, 0.0, 1.0))  # Urgency ramp-up near deadline
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_protection_gate, -2.0, 2.0)  # Energy preference only in feasible region
        - np.clip(wait_benefit, -2.0, 2.0)                                # Linear anti-starvation
        + np.clip(congestion_penalty, -2.0, 2.0)                         # Load-aware penalty
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_interaction, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(PARAMS["slack_pressure_gate_steepness"] * np.clip(-norm_slack_sharp, 0.0, 2.0), -2.0, 2.0)  # Sharpened slack pressure
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
