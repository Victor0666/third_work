def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: combines Parent 2's robust DDL protection and load gating with Parent 1's successor-release interaction;
       replaces hard clipping with bounded sigmoid coupling for slack-pressure modulation;
       introduces duration-aware starvation relief decoupled from congestion;
       adds risk-gated critical-path leverage requiring both DDL feasibility AND actionable uncertainty;
       retains median-MAD normalization, all validated gates, and strict numeric literal constraints."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Host-load gate: sigmoid on normalized duration as proxy for VM congestion ===
    host_load_gate = 1.0 / (1.0 + np.exp(-PARAMS["host_load_gate_steepness"] * norm_duration))
    
    # === Slack-pressure coupling: bounded sigmoid instead of clipping → smooth monotonic modulation ===
    # Maps norm_slack ∈ ℝ → [-1, 1] smoothly: 2/(1+exp(-2*x)) - 1
    slack_pressure_coupling = 2.0 / (1.0 + np.exp(-2.0 * norm_slack)) - 1.0
    
    # === Sharpened slack-pressure gate for pre-breach urgency ramp-up ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    
    # === Critical path leverage: gated by DDL feasibility AND actionable uncertainty ===
    ddl_breach = (slack <= 0.0).astype(float)
    risk_actionable = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_gate * risk_actionable
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Successor-release interaction (from Parent 1): fast-executing blockers of critical descendants ===
    successor_release = min_exec_time * norm_rank * ddl_breach
    
    # === Duration-aware starvation relief: anti-starvation only when VMs are NOT congested ===
    wait_benefit = (1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)) * (1.0 - host_load_gate)
    
    # === Uncertainty-duration interaction: gated by static uncertainty threshold + DDL feasibility ===
    uncert_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float) * ddl_gate
    duration_risk_score = norm_duration * uncert_gate * slack_pressure
    
    # === Energy-uncertainty coupling: gated by static uncertainty threshold and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    # Host-load gate now modulates energy_sensitivity to prevent low-energy bias on congested VMs
    # Also incorporates successor-release boost (Parent 1) and bounded sigmoid coupling (analysis insight)
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Risk-gated DDL-diagnostic leverage
        - np.clip(successor_release, -2.0, 2.0)                          # Fast-blocker release boost (Parent 1)
        - PARAMS["energy_sensitivity"] * host_load_gate * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Load-gated energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Duration-decoupled anti-starvation
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + PARAMS["criticality_scale"] * np.clip(norm_rank * slack_pressure_coupling, -2.0, 2.0)  # Smooth sigmoid-coupled rank
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
