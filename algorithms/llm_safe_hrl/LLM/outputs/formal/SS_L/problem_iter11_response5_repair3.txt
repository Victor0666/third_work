def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: integrates binary DDL breach response with calibrated urgency boost;
       replaces unbounded waiting bonuses with capped logistic relief;
       refines successor-release coupling using slack-gated residual work impact;
       uses N-aware median-MAD normalization with explicit scalar fallback for N=1;
       adds explicit starvation cap to preserve DDL feasibility hierarchy."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust per-feature normalization: median-MAD with safe scalar fallback for N=1 ===
    def robust_normalize(x):
        if N == 1:
            center = x[0]
            spread = np.abs(x[0]) + eps
        else:
            center = np.median(x)
            spread = np.median(np.abs(x - center))
        return (x - center) / (spread + eps)
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Hard deadline breach mask: binary activation for immediate DDL enforcement ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Slack-pressure gate: smooth sigmoid ramp for pre-breach urgency (eta=0.95 aligned) ===
    slack_gap = np.maximum(-slack, 0.0)
    slack_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_gap - eps)))
    
    # === Residual work impact: successor-release coupling gated by both breach and pressure ===
    # Stronger coupling under actual breach, softer under near-breach — improves stability
    residual_work_impact = norm_rank * norm_work * (ddl_breach + (1.0 - ddl_breach) * slack_pressure_gate)
    
    # === Lateness penalty: power-law on raw negative slack, boosted under hard breach ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    breach_amplified_penalty = raw_slack_penalty * (1.0 + PARAMS["starvation_relief_cap"] * ddl_breach)
    norm_slack_penalty = robust_normalize(breach_amplified_penalty)
    
    # === Boosted critical rank: only activated under slack pressure, with calibrated scaling ===
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_pressure_gate)
    
    # === Energy-uncertainty joint penalty: active only above threshold ===
    uncert_gate = (norm_uncert > PARAMS["uncertainty_gate_threshold"]).astype(float)
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate
    
    # === Duration-robustness penalty: requires both uncertainty and slack pressure ===
    duration_risk_score = norm_duration * norm_uncert * slack_pressure_gate
    
    # === Bounded starvation relief: logistic form with hard cap to prevent DDL violation ===
    wait_benefit = np.clip(
        1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"])),
        0.0,
        PARAMS["starvation_relief_cap"]
    )
    
    # === Composite score: smaller = higher priority
    # Priority order enforced: (1) DDL breach response, (2) residual work impact, (3) slack penalty,
    #                          (4) critical rank, (5) energy efficiency, (6) bounded starvation relief,
    #                          (7) risk penalties (duration & energy-uncertainty), (8) downstream work preference
    score = (
        + norm_slack_penalty                           # Primary DDL-constrained lateness penalty
        - residual_work_impact                         # Structural DAG pressure: unblock large workloads
        - boosted_rank                                 # Critical path awareness under pressure
        - PARAMS["energy_sensitivity"] * norm_energy  # Energy minimization within feasibility
        - wait_benefit                                 # Starvation mitigation with hard cap
        + PARAMS["duration_robustness"] * duration_risk_score  # Uncertain-slow tasks penalized under pressure
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Joint risk penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
