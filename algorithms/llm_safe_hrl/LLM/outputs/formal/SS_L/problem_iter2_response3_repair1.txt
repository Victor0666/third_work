def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all declared parameters are used; no unused entries.
    Introduces work-pressure term to prevent late-stage bottlenecks under deadline pressure.
    Uses unified slack_pressure signal, smooth rank_gate (with literal 2.0), tanh starvation relief, and robust normalization.
    All numeric literals are -2,-1,0,1,2; epsilon via PARAMS; no side effects or I/O."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median-center + MAD scale + epsilon ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        center = np.median(x_abs)
        scale = np.median(np.abs(x_abs - center)) + eps
        return (x_abs - center) / scale
    
    # Normalize base features (scale-invariant, finite, deterministic)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Unified slack pressure: clipped power law for numerical stability and monotonic urgency ===
    slack_pressure = np.power(np.clip(-norm_slack, 0.0, None), PARAMS["slack_penalty_exponent"])
    
    # === Critical-path acceleration gate: smooth transition from full boost to zero over [0, slack_gate_width] ===
    # Gate = 1.0 when norm_slack <= 0, linearly decays to 0 at norm_slack == slack_gate_width, clamped to [0,1]
    rank_gate = np.clip(1.0 - 2.0 * np.clip(norm_slack, 0.0, PARAMS["slack_gate_width"]), 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Work-pressure penalty: only active under deadline pressure — prevents high-work tasks from blocking critical path ===
    work_pressure = np.where(slack_pressure > 0, 
                           PARAMS["work_pressure_weight"] * norm_work * slack_pressure, 
                           0.0)
    
    # === Uncertainty-duration interaction: activate only under both high uncertainty AND deadline pressure ===
    uncert_gate = (
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) &
        (slack_pressure > 0)
    ).astype(float)
    duration_risk_interaction = norm_duration * uncert_gate * PARAMS["duration_robustness"]
    
    # === Starvation relief: bounded, smooth, and differentiable via tanh ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * norm_wait)
    
    # === Final score: smaller = higher priority ===
    score = (
        + slack_pressure                                  # Urgency: primary driver for deadline compliance
        + PARAMS["energy_sensitivity"] * norm_energy     # Energy cost: consistent penalty
        - boosted_rank                                   # Critical path acceleration: promotes vital tasks when needed
        + work_pressure                                  # Bottleneck prevention: penalizes large workloads under pressure
        + duration_risk_interaction                      # Risk-duration penalty: only when uncertainty and slack align
        - wait_benefit                                   # Starvation relief: rewards long-waiting tasks
    )
    
    # Final safeguard: ensure finite, deterministic, shape-(N,) output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required even for N=1
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    
    return score
