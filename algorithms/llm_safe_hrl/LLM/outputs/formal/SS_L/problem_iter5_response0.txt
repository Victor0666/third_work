def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's robust min-max normalization and raw slack fidelity
       with Parent 1's conditional energy-relaxation under DDL stress and explicit energy-uncertainty penalty.
       Introduces novel dual-gate structure: (1) slack-pressure-triggered energy relaxation,
       and (2) joint energy-uncertainty penalty only active under high risk AND negative slack.
       Uses bounded min-max scaling for all features except slack (kept raw for deadline signal integrity).
       All operations protected against NaN/inf; deterministic and shape-correct."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Bounded min-max normalization: preserves ordering and extreme value sensitivity ===
    # Critical: slack remains UNNORMALIZED to preserve physical deadline violation magnitude
    def bounded_minmax(x):
        x_min = np.min(x)
        x_max = np.max(x)
        range_val = x_max - x_min + eps
        return (x - x_min) / range_val
    
    norm_energy = bounded_minmax(min_incremental_energy)
    norm_duration = bounded_minmax(min_exec_time + min_comm_time)
    norm_rank = bounded_minmax(upward_rank)
    norm_wait = bounded_minmax(ready_wait_time)
    norm_uncert = bounded_minmax(uncertainty)
    
    # === Raw slack-based pressure: exact lateness magnitude matters for hard DDL constraints ===
    slack_pressure_raw = np.clip(-slack, 0.0, None)  # Only penalize lateness, preserve units
    slack_pressure = np.power(slack_pressure_raw + eps, PARAMS["slack_penalty_exponent"])
    
    # === Critical-path acceleration gate: smooth threshold on *normalized* slack for stability ===
    norm_slack_for_gate = bounded_minmax(slack)
    rank_gate = np.clip(1.0 - 2.0 * np.clip(norm_slack_for_gate, 0.0, PARAMS["slack_gate_width"]), 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Joint uncertainty-duration-slack gating: activated only when slack_pressure > 0 AND uncertainty high ===
    joint_risk_gate = (
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) &
        (slack_pressure > 0)
    ).astype(float)
    duration_risk_interaction = norm_duration * joint_risk_gate * PARAMS["duration_robustness"]
    
    # === Starvation relief: tanh-saturated benefit from wait time (preserves gradient near zero) ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * norm_wait)
    
    # === Rank-slack coupling: amplify criticality under pressure, but clip to prevent explosion ===
    coupled_rank = np.clip(norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure), 0.0, 2.0)
    
    # === Conditional energy relaxation: reduce energy sensitivity only when slack <= 0 (DDL stress) ===
    energy_weight = np.where(
        slack <= 0.0,
        PARAMS["energy_sensitivity"] * PARAMS["ddl_stress_energy_relaxation"],
        PARAMS["energy_sensitivity"]
    )
    
    # === Energy-uncertainty interaction: only penalize high-energy+high-uncertainty when DDL is violated ===
    # Matches hard constraint semantics: risk-adjusted energy matters most when deadline is at risk
    energy_uncert_active = (slack < 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"])
    energy_uncert_penalty = np.where(
        energy_uncert_active,
        norm_energy * norm_uncert,
        np.zeros_like(norm_energy)
    )
    
    # === Final score: smaller = higher priority ===
    # Dominant term: slack_pressure ensures deadline adherence first
    # Then: energy cost (relaxed under stress), critical path reward (coupled), starvation relief
    # Finally: risk penalties (duration + energy-uncertainty) only when both conditions hold
    score = (
        + slack_pressure                                  # Primary urgency driver (raw lateness penalty)
        + energy_weight * norm_energy                    # Energy cost, relaxed under DDL stress
        - coupled_rank                                   # Critical path reward (pressure-amplified)
        - wait_benefit                                   # Starvation relief (saturating)
        + duration_risk_interaction                      # Conditional duration risk penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Joint energy-uncertainty penalty
    )
    
    # Ensure finite output and correct shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(-1)
