def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: removes unused parameters; uses robust range normalization;
       applies unified congestion signal (ready_wait_time + uncertainty) gated by ddl_gate;
       replaces power-law slack penalty with clipped linear pressure for stability;
       enforces critical-path leverage only under slack <= 0 via ddl_breach;
       suppresses energy/uncertainty terms exclusively under ddl_gate (slack >= 0);
       retains sign-preserving structure, finite output, and strict (N,) shape."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust range-based normalization: (x - min) / (max - min + eps), sign-preserving ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            return np.zeros_like(x)
        x_min = np.min(x)
        x_max = np.max(x)
        denom = x_max - x_min + eps
        return (x - x_min) / denom
    
    norm_slack = robust_range_normalize(slack)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_duration = robust_range_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # === DDL-protection gate: active only when slack >= 0; enables energy/uncertainty use ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: upward_rank * remaining_work interaction, activated only under deadline breach (slack <= 0) ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Linear slack pressure: clipped [0, 1] to stabilize urgency coupling near violation boundary ===
    slack_pressure = np.clip(-norm_slack, 0.0, 1.0)
    
    # === Unified congestion signal: ready_wait_time + uncertainty, gated by ddl_gate ===
    congestion = (ready_wait_time + uncertainty) * ddl_gate
    
    # === Starvation relief: exponential decay without saturation offset (evidence shows it's inactive) ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Energy-uncertainty coupling: fully suppressed unless ddl_gate is active ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_gate
    
    # === Composite score: smaller = higher priority; all components bounded and stabilized ===
    score = (
        + PARAMS["slack_penalty_exponent"] * slack_pressure                      # Dominant lateness pressure
        - PARAMS["criticality_scale"] * critical_path_leverage                  # Critical-path unblocking only under breach
        - PARAMS["remaining_work_weight"] * norm_work * ddl_gate                # Downstream work preference only when feasible
        - PARAMS["energy_sensitivity"] * norm_energy * ddl_gate                 # Energy minimization only in feasible region
        - wait_benefit                                                            # Anti-starvation bonus
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty      # Risk-aware energy penalty only when safe
        + congestion                                                              # Congestion relief gated by DDL feasibility
        + norm_rank * (1.0 - ddl_gate)                                          # Mild rank boost when slack > 0
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)

    return score.reshape(-1)
