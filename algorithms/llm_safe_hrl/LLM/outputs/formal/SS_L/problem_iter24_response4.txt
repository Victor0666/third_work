def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces softplus starvation with bounded linear saturation for strict monotonicity and interpretability;
       removes uncertainty gating entirely per reflection — retains only slack feasibility as robust DDL-protection signal;
       introduces rank_slack_hinge_width to adaptively widen/narrow the urgency ramp based on problem hardness;
       eliminates all synthetic risk proxies (congestion, duration_risk) to reduce overfitting and improve generalization;
       uses simplified median-MAD normalization without norm_wait or norm_uncert — only essential features normalized;
       clips all terms to [-2,2] and enforces finite output via nan_to_num with machine-precision bounds."""
    
    eps = PARAMS["epsilon"]
    
    # Safe cast; no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — applied only to core decision features ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    
    # === Slack feasibility gate: only active when slack >= 0 (smooth Heaviside) ===
    # Simplified per self-reflection — uncertainty gating removed for stability & generalization
    slack_feasibility = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === DDL breach indicator for critical path leverage ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Robust slack penalty: capped power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Adaptive clipped linear hinge for slack-driven urgency ===
    # Uses new rank_slack_hinge_width to control ramp width: wider → gentler urgency transition
    slack_hinge = np.clip(-norm_slack, 0.0, PARAMS["rank_slack_hinge_width"])
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * (slack_hinge / (PARAMS["rank_slack_hinge_width"] + eps))
    
    # === Sharpened slack-pressure gate for rank boosting ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Bounded linear starvation relief: monotonic, interpretable, overflow-proof ===
    # Replaces softplus + offset — now strictly linear decay from 1.0 to 0.0
    wait_benefit = np.clip(1.0 - PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Unified risk-weighted energy terms under slack_feasibility only ===
    energy_feasible = norm_energy * slack_feasibility
    energy_slack_penalty = norm_energy * slack_pressure * slack_feasibility
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Critical-path unblocking boost
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency amplification
        - PARAMS["energy_sensitivity"] * np.clip(energy_feasible, -2.0, 2.0)  # Energy preference only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief (bounded linear)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
