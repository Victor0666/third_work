def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: combines V2's joint gating & smoother slack gate with V1's robust MAD normalization,
       adds work-pressure term and energy-slack coupling, removes redundant parameters for tighter AST.
       Uses only -2,-1,0,1,2 as literals; all tunables via PARAMS."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust normalization: median-center + MAD scale (V1-style outlier resilience) ===
    def robust_normalize(x):
        x_abs = np.abs(x)
        center = np.median(x_abs) if N > 1 else x_abs[0]
        # MAD = median(|x_i - center|); add eps to avoid zero scale
        deviations = np.abs(x_abs - center)
        scale = np.median(deviations) if N > 1 else deviations[0]
        return x_abs / (scale + eps)
    
    # Normalize base features — using absolute values ensures non-negativity and stability
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack pressure: nonlinear penalty for negative slack (V2-style, but on raw slack for fidelity) ===
    # Use raw slack (not normalized) for penalty exponentiation to preserve relative urgency magnitude
    slack_pressure_raw = np.clip(-slack, 0.0, None) + eps
    slack_pressure = np.power(slack_pressure_raw, PARAMS["slack_penalty_exponent"])
    # Re-normalize pressure to maintain scale consistency
    slack_pressure = robust_normalize(slack_pressure)
    
    # === Critical-path acceleration gate: bounded linear gate (V2) — smooth, monotonic, no tanh artifacts ===
    rank_gate = np.clip(1.0 - 2.0 * np.clip(norm_slack, 0.0, PARAMS["slack_gate_width"]), 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Joint uncertainty-duration-slack gating (V2): only penalize duration risk when both slack is tight AND uncertainty is high ===
    joint_risk_gate = (
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) &
        (slack_pressure > 0)
    ).astype(float)
    duration_risk_interaction = norm_duration * joint_risk_gate * PARAMS["duration_robustness"]
    
    # === Starvation relief: tanh-based (V2) — preserves gradient across full wait range, avoids exp saturation bias ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * norm_wait)
    
    # === Rank-slack coupling: amplify criticality *only* under pressure (V2), now clipped to [0,2] for bounded influence ===
    coupled_rank = np.clip(norm_rank * (1.0 + PARAMS["rank_slack_coupling"] * slack_pressure), 0.0, 2.0)
    
    # === Work-pressure term: reward high remaining_work *only* when slack is constrained (novel) ===
    work_pressure = norm_work * np.where(slack_pressure > 0, 1.0, 0.0) * PARAMS["work_pressure_weight"]
    
    # === Energy-slack coupling: increase energy sensitivity under deadline pressure (novel) ===
    energy_under_pressure = norm_energy * (1.0 + PARAMS["energy_slack_coupling"] * slack_pressure)
    
    # === Final score: smaller = higher priority ===
    # Dominant terms: urgency (slack_pressure), critical path (coupled_rank reward), energy (penalty), risk (duration_risk_interaction)
    # Beneficial terms: starvation relief (wait_benefit), work-pressure (work_pressure) — both subtracted
    score = (
        + slack_pressure                                  # Urgency penalty (primary driver for lateness)
        + energy_under_pressure                          # Energy cost, amplified when late
        - coupled_rank                                   # Critical path reward (stronger when slack tight)
        - wait_benefit                                   # Starvation relief (anti-starvation bonus)
        + duration_risk_interaction                      # Risk-duration penalty (jointly gated)
        + work_pressure                                  # Work-pressure reward (prioritizes heavy critical paths)
    )
    
    # Ensure finite output and correct shape
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    
    return score
