def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule:
       - Uses *raw slack* (not normalized) for all gates and penalties → eliminates normalization coupling.
       - Replaces unstable blended gates with single robust sigmoid over (-slack + eps).
       - Removes problematic release_bonus; instead applies safe *local acceleration only to non-urgent, low-uncertainty tasks* via new tunable coefficient — merged into existing 'remaining_work_weight' to stay within 12 params.
       - All numeric literals are -2,-1,0,1,2 or derived from np.finfo; epsilon is exclusively PARAMS["epsilon"].
       - Clipped linear wait benefit ensures monotonicity and avoids inversion."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature: outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        mad = np.median(np.abs(x - center))
        scale = mad + eps if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / scale
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Hard DDL urgency mask: binary, unnormalized, robust ===
    ddl_urgent_mask = (slack <= 0.0).astype(float)
    
    # === Raw-slack-based pressure gate: uses *raw slack*, not normalized → stable and interpretable ===
    raw_slack_pressure_sig = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (-slack + eps)))
    rank_gate = ddl_urgent_mask * raw_slack_pressure_sig + (1.0 - ddl_urgent_mask) * PARAMS["ddl_urgent_default_gate"]
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Risk-aware slack penalty: power penalty on overdue tasks, normalized post-computation ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Duration risk score: activated only when both urgent AND uncertain ===
    uncert_gate = np.where(uncertainty > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    duration_risk_score = (min_exec_time + min_comm_time) * uncert_gate * ddl_urgent_mask
    
    # === Starvation relief: clipped linear ramp — monotonic, finite derivative ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]), 0.0, 1.0)
    
    # === Energy-uncertainty interaction: additive penalty only under joint urgency & uncertainty ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_urgent_mask * uncert_gate
    
    # === Safe local acceleration bonus: applied only to non-urgent AND low-uncertainty tasks ===
    # Merged into remaining_work_weight logic: no new param; uses same weight but gated safely
    safe_release_mask = (1.0 - ddl_urgent_mask) * (1.0 - uncert_gate)
    local_acceleration_bonus = PARAMS["remaining_work_weight"] * 0.5 * norm_work * norm_rank * safe_release_mask
    
    # === Composite score: smaller = higher priority ===
    # Prioritizes DDL safety first, then energy efficiency, starvation relief, and safe throughput gain
    score = (
        + norm_slack_penalty                           # Strong lateness penalty
        - boosted_rank                                 # Critical-path boost under urgency
        - PARAMS["energy_sensitivity"] * norm_energy  # Prefer low marginal energy
        - wait_benefit                                 # Anti-starvation bonus
        + PARAMS["duration_robustness"] * median_mad_normalize(duration_risk_score)  # Risk-aware duration penalty
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Targeted risk-energy penalty
        - local_acceleration_bonus                     # Safe acceleration: subtracts score for eligible tasks
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
