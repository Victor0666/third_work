def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: combines Parent 2's robust linear-ramp slack activation and energy-uncertainty interaction
       with Parent 1's bounded energy-slack decay and rank-uncertainty coupling—both gated by DDL feasibility.
       Uses sign-aware median/MAD normalization throughout. All terms clipped to [-2,2] for ordinal stability.
       Introduces novel *dual-gated* rank-uncertainty coupling: activated only when (i) uncertainty > threshold AND (ii) slack >= 0.
       Eliminates redundant sigmoid gates; replaces with monotonic ramp + Heaviside for interpretability and CMA-ES convergence."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust sign-aware normalization: median centering + MAD scaling (preserves original sign) ===
    def robust_normalize(x):
        if N == 1:
            center = x[0]
            spread = eps
        else:
            center = np.median(x)
            spread = np.median(np.abs(x - center))
        return (x - center) / (spread + eps)
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware lateness penalty: power-law on overdue slack only ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Bounded linear-ramp slack interaction: monotonic activation signal for criticality scaling ===
    slack_ramp = np.clip((1.0 - norm_slack) * PARAMS["slack_ramp_scale"], 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_ramp)
    
    # === DDL-protection gate: smooth Heaviside on slack → disable risk-sensitive terms when slack < 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Dual-gated rank-uncertainty coupling: only active under DDL feasibility AND elevated uncertainty ===
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    rank_uncert_coupling = norm_rank * norm_uncert * uncert_gate * ddl_gate
    rank_uncert_boost = PARAMS["rank_uncertainty_coupling"] * np.clip(rank_uncert_coupling, -2.0, 2.0)
    
    # === Starvation relief: saturating exponential with safe offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Uncertainty-duration interaction: same dual-gating logic ===
    duration_risk_score = norm_duration * norm_uncert * uncert_gate * ddl_gate
    
    # === Energy-uncertainty interaction: gated by both DDL feasibility and uncertainty threshold ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy decay: softens energy penalty exponentially as slack grows (avoids over-penalizing surplus) ===
    energy_decay_factor = np.exp(-PARAMS["energy_slack_decay_rate"] * np.maximum(slack, 0.0))
    energy_preference = norm_energy * ddl_gate * energy_decay_factor
    
    # === Composite score: smaller = higher priority; all terms clipped individually to preserve ranking integrity ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty dominates
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Critical path amplified by slack ramp
        - PARAMS["energy_sensitivity"] * np.clip(energy_preference, -2.0, 2.0)  # Softened energy preference under slack
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation always active
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + rank_uncert_boost                                                # Dual-gated critical-risk coupling
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)   # Downstream work bias
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
