def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces global starvation relief with *conditional* anti-starvation gate — activated only when a task has waited longer than threshold_ratio × median_wait, preventing overcorrection;
       retains decoupled exec/comm signals and median-MAD normalization;
       introduces wait_benefit_steepness for sharper, more controllable saturation;
       removes redundant wait_decay and comm_priority_under_low_uncertainty (merged into existing logic via norm_comm + ddl_gate);
       all intermediate terms clipped to [-2,2] for bounded AST depth and stability."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_exec = median_mad_normalize(min_exec_time)
    norm_comm = median_mad_normalize(min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Slack pressure signal: sigmoid over clipped negative slack → sharp urgency transition ===
    raw_slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    slack_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (raw_slack_pressure - 1.0)))
    
    # === Tight-coupled critical-path booster: upward_rank × remaining_work × ddl_gate (feasibility-aware) ===
    critical_path_booster = norm_rank * norm_work * ddl_gate
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded rank amplification: linearly coupled, clipped for stability ===
    rank_amplifier = 1.0 + PARAMS["criticality_scale"] * slack_pressure_gate
    rank_amplifier = np.clip(rank_amplifier, 1.0, 2.0)
    boosted_rank = norm_rank * rank_amplifier
    
    # === Uncertainty gating: threshold-triggered for coupling terms only ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    
    # === Conditional anti-starvation gate: activated only when wait exceeds adaptive threshold ===
    # Replaces global wait_decay — avoids overcorrection while targeting true starvation
    wait_median = np.median(ready_wait_time) if N > 1 else ready_wait_time[0]
    wait_threshold = PARAMS["wait_activation_threshold_ratio"] * (wait_median + PARAMS["wait_saturation_offset"])
    wait_activation = 1.0 / (1.0 + np.exp(-PARAMS["wait_benefit_steepness"] * (ready_wait_time - wait_threshold)))
    wait_benefit = wait_activation * (1.0 - np.exp(-1.0 * (ready_wait_time - wait_threshold + PARAMS["wait_saturation_offset"])))
    
    # === Energy-uncertainty coupling: gated by both uncertainty and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * raw_slack_pressure) * ddl_gate
    
    # === Decoupled execution & communication penalties: individually gated and scaled ===
    exec_penalty = norm_exec * slack_pressure_gate * ddl_gate
    # Removed comm_priority_under_low_uncertainty: now use direct norm_comm * ddl_gate for I/O awareness under feasibility
    comm_penalty = norm_comm * ddl_gate
    
    # === Composite score: smaller = higher priority; all terms clipped to [-2,2] for stability and scale balance ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - np.clip(critical_path_booster, -2.0, 2.0)                       # Feasibility-gated critical path unblocking
        - np.clip(boosted_rank, -2.0, 2.0)                                  # HEFT-based urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Conditional anti-starvation (not global)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(exec_penalty, -2.0, 2.0)                                 # Compute-bound urgency
        + np.clip(comm_penalty, -2.0, 2.0)                                 # I/O priority under deadline feasibility
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
