def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule:
       - Replaces adaptive power-law slack penalty with *bounded linear urgency ramp* under dual gate for numerical stability and interpretability.
       - Introduces explicit *congestion-thresholded starvation relief*: wait_benefit scaled by (1 - risk_gate) to prevent interference with deadline pressure.
       - Removes redundant rank_boost_upper_bound; uses dual-gated rank scaling + lower-bound clipping only.
       - All nonlinearities are bounded, all divisions guarded, all outputs finite and shape-(N,)."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust range-based normalization: handles outliers better than MAD for small N ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            xmin = xmax = x[0]
        else:
            xmin = np.min(x)
            xmax = np.max(x)
        spread = xmax - xmin if (xmax - xmin) > eps else eps
        return (x - xmin) / spread
    
    norm_slack = robust_range_normalize(slack)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # === Unified congestion signal: waiting time + uncertainty — direct proxy for resource contention ===
    congestion = ready_wait_time + uncertainty
    norm_congestion = robust_range_normalize(congestion)
    
    # === Hard DDL protection gate: binary activation only when slack <= 0 ===
    ddl_gate = (slack <= 0.0).astype(float)
    
    # === Congestion-aware risk gate: soft activation above normalized threshold ===
    risk_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_congestion - PARAMS["uncertainty_gate_threshold"])))
    
    # === Dual-gated critical path leverage: activated only under BOTH deadline breach AND high congestion ===
    dual_gate = ddl_gate * risk_gate
    critical_path_leverage = norm_rank * norm_work * dual_gate
    
    # === Bounded linear urgency ramp: maps norm_slack ∈ [0,1] → penalty ∈ [0,1] only when slack <= 0, zero otherwise ===
    # Ensures monotonic, differentiable, numerically stable urgency signal without power-law sensitivity
    urgency_ramp = np.clip(PARAMS["slack_ramp_slope"] * (1.0 - norm_slack), 0.0, 1.0) * ddl_gate
    norm_urgency_ramp = robust_range_normalize(urgency_ramp)  # preserves relative ordering under gate
    
    # === Starvation relief: exponentially saturating, then *suppressed under high congestion* ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    congestion_suppressed_wait = wait_benefit * (1.0 - risk_gate)
    
    # === Energy-uncertainty interaction: gated by both DDL breach and congestion threshold ===
    energy_uncert_penalty = norm_energy * norm_uncert * dual_gate
    
    # === Duration-uncertainty coupling: new term penalizing tasks with high exec+comm time AND high uncertainty ===
    duration = min_exec_time + min_comm_time
    norm_duration = robust_range_normalize(duration)
    duration_uncert_penalty = norm_duration * norm_uncert * dual_gate
    
    # === Bounded rank amplification: linear coupling under dual pressure, clipped only at lower bound ===
    # Removed upper bound: dual_gate already bounds influence; lower bound prevents collapse under low pressure
    rank_boost = 1.0 + PARAMS["criticality_scale"] * urgency_ramp
    rank_boost = np.clip(rank_boost, PARAMS["rank_boost_lower_bound"], 2.0)  # cap at 2.0 for stability
    boosted_rank = norm_rank * rank_boost
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_urgency_ramp, -2.0, 2.0)                           # Dominant, stable lateness pressure
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Unblock critical paths only under dual pressure
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency-scaled rank under dual pressure
        - np.clip(congestion_suppressed_wait, -2.0, 2.0)                  # Anti-starvation only when congestion is low
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy only under breach
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["duration_uncertainty_coupling"] * np.clip(duration_uncert_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * ddl_gate, -2.0, 2.0)
        + np.clip(norm_congestion * ddl_gate, -2.0, 2.0)                   # Direct congestion penalty under breach
        + PARAMS["starvation_congestion_coupling"] * np.clip(-congestion_suppressed_wait, -2.0, 2.0)  # Explicit suppression term
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
