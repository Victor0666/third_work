def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust median-MAD normalization and sharpened gates with Parent 1's validated successor-release interaction.
       Structural novelty: (1) successor-release now couples wait time *and* descendant criticality under DDL breach, (2) unified dual-gated slack-pressure activation,
       (3) strict [-2,2] clipping per term for stability, (4) explicit anti-starvation + anti-lateness decoupling via orthogonal terms."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization — outlier-resilient, sign-preserving, preserves zero-crossings ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Unified DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === HARD DDL breach indicator for critical-path leverage and successor release ===
    ddl_breach = (slack <= 0.0).astype(float)
    
    # === Critical path leverage: upward_rank * remaining_work, activated only under breach ===
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack → normalized → clipped ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate: sigmoid centered at slack=0, steepened near deadline ===
    slack_pressure_raw = np.clip(-norm_slack, 0.0, 2.0)
    slack_pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_raw - 0.5)))
    
    # === Uncertainty-triggered duration risk: gated by both uncertainty threshold AND slack pressure ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * uncert_gate * slack_pressure_raw * ddl_gate
    
    # === Starvation relief: logistic-shaped anti-starvation bonus, robust to zero wait ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]))
    
    # === Successor-release coupling: prioritizes tasks whose delay blocks high-criticality descendants *and* have waited long, under DDL breach ===
    # Uses normalized rank (descendant importance) * normalized wait * breach flag → no extra normalization needed
    successor_release = norm_wait * norm_rank * ddl_breach * PARAMS["successor_release_coupling"]
    
    # === Energy-uncertainty coupling: only active when both energy and uncertainty are elevated AND slack permits ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty proportionally to urgency pressure, gated by feasibility ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure_raw) * ddl_gate
    
    # === Composite score: smaller = higher priority. All terms clipped to [-2,2] to prevent dominance and ensure bounded AST depth ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness core penalty
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Unblock critical downstream work
        - np.clip(norm_rank * (1.0 + PARAMS["criticality_scale"] * slack_pressure_gate), -2.0, 2.0)  # Urgency-amplified rank
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work * ddl_gate, -2.0, 2.0)
        - np.clip(successor_release, -2.0, 2.0)                            # Blocker-aware release under breach
    )
    
    # Final safeguard: ensure finite, deterministic output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
