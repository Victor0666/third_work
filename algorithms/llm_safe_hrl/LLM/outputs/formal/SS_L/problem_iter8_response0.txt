def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's hard DDL gate and successor-release logic with Parent 1's negative-slack-only activation and robust median-MAD normalization.
       Introduces novel *dual-gated criticality*: upward_rank boosted only under slack<=0 (hard) AND via smooth sigmoid (soft), weighted by uncertainty.
       Uses unified robust normalization; removes unstable power penalties; enforces deterministic shape and finite output."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization: handles outliers, zero-scales, and N=1 safely ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 0:
            return np.zeros(0, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        dev = x - center
        spread = np.median(np.abs(dev)) if N > 1 else np.abs(dev[0]) + eps
        return dev / (spread + eps)
    
    # Normalize base features — preserve sign semantics for directional scoring
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Hard DDL urgency gate: binary trigger for absolute deadline enforcement ===
    ddl_urgent = (slack <= 0.0).astype(float)
    
    # === Dual-gated criticality: combines hard and soft slack pressure with uncertainty modulation ===
    # Soft gate: sigmoid steepness on negative slack (smooth ramp for near-deadline tasks)
    soft_slack_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (-norm_slack)))
    # Uncertainty-modulated boost: stronger criticality amplification when uncertainty is high
    uncertainty_mod = 1.0 + PARAMS["negative_slack_only_activation"] * norm_uncert * (ddl_urgent + (1.0 - ddl_urgent) * soft_slack_gate)
    protected_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * uncertainty_mod)
    
    # === Successor-release interaction: unblocks downstream parallelism (Parent 2 strength) ===
    successor_release_score = norm_work * norm_rank
    
    # === Energy-uncertainty interaction: penalize high-energy only when uncertainty exceeds threshold (Parent 2) ===
    energy_uncert_penalty = norm_energy * np.where(norm_uncert > PARAMS["uncertainty_gate_threshold"], 1.0, 0.0)
    
    # === Negative-slack-only duration-risk coupling: prevents premature delay under safety margin (Parent 1) ===
    negative_slack_mask = ddl_urgent * PARAMS["negative_slack_only_activation"]
    duration_risk_score = norm_duration * norm_uncert * negative_slack_mask
    
    # === Starvation relief: saturating exponential with safe offset (Parent 2) ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Composite score: smaller = higher priority ===
    # Prioritizes hard deadline compliance first, then critical path acceleration, then starvation mitigation,
    # while penalizing risky/high-energy choices only when justified by constraint violation.
    score = (
        - ddl_urgent * PARAMS["ddl_urgency_boost"]           # Absolute deadline enforcement (dominant)
        - protected_rank                                     # Dual-gated criticality (robust & adaptive)
        - successor_release_score                            # Unblocks downstream work (parallelism gain)
        - PARAMS["energy_sensitivity"] * norm_energy       # Base energy efficiency
        - wait_benefit                                       # Anti-starvation reward
        + PARAMS["duration_robustness"] * duration_risk_score  # Risk-duration penalty *only* when overdue
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # Risk-aligned energy penalty
        + PARAMS["remaining_work_weight"] * norm_work        # Mild preference for large downstream load
    )
    
    # Final safeguard: ensure finite, deterministic output with strict shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    return score.reshape(-1)
