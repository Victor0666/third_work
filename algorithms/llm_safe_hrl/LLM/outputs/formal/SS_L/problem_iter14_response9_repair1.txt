def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's hard DDL-dominant gating with Parent 1's bounded rank-uncertainty coupling,
       enhanced by downstream-risk-aware amplification. Introduces novel 'downstream_risk_amplifier' logic via composition
       of existing parameters — avoids adding new parameter while preserving structural intent. Replaces global robust_normalize
       with feature-specific MAD-normalization that preserves sign semantics while bounding variance sensitivity."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Feature-specific robust normalization: per-feature median + MAD (sign-preserving, outlier-resistant) ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features individually
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Hard DDL-dominant energy disable gate: binary mask for strict feasibility-first behavior ===
    safe_slack_mask = (norm_slack > PARAMS["safe_slack_energy_disable_threshold"])
    energy_enabled = np.where(safe_slack_mask, 0.0, 1.0)
    
    # === Conditional urgency detection: tight or overdue slack AND sufficient relative margin pressure ===
    slack_margin_ratio = np.abs(slack) / (min_exec_time + min_comm_time + eps)
    ddl_urgent = np.where((slack <= 0.0) | (slack_margin_ratio < PARAMS["safe_slack_energy_disable_threshold"]), 1.0, 0.0)
    
    # === Downstream-risk-aware rank-uncertainty coupling: activated only when overdue AND high downstream work ===
    # Composed from existing parameters: uses norm_work > 0.0 (median threshold) and ddl_urgent → no new param needed
    work_above_median = np.where(norm_work > 0.0, 1.0, 0.0)
    downstream_risk_active = ddl_urgent * work_above_median
    rank_uncert_coupling = norm_rank * norm_uncert * downstream_risk_active
    # Bounded coupling using existing parameters only — no new parameter introduced
    rank_uncert_boost = PARAMS["criticality_scale"] * 0.5 * np.clip(rank_uncert_coupling, -2.0, 2.0)
    
    # === Sharpened slack-pressure gate: sigmoid over normalized slack for stable fuzzy timing transitions ===
    slack_pressure_norm = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure_norm - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate * ddl_urgent)
    
    # === Energy-uncertainty interaction: gated by urgency and safety, enforcing risk-aligned marginal penalty ===
    energy_modulation_gate = np.where(
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) & (ddl_urgent == 1.0),
        1.0,
        0.0
    )
    energy_uncert_penalty = norm_energy * norm_uncert * energy_modulation_gate * energy_enabled
    
    # === Starvation relief: logistic with explicit saturation offset to avoid numerical instability ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (norm_wait + PARAMS["wait_saturation_offset"]))
    
    # === Duration robustness: penalize long durations only under dual pressure (urgency + uncertainty) ===
    duration_risk_score = norm_duration * norm_uncert * ddl_urgent
    
    # === Composite score: smaller = higher priority ===
    # Strict hierarchy: (1) deadline bias dominates, (2) lateness penalty sharpens under urgency,
    # (3) critical path amplified only when overdue+high-work, (4) energy/uncertainty penalties gated tightly
    score = (
        + np.clip(norm_slack, -2.0, 2.0)                                  # Primary deadline bias
        + np.clip(norm_slack ** PARAMS["slack_penalty_exponent"], -2.0, 2.0) * ddl_urgent  # Nonlinear lateness penalty only when urgent
        - np.clip(boosted_rank, -2.0, 2.0)                                 # Critical path boost with sigmoid coupling
        - np.clip(norm_work * norm_rank, -2.0, 2.0)                       # Unblock high-work successors
        - np.clip(norm_energy * PARAMS["energy_sensitivity"] * energy_modulation_gate * energy_enabled, -2.0, 2.0)  # Hard-gated energy penalty
        - np.clip(norm_duration * (1.0 - ddl_urgent), -2.0, 2.0)          # Soft short-duration preference when not urgent
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation bonus
        + np.clip(PARAMS["duration_robustness"] * duration_risk_score, -2.0, 2.0)  # Risk-aware duration penalty
        + np.clip(PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty, -2.0, 2.0)  # Risk-aligned marginal energy penalty
        + np.clip(PARAMS["remaining_work_weight"] * norm_work, -2.0, 2.0)  # Mild preference for large downstream work
        + np.clip(rank_uncert_boost, -2.0, 2.0)                            # Downstream-risk-aware criticality boost
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
