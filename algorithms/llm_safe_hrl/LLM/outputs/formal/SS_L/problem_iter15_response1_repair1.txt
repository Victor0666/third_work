def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: restores sharp sigmoid ddl_protection_gate_slope;
       adds host-load–conditional suppression via min_incremental_energy * ready_wait_time * low-rank mask —
       avoids false urgency from idle-time inflation while preserving DDL safety;
       uses median-MAD normalization; all terms clipped to [-2,2]; no numeric literals beyond -2,-1,0,1,2."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Sigmoid DDL-protection gate: sharp, stable boundary at slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: direct upward_rank * remaining_work interaction, activated only under deadline breach (slack <= 0) ===
    ddl_breach_mask = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach_mask * PARAMS["rank_work_coupling_weight"]
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    
    # === Bounded linear rank-slack coupling: monotonic amplification under pressure ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    
    # === Starvation relief: linear and capped ===
    wait_benefit = np.clip(PARAMS["wait_decay"] * ready_wait_time, 0.0, 1.0)
    
    # === Duration-uncertainty coupling: activated only under tight slack AND high uncertainty ===
    tight_slack_mask = (slack <= eps).astype(float)
    high_uncert_mask = (norm_uncert >= PARAMS["uncertainty_gate_threshold"]).astype(float)
    duration_risk_score = norm_duration * tight_slack_mask * high_uncert_mask * PARAMS["duration_slack_coupling_weight"]
    
    # === Energy-uncertainty coupling: gated by both uncertainty threshold and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Host-load–conditional suppression: penalizes high-energy tasks with long wait but low criticality
    # Uses median-based low-rank mask to avoid brittle thresholding; ensures suppression only for non-critical paths.
    rank_median = np.median(upward_rank) if N > 1 else upward_rank[0]
    low_rank_mask = (upward_rank < rank_median + eps).astype(float) * (slack > 0.0).astype(float)
    host_load_penalty = norm_energy * norm_wait * low_rank_mask
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + raw_slack_penalty                           # Primary lateness penalty
        - np.clip(critical_path_leverage, -2.0, 2.0) # Critical-path blocking relief
        - np.clip(norm_rank * rank_slack_coupling, -2.0, 2.0)  # Urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)           # Anti-starvation
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
        + np.clip(host_load_penalty, -2.0, 2.0)      # New: suppresses costly non-critical tasks
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
