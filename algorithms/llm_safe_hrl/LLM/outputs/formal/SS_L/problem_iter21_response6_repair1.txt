def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces unstable power-law rank-slack coupling with *clipped linear hinge* for robust monotonic urgency;
       unifies all risk terms under a single *joint slack-uncertainty gate* to reduce over-parameterization;
       introduces *normalized wait-time saturation* using softplus instead of exp to prevent overflow;
       removes redundant duration_robustness — now fully absorbed into joint_gate and slack_pressure scaling;
       retains validated critical_path_leverage and DDL-protection semantics while tightening AST depth and eliminating inf-producing branches."""
    eps = PARAMS["epsilon"]
    
    # Safe cast; no in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization — outlier-resistant and sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Unified joint gate: active only when both slack >= 0 AND uncertainty is low enough ===
    # Replaces separate ddl_gate and uncert_gate → reduces interaction explosion & improves stability
    joint_gate = (
        1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack)) *
        1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    )
    
    # === DDL breach indicator for critical path leverage (validated diagnostic) ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Robust slack penalty: capped power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Stable hinge-based rank-slack coupling (replaces brittle power-law): linear ramp clipped at [-1,1] ===
    # Ensures monotonic, bounded, and numerically safe urgency scaling under pressure
    slack_hinge = np.clip(-norm_slack, 0.0, 1.0)  # [0,1] under pressure; 0 otherwise
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * slack_hinge
    
    # === Sharpened slack-pressure gate for rank boosting (same as before but now safer) ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Softplus-based starvation relief: avoids exp overflow, saturates smoothly ===
    # Replaces 1 - exp(-x) with log1p(exp(x)) → same asymptotics, stable at large x
    wait_benefit = np.log1p(np.exp(-PARAMS["wait_decay"] * (ready_wait_time + eps)))
    
    # === Unified risk terms under joint_gate — cleaner, fewer interactions ===
    duration_risk_score = norm_duration * norm_uncert * joint_gate
    energy_uncert_penalty = norm_energy * norm_uncert * joint_gate
    energy_slack_penalty = norm_energy * slack_pressure * joint_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty dominates
        - np.clip(critical_path_leverage, -2.0, 2.0)                      # Critical-path unblocking boost
        - np.clip(boosted_rank, -2.0, 2.0)                                  # Urgency amplification
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * joint_gate, -2.0, 2.0)  # Energy preference only when safe
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Starvation relief (softplus-saturated)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
