def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: inherits robust sign-preserving normalization and unified slack pressure from Parent 2;
       adds energy-uncertainty interaction and slack-gated work leverage from Parent 1;
       removes harmful work_pressure mask; uses linear slack reward + power-law penalty for balanced feasibility-first ranking;
       all literals are -2,-1,0,1,2; epsilon via PARAMS; no side effects."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Sign-preserving robust normalization: median-center + MAD scale + epsilon (fixes slack/wait distortion) ===
    def robust_normalize(x):
        center = np.median(x)
        scale = np.median(np.abs(x - center)) + eps
        return (x - center) / scale
    
    # Normalize base features — preserves physical sign semantics (e.g., negative slack → negative norm_slack)
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Unified slack pressure: power-law penalty only for negative slack (lateness risk), avoids over-penalizing safe tasks ===
    slack_pressure = np.power(np.clip(-norm_slack, 0.0, None), PARAMS["slack_penalty_exponent"])
    
    # === Critical-path acceleration gate: smooth transition from full boost to zero over [0, slack_gate_width] ===
    rank_gate = np.clip(1.0 - 2.0 * np.clip(norm_slack, 0.0, PARAMS["slack_gate_width"]), 0.0, 1.0)
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Uncertainty-duration interaction: activate only under both high uncertainty AND deadline pressure ===
    uncert_gate = (
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) &
        (slack_pressure > 0)
    ).astype(float)
    duration_risk_interaction = norm_duration * uncert_gate * PARAMS["duration_robustness"]
    
    # === Energy-uncertainty interaction: penalize high-energy tasks when uncertainty is high and slack is tight ===
    energy_uncert_gate = (
        (norm_uncert > PARAMS["uncertainty_gate_threshold"]) &
        (slack_pressure > 0)
    ).astype(float)
    energy_uncert_penalty = norm_energy * norm_uncert * energy_uncert_gate * PARAMS["energy_uncertainty_interaction"]
    
    # === Starvation relief: bounded, smooth, differentiable tanh using sign-preserving norm_wait ===
    wait_benefit = np.tanh(PARAMS["wait_decay"] * norm_wait)
    
    # === Slack sensitivity term: linear reward for positive slack (encourages early scheduling of safe tasks) ===
    # Prevents premature congestion and improves deadline feasibility
    slack_reward = PARAMS["slack_sensitivity"] * np.clip(norm_slack, None, 0.0)
    
    # === Slack-gated work leverage: prefer unblocking large downstream work only when slack is non-negative (safe context) ===
    # Avoids over-committing critical resources to large work under DDL stress
    work_leverage_gate = (slack >= 0.0).astype(float)
    work_leverage = norm_work * work_leverage_gate * PARAMS["work_leverage_factor"]
    
    # === Final score: smaller = higher priority ===
    # Dominant terms: slack_pressure (hard DDL enforcement), slack_reward (feasibility-first bias), boosted_rank (critical path)
    # Secondary refinement: energy cost, duration risk, energy-uncertainty penalty, starvation relief, work leverage
    score = (
        + slack_pressure                                  # Urgency penalty (dominant for late tasks)
        + PARAMS["energy_sensitivity"] * norm_energy     # Energy cost (penalty)
        - boosted_rank                                   # Critical path reward (only when needed)
        + duration_risk_interaction                      # Risk-duration penalty (conditional)
        - wait_benefit                                   # Starvation relief (reward)
        - slack_reward                                   # Positive-slack reward (prevents premature congestion)
        + energy_uncert_penalty                          # High-risk-high-energy penalty (conditional)
        - work_leverage                                  # Large-work unblocking benefit (only when safe)
    )
    
    # Final safeguard: ensure finite, deterministic, shape-(N,) output
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required even for N=1
    return score.reshape(-1)
