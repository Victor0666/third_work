def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces fragile binary gate with smooth sigmoid DDL urgency gate,
       adds rank_boost_floor to prevent critical-path suppression under mild pressure,
       reverts to pure raw-slack-penalty dominance for violation enforcement,
       and simplifies interactions to improve robustness and CMA-ES convergence.
       All 12 parameters are used; no numeric literals beyond -2,-1,0,1,2."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization: sign-preserving, outlier-resilient scaling ===
    def robust_normalize(x):
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Smooth DDL urgency gate: sigmoid transition instead of binary — avoids optimization cliffs and improves gradient flow ===
    # Gate activates gradually as slack approaches zero or becomes negative; steepness tunable
    urgency_input = -slack / (np.abs(np.median(slack)) + eps) if N > 1 else -slack[0] / (np.abs(slack[0]) + eps)
    ddl_urgency_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_urgency_steepness"] * urgency_input))
    
    # === Pure raw-slack-penalty dominance for hard deadline enforcement — no blending, no dilution ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    
    # === Critical-path boost with floor: ensures upward_rank contributes meaningfully even under non-critical slack ===
    # Prevents rank suppression by applying floor-scaled boost across all tasks, amplified under urgency
    rank_boost = PARAMS["rank_boost_floor"] + (1.0 - PARAMS["rank_boost_floor"]) * ddl_urgency_gate
    protected_rank = norm_rank * rank_boost
    
    # === Starvation relief: capped linear bonus with saturation offset ===
    wait_benefit = np.clip(
        PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]),
        0.0,
        1.0
    )
    
    # === Energy-uncertainty interaction: penalize only when both energy and uncertainty exceed thresholds ===
    energy_uncert_penalty = (
        norm_energy * norm_uncert *
        np.where(
            (norm_energy > 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]),
            1.0,
            0.0
        )
    )
    
    # === Duration robustness: penalize long durations only under combined risk (tight slack + high uncertainty) ===
    duration_risk_score = (
        norm_duration *
        np.where(
            (slack <= PARAMS["epsilon"]) & (uncertainty >= PARAMS["uncertainty_gate_threshold"]),
            1.0,
            0.0
        )
    )
    
    # === Composite score: smaller = higher priority ===
    score = (
        + raw_slack_penalty                           # Pure, unblended lateness penalty — preserves hard DDL enforcement
        - PARAMS["criticality_scale"] * protected_rank  # Critical-path boost with floor prevents suppression
        - PARAMS["energy_sensitivity"] * norm_energy   # Prefer low marginal energy
        - norm_duration                                # Prefer short duration
        - wait_benefit                                 # Anti-starvation bonus
        + PARAMS["duration_robustness"] * duration_risk_score  # Penalize risky+slow under combined pressure
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # High-risk-high-energy penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
