def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: replaces median-MAD with task-level min-max ranking for robust critical signal preservation;
       introduces successor-release-aware gating: (1 - exp(-remaining_work / (slack + ε))) to activate criticality only when slack permits unblocking;
       removes all duration-related terms per diagnostics; uses percentile-based relative ranking to eliminate outlier distortion in slack/energy;
       retains DDL-protection, power-law slack penalty, explicit critical-path leverage, and bounded inverse-time starvation relief."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Task-level min-max ranking: preserves ordinal structure, immune to outliers, monotonic ===
    # Maps each feature to [0,1] where 0 = best (e.g., smallest slack penalty, lowest energy), 1 = worst
    def rank_minmax(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            return np.array([0.0])
        x_min, x_max = np.min(x), np.max(x)
        spread = x_max - x_min
        if spread < eps:
            return np.full(N, 0.5)
        return (x - x_min) / (spread + eps)
    
    # Rank-based features: smaller rank value = better (e.g., low slack → high urgency → low rank)
    rank_slack = rank_minmax(np.maximum(-slack, 0.0))  # Higher penalty → higher rank → worse
    rank_energy = rank_minmax(min_incremental_energy)
    rank_rank = rank_minmax(-upward_rank)  # Higher rank → lower rank_score → better
    rank_work = rank_minmax(-remaining_work)  # Larger downstream work → lower rank → better
    rank_wait = rank_minmax(-ready_wait_time)  # Longer wait → lower rank → better
    rank_uncert = rank_minmax(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: explicit rank × work × ddl_breach term with dedicated weight ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = rank_rank * rank_work * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then ranked ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    rank_slack_penalty = rank_minmax(raw_slack_penalty)
    
    # === Slack-pressure gate: bounded sigmoid centered at fixed 0.5 in [0,1] rank space ===
    slack_pressure = np.clip(-rank_slack, 0.0, 1.0)  # Now in [0,1]
    pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 0.5)))
    
    # === Coupled rank: amplify criticality only under slack pressure ===
    coupled_rank = rank_rank * (1.0 + PARAMS["criticality_scale"] * pressure_gate)
    
    # === Successor-release-aware gating: activates criticality only when slack allows unblocking high-work paths ===
    # Gate = 1 - exp(-(remaining_work)/(slack + ε)) → near 1 when slack >> work, near 0 when slack ≈ 0 or negative
    release_gate_denom = slack + eps
    release_gate = 1.0 - np.exp(-PARAMS["successor_release_gate_steepness"] * (remaining_work / (release_gate_denom + eps)))
    # Clamp to [0,1] and apply DDL feasibility filter
    release_gate = np.clip(release_gate * ddl_gate, 0.0, 1.0)
    
    # === Starvation relief: bounded inverse-time form — monotonic, no singularities ===
    wait_benefit = np.clip(1.0 / (1.0 + np.exp(-PARAMS["wait_decay"] * (ready_wait_time + eps))), 0.0, 1.0)
    
    # === Energy-uncertainty coupling: gated by uncertainty threshold and DDL feasibility ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (rank_uncert - PARAMS["uncertainty_gate_threshold"])))
    energy_uncert_penalty = rank_energy * rank_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = rank_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms scaled to [-2,2] via multiplication by 2 ===
    score = (
        + rank_slack_penalty * 2.0                          # Primary lateness penalty
        - PARAMS["critical_path_leverage_weight"] * critical_path_leverage * 2.0  # Explicit downstream impact
        - coupled_rank * 2.0                                  # HEFT-based urgency under pressure
        - PARAMS["energy_sensitivity"] * rank_energy * ddl_gate * 2.0  # Energy preference only when feasible
        - wait_benefit * 2.0                                 # Anti-starvation
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty * 2.0
        + PARAMS["slack_penalty_exponent"] * energy_slack_penalty * 2.0
        + PARAMS["remaining_work_weight"] * rank_work * 2.0
        - release_gate * 2.0                                 # Successor-release boost: prioritize unblocking only when slack permits
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
