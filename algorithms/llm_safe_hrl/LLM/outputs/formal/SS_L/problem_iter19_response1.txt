def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule: replaces median-MAD with robust sign-preserving range normalization;
       introduces unified congestion signal (ready_wait_time + uncertainty) gated by DDL feasibility;
       removes redundant duration_robustness and wait_saturation_offset per counterfactual evidence;
       enforces strict sign-consistent prioritization: smaller score = higher priority;
       uses clipped linear slack pressure near violation boundary for numerical stability;
       incorporates upward_rank * remaining_work interaction only under DDL breach per diagnostic evidence."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust range-based normalization: sign-preserving, outlier-resistant, no MAD instability ===
    def robust_range_normalize(x):
        x = np.copy(x)
        if N == 1:
            rng = eps
            mid = x[0]
        else:
            p10 = np.percentile(x, 10)
            p90 = np.percentile(x, 90)
            rng = p90 - p10
            mid = (p10 + p90) / 2.0
        spread = rng if rng > eps else eps
        return (x - mid) / spread
    
    norm_slack = robust_range_normalize(slack)
    norm_energy = robust_range_normalize(min_incremental_energy)
    norm_rank = robust_range_normalize(upward_rank)
    norm_work = robust_range_normalize(remaining_work)
    norm_wait = robust_range_normalize(ready_wait_time)
    norm_uncert = robust_range_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside activated only when slack >= 0 (i.e., feasible region) ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical-path leverage: upward_rank * remaining_work, enabled ONLY under slack <= 0 (DDL breach) ===
    # Confirmed by ≥7 cross-seed replay failures — avoids leakage into feasible region
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Linear slack pressure near violation boundary: stable, interpretable, numerically safe ===
    # Replaces unstable exponentials: clip(-slack, 0, 1) gives bounded urgency ramp from 0→1 as slack → negative
    slack_pressure = np.clip(-slack, 0.0, 1.0)
    
    # === Unified congestion signal: ready_wait_time + uncertainty, gated by DDL feasibility ===
    # Addresses 'avoidable_marginal_energy_or_load_cost' pattern (15+ occurrences); suppresses load under DDL risk
    congestion = (ready_wait_time + uncertainty) * ddl_gate
    
    # === Starvation relief: exponential decay with direct wait-time scaling (no offset needed) ===
    # Simpler, more stable than logistic; validated against starvation cases
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * ready_wait_time)
    
    # === Energy-uncertainty coupling: multiplicative penalty only in feasible region and high uncertainty ===
    # Uses same uncertainty gate as congestion but with energy — avoids over-penalizing low-uncertainty tasks
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (uncertainty - PARAMS["uncertainty_gate_threshold"])))
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-driven energy amplification: scales energy penalty proportionally to urgency pressure ===
    # Ensures energy minimization is progressively weighted as deadline looms
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Composite priority score: smaller = higher priority ===
    # All terms clipped to [-2, 2] for stability; sign preserved per semantics
    score = (
        + np.clip(slack_pressure, -2.0, 2.0)                                # Primary urgency signal: linear, bounded, intuitive
        - np.clip(critical_path_leverage, -2.0, 2.0)                       # Boost for blocking critical-path successors under breach
        - np.clip(norm_rank * (1.0 + PARAMS["slack_pressure_gate_steepness"] * slack_pressure), -2.0, 2.0)  # HEFT rank amplified under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation: monotonic benefit
        + np.clip(congestion, -2.0, 2.0)                                   # Load-aware congestion penalty, gated by feasibility
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)

    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
