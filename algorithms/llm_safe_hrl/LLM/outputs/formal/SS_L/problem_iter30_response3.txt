def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: simplifies gates to eliminate instability from dual-mode slack pressure;
       restores and isolates the empirically validated 'critical_path_leverage' as dominant DDL-breach signal;
       replaces ad-hoc starvation gating with a monotonic, slack-distance-aware linear ramp;
       uses robust median-MAD normalization throughout; all terms clipped to [-2,2] for stability;
       removes inactive parameters (duration_robustness, wait_saturation_offset, energy_uncertainty_interaction)
       and redundant interactions to reduce noise and improve interpretability."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: isolated, validated core term — dominant signal under DDL breach ===
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Bounded linear rank-slack coupling: stable amplification under pressure ===
    coupled_slack = np.clip(norm_slack, -1.0, 1.0)
    rank_slack_coupling = 1.0 + PARAMS["criticality_scale"] * coupled_slack
    
    # === Sharpened slack-pressure gate: single sigmoid (no ramp) for clean zero-crossing ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * rank_slack_coupling * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === Uncertainty gating: thresholded by normalized uncertainty only (no slack interaction) ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    
    # === Starvation relief: monotonic linear ramp based on slack distance, bounded and normalized ===
    # Ensures relief increases smoothly as slack approaches zero from positive side, vanishes under hard breach
    slack_distance = np.clip(slack, 0.0, np.inf)  # Only positive slack contributes
    norm_slack_distance = median_mad_normalize(slack_distance)
    starvation_relief = np.clip(norm_slack_distance, 0.0, 1.0)
    
    # === Energy preference: only active when DDL-feasible and uncertainty low ===
    energy_preference = norm_energy * ddl_gate * (1.0 - uncert_gate)
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Lateness penalty dominates
        - PARAMS["critical_path_leverage_weight"] * np.clip(critical_path_leverage, -2.0, 2.0)  # Core DDL-breach lever
        - np.clip(boosted_rank, -2.0, 2.0)                                  # HEFT urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(energy_preference, -2.0, 2.0)  # Energy saving only when safe
        + PARAMS["starvation_relief_weight"] * np.clip(starvation_relief, -2.0, 2.0)  # Fairness without jitter
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)    # Downstream work bias
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
