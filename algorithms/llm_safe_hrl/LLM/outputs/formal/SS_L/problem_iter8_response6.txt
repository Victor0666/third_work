def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's binary DDL gate and sign-preserving normalization with Parent 1's slack-pressure gating insight,
       enhanced by a novel adaptive slack-mode switch and strengthened successor-release interaction.
       Introduces slack_sensitivity_threshold to smoothly transition between raw-penalty and normalized regimes — improving stability near zero slack.
       Replaces fragile sigmoid with piecewise linear activation for better CMA-ES convergence and interpretability.
       All 12 parameters are used; no numeric literals beyond -2,-1,0,1,2."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Sign-preserving median-MAD normalization: retains directionality for slack/energy tradeoffs ===
    def robust_normalize(x):
        center = np.median(x) if N > 1 else x[0]
        spread = np.median(np.abs(x - center)) if N > 1 else np.abs(x[0] - center) + eps
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Adaptive slack-mode switch: smooth transition from robust to raw-penalty dominance near critical slack ===
    # Uses piecewise linear blend: full raw penalty when slack <= 0, full robust when slack >= threshold, linear in between
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    slack_mode_weight = np.clip((PARAMS["slack_sensitivity_threshold"] - slack) / (PARAMS["slack_sensitivity_threshold"] + eps), 0.0, 1.0)
    primary_slack_score = (
        slack_mode_weight * raw_slack_penalty +
        (1.0 - slack_mode_weight) * norm_slack
    )
    
    # === Binary DDL-urgent trigger: activates under hard violation or high uncertainty + tight slack ===
    ddl_urgent = np.where(
        (slack <= 0.0) | 
        ((slack <= PARAMS["slack_sensitivity_threshold"]) & (uncertainty >= PARAMS["uncertainty_gate_threshold"])),
        1.0,
        0.0
    )
    
    # === Critical-path boost: apply only under urgency, using upward_rank * remaining_work to capture unblocking power ===
    successor_release_boost = norm_rank * norm_work * PARAMS["rank_work_interaction_strength"]
    protected_release_score = successor_release_boost * ddl_urgent
    
    # === Energy-uncertainty interaction: penalize only when both energy and uncertainty are above thresholds ===
    energy_uncert_penalty = (
        norm_energy * norm_uncert *
        np.where(
            (norm_energy > 0.0) & (norm_uncert > PARAMS["uncertainty_gate_threshold"]),
            1.0,
            0.0
        )
    )
    
    # === Starvation relief: capped linear bonus with saturation offset, preserving monotonicity ===
    wait_benefit = np.clip(
        PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]),
        0.0,
        1.0
    )
    
    # === Duration robustness: penalize long durations only under combined risk (tight slack + high uncertainty) ===
    duration_risk_score = (
        norm_duration *
        np.where(
            (slack <= PARAMS["slack_sensitivity_threshold"]) & (uncertainty >= PARAMS["uncertainty_gate_threshold"]),
            1.0,
            0.0
        )
    )
    
    # === Composite score: smaller = higher priority ===
    score = (
        + primary_slack_score                          # Dominant urgency signal, adaptively blended
        - PARAMS["criticality_scale"] * protected_release_score  # Boosts tasks that unblock critical downstream work *only* when urgent
        - PARAMS["energy_sensitivity"] * norm_energy   # Prefer low marginal energy
        - norm_duration                                # Prefer short duration
        - wait_benefit                                 # Anti-starvation bonus (reduces score)
        + PARAMS["duration_robustness"] * duration_risk_score  # Penalize risky+slow under combined pressure
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty  # High-risk-high-energy penalty
        + PARAMS["remaining_work_weight"] * norm_work  # Mild preference for large downstream work (adds load-awareness)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
