def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: combines Parent 2's DDL-protection gate with Parent 1's bounded linear starvation relief,
       uses unified MAD-based robust normalization, and enforces strict gating of all energy/uncertainty terms by ddl_gate.
       All numeric literals are -2,-1,0,1,2 or np.finfo guards; no hidden constants; exactly 13 parameters (added wait_benefit_cap)."""
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Unified robust normalization: MAD-based, sign-preserving, handles N=1 gracefully ===
    def robust_normalize(x):
        x = np.asarray(x)
        if N == 1:
            center = x[0]
            spread = eps
        else:
            center = np.median(x)
            spread = np.median(np.abs(x - center))
        return (x - center) / (spread + eps)
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === Risk-aware slack penalty: power penalty on overdue tasks, then robust-normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Sharpened slack-pressure gate: bounded [0,2] input for sigmoid to avoid extreme slopes ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * rank_gate)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty terms ONLY when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Uncertainty-duration interaction: activated only under dual pressure AND slack feasibility ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * uncert_gate * slack_pressure * ddl_gate
    
    # === Starvation relief: bounded linear ramp (from Parent 1), now capped via declared parameter ===
    wait_benefit_raw = PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"])
    wait_benefit = np.clip(wait_benefit_raw, 0.0, PARAMS["wait_benefit_cap"])
    
    # === Energy-uncertainty interaction: gated by both DDL feasibility and uncertainty threshold ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Composite score: smaller = higher priority ===
    score = (
        + norm_slack_penalty                           # Dominant lateness risk penalty
        - boosted_rank                                 # Critical-path boost under pressure
        - PARAMS["energy_sensitivity"] * norm_energy * ddl_gate  # Energy preference only when feasible
        - wait_benefit                                 # Bounded linear anti-starvation (cap = PARAMS)
        + PARAMS["duration_robustness"] * duration_risk_score
        + PARAMS["energy_uncertainty_interaction"] * energy_uncert_penalty
        + PARAMS["slack_penalty_exponent"] * energy_slack_penalty
        + PARAMS["remaining_work_weight"] * norm_work
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
