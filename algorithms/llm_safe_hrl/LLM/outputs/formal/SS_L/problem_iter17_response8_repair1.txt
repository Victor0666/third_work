def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's DDL-protection gate and robust normalization with Parent 1's smooth energy ramp;
       introduces *dual-gated energy activation*: both DDL-feasibility (smooth Heaviside) AND marginal slack alignment (linear ramp);
       replaces brittle critical-path leverage with *normalized successor-unblocking potential* — product of upward_rank and normalized remaining_work,
       gated by slack pressure only (not just breach), improving upstream prioritization under tight-but-feasible deadlines;
       uses median-MAD normalization throughout for outlier resilience;
       all terms clipped to [-2,2] for stability and bounded AST depth."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Median-MAD normalization per feature — outlier-resilient, sign-preserving ===
    def median_mad_normalize(x):
        x = np.copy(x)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad if mad > eps else eps
        return (x - med) / spread
    
    norm_slack = median_mad_normalize(slack)
    norm_energy = median_mad_normalize(min_incremental_energy)
    norm_duration = median_mad_normalize(min_exec_time + min_comm_time)
    norm_rank = median_mad_normalize(upward_rank)
    norm_work = median_mad_normalize(remaining_work)
    norm_wait = median_mad_normalize(ready_wait_time)
    norm_uncert = median_mad_normalize(uncertainty)
    
    # === Dual-gated energy activation ===
    # Gate 1: DDL-feasibility — smooth Heaviside (Parent 2 style)
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    # Gate 2: Marginal slack alignment — linear ramp around threshold (Parent 1 style)
    # Reuse existing parameters: no new param needed — center at 0, width from ddl_protection_gate_slope inverse scaling
    energy_ramp_center = 0.0
    energy_ramp_width = 1.0 / (PARAMS["ddl_protection_gate_slope"] + eps)
    energy_ramp = np.clip(
        (slack - energy_ramp_center) / (energy_ramp_width + eps),
        0.0,
        1.0
    )
    # Combined energy enablement: both conditions must support optimization
    energy_activation = ddl_gate * energy_ramp
    
    # === Sharpened slack-pressure gate for criticality modulation ===
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    rank_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    
    # === Successor-unblocking potential: upward_rank * remaining_work, activated by slack pressure (not breach-only) ===
    # Captures unblocking value of high-criticality tasks with large downstream scope — even when slack > 0 but tight
    unblocking_potential = norm_rank * norm_work * rank_gate
    
    # === DDL-protection gated duration robustness ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * norm_uncert * rank_gate * ddl_gate
    
    # === Starvation relief with saturation offset ===
    wait_benefit = 1.0 - np.exp(-PARAMS["wait_decay"] * (ready_wait_time + PARAMS["wait_saturation_offset"]))
    
    # === Energy-uncertainty coupling: dual-gated ===
    energy_uncert_penalty = norm_energy * norm_uncert * energy_activation
    
    # === Slack-aware energy scaling: enhances penalty near deadline margin ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * energy_activation
    
    # === Raw slack penalty (dominant term) ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = median_mad_normalize(raw_slack_penalty)
    
    # === Composite score: smaller = higher priority ===
    # Prioritizes: (1) lateness avoidance, (2) critical-path unblocking, (3) energy minimization only when safe & beneficial,
    # (4) anti-starvation, (5) uncertainty-aware robustness
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Dominant lateness penalty
        - np.clip(unblocking_potential, -2.0, 2.0)                        # Critical unblocking under pressure (not breach-only)
        - np.clip(PARAMS["energy_sensitivity"] * norm_energy * energy_activation, -2.0, 2.0)  # Dual-gated energy preference
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
