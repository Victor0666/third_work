def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule: merges Parent 2's proven DDL-protection gate and power-law slack penalty with Parent 1's robust median-MAD normalization;
       introduces explicit critical_path_leverage_weight to decouple downstream impact from rank amplification;
       replaces logistic starvation relief with bounded inverse-time form for monotonic, low-latency sensitivity;
       adds slack-aware duration penalty using clipped slack-pressure to avoid over-penalizing long-duration tasks under feasibility;
       all terms clipped to [-2,2] and safeguarded against inf/nan."""
    
    eps = PARAMS["epsilon"]
    
    # Cast inputs safely; avoid in-place mutation
    min_exec_time = np.asarray(min_exec_time, dtype=float)
    min_comm_time = np.asarray(min_comm_time, dtype=float)
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float)
    slack = np.asarray(slack, dtype=float)
    upward_rank = np.asarray(upward_rank, dtype=float)
    remaining_work = np.asarray(remaining_work, dtype=float)
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    uncertainty = np.asarray(uncertainty, dtype=float)
    
    N = len(min_exec_time)
    
    # === Robust median-MAD normalization: sign-preserving, outlier-resilient scaling ===
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        if N == 1:
            med = x[0]
            mad = eps
        else:
            med = np.median(x)
            mad = np.median(np.abs(x - med))
        spread = mad + eps
        return (x - med) / spread
    
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === DDL-protection gate: smooth Heaviside enabling energy/uncertainty only when slack >= 0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * slack))
    
    # === Critical path leverage: explicit rank × work × ddl_breach term with dedicated weight ===
    # Decouples downstream impact modeling from rank coupling — improves interpretability and CMA-ES convergence
    ddl_breach = (slack <= 0.0).astype(float)
    critical_path_leverage = norm_rank * norm_work * ddl_breach
    
    # === Risk-adjusted slack penalty: power law on raw negative slack, then normalized ===
    raw_slack_penalty = np.maximum(-slack, 0.0) ** PARAMS["slack_penalty_exponent"]
    norm_slack_penalty = robust_normalize(raw_slack_penalty)
    
    # === Slack-pressure gate: bounded sigmoid activation centered at zero slack ===
    # Maps slack ∈ [-∞, ∞] → pressure ∈ [0,1], steep near zero to sharpen deadline transitions
    slack_pressure = np.clip(-norm_slack, 0.0, 2.0)
    pressure_gate = 1.0 / (1.0 + np.exp(-PARAMS["slack_pressure_gate_steepness"] * (slack_pressure - 1.0)))
    
    # === Bounded linear rank-slack coupling: stable monotonic amplification under pressure ===
    coupled_rank = norm_rank * (1.0 + PARAMS["criticality_scale"] * pressure_gate)
    
    # === Uncertainty-duration interaction: dual-gated (uncertainty threshold + slack pressure + DDL feasibility) ===
    uncert_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate_slope"] * (norm_uncert - PARAMS["uncertainty_gate_threshold"])))
    duration_risk_score = norm_duration * uncert_gate * slack_pressure * ddl_gate
    
    # === Starvation relief: bounded inverse-time form — monotonic, avoids log singularities, preserves low-wait sensitivity ===
    wait_benefit = np.clip(1.0 / (1.0 + np.exp(-PARAMS["wait_decay"] * (ready_wait_time + eps))), 0.0, 1.0)
    
    # === Energy-uncertainty coupling: gated by both uncertainty and DDL feasibility ===
    energy_uncert_penalty = norm_energy * norm_uncert * uncert_gate * ddl_gate
    
    # === Slack-aware energy scaling: amplifies energy penalty only when slack is tight AND feasible ===
    energy_slack_penalty = norm_energy * (1.0 + PARAMS["slack_penalty_exponent"] * slack_pressure) * ddl_gate
    
    # === Composite score: smaller = higher priority; all intermediate terms clipped to [-2,2] for stability ===
    score = (
        + np.clip(norm_slack_penalty, -2.0, 2.0)                           # Primary lateness penalty
        - PARAMS["critical_path_leverage_weight"] * np.clip(critical_path_leverage, -2.0, 2.0)  # Explicit downstream impact
        - np.clip(coupled_rank, -2.0, 2.0)                                  # HEFT-based urgency under pressure
        - PARAMS["energy_sensitivity"] * np.clip(norm_energy * ddl_gate, -2.0, 2.0)  # Energy preference only when feasible
        - np.clip(wait_benefit, -2.0, 2.0)                                 # Anti-starvation (bounded inverse-time)
        + PARAMS["duration_robustness"] * np.clip(duration_risk_score, -2.0, 2.0)
        + PARAMS["energy_uncertainty_interaction"] * np.clip(energy_uncert_penalty, -2.0, 2.0)
        + PARAMS["slack_penalty_exponent"] * np.clip(energy_slack_penalty, -2.0, 2.0)
        + PARAMS["remaining_work_weight"] * np.clip(norm_work, -2.0, 2.0)
    )
    
    # Final safeguard: ensure finite, deterministic output and shape (N,)
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape (N,) explicitly — required for N=1 case
    return score.reshape(-1)
