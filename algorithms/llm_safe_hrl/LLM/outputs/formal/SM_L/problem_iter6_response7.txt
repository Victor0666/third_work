def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with smooth DDL feasibility gating, unified joint normalization,
    and eliminated percentile parameter redundancy.
    
    Key structural improvements:
      - Replaced hard `slk >= 0` gate with smooth logistic feasibility weight: 
        `sigma(slack * slack_feasibility_weight)` — avoids discontinuities near slack=0.
      - Unified joint normalization: all core features (duration, energy, rank, successor_load) 
        are normalized together using a single robust scale from the 90th quantile of their
        concatenated absolute values — improves stability and reduces parameter coupling.
      - Removed `pctl_low`/`pctl_high` entirely; fixed lower bound at 10th implicitly via median-centered
        scaling on quantile-range — reallocates DOF to structural smoothing.
      - All components now share identical normalization context, ensuring consistent relative weighting.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    
    # Duration: execution + communication
    duration = exec_t + comm_t
    
    # === Smooth DDL feasibility weight: logistic transition centered at slack=0 ===
    # Maps slack → [0,1] with steepness controlled by slack_feasibility_weight
    feasibility_weight = 1.0 / (1.0 + np.exp(-PARAMS["slack_feasibility_weight"] * slk))
    
    # === Joint robust normalization over key features: duration, energy, rank, successor_load ===
    # Concatenate absolute values to compute unified scale (robust to outliers and zero-mass cases)
    joint_features = np.abs(np.concatenate([
        duration + eps,
        energy + eps,
        rank + eps,
        rank * (work + eps) + eps
    ]))
    # Use 90th quantile as upper bound; lower bound assumed ~10th via centering logic
    q90 = np.percentile(joint_features, PARAMS["joint_norm_quantile"])
    rng_safe = np.where(np.isfinite(q90) & (q90 > eps), q90, eps)
    
    def joint_normalize(x):
        x = np.asarray(x)
        # Center around median of joint_features for symmetry
        center = np.median(joint_features)
        return (x - center) / (PARAMS["percentile_range_scale"] * rng_safe + eps)
    
    # === Slack processing: piecewise & risk-aware ===
    slack_norm = joint_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_risk_penalty"] * 0.3 * np.abs(slack_norm)  # softer deprioritization for positive slack
    )
    
    # === Energy efficiency: inverse scaling, weighted by feasibility ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * feasibility_weight * joint_normalize(inv_energy + eps)
    
    # === Criticality: smooth feasibility-weighted, not gated ===
    rank_score = -PARAMS["criticality_weight"] * feasibility_weight * joint_normalize(rank + eps)
    
    # === Joint duration-energy urgency: captures cost of delaying efficient tasks ===
    dur_energy_prod = (duration + eps) * (energy + eps)
    dur_energy_urgency = np.power(dur_energy_prod, PARAMS["duration_energy_coupling_power"])
    dur_energy_score = feasibility_weight * joint_normalize(dur_energy_urgency + eps)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -joint_normalize(wait_sat + eps)
    
    # === Successor-release load mitigation: reward unblocking of high-criticality descendants ===
    weighted_successor_load = rank * (work + eps)
    successor_release_score = -PARAMS["successor_release_weight"] * feasibility_weight * joint_normalize(weighted_successor_load + eps)
    
    # === Assemble final score: smaller = better ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_energy_score
        + wait_score
        + successor_release_score
    )
    
    # Final safeguard: replace NaN/inf with finite fallback using machine epsilon bounds
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
