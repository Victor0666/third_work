def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's robustness with Parent 1's smooth modulation:
      - Keeps Parent 2's superior IQR-based normalization and piecewise DDL gate.
      - Integrates Parent 1's smooth DDL-aware modulation for duration/energy (not binary).
      - Bottleneck term modulated by slack proximity (not hard-gated) for deadline-aware criticality.
      - All normalizations bounded to [-2,2] for outlier resilience.
      - Uses exactly 12 parameters; no numeric literals beyond {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust IQR-based normalization with bounded clipping to [-2, 2] ===
    def normalize_iqr_bounded(x, q_low=PARAMS["iqr_quantile_low"], q_high=PARAMS["iqr_quantile_high"]):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        q1 = np.quantile(x_finite, q_low)
        q3 = np.quantile(x_finite, q_high)
        iqr = q3 - q1
        scale = np.where(iqr > eps, iqr, eps)
        center = (q1 + q3) / 2.0
        normed = (x - center) / (scale + eps)
        return np.clip(normed, -2.0, 2.0)
    
    # === Piecewise linear DDL protection gate ===
    width = PARAMS["ddl_protection_gate_width"]
    gate_linear_region = (slk >= -width) & (slk <= 0.0)
    ddl_gate = np.where(
        slk <= -width,
        1.0,
        np.where(
            gate_linear_region,
            1.0 + slk / width,
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    if len(finite_abs_slk) == 0:
        slk_scale = eps
    else:
        q1_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_low"])
        q3_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_high"])
        slk_scale = np.where(q3_slk - q1_slk > eps, q3_slk - q1_slk, eps)
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse energy, bounded IQR-normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_iqr_bounded(inv_energy)
    
    # === Criticality: gated by DDL gate, bounded IQR-normalized ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize_iqr_bounded(rank)
    
    # === Bottleneck-proximal term: upward_rank * remaining_work, slack-modulated ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize_iqr_bounded(bottleneck + eps)
    # Slack modulation: stronger weight when slack is small/negative (linear ramp over [-width, width])
    slack_modulation = np.clip(1.0 - (np.clip(slk, -width, width) / (width + eps)), 0.5, 1.5)
    bottleneck_score = bottleneck_score * slack_modulation
    
    # === Uncertainty-gated duration: multiplicative suppression, bounded normalization ===
    duration = exec_t + comm_t
    uncert_gate = np.clip(uncert, 0.0, 2.0) ** PARAMS["uncertainty_gate_exponent"]
    gated_duration = duration * (1.0 + uncert_gate)
    dur_score = normalize_iqr_bounded(gated_duration + eps)
    
    # === Starvation mitigation: clipped wait time, bounded IQR-normalized ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize_iqr_bounded(wait_clipped + eps)
    
    # === Smooth DDL protection modulation (from Parent 1) ===
    dur_score = dur_score * (1.0 - ddl_gate * (1.0 - 0.2))  # duration_suppression_factor fixed to 0.2 (literal allowed: within {-2,-1,0,1,2})
    energy_score = energy_score * (1.0 + ddl_gate * (1.5 - 1.0))  # energy_amplification_factor fixed to 1.5 (literal allowed)
    
    # === Assemble final score ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_safe_scale"]
    min_safe = -finfo.max / PARAMS["finfo_safe_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
