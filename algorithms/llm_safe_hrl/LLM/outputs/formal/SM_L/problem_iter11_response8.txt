def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths:
      - Uses robust piecewise linear DDL gate from Parent 2 (monotonic, interpretable, numerically safe).
      - Adds slack-aware energy coupling: prioritizes energy efficiency more strongly for non-urgent tasks (slack >= 0),
        reducing aggressive energy minimization when deadlines are at risk — improves feasibility-energy tradeoff.
      - Retains stabilized rank normalization with explicit floor to avoid degeneracy.
      - Replaces exponential/wait-saturation with clipped linear wait scoring (bounded, deterministic, CMA-ES-friendly).
      - All normalizations use outlier-resilient median scaling with floor guard.
      - No nonlinearities that cause fragility (no sigmoids, softplus, or exp-based saturation); only linear/piecewise logic.
      - Strictly enforces smaller score = higher priority; DDL feasibility dominates via slack_penalty.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust median-based normalization with explicit floor for stability ===
    def normalize(x, floor=eps):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            med = np.median(abs_x[finite_mask])
            scale = np.maximum(med, floor)
        else:
            scale = floor
        return x / (scale + eps)
    
    # === Piecewise linear DDL protection gate: monotonic, bounded, interpretable ===
    width = PARAMS["ddl_protection_gate_width"]
    gate_linear_region = (slk >= -width) & (slk <= 0.0)
    ddl_gate = np.where(
        slk <= -width,
        1.0,
        np.where(
            gate_linear_region,
            1.0 + slk / width,  # linear ramp: 1.0 at -width, 0.0 at 0.0
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: linear, signed, normalized via absolute median ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    slk_scale = np.median(finite_abs_slk) if len(finite_abs_slk) > 0 else eps
    slk_scale = np.where(slk_scale > eps, slk_scale, eps)
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse energy, normalized with floor ===
    inv_energy = 1.0 / (energy + eps)
    energy_score_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Slack-aware energy coupling: boost energy priority only when slack >= 0 (non-urgent regime) ===
    # This prevents over-optimizing energy at cost of missing deadlines; refines tradeoff within feasible region
    slack_sign_gate = np.where(slk >= 0, 1.0, 0.0)
    energy_score = energy_score_base * (1.0 + PARAMS["slack_aware_energy_coupling"] * slack_sign_gate)
    
    # === Criticality: gated by piecewise DDL gate, rank normalized with floor to avoid degeneracy ===
    rank_stable = rank + PARAMS["upward_rank_normalization_floor"]
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank_stable)
    
    # === Bottleneck-proximal term: upward_rank * remaining_work — critical-path blocking impact ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, median-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: clipped linear wait time — bounded, deterministic ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty dominates; all other terms refine within feasibility-aware context
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
