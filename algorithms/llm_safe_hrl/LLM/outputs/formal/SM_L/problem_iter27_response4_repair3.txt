def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Bounded linear bottleneck modulation (replaces tanh) to avoid over-suppression of critical path under uncertainty.
      - Linear starvation saturation: min(wait / theta, 1.0) — avoids exponential over-penalization and aligns with DDL urgency.
      - Dual-gated energy score (DDL gate + exponential slack decay) preserved.
      - Robust slack normalization retained.
      - Successor-readiness penalty integrated via upward_rank * remaining_work, scaled by ddl_gate and tunable weight — merged into bottleneck_score to stay within 12 parameters.
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants.
      - Deterministic, finite, shape-(N,) output satisfying all interface contracts.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Max-abs normalization: robust, interpretable, and stable ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Robust slack normalization: |slack|^p / (|slack|^p + eps) preserves sign via multiplication by sign(slk) ===
    abs_slk = np.abs(slk) + eps
    robust_slk_mag = np.power(abs_slk, PARAMS["robust_slack_normalization"])
    robust_slk_norm = robust_slk_mag / (robust_slk_mag + eps)
    slk_robust = robust_slk_norm * np.sign(slk)
    
    # === Piecewise-linear DDL gate: applied on robustly normalized slack ===
    gate_width = PARAMS["piecewise_linear_gate_width"] + eps
    ddl_gate = np.where(
        slk_robust <= 0,
        1.0,
        np.where(
            slk_robust <= gate_width,
            1.0 - slk_robust / gate_width,
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: applied on robust slack ===
    slack_penalty = np.where(
        slk_robust < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk_robust),
        -PARAMS["slack_urgency_gain"] * slk_robust
    )
    
    # === Energy efficiency: inverse marginal energy, dual-gated ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    slack_decay = np.exp(-np.abs(slk_robust) * PARAMS["slack_aware_energy_decay"])
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate) * slack_decay
    
    # === Criticality: upward_rank raised to slack-aware power ===
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize(rank_powered)
    
    # === Bottleneck proximity + successor-readiness: rank * work / (|robust_slk| + eps), modulated by ddl_gate and bounded linear uncertainty gate ===
    robust_abs_slk = np.abs(slk_robust) + eps
    bottleneck_term = rank * work / robust_abs_slk
    # Merge successor-readiness: add rank * work (unnormalized) scaled by ddl_gate and bottleneck_weight
    successor_term = rank * work * ddl_gate
    combined_bottleneck = bottleneck_term + successor_term
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize(combined_bottleneck + eps)
    
    # === Bounded linear uncertainty modulation (replaces tanh): clamp [0,1] attenuation ===
    uncert_normalized = normalize(uncert + eps)
    uncert_gate = np.clip(1.0 - uncert_normalized, 0.0, 1.0)
    bottleneck_score = bottleneck_score * uncert_gate
    
    # === Duration-uncertainty coupling: bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: linear saturation — min(wait / theta, 1.0) ===
    wait_scaled = wait / (PARAMS["wait_saturation_time"] + eps)
    wait_score = -np.clip(wait_scaled, 0.0, 1.0)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2.0, neginf=finfo.min / 2.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
