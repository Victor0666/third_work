def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Piecewise adaptive energy gating: strong ddl_gate^exponent only when slack < 0.
      - Robust min-max normalization for bottleneck term (upward_rank * remaining_work) using configurable quantile.
      - Successor-release sharpening retained and correctly referenced via PARAMS["successor_release_sharpness"].
      - Deterministic clipping (-1e12, 1e12) instead of finfo-based scaling.
      - All numeric literals are in {-2,-1,0,1,2}; all tunables declared in PARAMETER_SCHEMA.
      - Exactly 12 parameters.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization (for most features) ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: piecewise adaptive gating — stronger suppression only when slack < 0 ===
    inv_energy = 1.0 / (energy + eps)
    # Apply strong gating (ddl_gate^exponent) only under deadline risk; preserve full sensitivity otherwise
    energy_gate = np.where(
        slk < 0,
        np.power(ddl_gate, PARAMS["energy_adaptive_gate_exponent"]),
        1.0
    )
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * energy_gate
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction with ROBUST MIN-MAX NORMALIZATION (not quantile) ===
    # Stabilizes bottleneck term across seeds: clamp to [min, q_upper] instead of scale by quantile
    bottleneck_raw = rank * work
    bottleneck_finite = bottleneck_raw[np.isfinite(bottleneck_raw)]
    if len(bottleneck_finite) > 0:
        b_min = np.min(bottleneck_finite)
        b_max = np.quantile(bottleneck_finite, PARAMS["bottleneck_minmax_clip_quantile"])
        b_range = np.where(b_max > b_min, b_max - b_min, eps)
        bottleneck_norm = (bottleneck_raw - b_min) / (b_range + eps)
        bottleneck_norm = np.clip(bottleneck_norm, 0.0, 1.0)  # unit interval
    else:
        bottleneck_norm = np.zeros_like(bottleneck_raw)
    
    # Sharpen with inverse slack magnitude using declared successor_release_sharpness
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = bottleneck_norm * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * bottleneck_sharpened
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear ramp instead of hard clip → monotonic & smooth ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: deterministic finite clipping (per reflection)
    score = np.clip(score, -1e12, 1e12)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
