import numpy as np

    "schema_version": 1,
    "parameters": [
        {
            "name": "epsilon",
            "initial_value": 1e-6,
            "lower_bound": 1e-9,
            "upper_bound": 1e-3,
            "semantic_description": "Numerical guard for safe normalization and division.",
            "parameter_type": "float",
            "transform": "log"
        },
        {
            "name": "slack_penalty_scale",
            "initial_value": 5.0,
            "lower_bound": 0.5,
            "upper_bound": 12.0,
            "semantic_description": "Amplification factor for negative slack (lateness risk).",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "slack_sensitivity",
            "initial_value": 1.8,
            "lower_bound": 0.1,
            "upper_bound": 4.0,
            "semantic_description": "Exponent controlling how sharply priority rises near zero slack.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "energy_norm_factor",
            "initial_value": 0.7,
            "lower_bound": 0.1,

            "parameter_type": "float",

            "parameter_type": "float",

            "initial_value": 0.05,

            "parameter_type": "float",

        },

            "initial_value": 0.9,
            "lower_bound": 0.0,

            "semantic_description": "Strength of multiplicative coupling between uncertainty and slack pressure.",
            "parameter_type": "float",

            "parameter_type": "float",

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: piecewise slack gating, IQR-normalized features, criticality-boosted rank,
       and uncertainty-aware interaction — all bounded, deterministic, and DDL-risk-first."""
    eps = PARAMS["epsilon"]
    
    # Safely cast inputs to float64, avoid in-place mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)

    # --- Robust normalization using interquartile range (IQR) to resist outliers ---
    def iqr_normalize(x):
        x = np.copy(x)
        q1, q3 = np.quantile(x, [0.25, 0.75])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr

    # Normalize only *finite* values; replace inf/nan with median before norm
    def safe_iqr_normalize(x):
        x_clean = np.where(np.isfinite(x), x, np.nanmedian(x))
        return iqr_normalize(x_clean)

    norm_energy = safe_iqr_normalize(energy)
    norm_exec_comm = safe_iqr_normalize(exec_t + comm_t)
    norm_rank = safe_iqr_normalize(rank)
    norm_work = safe_iqr_normalize(work)
    norm_wait = safe_iqr_normalize(wait)
    norm_uncert = safe_iqr_normalize(uncert)

    # --- Slack processing: prioritize negative slack strongly, smooth transition near zero ---
    # Use signed power: sign(slk) * |slk|^p → preserves direction & amplifies lateness
    # Clamp exponent to avoid overflow for large |slk|; use bounded sensitivity
    p = np.clip(PARAMS["slack_sensitivity"], 0.1, 4.0)
    signed_slack_power = np.sign(slk) * np.power(np.abs(slk) + eps, p)
    # Normalize the powered slack (still retains asymmetry: negative values dominate)
    norm_signed_slack = safe_iqr_normalize(signed_slack_power)

    # --- Lateness penalty: exponential-like but bounded and monotonic ---
    # For slk < 0: exp(|slk| * scale) → grows fast, but clipped to prevent overflow
    lateness_penalty = np.zeros_like(slk)
    neg_mask = slk < 0
    if np.any(neg_mask):
        abs_slk_neg = np.abs(slk[neg_mask])
        # Clip to avoid exp overflow: cap effective input at ~20 for float64 stability
        capped_input = np.clip(PARAMS["slack_penalty_scale"] * abs_slk_neg, 0.0, 20.0)
        lateness_penalty[neg_mask] = np.exp(capped_input) - 1.0  # zero at slk=0

    # --- Criticality boost: amplify upward_rank only under deadline pressure or high uncertainty ---
    # Gate: (|negative slack| + uncertainty) scaled and sigmoid-bounded → [0,1]
    pressure_score = np.clip(
        (np.abs(slk) * (slk < 0) + uncert) * PARAMS["uncertainty_slack_interaction"],
        0.0, 10.0
    )
    gate = 1.0 / (1.0 + np.exp(-pressure_score + 2.0))  # soft gate centered at pressure=2.0
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_boost"] * gate)

    # --- Anti-starvation: saturating wait bonus (log-like growth, bounded)
    # Prevents runaway priority from long waits; uses stable log1p
    wait_bonus = np.log1p(wait * PARAMS["wait_decay_rate"])  # grows slowly, finite
    norm_wait_bonus = safe_iqr_normalize(wait_bonus)

    # --- Duration balance: blend exec+comm into a single normalized driver ---
    duration_score = norm_exec_comm * PARAMS["duration_balance"]

    # --- Final score: smaller = higher priority ---
    # Order of terms emphasizes DDL risk first (lateness_penalty dominates),
    # then energy efficiency, then criticality & starvation mitigation.
    score = (
        lateness_penalty  # Dominant term: hard penalty for negative slack → pulls score up → lowers priority
        + PARAMS["energy_norm_factor"] * norm_energy
        + duration_score
        - boosted_rank  # Higher rank → lower score → higher priority
        - norm_wait_bonus  # Longer wait → lower score → higher priority
        + norm_signed_slack  # Negative slack → negative norm_signed_slack → lowers score → boosts priority
    )

    # Ensure finite output: replace NaN/inf with safe finite fallback (e.g., median)
    score = np.nan_to_num(score, nan=np.nanmedian(score), posinf=np.nanmax(score), neginf=np.nanmin(score))

    # Final safeguard: ensure shape is exactly (N,)
    return score.astype(np.float64, copy=False)
