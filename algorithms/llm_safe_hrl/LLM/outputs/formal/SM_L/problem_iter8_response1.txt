def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's smooth feasibility blending and joint normalization
    with Parent 1's bottleneck-aware successor release and uncertainty-slack coupling.
    
    Key novel improvements:
      - Added `bottleneck_boost_exponent` to amplify upward_rank * remaining_work under deadline stress,
        enhancing focus on high-impact blocking tasks.
      - Introduced `uncertainty_slack_coupling`: multiplies uncertainty by (1 - feasibility_mask)
        to selectively increase risk-awareness precisely when slack < 0 — avoids diluting urgency.
      - Retained smooth sigmoid feasibility blending for gradient-aware optimization.
      - Joint percentile normalization over [slack, upward_rank, remaining_work] ensures stable
        bottleneck interaction scaling.
      - All normalizations use identical percentile-based logic for consistency and outlier resilience.
      - Final clipping is symmetric and tunable via `score_clip_bound`.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Safe casting without in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Joint cross-feature normalization: aligns slack, rank, and work into shared scale
    joint_features = np.stack([slk, rank, work], axis=0)
    p_low_joint = np.percentile(joint_features, PARAMS["pctl_low"], axis=1)
    p_high_joint = np.percentile(joint_features, PARAMS["pctl_high"], axis=1)
    rng_joint = p_high_joint - p_low_joint
    rng_safe_joint = np.where(np.isfinite(rng_joint) & (rng_joint > eps), rng_joint, eps)
    center_joint = (p_low_joint + p_high_joint) / 2.0
    
    # Normalize each feature using joint statistics
    slk_norm = (slk - center_joint[0]) / (rng_safe_joint[0] + eps)
    rank_norm = (rank - center_joint[1]) / (rng_safe_joint[1] + eps)
    work_norm = (work - center_joint[2]) / (rng_safe_joint[2] + eps)
    
    # Robust percentile-based normalization for scalar features
    def percentile_normalize(x):
        x = np.asarray(x)
        p_low = np.percentile(x, PARAMS["pctl_low"])
        p_high = np.percentile(x, PARAMS["pctl_high"])
        rng = p_high - p_low
        rng_safe = np.where(np.isfinite(rng) & (rng > eps), rng, eps)
        center = (p_low + p_high) / 2.0
        return (x - center) / (rng_safe + eps)
    
    # Duration = execution + communication
    duration = exec_t + comm_t
    
    # === Smooth feasibility blending via sigmoid ===
    feasibility_mask = 1.0 / (1.0 + np.exp(-PARAMS["feasibility_blend_steepness"] * slk))
    
    # === Slack processing: quadratic penalty for lateness, linear gain for margin ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (np.abs(slk_norm) ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slk_norm)
    )
    
    # === Energy efficiency: smoothly blended — zero weight when slack << 0, full weight when slack >> 0 ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * percentile_normalize(inv_energy + eps)
    energy_score = feasibility_mask * energy_score
    
    # === Criticality: same smooth blending — active only in feasible region ===
    rank_score = -PARAMS["criticality_weight"] * percentile_normalize(rank_norm + eps)
    rank_score = feasibility_mask * rank_score
    
    # === Bottleneck-aware successor release: exponentiated interaction under deadline stress ===
    # Amplifies impact of high-rank + high-work tasks when slack < 0
    bottleneck_interaction = (rank_norm * work_norm) ** PARAMS["bottleneck_boost_exponent"]
    urgency_signal = np.clip(-slk, 0.0, np.inf)
    urgency_norm = percentile_normalize(urgency_signal + eps)
    # Coupled exclusively to latency-critical regime
    bottleneck_score = -PARAMS["successor_release_weight"] * bottleneck_interaction * urgency_norm * (1.0 - feasibility_mask)
    
    # === Uncertainty-slack coupling: enhances uncertainty impact only under deadline risk ===
    # Avoids diluting urgency with baseline uncertainty; focuses on risk amplification
    coupled_uncert = uncert * (1.0 - feasibility_mask) * PARAMS["uncertainty_slack_coupling"]
    coupled_uncert_norm = percentile_normalize(coupled_uncert + eps)
    
    # === Duration-uncertainty coupling: bounded harmonic blend with coupled uncertainty ===
    dur_norm = percentile_normalize(duration + eps)
    denom = dur_norm + PARAMS["duration_uncertainty_ratio"] * (uncert_norm + coupled_uncert_norm) + eps
    dur_uncert_blend = (1.0 + PARAMS["duration_uncertainty_ratio"] * (uncert_norm + coupled_uncert_norm)) / denom
    dur_score = percentile_normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -percentile_normalize(wait_sat + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: symmetric clip to bounded finite range
    score = np.clip(score, -PARAMS["score_clip_bound"], PARAMS["score_clip_bound"])
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
