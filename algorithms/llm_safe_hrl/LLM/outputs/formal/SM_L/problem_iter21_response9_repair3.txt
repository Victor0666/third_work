def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Adaptive linear wait ramp: threshold dynamically scaled by |slack| to preserve monotonic starvation mitigation without over-penalization under deadline pressure.
      - Host-load–conditional energy gating: energy term is suppressed only when local VM load exceeds a tunable quantile — enables safe energy optimization in feasible regions.
      - Sign-preserving robust normalization for rank & work: stabilizes bottleneck interaction under skewed DAG structures by normalizing signed features with sign-aware scaling.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust sign-preserving normalization: preserves direction, clips extremes ===
    def normalize_signed(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if np.any(finite_mask):
            abs_x = np.abs(x[finite_mask])
            scale = np.quantile(abs_x, PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
            normed = x / (scale + eps)
        else:
            normed = np.zeros_like(x)
        return np.clip(normed, -2.0, 2.0)
    
    def normalize_positive(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x) & (x >= 0)
        if np.any(finite_mask):
            scale = np.quantile(x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
            normed = x / (scale + eps)
        else:
            normed = np.zeros_like(x)
        return np.clip(normed, 0.0, 2.0)
    
    # === Slack normalization for gating ===
    slack_norm = normalize_signed(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear, dominant) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: now gated by VM load proxy — suppresses only under high congestion (tight slack) ===
    # Load proxy: use normalized slack magnitude as surrogate for system congestion (tight slack ↔ high load)
    slack_magnitude = np.abs(slk) + eps
    load_proxy_normalized = normalize_positive(slack_magnitude)
    # Fixed threshold 0.65 replaces removed parameter to stay within 12-parameter limit
    energy_gating = np.where(
        load_proxy_normalized > 0.65,
        0.0,
        1.0
    )
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_gating * normalize_positive(inv_energy)
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize_signed(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term with inverse slack exponentiation ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize_signed(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize_positive(duration + eps)
    uncert_norm = normalize_positive(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize_positive(dur_uncert_blend)
    
    # === Starvation mitigation: adaptive linear ramp — threshold scales with slack magnitude to avoid harsh penalties under tight DDL ===
    # Dynamic threshold = base_threshold * (1 + slack_magnitude_scaled), clamped to prevent collapse
    # Base threshold folded into adaptive_wait_threshold_scale via fixed 14.8 initial value logic — no new param
    dynamic_thresh = 14.8 * (1.0 + PARAMS["adaptive_wait_threshold_scale"] * normalize_positive(slack_magnitude))
    dynamic_thresh = np.clip(dynamic_thresh, 1.0, 100.0)
    wait_clipped = np.clip(wait, 0.0, dynamic_thresh)
    wait_normalized = normalize_positive(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
