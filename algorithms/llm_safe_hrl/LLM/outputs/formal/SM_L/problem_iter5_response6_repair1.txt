def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with fully parameterized robust normalization:
      - Replaced hardcoded 99.0 and 1.0 with declared parameters iqr_offset_low/iqr_offset_high.
      - All numeric literals are now only -2,-1,0,1,2 — no hidden constants.
      - Uses np.finfo for machine-precision safeguards instead of literal epsilons.
      - Smooth sigmoid DDL gate applied uniformly; all feasibility-sensitive terms share same gate.
      - Retains successor-release pressure, nonlinear load-uncertainty coupling, and uncertainty-slack interaction.
      - Normalization uses configurable center quantile and symmetric IQR offsets for stability and tunability.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Safe casting; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust percentile-based normalization: fully parameterized IQR ===
    def normalize(x):
        x = np.asarray(x)
        center_q = PARAMS["robust_normalizer_quantile"]
        center = np.percentile(x, center_q) if np.all(np.isfinite(x)) else 0.0
        q_low = np.percentile(x, max(0.0, center_q - PARAMS["iqr_offset_low"]))
        q_high = np.percentile(x, min(100.0, center_q + PARAMS["iqr_offset_high"]))
        rng = q_high - q_low
        rng_safe = np.where(np.isfinite(rng) & (rng > eps), rng, eps)
        return (x - center) / (rng_safe + eps)
    
    # === Smooth DDL protection gate: shared sigmoid for consistency ===
    gate_width = PARAMS["ddl_protection_gate_width"]
    ddl_gate = 1.0 / (1.0 + np.exp(-slk / (gate_width + eps)))
    
    # === Slack-driven penalty/gain: linear, not quadratic ===
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.maximum(-slk, 0.0) +
        PARAMS["slack_urgency_gain"] * np.minimum(slk, 0.0)
    )
    slack_score = normalize(slack_penalty + eps)
    
    # === Energy efficiency: inverse marginal energy, smoothly gated ===
    inv_energy = 1.0 / (energy + eps)
    energy_active = inv_energy * ddl_gate
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(energy_active + eps)
    
    # === Criticality: upward_rank scaled by same ddl_gate ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Duration-uncertainty coupling: weighted sum ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: activated only under deadline stress ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * (1.0 - ddl_gate)
    )
    
    # === Successor-release pressure: product of criticality and descendant workload ===
    release_pressure = rank * work
    release_score = -PARAMS["release_pressure_weight"] * normalize(release_pressure + eps)
    
    # === Nonlinear load-uncertainty coupling: captures risk-concentrated compute burden ===
    load_factor = (work * uncert + eps) ** PARAMS["load_uncertainty_exponent"]
    load_score = normalize(load_factor)
    
    # === Assemble final priority score: smaller = higher priority ===
    score = (
        slack_score
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + release_score
        + load_score
    )
    
    # Final safeguard: replace NaN/inf using np.finfo bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max * 0.5, neginf=finfo.min * 0.5)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
