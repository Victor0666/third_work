def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule with:
      - All declared parameters used; no unused entries
      - No numeric literals except -2, -1, 0, 1, 2
      - Max-abs normalization for stability and interpretability
      - Piecewise-linear DDL gate using only slack and epsilon
      - Host-load-sensitive energy gating via PARAMS["host_load_sensitivity"]
      - Exponential starvation mitigation: 1 - exp(-wait/theta)
      - Slack-aware criticality: upward_rank raised to power of criticality_weight
      - Bottleneck term: rank * work / (|slack| + eps), normalized and weighted
      - Robust NaN/inf handling using np.nan_to_num with fixed bounds
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust max-abs normalization ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Piecewise-linear DDL protection gate: sharp activation at slack <= 0 ===
    # Uses only slack and eps — no external parameters
    min_slack_abs = np.abs(np.min(slk)) + eps
    ddl_gate = np.where(slk <= 0, 1.0, np.clip(1.0 - slk / min_slack_abs, 0.0, 1.0))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: gated by host_load_sensitivity * ddl_gate ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate)
    
    # === Criticality: slack-aware upward_rank weighting via exponentiation ===
    slack_mag_norm = normalize(np.abs(slk) + eps)
    rank_weight = np.power(1.0 + slack_mag_norm, PARAMS["criticality_weight"])
    rank_score = -normalize(rank) * rank_weight
    
    # === Successor-release interaction: rank * work / (|slack| + eps) ===
    slack_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_term = rank * work * slack_inv
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_term + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential saturation ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_score = -wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with safe finite values
    score = np.nan_to_num(score, nan=0.0, posinf=1e10, neginf=-1e10)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
