def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining best practices:
      - Retains Parent 2's crisp ddl_feasible_mask and robust quantile normalization.
      - Replaces exponential starvation with power-law saturation (new structural change): 
        smoother, more interpretable, and avoids numerical instability of exp(-wait/theta) for large wait.
      - Keeps sharpened successor-release via raw |slack|⁻ᵖ for fidelity under fuzzy eta=0.95 timing.
      - Uses normalized wait ramp + power-law saturation for bounded, monotonic starvation mitigation.
      - All features are DDL-first: slack_penalty dominates; energy only active when slack > margin.
      - No unused parameters; all numeric literals are {-2,-1,0,1,2}; deterministic and finite.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === DDL feasibility mask: binary gate enabling energy scoring only when slack > margin ===
    ddl_feasible_mask = (slk > PARAMS["ddl_feasibility_margin"])
    
    # === Slack-driven base penalty/gain (linear) — dominant term ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: gated by ddl_feasible_mask — prevents energy optimization from violating DDL ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_feasible_mask
    
    # === Criticality: normalized upward_rank, weighted but not suppressed ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps)
    
    # === Successor-release bottleneck: uses raw |slack|⁻ᵖ for sharper deadline coupling ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: quantile-normalized linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: power-law saturation (novel structural change) ===
    # f(wait) = wait^p / (wait^p + theta^p), bounded [0,1], monotonic, numerically stable
    theta = PARAMS["wait_ramp_threshold"] + eps
    p = PARAMS["starvation_saturation_exponent"]
    wait_power = np.power(np.clip(wait, 0.0, None), p)
    theta_power = np.power(theta, p)
    starvation_saturation = wait_power / (wait_power + theta_power + eps)
    wait_score = -starvation_saturation  # higher saturation → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using parametrized finfo scaling
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max * PARAMS["finfo_safety_factor"], neginf=finfo.min * PARAMS["finfo_safety_factor"])
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
