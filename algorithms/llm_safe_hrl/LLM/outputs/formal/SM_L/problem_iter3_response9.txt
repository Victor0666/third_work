def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
    1. Percentile-based normalization (10th/90th) to suppress outlier distortion in duration & uncertainty.
    2. Strict criticality gate: upward_rank only active when slack > 0 (not ≥ -ε), enforcing hard DDL feasibility first.
    3. Successor-release pressure term: bounded interaction between local slack and descendant remaining_work,
       capturing cascade risk — i.e., 'if my successors are heavy AND tight, I must go now'.
    All operations use np.finfo guards, avoid NaN/inf, and guarantee finite (N,) output.
    Smaller score = higher priority.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs to float64 without mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust percentile-based normalization: resistant to outliers ===
    def percentile_normalize(x):
        x = np.asarray(x)
        p10 = np.percentile(x, 10)
        p90 = np.percentile(x, 90)
        scale = p90 - p10
        scale = max(scale, eps)
        center = (p10 + p90) / 2.0
        return (x - center) / scale
    
    # === Duration & uncertainty coupling with percentile normalization ===
    duration = exec_t + comm_t
    dur_norm = percentile_normalize(duration + eps)
    uncert_norm = percentile_normalize(uncert + eps)
    # Harmonic blend: stable under extreme values
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * uncert_norm) 
        / (np.abs(dur_norm) + eps + PARAMS["duration_uncertainty_ratio"] * np.abs(uncert_norm) + eps)
    )
    # Fairness: power-law penalty on |dur_norm| to discourage bias toward ultra-short/long durations
    fairness_term = np.sign(dur_norm) * np.abs(dur_norm) ** PARAMS["fairness_exponent"]
    
    # === Slack processing: piecewise with strict DDL-first logic ===
    slack_norm = percentile_normalize(slk + eps)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (np.abs(slack_norm) ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse with clipping and percentile normalization ===
    inv_energy = 1.0 / (np.clip(energy, eps, None) + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * percentile_normalize(inv_energy)
    
    # === Strict criticality gating: activate ONLY when slack > 0 (hard feasibility barrier) ===
    rank_active = np.where(slk > 0, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * percentile_normalize(rank_active + eps)
    
    # === Wait-time mitigation: exponential saturation, normalized ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -percentile_normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: only under deadline stress (slk < 0) ===
    slack_stress = np.where(slk < 0, np.abs(slack_norm), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * uncert_norm * slack_stress
    
    # === Successor-release pressure: bounded min-max interaction capturing cascade risk ===
    # High pressure = high descendant work AND low descendant slack → prioritize now to unblock
    # Use median slack/work across all ready tasks as robust proxy for descendant conditions
    # Since true descendant slack isn't provided, we approximate pressure as local slack * local work,
    # but clamp and normalize to prevent dominance
    pressure_raw = np.clip(slk * work, -1e6, 1e6)  # Bounded interaction
    pressure_norm = percentile_normalize(pressure_raw + eps)
    successor_pressure = PARAMS["successor_pressure_coeff"] * np.where(slk < 0, pressure_norm, -pressure_norm)
    
    # === Assemble final score: smaller = better ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + percentile_normalize(fairness_term)
        + wait_score
        + unc_slack_interaction
        + successor_pressure
    )
    
    # Final safeguard: replace NaN/inf using machine limits; fallback to median if needed
    finfo = np.finfo(float)
    score = np.nan_to_num(
        score,
        nan=np.median(score) if np.isfinite(np.median(score)) else 0.0,
        posinf=finfo.max,
        neginf=finfo.min
    )
    
    # Ensure shape and finiteness
    assert score.shape == (N,), f"Expected (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values in final priority score"
    
    return score
