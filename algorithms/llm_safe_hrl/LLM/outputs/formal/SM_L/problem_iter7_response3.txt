def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with:
      - Hard wait-time clipping instead of exponential decay (more stable & interpretable)
      - Bottleneck-proximal term: upward_rank * remaining_work → identifies high-impact critical-path bottlenecks
      - Smooth sigmoid DDL protection gate → softly suppresses non-urgent tasks without hard thresholds
      - Removed inactive uncertainty-slack interaction per evidence; replaced with robust bottleneck term
      - All normalizations use median-based scaling (more outlier-resilient than mean-abs)
      - Strict adherence to feasibility-first: no energy terms dominate negative-slack tasks
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: uses median to resist outliers, guards against zero-scale ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        # Use median instead of mean for outlier resilience per design insight
        scale = np.median(abs_x[np.isfinite(abs_x)]) if np.any(np.isfinite(abs_x)) else eps
        scale = np.where(scale > eps, scale, eps)
        return x / (scale + eps)
    
    # === Slack normalization for gating — same as before but median-scaled ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0, full suppression for large positive slack ===
    # Gate ∈ [0, 1]; ~1 near zero/negative slack, decays smoothly for positive slack → ensures DDL-critical tasks always win
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear, no quadratic) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling with median-normalized inv-energy ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: now gated by ddl_gate (not just slack>=0), enabling smooth transition ===
    # This replaces hard gating with continuous feasibility-aware weighting
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Bottleneck-proximal term: upward_rank * remaining_work → identifies tasks whose delay blocks much work ===
    # Strongly prioritizes tasks that are both high-rank AND have large descendant workload
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck + eps)
    
    # === Duration-uncertainty coupling: same bounded linear blend, median-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: clipped linear wait time → bounded, deterministic, no exponentials ===
    # Replaces fragile exponential decay per design insight ("bounded linear clipping")
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # All components contribute additively; ddl_gate ensures slack_penalty dominates feasibility decisions
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values scaled by finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
