import numpy as np

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule: DDL-hard constraint first, then energy minimization.
    
    Key structural improvements:
      - Criticality uses smooth slack-normalized scaling: `rank * clip(slk / slack_margin, 0, 1)`.
      - Duration term is linearly gated by normalized slack (0 when slack <= 0), not power-law.
      - All terms clipped individually to [-1e6, 1e6] — no `nan_to_num` scaling artifacts.
      - Uncertainty-slack interaction strictly zero unless slack < 0 → enforces hard DDL-first policy.
      - Robust median-based slack margin with fallback; no mean-sensitive outliers.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Safe casting; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust slack margin estimation: median of positive slack, with fallback ===
    pos_slack = slk[slk > eps]
    slack_margin = np.median(pos_slack) if len(pos_slack) > 0 else PARAMS["median_slack_fallback"]
    slack_margin = max(slack_margin, eps)
    
    # === Slack processing: quadratic penalty for negative, linear gain for positive ===
    norm_slk = slk / (slack_margin + eps)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * norm_slk ** 2,
        -PARAMS["slack_urgency_gain"] * np.abs(norm_slk)
    )
    
    # === Criticality: smooth slack-normalized scaling (0 when slack <= 0, linear up to 1) ===
    scaled_rank = rank * np.clip(norm_slk, 0.0, 1.0)
    rank_score = -PARAMS["criticality_weight"] * scaled_rank
    
    # === Energy efficiency: raw inverse marginal energy → preserves relative discrimination ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * inv_energy
    
    # === Duration term: linearly gated by normalized slack (0 when slack <= 0) ===
    duration = exec_t + comm_t
    duration_gate = np.clip(norm_slk, 0.0, 1.0)
    duration_term = duration_gate * duration
    dur_mean = np.mean(np.abs(duration_term)) + eps
    duration_score = duration_term / dur_mean
    
    # === Uncertainty-slack interaction: strictly zero unless slack < 0 (DDL-first enforcement) ===
    unc_slack_interaction = np.where(
        slk < 0,
        PARAMS["uncertainty_slack_interaction"] * uncert * np.abs(norm_slk),
        0.0
    )
    
    # === Wait-time saturation: exponential anti-starvation with soft cap ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_saturation_rate"] * wait)
    wait_score = -wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    # Clip each term individually to avoid inf/NaN propagation and artificial scaling
    score = (
        np.clip(slack_penalty, -1e6, 1e6) +
        np.clip(energy_score, -1e6, 1e6) +
        np.clip(rank_score, -1e6, 1e6) +
        np.clip(duration_score, -1e6, 1e6) +
        np.clip(wait_score, -1e6, 1e6) +
        np.clip(unc_slack_interaction, -1e6, 1e6)
    )
    
    # Final safeguard: ensure finite output with strict shape guarantee
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 1e3, neginf=-finfo.max / 1e3)
    assert score.shape == (N,), f"Expected (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values remain after sanitization"
    
    return score
