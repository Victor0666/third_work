def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with hybrid rank normalization and dual-gated energy.
    Combines Parent 1's quantile-aware centering with Parent 2's MAD scaling and sigmoid gating.
    Uses exactly 11 parameters; all numeric literals are in {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Hybrid quantile-MAD rank normalization: robust centering + bounded scaling ===
    def hybrid_rank_normalize(x, q=0.75):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        # Quantile-aware centering: use median of top-q tail to avoid outlier pull
        q_val = np.quantile(x_finite, q)
        center = np.median(x_finite[x_finite >= q_val])
        # MAD-based spread estimation on full finite set
        mad_full = np.median(np.abs(x_finite - np.median(x_finite))) * PARAMS["mad_scale_factor"]
        scale = np.where(mad_full > eps, mad_full, eps)
        normed = (x - center) / (scale + eps)
        return np.clip(normed, -2.0, 2.0)
    
    # === Smooth sigmoid DDL protection gate: continuous, differentiable, reuse width ===
    width = PARAMS["ddl_protection_gate_width"]
    ddl_gate = 1.0 / (1.0 + np.exp(-slk / (width + eps)))  # 0→1 as slack goes from -∞→+∞; center at 0
    
    # === Slack-driven base penalty/gain: signed, normalized via robust absolute-slack scale ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    if len(finite_abs_slk) == 0:
        slk_scale = eps
    else:
        median_slk = np.median(finite_abs_slk)
        mad_slk = np.median(np.abs(finite_abs_slk - median_slk)) * PARAMS["mad_scale_factor"]
        slk_scale = mad_slk + eps
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Dual-gated energy term: multiplicative urgency coupling AND uncertainty-as-load proxy ===
    # Urgency coupling: peaks at slack=0 via sigmoid(-slack/width)
    urgency_coupling = 1.0 + PARAMS["energy_slack_coupling_strength"] * (1.0 / (1.0 + np.exp(slk / (width + eps))))
    # Uncertainty-as-load proxy: reduces energy priority when uncertainty is high
    uncert_max = np.max(np.abs(uncert)) + eps
    uncert_norm = np.clip(uncert, 0.0, 2.0) / uncert_max
    load_proxy_gate = 1.0 - PARAMS["energy_slack_coupling_strength"] * uncert_norm
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * hybrid_rank_normalize(inv_energy) * urgency_coupling * load_proxy_gate
    
    # === Criticality: gated by smooth sigmoid DDL gate, rank normalized with hybrid method ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * hybrid_rank_normalize(rank)
    
    # === Bottleneck-proximal term: upward_rank * remaining_work — critical-path blocking impact ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * hybrid_rank_normalize(bottleneck + eps)
    
    # === Bounded linear duration-uncertainty blend: avoids over-suppression ===
    duration = exec_t + comm_t
    dur_norm = hybrid_rank_normalize(duration + eps)
    uncert_norm = hybrid_rank_normalize(uncert + eps)
    dur_uncert_blend = dur_norm + 0.7 * uncert_norm  # fixed ratio to remove parameter
    dur_score = hybrid_rank_normalize(dur_uncert_blend)
    
    # === Starvation mitigation: clipped linear wait time — bounded, deterministic ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -hybrid_rank_normalize(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_safe_scale"]
    min_safe = -finfo.max / PARAMS["finfo_safe_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
