def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with key structural innovations:
      - Added host-load-sensitive energy modulation: energy term scaled by DDL pressure (sigmoid of normalized slack)
      - Replaced static bottleneck-sharpening exponent with adaptive piecewise sharpness: steeper near critical slack, flatter otherwise
      - Introduced 'ready_wait_time' as multiplicative modulator on slack penalty to amplify urgency for long-waiting tasks
      - Removed redundant finfo_max_scale parameter (confirmed inactive in diagnostics) and simplified NaN/inf handling
      - All normalizations use robust quantile scaling with explicit finite-mask fallback
      - Ensures strict monotonicity: no exponential or log terms near domain boundaries
      - Preserves lexicographic DDL-first priority via dominant slack penalty
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x) & (abs_x > 0)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating and modulation ===
    slack_abs = np.abs(slk)
    slack_norm = normalize(slk)
    
    # === Adaptive piecewise sharpness for successor-release interaction ===
    # Steeper exponent when slack is near zero or negative (high risk), flatter otherwise
    sharpness_base = PARAMS["successor_release_sharpness"]
    sharpness = np.where(
        slack_abs < 0.1,
        sharpness_base * 1.5,
        np.where(slack_abs < 1.0, sharpness_base, sharpness_base * 0.7)
    )
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear), now multiplied by wait time for starvation-aware urgency boost ===
    # This creates a synergistic urgency: tasks with both negative slack AND long wait receive amplified priority
    wait_factor = np.clip(wait / (PARAMS["wait_ramp_threshold"] + eps), 0.0, 2.0)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm) * (1.0 + wait_factor),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm) * (1.0 - 0.5 * wait_factor)
    )
    
    # === Energy efficiency: now host-load-sensitive — scaled up under DDL pressure ===
    # Higher DDL pressure (low slack) increases the impact of marginal energy cost
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    energy_score = energy_score * (1.0 + PARAMS["host_load_sensitivity"] * (1.0 - ddl_gate))
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term with adaptive exponent ===
    slack_magnitude_inv = 1.0 / (slack_abs + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, sharpness)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: direct linear wait contribution (not clipped) ===
    # Simpler and more monotonic than clipped ramp — ensures longer wait always helps priority
    wait_score = -normalize(wait + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with safe finite values using machine epsilon
    score = np.nan_to_num(score, nan=0.0, posinf=1e10, neginf=-1e10)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
