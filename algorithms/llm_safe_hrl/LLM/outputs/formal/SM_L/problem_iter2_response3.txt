def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining best practices from both parents:
    - Uses inverse-energy efficiency (Parent 2) for sharper discrimination.
    - Applies exponential wait decay (Parent 2) for bounded starvation mitigation.
    - Integrates robust MAD-based normalization (Parent 1) with tunable scale.
    - Adds fairness exponent to penalize extreme durations non-linearly.
    - Keeps slack-gated criticality and uncertainty-slack interaction (both).
    - All components are finite, deterministic, and shape-strict (N,)."""
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Copy inputs to float64, avoid in-place mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization using MAD + tunable scale (improved over Parent 1/2)
    def robust_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        dev = np.abs(x - center)
        scale = np.median(dev) if len(dev) > 0 else 1.0
        # Apply tunable scale factor for stability-sensitivity tradeoff
        scale = max(scale * PARAMS["robust_scale_factor"], eps)
        return (x - center) / (scale + eps)
    
    # Duration: execution + communication
    duration = exec_t + comm_t
    
    # === Slack processing: piecewise & risk-aware (Parent 2 style, enhanced) ===
    # Quadratic penalty for negative slack; linear de-prioritization for positive
    slack_norm = robust_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling with robust normalization (Parent 2) ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * robust_normalize(inv_energy)
    
    # === Criticality: slack-gated activation (Parent 2) ===
    rank_active = np.where(slk > -eps, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * robust_normalize(rank_active + eps)
    
    # === Duration-uncertainty coupling: weighted harmonic blend (Parent 2) ===
    dur_norm = robust_normalize(duration + eps)
    uncert_norm = robust_normalize(uncert + eps)
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * uncert_norm) 
        / (dur_norm + eps + PARAMS["duration_uncertainty_ratio"] * uncert_norm + eps)
    )
    dur_score = robust_normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time (Parent 2) ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -robust_normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: multiplicative boost under deadline stress (Parent 2) ===
    slack_stress = np.where(slk < 0, np.abs(slack_norm), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * uncert_norm * slack_stress
    
    # === Fairness term: power-law penalization of duration outliers (novel) ===
    # Encourages balanced load by penalizing |duration| deviations more sharply
    norm_dur_dev = robust_normalize(duration)
    fairness_term = np.sign(norm_dur_dev) * (np.abs(norm_dur_dev) ** PARAMS["fairness_exponent"])
    
    # === Assemble final score: smaller = better ===
    # All terms are finite, normalized, and contribute deterministically
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + fairness_term
    )
    
    # Final safeguard: ensure finite, replace NaN/inf with median fallback
    score = np.nan_to_num(score, nan=np.median(score), posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
