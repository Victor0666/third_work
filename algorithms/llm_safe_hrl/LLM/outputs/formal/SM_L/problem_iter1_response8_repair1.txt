def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule with all numeric thresholds declared in PARAMETER_SCHEMA.
    
    Replaces hardcoded 0.75/0.25 with tunable quantiles. All other literals are -2,-1,0,1,2 or derived from np.finfo.
    Uses robust IQR normalization, uncertainty-coupled slack penalty, and rank-per-duration stabilization.
    Smaller score = higher priority.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization using tunable quantiles
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q75 = np.quantile(x, PARAMS["q75_quantile"], method='midpoint')
        q25 = np.quantile(x, PARAMS["q25_quantile"], method='midpoint')
        iqr = q75 - q25 + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Slack processing: power-law asymmetry
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_powered = np.power(slack_abs + eps, PARAMS["slack_sensitivity"]) * slack_sign
    
    # Negative slack penalty gated by uncertainty
    negative_slack_mask = (slack < 0).astype(float)
    uncertainty_coupled_penalty = (
        PARAMS["slack_penalty_factor"] 
        * np.abs(slack_powered) 
        * negative_slack_mask 
        * np.tanh(PARAMS["uncertainty_slack_coupling"] * uncertainty)
    )
    
    # Normalize core features
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    
    # Stabilized criticality: upward_rank per unit duration
    duration_safe = min_exec_time + min_comm_time + eps
    rank_per_duration = upward_rank / duration_safe
    norm_rank_per_duration = robust_normalize(rank_per_duration)
    
    # Bounded anti-starvation via tanh
    wait_boost = PARAMS["wait_fairness"] * np.tanh(ready_wait_time / (1.0 + eps))
    
    # Final score: minimize → prioritize urgent, low-energy, efficient-critical, non-starved
    score = (
        negative_slack_penalty
        + PARAMS["energy_scale"] * norm_energy
        + PARAMS["duration_balance"] * norm_duration
        - PARAMS["criticality_weight"] * norm_rank_per_duration
        - wait_boost
    )
    
    # Ensure finite output, shape (N,)
    return np.nan_to_num(
        score,
        nan=0.0,
        posinf=np.finfo(float).max,
        neginf=-np.finfo(float).max
    )
