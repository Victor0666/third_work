def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining best practices:
      - Soft sigmoid DDL gating from Parent 2 (superior gradient stability) instead of hard mask or Heaviside.
      - Robust z-score + sigma clipping for discriminative normalization (Parent 2), but with fallback to MAD-based norm
        when variance collapses (e.g., uniform inputs) — adaptive hybrid normalization.
      - Bottleneck boost gated by *both* slack < 0 AND work > eps (Parent 2), now combined with release pressure term
        (rank × work) to unify critical path load and successor pressure.
      - Uncertainty-slack interaction uses *signed slack_norm* (not abs), preserving sign-sensitive urgency modulation.
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants.
      - Deterministic, finite, shape-compliant, fully parameterized.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Adaptive hybrid normalization: z-score with sigma clipping, fallback to MAD if std ≈ 0 ===
    def normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x) & (np.abs(x) > eps)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        mu = np.mean(x_finite)
        std = np.std(x_finite) if len(x_finite) > 1 else eps
        if std < eps:
            # Fallback: robust MAD normalization (Parent 1 style)
            abs_x = np.abs(x_finite)
            med = np.median(abs_x)
            mad = np.median(np.abs(abs_x - med)) if np.all(np.isfinite(abs_x)) else eps
            scale = max(med, PARAMS["robust_norm_mad_scale"] * mad, eps)
            return x / (scale + eps)
        else:
            # Primary: z-score with sigma clipping (Parent 2 style)
            z = (x - mu) / (std + eps)
            clip_bound = PARAMS["z_score_clip_sigma"]
            z_clipped = np.clip(z, -clip_bound, clip_bound)
            max_abs = np.maximum(np.max(np.abs(z_clipped)), eps)
            return z_clipped / (max_abs + eps)
    
    # === Normalized slack for gating — used in sigmoid and interactions ===
    slack_norm = normalize(slk)
    
    # === Soft DDL-protection gating: sigmoid(-slack_norm * scale) → 1 when slack high, → 0 when slack low ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_sigmoid_scale"] * slack_norm))
    
    # === Slack-driven priority: signed linear response (penalty for neg, gain for pos) ===
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.where(slk < 0, -slk, 0.0) +
        PARAMS["slack_urgency_gain"] * np.where(slk > 0, slk, 0.0)
    )
    slack_score = normalize(slack_penalty + eps)
    
    # === Energy efficiency: inverse marginal energy, soft-gated for graceful tradeoff ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_gate
    
    # === Criticality: upward_rank weighted by ddl_gate (smooth, not binary) ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank) * ddl_gate
    
    # === Bottleneck + release pressure: unified term gated by deadline stress and work volume ===
    # Combines Parent 2's bottleneck boost (rank * work, active only on urgent + heavy nodes)
    # and Parent 1's release pressure (same product, softly gated by ddl_gate)
    pressure_base = rank * work
    pressure_active = np.where((slk < 0) & (work > eps), pressure_base, 0.0)
    pressure_score = -PARAMS["bottleneck_boost_weight"] * normalize(pressure_active + eps)
    # Add soft-gated release component to retain global balance
    release_score = -PARAMS["release_pressure_weight"] * ddl_gate * normalize(pressure_base + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend of normalized terms ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation, *ungated* to ensure fairness even under slack ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: uses *signed slack_norm*, amplifying uncertainty only when slack < 0 ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * np.where(slack_norm < 0, slack_norm, 0.0)  # preserves directional urgency
    )
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_score
        + energy_score
        + rank_score
        + pressure_score
        + release_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
