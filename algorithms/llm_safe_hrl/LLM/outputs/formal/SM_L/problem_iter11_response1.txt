def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robustness from Parent 2 and bottleneck-awareness from Parent 1:
      - Preserves bounded MAD+median normalization with [-2,2] clipping for stability.
      - Reintroduces bottleneck term (upward_rank * remaining_work) but gates it under DDL protection only.
      - Uses unified normalized slack for all gating — avoids dual scales (width vs threshold).
      - All amplifications/suppressions apply *only* when slack_norm < ddl_protection_gate (tight-deadline regime).
      - Load pressure now includes bottleneck coupling: (rank * work * uncertainty), enabling joint critical-path + risk awareness.
      - Starvation mitigation remains bounded and deterministic: clipped wait time, not exponential.
      - No finfo_max_scale or nan_to_num — uses explicit finite masking for safety and simplicity.
      - All numeric literals are restricted to {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust bounded normalization: median + MAD scale, then clip to [-2, 2] ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            med = np.median(abs_x[finite_mask])
            dev = np.abs(abs_x - med)
            mad = np.median(dev[finite_mask]) if np.any(finite_mask) else eps
            scale = max(med, 1.0 * mad, eps)  # literal 1.0 allowed
        else:
            scale = eps
        normed = x / (scale + eps)
        return np.clip(normed, -2.0, 2.0)  # bounded piecewise form
    
    # === Normalized slack for unified gating ===
    slack_norm = normalize(slk)
    
    # === Slack-driven base penalty/gain: signed, linear, monotonic ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse marginal energy, normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: gated by slack_norm < ddl_protection_gate ===
    rank_active = np.where(slack_norm < PARAMS["ddl_protection_gate"], rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # === Bottleneck-proximal term: gated, normalized product of rank × work ===
    bottleneck_raw = rank * work
    bottleneck_active = np.where(slack_norm < PARAMS["ddl_protection_gate"], bottleneck_raw, 0.0)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_active + eps)
    
    # === Duration-uncertainty coupling: simplified bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear clip ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_normalized = wait_clipped / (PARAMS["wait_clip_threshold"] + eps)
    wait_score = -normalize(wait_normalized + eps)
    
    # === Joint load-pressure: (upward_rank × remaining_work × uncertainty), saturated and gated ===
    load_pressure_raw = rank * work * (uncert + eps)
    load_pressure_norm = normalize(load_pressure_raw + eps)
    # Saturate at adaptive threshold to avoid outlier dominance
    saturation_bound = PARAMS["load_saturation_threshold"] * np.median(np.abs(load_pressure_norm) + eps)
    load_pressure_saturated = np.clip(load_pressure_norm, -saturation_bound, saturation_bound)
    load_score = np.where(
        slack_norm < PARAMS["ddl_protection_gate"],
        load_pressure_saturated,
        0.0
    )
    
    # === DDL protection gate: soft suppression/amplification only in tight-slack regime ===
    protection_mask = (slack_norm < PARAMS["ddl_protection_gate"])
    dur_score = np.where(protection_mask, dur_score * PARAMS["duration_suppression_factor"], dur_score)
    energy_score = np.where(protection_mask, energy_score * PARAMS["energy_amplification_factor"], energy_score)
    rank_score = np.where(protection_mask, rank_score * PARAMS["criticality_amplification_factor"], rank_score)
    bottleneck_score = np.where(protection_mask, bottleneck_score * PARAMS["criticality_amplification_factor"], bottleneck_score)
    load_score = np.where(protection_mask, load_score * PARAMS["energy_amplification_factor"], load_score)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + load_score
    )
    
    # Final safeguard: replace NaN/inf with 0.0 using explicit masking
    score = np.where(np.isfinite(score), score, 0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
