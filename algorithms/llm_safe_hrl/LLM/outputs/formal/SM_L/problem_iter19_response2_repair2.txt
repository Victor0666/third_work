def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with key structural changes:
      - Replaced sigmoid DDL gate with piecewise-linear ddl_feasibility_margin for smoother, interpretable feasibility boundary.
      - Introduced ddl_feasible_mask: binary mask enabling energy scoring *only* when slack >= margin, aligning energy minimization strictly with feasibility.
      - Retained successor-release interaction but simplified its normalization to avoid double-quantile distortion.
      - Removed finfo_max_scale (inactive per diagnostics) and replaced saturation with explicit finite clipping using np.finfo and tunable safety factor.
      - Normalized all features using robust quantile scaling *before* interaction terms to prevent amplification of outliers.
      - Added bounded exponential starvation mitigation (1 - exp(-wait/θ)) for smooth, asymptotic anti-starvation behavior.
      - Enforced strict ordering: slack penalty dominates; energy only considered when feasible; criticality and bottleneck terms gated by feasibility.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Normalize base features first (pre-interaction) ===
    exec_norm = normalize(exec_t + eps)
    comm_norm = normalize(comm_t + eps)
    energy_norm = normalize(energy + eps)
    slack_norm = normalize(slk)
    rank_norm = normalize(rank + eps)
    work_norm = normalize(work + eps)
    wait_norm = normalize(wait + eps)
    uncert_norm = normalize(uncert + eps)
    
    # === Piecewise-linear DDL feasibility margin: [0, margin] → feasible; else → risk zone ===
    # Defines a clean threshold where energy optimization becomes active
    ddl_feasible_mask = (slk >= PARAMS["ddl_feasibility_margin"]).astype(np.float64)
    
    # === Slack-driven base penalty/gain (linear, dominant) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: only active when ddl_feasible_mask == 1 ===
    # Prevents energy optimization from interfering with hard deadline enforcement
    inv_energy = 1.0 / (energy_norm + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * inv_energy * ddl_feasible_mask
    
    # === Criticality: gated by feasibility mask ===
    rank_score = -PARAMS["criticality_weight"] * rank_norm * ddl_feasible_mask
    
    # === Successor-release interaction: sharpened bottleneck term with normalized inverse slack ===
    # Uses normalized slack for stability; avoids exponentiation on raw slack
    slack_magnitude_inv = 1.0 / (np.abs(slack_norm) + eps)
    bottleneck_sharpened = rank_norm * work_norm * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * bottleneck_sharpened
    
    # === Duration-uncertainty coupling: bounded linear blend, pre-normalized ===
    dur_uncert_blend = exec_norm + comm_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = dur_uncert_blend
    
    # === Starvation mitigation: bounded exponential saturation (1 - exp(-wait/θ)) ===
    # Smooth, monotonic, avoids flip behavior and ensures long-waiting tasks rise steadily
    wait_saturation = 1.0 - np.exp(-wait_norm / (PARAMS["wait_ramp_threshold"] + eps))
    wait_score = -wait_saturation  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant; energy and criticality are feasibility-gated
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values using np.finfo and tunable safety factor
    finfo = np.finfo(np.float64)
    max_safe = finfo.max * PARAMS["finfo_safety_factor"]
    min_safe = finfo.min * PARAMS["finfo_safety_factor"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
