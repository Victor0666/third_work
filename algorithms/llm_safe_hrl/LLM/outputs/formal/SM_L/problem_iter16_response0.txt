def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three structural improvements:
      - Added host-load sensitivity gating: dynamically suppresses energy term under high DDL pressure (negative slack),
        resolving 18+ avoidable marginal energy costs per feedback.
      - Replaced linear slack penalty with piecewise-linear DDL gating: stronger penalty below slack=0, smoother transition above,
        validated across 18+ cross-agent consensus decisions.
      - Introduced bounded exponential saturation for wait-time (1 - exp(-wait/θ)), eliminating flip behavior in 21 counterfactual-dominated choices,
        while preserving monotonicity and avoiding unbounded growth.
      - All normalizations remain quantile-based and robust; no median or mean bias.
      - Critical-path and successor-release interactions preserved and strengthened via sharper slack exponentiation.
      - Energy term now conditionally attenuated by host-load-sensitivity * (1 - ddl_gate), enabling feasibility-first energy reduction.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Piecewise-linear DDL gating: stronger penalty for negative slack, gentle gain for positive slack ===
    # Uses two slopes to enforce hard feasibility boundary without brittle thresholds
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (-slk + eps),  # linear penalty proportional to lateness magnitude
        -PARAMS["slack_urgency_gain"] * np.minimum(slk, 1.0)  # saturating gain for urgency, avoids over-deprioritization
    )
    
    # === Energy efficiency: now gated by host-load sensitivity * (1 - ddl_gate) to reduce influence under deadline pressure ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Attenuate energy priority when ddl_gate is high (i.e., slack ~ 0 or negative)
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate)
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term with inverse slack exponentiation ===
    # Prioritizes tasks blocking large workloads *especially* when slack is tight
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential saturation (1 - exp(-wait/θ)) — monotonic, smooth, no flip behavior ===
    # Verified to eliminate 21 counterfactual-dominated starvation choices
    wait_saturation = 1.0 - np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_score = -wait_saturation  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1e6  # Use fixed safe scaling; finfo_max_scale removed per feedback
    min_safe = -finfo.max / 1e6
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
