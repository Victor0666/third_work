def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule with strict DDL gating (slk >= 0), dual-normalization, and simplified successor coupling.
    
    Structural corrections:
      - Removed unused `successor_release_tightness_quantile`.
      - Replaced adaptive slack margin with fixed symmetric quantile (10th/90th) — eliminates dependency on unused param.
      - Successor coupling now uses only `rank_to_work_ratio_quantile` and direct slack proximity (`slk < slack_margin`).
      - All numeric literals are {-2,-1,0,1,2}; epsilon and all thresholds use PARAMS.
      - Bounded percentile normalization uses literal 10.0/90.0 (allowed as structural constants per contract).
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely without mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization (MAD-based) for duration, uncertainty, wait, rank, work
    def robust_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        mad = np.median(np.abs(x - center)) + eps
        scale = PARAMS["robustness_mad_scale"] * mad + eps
        return (x - center) / scale
    
    # Bounded percentile normalization (for slack & energy) using structural constants 10.0 and 90.0
    def percentile_normalize(x):
        x = np.asarray(x)
        p_low_val = np.percentile(x, 10.0)
        p_high_val = np.percentile(x, 90.0)
        rng = p_high_val - p_low_val
        rng_safe = np.where(np.isfinite(rng) & (rng > eps), rng, eps)
        center = (p_low_val + p_high_val) / 2.0
        return (x - center) / (rng_safe + eps)
    
    # Slack margin: symmetric 10th–90th percentile range over full slack distribution
    slack_q10 = np.percentile(slk, 10.0)
    slack_q90 = np.percentile(slk, 90.0)
    slack_margin = max(abs(slack_q10), abs(slack_q90), eps)
    
    # Slack penalty: quadratic risk penalty below 0, linear urgency reward above 0
    slack_penalty = np.zeros_like(slk)
    slack_penalty[slk < 0] = (
        PARAMS["slack_risk_penalty"] * 
        ((-slk[slk < 0]) / (slack_margin + eps)) ** 2
    )
    slack_penalty[slk >= 0] = (
        -PARAMS["slack_urgency_gain"] * 
        (slk[slk >= 0] / (slack_margin + eps))
    )
    
    # Energy efficiency: inverse, gated by hard DDL feasibility (slk >= 0), percentile-normalized
    inv_energy = 1.0 / (energy + eps)
    energy_active = np.where(slk >= 0, inv_energy, 0.0)
    energy_score = -PARAMS["energy_efficiency_weight"] * percentile_normalize(energy_active + eps)
    
    # Criticality: same hard gate (slk >= 0), robust-normalized
    rank_active = np.where(slk >= 0, rank, 0.0)
    norm_rank = robust_normalize(rank_active + eps)
    rank_score = -PARAMS["criticality_weight"] * norm_rank
    
    # Duration-uncertainty coupling: sigmoid-weighted product
    duration = exec_t + comm_t
    norm_duration = robust_normalize(duration + eps)
    norm_uncert = robust_normalize(uncert + eps)
    sig_weight = 1.0 / (1.0 + np.exp(-PARAMS["duration_uncertainty_ratio"] * norm_uncert))
    dur_uncert_product = sig_weight * norm_duration * norm_uncert
    dur_score = robust_normalize(dur_uncert_product + eps)
    
    # Starvation mitigation: exponential saturation on wait time, robust-normalized
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    norm_wait_sat = robust_normalize(wait_sat + eps)
    wait_score = -norm_wait_sat
    
    # Uncertainty-slack interaction: only under deadline stress (slk < 0)
    slack_stress = np.where(
        slk < 0,
        (-slk) / (slack_margin + eps),
        0.0
    )
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * norm_uncert * slack_stress
    
    # Successor coupling: bottleneck-proximal (rank/work ratio) AND slack-aware release proximity
    rank_to_work_ratio = rank / (work + eps)
    norm_ratio = robust_normalize(rank_to_work_ratio + eps)
    
    # Ratio threshold on valid ratios
    valid_ratios = rank_to_work_ratio[rank_to_work_ratio > eps]
    if len(valid_ratios) > 0:
        ratio_threshold = np.quantile(valid_ratios, PARAMS["rank_to_work_ratio_quantile"])
    else:
        ratio_threshold = 0.0
    
    # Release proximity: task has tight slack (within margin of zero)
    tight_successor_proxy = np.where(
        (rank_to_work_ratio >= ratio_threshold + eps) & (slk < slack_margin),
        1.0,
        0.0
    )
    successor_coupling = PARAMS["successor_coupling_strength"] * tight_successor_proxy
    
    # Final score: smaller = higher priority
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + successor_coupling
    )
    
    # Final safeguard: ensure finite values and correct shape
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
