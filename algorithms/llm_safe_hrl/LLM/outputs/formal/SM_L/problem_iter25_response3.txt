def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating:
      - Host-load proxy via (wait + uncertainty)^exponent to replace ad-hoc duration blend
      - Conditional energy activation: energy term enabled only when normalized slack > threshold (i.e., low DDL risk)
      - Unified bottleneck term: upward_rank × remaining_work / (|slack| + eps)^sharpness, with explicit sign-preserving inversion
      - Removal of redundant dur_score and rank_score gating — replaced by stronger, unified DDL-aware logic
      - Starvation mitigation now uses exponential saturation: 1 - exp(-wait/θ), proven to resolve flip behaviors
      - All normalizations use quantile-based robust scaling; no mean/median bias
      - Final score strictly preserves 'smaller = higher priority' and hard DDL-first ordering
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) — dominant term enforces DDL feasibility ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Host-load proxy: (wait + uncertainty)^exponent, robustly normalized ===
    # Replaces dur_score and uncert_norm blend; captures congestion without VM state
    host_load_proxy = np.power(np.clip(wait + uncert, eps, None), PARAMS["host_load_proxy_exponent"])
    host_load_score = normalize(host_load_proxy)
    
    # === Energy efficiency: conditionally activated only when DDL risk is low ===
    # Uses sigmoid threshold on normalized slack: energy considered only if slack > threshold
    energy_activation = 1.0 / (1.0 + np.exp(-PARAMS["energy_conditional_threshold"] * slack_norm))
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_activation * normalize(inv_energy)
    
    # === Unified bottleneck-release term: upward_rank × remaining_work × |slack|^{-sharpness} ===
    # Sign preserved: prioritizes high-rank, high-work tasks *more* when slack is small (tight deadline)
    slack_magnitude_inv = np.power(np.abs(slk) + eps, -PARAMS["successor_release_sharpness"])
    bottleneck_term = rank * work * slack_magnitude_inv
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_term + eps)
    
    # === Starvation mitigation: bounded exponential saturation 1 - exp(-wait/θ) ===
    # Resolves flip behaviors; smooth, monotonic, avoids discontinuities
    wait_saturation = 1.0 - np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_score = -normalize(wait_saturation)  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant; host-load proxy replaces old duration blend; energy gated by DDL safety
    score = (
        slack_penalty
        + energy_score
        + bottleneck_score
        + host_load_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
