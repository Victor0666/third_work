def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Uncertainty-gated duration: uses `uncert^p` as multiplicative suppression factor on duration,
        replacing fragile additive blend — enforces sharper risk-aware pruning without coupling bias.
      - IQR-based rank normalization: replaces median/floor with interquartile range scaling,
        eliminating degeneracy and improving robustness to skewed or near-zero upward_rank distributions.
      - Removed slack-aware energy coupling (per self-reflection): restores monotonic feasibility-first priority;
        energy remains uniformly weighted, ensuring consistent tradeoff behavior across slack regimes.
      - All normalizations are finite, bounded, and use only {-2,-1,0,1,2} literals.
      - No hidden constants; all tunables declared in PARAMETER_SCHEMA.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust IQR-based normalization: handles skew, zeros, and outliers better than median+floor ===
    def normalize_iqr(x, q_low=PARAMS["iqr_quantile_low"], q_high=PARAMS["iqr_quantile_high"]):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        q1 = np.quantile(x_finite, q_low)
        q3 = np.quantile(x_finite, q_high)
        iqr = q3 - q1
        scale = np.where(iqr > eps, iqr, eps)
        center = (q1 + q3) / 2.0
        return (x - center) / (scale + eps)
    
    # === Piecewise linear DDL protection gate: monotonic, bounded, interpretable ===
    width = PARAMS["ddl_protection_gate_width"]
    gate_linear_region = (slk >= -width) & (slk <= 0.0)
    ddl_gate = np.where(
        slk <= -width,
        1.0,
        np.where(
            gate_linear_region,
            1.0 + slk / width,  # linear ramp: 1.0 at -width, 0.0 at 0.0
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: linear, signed, normalized via absolute IQR ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    if len(finite_abs_slk) == 0:
        slk_scale = eps
    else:
        q1_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_low"])
        q3_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_high"])
        slk_scale = np.where(q3_slk - q1_slk > eps, q3_slk - q1_slk, eps)
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse energy, IQR-normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_iqr(inv_energy)
    
    # === Criticality: gated by DDL gate, rank normalized with IQR (no floor needed) ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize_iqr(rank)
    
    # === Bottleneck-proximal term: upward_rank * remaining_work — critical-path blocking impact ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize_iqr(bottleneck + eps)
    
    # === Uncertainty-gated duration: replaces additive blend with bounded multiplicative suppression ===
    # Higher uncertainty → stronger suppression of long-duration tasks (risk-aware pruning)
    duration = exec_t + comm_t
    uncert_gate = np.clip(uncert, 0.0, 2.0) ** PARAMS["uncertainty_gate_exponent"]
    gated_duration = duration * (1.0 + uncert_gate)
    dur_score = normalize_iqr(gated_duration + eps)
    
    # === Starvation mitigation: clipped linear wait time — bounded, deterministic ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize_iqr(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty dominates; all other terms refine within feasibility-aware context
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1e6  # Use fixed safe scale (no PARAMS dependency)
    min_safe = -finfo.max / 1e6
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
