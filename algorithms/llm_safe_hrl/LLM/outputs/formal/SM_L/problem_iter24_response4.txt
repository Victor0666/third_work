def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaces sigmoid DDL gate with a bounded linear ramp (soft clamp) for improved numerical stability and interpretability.
      - Replaces multiplicative uncertainty gating with power-law uncertainty modulation: (1 + uncert)^sensitivity → smoother, more robust scaling.
      - Introduces additive slack-dependent energy modulation (not multiplicative coupling) to avoid explosion near zero slack — aligns with reflection.
      - All features remain strictly DDL-first: slack_penalty dominates; others only activate under feasibility.
      - No hidden constants: only -2,-1,0,1,2 allowed; all tunables via PARAMS.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: preserves relative order, no hard clipping ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Bounded linear-ramp DDL feasibility gate: smooth [0,1] transition centered at margin ===
    # Ramp rises linearly from 0 at (margin - 1/steepness) to 1 at (margin + 1/steepness), then clamps
    ramp_width = 1.0 / (PARAMS["ddl_feasibility_ramp_steepness"] + eps)
    low_bound = PARAMS["ddl_feasibility_margin"] - ramp_width
    high_bound = PARAMS["ddl_feasibility_margin"] + ramp_width
    ddl_feasible_gate = np.clip((slk - low_bound) / (2.0 * ramp_width + eps), 0.0, 1.0)
    
    # === Slack-driven base penalty — dominant, raw units, linear ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        0.0
    )
    
    # === Additive slack-dependent energy modulation: avoids multiplicative blowup near slack=0 ===
    # Modulates energy_score additively using normalized slack proximity: max(0, 1 - |slk| / (margin + eps))
    slack_proximity = np.clip(1.0 - np.abs(slk) / (PARAMS["ddl_feasibility_margin"] + eps), 0.0, 1.0)
    inv_energy = 1.0 / (energy + eps)
    base_energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    energy_score = base_energy_score * ddl_feasible_gate + slack_proximity * base_energy_score
    
    # === Criticality: normalized upward_rank, gated by slack >= 0 (hard binary) for DDL-first safety ===
    criticality_mask = (slk >= 0).astype(np.float64)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps) * criticality_mask
    
    # === Successor-release interaction: sharpens under tight slack using |slack|^{-sharpness} ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Power-law uncertainty-modulated duration: (exec + comm) * (1 + uncert)^sensitivity ===
    # More physically grounded than linear blend; bounded and monotonic
    duration = exec_t + comm_t
    uncertainty_factor = np.power(1.0 + np.clip(uncert, 0.0, 2.0), PARAMS["duration_uncertainty_sensitivity"])
    gated_duration = duration * uncertainty_factor
    dur_norm = normalize(gated_duration + eps)
    dur_score = dur_norm * ddl_feasible_gate
    
    # === Starvation mitigation: bounded linear ramp, normalized to [0,1] range ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using parametrized finfo safety factor
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=finfo.max * PARAMS["finfo_safety_factor"],
        neginf=finfo.min * PARAMS["finfo_safety_factor"]
    )
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
