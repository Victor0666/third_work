def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all tunable constants declared; only -2,-1,0,1,2 used as literals."""
    eps = PARAMS["epsilon"]
    
    # Safely cast inputs to float64, avoid in-place mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)

    # --- Robust normalization using interquartile range (IQR) to resist outliers ---
    def iqr_normalize(x):
        x = np.copy(x)
        q1, q3 = np.quantile(x, [0.25, 0.75])
        iqr = q3 - q1 + eps
        center = np.median(x)
        return (x - center) / iqr

    # Normalize only *finite* values; replace inf/nan with median before norm
    def safe_iqr_normalize(x):
        x_clean = np.where(np.isfinite(x), x, np.nanmedian(x))
        return iqr_normalize(x_clean)

    norm_energy = safe_iqr_normalize(energy)
    norm_exec_comm = safe_iqr_normalize(exec_t + comm_t)
    norm_rank = safe_iqr_normalize(rank)
    norm_work = safe_iqr_normalize(work)
    norm_wait = safe_iqr_normalize(wait)
    norm_uncert = safe_iqr_normalize(uncert)

    # --- Slack processing: prioritize negative slack strongly, smooth transition near zero ---
    p = np.clip(PARAMS["slack_sensitivity"], 0.1, 4.0)
    signed_slack_power = np.sign(slk) * np.power(np.abs(slk) + eps, p)
    norm_signed_slack = safe_iqr_normalize(signed_slack_power)

    # --- Lateness penalty: exponential-like but bounded and monotonic ---
    lateness_penalty = np.zeros_like(slk)
    neg_mask = slk < 0
    if np.any(neg_mask):
        abs_slk_neg = np.abs(slk[neg_mask])
        capped_input = np.clip(PARAMS["slack_penalty_scale"] * abs_slk_neg, 0.0, PARAMS["lateness_cap"])
        lateness_penalty[neg_mask] = np.exp(capped_input) - 1.0

    # --- Criticality boost: amplify upward_rank only under deadline pressure or high uncertainty ---
    pressure_score = np.clip(
        (np.abs(slk) * (slk < 0) + uncert) * PARAMS["uncertainty_slack_interaction"],
        0.0, 10.0
    )
    # Sigmoid gate: centered at PARAMS["sigmoid_center_offset"], slope ~1
    gate = 1.0 / (1.0 + np.exp(-pressure_score + PARAMS["sigmoid_center_offset"]))
    boosted_rank = norm_rank * (1.0 + PARAMS["criticality_boost"] * gate)

    # --- Anti-starvation: saturating wait bonus (log1p-based, stable) ---
    wait_bonus = np.log1p(wait * PARAMS["wait_decay_rate"])
    norm_wait_bonus = safe_iqr_normalize(wait_bonus)

    # --- Duration balance ---
    duration_score = norm_exec_comm * PARAMS["duration_balance"]

    # --- Final score: smaller = higher priority ---
    score = (
        lateness_penalty
        + PARAMS["energy_norm_factor"] * norm_energy
        + duration_score
        - boosted_rank
        - norm_wait_bonus
        + norm_signed_slack
    )

    # Ensure finite output
    score = np.nan_to_num(
        score,
        nan=np.nanmedian(score),
        posinf=np.nanmax(score),
        neginf=np.nanmin(score)
    )

    return score.astype(np.float64, copy=False)
