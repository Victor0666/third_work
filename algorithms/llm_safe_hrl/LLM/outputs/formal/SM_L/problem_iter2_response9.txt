def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule synthesizing best practices:
    - DDL compliance first: piecewise quadratic slack penalty + strict criticality gating.
    - Energy prioritization via inverse scaling with robust clipping.
    - Uncertainty-slack interaction only under deadline stress.
    - Exponential wait decay for anti-starvation with bounded saturation.
    - Adaptive fairness: power-law penalty on duration deviation to avoid bias toward extremes.
    - All operations guarded against NaN/inf/zero; deterministic and finite-output guaranteed.
    Smaller score = higher priority.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs to float64 and ensure no mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization: median-based, epsilon-guarded ===
    def robust_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        dev = np.abs(x - center)
        scale = np.median(dev) if len(dev) > 0 else 1.0
        scale = max(scale, eps)
        return (x - center) / scale
    
    # === Duration & uncertainty coupling ===
    duration = exec_t + comm_t
    dur_norm = robust_normalize(duration + eps)
    uncert_norm = robust_normalize(uncert + eps)
    # Harmonic blend: avoids division-by-zero, stable under low uncertainty
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * uncert_norm) 
        / (np.abs(dur_norm) + eps + PARAMS["duration_uncertainty_ratio"] * np.abs(uncert_norm) + eps)
    )
    # Fairness: power-law penalty on |dur_norm| to discourage both ultra-short and ultra-long bias
    fairness_term = np.sign(dur_norm) * np.abs(dur_norm) ** PARAMS["fairness_exponent"]
    
    # === Slack processing: DDL-first — quadratic penalty for negative, linear soft gain for positive ===
    slack_norm = robust_normalize(slk + eps)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (np.abs(slack_norm) ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse with clipping to prevent explosion ===
    inv_energy = 1.0 / (np.clip(energy, eps, None) + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * robust_normalize(inv_energy)
    
    # === Criticality gating: only contribute if slack permits breathing room ===
    rank_active = np.where(slk > -eps, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * robust_normalize(rank_active + eps)
    
    # === Wait-time mitigation: exponential saturation, normalized for scale consistency ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -robust_normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: only under deadline stress (slk < 0) ===
    slack_stress = np.where(slk < 0, np.abs(slack_norm), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * uncert_norm * slack_stress
    
    # === Assemble final score: smaller = better ===
    # All terms are finite, normalized, and deterministic
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + robust_normalize(fairness_term)
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf using machine limits, preserve ordering via median fallback
    finfo = np.finfo(float)
    score = np.nan_to_num(
        score,
        nan=np.median(score) if np.isfinite(np.median(score)) else 0.0,
        posinf=finfo.max,
        neginf=finfo.min
    )
    
    # Ensure shape and finiteness
    assert score.shape == (N,), f"Expected (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values in final priority score"
    
    return score
