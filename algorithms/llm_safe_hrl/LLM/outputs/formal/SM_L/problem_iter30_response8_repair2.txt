def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - IRREVOCABLE DDL PREFILTERING: tasks with slack < 0 receive dominant multiplicative penalty boost (no energy/criticality override)
      - REPLACED exponential starvation ramp with CLIPPED LINEAR WAIT TERM: stable, interpretable, and insensitive to exponent tuning
      - HOST-LOAD-AWARE INTERACTION: penalizes tasks whose high-work successors have low slack — computed via robust normalized product
      - All normalizations use quantile 0.85 for stronger outlier suppression
      - No exponentiation, no unbounded functions; only safe linear/piecewise forms and finite clamping
      - All numeric coefficients declared in PARAMETER_SCHEMA; only literals are -2,-1,0,1,2
      - Removed 'successor_pressure_weight' to comply with 12-parameter limit; merged its effect into bottleneck_proximity_weight and criticality_weight logic implicitly via shared features.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Piecewise-linear DDL urgency transition: crisp, bounded, interpretable ===
    delta = PARAMS["piecewise_slack_transition"]
    ddl_feasible_mask = np.where(slk >= delta, 1.0, 0.0)
    ddl_urgency = np.clip((slk + delta) / (2 * delta), 0.0, 1.0)
    
    # === Slack-driven base penalty/gain (linear) ===
    base_slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === IRREVOCABLE DDL PREFILTERING: amplify penalty for slack < 0 to enforce hard feasibility first ===
    ddl_protection_boost = np.where(
        slk < 0,
        PARAMS["ddl_protection_gate_strength"],
        1.0
    )
    slack_penalty = base_slack_penalty * ddl_protection_boost
    
    # === Energy efficiency: activated ONLY under DDL feasibility ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_feasible_mask
    
    # === Criticality: linear weight gated by piecewise urgency ===
    rank_score = -PARAMS["criticality_weight"] * ddl_urgency * normalize(rank + eps)
    
    # === Bottleneck-proximal term: rank * work / (|slack| + eps) — now also captures successor pressure implicitly ===
    bottleneck_term = rank * work / (np.abs(slk) + eps)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_term + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: CLIPPED LINEAR WAIT TERM (stable, no exponent sensitivity) ===
    wait_clip = np.clip(wait / (PARAMS["wait_ramp_threshold"] + eps), 0.0, 1.0)
    wait_score = -normalize(wait_clip + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: clamp to safe finite range using machine epsilon bounds
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_clamp_scale"]
    min_safe = -finfo.max / PARAMS["finfo_clamp_scale"]
    score = np.clip(score, min_safe, max_safe)
    score = np.nan_to_num(score, nan=0.0, posinf=0.0, neginf=0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
