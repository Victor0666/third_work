def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robust IQR normalization and multiplicative uncertainty gating
    with Parent 1's normalized bottleneck ratio and direct wait-time monotonicity — while removing fragile slack-energy coupling.
    
    Key improvements:
      - Unified normalization: IQR-first, fallback to quantile-based scaling if IQR collapses (e.g., constant inputs).
      - Bottleneck proximity redefined as ratio (rank * work) / (|slack| + eps), preserving deadline-driven sharpness 
        without overflow or instability; normalized via same IQR/quantile logic.
      - Starvation mitigation uses clipped wait time *without* normalization inversion — instead directly penalizes long waits
        via monotonic -wait_clipped term (stronger than inverse-normalized wait), ensuring strict FIFO bias when safe.
      - Duration-uncertainty coupling remains multiplicative (Parent 2's superior pruning), but now gated by ddl_gate 
        to avoid over-suppressing tasks when deadlines are relaxed.
      - All terms retain DDL-first dominance: slack_penalty is always leading term.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust unified normalization: IQR first, fallback to quantile scale if degenerate ===
    def normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        q1 = np.quantile(x_finite, PARAMS["iqr_quantile_low"])
        q3 = np.quantile(x_finite, PARAMS["iqr_quantile_high"])
        iqr = q3 - q1
        if np.allclose(iqr, 0.0, atol=eps):
            # Fallback: use quantile-based scale to avoid division by zero
            scale = np.quantile(np.abs(x_finite), PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
            center = np.median(x_finite)
            return (x - center) / (scale + eps)
        else:
            center = (q1 + q3) / 2.0
            return (x - center) / (iqr + eps)
    
    # === Piecewise linear DDL protection gate: monotonic, bounded, interpretable ===
    width = PARAMS["ddl_protection_gate_width"]
    gate_linear_region = (slk >= -width) & (slk <= 0.0)
    ddl_gate = np.where(
        slk <= -width,
        1.0,
        np.where(
            gate_linear_region,
            1.0 + slk / width,  # linear ramp: 1.0 at -width, 0.0 at 0.0
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: linear, signed, normalized via absolute IQR ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    if len(finite_abs_slk) == 0:
        slk_scale = eps
    else:
        q1_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_low"])
        q3_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_high"])
        slk_scale = np.where(q3_slk - q1_slk > eps, q3_slk - q1_slk, eps)
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse energy, robustly normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: gated by DDL gate, robustly normalized ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank)
    
    # === Bottleneck proximity: ratio form — prioritizes critical-path blockers *especially* under tight slack ===
    # Uses same robust normalization as other features; avoids power-law instability and preserves ordering
    bottleneck_ratio = (rank * work) / (abs_slk + eps)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_ratio + eps)
    
    # === Uncertainty-gated duration: multiplicative suppression, *only active under DDL pressure* ===
    # Prevents unnecessary suppression when slack is healthy; improves energy-efficiency tradeoff in safe regions
    duration = exec_t + comm_t
    uncert_gate = np.clip(uncert, 0.0, 2.0) ** PARAMS["uncertainty_gate_exponent"]
    gated_duration = duration * (1.0 + ddl_gate * uncert_gate)  # suppression only when ddl_gate > 0
    dur_score = normalize(gated_duration + eps)
    
    # === Starvation mitigation: clipped linear wait penalty — stronger and monotonic ===
    # Direct penalty: longer wait → lower priority score (more negative), no normalization inversion needed
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -wait_clipped  # monotonic, bounded, avoids normalization artifacts
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty dominates; all other terms refine within feasibility-aware context
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max * 0.5  # Use fixed safe scale (no PARAMS dependency for stability)
    min_safe = finfo.min * 0.5
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
