def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with smooth DDL protection gate, successor-release interaction, and host-load-aware energy scaling.
    
    Key structural mutations:
      - Replaces hard `slack_norm >= 0` gating with *smooth sigmoid gate* centered at slack=0 (controlled by ddl_protection_gate_width)
        → eliminates discontinuities, improves gradient stability for CMA-ES, and enables graceful degradation under marginal slack.
      - Adds *successor-release interaction*: prioritizes tasks whose completion releases many successors *only when system load is high* 
        (estimated via median-normalized remaining_work), addressing 'critical_path starvation' observed in replay failures.
      - Introduces *host-load-aware energy scaling*: applies power-law compression to energy_score when total remaining_work is high,
        preventing over-aggressive low-energy selection that delays critical path tasks under congestion.
      - Uses *median-based robust normalization* (instead of mean) for all feature scalings to suppress outlier influence.
      - All divisions guarded; all NaN/inf replaced; shape strictly enforced; deterministic.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust median-based normalization helper: resistant to outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        # Use median absolute deviation scaled by median for robustness
        med_abs = np.median(abs_x)
        scale = med_abs if np.all(np.isfinite(abs_x)) and med_abs > eps else eps
        return x / (scale + eps)
    
    # === Smooth DDL protection gate: sigmoid transition around slack = 0 ===
    # Enables gradual activation instead of binary on/off — avoids instability near boundary
    slack_centered = slk / (np.median(np.abs(slk)) + eps)
    gate_smooth = 1.0 / (1.0 + np.exp(-slack_centered / PARAMS["ddl_protection_gate_width"]))
    
    # === Slack-driven components: symmetric penalty/gain using smooth gate ===
    # Negative slack → strong penalty; positive slack → mild gain; smooth transition
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.maximum(0.0, -slack_centered) 
        - PARAMS["slack_urgency_gain"] * np.maximum(0.0, slack_centered)
    )
    
    # === Energy efficiency: inverse scaling, now load-aware ===
    inv_energy = 1.0 / (energy + eps)
    energy_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Compress energy priority under high system load (large remaining_work)
    load_factor = np.clip(normalize(work), 0.0, 2.0)  # [0,2] normalized load estimate
    energy_score = energy_base * (1.0 - load_factor ** PARAMS["host_load_sensitivity"])
    
    # === Criticality: smooth-gated upward_rank (not binary) ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps) * gate_smooth
    
    # === Duration-uncertainty coupling: bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: activated only under negative slack region ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * np.maximum(0.0, -slack_centered)
    )
    
    # === Successor-release interaction: promotes tasks that unblock many successors under load ===
    # Approximated by upward_rank * (1 - gate_smooth) * load_factor — i.e., critical but non-DDL-urgent tasks get boost when load is high
    successor_release_boost = (
        PARAMS["load_successor_release_weight"] 
        * normalize(rank + eps) 
        * (1.0 - gate_smooth) 
        * load_factor
    )
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + successor_release_boost
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
