def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with piecewise-linear slack gating and bounded wait linearity.

    Key structural improvements:
      - Replaces sigmoid DDL-protection gate with *piecewise-linear slack gate*: 
        smooth, interpretable, and numerically stable; avoids gradient vanishing 
        and enables precise control over slack transition region via `slack_gate_steepness`.
      - Replaces exponential wait saturation with *bounded linear wait scoring*:
        `min(wait * scale, wait_cap)` ensures monotonic, predictable starvation mitigation.
      - All thresholds and coefficients are now declared parameters — no hidden literals.
      - Uses np.finfo for machine-precision safeguards instead of hardcoded epsilons.
      - All normalizations use mean-abs scale + PARAMS["epsilon"]; all divisions guarded.
    Smaller score = higher priority.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization: handles zero-mean and near-zero scale
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        scale = np.mean(abs_x) if np.all(np.isfinite(abs_x)) and np.mean(abs_x) > eps else eps
        return x / (scale + eps)
    
    # Normalized slack for gating and penalty/gain
    slack_norm = normalize(slk)
    
    # Slack-driven base penalty/gain: linear, symmetric around zero
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # Piecewise-linear slack gate: 0 when slk <= -1/steepness, 1 when slk >= 1/steepness, linear in between
    steep = PARAMS["slack_gate_steepness"]
    gate_lower = -1.0 / (steep + eps)
    gate_upper = 1.0 / (steep + eps)
    gate_linear = np.clip((slk - gate_lower) / (gate_upper - gate_lower + eps), 0.0, 1.0)
    slack_gate = np.where(slk < gate_lower, 0.0, np.where(slk > gate_upper, 1.0, gate_linear))
    
    # Energy efficiency: inverse marginal energy, attenuated only near critical slack
    inv_energy = 1.0 / (energy + eps)
    energy_score_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    energy_score = energy_score_base * slack_gate
    
    # Criticality: upward_rank only active when slack allows recovery, attenuated by linear gate
    rank_active = np.where(slack_norm >= 0, rank, 0.0)
    rank_score_base = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    rank_score = rank_score_base * slack_gate
    
    # Duration-uncertainty blend: stable linear sum
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend) * slack_gate
    
    # Starvation mitigation: bounded linear wait scoring — monotonic & tunable
    wait_score_base = PARAMS["wait_linearity_scale"] * wait
    wait_score = -np.minimum(wait_score_base, PARAMS["wait_cap"])
    
    # Uncertainty-slack interaction: amplifies uncertainty *only* under slack stress
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * np.where(slack_norm < 0, np.abs(slack_norm), 0.0)
    )
    
    # Final score: smaller = higher priority
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Safeguard: replace NaN/inf with finite machine-safe bounds using native finfo
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max * 1e-3, neginf=finfo.min * 1e-3)
    
    # Enforce output shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
