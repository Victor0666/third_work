def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaced sigmoid DDL gate with a bounded piecewise-linear ramp (0→1 over [-δ, δ]) using explicit center bias parameter.
      - Introduced load-proxy normalization: `min_incremental_energy` is normalized using a dynamic scale that grows
        with `ready_wait_time / (|slack| + eps)` raised to tunable exponent — captures system congestion without requiring VM-type flags.
      - Kept successor-release interaction and energy modulation via the new load-aware scaling instead of exponentiation,
        yielding better interpretability and numerical stability.
      - All terms remain strictly DDL-first: slack_penalty dominates, others refine within feasibility.
      - No unbounded operations; all divisions, powers, and clamps use ε-safeguards and finite bounds.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Piecewise-linear DDL ramp gate: (bias + (1-bias)*(1 + slk/delta)/2) clipped to [0,1] ===
    delta = PARAMS["ddl_ramp_halfwidth"]
    # Linear mapping: slk ∈ [-delta, delta] → [0,1], centered at bias
    ramp_linear = PARAMS["ddl_ramp_center_bias"] + (1.0 - PARAMS["ddl_ramp_center_bias"]) * 0.5 * (1.0 + slk / (delta + eps))
    ddl_ramp = np.clip(ramp_linear, 0.0, 1.0)
    
    # === Slack-driven base penalty/gain (linear) — dominant enforcement of DDL feasibility ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk) / (delta + eps),
        -PARAMS["slack_urgency_gain"] * np.abs(slk) / (delta + eps)
    )
    
    # === Load-aware energy normalization proxy: higher wait + tighter slack → stronger energy focus ===
    # Proxy scale = (wait / (|slack| + eps))^p → amplifies energy term when system is congested & deadline tight
    load_proxy = np.power((wait + eps) / (np.abs(slk) + eps), PARAMS["load_proxy_exponent"])
    # Normalize energy using quantile scale *modulated by load proxy* for context-sensitive weighting
    energy_scale_base = np.quantile(np.abs(energy), PARAMS["robust_normalization_quantile"])
    energy_scale = np.where(energy_scale_base > eps, energy_scale_base, eps) * (1.0 + load_proxy)
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * (inv_energy / (energy_scale + eps))
    
    # === Criticality: gated by piecewise-linear ddl_ramp (smooth, no gradient loss) ===
    rank_score = -PARAMS["criticality_weight"] * ddl_ramp * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term with inverse slack exponentiation ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear ramp → monotonic & smooth ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant; others provide fine-grained trade-offs within feasibility
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with zeros (safe fallback, no unbounded scaling)
    score = np.nan_to_num(score, nan=0.0, posinf=0.0, neginf=0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
