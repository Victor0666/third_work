def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three structural improvements:
      - Added conditional energy_feasibility_gate: energy term only activated when slack > margin, preventing energy optimization from compromising DDL safety.
      - Replaced linear slack_penalty with piecewise smooth penalty: quadratic for negative slack (stronger risk penalty), linear for positive slack (mild urgency decay).
      - Introduced normalized bottleneck proximity via (upward_rank * remaining_work) / (min_exec_time + min_comm_time + eps), emphasizing throughput-critical tasks.
      - Removed finfo_max_scale parameter (evidence showed it was inactive) and simplified NaN/inf handling using np.finfo directly.
      - All normalizations use quantile-based scaling with explicit finite-mask safety.
      - Wait ramp now uses sigmoid instead of clip+normalize for smoother starvation mitigation without hard thresholds.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            # Use quantile for robust scale; avoid zero or inf
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Piecewise smooth slack penalty: quadratic for negative slack, linear for positive ===
    # Ensures stronger penalty for violation risk while preserving monotonicity
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),  # quadratic penalty amplifies lateness risk
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)  # linear decay for safe margin
    )
    
    # === Energy efficiency: gated by energy_feasibility_gate — only active when slack > margin ===
    # Prevents energy minimization from dominating when deadline pressure is high
    energy_feasibility_margin = PARAMS["energy_feasibility_gate"] * np.quantile(np.abs(slk), 0.9)
    energy_active = (slk > energy_feasibility_margin).astype(np.float64)
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * energy_active
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Bottleneck proximity: normalized by estimated duration to favor throughput-critical tasks ===
    # Upward rank * remaining work per unit execution+comm time emphasizes efficiency on critical path
    duration = exec_t + comm_t + eps
    bottleneck_raw = (rank * work) / duration
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_raw)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    dur_norm = normalize(duration)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: smooth sigmoid ramp over wait time — no hard thresholds ===
    # Ensures monotonic priority boost with increasing wait, avoids discontinuities
    wait_sigmoid = 1.0 / (1.0 + np.exp(-0.1 * (wait - PARAMS["wait_ramp_threshold"])))
    wait_score = -wait_sigmoid  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max * 0.5, neginf=finfo.min * 0.5)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
