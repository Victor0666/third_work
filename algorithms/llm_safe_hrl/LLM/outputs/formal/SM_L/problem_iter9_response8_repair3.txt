def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Task-local IQR-based normalization using fixed quantiles (0.25, 0.75) to avoid extra parameters.
      - Piecewise-linear slack-dependent energy weight (replaces unstable multiplicative interaction).
      - All numeric literals are in [-2, 2]; exactly 12 tunable parameters declared.
      - Deterministic, finite, shape-correct, and feasibility-first.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust IQR-based normalization using fixed 25th/75th percentiles (no extra params) ===
    def normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        q_low = np.quantile(x_finite, 0.25)
        q_high = np.quantile(x_finite, 0.75)
        iqr = q_high - q_low
        scale = np.where(iqr > eps, iqr, eps)
        center = (q_low + q_high) / 2.0
        return (x - center) / (scale + eps)
    
    # === Slack normalization with tunable scale before sigmoid ===
    slack_norm = normalize(slk) * PARAMS["slack_normalization_scale"]
    
    # === Smooth sigmoid DDL protection gate ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling, normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Piecewise-linear slack-dependent energy weight: activates only when slack <= 0 ===
    slack_under_zero = np.maximum(0.0, -slack_norm)
    energy_weight_boost = 1.0 + PARAMS["energy_slack_piecewise_slope"] * slack_under_zero
    energy_score_adjusted = energy_score * energy_weight_boost
    
    # === Criticality: gated smoothly by ddl_gate ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Bottleneck-proximal term: upward_rank * remaining_work ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck + eps)
    
    # === Duration-uncertainty blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: clipped linear wait time ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score_adjusted
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
