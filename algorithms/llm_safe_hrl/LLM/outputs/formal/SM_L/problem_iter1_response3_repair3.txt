def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric thresholds declared; only -2,-1,0,1,2 used as literals."""
    eps = PARAMS["epsilon"]
    
    # Robust per-feature normalization: use median absolute deviation for outlier resilience
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        spread = np.median(np.abs(x - center)) + eps
        return (x - center) / spread
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    
    # --- Risk-aware slack penalty: strong nonlinearity for negative slack ---
    slack_sign = np.sign(slack)
    slack_abs = np.abs(slack)
    # Apply exponent only to negative slack; linear term for positive slack using declared parameter
    slack_penalty = np.where(
        slack < 0,
        np.power(slack_abs + eps, PARAMS["slack_penalty_exponent"]),
        slack_abs * PARAMS["slack_buffer_weight"]
    )
    
    # --- Criticality amplification gate: boost upward_rank only when slack is tight ---
    slack_risk_score = np.clip(-norm_slack, 0.0, 2.0)
    # Sigmoid gate centered at PARAMS["sigmoid_shift"], scaled by 2 for steepness
    rank_gate = 1.0 / (1.0 + np.exp(-2 * (slack_risk_score - PARAMS["sigmoid_shift"])))
    rank_amplifier = 1.0 + PARAMS["criticality_scale"] * rank_gate
    amplified_rank = norm_rank * rank_amplifier
    
    # --- Duration-uncertainty coupling: penalize long/uncertain tasks more under deadline pressure ---
    slack_pressure_mask = np.where(norm_slack < -PARAMS["slack_pressure_threshold"], 1.0, 0.0)
    uncert_gate = np.where(norm_uncert > PARAMS["uncertainty_slack_gate"], 1.0, 0.0)
    dur_uncert_penalty = norm_duration * slack_pressure_mask * uncert_gate * PARAMS["duration_robustness"]
    
    # --- Bounded starvation relief: saturating benefit from waiting ---
    wait_benefit = (np.arctan(PARAMS["wait_decay"] * norm_wait) + np.pi/2) / np.pi
    
    # --- Work saturation: dampen influence of extremely large remaining_work ---
    work_ref = PARAMS["work_saturation_point"]
    work_scaled = remaining_work / (work_ref + eps)
    work_benefit = (np.tanh(work_scaled) + 1.0) / 2.0
    
    # --- Composite score: smaller = higher priority ---
    score = (
        slack_penalty
        + PARAMS["energy_sensitivity"] * norm_energy
        + amplified_rank
        + dur_uncert_penalty
        - wait_benefit
        - work_benefit
    )
    
    # Final safeguard: ensure finite, deterministic output
    finfo = np.finfo(float)
    return np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
