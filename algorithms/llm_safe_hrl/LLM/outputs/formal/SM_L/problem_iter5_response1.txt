def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robustness from Parent 1 and performance from Parent 2:
      - Uses median-based normalization with MAD scale (Parent 1) for outlier resilience.
      - Adopts smooth sigmoid DDL gate (both parents) but applies it *only* to criticality and uncertainty-slack terms — avoids over-gating energy/duration.
      - Replaces brittle `host_load_sensitivity` threshold with exponentiated `(work * uncert)` (Parent 2), now scaled by `normalized_slack_sensitivity` to modulate load impact based on deadline pressure.
      - Introduces *normalized slack sensitivity*: scales the input to sigmoid gate to sharpen urgency discrimination when slack is near zero.
      - Keeps simplified `rank * work` successor release term (Parent 2) — empirically stable and interpretable.
      - All components are finite, deterministic, and shape-preserving.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust median-based normalization (Parent 1): resistant to skew & outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        med = np.median(abs_x)
        mad = np.median(np.abs(abs_x - med)) if np.all(np.isfinite(abs_x)) else eps
        scale = max(med, mad, eps)
        return x / (scale + eps)
    
    # === Normalized slack with adaptive sensitivity for sharper DDL transition ===
    # Sharper centering improves urgency resolution near deadline boundary
    abs_slk = np.abs(slk)
    slk_norm = slk / (np.median(abs_slk) + eps)
    slk_scaled = slk_norm * PARAMS["normalized_slack_sensitivity"]
    
    # === Smooth DDL protection gate: centered at zero, width-tuned sigmoid ===
    ddl_gate = 1.0 / (1.0 + np.exp(-slk_scaled / (PARAMS["ddl_protection_gate_width"] + eps)))
    
    # === Slack-driven penalty/gain: unnormalized to preserve physical scale alignment ===
    # Avoids double-normalization; uses raw slack for direct time-unit semantics
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.maximum(-slk, 0.0) +
        PARAMS["slack_urgency_gain"] * np.minimum(slk, 0.0)
    )
    
    # === Energy efficiency: inverse scaling, *ungated* to ensure low-energy tasks always win when feasible ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: gated *only* where DDL margin exists (smoothly via ddl_gate) ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Duration-uncertainty coupling: *ungated* linear blend for reliability ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation, *ungated* to universally prevent starvation ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: activated *only* under deadline stress (1 - ddl_gate) ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * (1.0 - ddl_gate)
    )
    
    # === Successor-release pressure: product of criticality and workload (Parent 2) ===
    release_pressure = rank * work
    release_score = -PARAMS["load_successor_release_weight"] * normalize(release_pressure + eps)
    
    # === Nonlinear host-load interaction: exponentiated risk-concentrated load, scaled by slack pressure ===
    # Higher `normalized_slack_sensitivity` amplifies load penalty when deadlines are tight
    load_factor = (work * uncert + eps) ** PARAMS["host_load_interaction_exponent"]
    load_score = PARAMS["normalized_slack_sensitivity"] * normalize(load_factor)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + release_score
        + load_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
