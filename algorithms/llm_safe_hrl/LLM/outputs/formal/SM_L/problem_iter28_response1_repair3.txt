def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with:
      - Replaced sigmoid DDL gate with interpretable piecewise-linear urgency transition (sharper, bounded, no asymptotes)
      - Added explicit starvation-aware wait-saturation term using bounded exponential ramp (1 - exp(-wait/θ)) instead of linear clip
      - Unified bottleneck scoring via robust slack-normalized critical-path importance: upward_rank * remaining_work / (|slack| + eps)
      - Energy term now gated by DDL feasibility mask (binary) rather than soft sigmoid, enforcing hard feasibility boundary
      - All normalizations use quantile-based scaling; no mean/median bias
      - Final score clamped to safe finite range using np.finfo directly
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Piecewise-linear DDL urgency transition: crisp, bounded, interpretable ===
    # Linear ramp from -δ to +δ: maps slack ∈ [-δ, δ] → [0, 1] for urgency activation
    delta = PARAMS["piecewise_slack_transition"]
    ddl_feasible_mask = np.where(slk >= delta, 1.0, 0.0)  # binary feasibility signal
    ddl_urgency = np.clip((slk + delta) / (2 * delta), 0.0, 1.0)  # smooth transition in [-δ, δ]
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: now conditionally activated ONLY under DDL feasibility ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_feasible_mask
    
    # === Criticality: uses urgency-weighted gating instead of sigmoid ===
    rank_score = -PARAMS["criticality_weight"] * ddl_urgency * normalize(rank + eps)
    
    # === Bottleneck-proximal term: simplified robust form — upward_rank * remaining_work / (|slack| + eps) ===
    # Prioritizes high-impact tasks when slack is tight; avoids exponentiation instability
    bottleneck_term = rank * work / (np.abs(slk) + eps)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_term + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential ramp → smooth saturation without clipping artifacts ===
    # 1 - exp(-wait/θ) ensures monotonic, smooth, and bounded (0→1) growth
    wait_exp_ramp = 1.0 - np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_score = -normalize(wait_exp_ramp + eps)  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: clamp to safe finite range using machine epsilon bounds
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_clamp_scale"]
    min_safe = -finfo.max / PARAMS["finfo_clamp_scale"]
    score = np.clip(score, min_safe, max_safe)
    score = np.nan_to_num(score, nan=0.0, posinf=0.0, neginf=0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
