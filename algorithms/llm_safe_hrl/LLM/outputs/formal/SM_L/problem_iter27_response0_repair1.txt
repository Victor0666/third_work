def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Tunable bottleneck_offset and starvation_gain (replacing fixed 0.5/0.85).
      - Removed redundant piecewise_linear_gate_width and slack_aware_energy_decay to meet 12-parameter limit.
      - Reintroduced robust_slack_normalization as sole slack-shaping parameter — sufficient for monotonic, outlier-softened urgency.
      - DDL gating now uses direct sign-based binary gate: 1.0 if slk_robust <= 0, else 0.0 — simpler, more decisive, and avoids width tuning.
      - Energy decay simplified to hard cutoff: energy term fully suppressed when slack_robust < 0, else full weight — no exponential decay needed.
      - All numeric literals are {-2,-1,0,1,2}; epsilon handled via PARAMS; machine bounds via np.finfo.
      - Deterministic, finite, shape-correct, and side-effect-free.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Max-abs normalization: robust, interpretable, and stable ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Robust slack normalization: |slack|^p / (|slack|^p + eps) preserves sign via multiplication by sign(slk) ===
    abs_slk = np.abs(slk) + eps
    robust_slk_mag = np.power(abs_slk, PARAMS["robust_slack_normalization"])
    robust_slk_norm = robust_slk_mag / (robust_slk_mag + eps)
    slk_robust = robust_slk_norm * np.sign(slk)
    
    # === Binary DDL gate: 1.0 if at-risk or late, else 0.0 — decisive and parameter-efficient ===
    ddl_gate = np.where(slk_robust <= 0, 1.0, 0.0)
    
    # === Slack-driven base penalty/gain: applied on robust slack (preserves sign & monotonicity) ===
    slack_penalty = np.where(
        slk_robust < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk_robust),
        -PARAMS["slack_urgency_gain"] * slk_robust
    )
    
    # === Energy efficiency: inverse marginal energy, gated by binary ddl_gate only ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Hard gating: zero energy contribution when slack_robust <= 0
    energy_score = energy_score * (1.0 - ddl_gate)
    
    # === Criticality: upward_rank raised to slack-aware power ===
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|robust_slk| + bottleneck_offset) — now fully tunable and stable ===
    robust_abs_slk = np.abs(slk_robust) + PARAMS["bottleneck_offset"]
    bottleneck_term = rank * work / robust_abs_slk
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize(bottleneck_term + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on raw wait time, scaled by tunable gain ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -PARAMS["starvation_gain"] * wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2.0, neginf=finfo.min / 2.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
