def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with strict DDL-protection gating, percentile-based normalization,
    and eliminated fragile uncertainty-slack interaction.
    
    Key structural improvements:
      - Replaced MAD-based normalization with bounded configurable percentile scaling (pctl_low/pctl_high).
      - Hard DDL-protection gate: zero out *both* upward_rank and energy_score contributions unless
        slack >= 0 — enforces strict feasibility-first ordering before any energy optimization.
      - Removed uncertainty_slack_interaction entirely — confirmed inactive and adds numerical fragility.
      - All components now use the same robust percentile-based normalizer for consistency and stability.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Copy inputs with float64 precision; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Bounded percentile-based normalization: uses configurable pctl_low–pctl_high range
    def percentile_normalize(x):
        x = np.asarray(x)
        p_low = np.percentile(x, PARAMS["pctl_low"])
        p_high = np.percentile(x, PARAMS["pctl_high"])
        rng = p_high - p_low
        # Ensure finite, non-zero range
        rng_safe = np.where(np.isfinite(rng) & (rng > eps), rng, eps)
        center = (p_low + p_high) / 2.0
        return (x - center) / (PARAMS["percentile_range_scale"] * rng_safe + eps)
    
    # Duration: execution + communication
    duration = exec_t + comm_t
    
    # === Slack processing: piecewise & risk-aware ===
    slack_norm = percentile_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling, but ONLY if DDL-feasible (slack >= 0) ===
    inv_energy = 1.0 / (energy + eps)
    energy_active = np.where(slk >= 0, inv_energy, 0.0)
    energy_score = -PARAMS["energy_efficiency_weight"] * percentile_normalize(energy_active + eps)
    
    # === Criticality: strict DDL gate — only active when slack >= 0 (hard feasibility requirement) ===
    rank_active = np.where(slk >= 0, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * percentile_normalize(rank_active + eps)
    
    # === Duration-uncertainty coupling: harmonic blend using percentile normalization ===
    dur_norm = percentile_normalize(duration + eps)
    uncert_norm = percentile_normalize(uncert + eps)
    # Weighted harmonic blend: smooth, bounded, avoids singularity
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * uncert_norm) 
        / (dur_norm + eps + PARAMS["duration_uncertainty_ratio"] * uncert_norm + eps)
    )
    dur_score = percentile_normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -percentile_normalize(wait_sat + eps)
    
    # === Assemble final score: smaller = better ===
    # All components are finite, consistently normalized, and strictly gated by DDL feasibility
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with finite fallback using machine epsilon bounds
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure shape (N,) and finite values
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
