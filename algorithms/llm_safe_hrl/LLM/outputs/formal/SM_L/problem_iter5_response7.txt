def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 2's strict DDL gating and percentile normalization
    with Parent 1's successor-release awareness and novel duration-energy coupling.
    
    Structural improvements:
      - Strict DDL gating retained: all optimization components (energy, criticality) inactive unless slack >= 0.
      - Added successor-release term: rewards tasks whose descendants carry high cumulative upward_rank density,
        computed as rank-weighted remaining_work — avoids over-reliance on raw work alone.
      - Introduced joint duration-energy urgency term: (duration * energy)^p normalized and gated by DDL feasibility,
        capturing trade-off between fast-but-wasteful vs slow-but-efficient scheduling.
      - Removed all fragile interactions (uncertainty-slack, host-load-sensitivity) and sigmoid gates;
        uses only hard `slk >= 0` logic for interpretability and robustness.
      - All normalization uses same configurable percentile scheme for consistency.
      - No infinite loops, no I/O, no global state; fully deterministic and finite.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Bounded percentile-based normalization: uses configurable pctl_low–pctl_high range
    def percentile_normalize(x):
        x = np.asarray(x)
        p_low = np.percentile(x, PARAMS["pctl_low"])
        p_high = np.percentile(x, PARAMS["pctl_high"])
        rng = p_high - p_low
        # Ensure finite, non-zero range
        rng_safe = np.where(np.isfinite(rng) & (rng > eps), rng, eps)
        center = (p_low + p_high) / 2.0
        return (x - center) / (PARAMS["percentile_range_scale"] * rng_safe + eps)
    
    # Duration: execution + communication
    duration = exec_t + comm_t
    
    # === Slack processing: piecewise & risk-aware ===
    slack_norm = percentile_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling, ONLY if DDL-feasible (slack >= 0) ===
    inv_energy = 1.0 / (energy + eps)
    energy_active = np.where(slk >= 0, inv_energy, 0.0)
    energy_score = -PARAMS["energy_efficiency_weight"] * percentile_normalize(energy_active + eps)
    
    # === Criticality: strict DDL gate — only active when slack >= 0 ===
    rank_active = np.where(slk >= 0, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * percentile_normalize(rank_active + eps)
    
    # === Joint duration-energy urgency: captures cost of delaying efficient tasks ===
    # Only meaningful under DDL safety: (duration * energy)^p, normalized
    dur_energy_prod = (duration + eps) * (energy + eps)
    dur_energy_urgency = np.where(
        slk >= 0,
        np.power(dur_energy_prod, PARAMS["duration_energy_coupling_power"]),
        0.0
    )
    dur_energy_score = percentile_normalize(dur_energy_urgency + eps)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -percentile_normalize(wait_sat + eps)
    
    # === Successor-release load mitigation: reward unblocking of high-criticality descendants ===
    # Uses rank-weighted work as proxy for descendant criticality density
    weighted_successor_load = rank_active * (work + eps)
    successor_release_score = -PARAMS["successor_release_weight"] * percentile_normalize(weighted_successor_load + eps)
    
    # === Assemble final score: smaller = better ===
    # All components are finite, consistently normalized, and strictly gated by DDL feasibility
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_energy_score
        + wait_score
        + successor_release_score
    )
    
    # Final safeguard: replace NaN/inf with finite fallback using machine epsilon bounds
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
