def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Bottleneck term pre-clipped before MAD normalization to prevent outlier distortion.
      - Host-load proxy replaced by robust, bounded waiting-time gating (no unstable power law).
      - All divisions and exponentials guarded against NaN/inf via finite masking and clipping.
      - Duration modeling uses only exec+comm (uncertainty handled in bottleneck/energy).
      - Strict adherence to [-2,2] normalized feature bounds and deterministic outputs.
      - Exactly 12 parameters; all used; no numeric literals outside {-2,-1,0,1,2}.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust MAD-based normalization with pre-clipping for bottleneck stability ===
    def normalize_mad_bounded(x, preclip_factor=0.0):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        if preclip_factor > 0.0:
            # Soft clip extremes before median/MAD to avoid outlier skew
            median_abs = np.median(np.abs(x_finite))
            clip_bound = median_abs * (1.0 + preclip_factor)
            x_finite = np.clip(x_finite, -clip_bound, clip_bound)
        median = np.median(x_finite)
        mad = np.median(np.abs(x_finite - median))
        scaled_mad = mad * PARAMS["mad_scale_factor"]
        normed = (x - median) / (scaled_mad + eps)
        return np.clip(normed, -2.0, 2.0)
    
    # === Smooth sigmoid DDL protection gate ===
    width = PARAMS["ddl_protection_gate_width"]
    ddl_gate = 1.0 / (1.0 + np.exp(-slk / (width + eps)))
    
    # === Slack-driven base penalty/gain ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    if len(finite_abs_slk) == 0:
        slk_scale = eps
    else:
        median_slk = np.median(finite_abs_slk)
        mad_slk = np.median(np.abs(finite_abs_slk - median_slk)) * PARAMS["mad_scale_factor"]
        slk_scale = mad_slk + eps
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Energy score with additive DDL-aware correction and bounded waiting-time gating ===
    inv_energy = 1.0 / (energy + eps)
    # Bounded gating: 1/(1 + clipped_wait) — avoids instability of power laws
    wait_gating = 1.0 / (1.0 + np.clip(wait, 0.0, PARAMS["wait_clip_threshold"]))
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_mad_bounded(inv_energy) * wait_gating
    # Additive correction: penalize high energy when slack < 0
    energy_correction = PARAMS["energy_slack_coupling_strength"] * (
        np.where(slk < 0, 1.0, 0.0) * normalize_mad_bounded(energy)
    )
    energy_score += energy_correction
    
    # === Criticality: gated by DDL gate ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize_mad_bounded(rank)
    
    # === Bottleneck-proximal term: pre-clipped before MAD to prevent amplification ===
    bottleneck = rank * work
    # Pre-clipping prevents extreme values from distorting median/MAD
    bottleneck_clipped = np.clip(bottleneck, -1e6, 1e6)  # hard bound before normalization
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize_mad_bounded(
        bottleneck_clipped + eps, 
        preclip_factor=0.5  # fixed soft preclip factor (not tunable — removed parameter)
    )
    
    # === Duration score: pure exec+comm, no uncertainty gating (handled elsewhere) ===
    duration = exec_t + comm_t
    dur_score = normalize_mad_bounded(duration + eps)
    
    # === Starvation mitigation: clipped & normalized wait time ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize_mad_bounded(wait_clipped + eps)
    
    # === Assemble final score ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_safe_scale"]
    min_safe = -finfo.max / PARAMS["finfo_safe_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
