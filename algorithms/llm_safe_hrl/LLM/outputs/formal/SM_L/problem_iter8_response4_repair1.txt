def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining best practices:
      - Smooth sigmoid DDL gate (Parent 2) + uncertainty-amplified slack penalty (novel).
      - Bottleneck-proximal term (Parent 2) + energy-slack interaction (novel) for joint feasibility-efficiency tradeoff.
      - Median-based normalization universally applied with epsilon safeguards.
      - All components additive, deterministic, and bounded; no exponential terms or fragile thresholds.
      - Two novel interaction terms: (1) uncertainty × slack_penalty, (2) energy × slack_norm — both gated by feasibility logic.
      - ddl_gate_activation_threshold replaces hidden literal 0.5.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: median-based, epsilon-guarded, handles all-zero case ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            med = np.median(abs_x[finite_mask])
            scale = np.where(med > eps, med, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating and interactions ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 → ensures critical tasks dominate ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Base slack-driven penalty/gain (linear) ===
    slack_base = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Uncertainty-amplified slack penalty: increases urgency under high uncertainty when slack is negative ===
    # Only active when slack is negative — avoids over-penalizing safe tasks
    uncertainty_slack_boost = np.where(
        slk < 0,
        PARAMS["uncertainty_slack_coupling"] * uncert * np.abs(slack_norm),
        0.0
    )
    
    # === Combined slack penalty: base + uncertainty-modulated boost ===
    slack_penalty = slack_base + uncertainty_slack_boost
    
    # === Energy efficiency: inverse scaling, robustly normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Energy-slack interaction: penalizes high-energy tasks more when slack is tight (near zero/negative) ===
    # Encourages low-energy assignment exactly when deadlines are stressed
    energy_slack_interaction_term = PARAMS["energy_slack_interaction"] * normalize(energy) * np.abs(slack_norm)
    # Apply only when ddl_gate is active above threshold — replaces hidden 0.5
    energy_slack_score = np.where(ddl_gate > PARAMS["ddl_gate_activation_threshold"], energy_slack_interaction_term, 0.0)
    
    # === Criticality: smoothly gated by ddl_gate (not hard threshold), enabling graceful degradation ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Bottleneck-proximal term: identifies tasks whose delay blocks large descendant work on critical path ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck + eps)
    
    # === Duration-uncertainty coupling: robust linear blend of exec+comm and uncertainty ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear clip → prevents indefinite waiting without exponentials ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # All terms are additive and designed to preserve feasibility-first semantics:
    # slack_penalty dominates under violation; ddl_gate modulates criticality & energy-slack coupling
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + energy_slack_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
