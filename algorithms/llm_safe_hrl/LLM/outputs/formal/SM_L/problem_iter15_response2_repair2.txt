def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Bounded concave bottleneck coupling: (rank * work)^β with tunable β ∈ (0,1), improving robustness to scale outliers.
      - Starvation mitigation now gated by slack non-negativity — avoids competing with deadline urgency.
      - Removed fragile uncertainty-gated duration and redundant parameters (finfo_safe_scale dropped to meet 12-param limit).
      - Uses np.finfo-based inf/NaN replacement with fixed safe scaling (no param needed).
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants.
      - Final score is finite, shape-(N), deterministic, and satisfies all interface contracts.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust bounded normalization: median + MAD scale, then clip to [-2,2] ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        med = np.median(x_finite)
        dev = np.abs(x_finite - med)
        mad = np.median(dev) if len(dev) > 0 else eps
        scale = np.where(mad > eps, mad, eps)
        normed = (x - med) / (scale + eps)
        return np.clip(normed, -2.0, 2.0)
    
    # === Normalized slack for gating only ===
    slack_norm = normalize(slk)
    
    # === Slack-driven components: linear penalty/gain with explicit zero-crossing ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling → sharper preference for low marginal energy ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: only active when slack is non-negative AND above protection threshold ===
    rank_active = np.where(
        (slk >= 0) & (slack_norm >= PARAMS["ddl_protection_gate"]), 
        rank, 
        0.0
    )
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # === Duration score: pure (exec + comm), normalized — no unstable uncertainty coupling ===
    duration = exec_t + comm_t
    dur_score = normalize(duration + eps)
    
    # === Starvation mitigation: only active when slack ≥ 0 (avoids conflicting with urgency) ===
    wait_score = np.where(
        slk >= 0,
        -PARAMS["wait_starvation_weight"] * normalize(wait + eps),
        0.0
    )
    
    # === Bounded concave bottleneck term: (rank * work)^β / (|slack|+eps)^sensitivity, fully normalized ===
    # Concave coupling prevents explosion from large rank/work products while preserving ordering
    coupled_bottleneck_base = np.power(np.maximum(rank * work, eps), PARAMS["bottleneck_coupling_exponent"])
    slack_base = np.abs(slk) + eps
    bottleneck_raw = coupled_bottleneck_base / (np.power(slack_base, PARAMS["bottleneck_sensitivity"]) + eps)
    bottleneck_score = -PARAMS["criticality_weight"] * normalize(bottleneck_raw + eps)
    
    # === DDL protection gate: overrides duration dominance when slack is critically low ===
    protection_mask = slack_norm < PARAMS["ddl_protection_gate"]
    dur_score = np.where(protection_mask, dur_score * PARAMS["duration_suppression_factor"], dur_score)
    energy_score = np.where(protection_mask, energy_score * PARAMS["energy_amplification_factor"], energy_score)
    rank_score = np.where(protection_mask, rank_score * PARAMS["criticality_amplification_factor"], rank_score)
    bottleneck_score = np.where(protection_mask, bottleneck_score * PARAMS["criticality_amplification_factor"], bottleneck_score)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using finfo-safe fixed scaling (no param needed)
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1000.0
    min_safe = -finfo.max / 1000.0
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
