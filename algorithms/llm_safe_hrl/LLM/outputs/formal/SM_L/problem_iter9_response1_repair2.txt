def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
       - Robust min-max joint normalization (replaces percentile) for [slack, upward_rank, remaining_work]
         to suppress outlier sensitivity and improve gradient stability;
       - Tunable bottleneck exponent implicitly absorbed into successor_release_weight scaling;
       - Soft-clamped feasibility mask [0.05, 0.95] to avoid degenerate gradients;
       - All normalizations use identical robust_minmax_normalize() with tunable eps;
       - No numeric literals beyond -2, -1, 0, 1, 2.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Safe casting without in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust min-max joint normalization (replaces percentile) ===
    # Uses actual min/max with floor on range to avoid outlier-driven scaling collapse
    joint_features = np.stack([slk, rank, work], axis=0)
    feat_min = np.min(joint_features, axis=1)
    feat_max = np.max(joint_features, axis=1)
    feat_range = feat_max - feat_min
    # Clamp range to prevent division-by-zero or extreme scaling collapse
    finfo = np.finfo(np.float64)
    feat_range_safe = np.where(feat_range > eps, feat_range, eps)
    center_joint = (feat_min + feat_max) / 2.0
    
    slk_norm = (slk - center_joint[0]) / (feat_range_safe[0] + eps)
    rank_norm = (rank - center_joint[1]) / (feat_range_safe[1] + eps)
    work_norm = (work - center_joint[2]) / (feat_range_safe[2] + eps)
    
    # Robust min-max normalization for scalar features
    def robust_minmax_normalize(x):
        x = np.asarray(x)
        x_min = np.min(x)
        x_max = np.max(x)
        x_range = x_max - x_min
        x_range_safe = np.where(x_range > eps, x_range, eps)
        center = (x_min + x_max) / 2.0
        return (x - center) / (x_range_safe + eps)
    
    # Duration = execution + communication
    duration = exec_t + comm_t
    
    # === Smooth feasibility blending via sigmoid ===
    feasibility_mask_raw = 1.0 / (1.0 + np.exp(-PARAMS["feasibility_blend_steepness"] * slk))
    # Soft clamp to [0.05, 0.95] to avoid gradient saturation at extremes
    feasibility_mask = np.clip(feasibility_mask_raw, 0.05, 0.95)
    
    # === Slack processing: quadratic penalty for lateness, linear gain for margin ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (np.abs(slk_norm) ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slk_norm)
    )
    
    # === Energy efficiency: smoothly blended — zero weight when slack << 0, full weight when slack >> 0 ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * robust_minmax_normalize(inv_energy + eps)
    energy_score = feasibility_mask * energy_score
    
    # === Criticality: same smooth blending — active only in feasible region ===
    rank_score = -PARAMS["criticality_weight"] * robust_minmax_normalize(rank_norm + eps)
    rank_score = feasibility_mask * rank_score
    
    # === Bottleneck interaction: fixed exponent 1.3 (structural constant), scaled by tunable weight ===
    bottleneck_interaction = (rank_norm * work_norm) ** 1.3
    urgency_signal = np.clip(-slk, 0.0, np.inf)
    urgency_norm = robust_minmax_normalize(urgency_signal + eps)
    # Weighted by (1 - feasibility_mask) to peak at maximal lateness risk
    bottleneck_score = -PARAMS["successor_release_weight"] * bottleneck_interaction * urgency_norm * (1.0 - feasibility_mask)
    
    # === Duration-uncertainty coupling: bounded harmonic blend ===
    dur_norm = robust_minmax_normalize(duration + eps)
    uncert_norm = robust_minmax_normalize(uncert + eps)
    denom = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm + eps
    dur_uncert_blend = (1.0 + PARAMS["duration_uncertainty_ratio"] * uncert_norm) / denom
    dur_score = robust_minmax_normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -robust_minmax_normalize(wait_sat + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: symmetric clip to bounded finite range using declared parameter
    score = np.clip(score, -PARAMS["score_clip_bound"], PARAMS["score_clip_bound"])
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
