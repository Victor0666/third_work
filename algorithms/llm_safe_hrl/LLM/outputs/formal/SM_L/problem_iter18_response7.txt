def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaces piecewise-linear DDL mask with a *conditional sigmoid gate*: active only when slack < 0,
        avoiding dilution of urgency signals for non-violating tasks (per reflection).
      - Introduces dedicated `energy_ddl_coupling_exponent` — decoupled from bottleneck sharpening — 
        enabling independent tuning of deadline sensitivity in energy-aware prioritization.
      - Uses `np.finfo` for machine-precision epsilon safeguards instead of literal 0.001 or eps-based fallbacks.
      - All numeric literals remain in {-2,-1,0,1,2}; no hidden constants.
      - Preserves semantic-aware quantile normalization and starvation-aware wait ramp.
    """
    eps = np.finfo(np.float64).tiny
    N = len(min_exec_time)
    
    # Cast safely without in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile normalization: two variants for semantic fidelity ===
    # For signed/temporal features (slack, duration, wait, uncertainty): use abs-value quantile
    def normalize_abs_quantile(x, q=PARAMS["robust_normalization_quantile"]):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], q)
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # For non-negative structural features (rank, work, energy-derived): use value-domain quantile
    def normalize_pos_quantile(x, q=PARAMS["robust_normalization_quantile"]):
        x = np.asarray(x)
        finite_mask = np.isfinite(x) & (x >= 0)
        if np.any(finite_mask):
            x_clean = x[finite_mask]
            if len(x_clean) > 0:
                scale = np.quantile(x_clean, q)
                scale = np.where(scale > eps, scale, eps)
            else:
                scale = eps
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack: signed temporal → abs-quantile ===
    slack_norm = normalize_abs_quantile(slk)
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Conditional sigmoid DDL protection gate: active ONLY when slack < 0 ===
    # Smooth, differentiable, and avoids signal dilution for non-violating tasks
    sigmoid_gate = np.where(
        slk < 0,
        1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_sigmoid_slope"] * slk)),
        0.0
    )
    
    # === Energy efficiency: inv_energy is positive → use pos-quantile ===
    # Now uses dedicated `energy_ddl_coupling_exponent`, decoupled from bottleneck logic
    slack_magnitude = np.abs(slk) + eps
    slack_coupling_factor = np.power(slack_magnitude, -PARAMS["energy_ddl_coupling_exponent"])
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_pos_quantile(inv_energy) * sigmoid_gate * slack_coupling_factor
    
    # === Criticality: upward_rank ≥ 0 → pos-quantile, gated by DDL protection ===
    rank_score = -PARAMS["criticality_weight"] * normalize_pos_quantile(rank + eps) * sigmoid_gate
    
    # === Bottleneck term: rank * work ≥ 0 → pos-quantile, uses original successor_release_sharpness (not reused) ===
    # Note: `successor_release_sharpness` was removed from PARAMETER_SCHEMA to stay within 12 params;
    # replaced by dedicated `energy_ddl_coupling_exponent`, so bottleneck uses fixed exponent 1.0 for clarity and stability
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, 1.0)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize_pos_quantile(bottleneck_sharpened + eps)
    
    # === Duration & uncertainty: signed temporal → abs-quantile ===
    duration = exec_t + comm_t
    dur_norm = normalize_abs_quantile(duration + eps)
    uncert_norm = normalize_abs_quantile(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize_abs_quantile(dur_uncert_blend)
    
    # === Starvation mitigation: wait ≥ 0 → pos-quantile on clipped input ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize_pos_quantile(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with bounded finite values
    score = np.nan_to_num(score, nan=0.0, posinf=PARAMS["score_clipping_bound"], neginf=-PARAMS["score_clipping_bound"])
    score = np.clip(score, -PARAMS["score_clipping_bound"], PARAMS["score_clipping_bound"])
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
