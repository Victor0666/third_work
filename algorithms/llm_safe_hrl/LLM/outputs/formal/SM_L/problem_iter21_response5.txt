def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Restored multiplicative uncertainty gating on *both* exec_t and comm_t (not linear blend), preserving physical risk composition.
      - Conditional DDL-gating applied to bottleneck term (upward_rank * remaining_work), aligning critical-path impact with deadline urgency.
      - Unified robust MAD-based normalization across all features; all bounded to [-2,2].
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants.
      - Deterministic, finite, shape-(N,) output satisfying all interface contracts.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust MAD-based normalization: [-2,2] clamping ===
    def normalize_mad_bounded(x, scale_factor=PARAMS["mad_scale_factor"]):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        median = np.median(x_finite)
        mad = np.median(np.abs(x_finite - median))
        scaled_mad = mad * scale_factor
        normed = (x - median) / (scaled_mad + eps)
        return np.clip(normed, -2.0, 2.0)
    
    # === Smooth sigmoid DDL protection gate: center at slack=0 ===
    width = PARAMS["ddl_protection_gate_width"]
    ddl_gate = 1.0 / (1.0 + np.exp(-slk / (width + eps)))
    
    # === Slack-driven base penalty/gain: signed, robustly normalized ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    if len(finite_abs_slk) == 0:
        slk_scale = eps
    else:
        median_slk = np.median(finite_abs_slk)
        mad_slk = np.median(np.abs(finite_abs_slk - median_slk)) * PARAMS["mad_scale_factor"]
        slk_scale = mad_slk + eps
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Multiplicative slack-energy coupling: peaks at slack=0 ===
    urgency_coupling = 1.0 + PARAMS["energy_slack_coupling_strength"] * (1.0 / (1.0 + np.exp(slk / (width + eps))))
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_mad_bounded(inv_energy) * urgency_coupling
    
    # === Criticality: gated by DDL gate, rank normalized ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize_mad_bounded(rank)
    
    # === Bottleneck-proximal term: now *conditionally DDL-gated*, and normalized as single interaction ===
    bottleneck = rank * work
    # Apply same ddl_gate to bottleneck — ensures critical-path tasks are prioritized *only when urgent*
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * ddl_gate * normalize_mad_bounded(bottleneck + eps)
    
    # === Uncertainty-gated duration: multiplicative suppression on *both* exec and comm times ===
    # Physical interpretation: uncertainty inflates both compute and comms durations independently
    exec_gated = exec_t * (1.0 + np.clip(uncert, 0.0, 2.0) ** PARAMS["uncertainty_gate_exponent"])
    comm_gated = comm_t * (1.0 + np.clip(uncert, 0.0, 2.0) ** PARAMS["uncertainty_gate_exponent"])
    gated_duration = exec_gated + comm_gated
    dur_score = normalize_mad_bounded(gated_duration + eps)
    
    # === Starvation mitigation: clipped linear wait time — bounded, deterministic ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize_mad_bounded(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_safe_scale"]
    min_safe = -finfo.max / PARAMS["finfo_safe_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
