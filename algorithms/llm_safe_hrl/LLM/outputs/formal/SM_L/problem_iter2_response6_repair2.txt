def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's sharp deadline risk handling and inverse-energy scoring
    with Parent 1's robust fairness term and anti-starvation wait bias. Key improvements:
      - Quadratic slack penalty (Parent 2) + soft linear gain for positive slack
      - Inverse-energy scoring (1/(energy+eps)) for sharper low-energy preference
      - Criticality gated by slack >= -epsilon (not just > 0), improving boundary stability
      - Uncertainty-slack interaction applied *only* under negative slack (tighter coupling)
      - Duration fairness via tanh-scaled normalized duration (Parent 1) to avoid load skew
      - Wait-time exponential saturation (Parent 2) + linear fallback near zero for stability
      - All normalizations use mean-abs scale + epsilon for stability and boundedness
    Smaller score = higher priority.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast to float64 and ensure no in-place mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization: mean absolute value + epsilon, avoids median/MAD cost & outliers
    def normalize(x):
        x = np.asarray(x)
        scale = np.mean(np.abs(x)) + eps
        return x / scale
    
    # Duration: execution + communication
    duration = exec_t + comm_t
    norm_duration = normalize(duration + eps)
    
    # Slack processing: quadratic penalty for negative, linear gain for positive
    slack_norm = normalize(slk + eps)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # Inverse energy efficiency: higher priority for lower energy; clipped to avoid blowup
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # Criticality: activate upward_rank only when slack allows recovery (slk >= -epsilon)
    rank_active = np.where(slk >= -eps, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # Uncertainty-slack interaction: only amplify uncertainty under deadline stress (slk < 0)
    slack_stress_mask = (slk < 0).astype(float)
    norm_uncert = normalize(uncert + eps)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * norm_uncert * slack_stress_mask
    
    # Wait-time saturation: exponential decay with linear fallback for very small wait
    # Use linear scale near zero to avoid flat derivative and preserve ordering
    wait_linear_fallback = PARAMS["wait_linear_fallback_scale"] * wait
    wait_sat = np.where(
        wait < eps,
        wait_linear_fallback,
        1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    )
    wait_score = -normalize(wait_sat + eps)
    
    # Duration fairness: tanh-scaled deviation to penalize extremes symmetrically and bound [-1,1]
    fairness_term = np.tanh(norm_duration) * norm_duration
    
    # Final score: sum all components — smaller is better
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + unc_slack_interaction
        + wait_score
        + PARAMS["duration_fairness_strength"] * fairness_term
    )
    
    # Final safeguards: replace NaN/inf with finite values using machine epsilon bounds scaled by finfo_max_scale
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = finfo.min / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(
        score,
        nan=np.median(score) if np.any(np.isfinite(score)) else 0.0,
        posinf=max_safe,
        neginf=min_safe
    )
    
    # Ensure shape and finiteness
    assert score.shape == (N,), f"Expected (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values remain after sanitization"
    
    return score
