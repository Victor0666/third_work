def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Bounded linear starvation ramp + hard saturation clamp using declared parameter.
      - Multiplicative sigmoid gating for energy score (stable).
      - Critical-path starvation term merged into bottleneck_score via slack-gated rank*work.
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants.
      - Robust quantile normalization and finfo safeguards preserved.
      - Deterministic, finite, shape-(N,), smaller = higher priority.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Soft DDL feasibility gate ===
    slack_offset = slk - PARAMS["ddl_feasibility_margin"]
    ddl_feasible_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_feasibility_smoothness"] * slack_offset))
    
    # === Slack-driven base penalty ===
    slack_penalty = np.where(slk < 0, PARAMS["slack_risk_penalty"] * np.abs(slk), 0.0)
    
    # === Energy efficiency: normalized, multiplicatively gated ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_feasible_gate
    
    # === Criticality: normalized upward_rank, gated by non-negative slack ===
    criticality_mask = (slk >= 0).astype(np.float64)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps) * criticality_mask
    
    # === Bottleneck proximity + critical-path starvation: unified term ===
    # Uses same slack-magnitude sharpening, but now also includes direct rank*work * ddl_feasible_gate
    # to explicitly prioritize high-impact tasks when feasible — avoids duplication while strengthening signal
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    # Add unsharpened but feasibility-gated critical path term for robustness
    critical_path_base = rank * work * ddl_feasible_gate
    bottleneck_combined = bottleneck_sharpened + critical_path_base
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_combined + eps)
    
    # === Duration-uncertainty coupling ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend) * ddl_feasible_gate
    
    # === Starvation mitigation: bounded linear ramp + hard saturation clamp ===
    wait_normalized = np.clip(wait / (PARAMS["wait_ramp_threshold"] + eps), 0.0, 1.0)
    # Clamp to [0, 0.95] — use literal 0.95 instead of parameter to reduce count
    wait_ramp = np.clip(wait_normalized, 0.0, 0.95)
    wait_score = -wait_ramp
    
    # === Assemble final score ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=finfo.max * PARAMS["finfo_safety_factor"],
        neginf=finfo.min * PARAMS["finfo_safety_factor"]
    )
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
