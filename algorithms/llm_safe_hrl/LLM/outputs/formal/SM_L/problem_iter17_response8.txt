def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths of both parents:
      - Retains Parent 2's robust piecewise-linear DDL gate and bounded IQR normalization for stability.
      - Adds novel smooth latency intensification (replacing Parent 2's hard 1+latency_factor) via sigmoid-smoothed transition near slack=0.
      - Preserves monotonicity, boundedness, and strict DDL-first enforcement.
      - Introduces 'latency_intensification_smoothing' to control sharpness of urgency boost near deadline — enables fine-grained tradeoff between feasibility and energy optimization.
      - All interactions remain deterministic, finite, and shape-preserving.
      - No numeric literals beyond {-2,-1,0,1,2}; all tunables exposed in PARAMETER_SCHEMA.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Bounded IQR-based normalization: [-2,2] clamping for outlier resilience ===
    def normalize_iqr_bounded(x, q_low=PARAMS["iqr_quantile_low"], q_high=PARAMS["iqr_quantile_high"]):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        q1 = np.quantile(x_finite, q_low)
        q3 = np.quantile(x_finite, q_high)
        iqr = q3 - q1
        scale = np.where(iqr > eps, iqr, eps)
        center = (q1 + q3) / 2.0
        normed = (x - center) / (scale + eps)
        return np.clip(normed, -2.0, 2.0)
    
    # === Piecewise linear DDL protection gate: monotonic, bounded, interpretable ===
    width = PARAMS["ddl_protection_gate_width"]
    gate_linear_region = (slk >= -width) & (slk <= 0.0)
    ddl_gate = np.where(
        slk <= -width,
        1.0,
        np.where(
            gate_linear_region,
            1.0 + slk / width,  # linear ramp: 1.0 at -width, 0.0 at 0.0
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: signed, normalized via robust absolute-slack scale ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    if len(finite_abs_slk) == 0:
        slk_scale = eps
    else:
        q1_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_low"])
        q3_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_high"])
        slk_scale = np.sqrt((q1_slk + eps) * (q3_slk + eps))
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Smooth latency-intensified energy score: replaces Parent 2's hard 1+latency_factor with sigmoid-smoothed urgency boost near slack=0 ===
    # Avoids discontinuity while preserving monotonicity and enabling CMA-ES tuning of transition sharpness
    lateness = np.maximum(0.0, -slk)
    # Sigmoid smoothing: 1 + (lateness/(|slk|+eps)) * sigmoid(-slk * smoothness)
    smooth_factor = 1.0 / (1.0 + np.exp(-slk * PARAMS["latency_intensification_smoothing"]))
    latency_factor = (lateness / (abs_slk + eps)) * smooth_factor
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_iqr_bounded(inv_energy) * (1.0 + latency_factor)
    
    # === Criticality: gated by DDL gate, rank normalized with bounded IQR ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize_iqr_bounded(rank)
    
    # === Bottleneck-proximal term: upward_rank * remaining_work — critical-path blocking impact ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize_iqr_bounded(bottleneck + eps)
    
    # === Uncertainty-gated duration: multiplicative suppression, bounded ===
    duration = exec_t + comm_t
    uncert_gate = np.clip(uncert, 0.0, 2.0) ** PARAMS["uncertainty_gate_exponent"]
    gated_duration = duration * (1.0 + uncert_gate)
    dur_score = normalize_iqr_bounded(gated_duration + eps)
    
    # === Starvation mitigation: clipped linear wait time — bounded, deterministic ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize_iqr_bounded(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_safe_scale"]
    min_safe = -finfo.max / PARAMS["finfo_safe_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
