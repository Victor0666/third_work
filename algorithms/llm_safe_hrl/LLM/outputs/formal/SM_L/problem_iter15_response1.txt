def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with adaptive deadline gating and scenario-aware starvation mitigation.
    
    Key structural improvements:
    - Replaces fixed `ddl_protection_center` with task-specific adaptive offset: 
      `offset = adaptive_gate_offset * normalize(upward_rank)`, making deadline protection severity 
      proportional to critical-path importance — high-rank tasks get earlier gating activation.
    - Replaces global `wait_ramp_threshold` with piecewise linear starvation activation triggered only when 
      `ready_wait_time > quantile(slack, starvation_activation_quantile)` — ensures starvation response is 
      contextually aware of workflow's current deadline pressure, not absolute time.
    - Uses bounded min-max normalization *only* for `slack` and `uncertainty` (per reflection) to prevent rank inversion 
      under extreme outliers, while retaining quantile normalization for all other features for robustness.
    - All numeric literals are {-2,-1,0,1,2}; no hidden constants; all parameters referenced exactly once.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization (for most features) ===
    def normalize_quantile(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Bounded min-max normalization for slack and uncertainty only (per reflection) ===
    def normalize_mm(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if np.any(finite_mask):
            x_finite = x[finite_mask]
            x_min = np.min(x_finite)
            x_max = np.max(x_finite)
            range_val = x_max - x_min
            range_val = np.where(range_val > eps, range_val, eps)
            return (x - x_min) / range_val
        else:
            return np.zeros_like(x)
    
    # Apply mm-normalization to slack and uncertainty only
    slack_mm = normalize_mm(slk)
    uncert_mm = normalize_mm(uncert)
    
    # === Adaptive sigmoid DDL protection gate: center shifts per-task based on criticality ===
    # Task-specific offset = adaptive_gate_offset * normalized upward_rank
    rank_norm = normalize_quantile(rank + eps)
    adaptive_center = PARAMS["adaptive_gate_offset"] * rank_norm
    # Gate input: steepness * (adaptive_center - slack_mm)
    gate_input = PARAMS["ddl_protection_steepness"] * (adaptive_center - slack_mm)
    # Clip input to avoid overflow in exp
    finfo = np.finfo(float)
    gate_input_clipped = np.clip(gate_input, -np.log(finfo.max), np.log(finfo.max))
    ddl_gate = 1.0 / (1.0 + np.exp(-gate_input_clipped))
    
    # === Slack-driven base penalty/gain: signed, linear, monotonic ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_mm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_mm)
    )
    
    # === Energy efficiency: inverse marginal energy, quantile-normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_quantile(inv_energy)
    
    # === Criticality: smoothly gated by adaptive sigmoid ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize_quantile(rank + eps)
    
    # === Enhanced bottleneck term: upward_rank * remaining_work * (1 + uncertainty) * slack_magnitude_inv ===
    # Softplus ensures convex, stable amplification of high-pressure bottlenecks
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_base = rank * work * (1.0 + uncert) * slack_magnitude_inv
    bottleneck_sharpened = np.log1p(np.exp(bottleneck_base))  # softplus
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize_quantile(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: simplified bounded linear blend (quantile-normalized) ===
    duration = exec_t + comm_t
    dur_norm = normalize_quantile(duration + eps)
    uncert_norm = normalize_quantile(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize_quantile(dur_uncert_blend)
    
    # === Starvation mitigation: piecewise linear activation, scenario-aware threshold ===
    # Threshold = quantile(slack, starvation_activation_quantile) → adapts to workflow deadline pressure
    slack_finite = slk[np.isfinite(slk)]
    if len(slack_finite) > 0:
        threshold_val = np.quantile(slack_finite, PARAMS["starvation_activation_quantile"])
    else:
        threshold_val = 0.0
    # Activate only when wait exceeds threshold AND slack is finite
    starvation_active = (wait > threshold_val) & np.isfinite(slk)
    # Linear ramp: (wait - threshold) / (threshold + eps), capped at 1.0
    wait_ramp = np.clip((wait - threshold_val) / (threshold_val + eps), 0.0, 1.0)
    wait_score = -np.where(starvation_active, wait_ramp, 0.0)
    
    # === Host-load sensitivity: energy-per-duration ratio, feasibility-gated ===
    energy_per_duration = energy / (duration + eps)
    load_proxy = normalize_quantile(energy_per_duration + eps)
    load_score = -PARAMS["host_load_sensitivity"] * ddl_gate * load_proxy
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + load_score
    )
    
    # Final safeguard using np.finfo (allowed literal 2 per contract)
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 2.0
    min_safe = finfo.min / 2.0
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
