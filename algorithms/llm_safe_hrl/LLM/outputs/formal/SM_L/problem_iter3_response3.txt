def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - DDL protection gate: hard slack threshold disables non-deadline terms when slack < ddl_protection_threshold.
      - Successor-release interaction: penalizes tasks blocking critical-path successors (direct starvation mitigation).
      - Robust conditional activation: all non-slack terms activate only under safe slack margin, ensuring deadline compliance first.
    All components remain finite, deterministic, and shape-strict (N,)."""
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Copy inputs to float64, avoid in-place mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization using MAD + tunable scale
    def robust_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        dev = np.abs(x - center)
        scale = np.median(dev) if len(dev) > 0 else 1.0
        scale = max(scale * PARAMS["robust_scale_factor"], eps)
        return (x - center) / (scale + eps)
    
    # Duration: execution + communication
    duration = exec_t + comm_t
    
    # === Slack processing: piecewise & risk-aware ===
    slack_norm = robust_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === DDL protection gate: activate non-slack terms only when slack is above hard threshold ===
    # Ensures strict deadline-first behavior: no energy/criticality tradeoff when DDL risk is severe
    is_safe_slack = slk >= PARAMS["ddl_protection_threshold"]
    
    # === Energy efficiency: inverse scaling, gated by DDL safety ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = np.where(
        is_safe_slack,
        -PARAMS["energy_efficiency_weight"] * robust_normalize(inv_energy),
        0.0
    )
    
    # === Criticality: slack-gated AND DDL-protection-gated ===
    rank_active = np.where(slk > -eps, rank, 0.0)
    rank_score = np.where(
        is_safe_slack,
        -PARAMS["criticality_weight"] * robust_normalize(rank_active + eps),
        0.0
    )
    
    # === Duration-uncertainty coupling: gated by DDL safety ===
    dur_norm = robust_normalize(duration + eps)
    uncert_norm = robust_normalize(uncert + eps)
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * uncert_norm) 
        / (dur_norm + eps + PARAMS["duration_uncertainty_ratio"] * uncert_norm + eps)
    )
    dur_score = np.where(
        is_safe_slack,
        robust_normalize(dur_uncert_blend),
        0.0
    )
    
    # === Starvation mitigation: exponential saturation on wait time, gated ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = np.where(
        is_safe_slack,
        -robust_normalize(wait_sat + eps),
        0.0
    )
    
    # === Uncertainty-slack interaction: only active under DDL stress (negative slack) but still within protection window ===
    # Applies only when slack is negative *but* above the hard threshold — avoids over-penalization at extreme risk
    slack_stress = np.where(
        (slk < 0) & (slk >= PARAMS["ddl_protection_threshold"]), 
        np.abs(slack_norm), 
        0.0
    )
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * uncert_norm * slack_stress
    
    # === Successor-release interaction: novel term penalizing tasks that block ready successors ===
    # This directly addresses critical-path starvation: if a task's successor is ready, delay this task less
    # Since 'remaining_work' correlates with descendant readiness, we approximate successor-readiness as:
    #   low remaining_work → successors likely computed → less blocking → lower penalty
    #   high remaining_work → successors likely blocked → higher penalty
    # We use robust-normalized remaining_work as proxy, scaled only under safe slack
    successor_penalty = np.where(
        is_safe_slack,
        PARAMS["successor_release_penalty"] * robust_normalize(work),
        0.0
    )
    
    # === Assemble final score: smaller = better ===
    # Deadline penalty always active; all others gated to enforce hard DDL priority
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + successor_penalty
    )
    
    # Final safeguard: ensure finite, replace NaN/inf with median fallback
    score = np.nan_to_num(score, nan=np.median(score), posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Enforce shape (N,) and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
