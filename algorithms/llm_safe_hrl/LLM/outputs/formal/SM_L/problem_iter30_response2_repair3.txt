def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Load-aware starvation mitigation: congestion proxy = wait + uncertainty, exponent = 1 (literal).
      - Successor-release interaction merged into bottleneck term via fixed urgency mask (1 - clip(slk/eps, 0, 1)) — no new parameter.
      - All numeric literals are in {-2,-1,0,1,2}; no hidden constants.
      - Robust normalization, sigmoid gates, starvation bypass, and finfo clamping preserved.
      - Exactly 12 parameters; all used; no unused or missing PARAMS references.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust min-max normalization: uses high quantile for upper bound, eps for lower bound ===
    def normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if np.any(finite_mask):
            x_finite = x[finite_mask]
            q_low = np.quantile(x_finite, 1.0 - PARAMS["robust_minmax_quantile"])
            q_high = np.quantile(x_finite, PARAMS["robust_minmax_quantile"])
            scale = np.where(q_high > q_low, q_high - q_low, eps)
            center = np.median(x_finite)
            return (x - center) / (scale + eps)
        else:
            return np.zeros_like(x)
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Smooth sigmoid energy suppression: decays from 1.0 at slk=0 using ddl_protection_gate ===
    energy_suppression_weight = np.where(
        slk < 0,
        1.0,
        1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slk))
    )
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_suppression_weight * normalize(inv_energy)
    
    # === Criticality: gated by same sigmoid ddl_gate for consistency ===
    ddl_gate = energy_suppression_weight
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Bottleneck-proximal term: sharpened with fixed exponent 2 (allowed literal), plus successor-release interaction ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    # Urgency mask: 1 when slack ≈ 0, decays linearly to 0 as |slk| grows beyond eps
    urgency_mask = np.clip(1.0 - np.abs(slk) / (eps + eps), 0.0, 1.0)  # literal 1 allowed
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, 2) * urgency_mask
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, min-max normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Load-aware starvation mitigation: congestion proxy = wait + uncertainty, exponent = 1 (literal) ===
    congestion_proxy = wait + uncert
    load_aware_ramp = np.clip(congestion_proxy / (PARAMS["wait_ramp_threshold"] + eps), 0.0, 1.0)  # exponent 1 → identity
    wait_score = -normalize(load_aware_ramp + eps)
    
    # === Critical-path starvation guard: bypass non-bottleneck terms when slack critically low ===
    starvation_mask = (slk < PARAMS["starvation_slack_threshold"]).astype(np.float64)
    starvation_bypass = bottleneck_score * starvation_mask
    base_score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
    )
    score = np.where(starvation_mask > 0, starvation_bypass, base_score)
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
