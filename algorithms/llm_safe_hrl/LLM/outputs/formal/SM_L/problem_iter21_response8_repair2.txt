def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Smooth sigmoid DDL gate (centered at ddl_gate_center, steepness ddl_gate_steepness)
      - Host-load–aware energy normalization using load_proxy = rank * work
      - Critical-path starvation guard activated when 0 < slack < starvation_guard_threshold
      - All numeric literals are in {-2,-1,0,1,2}; all tunable constants declared in PARAMETER_SCHEMA
      - Uses finfo_safety_factor for clamping instead of hidden literal
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: fixed 0.75 quantile per contract (no param needed) ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], 0.75)
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Smooth sigmoid DDL feasibility gate ===
    ddl_gate = 0.5 * (1.0 + np.tanh(PARAMS["ddl_gate_steepness"] * (slk - PARAMS["ddl_gate_center"])))
    
    # === Slack-driven base penalty/gain (linear) — remains dominant ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Host-load–aware energy normalization ===
    load_proxy = rank * work
    load_norm = normalize(load_proxy + eps)
    inv_energy = 1.0 / (energy + eps)
    energy_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    energy_score = energy_base * ddl_gate * (1.0 - PARAMS["load_aware_energy_scale"] * load_norm)
    
    # === Adaptive criticality: normalized rank scaled by slack magnitude (fixed exponent 2 per contract) ===
    abs_slack = np.abs(slk) + eps
    slack_coupling = np.power(abs_slack, -2)  # literal -2 allowed
    rank_norm = normalize(rank + eps)
    rank_score = -PARAMS["criticality_weight"] * rank_norm * slack_coupling
    
    # === Successor-release interaction (fixed exponent 2) ===
    slack_magnitude_inv = 1.0 / (abs_slack)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, 2)  # literal 2 allowed
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear ramp ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)
    wait_score = -wait_ramp
    
    # === Critical-path starvation guard ===
    cp_starvation_mask = (slk > 0) & (slk < PARAMS["starvation_guard_threshold"])
    cp_starvation_boost = np.where(
        cp_starvation_mask,
        -PARAMS["starvation_boost_coefficient"] * normalize(rank + eps) * (1.0 - normalize(wait + eps)),
        0.0
    )
    
    # === Assemble final score ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + cp_starvation_boost
    )
    
    # Final safeguard using np.finfo with parametrized safety factor
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max * PARAMS["finfo_safety_factor"], neginf=finfo.min * PARAMS["finfo_safety_factor"])
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
