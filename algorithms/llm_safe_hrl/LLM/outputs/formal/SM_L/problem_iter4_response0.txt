def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three structural improvements:
      - Smooth sigmoid DDL protection gate replaces hard sign-based gating → eliminates discontinuity at slack=0.
      - Successor-release interaction: weighted remaining_work * urgency signal promotes unblocking critical paths.
      - Host-load-aware energy modulation: activates only under high combined uncertainty+slack stress, preventing over-correction.
    All normalizations use median absolute deviation (MAD) scaling for robustness against outliers.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: median absolute deviation (MAD) → outlier-resistant ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        # Use median-based scale for robustness
        med = np.median(abs_x)
        mad = np.median(np.abs(abs_x - med)) if np.all(np.isfinite(abs_x)) and med > eps else eps
        scale = mad + eps
        return x / scale
    
    # === Smooth DDL protection gate: sigmoid centered at slack=0 with tunable width ===
    # Replaces brittle sign() with continuous transition for stability near deadline boundary
    gate_center = 0.0
    gate_width = PARAMS["ddl_protection_gate_width"]
    ddl_gate = 1.0 / (1.0 + np.exp(-(slk - gate_center) / (gate_width + eps)))
    
    # === Slack-driven components: soft-clipped linear penalty/gain via gate interpolation ===
    # Interpolates between risk-penalty (slack<0) and urgency-gain (slack>0) smoothly
    slack_penalty_base = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * np.abs(slk)
    )
    # Apply smooth gate: full penalty near negative slack, reduced influence near/above zero
    slack_penalty = slack_penalty_base * ddl_gate
    
    # === Energy efficiency: inverse scaling with host-load-aware modulation ===
    inv_energy = 1.0 / (energy + eps)
    energy_norm = normalize(inv_energy)
    
    # Host-load interaction: amplifies energy preference only under high stress (negative slack & high uncertainty)
    stress_signal = np.abs(slk) * uncert
    stress_normalized = normalize(stress_signal)
    host_load_mod = np.where(
        stress_normalized > PARAMS["host_load_interaction_threshold"],
        1.0 + stress_normalized,
        1.0
    )
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_norm * host_load_mod
    
    # === Criticality: gated by smooth DDL protection (not binary), scaled by rank importance ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps) * ddl_gate
    
    # === Successor-release interaction: promotes tasks whose execution unlocks large remaining_work ===
    # Weighted by urgency: high remaining_work + negative slack → high priority
    urgency_factor = np.clip(-slk, 0.0, np.inf)  # only positive when slack < 0
    successor_release = PARAMS["successor_release_weight"] * normalize(work + eps) * normalize(urgency_factor + eps)
    # Add as *positive* contribution → reduces score (i.e., increases priority) for high-impact releases
    successor_score = -successor_release
    
    # === Duration-uncertainty coupling: bounded linear blend (stable & interpretable) ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: activates only under normalized deadline stress ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * (1.0 - ddl_gate)  # active where gate is low → i.e., strong negative slack
    )
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + successor_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values scaled by finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
