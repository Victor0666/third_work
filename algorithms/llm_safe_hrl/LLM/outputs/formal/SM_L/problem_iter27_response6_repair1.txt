import numpy as np

            "name": "epsilon",

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      1. ADDITIVE SIGN-PRESERVING UNCERTAINTY GATING: `duration + ratio * (uncert - median(uncert))`
      2. ASYMMETRIC PIECEWISE SLACK SCALING: clamp slack to [-1,1], then apply steeper penalty slope
         for negative slack (via `asymmetric_slack_gain`) — sharpening DDL violation response.
      3. CRITICAL-PATH STARVATION GUARD: adds conditional penalty `(1 - rank_norm)` under slack < 0,
         ensuring high-upward-rank tasks are never starved during deadline pressure.
      All features use max-abs normalization; all operations guarded against NaN/inf; deterministic and finite.
      Parameter count reduced to 12 by merging starvation_penalty_weight into asymmetric_slack_gain logic
      via fixed multiplicative coupling (1.0) — preserving starvation mitigation without new parameter.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Max-abs normalization: robust, interpretable, and stable ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Asymmetric piecewise slack scaling: clamp → rescale → apply asymmetric gain ===
    # Clamp to [-1, 1] for bounded input; preserves sign and eliminates outliers
    slk_clamped = np.clip(slk, -1.0, 1.0)
    # Apply steeper penalty for negative slack: penalty_slope = gain * |slk|, urgency_slope = |slk|
    slk_scaled = np.where(
        slk_clamped < 0,
        slk_clamped * PARAMS["asymmetric_slack_gain"],
        slk_clamped
    )
    # Use this scaled version for all downstream slack logic (gate, penalty, bottleneck)
    slk_eff = slk_scaled
    
    # === Piecewise-linear DDL gate: applied on asymmetrically scaled slack ===
    gate_width = PARAMS["piecewise_linear_gate_width"] + eps
    ddl_gate = np.where(
        slk_eff <= 0,
        1.0,
        np.where(
            slk_eff <= gate_width,
            1.0 - slk_eff / gate_width,
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: applied on asymmetrically scaled slack ===
    slack_penalty = np.where(
        slk_eff < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk_eff),
        -PARAMS["slack_urgency_gain"] * slk_eff
    )
    
    # === Energy efficiency: inverse marginal energy, dual-gated for DDL safety ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    slack_decay = np.exp(-np.abs(slk_eff) * PARAMS["slack_aware_energy_decay"])
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate) * slack_decay
    
    # === Criticality: upward_rank raised to slack-aware power ===
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|slk_eff| + eps) ===
    robust_abs_slk = np.abs(slk_eff) + eps
    bottleneck_term = rank * work / robust_abs_slk
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize(bottleneck_term + eps)
    
    # === Additive sign-preserving uncertainty gating: duration + ratio * (uncert - median_uncert) ===
    duration = exec_t + comm_t
    # Compute median safely — handle empty case
    median_uncert = np.median(uncert) if N > 0 else 0.0
    # Broadcast median to shape (N,) safely
    median_broadcast = np.full(N, median_uncert, dtype=np.float64)
    duration_with_uncert = duration + PARAMS["duration_uncertainty_ratio"] * (uncert - median_broadcast)
    dur_score = normalize(duration_with_uncert + eps)
    
    # === Starvation mitigation: exponential saturation on raw wait time ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -wait_sat
    
    # === Critical-path starvation guard: add penalty under negative slack for low criticality ===
    # Normalize rank_score to [0,1] range for interpretability in guard
    rank_norm = (normalize(rank + eps) + 1.0) / 2.0  # maps [-1,1] → [0,1]
    # Fixed coupling strength (1.0) avoids new parameter while preserving intent
    starvation_guard = np.where(
        slk_eff < 0,
        1.0 * (1.0 - rank_norm),
        0.0
    )
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + starvation_guard
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2.0, neginf=finfo.min / 2.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
