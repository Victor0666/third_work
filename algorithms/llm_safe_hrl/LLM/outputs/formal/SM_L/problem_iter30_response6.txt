def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaced hard threshold gating on energy with smooth sigmoid activation based on slack → avoids discontinuities & enables gradient-based tuning.
      - Introduced load-aware energy normalization: scales `min_incremental_energy` by quantile-normalized `upward_rank * remaining_work`, 
        so energy scoring reflects *relative contention pressure* on critical paths, not absolute joules.
      - Redesigned starvation mitigation as `wait_normalized / (|slack_norm| + eps)` scaled by `wait_pressure_ratio` → makes waiting urgency adapt to deadline pressure,
        ensuring fairness under tight deadlines and stronger mitigation when slack is abundant.
      - All normalizations use robust quantile scaling; no mean/median bias.
      - Slack penalty remains dominant to enforce hard deadline feasibility first.
      - No numeric literals beyond -2,-1,0,1,2; all thresholds/weights are parameters.
      - Uses np.finfo for safe clamping instead of hardcoded epsilons.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    slack_norm = normalize(slk)
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: smooth sigmoid activation (replaces brittle threshold) ===
    # Activates gradually as slack increases — differentiable, tunable, avoids artifacts
    energy_activation = 1.0 / (1.0 + np.exp(-PARAMS["energy_activation_steepness"] * slack_norm))
    inv_energy = 1.0 / (energy + eps)
    # Load-aware normalization: energy cost is weighted by critical path workload density
    load_density = rank * work
    load_norm = normalize(load_density + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * load_norm * energy_activation
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term ===
    # Prioritizes tasks blocking large workloads *especially* when slack is tight
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: adaptive wait-pressure ramp ===
    # Wait urgency now scales inversely with deadline pressure: more aggressive when slack is large
    wait_clipped = np.clip(wait, 0.0, 2.0 * np.abs(np.median(slk)) + eps)  # dynamic upper bound
    wait_normalized = normalize(wait_clipped + eps)
    wait_pressure = wait_normalized / (np.abs(slack_norm) + eps)  # relative to deadline tension
    wait_ramp = np.clip(PARAMS["wait_pressure_ratio"] * wait_pressure, 0.0, 1.0)
    wait_score = -wait_ramp
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
