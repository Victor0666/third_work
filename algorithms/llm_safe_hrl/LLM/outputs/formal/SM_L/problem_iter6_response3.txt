def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with MAD-based slack gating, adaptive successor coupling, and bounded sigmoid smoothing.
    
    Key structural improvements:
      - Replaces percentile slack margin with robust MAD-normalized slack gating for cross-seed stability.
      - Restores `successor_release_tightness_quantile`-driven adaptive successor coupling, strictly gated by `slk > -eps`.
      - Introduces bounded sigmoid smoothing on rank and energy signals to suppress outlier sensitivity and improve gradient flow for CMA-ES.
      - All numeric literals are {-2,-1,0,1,2}; epsilon and all tunables use PARAMS.
      - Uses `np.finfo` for safe inf/nan handling; no unbounded operations.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs to float64 without mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization using median and scaled MAD (MAD-based slack gating replaces percentile margin)
    def robust_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        mad = np.median(np.abs(x - center)) + eps
        scale = PARAMS["robustness_mad_scale"] * mad + eps
        return (x - center) / scale
    
    # MAD-based slack gating: use normalized slack for DDL feasibility and margin estimation
    norm_slk = robust_normalize(slk + eps)
    # Slack margin derived from MAD-scaled spread — stable across distributions
    slack_margin = np.abs(np.median(norm_slk)) + np.median(np.abs(norm_slk - np.median(norm_slk))) + eps
    
    # Slack penalty: quadratic for negative slack, linear gain for positive slack
    slack_penalty = np.zeros_like(slk)
    slack_penalty[slk < 0] = PARAMS["slack_risk_penalty"] * ((-slk[slk < 0]) / (slack_margin + eps)) ** 2
    slack_penalty[slk >= 0] = -PARAMS["slack_urgency_gain"] * (slk[slk >= 0] / (slack_margin + eps))
    
    # Bounded sigmoid-smoothed inverse energy (suppresses extreme outliers)
    inv_energy = 1.0 / (energy + eps)
    energy_active = np.where(slk > -eps, inv_energy, 0.0)
    # Sigmoid scaling: maps to [-1,1], then linearly rescales for balanced contribution
    energy_sigmoid = 2.0 * (1.0 / (1.0 + np.exp(-PARAMS["sigmoid_steepness"] * (robust_normalize(energy_active + eps))))) - 1.0
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_sigmoid
    
    # Bounded sigmoid-smoothed criticality (gated by DDL feasibility)
    rank_active = np.where(slk > -eps, rank, 0.0)
    norm_rank = robust_normalize(rank_active + eps)
    rank_sigmoid = 2.0 * (1.0 / (1.0 + np.exp(-PARAMS["sigmoid_steepness"] * norm_rank))) - 1.0
    rank_score = -PARAMS["criticality_weight"] * rank_sigmoid
    
    # Duration-uncertainty coupling via harmonic-weighted signal
    duration = exec_t + comm_t
    norm_duration = robust_normalize(duration + eps)
    norm_uncert = robust_normalize(uncert + eps)
    # Harmonic blending: avoids division-by-zero and amplifies synergy under high uncertainty
    dur_uncert_harmonic = (
        (norm_duration + eps) * (norm_uncert + eps) 
        / (norm_duration + norm_uncert + 2.0 * eps)
    )
    dur_score = robust_normalize(dur_uncert_harmonic)
    
    # Starvation mitigation with bounded saturation
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    norm_wait_sat = robust_normalize(wait_sat + eps)
    wait_score = -norm_wait_sat
    
    # Uncertainty-slack interaction under deadline stress
    slack_stress = np.where(slk < 0, np.abs(slk) / (slack_margin + eps), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * norm_uncert * slack_stress
    
    # Adaptive successor-release coupling: uses both quantiles, gated by DDL feasibility
    pos_slack = slk[slk > eps]
    if len(pos_slack) > 0:
        successor_slack_threshold = np.quantile(pos_slack, PARAMS["successor_release_tightness_quantile"])
        successor_slack_threshold = max(successor_slack_threshold, eps)
    else:
        successor_slack_threshold = 1.0
    
    rank_to_work_ratio = rank / (work + eps)
    valid_ratios = rank_to_work_ratio[rank_to_work_ratio > eps]
    if len(valid_ratios) > 0:
        ratio_threshold = np.quantile(valid_ratios, PARAMS["rank_to_work_ratio_quantile"])
    else:
        ratio_threshold = 0.0
    
    # Coupling only when task is DDL-feasible AND its earliest successor has tight release window
    tight_successor_proxy = np.where(
        (slk > -eps) & 
        (rank_to_work_ratio >= ratio_threshold + eps) & 
        (slk < successor_slack_threshold),
        1.0,
        0.0
    )
    successor_coupling = PARAMS["successor_coupling_strength"] * tight_successor_proxy
    
    # Final score: smaller = higher priority
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + successor_coupling
    )
    
    # Final safeguard: finite, shape-correct, deterministic
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
