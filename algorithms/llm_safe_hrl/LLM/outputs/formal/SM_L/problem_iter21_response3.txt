def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Bounded differentiable sigmoidal DDL gate using tanh (replaces piecewise linear) for smoother gradient flow and crisp boundary control.
      - Adaptive percentile-based normalization for temporal_risk_sharpened and bottleneck_term — improves outlier robustness.
      - Decoupled load-bottleneck coupling: two distinct terms — one active in deadline-feasible regime (ddl_gate > th), one in critical regime (ddl_gate <= th).
      - All features use only {-2,-1,0,1,2} literals; no unused parameters; fully deterministic and finite.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Adaptive percentile-based normalization: robust to outliers ===
    def normalize_adaptive(x, pctl=PARAMS["adaptive_percentile"]):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if np.any(finite_mask):
            # Use percentile instead of max-abs for outlier resilience
            valid_x = x[finite_mask]
            scale = np.percentile(np.abs(valid_x), min(99.9, max(1.0, pctl)))
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Sigmoidal DDL gate: tanh-based, bounded, differentiable, crisp near zero ===
    # Maps slack → [0,1], with steepness controlled by sigmoid_steepness
    # tanh(k * slack) ∈ (-1,1) → shift/scale to [0,1]; symmetric steep falloff
    k = PARAMS["sigmoid_steepness"]
    # Gate = 0.5 * (1 - tanh(k * slack)) → 1 when slack << 0, ~0 when slack >> 0, smooth transition
    ddl_gate = 0.5 * (1.0 - np.tanh(k * slk))
    
    # === Slack-driven base penalty/gain: linear, signed, monotonic ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: inverse marginal energy, attenuated by host-load-sensitive ddl_gate ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_adaptive(inv_energy)
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate)
    
    # === Criticality: upward_rank raised to slack-aware power ===
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize_adaptive(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|slack| + eps), adaptively normalized ===
    slack_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_term = rank * work * slack_inv
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize_adaptive(bottleneck_term + eps)
    
    # === Temporal risk: sharpened blend of exec+comm+uncert, adaptively normalized ===
    temporal_risk = exec_t + comm_t + uncert
    temporal_risk_sharpened = np.power(temporal_risk + eps, PARAMS["temporal_risk_sharpening"])
    dur_score = normalize_adaptive(temporal_risk_sharpened)
    
    # === Starvation mitigation: exponential saturation ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -wait_sat
    
    # === Decoupled load-bottleneck coupling: two gated components ===
    energy_per_duration = energy / (exec_t + comm_t + eps)
    load_proxy = normalize_adaptive(energy_per_duration + eps)
    
    # Feasible regime coupling: active when ddl_gate > threshold
    feasible_mask = ddl_gate > PARAMS["critical_regime_gate_threshold"]
    coupled_feasible = np.where(feasible_mask, load_proxy * bottleneck_term * ddl_gate, 0.0)
    
    # Critical regime coupling: active when ddl_gate <= threshold, uses inverted gate for urgency
    critical_mask = ~feasible_mask
    coupled_critical = np.where(critical_mask, load_proxy * bottleneck_term * (1.0 - ddl_gate), 0.0)
    
    coupled_feasible_score = -PARAMS["load_bottleneck_coupling_strength"] * normalize_adaptive(coupled_feasible + eps)
    coupled_critical_score = -PARAMS["load_bottleneck_coupling_strength"] * normalize_adaptive(coupled_critical + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + coupled_feasible_score
        + coupled_critical_score
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2, neginf=-finfo.max / 2)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
