def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with unified uncertainty-augmented duration, tanh-based urgency gating,
       and simplified robust normalization — no MAD fallback, no epsilon/finfo_max_scale bloat.
    
    Key structural improvements:
      - Replaces sigmoid DDL gating with **tanh-based urgency modulation**: tanh(slack_norm) provides smooth,
        bounded (-1,1) urgency signal — avoids vanishing gradients near zero slack and eliminates sigmoid scale tuning.
      - Introduces **unified uncertainty-augmented duration**: `exec_t + comm_t + PARAMS["duration_uncertainty_ratio"] * uncert`
        replaces separate duration/uncertainty terms and interaction — reduces parameter coupling and improves interpretability.
      - Removes redundant `epsilon` and `finfo_max_scale` parameters per reflection; uses `np.finfo` directly for safe clamping.
      - Simplifies normalization: pure z-score + sigma clipping only; drops fragile MAD fallback — sufficient for finite, 
        well-conditioned workflow features.
      - Slack penalty now directly uses raw slack (not normalized) for physical interpretability, then normalized once.
      - All numeric literals strictly ∈ {-2,-1,0,1,2}; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust z-score normalization with outlier clipping (±sigma) ===
    def normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x) & (np.abs(x) > eps)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        mu = np.mean(x_finite)
        std = np.std(x_finite) if len(x_finite) > 1 else eps
        z = (x - mu) / (std + eps)
        clip_bound = PARAMS["z_score_clip_sigma"]
        z_clipped = np.clip(z, -clip_bound, clip_bound)
        max_abs = np.maximum(np.max(np.abs(z_clipped)), eps)
        return z_clipped / (max_abs + eps)
    
    # === Unified uncertainty-augmented duration: physical, interpretable, single term ===
    dur_uncert = exec_t + comm_t + PARAMS["duration_uncertainty_ratio"] * uncert
    dur_score = normalize(dur_uncert + eps)
    
    # === Raw slack penalty (physical units) → normalized once → preserves urgency semantics ===
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.where(slk < 0, -slk, 0.0) +
        PARAMS["slack_urgency_gain"] * np.where(slk > 0, slk, 0.0)
    )
    slack_score = normalize(slack_penalty + eps)
    
    # === tanh-based urgency modulation: bounded (-1,1), symmetric, differentiable, no scale tuning ===
    # tanh(slack_norm) ≈ -1 for high lateness risk, ≈ 0 at deadline, ≈ +1 for large slack → direct urgency signal
    slack_norm = normalize(slk)
    urgency_gate = np.tanh(slack_norm)
    
    # === Energy efficiency: inverse marginal energy, gated by urgency (not sigmoid) ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * (1.0 + urgency_gate) * 0.5
    
    # === Criticality: upward_rank weighted by urgency gate — stronger boost when slack is tight (urgency_gate ≈ -1) ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank) * (1.0 - urgency_gate) * 0.5
    
    # === Bottleneck boost: activated only under deadline stress (urgency_gate < 0) AND non-negligible work ===
    pressure_base = rank * work
    pressure_active = np.where((urgency_gate < 0.0) & (work > eps), pressure_base, 0.0)
    pressure_score = -PARAMS["bottleneck_boost_weight"] * normalize(pressure_active + eps)
    
    # === Starvation mitigation: exponential saturation, normalized and inverted ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_score
        + energy_score
        + rank_score
        + pressure_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values using native finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1e6
    min_safe = -finfo.max / 1e6
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
