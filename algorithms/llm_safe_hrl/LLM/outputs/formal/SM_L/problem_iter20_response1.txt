def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's crisp DDL gate and max-abs stability with Parent 1's robust quantile normalization and temporal risk fusion.
    Key innovations:
      - Dual-mode normalization: quantile-based for temporal features (slack, merged risk) and max-abs for structural features (rank, work, energy),
        balancing outlier resilience and interpretability.
      - Temporal risk sharpening: power-transformed fused duration+uncertainty term improves discriminative capacity for high-risk tasks.
      - Reinstated robust quantile for slack and temporal risk to prevent over-sensitivity to outliers in fuzzy timing estimates.
      - All numeric literals strictly limited to {-2,-1,0,1,2}; no hidden constants.
      - Final score enforces strict finiteness and shape compliance.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Quantile normalization for signed/temporal features (slack, fused temporal risk) ===
    def normalize_quantile(x, q=PARAMS["robust_normalization_quantile"]):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], q)
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Max-abs normalization for non-negative structural features ===
    def normalize_max_abs(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack: quantile-normalized for outlier-robust deadline signaling ===
    slack_norm = normalize_quantile(slk)
    
    # === Slack-driven base penalty/gain: linear, signed ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * slack_norm
    )
    
    # === Piecewise-linear DDL gate (Parent 2): crisp, bounded, monotonic ===
    gate_width = PARAMS["piecewise_linear_gate_width"] + eps
    ddl_gate = np.where(
        slk <= 0,
        1.0,
        np.where(
            slk <= gate_width,
            1.0 - slk / gate_width,
            0.0
        )
    )
    
    # === Energy efficiency: inverse marginal energy, max-abs normalized, host-load gated ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_max_abs(inv_energy)
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate)
    
    # === Criticality: exponentiated rank, max-abs normalized ===
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize_max_abs(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|slack| + eps), max-abs normalized ===
    slack_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_term = rank * work * slack_inv
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize_max_abs(bottleneck_term + eps)
    
    # === Temporal risk: fused & sharpened (exec_t + comm_t + uncert)^p, quantile-normalized ===
    temporal_risk = exec_t + comm_t + uncert
    temporal_risk_sharp = np.power(temporal_risk + eps, PARAMS["temporal_risk_sharpening"])
    dur_score = normalize_quantile(temporal_risk_sharp)
    
    # === Starvation mitigation: exponential saturation ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2.0, neginf=finfo.min / 2.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
