def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths:
      - Smooth sigmoid DDL gate (Parent 2) for monotonic feasibility enforcement.
      - Median-based normalization with floor (Parent 2) for robustness on skewed/zero rank.
      - Bounded linear duration-uncertainty blend (Parent 2) for stable risk suppression.
      - Starvation exponent ramp (Parent 2) for stronger long-wait bias without hard clipping.
      - Novel energy-DDL coupling: multiplies energy score by sigmoid gate to defer low-energy tasks until deadline pressure arises — prevents premature low-energy selection that may harm DDL feasibility.
      - All normalizations and gates are finite, bounded, and use only {-2,-1,0,1,2} literals.
      - No hidden constants; all tunables declared in PARAMETER_SCHEMA.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust median-based normalization with explicit floor ===
    def normalize(x, floor=eps):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            med = np.median(abs_x[finite_mask])
            scale = np.maximum(med, floor)
        else:
            scale = floor
        return x / (scale + eps)
    
    # === Smooth sigmoid DDL protection gate: differentiable, monotonic, bounded ∈ [0,1] ===
    steep = PARAMS["ddl_sigmoid_steepness"]
    ddl_gate = 1.0 / (1.0 + np.exp(-steep * slk))
    
    # === Slack-driven base penalty/gain: signed, normalized via absolute median ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    slk_scale = np.median(finite_abs_slk) if len(finite_abs_slk) > 0 else eps
    slk_scale = np.where(slk_scale > eps, slk_scale, eps)
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse energy, normalized with floor, then gated by DDL feasibility ===
    inv_energy = 1.0 / (energy + eps)
    energy_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Coupling: only reward energy efficiency when DDL headroom exists (ddl_gate ≈ 1 → high urgency → uncoupled)
    # When ddl_gate ≈ 0 (plenty of slack), suppress energy signal to avoid premature scheduling
    energy_score = energy_base * (1.0 - PARAMS["energy_slack_coupling_strength"] * (1.0 - ddl_gate))
    
    # === Criticality: gated by smooth sigmoid, rank normalized with floor to avoid degeneracy ===
    rank_stable = rank + PARAMS["upward_rank_normalization_floor"]
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank_stable)
    
    # === Bottleneck-proximal term: upward_rank * remaining_work — critical-path blocking impact ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, median-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: sharpened power-law ramp — monotonic, bounded, stronger bias than linear ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_sharpened = np.power(wait_clipped + eps, PARAMS["starvation_exponent"])
    wait_score = -normalize(wait_sharpened)
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty dominates; other terms refine within feasibility-aware context
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
