    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Bounded successor-coupling term: adds robust multiplicative interaction `upward_rank * remaining_work`

        enforcing strict deadline feasibility-first ordering when lateness risk is extreme.
      - All slack-dependent logic now operates exclusively on robust_slk (power-normalized, signed) for consistency.
      - Removed unstable `risk_boost` (clipped uncertainty × penalty) — replaced by cleaner hard gating + successor coupling.
      - Preserves max-abs normalization, piecewise DDL gate, exponential decay above threshold, and all stable features.
      - No new parameters beyond the two structural additions; all numeric literals are {-2,-1,0,1,2}.
      - Deterministic, finite, shape-correct, and immune to NaN/inf via np.nan_to_num + machine-precision bounds.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)

    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Robust slack normalization: |slack|^p / (|slack|^p + eps) preserves sign via multiplication by sign(slk) ===
    abs_slk = np.abs(slk) + eps
    robust_slk_mag = np.power(abs_slk, PARAMS["robust_slack_normalization"])
    robust_slk_norm = robust_slk_mag / (robust_slk_mag + eps)
    slk_robust = robust_slk_norm * np.sign(slk)
    
    # === Piecewise-linear DDL gate: applied on robustly normalized slack ===
    gate_width = PARAMS["piecewise_linear_gate_width"] + eps
    ddl_gate = np.where(
        slk_robust <= 0,
        1.0,
        np.where(
            slk_robust <= gate_width,
            1.0 - slk_robust / gate_width,
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: applied on robust slack ===
    slack_penalty = np.where(

        PARAMS["slack_risk_penalty"] * np.abs(slk_robust),
        -PARAMS["slack_urgency_gain"] * slk_robust
    )

    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Hard gate: zero energy contribution if robust slack falls below threshold (strict feasibility-first)
    # Threshold derived from robust_slack_normalization: bounded in [-1.0, -0.5]
    energy_gating_threshold = -0.5 * (1.0 + PARAMS["robust_slack_normalization"])
    hard_gate_mask = (slk_robust < energy_gating_threshold)
    # Exponential decay only where slack > threshold (soft suppression in moderate risk zone)
    slack_decay = np.exp(-np.abs(slk_robust) * PARAMS["slack_aware_energy_decay"])
    slack_decay = np.where(hard_gate_mask, 0.0, slack_decay)
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate) * slack_decay
    
    # === Criticality: upward_rank raised to slack-aware power ===

    rank_score = -normalize(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|robust_slk| + eps) — monotonic, bounded ===

    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize(bottleneck_term + eps)
    
    # === Successor-coupling term: bounded multiplicative interaction capturing combined criticality & workload pressure ===
    # This directly mitigates starvation of high-rank, high-work successors even under positive slack

    # Reuse bottleneck_weight and apply fixed 0.5 scaling (allowed literal per contract)

    # === Starvation mitigation: exponential saturation on raw wait time ===

        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score

    # Instead: use finfo.max / 1000000.0 violates literal rule.
