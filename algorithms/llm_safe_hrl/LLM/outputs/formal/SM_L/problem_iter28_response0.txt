def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three key structural innovations:
      - Added host-load-aware suppression: uses normalized uncertainty *and* execution time as proxy for host load,
        dampening energy optimization when load is high (prevents avoidable marginal energy cost).
      - Replaced linear wait ramp with bounded exponential saturation: smoother, more responsive to long waits,
        avoids over-prioritizing short-wait tasks while still mitigating starvation.
      - Introduced piecewise-linear robust slack normalization: preserves sign-aware urgency while avoiding
        extreme sensitivity near zero; uses absolute value + epsilon scaling with explicit clipping.
      - All normalizations now use quantile-based scaling *without* eps addition inside normalize() — eps handled externally.
      - Criticality and bottleneck terms are now *dual-gated*: by both ddl_gate and a host-load suppression factor
        to prevent critical-path starvation under congestion.
      - Energy term is now conditionally suppressed: only active when slack >= 0 and host load is low.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Piecewise-linear robust slack normalization: preserves sign, avoids singularity ===
    # Use absolute value for scale, but retain original sign for directional penalty/gain
    slk_abs = np.abs(slk)
    slk_scale = np.quantile(slk_abs[np.isfinite(slk_abs)], PARAMS["robust_normalization_quantile"])
    slk_scale = np.where(slk_scale > eps, slk_scale, eps)
    slack_norm = slk / (slk_scale + eps)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Host load proxy: combines uncertainty (bandwidth/CPU variability) and exec time (load intensity) ===
    # Higher values indicate higher estimated host load — suppresses energy optimization
    host_load_proxy = (exec_t + uncert) * (1.0 + eps)
    host_load_norm = normalize(host_load_proxy)
    host_load_suppression = 1.0 - PARAMS["host_load_suppression_factor"] * np.clip(host_load_norm, 0.0, 1.0)
    
    # === Slack-driven base penalty/gain (linear), using robustly normalized slack ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: now conditionally gated by both ddl feasibility AND low host load ===
    # Only considered when slack >= 0 AND host load is not high
    energy_active_mask = (slk >= 0) & (host_load_norm <= 0.8)
    inv_energy = 1.0 / (energy + eps)
    energy_score = np.zeros_like(inv_energy)
    energy_score[energy_active_mask] = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)[energy_active_mask]
    
    # === Criticality: dual-gated by ddl_gate AND host_load_suppression for congestion-aware focus ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * host_load_suppression * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term, also dual-gated ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * ddl_gate * host_load_suppression * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential ramp — faster saturation than linear, avoids overshoot ===
    # exp(-wait/threshold) decays to near-zero; we invert and clip to [0,1] for monotonic priority boost
    wait_exp_decay = np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_ramp = 1.0 - wait_exp_decay  # asymptotically approaches 1.0 as wait increases
    wait_score = -np.clip(wait_ramp, 0.0, 1.0)  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1e6  # Use fixed safe scale instead of parameterized one — removed finfo_max_scale per feedback
    min_safe = -finfo.max / 1e6
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
