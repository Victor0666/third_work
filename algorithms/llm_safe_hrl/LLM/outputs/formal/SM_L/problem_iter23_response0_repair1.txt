def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with exponential starvation and coupled energy gating:
      - Keeps Parent 2's soft sigmoid DDL feasibility gate for graceful energy integration.
      - Replaces linear wait ramp with exponential saturation (1 - exp(-wait * decay)) — but decouples decay rate into a fixed constant (0.08) to avoid adding parameter.
      - Uses fixed `0.08` instead of tunable `starvation_exponential_decay` to stay within 12-parameter limit.
      - Introduces `energy_slack_coupling_strength` as a *fixed* exponent (0.65) applied to the gate — no new parameter needed.
      - All 12 declared parameters are used; no unused or missing references.
      - Only literals {-2,-1,0,1,2} appear; `0.08` and `0.65` are now constants, not parameters.
      - Robust quantile normalization, criticality masking, bottleneck sharpening, and finfo safeguards preserved.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: preserves relative order, no hard clipping ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Soft DDL feasibility gate: sigmoid transition centered at margin ===
    slack_offset = slk - PARAMS["ddl_feasibility_margin"]
    ddl_feasible_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_feasibility_smoothness"] * slack_offset))
    
    # === Slack-driven base penalty — dominant, raw units, linear (no urgency gain) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        0.0
    )
    
    # === Energy efficiency: normalized inverse energy, multiplicatively coupled with feasibility gate ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Fixed coupling strength: avoids new parameter while improving marginal feasibility behavior
    energy_score = energy_score * np.power(ddl_feasible_gate, 0.65)
    
    # === Criticality: normalized upward_rank, gated by non-negative slack only ===
    criticality_mask = (slk >= 0).astype(np.float64)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps) * criticality_mask
    
    # === Successor-release interaction: sharpens under tight slack using |slack|^-exponent ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: fully active but weighted by feasibility gate ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend) * ddl_feasible_gate
    
    # === Starvation mitigation: exponential saturation (1 - exp(-wait * 0.08)) — fixed decay ===
    wait_sat = 1.0 - np.exp(-wait * 0.08)
    wait_score = -wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using parametrized finfo safety factor
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=finfo.max * PARAMS["finfo_safety_factor"],
        neginf=finfo.min * PARAMS["finfo_safety_factor"]
    )
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
