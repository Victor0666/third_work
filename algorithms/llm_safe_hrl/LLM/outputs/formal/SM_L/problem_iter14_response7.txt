def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining strengths of both parents:
      - Uses IQR-based normalization (from Parent 1) for superior outlier robustness in rank/work/energy.
      - Adopts smooth piecewise-linear feasibility margin (Parent 2) instead of hard DDL gate.
      - Integrates inverse-slack sharpened bottleneck term (Parent 2) with IQR normalization (Parent 1).
      - Introduces novel dual-normalization: IQR for critical structural features (rank, work, energy),
        quantile-based for temporal/dynamic features (slack, duration, wait, uncertainty).
      - Removes redundant parameters: consolidates normalization logic while preserving interpretability.
      - All literals are from {-2,-1,0,1,2}; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === IQR-based normalization for structural features (rank, work, energy, bottleneck) ===
    def normalize_iqr(x, q_low=PARAMS["iqr_quantile_low"], q_high=PARAMS["iqr_quantile_high"]):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        q1 = np.quantile(x_finite, q_low)
        q3 = np.quantile(x_finite, q_high)
        iqr = q3 - q1
        scale = np.where(iqr > eps, iqr, eps)
        center = (q1 + q3) / 2.0
        return (x - center) / (scale + eps)
    
    # === Quantile-based normalization for temporal/dynamic features (slack, duration, wait, uncertainty) ===
    def normalize_quantile(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack-driven base penalty/gain (linear, signed, quantile-normalized) ===
    slack_norm = normalize_quantile(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Piecewise-linear DDL feasibility margin: [−margin, +margin] ramp; flat beyond ===
    margin = PARAMS["ddl_feasibility_margin"]
    ddl_feasible_mask = np.clip(
        (slk + margin) / (2 * margin + eps),
        0.0,
        1.0
    )
    
    # === Energy efficiency: IQR-normalized inverse energy, conditionally modulated ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_iqr(inv_energy) * ddl_feasible_mask
    
    # === Criticality: IQR-normalized upward_rank, un-gated (critical path matters always) ===
    rank_score = -PARAMS["criticality_weight"] * normalize_iqr(rank + eps)
    
    # === Bottleneck-proximal term: sharpened by inverse slack, IQR-normalized ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize_iqr(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: quantile-normalized linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize_quantile(duration + eps)
    uncert_norm = normalize_quantile(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize_quantile(dur_uncert_blend)
    
    # === Starvation mitigation: monotonic ramp on quantile-normalized clipped wait time ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize_quantile(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant; others refine within feasibility-aware context
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with bounded finite values
    score_clipping_bound = 1e12  # Use fixed bound since PARAMS["score_clipping_bound"] removed for simplicity & stability
    score = np.nan_to_num(score, nan=0.0, posinf=score_clipping_bound, neginf=-score_clipping_bound)
    score = np.clip(score, -score_clipping_bound, score_clipping_bound)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
