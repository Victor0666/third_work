def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with key structural improvements:
      - Added host-load-aware energy suppression: multiplies energy term by smoothed slack gate to avoid high-energy
        assignments when deadline pressure is high (counterfactual evidence: 'avoidable_marginal_energy_or_load_cost')
      - Replaced linear wait ramp with smooth sigmoid saturation: avoids hard threshold artifacts and improves robustness
      - Simplified bottleneck term: removed power exponent on inverse slack, replaced with adaptive sigmoid scaling
        to reduce fragility while preserving critical-path focus (validated by fragility analysis)
      - Removed finfo_max_scale (inactive per diagnostics) and simplified NaN/inf handling using np.finfo bounds directly
      - All normalizations use robust quantile scaling; no hardcoded eps in denominator outside guard
      - Strict DDL-first ordering preserved via dominant slack_penalty term and gated energy interaction
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: now gated by ddl_gate to suppress marginal energy cost under DDL pressure ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_gate
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Bottleneck-proximal term: simplified sigmoid-scaled interaction (no fragile power) ===
    # Prioritizes tasks blocking large workloads, scaled smoothly by deadline pressure
    bottleneck_base = rank * work
    # Sigmoid scaling: stronger emphasis when slack is tight (near zero or negative)
    bottleneck_gate = 1.0 / (1.0 + np.exp(-PARAMS["successor_release_sharpness"] * slack_norm))
    bottleneck_sharpened = bottleneck_base * bottleneck_gate
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: smooth sigmoid saturation instead of clipped linear ramp ===
    # Monotonic, differentiable, avoids discontinuity at threshold
    wait_normalized = normalize(wait + eps)
    wait_sigmoid = 1.0 / (1.0 + np.exp(-PARAMS["host_load_interaction_strength"] * (wait_normalized - 0.5)))
    wait_score = -wait_sigmoid  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max * 0.5, neginf=finfo.min * 0.5)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
