def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining DDL-first safety, successor-aware duration,
    inverse-energy efficiency, and robust median-based normalization.
    
    Key improvements:
      - Robust MAD-normalization using np.median(abs(x - median)) with tunable scale
      - Successor-release-aware duration: exec + comm + successor_work_coupling * remaining_work
      - Slack-gated criticality: upward_rank contributes only if slack >= -epsilon
      - Quadratic slack penalty for violation risk; linear gain for positive slack
      - Exponential wait saturation prevents runaway priority drift
      - Uncertainty-slack interaction only under deadline stress (slk < 0)
      - All operations guarded against NaN/inf/zero; deterministic & finite output
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization using median absolute deviation (MAD) with tunable scale
    def robust_normalize(x):
        x = np.asarray(x)
        med = np.median(x)
        dev = x - med
        mad = np.median(np.abs(dev))
        scale = PARAMS["robust_mad_scale"] * (mad + eps)
        return dev / scale
    
    # Duration: include successor workload to anticipate downstream resource pressure
    duration = exec_t + comm_t + PARAMS["successor_work_coupling"] * work
    norm_duration = robust_normalize(duration + eps)
    
    # Energy: inverse efficiency — strongly favors low marginal energy
    inv_energy = 1.0 / (energy + eps)
    norm_energy = robust_normalize(inv_energy)
    energy_score = -PARAMS["energy_efficiency_weight"] * norm_energy
    
    # Slack: piecewise — quadratic penalty for negative, linear gain for positive
    norm_slack = robust_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (norm_slack ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(norm_slack)
    )
    
    # Criticality: gated by slack feasibility (only active when slack allows breathing room)
    rank_active = np.where(slk >= -eps, rank, 0.0)
    norm_rank = robust_normalize(rank_active + eps)
    rank_score = -PARAMS["criticality_weight"] * norm_rank
    
    # Duration-uncertainty coupling: weighted harmonic blend for stable timing signal
    norm_uncert = robust_normalize(uncert + eps)
    # Harmonic blend avoids division-by-zero; bounded between ~0 and ~2
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * norm_uncert) 
        / (np.abs(norm_duration) + eps + PARAMS["duration_uncertainty_ratio"] * norm_uncert + eps)
    )
    dur_score = robust_normalize(dur_uncert_blend)
    
    # Starvation mitigation: exponential saturation on wait time
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    norm_wait = robust_normalize(wait_sat + eps)
    wait_score = -norm_wait
    
    # Uncertainty-slack interaction: only activated under deadline stress (slk < 0)
    slack_stress = np.where(slk < 0, np.abs(norm_slack), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * norm_uncert * slack_stress
    
    # Assemble final score: smaller = higher priority
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguards: ensure finite, shape (N,), and deterministic
    score = np.nan_to_num(
        score,
        nan=np.finfo(float).smallest_subnormal,
        posinf=np.finfo(float).max,
        neginf=-np.finfo(float).max
    )
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
