def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Restored max-abs normalization (per reflection) for stability across all features.
      - Reintroduced `duration_uncertainty_ratio` as a robust linear blend — critical for fuzzy deadline handling.
      - Added `slack_aware_energy_decay`: multiplicative exponential attenuation of energy term under high lateness risk,
        replacing linear gating for smoother, physics-aligned suppression of energy optimization when deadlines dominate.
      - All features normalized via max-abs; no quantile, softplus, or unstable transforms.
      - Criticality uses exponentiated rank, bottleneck uses rank*work/(|slack|+eps), starvation uses exp(-wait/theta).
      - No unused parameters; all numeric literals are {-2,-1,0,1,2}; deterministic and finite.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Max-abs normalization: robust, interpretable, and stable ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Piecewise-linear DDL gate (restored per self-reflection): crisp, bounded, monotonic ===
    # Gate = 1.0 for slack <= 0, linear ramp from 1→0 over [0, width], 0 beyond width
    gate_width = PARAMS["piecewise_linear_gate_width"] + eps
    ddl_gate = np.where(
        slk <= 0,
        1.0,
        np.where(
            slk <= gate_width,
            1.0 - slk / gate_width,
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: linear, signed, monotonic ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: inverse marginal energy, gated by both ddl_gate AND slack-aware exponential decay ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Dual gating: linear ddl_gate * exponential slack decay for stronger, smoother suppression under risk
    slack_decay = np.exp(-np.abs(slk) * PARAMS["slack_aware_energy_decay"])
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate) * slack_decay
    
    # === Criticality: upward_rank raised to slack-aware power ===
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|slack| + eps) — monotonic, bounded, intuitive ===
    slack_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_term = rank * work * slack_inv
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize(bottleneck_term + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend (reinstated per reflection) ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2.0, neginf=finfo.min / 2.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
