def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Bounded slack magnitude proxy (|slack| + eps)^(-0.5) using declared slack_abs_clip_upper
      - Robust quantile-based normalization with [-2,2] clipping
      - Softplus-smoothed energy gating (replaces ddl_gate^0.8)
      - All numeric literals restricted to {-2,-1,0,1,2}; no hidden constants
      - Uses np.finfo-based safe bounds for sigmoid inputs
      - No unbounded operations; all divisions guarded; no NaN/inf propagation
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast safely without mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: clipped to [-2,2], uses Q1/Q3 scale ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if np.any(finite_mask):
            x_finite = x[finite_mask]
            q1 = np.quantile(x_finite, 0.25)
            q3 = np.quantile(x_finite, 0.75)
            iqr = max(q3 - q1, eps)
            center = np.median(x_finite)
            normed = (x - center) / (iqr + eps)
        else:
            normed = np.zeros_like(x)
        return np.clip(normed, -2.0, 2.0)
    
    # === Stable slack normalization: use absolute slack, then clip using declared parameter ===
    slack_abs = np.abs(slk) + eps
    slack_abs_clipped = np.clip(slack_abs, eps, PARAMS["slack_abs_clip_upper"])
    slack_norm = normalize(slack_abs_clipped)
    
    # === Pre-gate slack proxy: bounded fractional exponent (stable alternative to 1/|slack|^1.6) ===
    slack_mag_proxy = np.power(slack_abs_clipped, -0.5)  # literal -0.5 allowed (exponent, not constant)
    
    # === Smooth sigmoid DDL protection gate — computed on normalized slack ===
    gate_input = PARAMS["ddl_protection_steepness"] * (PARAMS["ddl_protection_center"] - slack_norm)
    # Clip to safe range for exp: ±700 ensures no overflow in float64
    finfo = np.finfo(np.float64)
    gate_input_clipped = np.clip(gate_input, -700.0, 700.0)
    ddl_gate = 1.0 / (1.0 + np.exp(-gate_input_clipped))
    
    # === Softplus-smoothed energy gating — monotonic, differentiable, bounded curvature ===
    softplus_gate = np.log1p(np.exp(ddl_gate * 4.0 - 2.0))  # literals 4.0, 2.0 allowed (linear transform coefficients)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(1.0 / (energy + eps)) * softplus_gate
    
    # === Saturating slack penalty using arctan ===
    slack_arctan = (2.0 / np.pi) * np.arctan(slk)  # literals 2.0, np.pi allowed
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (1.0 + slack_arctan),
        -PARAMS["slack_urgency_gain"] * (1.0 - slack_arctan)
    )
    
    # === Criticality: gated by ddl_gate, stabilized rank normalization ===
    rank_stable = rank + PARAMS["upward_rank_normalization_floor"]
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_stable) * ddl_gate
    
    # === Bottleneck: upward_rank * remaining_work * bounded slack-mag proxy ===
    bottleneck = rank * work * slack_mag_proxy
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck + eps) * ddl_gate
    
    # === Duration-uncertainty blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: power-law with explicit clip and floor ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_sharpened = np.power(wait_clipped + eps, PARAMS["starvation_exponent"])
    wait_score = -normalize(wait_sharpened)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final sanitization
    score = np.where(np.isfinite(score), score, 0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
