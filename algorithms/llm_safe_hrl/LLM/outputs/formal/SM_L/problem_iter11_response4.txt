def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining strengths of both parents:
      - Retains Parent 2's quantile-based robust normalization and host-load-aware energy modulation.
      - Integrates Parent 1's smooth ddl_gate activation centered at slack=0 (via symmetric sigmoid).
      - Introduces novel *uncertainty-slack coupling exponent* for more expressive joint risk modeling.
      - Adds *release pressure saturation* to prevent dominance under ultra-tight deadlines.
      - Replaces static wait ramp with *saturating linear ramp*, preserving monotonicity while bounding influence.
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants.
      - Deterministic, finite, shape-compliant, and fully parameterized.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Symmetric sigmoid DDL protection gate: smooth activation centered at slack=0 (Parent 1 style) ===
    # More balanced than asymmetric gate — critical for fuzzy deadline boundaries
    gate_width = PARAMS["ddl_protection_gate"] + eps
    ddl_gate = 1.0 / (1.0 + np.exp(-slack_norm / gate_width))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: conditionally modulated by uncertainty-slack coupling ===
    # High uncertainty *and* tight/negative slack reduces energy priority — prevents risky low-energy choices
    slack_magnitude = np.abs(slk) + eps
    unc_slk_coupling = PARAMS["host_load_interaction_weight"] * normalize(uncert) * (1.0 / slack_magnitude)
    # Modulate energy score: stronger suppression when coupling is high
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    energy_score = energy_score * (1.0 - np.clip(unc_slk_coupling, 0.0, 1.0))
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term with inverse slack exponentiation ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    # Apply saturation to avoid explosion under extreme deadlines
    bottleneck_saturated = np.clip(bottleneck_sharpened, 0.0, np.inf) * PARAMS["release_pressure_saturation"]
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_saturated + eps)
    
    # === Novel uncertainty-slack-coupled urgency term: exponentiated joint signal ===
    # Captures multiplicative risk: (uncertainty)^α × |normalized slack|^β → tuned via single exponent
    coupled_urgency = (
        np.power(normalize(uncert) + eps, PARAMS["uncertainty_slack_coupling_exponent"]) *
        np.power(np.abs(slack_norm) + eps, PARAMS["uncertainty_slack_coupling_exponent"])
    )
    coupled_score = -PARAMS["host_load_interaction_weight"] * normalize(coupled_urgency)
    
    # === Starvation mitigation: saturating linear ramp (monotonic & bounded) ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + coupled_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with safe finite bounds using tunable finfo scaling
    finfo = np.finfo(np.float64)
    # Compute max_safe = finfo.max / (2 ** finfo_safety_scale)
    max_safe = finfo.max / np.power(2.0, PARAMS["finfo_safety_scale"])
    min_safe = -max_safe
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
