def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Conditional criticality gate: upward_rank exponentiation *only* when slack <= threshold (no softening — hard activation per reflection)
      - Host-load-sensitive energy gating (reinstated): suppresses energy term when ready_wait_time is large → decouples urgency from load pressure
      - Unified bottleneck term gated by same slack threshold → avoids overloading non-urgent paths
      - Removed redundant clipping parameters: rely on robust normalization + np.finfo-based safeguards instead
      - All numeric literals strictly limited to -2,-1,0,1,2; no unbounded loops or state
      - Uses np.finfo for machine-precision epsilon fallbacks where needed
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust max-abs normalization using machine epsilon fallback ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            # Use np.finfo for safe minimal scale bound
            finfo = np.finfo(np.float64)
            scale = np.where(scale > eps, scale, finfo.tiny)
        else:
            finfo = np.finfo(np.float64)
            scale = finfo.tiny
        return x / (scale + eps)
    
    # === Hard DDL activation gate: binary-like but numerically smooth via clip ===
    # Activates full criticality & bottleneck logic only when slack <= threshold
    slack_threshold = PARAMS["criticality_threshold_slack"]
    ddl_pressure = np.where(slk <= slack_threshold, 1.0, 0.0)
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: gated by host load (not DDL) — reinstated per reflection ===
    # Reduces energy emphasis when task has waited long → avoids starving low-energy but high-wait tasks
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    wait_norm = normalize(wait + eps)
    host_load_gate = np.clip(wait_norm, 0.0, 1.0)  # bounded [0,1] load proxy
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * host_load_gate)
    
    # === Criticality: hard-gated exponentiation — only active under deadline pressure ===
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize(rank_powered) * ddl_pressure
    
    # === Bottleneck term: also hard-gated by ddl_pressure → avoids spurious coupling when slack is ample ===
    slack_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_term = rank_powered * work * slack_inv
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize(bottleneck_term + eps) * ddl_pressure
    
    # === Duration-uncertainty coupling: bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation — now scaled by ddl_pressure to avoid over-compensation ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -wait_sat * ddl_pressure  # only mitigates starvation *under pressure*
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with safe finite values using machine limits
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=-finfo.max)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
