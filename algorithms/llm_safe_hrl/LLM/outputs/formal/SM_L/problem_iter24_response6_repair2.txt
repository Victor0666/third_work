def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Smooth sigmoidal DDL gate using tunable steepness (no hidden literals).
      - Slack-aware criticality modulation: upward_rank scaled by (1 + boost * slack_decay) before exponentiation.
      - Conditional starvation gating: wait_saturation only active when slack > 0.
      - All features max-abs normalized; dual-gated energy scoring with robust scaling; duration-uncertainty blend retained.
      - No unused parameters; all numeric literals are {-2,-1,0,1,2}; deterministic and finite.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Max-abs normalization: robust, interpretable, and stable ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Smooth sigmoidal DDL gate: tunable steepness, no hidden literals ===
    # Uses PARAMS["sigmoid_steepness"] instead of hardcoded 8.0 or 2.0
    gate_center = 0.0
    sigmoid_input = PARAMS["sigmoid_steepness"] * (gate_center - slk)
    # Clamp input to avoid overflow
    finfo = np.finfo(np.float64)
    sigmoid_input = np.clip(sigmoid_input, -np.log(finfo.max), np.log(finfo.max))
    ddl_gate = 1.0 / (1.0 + np.exp(-sigmoid_input))
    
    # === Slack-driven base penalty/gain: linear, signed, monotonic ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: inverse marginal energy, robustly scaled and dual-gated ===
    inv_energy = 1.0 / (energy + eps)
    energy_norm = normalize(inv_energy) * PARAMS["robust_energy_normalization_scale"]
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_norm
    # Dual gating: sigmoid ddl_gate * exponential slack decay * host-load sensitivity
    slack_decay = np.exp(-np.abs(slk) * PARAMS["slack_aware_energy_decay"])
    host_load_gate = (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate)
    energy_score = energy_score * host_load_gate * slack_decay
    
    # === Slack-aware criticality modulation: boost rank under deadline pressure ===
    rank_boosted = rank * (1.0 + PARAMS["slack_aware_criticality_boost"] * slack_decay)
    rank_powered = np.power(rank_boosted + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|slack| + eps) — monotonic, bounded, intuitive ===
    slack_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_term = rank * work * slack_inv
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize(bottleneck_term + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Conditional starvation mitigation: only when slack > 0 (non-urgent phase) ===
    wait_sat = np.where(
        slk > 0,
        1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps)),
        0.0
    )
    wait_score = -wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2.0, neginf=finfo.min / 2.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
