def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with:
      - Smooth sigmoidal energy gating centered at −ddl_feasibility_margin (replaces piecewise).
      - Quantile-based normalization for all features to resist outliers.
      - Clamped inverse slack sharpening on normalized slack for scale-invariant bottleneck coupling.
      - Exponential starvation saturation with tunable threshold (no exponent parameter — fixed curvature).
      - All numeric literals are {-2,-1,0,1,2}; no hidden constants; deterministic and finite.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    slk_norm = normalize(slk)
    rank_norm = normalize(rank + eps)
    work_norm = normalize(work + eps)
    exec_norm = normalize(exec_t + eps)
    comm_norm = normalize(comm_t + eps)
    energy_norm = normalize(energy + eps)
    wait_norm = normalize(wait + eps)
    uncert_norm = normalize(uncert + eps)
    
    # === Slack-driven base penalty/gain (linear, signed) ===
    slack_penalty = np.where(
        slk_norm < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk_norm),
        -PARAMS["slack_urgency_gain"] * slk_norm
    )
    
    # === Smooth energy attenuation gate: sigmoid centered at −ddl_feasibility_margin ===
    gate_center = -PARAMS["ddl_feasibility_margin"]
    gate_slope = 2.0  # Fixed steepness: avoids extra param, ensures stability & interpretability
    energy_gate = 1.0 / (1.0 + np.exp(gate_slope * (slk_norm - gate_center)))
    
    # === Energy efficiency: inverse marginal energy, gated smoothly ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * energy_gate
    
    # === Criticality: always active, normalized ===
    rank_score = -PARAMS["criticality_weight"] * rank_norm
    
    # === Successor-release interaction: clamped |slk_norm|⁻ˢʰᵃʳᵖⁿᵉˢˢ × rank × work ===
    slack_magnitude_inv = 1.0 / (np.abs(slk_norm) + eps)
    slack_magnitude_inv_clamped = np.clip(slack_magnitude_inv, 0.0, 1.0 / eps)
    bottleneck_sharpened = rank_norm * work_norm * np.power(slack_magnitude_inv_clamped, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential saturation (fixed curvature) ===
    wait_saturation = 1.0 - np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_score = -normalize(wait_saturation + eps)  # monotonic, smooth, no extra exponent param
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finite_safeguard_scale"]
    min_safe = -finfo.max / PARAMS["finite_safeguard_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
