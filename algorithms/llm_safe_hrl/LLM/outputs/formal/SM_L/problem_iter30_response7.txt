def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Restored `slack_urgency_gain` for balanced prioritization under tight-but-feasible deadlines.
      - Smooth sigmoidal DDL feasibility gate (not binary) to preserve CMA-ES gradient near slack=0.
      - Successor-aware criticality: modulates upward_rank by normalized mean successor slack (simulated via input slack as proxy for local context).
      - All features max-abs normalized; no literals beyond {-2,-1,0,1,2}.
      - Deterministic, finite, shape-correct, and DDL-hard-constraint safe.
      - Explicit starvation weight and bottleneck power decoupled (no reuse of same param for different roles).
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Max-abs normalization: robust, interpretable, and stable ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Robust slack normalization: |slack|^p / (|slack|^p + eps), sign-preserving ===
    abs_slk = np.abs(slk) + eps
    robust_slk_mag = np.power(abs_slk, PARAMS["robust_slack_normalization"])
    robust_slk_norm = robust_slk_mag / (robust_slk_mag + eps)
    slk_robust = robust_slk_norm * np.sign(slk)
    
    # === Smooth DDL feasibility gate: sigmoidal (not binary) — preserves gradient near zero slack ===
    ddl_feasibility_gate = 1.0 / (1.0 + np.exp(-slk_robust * PARAMS["criticality_slack_modulation"]))
    
    # === Slack-driven base penalty/gain: restored urgency gain for positive slack (tight-but-feasible regime) ===
    slack_penalty = np.where(
        slk_robust < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk_robust),
        -PARAMS["slack_urgency_gain"] * slk_robust
    )
    
    # === Energy efficiency: inverse marginal energy, gated by smooth feasibility (not hard binary) ===
    inv_energy = 1.0 / (energy + eps)
    energy_score_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    energy_score = energy_score_base * ddl_feasibility_gate
    
    # === Criticality: upward_rank powered, then modulated by both slack feasibility AND successor readiness ===
    # Simulate successor slack influence: use current task's slack as local proxy (in absence of explicit successor array);
    # stronger reduction when successors are *less urgent* (i.e., higher slack) → reduces over-committing to critical paths that aren't time-critical downstream
    successor_slack_influence = 1.0 - PARAMS["successor_slack_sensitivity"] * np.clip(normalize(slk), 0.0, 1.0)
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score_unmod = -normalize(rank_powered)
    rank_score = rank_score_unmod * ddl_feasibility_gate * successor_slack_influence
    
    # === Bottleneck proximity: tunable power-law interaction (rank × work)^p — decoupled parameter usage ===
    bottleneck_product = rank * work + eps
    bottleneck_powered = np.power(bottleneck_product, PARAMS["bottleneck_power"])
    bottleneck_score = -PARAMS["bottleneck_power"] * normalize(bottleneck_powered)
    
    # === Duration-uncertainty coupling: bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on raw wait time, explicitly weighted ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -PARAMS["starvation_weight"] * wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2.0, neginf=finfo.min / 2.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
