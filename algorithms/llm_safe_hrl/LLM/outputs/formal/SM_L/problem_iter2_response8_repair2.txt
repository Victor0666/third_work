def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule fusing best-in-class deadline risk containment and energy efficiency.
    
    Key improvements over parents:
      - Replaces tanh-based fairness (Parent 1) and harmonic blend (Parent 2) with signed power-law fairness:
        |duration|^p * sign(duration), enabling tunable convexity/concavity for load balancing.
      - Combines Parent 2's inverse energy and exponential wait decay with Parent 1's robust median-based normalization
        for outlier resilience in heterogeneous workloads.
      - Introduces *slack-margin adaptive gating*: upward_rank activation threshold shifts with median positive slack,
        scaled by criticality_slack_threshold — improves adaptivity to workflow-scale deadlines.
      - Uses clipped uncertainty coupling: avoids unbounded amplification by capping uncert_norm at [0, clip_upper].
      - All components are finite, deterministic, and shape-preserving.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Copy inputs with float64 precision; no in-place mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization using median + MAD (more outlier-resistant than mean/std)
    def robust_normalize(x):
        x = np.asarray(x)
        center = np.median(x)
        scale = np.median(np.abs(x - center)) + eps
        return (x - center) / scale
    
    # Duration: execution + communication
    duration = exec_t + comm_t
    
    # Slack margin for adaptive gating: median of *positive* slack (fallback if none)
    pos_slack = slk[slk > eps]
    slack_margin = np.median(pos_slack) if len(pos_slack) > 0 else 1.0
    slack_margin = max(slack_margin, eps)
    
    # === Slack processing: quadratic penalty for negative, linear gain for small positive ===
    slack_norm = robust_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm) * (np.abs(slack_norm) <= 1.0)
    )
    
    # === Energy efficiency: inverse scaling, robustly normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * robust_normalize(inv_energy)
    
    # === Criticality: gated by slack margin — active only when slack > -slack_margin * threshold ===
    rank_active = np.where(slk > -PARAMS["criticality_slack_threshold"] * slack_margin, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * robust_normalize(rank_active + eps)
    
    # === Duration fairness: signed power-law to avoid bias toward short/long tasks ===
    # Ensures both extremes are penalized symmetrically in magnitude but preserve sign for ordering
    dur_norm = robust_normalize(duration + eps)
    fairness_term = np.sign(dur_norm) * (np.abs(dur_norm) ** PARAMS["fairness_exponent"])
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -robust_normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: only activated under deadline stress ===
    slack_stress = np.where(slk < 0, np.abs(slack_norm), 0.0)
    uncert_norm = robust_normalize(uncert + eps)
    uncert_norm_clipped = np.clip(uncert_norm, 0.0, PARAMS["uncertainty_clip_upper"])
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * uncert_norm_clipped * slack_stress
    
    # === Duration-uncertainty coupling: weighted sum (simpler & more stable than harmonic mean) ===
    dur_uncert_blend = (
        (1.0 - PARAMS["duration_uncertainty_ratio"]) * robust_normalize(duration + eps) +
        PARAMS["duration_uncertainty_ratio"] * uncert_norm_clipped
    )
    dur_score = robust_normalize(dur_uncert_blend)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + fairness_term
        + wait_score
        + unc_slack_interaction
        + dur_score
    )
    
    # Final safeguard: ensure finiteness without breaking ordering
    score = np.nan_to_num(
        score,
        nan=np.median(score) if np.any(np.isfinite(score)) else 0.0,
        posinf=np.finfo(np.float64).max * 0.5,
        neginf=-np.finfo(np.float64).max * 0.5
    )
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values remain after sanitization"
    
    return score
