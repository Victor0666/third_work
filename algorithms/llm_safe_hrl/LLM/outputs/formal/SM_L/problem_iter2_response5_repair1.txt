def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule: DDL-hard constraint first, then energy minimization.
    
    Key structural improvements:
      - Replaces double normalization with single, stable `safe_normalize` using mean-abs + epsilon.
      - Uses bounded slack-criticality gate: criticality suppressed only when slack is severely negative.
      - Energy term uses raw `1/(energy + eps)` — preserves relative discrimination without destructive rescaling.
      - Introduces fairness power-law on duration: `sign(duration_dev) * |duration_dev|^p` for asymmetric bias control.
      - Uncertainty-slack interaction is strictly multiplicative *only* under stress (`slack < 0`).
      - Wait-time uses saturating exponential: smooth, bounded, and avoids runaway priority.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Safe casting; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization: mean-abs scale with epsilon fallback ===
    def safe_normalize(x):
        x = np.asarray(x)
        scale = np.mean(np.abs(x)) + eps
        return x / scale
    
    # === Duration & fairness ===
    duration = exec_t + comm_t
    dur_center = np.mean(duration)
    dur_dev = duration - dur_center
    # Asymmetric fairness: penalize long durations more strongly than short ones
    fairness_term = np.sign(dur_dev) * np.abs(dur_dev) ** PARAMS["duration_fairness_power"]
    # Normalize fairness to avoid dominance
    norm_fairness = safe_normalize(fairness_term + eps)
    
    # === Slack processing: piecewise quadratic/linear with bounded support ===
    # Identify median positive slack for robust gating
    pos_slack = slk[slk > eps]
    slack_margin = np.median(pos_slack) if len(pos_slack) > 0 else PARAMS["slack_margin_fallback"]
    slack_margin = max(slack_margin, eps)
    
    # Quadratic penalty for negative slack, linear de-prioritization for positive slack
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (np.abs(slk) / (slack_margin + eps)) ** 2,
        -PARAMS["slack_urgency_gain"] * (np.abs(slk) / (slack_margin + eps))
    )
    
    # === Criticality gating: only activate when slack allows margin ===
    # Gate: suppress rank when slack < -slack_margin (i.e., severe violation expected)
    gate = np.clip(1.0 - np.maximum(-slk - slack_margin, 0.0) / (slack_margin + eps), 0.0, 1.0)
    gated_rank = rank * gate
    rank_score = -PARAMS["criticality_weight"] * safe_normalize(gated_rank + eps)
    
    # === Energy efficiency: inverse marginal energy, unnormalized → preserves ratio fidelity ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * inv_energy
    
    # === Uncertainty-slack interaction: only activated under deadline stress ===
    # Multiplicative boost: high uncertainty × high negative slack → urgent placement
    unc_slack_interaction = np.where(
        slk < 0,
        PARAMS["uncertainty_slack_interaction"] * uncert * (np.abs(slk) / (slack_margin + eps)),
        0.0
    )
    
    # === Wait-time saturation: exponential anti-starvation with soft cap ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_saturation_rate"] * wait)
    wait_score = -safe_normalize(wait_sat + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # All terms are finite, deterministic, and shape-(N,)
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + norm_fairness
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf using machine-precision bounds
    score = np.nan_to_num(
        score,
        nan=np.finfo(float).tiny,
        posinf=np.finfo(float).max / 1e3,
        neginf=-np.finfo(float).max / 1e3
    )
    
    # Ensure output shape and finiteness
    assert score.shape == (N,), f"Expected (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values remain after sanitization"
    
    return score
