def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with key structural changes:
      - Replaced global energy scoring with conditional host-load-aware suppression: energy priority attenuated
        when |slack| is small (i.e., deadline pressure high), using a smooth piecewise-linear decay.
      - Introduced robust slack-based host load suppression factor: avoids over-optimizing energy under feasibility risk.
      - Simplified wait ramp: direct normalized wait clipped to [0,1], no intermediate normalize() call — improves monotonicity.
      - Tightened bottleneck term: uses sign-aware robust slack normalization (not absolute) to preserve directionality.
      - All interactions now use only safe, epsilon-guarded operations; no unbounded exponentials or divisions.
      - Removed finfo_max_scale (deemed inactive per diagnostics); replaced finite safeguard with np.clip against machine bounds.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating and directional awareness ===
    # Use signed slack for directional urgency (negative = urgent), but clamp magnitude for stability
    slack_norm = normalize(slk)
    slack_abs_norm = np.abs(slack_norm)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * slack_abs_norm,
        -PARAMS["slack_urgency_gain"] * slack_abs_norm
    )
    
    # === Energy efficiency: inverse scaling, now suppressed under tight slack ===
    # Suppression factor: linear decay from 1.0 (large slack) to PARAMS["host_load_suppression_factor"] (near-zero slack)
    suppression_factor = np.clip(
        1.0 - (1.0 - PARAMS["host_load_suppression_factor"]) * np.clip(slack_abs_norm, 0.0, 1.0),
        PARAMS["host_load_suppression_factor"],
        1.0
    )
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * suppression_factor * normalize(inv_energy)
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sign-aware sharpening — uses signed slack to boost priority for negative slack ===
    # Prioritizes tasks blocking large workloads *especially* when slack is negative
    slack_signed_inv = 1.0 / (slk + np.sign(slk) * eps)  # preserves sign, avoids zero-crossing instability
    slack_signed_inv_safe = np.where(np.abs(slk) < eps, np.sign(slk) * (1.0 / eps), slack_signed_inv)
    bottleneck_sharpened = rank * work * np.power(np.abs(slack_signed_inv_safe), PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: direct bounded linear ramp on wait time ===
    # Normalize wait relative to threshold, then clip to [0,1] — monotonic, no quantile bias
    wait_normalized = np.clip(wait / (PARAMS["wait_ramp_threshold"] + eps), 0.0, 1.0)
    wait_score = -wait_normalized  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: clip to safe finite bounds using machine epsilon limits
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1e6
    min_safe = -finfo.max / 1e6
    score = np.clip(score, min_safe, max_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
