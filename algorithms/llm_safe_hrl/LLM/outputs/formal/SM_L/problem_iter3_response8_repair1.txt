def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with strict DDL-first enforcement and robust normalization.
    
    Key self-evolution improvements:
      - Replaces median-based normalization with clipped mean±std using tunable outlier_clip_factor.
      - Introduces hard DDL-protection gate: when slack is critically low (< ddl_protection_gate * ref_scale),
        only slack_penalty and energy_score remain active; all other components zero out → enforces feasibility-first.
      - Uses dynamic reference scale for gate threshold: computed from finite slack stats, avoiding NaN on all-negative slack.
      - Retains inverse energy + linear slack penalty but simplifies fairness term to linear duration-uncertainty blend.
      - Removes wait_decay_rate and uncertainty_slack_interaction per reflection — reduces overfitting and improves stability.
      - All operations are bounded, finite, and shape-preserving; no unbounded branches or hidden state.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Safe casting; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization: use clipped mean/std to avoid NaN/inf on sparse or constant inputs
    def robust_normalize(x):
        x = np.asarray(x)
        valid_mask = np.isfinite(x)
        if not np.any(valid_mask):
            return np.zeros_like(x)
        x_clean = x[valid_mask]
        center = np.mean(x_clean)
        scale = np.std(x_clean, ddof=0) + eps
        # Clip extreme deviations using tunable factor instead of literal 3
        clip_bound = PARAMS["outlier_clip_factor"] * scale
        x_centered = np.clip(x - center, -clip_bound, clip_bound)
        return x_centered / (scale + eps)
    
    # Compute reference scale for DDL protection gate: robust magnitude of slack
    slk_abs_clean = np.abs(slk[np.isfinite(slk)])
    ref_scale = np.mean(slk_abs_clean) if len(slk_abs_clean) > 0 else 1.0
    ref_scale = max(ref_scale, eps)
    
    # DDL-protection activation mask: True only when slack is *not* critically tight
    ddl_safe_mask = (slk >= -PARAMS["ddl_protection_gate"] * ref_scale)
    
    # Slack processing: quadratic penalty for violation, linear gain for margin
    slack_norm = robust_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # Energy efficiency: inverse, normalized
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * robust_normalize(inv_energy)
    
    # Criticality: gated by both absolute slack margin and adaptive threshold
    pos_slack_clean = slk[slk > eps]
    slack_margin = np.mean(pos_slack_clean) if len(pos_slack_clean) > 0 else ref_scale
    slack_margin = max(slack_margin, eps)
    rank_activation_threshold = -PARAMS["slack_margin_sensitivity"] * slack_margin
    rank_active = np.where(slk > rank_activation_threshold, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * robust_normalize(rank_active + eps)
    # Apply DDL gate: zero out rank_score when unsafe
    rank_score = np.where(ddl_safe_mask, rank_score, 0.0)
    
    # Duration-uncertainty blend: linear weighted sum (simpler, more stable than harmonic/tanh)
    duration = exec_t + comm_t
    dur_norm = robust_normalize(duration + eps)
    uncert_norm = robust_normalize(uncert + eps)
    dur_uncert_blend = (
        (1.0 - PARAMS["duration_uncertainty_ratio"]) * dur_norm
        + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    )
    dur_score = robust_normalize(dur_uncert_blend)
    # Apply DDL gate: zero out dur_score when unsafe
    dur_score = np.where(ddl_safe_mask, dur_score, 0.0)
    
    # Final score: smaller = better; all components finite & bounded
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
    )
    
    # Safeguard: replace NaN/inf using machine precision bounds
    score = np.nan_to_num(
        score,
        nan=np.median(score) if np.any(np.isfinite(score)) else 0.0,
        posinf=np.finfo(np.float64).max * 0.5,
        neginf=-np.finfo(np.float64).max * 0.5
    )
    
    # Enforce output contract
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in final priority score"
    
    return score
