def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining strengths of both parents:
      - Max-abs normalization (Parent 2) for stability and interpretability.
      - Piecewise-linear DDL gate (Parent 2) for crisp, bounded deadline protection.
      - Exponential slack-aware energy decay (Parent 2) for physics-aligned suppression.
      - Added fuzzy completion bias: blends slack with fuzzy tail-adjusted deadline margin.
      - Bottleneck term uses robust slack-normalized rank*work (Parent 2) with implicit clipping via max-abs norm.
      - Starvation mitigation retains exponential saturation (Parent 2), with wait clipped using tunable scale.
      - All numeric literals strictly in {-2,-1,0,1,2}; no loops, I/O, or state mutation.
      - Exactly 13 parameters; all used; no unused references.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Max-abs normalization: robust, interpretable, and stable ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Fuzzy completion bias: adjust slack using tail-aware margin (novel) ===
    # slk_biased = slk * (1 - bias) + |slk| * bias * sign(slk < 0) → simplifies to:
    # For slk >= 0: slk * (1 - bias)
    # For slk < 0: slk * (1 - bias) + |slk| * bias = slk * (1 - bias) - slk * bias = slk * (1 - 2*bias)
    sign_neg_slk = np.where(slk < 0, 1.0, 0.0)
    slk_biased = slk * (1.0 - PARAMS["fuzzy_completion_bias"]) + np.abs(slk) * PARAMS["fuzzy_completion_bias"] * (2.0 * sign_neg_slk - 1.0)
    
    # === Piecewise-linear DDL gate: crisp, bounded, monotonic ===
    gate_width = PARAMS["piecewise_linear_gate_width"] + eps
    ddl_gate = np.where(
        slk_biased <= 0,
        1.0,
        np.where(
            slk_biased <= gate_width,
            1.0 - slk_biased / gate_width,
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: linear, signed, monotonic on biased slack ===
    slack_penalty = np.where(
        slk_biased < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk_biased),
        -PARAMS["slack_urgency_gain"] * slk_biased
    )
    
    # === Energy efficiency: inverse marginal energy with dual gating ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Dual gating: linear ddl_gate * exponential slack decay for smoother suppression
    slack_decay = np.exp(-np.abs(slk_biased) * PARAMS["slack_aware_energy_decay"])
    energy_score = energy_score * (1.0 - PARAMS["host_load_sensitivity"] * ddl_gate) * slack_decay
    
    # === Criticality: upward_rank raised to power, normalized ===
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|biased_slack| + eps), normalized ===
    abs_slk_biased = np.abs(slk_biased) + eps
    bottleneck_term = rank * work / abs_slk_biased
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize(bottleneck_term + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation with wait clipped to avoid overflow ===
    finfo = np.finfo(np.float64)
    wait_clipped = np.clip(wait, 0.0, finfo.max / PARAMS["wait_clipping_scale"])
    wait_sat = 1.0 - np.exp(-wait_clipped / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2.0, neginf=finfo.min / 2.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
