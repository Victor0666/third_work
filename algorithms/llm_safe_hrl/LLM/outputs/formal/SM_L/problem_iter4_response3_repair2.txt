def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with:
      - Smooth sigmoid DDL protection gate (replaces hard sign gating) for robust slack transition;
      - Load-aware successor release term: rewards tasks that unlock high-criticality descendants;
      - Host-load-sensitive conditional gating: applies energy/criticality penalties only when system load is high;
      - Median-based normalization (more robust than mean under skew);
      - All components explicitly bounded and finite.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust median-based normalization: handles skew better than mean ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        # Use median absolute deviation scaled by median for stability
        med = np.median(abs_x)
        mad = np.median(np.abs(abs_x - med)) if np.all(np.isfinite(abs_x)) else eps
        scale = med + mad + eps
        return x / scale
    
    # === Smooth sigmoid DDL protection gate around slack = 0 ===
    # Replaces brittle sign-based gating with differentiable transition
    slack_centered = slk / (np.median(np.abs(slk)) + eps)
    ddl_gate = 1.0 / (1.0 + np.exp(-slack_centered / PARAMS["ddl_protection_gate_width"]))
    
    # === Slack-driven components: now smoothly modulated by sigmoid gate ===
    # Negative slack → strong penalty; positive slack → mild de-prioritization
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.maximum(0.0, -slack_centered) 
        - PARAMS["slack_urgency_gain"] * np.maximum(0.0, slack_centered)
    )
    
    # === Energy efficiency: inverse scaling, gated by DDL safety ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_gate
    
    # === Criticality: upward_rank weighted only where DDL safety margin exists (smoothly) ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps) * ddl_gate
    
    # === Duration-uncertainty coupling: robust linear blend with median scaling ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: saturating exponential, gated by DDL urgency ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps) * (1.0 - ddl_gate)  # Only active when DDL is tight
    
    # === Uncertainty-slack interaction: amplified under DDL stress, smoothed ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * np.maximum(0.0, -slack_centered) 
        * (1.0 - ddl_gate)  # Reinforce under high risk
    )
    
    # === Load-aware successor release: reward tasks whose descendants have high upward_rank ===
    # Uses work as proxy for descendant count & criticality density
    successor_release_score = -PARAMS["load_successor_release_weight"] * normalize(work + eps) * rank * ddl_gate
    
    # === Host-load-sensitive conditional gating: apply extra penalty only under congestion ===
    # Simulate load via normalized uncertainty + duration; activate penalty above threshold
    system_load_proxy = normalize(duration + eps) + normalize(uncert + eps)
    load_gate = np.clip(system_load_proxy - PARAMS["host_load_sensitivity"], 0.0, 1.0)  # Threshold now parameterized
    host_load_penalty = PARAMS["host_load_sensitivity"] * load_gate * energy_score
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + successor_release_score
        + host_load_penalty
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
