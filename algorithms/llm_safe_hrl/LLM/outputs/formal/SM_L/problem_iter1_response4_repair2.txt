def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """
    Novel priority rule emphasizing:
      - Strong nonlinear risk penalty for negative slack (DDL violation)
      - Critical-path boosting only for top-quartile upward_rank (avoids over-prioritizing medium ranks)
      - Energy efficiency ratio (marginal J / remaining MI) as a primary signal
      - Robust slack normalization blending mean-abs and IQR for stability under outliers
      - Saturating anti-starvation term with linear ramp beyond threshold
      - Uncertainty-duration coupling to favor predictable short tasks under high risk
    All operations are finite, deterministic, epsilon-guarded, and preserve shape (N,).
    """
    eps = PARAMS["epsilon"]
    N = len(slack)

    # Safeguard inputs: replace NaN/inf with finite bounds
    def safe_clip(x):
        x = np.asarray(x, dtype=float)
        finfo = np.finfo(float)
        return np.nan_to_num(x, nan=0.0, posinf=finfo.max, neginf=finfo.min)

    min_exec_time = safe_clip(min_exec_time)
    min_comm_time = safe_clip(min_comm_time)
    min_incremental_energy = safe_clip(min_incremental_energy)
    slack = safe_clip(slack)
    upward_rank = safe_clip(upward_rank)
    remaining_work = safe_clip(remaining_work)
    ready_wait_time = safe_clip(ready_wait_time)
    uncertainty = safe_clip(uncertainty)

    # --- Robust normalization: blend mean-abs and IQR ---
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        # Mean absolute baseline
        mean_abs = np.mean(np.abs(x)) + eps
        # IQR baseline (Q3-Q1), more outlier-resistant
        q1 = np.percentile(x, 25)
        q3 = np.percentile(x, 75)
        iqr = q3 - q1 + eps
        # Blend using learned mode parameter
        alpha = np.clip(PARAMS["slack_normalization_mode"], 0.0, 1.0)
        denom = alpha * iqr + (1.0 - alpha) * mean_abs
        return x / denom

    # --- Slack risk: strong nonlinearity for negative values only ---
    # Use clipped power: sign-preserving but penalize negative slack quadratically
    slack_norm = robust_normalize(slack)
    slack_risk = np.where(
        slack_norm < 0,
        -np.power(-slack_norm, PARAMS["slack_risk_exponent"]),
        0.0
    )

    # --- Criticality boost: only activate for top quartile upward_rank ---
    # Use configurable percentile (e.g., 75.0 → Q3)
    urank_sorted = np.sort(upward_rank)
    if N == 0:
        urank_q = 0.0
    else:
        # Compute percentile index: floor((p/100)*(N-1)) for 0-indexed array
        p = PARAMS["quartile_percentile"]
        idx = int(np.floor((p / 100.0) * (N - 1)))
        idx = np.clip(idx, 0, N - 1)
        urank_q = urank_sorted[idx]
    critical_mask = (upward_rank >= urank_q - eps).astype(float)
    critical_boost = critical_mask * PARAMS["criticality_boost"] * robust_normalize(upward_rank)

    # --- Energy efficiency ratio: J / MI — lower is better, so invert sign for priority
    # Avoid div-by-zero: add eps to denominator, clip numerator
    eff_ratio = np.divide(
        min_incremental_energy,
        np.maximum(remaining_work, eps),
        out=np.full_like(remaining_work, np.inf, dtype=float),
        where=(remaining_work > 0)
    )
    # Clip extreme ratios to avoid dominance
    eff_ratio = np.clip(eff_ratio, 0.0, PARAMS["eff_ratio_clip_upper"])
    eff_score = -robust_normalize(eff_ratio)  # negative → smaller score = higher priority

    # --- Duration-uncertainty coupling: prefer low-uncertainty short tasks
    duration = min_exec_time + min_comm_time
    duration_norm = robust_normalize(duration)
    unc_norm = robust_normalize(uncertainty)
    # Coupling term: penalize high duration * high uncertainty
    coupling_penalty = PARAMS["duration_uncertainty_coupling"] * duration_norm * unc_norm

    # --- Anti-starvation: saturating linear ramp beyond threshold
    wait_norm = robust_normalize(ready_wait_time)
    # Threshold in normalized units
    sat_thresh = PARAMS["wait_saturation_threshold"]
    wait_relief = np.where(
        wait_norm > sat_thresh,
        PARAMS["anti_starvation_slope"] * (wait_norm - sat_thresh),
        0.0
    )

    # --- Assemble final score: smaller = higher priority
    # Prioritize: 1) DDL risk (penalty → negative), 2) critical path, 3) energy efficiency,
    #             4) anti-starvation, then penalize: 5) duration-uncertainty coupling
    score = (
        slack_risk                  # Strong penalty for missed deadlines (negative = good)
        + critical_boost           # Boost top critical tasks
        + PARAMS["energy_efficiency_ratio_weight"] * eff_score  # Prefer energy-efficient per work
        + wait_relief              # Anti-starvation relief
        + coupling_penalty         # Penalize risky long tasks
    )

    # Final safeguard: ensure finite output
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    return score
