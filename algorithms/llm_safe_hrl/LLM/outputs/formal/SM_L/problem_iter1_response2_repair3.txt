def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: combines deadline-driven urgency gating, criticality-boosted ranking,
    risk-coupled duration-energy tradeoff, and starvation-avoiding wait decay.
    
    Key innovations:
    - Uses piecewise slack gating: strong linear penalty for negative slack, soft relaxation for positive slack.
    - Upward rank is amplified *only* when slack < 0 or uncertainty > median — avoids over-prioritizing
      critical paths during safe periods.
    - Energy term is downweighted when slack is positive, reflecting relaxed optimization priority.
    - Duration and uncertainty interact multiplicatively to penalize high-risk, long-duration tasks.
    - Ready wait time decays exponentially with slack pressure, preventing starvation without overwhelming DDL.
    - Work density (remaining_work / (exec + comm + epsilon)) prioritizes throughput-critical tasks.
    - All normalizations use robust mean-abs scaling with epsilon guard; no divisions by zero or inf.
    """
    eps = PARAMS["epsilon"]
    finfo = np.finfo(float)
    eps_safe = max(eps, finfo.tiny)

    # Ensure finite, non-NaN inputs — clip extreme outliers to avoid blowup
    def safe_clip(x):
        return np.clip(x, finfo.min + eps_safe, finfo.max - eps_safe)

    min_exec_time = safe_clip(min_exec_time)
    min_comm_time = safe_clip(min_comm_time)
    min_incremental_energy = safe_clip(min_incremental_energy)
    slack = safe_clip(slack)
    upward_rank = safe_clip(upward_rank)
    remaining_work = safe_clip(remaining_work)
    ready_wait_time = safe_clip(ready_wait_time)
    uncertainty = safe_clip(uncertainty)

    # Robust normalization: scale by mean absolute value + epsilon
    def robust_norm(x):
        x_abs = np.abs(x)
        norm_denom = np.mean(x_abs) + eps_safe
        return x / norm_denom

    # 1. Slack: piecewise-linear urgency signal — strong penalty for negative slack, mild benefit for large slack
    slack_norm = robust_norm(slack)
    slack_penalty = np.where(slack < 0, 
                            PARAMS["slack_risk_penalty"] * (-slack_norm), 
                            PARAMS["slack_relaxation_weight"] * slack_norm)

    # 2. Criticality boost: apply nonlinear gain to upward_rank only under risk (tight slack or high uncertainty)
    slack_tight = (slack < 0)
    uncertainty_high = (uncertainty > np.median(uncertainty) + eps_safe)
    risk_condition = np.logical_or(slack_tight, uncertainty_high)
    rank_boosted = upward_rank * (1.0 + PARAMS["criticality_boost"] * risk_condition.astype(float))
    rank_norm = robust_norm(rank_boosted)
    rank_score = -rank_norm  # higher rank → higher priority → lower score

    # 3. Energy: downweight when slack > 0 (relaxed regime), full weight when slack <= 0 (urgent regime)
    energy_weight_adj = np.where(slack > 0, 
                                PARAMS["energy_efficiency_scale"] * 0.5, 
                                PARAMS["energy_efficiency_scale"])
    energy_norm = robust_norm(min_incremental_energy)
    energy_score = energy_weight_adj * energy_norm

    # 4. Duration-uncertainty coupling: penalize long + uncertain tasks more than sum of parts
    duration = min_exec_time + min_comm_time + eps_safe
    dur_uncert_coupling = duration * uncertainty * PARAMS["duration_uncertainty_coupling"]
    dur_uncert_norm = robust_norm(dur_uncert_coupling)
    dur_score = dur_uncert_norm

    # 5. Anti-starvation: ready_wait_time decays under urgency (i.e., when slack is negative)
    # so long waits matter most when deadlines are tight
    wait_decay = np.exp(-PARAMS["wait_decay_rate"] * np.maximum(-slack, 0))
    wait_norm = robust_norm(ready_wait_time)
    wait_score = -wait_norm * wait_decay  # negative → higher priority

    # 6. Work density: remaining_work per unit duration — favors high-throughput tasks
    work_density = remaining_work / (duration + eps_safe)
    work_density_norm = robust_norm(work_density)
    work_score = -work_density_norm * PARAMS["work_density_ratio"]

    # 7. Uncertainty-slack gate: activate extra penalty only when both slack is low AND uncertainty is high
    # uses percentile-based threshold (tunable, not hardcoded)
    slack_threshold = np.percentile(slack, PARAMS["slack_percentile_threshold"])
    gate_input = PARAMS["uncertainty_slack_gate"] * (slack_threshold - slack)
    # bounded sigmoid approximation: clip input to avoid overflow, use tanh for smoothness
    gate_input_clipped = np.clip(gate_input, -PARAMS["tanh_clip_bound"], PARAMS["tanh_clip_bound"])
    uncertainty_gate = np.tanh(gate_input_clipped) * uncertainty
    uncertainty_gate_norm = robust_norm(uncertainty_gate)
    gate_score = uncertainty_gate_norm

    # Final score: sum all components — smaller = better
    score = (
        slack_penalty +
        rank_score +
        energy_score +
        dur_score +
        wait_score +
        work_score +
        gate_score
    )

    # Final safeguard: replace NaN/inf with finite values, preserve shape
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 1e3, neginf=finfo.min / 1e3)

    return score
