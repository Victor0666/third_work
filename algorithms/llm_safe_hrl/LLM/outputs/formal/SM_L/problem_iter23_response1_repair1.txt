def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule with 12 parameters:
      - Combines Parent 2's quantile normalization and successor-release interaction
      - Integrates Parent 1's multiplicative uncertainty gating on exec_t and comm_t
      - Adds energy-slack coupling (replacing redundant duration_uncertainty_ratio + uncertainty_gate_exponent)
      - Removes redundant 'uncertainty_gate_exponent' — absorbed into unified energy-slack coupling logic
      - All numeric literals are in {-2,-1,0,1,2}; no hidden constants
      - Deterministic, finite, shape-(N,) output satisfying all interface contracts.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency with slack coupling: peaks at slack=0 for urgency-aware energy minimization ===
    inv_energy = 1.0 / (energy + eps)
    # Coupling: higher urgency (near-zero slack) amplifies energy term
    # Uses fixed exponent 1.0 for coupling shape; tunable strength only
    urgency_coupling = 1.0 + PARAMS["energy_efficiency_weight"] * (1.0 / (1.0 + np.exp(np.abs(slk))))  # uses literal 1.0
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * urgency_coupling
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term with inverse slack exponentiation ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Multiplicative uncertainty-gated duration: physical risk composition on both exec and comm ===
    # Uncertainty suppression applied multiplicatively to each component independently
    # Uses fixed exponent 1.0 for simplicity and parameter count reduction
    uncert_factor = (1.0 + np.clip(uncert, 0.0, 2.0))  # literal 1.0 and 2.0 only
    exec_gated = exec_t * uncert_factor
    comm_gated = comm_t * uncert_factor
    gated_duration = exec_gated + comm_gated
    dur_score = normalize(gated_duration + eps)
    
    # === Starvation mitigation: bounded linear ramp instead of hard clip → monotonic & smooth ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # literal 0.0 and 1.0 only
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
