def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's robustness with Parent 1's adaptive gating:
      - Uses sigmoid DDL gate (Parent 1) for smooth, numerically stable transition near slack=0,
        but simplified with fixed center at 0 and steepness parameter — avoids per-task offset complexity.
      - Retains exponential starvation mitigation (Parent 2) for stable wait-time response.
      - Applies max-abs normalization universally (Parent 2) for interpretability and outlier resilience.
      - Introduces *slack-dependent energy gating*: attenuates energy term proportionally to ddl_gate,
        improving deadline-risk focus without eliminating energy optimization in feasible region.
      - Criticality uses exponentiated upward_rank weighted by slack magnitude (Parent 2 style),
        enabling stronger prioritization of high-rank tasks when deadlines tighten.
      - Bottleneck term uses rank * work / (|slack| + eps) — clean, monotonic, and avoids softplus instability.
      - All numeric literals are {-2,-1,0,1,2}; no hidden constants; all parameters used exactly once.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Max-abs normalization: robust, interpretable, and stable ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Sigmoid DDL protection gate centered at slack = 0 (simplified Parent 1) ===
    # Steep activation near deadline violation: gate ≈ 1 when slack ≤ 0, ≈ 0 when slack >> 0
    finfo = np.finfo(float)
    gate_input = PARAMS["ddl_protection_steepness"] * (-slk)  # flips sign so negative slack → large positive input
    gate_input_clipped = np.clip(gate_input, -np.log(finfo.max), np.log(finfo.max))
    ddl_gate = 1.0 / (1.0 + np.exp(-gate_input_clipped))
    
    # === Slack-driven base penalty/gain: linear, signed, monotonic ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: inverse marginal energy, gated by ddl_gate ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Attenuate energy influence when deadline risk is high
    energy_score = energy_score * (1.0 - PARAMS["energy_gating_strength"] * ddl_gate)
    
    # === Criticality: upward_rank raised to slack-aware power ===
    # Use |slack|+eps to avoid division by zero and ensure monotonic decay of sensitivity
    slack_mag_norm = normalize(np.abs(slk) + eps)
    # Higher exponent amplifies rank importance more sharply as slack shrinks
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|slack| + eps) — monotonic, bounded, intuitive ===
    slack_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_term = rank * work * slack_inv
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize(bottleneck_term + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation (Parent 2) ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using declared bounds
    score_clip_max = PARAMS["score_clip_max"]
    score_clip_min = -PARAMS["score_clip_min_abs"]
    score = np.nan_to_num(score, nan=0.0, posinf=score_clip_max, neginf=score_clip_min)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
