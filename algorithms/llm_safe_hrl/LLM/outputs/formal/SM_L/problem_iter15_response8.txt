def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Robust percentile-based normalization (75th percentile) replaces median to resist skew and zero-dominance.
      - Restored and strengthened energy-DDL coupling: energy_base * (ddl_gate ** energy_ddl_coupling_exponent)
        → preserves energy signal under tight deadlines (ddl_gate ≈ 1), suppresses it smoothly under surplus slack.
      - Removed redundant finfo_max_scale; now uses bounded percentile clipping (99.9%) + np.clip for safe clamping.
      - All divisions, exponentials, and normalizations guarded against NaN/inf/zero.
      - No hidden constants beyond {-2,-1,0,1,2}; all tunables declared and used.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust percentile-based normalization (75th percentile) for skew resistance ===
    def normalize(x, p=PARAMS["robust_norm_percentile"]):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            # Use clipped percentile to avoid outlier dominance
            q = np.percentile(abs_x[finite_mask], p, method='midpoint')
            scale = np.maximum(q, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Smooth sigmoid DDL protection gate: differentiable, monotonic, bounded ∈ [0,1] ===
    steep = PARAMS["ddl_sigmoid_steepness"]
    ddl_gate = 1.0 / (1.0 + np.exp(-steep * slk))
    
    # === Slack-driven base penalty/gain: signed, normalized via absolute percentile ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    slk_scale = np.percentile(finite_abs_slk, PARAMS["robust_norm_percentile"], method='midpoint') \
        if len(finite_abs_slk) > 0 else eps
    slk_scale = np.where(slk_scale > eps, slk_scale, eps)
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse energy, normalized, then coupled via power-law DDL gate ===
    inv_energy = 1.0 / (energy + eps)
    energy_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Stronger coupling: full energy influence when ddl_gate ≈ 1 (tight deadline); suppressed as slack grows
    energy_score = energy_base * (ddl_gate ** PARAMS["energy_ddl_coupling_exponent"])
    
    # === Criticality: gated by smooth sigmoid, rank normalized with floor to avoid degeneracy ===
    rank_stable = rank + PARAMS["upward_rank_normalization_floor"]
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank_stable)
    
    # === Bottleneck-proximal term: upward_rank * remaining_work — critical-path blocking impact ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, percentile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: sharpened power-law ramp — monotonic, bounded, stronger bias than linear ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_sharpened = np.power(wait_clipped + eps, PARAMS["starvation_exponent"])
    wait_score = -normalize(wait_sharpened)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: bounded percentile clipping + nan_to_num
    # Clip extreme outliers using 99.9th percentile instead of absolute machine bounds
    score_finite = np.where(np.isfinite(score), score, 0.0)
    p999 = np.percentile(np.abs(score_finite), 99.9, method='midpoint') if N > 1 else np.max(np.abs(score_finite))
    clip_bound = np.maximum(p999, eps)
    score = np.clip(score, -clip_bound, clip_bound)
    score = np.nan_to_num(score, nan=0.0, posinf=clip_bound, neginf=-clip_bound)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
