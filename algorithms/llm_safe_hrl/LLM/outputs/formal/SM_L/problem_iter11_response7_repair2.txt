def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with 13 parameters:
      - Added 'robust_normalization_quantile' to replace hidden literal 0.75.
      - All numeric literals in function body are now strictly in {-2,-1,0,1,2}.
      - Uses quantile-based normalization consistently across all features.
      - Bottleneck term uses softplus sharpening (log1p(exp(x))) — no new parameter, uses only built-in functions and allowed literals.
      - DDL protection applies amplification to energy, criticality, and bottleneck terms uniformly.
      - Final score is finite, shape-(N), deterministic, and satisfies all interface contracts.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: unified scale using tunable quantile ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        normed = x / (scale + eps)
        # Clip to [-2, 2] to suppress outlier-induced rank flips
        return np.clip(normed, -2.0, 2.0)
    
    # === Normalized slack for gating and comparisons ===
    slack_norm = normalize(slk)
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling → sharper preference for low marginal energy ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: only active when slack is non-negative AND above protection threshold ===
    rank_active = np.where(
        (slk >= 0) & (slack_norm >= PARAMS["ddl_protection_gate"]), 
        rank, 
        0.0
    )
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear clip ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_normalized = wait_clipped / (PARAMS["wait_clip_threshold"] + eps)
    wait_score = -normalize(wait_normalized + eps)
    
    # === Bounded piecewise load-pressure: (work * uncertainty), saturated above learned threshold ===
    load_pressure_raw = work * (uncert + eps)
    load_pressure_norm = normalize(load_pressure_raw + eps)
    saturation_bound = PARAMS["load_saturation_threshold"] * np.median(np.abs(load_pressure_norm) + eps)
    load_pressure_saturated = np.clip(load_pressure_norm, -saturation_bound, saturation_bound)
    # Gate: apply only under DDL protection (critically tight slack)
    load_score = np.where(
        slack_norm < PARAMS["ddl_protection_gate"],
        load_pressure_saturated,
        0.0
    )
    
    # === Bottleneck proximity: sharpened interaction (rank * work) / (|slack|+eps), then softplus ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_base = rank * work * slack_magnitude_inv
    bottleneck_sharpened = np.log1p(np.exp(bottleneck_base))  # softplus: uses only 0,1 and built-ins
    bottleneck_score = -PARAMS["criticality_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === DDL protection gate: overrides duration dominance when slack is critically low ===
    protection_mask = (slack_norm < PARAMS["ddl_protection_gate"])
    # Apply soft suppression: reduce duration impact, amplify energy, criticality, and bottleneck
    dur_score = np.where(protection_mask, dur_score * PARAMS["duration_suppression_factor"], dur_score)
    energy_score = np.where(protection_mask, energy_score * PARAMS["energy_amplification_factor"], energy_score)
    rank_score = np.where(protection_mask, rank_score * PARAMS["criticality_amplification_factor"], rank_score)
    bottleneck_score = np.where(protection_mask, bottleneck_score * PARAMS["criticality_amplification_factor"], bottleneck_score)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + load_score
    )
    
    # Final safeguard: replace NaN/inf with 0.0 using explicit masking
    score = np.where(np.isfinite(score), score, 0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
