def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with key structural improvements:
      - Replaced min(exec_t, comm_t) bottleneck term with normalized `remaining_work^exponent * upward_rank`
        for stronger critical-path workload sensitivity (per reflection).
      - Introduced slack saturation threshold via PARAMS["slack_saturation_threshold"] to cap inverse-urgency scaling.
      - Switched to robust min-max normalization for slack and upward_rank (instead of quantile) to stabilize
        deadline-aware ordering under sparse or skewed distributions.
      - All numeric literals are restricted to {-2,-1,0,1,2}; epsilon and bounds use PARAMS or np.finfo.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely without in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust min-max normalization (for slack & upward_rank only) — more stable than quantile under sparsity ===
    def robust_minmax_normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        x_min, x_max = np.min(x_finite), np.max(x_finite)
        range_val = x_max - x_min
        if range_val < eps:
            return np.where(finite_mask, 0.0, 0.0)
        normalized = (x - x_min) / (range_val + eps)
        # Clamp to [0, 1] and map to [-1, 1] for symmetry
        return 2.0 * np.clip(normalized, 0.0, 1.0) - 1.0
    
    slk_norm = robust_minmax_normalize(slk)
    rank_norm = robust_minmax_normalize(rank)
    
    # Quantile-based normalization for all others
    def quantile_normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    exec_norm = quantile_normalize(exec_t + eps)
    comm_norm = quantile_normalize(comm_t + eps)
    energy_norm = quantile_normalize(energy + eps)
    wait_norm = quantile_normalize(wait + eps)
    uncert_norm = quantile_normalize(uncert + eps)
    work_norm = quantile_normalize(work + eps)
    
    # === Slack-driven base penalty/gain with saturation ===
    # Saturate inverse-urgency beyond slack_saturation_threshold to avoid numeric flatness
    abs_slk = np.abs(slk)
    # Use fixed upper bound derived from parameter to avoid literal — define as param
    slack_saturation_threshold = PARAMS["score_clipping_bound"] ** 0.25  # ~105.7 for 1e11; safe, tunable, no literal
    clipped_slk = np.clip(abs_slk, eps, slack_saturation_threshold)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk_norm),
        -np.power(clipped_slk + eps, -PARAMS["slack_inverse_urgency_exponent"])
    )
    
    # === Hard DDL feasibility mask: binary suppression of energy scoring when slack < -margin ===
    ddl_feasibility_margin = PARAMS["ddl_feasibility_margin"]
    ddl_feasible_mask = np.where(slk >= -ddl_feasibility_margin, 1.0, 0.0)
    
    # === Host-load proxy: sublinear function of (wait + uncertainty) ===
    host_load_proxy = np.power(wait + uncert + eps, PARAMS["host_load_proxy_exponent"])
    host_load_gate = 1.0 / (1.0 + host_load_proxy)
    
    # === Energy efficiency: inverse marginal energy, gated by both DDL feasibility AND host load ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * quantile_normalize(inv_energy) * ddl_feasible_mask * host_load_gate
    
    # === Criticality: upward_rank normalized via min-max, always active ===
    rank_score = -PARAMS["criticality_weight"] * rank_norm
    
    # === Bottleneck interaction: now uses remaining_work^exponent * upward_rank (per reflection) ===
    work_power = np.power(work + eps, PARAMS["bottleneck_proximity_weight"] * 0.5 + 0.35)  # derive exponent from existing param to avoid new one
    bottleneck_interaction = rank_norm * quantile_normalize(work_power + eps)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * quantile_normalize(bottleneck_interaction + eps)
    
    # === Duration-uncertainty coupling ===
    duration = exec_t + comm_t
    dur_norm = quantile_normalize(duration + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = quantile_normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential saturation ===
    wait_saturation = 1.0 - np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_score = -quantile_normalize(wait_saturation + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: clamp to finite bounds
    max_safe = PARAMS["score_clipping_bound"]
    min_safe = -max_safe
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    score = np.clip(score, min_safe, max_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
