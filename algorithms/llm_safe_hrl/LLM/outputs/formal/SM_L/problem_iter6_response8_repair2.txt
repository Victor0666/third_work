def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with quantile-IQR normalization, smooth sigmoid bottleneck coupling,
    and raw-slack–preserving deadline urgency encoding.
    
    Key evolutions:
      - Replaced robust MAD normalization with quantile-based IQR scaling using tunable q25/q75.
      - Replaced binary bottleneck boost with smooth sigmoidal coupling using PARAMS["bottleneck_sigmoid_steepness"].
      - Slack penalty uses *raw slack* directly (preserves absolute urgency) inside gated quadratic/linear form.
      - All numeric literals restricted to {-2,-1,0,1,2}; epsilon & parameters strictly used; no hardcoded thresholds.
      - Quantile normalizations use np.quantile with explicit method='linear' for determinism.
      - Bottleneck ratio threshold now tunable via PARAMS["bottleneck_ratio_quantile"].
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs to float64; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Quantile-based normalization using tunable IQR bounds
    def quantile_normalize(x):
        x = np.asarray(x)
        q25 = np.quantile(x, PARAMS["q25_quantile"], method='linear')
        q75 = np.quantile(x, PARAMS["q75_quantile"], method='linear')
        iqr = q75 - q25 + eps
        center = np.quantile(x, 0.5, method='linear')
        scale = PARAMS["iqr_scale_factor"] * iqr + eps
        return (x - center) / scale
    
    # Adaptive slack margin: configurable quantile of positive slack
    pos_slack = slk[slk > eps]
    if len(pos_slack) > 0:
        slack_margin = np.quantile(pos_slack, PARAMS["successor_release_tightness_quantile"], method='linear')
        slack_margin = max(slack_margin, eps)
    else:
        slack_margin = 1.0
    
    # Slack penalty: uses *raw slack* directly (preserves absolute urgency), gated by sign
    slack_penalty = np.zeros_like(slk)
    # Negative slack → quadratic penalty
    neg_mask = slk < 0
    slack_penalty[neg_mask] = PARAMS["slack_risk_penalty"] * ((-slk[neg_mask]) / (slack_margin + eps)) ** 2
    # Positive slack → linear urgency gain (deprioritizes idle time)
    pos_mask = slk >= 0
    slack_penalty[pos_mask] = -PARAMS["slack_urgency_gain"] * (slk[pos_mask] / (slack_margin + eps))
    
    # Inverse energy efficiency — DDL-feasibility gated
    inv_energy = 1.0 / (energy + eps)
    energy_active = np.where(slk > -eps, inv_energy, 0.0)
    norm_energy = quantile_normalize(energy_active + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * norm_energy
    
    # Criticality: strictly gated by slack > -epsilon
    rank_active = np.where(slk > -eps, rank, 0.0)
    norm_rank = quantile_normalize(rank_active + eps)
    rank_score = -PARAMS["criticality_weight"] * norm_rank
    
    # Duration-uncertainty coupling: harmonic blend on quantile-normalized signals
    duration = exec_t + comm_t
    norm_duration = quantile_normalize(duration + eps)
    norm_uncert = quantile_normalize(uncert + eps)
    dur_uncert_signal = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * norm_uncert) 
        / (norm_duration + eps + PARAMS["duration_uncertainty_ratio"] * norm_uncert + eps)
    )
    dur_score = quantile_normalize(dur_uncert_signal)
    
    # Starvation mitigation with saturation
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    norm_wait_sat = quantile_normalize(wait_sat + eps)
    wait_score = -norm_wait_sat
    
    # Uncertainty-slack interaction under deadline stress
    slack_stress = np.where(slk < 0, np.abs(slk) / (slack_margin + eps), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * norm_uncert * slack_stress
    
    # Smooth sigmoid bottleneck coupling (replaces brittle binary gate)
    rank_to_work_ratio = rank / (work + eps)
    # Compute quantile threshold on valid ratios
    valid_ratios = rank_to_work_ratio[rank_to_work_ratio > eps]
    if len(valid_ratios) > 0:
        ratio_threshold = np.quantile(valid_ratios, PARAMS["bottleneck_ratio_quantile"], method='linear')
    else:
        ratio_threshold = 0.0
    # Sigmoid: soft transition around threshold, scaled by slack tightness
    ratio_deviation = (rank_to_work_ratio - ratio_threshold) / (slack_margin + eps)
    sigmoid_coupling = 1.0 / (1.0 + np.exp(-PARAMS["bottleneck_sigmoid_steepness"] * ratio_deviation))
    # Only activate coupling when slack <= slack_margin (tight-deadline context)
    bottleneck_boost = np.where(slk <= slack_margin, sigmoid_coupling, 0.0)
    successor_coupling = PARAMS["successor_coupling_strength"] * bottleneck_boost
    
    # Final score: smaller = higher priority
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + successor_coupling
    )
    
    # Final safeguard: replace NaN/inf with finite bounds
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
