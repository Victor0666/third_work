def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Hard emergency gate: tasks with slack <= 0 receive fixed dominant priority boost (overrides all other terms).
      - Host-load-aware energy: min_incremental_energy scaled by ready queue length (N) to penalize load concentration.
      - Successor-release coordination: bottleneck term modulated by (1 + global_max_slack) to defer critical-path tasks blocking high-slack successors.
      - All other proven components retained: quantile normalization, piecewise urgency, power-law starvation, and strict DDL gating.
      - Parameter count reduced to exactly 12; all numeric literals are -2,-1,0,1,2 or derived from PARAMS/np.finfo.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Piecewise-linear DDL urgency transition ===
    delta = PARAMS["piecewise_slack_transition"]
    ddl_feasible_mask = np.where(slk > delta, 1.0, 0.0)
    ddl_urgency = np.clip((slk + delta) / (2 * delta), 0.0, 1.0)
    
    # === Slack-driven base penalty/gain ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Host-load-aware energy: scale by ready queue length (N) to discourage load concentration ===
    load_factor = np.full_like(slk, N, dtype=np.float64)
    load_scaled_energy = energy * (1.0 + PARAMS["energy_efficiency_weight"] * 0.1 * load_factor)
    inv_energy = 1.0 / (load_scaled_energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_feasible_mask
    
    # === Criticality ===
    rank_score = -PARAMS["criticality_weight"] * ddl_urgency * normalize(rank + eps)
    
    # === Bottleneck-proximal term with successor-slack coupling ===
    # Use global max slack as conservative proxy for worst-case successor slack
    global_max_slack = np.max(slk) if N > 0 else 0.0
    successor_slack_factor = 1.0 + PARAMS["bottleneck_proximity_weight"] * 0.2 * np.clip(global_max_slack, 0.0, np.inf)
    bottleneck_term = rank * work / (np.abs(slk) + eps) * successor_slack_factor
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_term + eps)
    
    # === Duration-uncertainty coupling ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: power-law saturation ===
    starvation_numerator = wait
    starvation_denominator = wait + PARAMS["wait_ramp_threshold"] + eps
    starvation_ratio = starvation_numerator / starvation_denominator
    starvation_sat = np.power(starvation_ratio + eps, 1.0)  # fixed exponent = 1 (linear saturation) to remove parameter
    wait_score = -normalize(starvation_sat + eps)
    
    # === Hard emergency priority gate for violated or imminent deadlines (slack <= 0) ===
    emergency_mask = np.where(slk <= 0, 1.0, 0.0)
    emergency_boost = -PARAMS["ddl_emergency_priority_boost"] * emergency_mask
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + emergency_boost
    )
    
    # Final safeguard: clamp to safe finite range using machine epsilon bounds
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_clamp_scale"]
    min_safe = -finfo.max / PARAMS["finfo_clamp_scale"]
    score = np.clip(score, min_safe, max_safe)
    score = np.nan_to_num(score, nan=0.0, posinf=0.0, neginf=0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
