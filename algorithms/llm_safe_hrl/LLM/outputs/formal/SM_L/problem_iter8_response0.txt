def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robustness from Parent 1 and expressivity from Parent 2:
      - Hybrid bottleneck-criticality: uses both upward_rank (for path length) AND upward_rank * remaining_work (for workload pressure)
      - Dual-gated structure: hard feasibility mask for energy/criticality AND soft sigmoid gate for starvation/interaction terms
      - Enhanced normalization: median/MAD-based for all features, plus explicit clipping for wait time before saturation
      - Starvation mitigation uses clipped wait time + exponential saturation (more stable than pure exp on raw wait)
      - All deadline-sensitive terms (energy, criticality, release, load) share unified ddl_gate; bottleneck term uses hard slack mask for strict DDL safety
      - Uncertainty-slack interaction applies only under deadline stress (1-ddl_gate), enhancing robustness to fuzzy lateness
      - No numeric literals beyond {-2,-1,0,1,2}; all tunables declared in PARAMETER_SCHEMA.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust median/MAD normalization: resistant to outliers, preserves ordering ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        med = np.median(abs_x)
        mad = np.median(np.abs(abs_x - med)) if np.all(np.isfinite(abs_x)) else eps
        scale = max(med, 1.0 * mad, eps)  # literal 1.0 per constraint
        return x / (scale + eps)
    
    # === Unified smooth DDL protection gate: centered at 0, width-tunable ===
    gate_width = PARAMS["ddl_protection_gate_width"]
    ddl_gate = 1.0 / (1.0 + np.exp(-slk / (gate_width + eps)))
    
    # === Hard slack feasibility mask: binary gating for strict DDL-first enforcement on bottleneck & energy ===
    slack_feasible_mask = (slk >= 0).astype(np.float64)
    
    # === Slack-driven priority: linear penalty for negative slack, mild gain for large positive slack ===
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.maximum(-slk, 0.0) +
        PARAMS["slack_urgency_gain"] * np.minimum(slk, 0.0)
    )
    slack_score = normalize(slack_penalty + eps)
    
    # === Energy efficiency: gated by ddl_gate to avoid over-prioritizing low-energy tasks under deadline stress ===
    inv_energy = 1.0 / (energy + eps)
    energy_active = inv_energy * ddl_gate
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(energy_active + eps)
    
    # === Pure criticality (upward_rank): fully soft-gated by ddl_gate ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Bottleneck-proximal criticality (upward_rank * remaining_work): hard-gated by slack feasibility for strict DDL safety ===
    bottleneck_score = rank * work
    bottleneck_score_gated = bottleneck_score * slack_feasible_mask
    bottleneck_norm = normalize(bottleneck_score_gated + eps)
    bottleneck_score_final = -PARAMS["bottleneck_proximity_weight"] * bottleneck_norm
    
    # === Duration-uncertainty coupling: robustly normalized blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: clipped wait time + exponential saturation, then soft-gated ===
    wait_clipped = np.clip(wait, 0.0, 2.0 * PARAMS["wait_decay_rate"]**(-1.0))  # bounded clip using literal 2.0
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait_clipped)
    wait_score = -normalize(wait_sat + eps) * ddl_gate
    
    # === Uncertainty-slack interaction: activated only when ddl_gate < 0.5 (i.e., slack negative or near-zero) ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * (1.0 - ddl_gate)
    )
    
    # === Successor-release pressure: fully gated to prevent premature scheduling of low-criticality releases ===
    release_pressure = rank * work
    release_score = -PARAMS["release_pressure_weight"] * ddl_gate * normalize(release_pressure + eps)
    
    # === Nonlinear host-load interaction: gated and exponentiated for risk-concentrated work ===
    load_factor = (work * uncert + eps) ** PARAMS["load_uncertainty_exponent"]
    load_score = normalize(load_factor) * ddl_gate
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty dominates to enforce DDL feasibility first
    score = (
        slack_score
        + energy_score
        + rank_score
        + bottleneck_score_final
        + dur_score
        + wait_score
        + unc_slack_interaction
        + release_score
        + load_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
