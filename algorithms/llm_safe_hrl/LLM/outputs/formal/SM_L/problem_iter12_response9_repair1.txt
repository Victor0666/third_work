def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Bounded sigmoid urgency scaling (replacing fragile 1/|slack|^sharpness) for uncertainty-slack coupling.
      - Reintroduced duration-uncertainty blend, now *gated by ddl_gate* to preserve DDL safety while reducing energy variance.
      - All terms strictly additive, sign-consistent, and shape-preserving; no loops or hidden state.
      - Robust z-score normalization with ±sigma clipping remains for discriminability and stability.
      - Final score enforces strict finiteness and shape (N,).
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust z-score normalization with outlier clipping (±sigma) ===
    def normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x) & (np.abs(x) > eps)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        mu = np.mean(x_finite)
        std = np.std(x_finite) if len(x_finite) > 1 else eps
        z = (x - mu) / (std + eps)
        clip_bound = PARAMS["z_score_clip_sigma"]
        z_clipped = np.clip(z, -clip_bound, clip_bound)
        # Re-normalize clipped z-scores to [−1, 1] for consistent scaling
        max_abs = np.maximum(np.max(np.abs(z_clipped)), eps)
        return z_clipped / (max_abs + eps)
    
    # === Normalized slack for gating — used in sigmoid and interactions ===
    slack_norm = normalize(slk)
    
    # === Soft DDL-protection gating: smooth attenuation of criticality & energy under stress ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_sigmoid_scale"] * slack_norm))
    
    # === Slack-driven penalty/gain: linear, symmetric around zero ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse marginal energy, scaled by ddl_gate ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_gate
    
    # === Criticality: upward_rank weighted by ddl_gate (soft) ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank) * ddl_gate
    
    # === Bottleneck boost: activated only under deadline stress, non-negligible work, and sharpened by bounded sigmoid ===
    # Replace unstable power law with steep but bounded sigmoid on |slack| to avoid blowup near zero
    slack_magnitude = np.abs(slk) + eps
    # Sigmoid(-slack_magnitude * steepness) → high when slack small (tight), low when slack large
    bottleneck_urgency = 1.0 / (1.0 + np.exp(PARAMS["urgency_sigmoid_steepness"] * (slack_magnitude - eps)))
    bottleneck_sharpened = rank * work * bottleneck_urgency
    bottleneck_active = np.where((slk < 0) & (work > eps), bottleneck_sharpened, 0.0)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_active + eps)
    
    # === Duration-uncertainty blend: reintroduced and gated to reduce energy variance without violating DDL ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_blend_weight"] * uncert_norm
    dur_score = -normalize(dur_uncert_blend) * ddl_gate  # gated to avoid over-prioritizing under tight deadlines
    
    # === Uncertainty-slack-coupled urgency term: now uses bounded sigmoid instead of raw magnitude ===
    # Prioritizes tasks with high uncertainty *and* tight/negative slack, but capped smoothly
    unc_slack_coupling = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * (1.0 / (1.0 + np.exp(-PARAMS["urgency_sigmoid_steepness"] * slack_norm)))  # ~1 when slack_norm << 0
    )
    coupled_score = -unc_slack_coupling
    
    # === Starvation mitigation: exponential saturation with bounded linear fallback ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_fallback = np.clip(wait / (PARAMS["wait_decay_rate"] + eps), 0.0, 1.0)
    wait_combined = np.where(np.isfinite(wait_sat), wait_sat, wait_fallback)
    wait_score = -normalize(wait_combined + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + coupled_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
