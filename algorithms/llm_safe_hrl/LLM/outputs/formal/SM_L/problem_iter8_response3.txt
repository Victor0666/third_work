def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining robust slack gating (Parent 2) with bottleneck-aware load coupling (Parent 1).
    
    Structural improvements:
      - Retains Parent 2's stable exponential wait saturation and slack-sign gating.
      - Integrates Parent 1's validated bottleneck_boost (upward_rank * remaining_work), but gated *only* on negative slack
        — avoids over-activation during slack-rich phases while targeting critical path bottlenecks under DDL stress.
      - Uses median-based normalization (more outlier-resilient than mean-abs) from Parent 1, preserving stability across heterogeneous workloads.
      - Keeps uncertainty-slack interaction from Parent 2 for direct DDL-risk–energy tradeoff modeling.
      - All components are additive, sign-consistent, and shape-preserving; no hidden state or loops.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: median-based, epsilon-guarded (Parent 1 style) ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        med = np.median(abs_x[finite_mask]) if np.any(finite_mask) else 0.0
        scale = med if np.isfinite(med) and med > eps else eps
        return x / (scale + eps)
    
    # === Normalized slack for gating only — clean sign-based activation (Parent 2) ===
    slack_norm = normalize(slk)
    
    # === Slack-driven penalty/gain: linear, symmetric around zero, no clipping ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse marginal energy, robustly normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: activated only when slack_norm >= 0 (Parent 2 robust gating) ===
    rank_active = np.where(slack_norm >= 0, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # === Bottleneck boost: upward_rank * remaining_work, activated *only* under deadline stress (Parent 1 insight, refined gating) ===
    bottleneck_boost = rank * work
    bottleneck_active = np.where(slk < 0, bottleneck_boost, 0.0)
    bottleneck_score = -PARAMS["bottleneck_boost_weight"] * normalize(bottleneck_active + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend (stable & interpretable) ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation (Parent 2 improvement over linear clip) ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: amplifies uncertainty impact only under deadline stress (Parent 2) ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * np.where(slack_norm < 0, np.abs(slack_norm), 0.0)
    )
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
