def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Replaced hard mask with *smooth, differentiable slack gating* (sigmoid) for all deadline-sensitive terms,
        improving gradient stability for offline CMA-ES optimization.
      - Unified robust normalization using *MAD-scaled median absolute deviation*, now parameterized via
        `robust_norm_mad_scale` to preserve statistical consistency across features.
      - Removed bottleneck-proximal term (rank * work) as redundant per reflection; instead strengthened
        release pressure via *normalized product of rank and work*, fully soft-gated for smoother DDL adaptation.
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants.
      - Deterministic, finite, shape-compliant, and fully parameterized.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust median/MAD normalization with tunable MAD scaling ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        med = np.median(abs_x)
        mad = np.median(np.abs(abs_x - med)) if np.all(np.isfinite(abs_x)) else eps
        # Use parameterized MAD-to-std scaling (1.4826 ≈ normal dist. consistency)
        scale = max(med, PARAMS["robust_norm_mad_scale"] * mad, eps)
        return x / (scale + eps)
    
    # === Unified smooth DDL protection gate: differentiable, centered at 0 ===
    gate_width = PARAMS["ddl_protection_gate_width"]
    ddl_gate = 1.0 / (1.0 + np.exp(-slk / (gate_width + eps)))
    
    # === Slack-driven priority: raw penalty/gain, then robustly normalized ===
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.maximum(-slk, 0.0) +
        PARAMS["slack_urgency_gain"] * np.minimum(slk, 0.0)
    )
    slack_score = normalize(slack_penalty + eps)
    
    # === Energy efficiency: soft-gated and inverted ===
    inv_energy = 1.0 / (energy + eps)
    energy_active = inv_energy * ddl_gate
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(energy_active + eps)
    
    # === Criticality (upward_rank): soft-gated and normalized ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Duration-uncertainty coupling: robustly normalized linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: pure exponential saturation, soft-gated ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps) * ddl_gate
    
    # === Uncertainty-slack interaction: activated only under deadline stress (1-ddl_gate) ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * (1.0 - ddl_gate)
    )
    
    # === Successor-release pressure: normalized rank × work, fully soft-gated (no hard mask) ===
    release_pressure = rank * work
    release_score = -PARAMS["release_pressure_weight"] * ddl_gate * normalize(release_pressure + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_score
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + release_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
