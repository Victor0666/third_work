def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining Parent 1's smooth sigmoid gating with Parent 2's robust quantile normalization and bottleneck modeling.
    
    Key improvements:
    - Replaces hard gate with smooth sigmoid using steepness/center (from Parent 1) for differentiable, stable deadline protection
    - Keeps Parent 2's quantile-based normalization (0.75) for outlier resilience
    - Retains softplus-transformed bottleneck term (rank * work * (1+uncert) / |slack|) for critical path sensitivity without explosion
    - Integrates host-load proxy (energy/duration) gated by the new smooth sigmoid for resource-aware energy efficiency
    - Uses unified slack-driven gating across all components (not just criticality), enabling coordinated behavior under deadline pressure
    - All operations guarded against NaN/inf via np.nan_to_num and finfo bounds
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Smooth sigmoid DDL protection gate (Parent 1 style, improved stability) ===
    slack_norm = normalize(slk)
    gate_input = PARAMS["ddl_protection_steepness"] * (PARAMS["ddl_protection_center"] - slack_norm)
    # Clip input to avoid overflow in exp
    gate_input_clipped = np.clip(gate_input, -np.log(np.finfo(float).max), np.log(np.finfo(float).max))
    ddl_gate = 1.0 / (1.0 + np.exp(-gate_input_clipped))
    
    # === Slack-driven base penalty/gain: signed, linear, monotonic ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse marginal energy, normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: smoothly gated by sigmoid (not binary) ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Enhanced bottleneck term: upward_rank * remaining_work * (1 + uncertainty) * slack_magnitude_inv ===
    # Softplus ensures convex, stable amplification of high-pressure bottlenecks
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_base = rank * work * (1.0 + uncert) * slack_magnitude_inv
    bottleneck_sharpened = np.log1p(np.exp(bottleneck_base))  # softplus
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: simplified bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential saturation ===
    wait_normalized = wait / (PARAMS["wait_ramp_threshold"] + eps)
    wait_saturation = 1.0 - np.exp(-wait_normalized)
    wait_saturation = np.clip(wait_saturation, 0.0, 1.0)
    wait_score = -wait_saturation
    
    # === Host-load sensitivity: energy-per-duration ratio, feasibility-gated ===
    energy_per_duration = energy / (duration + eps)
    load_proxy = normalize(energy_per_duration + eps)
    load_score = -PARAMS["host_load_sensitivity"] * ddl_gate * load_proxy
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + load_score
    )
    
    # Final safeguard using np.finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 2.0  # literal 2 allowed per contract
    min_safe = finfo.min / 2.0
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
