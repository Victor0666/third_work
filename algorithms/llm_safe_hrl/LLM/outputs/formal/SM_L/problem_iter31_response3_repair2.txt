def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule incorporating:
      - Critical-path starvation guard: when slack <= min_slack_feasibility_margin, bypass energy/uncertainty and prioritize bottleneck unconditionally.
      - Starvation-sensitivity sigmoid replaced with fixed steepness (1.0) to reduce parameter count while preserving smooth anti-starvation.
      - Risk-weighted duration: execution + communication scaled by uncertainty ratio, normalized separately.
      - All feature interactions use quantile-normalized inputs; no unbounded exponentials or fragile denominators.
      - Strict DDL feasibility preserved: slack penalty dominates; starvation guard activates only when slack is critically low.
      - finfo_safeguard_scale replaces hidden literal 1e-12.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Critical-path starvation guard: activate unconditionally for deeply negative or near-zero slack ===
    # This overrides energy/uncertainty scoring to prevent starvation of bottleneck tasks under deadline pressure
    starvation_guard_active = (slk <= PARAMS["min_slack_feasibility_margin"])
    # Compute bottleneck score without normalization scaling (raw importance dominates)
    bottleneck_raw = rank * work * (1.0 / (np.abs(slk) + eps)) ** PARAMS["successor_release_sharpness"]
    bottleneck_score = np.where(
        starvation_guard_active,
        -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_raw + eps),
        0.0
    )
    
    # === Energy efficiency: inverse scaling with quantile-normalized inv-energy, gated by ddl_gate ===
    # Energy considered only when DDL risk is manageable (soft gating instead of hard exclusion)
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * ddl_gate * normalize(inv_energy)
    
    # === Criticality: now unconditional but scaled by ddl_gate for consistency ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Risk-weighted duration: execution + communication weighted by uncertainty ratio ===
    # Avoids conflating duration and uncertainty scales — instead, uses uncertainty to modulate duration impact
    dur = exec_t + comm_t
    dur_norm = normalize(dur + eps)
    dur_score = dur_norm * (1.0 + PARAMS["duration_uncertainty_ratio"] * normalize(uncert + eps))
    
    # === Starvation mitigation: sigmoid ramp with fixed steepness (1.0) → smooth, differentiable, no extra param ===
    wait_sigmoid = 1.0 / (1.0 + np.exp(-(wait - PARAMS["wait_ramp_threshold"])))
    wait_score = -wait_sigmoid  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant; starvation guard adds strong bottleneck bias when slack is critical
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values using finfo directly
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=finfo.max * PARAMS["finfo_safeguard_scale"],
        neginf=finfo.min * PARAMS["finfo_safeguard_scale"]
    )
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
