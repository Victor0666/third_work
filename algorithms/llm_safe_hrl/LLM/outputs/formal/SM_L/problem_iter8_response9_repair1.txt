def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining robust slack gating (Parent 2) with bottleneck-aware criticality (Parent 1).
    
    Key structural improvements:
      - Introduces *bottleneck_activation_threshold*: replaces binary sign-gating with soft threshold-based activation
        of bottleneck importance (upward_rank * remaining_work), enabling graceful ramp-up of criticality near deadlines.
      - Retains uncertainty-slack interaction from Parent 2 but applies it to *both* uncertainty and bottleneck terms
        under deadline stress — captures joint risk amplification in constrained regions.
      - Uses median-based normalization (Parent 1) for all features — more robust to outlier-dominated DAGs than mean-abs.
      - Combines exponential wait saturation (Parent 2) with bounded linear duration-uncertainty blend (Parent 1/2).
      - Removes redundant components (e.g., separate comm/exec bias, fairness terms) to reduce noise and improve CMA-ES convergence.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: uses median absolute deviation scale to resist outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        # Use median of absolute values as robust scale; fallback to eps if degenerate
        scale = np.median(abs_x) if np.all(np.isfinite(abs_x)) and np.median(abs_x) > eps else eps
        return x / (scale + eps)
    
    # === Normalized slack for gating — preserves ordering and avoids magnitude distortion ===
    slack_norm = normalize(slk)
    
    # === Slack-driven penalty/gain: linear, symmetric around zero, scaled by normalized magnitude ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: prioritize low marginal energy via inverse scaling ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Bottleneck-criticality: upward_rank * remaining_work, activated only when slack is tight ===
    # Soft activation: gate = max(0, 1 - |slack_norm| / threshold) → smooth ramp near deadline
    abs_slack_norm = np.abs(slack_norm)
    bottleneck_gate = np.clip(1.0 - abs_slack_norm / (PARAMS["bottleneck_activation_threshold"] + eps), 0.0, 1.0)
    bottleneck_importance = rank * work
    bottleneck_active = bottleneck_gate * bottleneck_importance
    bottleneck_score = -PARAMS["criticality_weight"] * normalize(bottleneck_active + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend (stable & interpretable) ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time (Parent 2) ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: amplifies both uncertainty and bottleneck under deadline stress ===
    # Activates only when slack_norm < 0 → captures risk amplification in violation-prone region
    unc_slack_mask = np.where(slack_norm < 0, np.abs(slack_norm), 0.0)
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * (uncert_norm + PARAMS["bottleneck_uncertainty_coupling_weight"] * bottleneck_active)
        * unc_slack_mask
    )
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + bottleneck_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values scaled by finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
