def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule with all thresholds declared as parameters.
    
    Fixes: replaced literal 3600 with PARAMS["max_ready_wait_cap_seconds"].
    All numeric literals in body are now only -2, -1, 0, 1, or 2.
    Uses robust MAD-based normalization and piecewise slack gating.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization: use median abs deviation (MAD) for scale stability
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        center = np.median(x)
        scale = np.median(np.abs(x - center)) + eps
        return (x - center) / scale
    
    # --- Slack processing: piecewise & risk-aware ---
    slack = np.asarray(slack, dtype=float)
    # Negative slack → strong exponential penalty (higher score = lower priority)
    neg_slack_penalty = np.where(
        slack < 0,
        PARAMS["slack_penalty_scale"] * np.exp(-PARAMS["slack_sensitivity"] * slack),
        0.0
    )
    # Near-zero positive slack: sharp power-law increase (urgency zone)
    near_zero_mask = (slack >= 0) & (slack <= PARAMS["slack_critical_threshold"])
    near_zero_urgency = np.where(
        near_zero_mask,
        PARAMS["slack_penalty_scale"] * (PARAMS["slack_critical_threshold"] - slack) ** PARAMS["slack_sensitivity"],
        0.0
    )
    # Large slack: flat baseline (no urgency bonus), but still normalized for fairness
    large_slack_baseline = np.where(slack > PARAMS["slack_critical_threshold"], robust_normalize(slack), 0.0)
    slack_score = neg_slack_penalty + near_zero_urgency + large_slack_baseline

    # --- Energy: normalized marginal risk-adjusted energy ---
    energy_score = PARAMS["energy_norm_factor"] * robust_normalize(min_incremental_energy)

    # --- Duration: split by uncertainty ratio, then robust-normalize sum ---
    duration_risk_ratio = PARAMS["duration_risk_ratio"]
    weighted_duration = (
        duration_risk_ratio * min_exec_time +
        (1.0 - duration_risk_ratio) * min_comm_time
    )
    duration_score = robust_normalize(weighted_duration)

    # --- Upward rank: boosted only under tight slack ---
    upward_rank = np.asarray(upward_rank, dtype=float)
    critical_boost_mask = slack < PARAMS["slack_critical_threshold"]
    rank_score = -robust_normalize(upward_rank) * (
        1.0 + PARAMS["criticality_boost"] * critical_boost_mask.astype(float)
    )

    # --- Ready wait time: anti-starvation with exponential decay ---
    ready_wait_time = np.asarray(ready_wait_time, dtype=float)
    # Exp decay: priority gain saturates; avoids infinite growth for old tasks
    capped_wait = np.clip(ready_wait_time, 0.0, PARAMS["max_ready_wait_cap_seconds"])
    wait_score = -np.exp(-PARAMS["wait_decay"] * capped_wait)

    # --- Uncertainty-slack interaction: amplify penalty when both high ---
    uncertainty = np.asarray(uncertainty, dtype=float)
    uncertainty_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] *
        np.abs(robust_normalize(uncertainty)) *
        np.abs(neg_slack_penalty + near_zero_urgency)
    )

    # Assemble final score — smaller = higher priority
    score = (
        slack_score
        + energy_score
        + duration_score
        + rank_score
        + wait_score
        + uncertainty_slack_interaction
    )

    # Final safeguard: replace NaN/inf with finite values; ensure shape (N,)
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    return score
