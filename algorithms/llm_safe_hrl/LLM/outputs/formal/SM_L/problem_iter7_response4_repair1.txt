import numpy as np

PARAMETER_SCHEMA = {
    "schema_version": 1,
    "parameters": [
        {
            "name": "epsilon",
            "initial_value": 0.001,
            "lower_bound": 1e-6,
            "upper_bound": 0.1,
            "semantic_description": "Numerical guard for stable normalization and division.",
            "parameter_type": "float",
            "transform": "log"
        },
        {
            "name": "slack_risk_penalty",
            "initial_value": 5.0,
            "lower_bound": 0.5,
            "upper_bound": 12.0,
            "semantic_description": "Strong linear penalty multiplier for negative slack (deadline violation risk).",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "slack_urgency_gain",
            "initial_value": 2.5,
            "lower_bound": 0.1,
            "upper_bound": 6.0,
            "semantic_description": "Linear gain for positive slack to mildly de-prioritize non-urgent tasks.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "energy_efficiency_weight",
            "initial_value": 1.2,
            "lower_bound": 0.05,
            "upper_bound": 4.0,
            "semantic_description": "Weight for inverse energy efficiency: lower incremental energy → higher priority.",
            "parameter_type": "float",
            "transform": "identity"
        },
        {
            "name": "bottleneck_proximity_weight",
            "initial_value": 0.8,

            "parameter_type": "float",

            "initial_value": 1.5,

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with:
      - Slack gating applied to *bottleneck-proximal criticality* (upward_rank * remaining_work)
      - Ready-wait time capped by bounded linear clipping instead of exponential saturation
      - Removed fragile uncertainty-slack interaction and inactive criticality_weight
      - Added robust median-based normalization for improved outlier resilience
      - Enforced hard DDL-first feasibility via slack sign gating before energy/criticality terms
      - All features scaled compatibly via median absolute scaling (more stable than mean-abs)
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust median-based normalization: resistant to outliers, preserves ordering ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        # Use median of absolute values as scale for robustness
        scale = np.median(abs_x) if np.all(np.isfinite(abs_x)) and np.median(abs_x) > eps else eps
        return x / (scale + eps)
    
    # === Slack sign gating: strict feasibility first — only activate criticality/energy when slack >= 0 ===
    # This implements hard DDL-first feasibility per design insight
    slack_feasible_mask = (slk >= 0).astype(np.float64)
    
    # === Bottleneck-proximal criticality: upward_rank * remaining_work, gated by slack feasibility ===
    bottleneck_score = rank * work
    bottleneck_score_gated = bottleneck_score * slack_feasible_mask
    bottleneck_norm = normalize(bottleneck_score_gated + eps)
    criticality_score = -PARAMS["bottleneck_proximity_weight"] * bottleneck_norm
    
    # === Slack-driven priority: linear penalty for negative slack, mild de-prioritization for large positive slack ===
    # Negative slack → strong penalty (smaller score = higher priority → so we add positive penalty)
    # Positive slack → small gain (i.e., less negative contribution → lowers priority)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(normalize(slk)),
        -PARAMS["slack_urgency_gain"] * np.abs(normalize(slk))
    )
    
    # === Energy efficiency: prefer low marginal energy, but only when slack feasible (DDL-safe zone) ===
    inv_energy = 1.0 / (energy + eps)
    energy_norm = normalize(inv_energy)
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_norm * slack_feasible_mask
    
    # === Duration-uncertainty coupling: robust linear blend of normalized duration and uncertainty ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear clipping instead of exponential decay ===
    # Prevents over-correction while ensuring long-waiting tasks get fair share
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty dominates to enforce DDL feasibility first
    score = (
        slack_penalty
        + criticality_score
        + energy_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
