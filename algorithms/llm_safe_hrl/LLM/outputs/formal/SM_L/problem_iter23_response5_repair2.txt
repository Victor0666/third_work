def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule with:
      - Soft DDL feasibility gating (sigmoid + margin).
      - Quantile-based normalization (no clipping) preserving ordinal fidelity.
      - Multiplicative uncertainty gating on both exec_t and comm_t using tunable exponent.
      - Slack-energy coupling via urgency_coupling, recentered using ddl_feasibility_margin.
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants.
      - Deterministic, finite, shape-(N,) output satisfying all interface contracts.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: preserves relative order ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Soft DDL feasibility gate: sigmoid transition centered at margin ===
    slack_offset = slk - PARAMS["ddl_feasibility_margin"]
    ddl_feasible_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_feasibility_smoothness"] * slack_offset))
    
    # === Slack-driven base penalty — dominant, raw units, linear ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        0.0
    )
    
    # === Multiplicative slack-energy coupling: peaks near zero slack, using same margin ===
    urgency_coupling = 1.0 / (1.0 + np.exp(np.abs(slk) / (PARAMS["ddl_feasibility_margin"] + eps)))
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * (
        ddl_feasible_gate + 0.5 * urgency_coupling
    )
    
    # === Criticality: normalized upward_rank, gated by DDL feasibility ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps) * ddl_feasible_gate
    
    # === Successor-release interaction: sharpens under tight slack ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Multiplicative uncertainty-gated duration: physical risk composition ===
    unc_factor = 1.0 + np.clip(uncert, 0.0, 2.0) ** PARAMS["uncertainty_gate_exponent"]
    exec_gated = exec_t * unc_factor
    comm_gated = comm_t * unc_factor
    gated_duration = exec_gated + comm_gated
    dur_norm = normalize(gated_duration + eps)
    dur_score = dur_norm * ddl_feasible_gate
    
    # === Starvation mitigation: bounded linear ramp ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)
    wait_score = -wait_ramp
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using parametrized finfo safety factor
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=finfo.max * PARAMS["finfo_safety_factor"],
        neginf=finfo.min * PARAMS["finfo_safety_factor"]
    )
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
