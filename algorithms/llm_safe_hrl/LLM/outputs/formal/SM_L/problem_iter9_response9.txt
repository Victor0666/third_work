def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with hard DDL-protection gating and simplified robustness.
    
    Key structural improvements:
      - Replaces soft bottleneck gating with *hard slack < 0 activation* for criticality and interaction:
        upward_rank only contributes when slack < 0 — strictly enforces DDL-first scheduling discipline.
      - Removes remaining_work entirely per reflection: no cross-seed benefit, adds noise, violates minimalism.
      - Reverts to mean-abs normalization (more stable under workload shifts than median) and removes fragile clipping/gating logic.
      - Simplifies uncertainty-slack interaction to pure slack-sign masking (no abs(slack_norm) scaling), improving monotonicity.
      - All components are additive, bounded, and use only safe arithmetic; no redundant features or nonlinear distortion.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: uses mean absolute value scale — stable under shifts, avoids outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        scale = np.mean(abs_x) if np.all(np.isfinite(abs_x)) and np.mean(abs_x) > eps else eps
        return x / (scale + eps)
    
    # === Slack-driven penalty/gain: linear, sign-based, magnitude-normalized ===
    slack_norm = normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse marginal energy prioritization ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Hard DDL-protection gate: upward_rank active ONLY when slack < 0 (strict deadline-first enforcement) ===
    rank_active = np.where(slk < 0, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # === Duration-uncertainty coupling: stable bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: activated *only* under hard DDL violation (slk < 0), no scaling distortion ===
    unc_slack_mask = np.where(slk < 0, 1.0, 0.0)
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * unc_slack_mask
    )
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
