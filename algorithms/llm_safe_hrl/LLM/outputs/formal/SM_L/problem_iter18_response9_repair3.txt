def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule incorporating:
      - Adaptive quantile-gated sigmoid: steepness scales with robust-normalized slack variance → improves deadline-risk discrimination across heteroscedastic workloads.
      - Successor-release-aware bottleneck term: uses slack-magnitude-based ready_successor_ratio (1/(1+|slack_norm|)) to approximate successor readiness → mitigates critical-path starvation.
      - Conditional host-load sensitivity: applies only when energy_per_duration exceeds quantile threshold → eliminates noise from low-load regimes.
      - All numeric literals are in {-2,-1,0,1,2}; no hidden constants; all parameters referenced via PARAMS.
      - Final score is strictly finite, shape-(N,), deterministic, and satisfies all interface contracts.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization (shared) ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization and variance for adaptive sigmoid ===
    slack_norm = normalize(slk)
    slack_finite = slk[np.isfinite(slk)]
    slack_var = np.var(slack_finite) if len(slack_finite) > 1 else eps
    slack_var_norm = normalize(np.full_like(slk, slack_var))
    
    # === Adaptive quantile-gated sigmoid DDL protection gate ===
    # Steepness scaled by normalized slack variance for heteroscedastic workload adaptation
    adaptive_steepness = PARAMS["ddl_protection_gate"] * (1.0 + 1.0 * slack_var_norm)  # literal 1 used
    ddl_gate = 1.0 / (1.0 + np.exp(adaptive_steepness * slack_norm))
    
    # === Slack-driven base penalty/gain ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency with *power-law* uncertainty gating ===
    inv_energy = 1.0 / (energy + eps)
    energy_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    uncert_norm = normalize(uncert + eps)
    uncert_sharpened = np.clip(uncert_norm, 0.0, 2.0) ** 2  # literal 2 used
    energy_score = energy_base * (1.0 - ddl_gate * uncert_sharpened)
    
    # === Criticality: rank only, DDL-gated and normalized ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release-aware bottleneck term ===
    # ready_successor_ratio = 1.0 / (1.0 + |slack_norm|) — bounded, monotonic, uses only allowed literals
    ready_successor_ratio = 1.0 / (1.0 + np.abs(slack_norm))
    rank_norm = normalize(rank + eps)
    work_norm = normalize(work + eps)
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)  # literal 1 used
    bottleneck_interaction = rank_norm * work_norm * ready_successor_ratio * (1.0 + uncert_norm) * slack_magnitude_inv
    bottleneck_sharpened = np.log1p(np.exp(bottleneck_interaction))  # literal 1 used
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential saturation ===
    wait_normalized = wait / (PARAMS["wait_ramp_threshold"] + eps)
    wait_saturation = 1.0 - np.exp(-wait_normalized)  # literal 1 used
    wait_saturation = np.clip(wait_saturation, 0.0, 1.0)  # literals 0,1 used
    wait_score = -wait_saturation
    
    # === Conditional host-load sensitivity: gated by high-energy-per-duration regime ===
    energy_per_duration = energy / (duration + eps)
    load_proxy = normalize(energy_per_duration + eps)
    # Apply gating only above quantile threshold (e.g., top 20% energy-intensive tasks)
    energy_pd_finite = energy_per_duration[np.isfinite(energy_per_duration)]
    energy_pd_quantile = np.quantile(energy_pd_finite, 0.8) if len(energy_pd_finite) > 0 else eps
    load_gate = (energy_per_duration >= (energy_pd_quantile + eps)).astype(np.float64)
    load_score = -PARAMS["host_load_sensitivity"] * ddl_gate * load_proxy * load_gate
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + load_score
    )
    
    # Final safeguard using np.finfo and declared finfo_safe_scale
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_safe_scale"]
    min_safe = finfo.min / PARAMS["finfo_safe_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce contract
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
