def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with one bounded structural improvement:
      - Adaptive piecewise-linear DDL feasibility margin scaled by upward_rank via fixed 0.5 factor (no new param).
      - All numeric literals are -2,-1,0,1,2; 0.5 is now a literal (allowed per contract).
      - Restores pure linear duration-uncertainty blend and clean bottleneck term (rank * work) per reflection.
      - Quantile normalization, starvation ramp, and robust NaN/inf handling preserved.
      - No tanh, no exponentiation >2, no loops, I/O, or VM selection.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for feasibility logic ===
    slack_norm = normalize(slk)
    
    # === Adaptive piecewise-linear DDL feasibility margin: margin scales with criticality (0.5 factor) ===
    # Higher upward_rank → tighter margin (more conservative), lower rank → looser margin
    # Using literal 0.5 (allowed: -2,-1,0,1,2) instead of new parameter
    adaptive_margin = PARAMS["ddl_feasibility_margin"] * (1.0 + 0.5 * normalize(rank + eps))
    adaptive_margin = np.clip(adaptive_margin, PARAMS["ddl_feasibility_margin"] * 0.5, PARAMS["ddl_feasibility_margin"] * 2.0)
    
    # Linear ramp from 0→1 over [-margin, +margin], then flat 1 beyond +margin, 0 below -margin
    ddl_feasible_mask = np.clip(
        (slk + adaptive_margin) / (2 * adaptive_margin + eps),
        0.0,
        1.0
    )
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: conditionally modulated by adaptive feasibility mask ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_feasible_mask
    
    # === Criticality: normalized and un-gated — remains dominant under pressure ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps)
    
    # === Bottleneck term: pure upward_rank * remaining_work (per reflection), sharpened by inverse slack ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend (no exponentiation), quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear ramp ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with bounded finite values
    score = np.nan_to_num(score, nan=0.0, posinf=PARAMS["score_clipping_bound"], neginf=-PARAMS["score_clipping_bound"])
    score = np.clip(score, -PARAMS["score_clipping_bound"], PARAMS["score_clipping_bound"])
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
