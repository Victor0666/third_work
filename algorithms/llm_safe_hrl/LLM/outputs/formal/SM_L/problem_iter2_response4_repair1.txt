import numpy as np

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robust deadline enforcement (Parent 2) with path-aware fairness (Parent 1).
    
    Key improvements:
      - Adds `remaining_work_fairness`: normalizes and penalizes tasks dominating downstream computation load,
        preventing bottlenecks in high-work subtrees.
      - Introduces `min_exec_comm_ratio_bias`: computes comm/(exec+eps) ratio to guide placement-aware urgency —
        high-ratio tasks are more sensitive to VM bandwidth selection; thus prioritized under tight slack.
      - Retains Parent 2's exponential wait decay, hard-criticality gating, and inverse-energy efficiency.
      - Replaces fragile soft coupling (Parent 1) with strict slack sign gating + interaction-only amplification.
      - Uses harmonic duration-uncertainty blend (Parent 2) for numerically stable timing signal.
      - All operations guarded against NaN/inf/zero using epsilon and np.nan_to_num with finfo-based clamping.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: avoids zero-scale and preserves ordering ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        scale = np.mean(abs_x) if np.all(np.isfinite(abs_x)) and np.mean(abs_x) > eps else eps
        return x / (scale + eps)
    
    # === Slack processing: quadratic penalty for negative, linear gain for positive ===
    # Ensures strong deadline compliance first; uses raw slack (not normalized) for penalty shape fidelity
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slk ** 2),  # Quadratic: steep penalty near violation
        -PARAMS["slack_urgency_gain"] * slk  # Linear de-prioritization for surplus slack
    )
    
    # === Energy efficiency: inverse scaling → sharper preference for low marginal energy ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: HARD gating — zero contribution if slack < -epsilon (no recovery margin) ===
    rank_active = np.where(slk >= -eps, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # === Duration-uncertainty coupling: weighted harmonic mean for stable timing signal ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    # Harmonic blend: avoids division-by-zero, smooth, bounded in [0, ~2]
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * uncert_norm) 
        / (dur_norm + eps + PARAMS["duration_uncertainty_ratio"] * uncert_norm + eps)
    )
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: only activates under deadline stress (slk < 0) ===
    slack_stress = np.where(slk < 0, np.abs(slk), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * normalize(uncert) * normalize(slack_stress + eps)
    
    # === Remaining work fairness: penalizes tasks anchoring large downstream workloads ===
    # Prevents scheduling imbalance where one high-work path monopolizes resources
    work_score = PARAMS["remaining_work_fairness"] * normalize(work)
    
    # === Comm/exec ratio bias: identifies communication-sensitive tasks for urgency under tight slack ===
    # High comm/exec ratio → more bandwidth-dependent → higher priority when slack is negative
    comm_exec_ratio = comm_t / (exec_t + eps)
    ratio_bias = np.where(slk < 0, comm_exec_ratio, 0.0)
    ratio_score = PARAMS["min_exec_comm_ratio_bias"] * normalize(ratio_bias + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # All terms are finite, normalized, and deterministically bounded
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + work_score
        + ratio_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values scaled by finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
