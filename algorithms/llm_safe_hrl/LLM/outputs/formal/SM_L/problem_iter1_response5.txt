def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: prioritizes DDL-critical tasks with risk-gated energy efficiency.
    
    Key innovations:
    - Nonlinear slack penalty: strong convex penalty for negative slack (deadline risk).
    - Criticality gate: activates rank-energy coupling only when both slack is tight AND 
      uncertainty is above threshold, avoiding over-prioritization of low-risk critical tasks.
    - Saturation-based wait boost: anti-starvation via smooth arctan-like saturation, not linear.
    - Robust Lp-style normalization: uses power-normalized spread instead of mean-abs scaling.
    - Energy-efficiency bias: prioritizes *low* incremental energy via inverse normalization.
    - Duration-uncertainty pacing: multiplies normalized duration and uncertainty to delay 
      high-risk long tasks unless they're critical.
    """
    epsilon = PARAMS["epsilon"]
    p = PARAMS["robustness_normalization_power"]

    # Cast to float64 and ensure finite inputs
    inputs = [
        np.asarray(min_exec_time, dtype=np.float64),
        np.asarray(min_comm_time, dtype=np.float64),
        np.asarray(min_incremental_energy, dtype=np.float64),
        np.asarray(slack, dtype=np.float64),
        np.asarray(upward_rank, dtype=np.float64),
        np.asarray(remaining_work, dtype=np.float64),
        np.asarray(ready_wait_time, dtype=np.float64),
        np.asarray(uncertainty, dtype=np.float64),
    ]

    # Safeguard against NaN/inf in inputs
    for i in range(len(inputs)):
        inputs[i] = np.nan_to_num(inputs[i], nan=0.0, posinf=1e6, neginf=-1e6)

    # Robust normalization: x / (||x||_p + epsilon), where ||x||_p = (mean(|x|^p))^(1/p)
    def robust_normalize(x):
        x_abs_p = np.abs(x) ** p
        norm = (np.mean(x_abs_p) + epsilon) ** (1.0 / p)
        return x / (norm + epsilon)

    # Normalize all features
    exec_comm = min_exec_time + min_comm_time
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(exec_comm)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # Nonlinear slack penalty: convex penalty for negative slack; zero for positive slack
    slack_penalty = np.where(
        norm_slack < 0,
        -np.abs(norm_slack) ** PARAMS["slack_penalty_exponent"],
        0.0
    )

    # Criticality gate: activate rank-energy coupling only under joint risk
    # (tight slack AND high uncertainty) — prevents over-committing to low-risk critical paths
    critical_gate = np.where(
        (norm_slack < PARAMS["criticality_gate_threshold"]) & 
        (norm_uncert > PARAMS["criticality_gate_threshold"]),
        1.0,
        0.0
    )

    # Energy efficiency: prioritize *low* energy → use inverse (avoid division by zero)
    inv_energy = 1.0 / (norm_energy + epsilon)

    # Duration-uncertainty pacing: penalize long + uncertain tasks unless critical
    dur_uncert_pace = norm_duration * norm_uncert * (1.0 - critical_gate)

    # Anti-starvation: saturating arctan-like boost (bounded, smooth, monotonic)
    wait_boost = np.arctan(norm_wait / PARAMS["wait_saturation_point"])

    # Composite score: smaller = higher priority
    # Base priority: slack risk dominates
    score = (
        # Primary: deadline risk penalty (strong, convex, negative slack only)
        slack_penalty
        # Secondary: energy efficiency bias (inverted, scaled)
        - PARAMS["energy_efficiency_bias"] * inv_energy
        # Tertiary: duration-uncertainty pacing (delays non-critical risky long tasks)
        + PARAMS["duration_uncertainty_coupling"] * dur_uncert_pace
        # Quaternary: critical-path awareness *only* when jointly risky
        - critical_gate * PARAMS["rank_energy_tradeoff"] * norm_rank
        # Anti-starvation: bounded, smooth boost for waiting tasks
        - wait_boost
        # Optional light work-load de-prioritization (larger work → lower urgency if not critical)
        + 0.2 * norm_work  # fixed small weight — structural constant per contract
    )

    # Final safeguard: ensure finite output
    score = np.nan_to_num(score, nan=1e6, posinf=1e6, neginf=-1e6)

    return score
