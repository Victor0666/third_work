def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule featuring:
      - Replaced exponential wait decay with bounded linear clipping (more stable & interpretable)
      - Added conditional DDL protection gate: when normalized slack < ddl_protection_gate,
        energy and criticality terms dominate duration/uncertainty to prevent late scheduling
      - Removed uncertainty-slack interaction (diagnosed inactive) and simplified duration blend
      - Used median-based normalization for outlier resilience per design insight
      - All components explicitly gated by slack sign or magnitude for DDL-first enforcement
      - All tunable coefficients now declared in PARAMETER_SCHEMA (no hidden literals)
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust median-based normalization: resistant to outliers and preserves ordering ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        # Use median instead of mean for scale — more robust to extreme values
        scale = np.median(abs_x) if np.all(np.isfinite(abs_x)) and np.median(abs_x) > eps else eps
        return x / (scale + eps)
    
    # === Normalized slack for gating only — used in all conditionals ===
    slack_norm = normalize(slk)
    
    # === Slack-driven components: linear penalty/gain with explicit zero-crossing ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling → sharper preference for low marginal energy ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: only active when slack is non-negative AND above protection threshold ===
    # This enforces DDL feasibility first: if slack is too tight (< gate), skip rank penalty
    rank_active = np.where(
        (slk >= 0) & (slack_norm >= PARAMS["ddl_protection_gate"]), 
        rank, 
        0.0
    )
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # === Duration-uncertainty coupling: simplified bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear clip instead of exponential decay ===
    # More stable, predictable, and avoids underflow/overflow — matches design insight
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_normalized = wait_clipped / (PARAMS["wait_clip_threshold"] + eps)
    wait_score = -normalize(wait_normalized + eps)
    
    # === DDL protection gate: overrides duration dominance when slack is critically low ===
    # When slack_norm < ddl_protection_gate, suppress duration impact, amplify energy & criticality
    protection_mask = (slack_norm < PARAMS["ddl_protection_gate"])
    # Apply soft suppression: reduce duration impact, amplify energy & criticality
    dur_score = np.where(protection_mask, dur_score * PARAMS["duration_suppression_factor"], dur_score)
    energy_score = np.where(protection_mask, energy_score * PARAMS["energy_amplification_factor"], energy_score)
    rank_score = np.where(protection_mask, rank_score * PARAMS["criticality_amplification_factor"], rank_score)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values scaled by finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
