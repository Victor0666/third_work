def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Smooth sigmoidal energy suppression (replacing threshold gate) for robust feasibility-energy tradeoff.
      - Robust min-max normalization for slack, energy, and uncertainty to stabilize cross-scenario behavior.
      - Unified risk-weighted execution score via bounded tanh: avoids linear blending fragility and ensures monotonic, saturating response to duration+uncertainty.
      - All parameters used; no unused entries; only {-2,-1,0,1,2} literals; epsilon & machine bounds via PARAMS/np.finfo.
      - Deterministic, finite, shape-(N,), side-effect-free.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust min-max normalization: stable, bounded, less epsilon-sensitive than max-abs ===
    def normalize_mm(x):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if np.any(finite_mask):
            x_min = np.min(x[finite_mask])
            x_max = np.max(x[finite_mask])
            range_val = x_max - x_min
            range_val = np.where(range_val > eps, range_val, eps)
            return (x - x_min) / range_val
        else:
            return np.zeros_like(x)
    
    # === Robust slack normalization: |slack|^p / (|slack|^p + eps) preserves sign via multiplication by sign(slk) ===
    abs_slk = np.abs(slk) + eps
    robust_slk_mag = np.power(abs_slk, PARAMS["robust_slack_normalization"])
    robust_slk_norm = robust_slk_mag / (robust_slk_mag + eps)
    slk_robust = robust_slk_norm * np.sign(slk)
    
    # === Smooth sigmoidal energy suppression gate: soft transition near zero robust slack ===
    # Gate = 1/(1 + exp(-steepness * |slk_robust|)) → approaches 1 when |slk_robust| large, ~0.5 at boundary, ~0 when tiny
    energy_suppress_gate = 1.0 / (1.0 + np.exp(-PARAMS["energy_suppression_steepness"] * np.abs(slk_robust)))
    
    # === Slack-driven base penalty/gain: applied on robust slack (preserves sign & monotonicity) ===
    slack_penalty = np.where(
        slk_robust < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk_robust),
        -PARAMS["slack_urgency_gain"] * slk_robust
    )
    
    # === Energy efficiency: inverse marginal energy, smoothly gated by robust slack magnitude ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_mm(inv_energy)
    energy_score = energy_score * energy_suppress_gate
    
    # === Criticality: upward_rank raised to slack-aware power ===
    rank_powered = np.power(rank + eps, PARAMS["criticality_exponent"])
    rank_score = -normalize_mm(rank_powered)
    
    # === Bottleneck proximity: rank * work / (|robust_slk| + bottleneck_offset) — stable, tunable ===
    robust_abs_slk = np.abs(slk_robust) + PARAMS["bottleneck_offset"]
    bottleneck_term = rank * work / robust_abs_slk
    bottleneck_score = -PARAMS["bottleneck_weight"] * normalize_mm(bottleneck_term + eps)
    
    # === Risk-weighted execution score: tanh-bounded combination of duration and uncertainty ===
    # Ensures monotonic, saturating response; avoids unbounded linear growth and fragile ratio tuning
    duration = exec_t + comm_t
    dur_norm = normalize_mm(duration + eps)
    uncert_norm = normalize_mm(uncert + eps)
    # tanh is bounded in [-1,1]; scale to [0,2] then shift+scale to meaningful range
    risk_execution_raw = np.tanh(PARAMS["risk_execution_sensitivity"] * (dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm))
    risk_execution_score = normalize_mm(risk_execution_raw + 1.0)  # map [-1,1] → [0,1] → normalized again for consistency
    
    # === Starvation mitigation: exponential saturation on raw wait time, scaled by tunable gain ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -PARAMS["starvation_gain"] * wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + risk_execution_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf using machine-precision-safe bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 2.0, neginf=finfo.min / 2.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
