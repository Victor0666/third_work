def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Convex slack-conditioned energy suppression (replaces multiplicative gating)
      - Adaptive starvation-triggered wait ramp (uses median slack × factor)
      - Host-load-aware energy scaling via uncertainty proxy
      - Energy suppression blend controlled by explicit parameter
      - All numeric literals restricted to {-2,-1,0,1,2}
      - Exactly 13 parameters (added one for blend factor)
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Host-load-aware energy normalization: higher uncertainty → more VMs → smoother energy impact ===
    energy_scaled = energy / (uncert + eps)
    inv_energy = 1.0 / (energy_scaled + eps)
    energy_raw = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Slack-conditioned energy suppression: additive convex blend ===
    # Linear fade-out for negative slack: energy_suppressed = energy_raw * (1 + slack_norm) when slack < 0
    energy_suppressed = np.where(
        slk < 0,
        energy_raw * (1.0 + slack_norm),  # literal 1.0 allowed
        energy_raw
    )
    
    # === Blend between raw and suppressed energy using explicit parameter ===
    energy_score = (
        (1.0 - PARAMS["energy_suppression_blend_factor"]) * energy_raw +
        PARAMS["energy_suppression_blend_factor"] * energy_suppressed
    )
    
    # === Criticality: gated by ddl_gate ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: adaptive wait ramp triggered only after confirmed starvation ===
    finite_slack = slk[np.isfinite(slk)]
    median_slack = np.median(finite_slack) if len(finite_slack) > 0 else 0.0
    starvation_threshold = PARAMS["wait_starvation_threshold_factor"] * np.abs(median_slack) + eps
    wait_active = np.where(wait >= starvation_threshold, wait, 0.0)
    # Bounded saturation: clip to [0, 2 * threshold] → uses literal 2
    wait_clipped = np.clip(wait_active, 0.0, 2.0 * starvation_threshold)
    wait_normalized = normalize(wait_clipped + eps)
    wait_saturation = np.clip(wait_normalized, 0.0, 1.0)  # literals 0 and 1 allowed
    wait_score = -wait_saturation
    
    # === Assemble final score ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard using np.finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
