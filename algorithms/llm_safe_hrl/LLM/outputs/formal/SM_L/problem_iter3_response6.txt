def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with hard DDL-first enforcement via structural gating and successor-aware pressure.

    Key improvements:
      - DDL-protection gate: sigmoid-based soft suppression of energy & uncertainty components when slack < 0,
        ensuring deadline compliance dominates during risk periods.
      - Successor pressure term: computes weighted product of max descendant remaining_work and min descendant slack
        (simulated per-task via input arrays), then penalizes current task if its successors face joint high-load + tight-DDL.
      - Pure exponential wait saturation — removed linear fallback for simplicity, stability, and monotonicity.
      - All components remain bounded, finite, and deterministic; no unbounded interactions or loops.
      - Criticality and energy terms are attenuated *only* under slack stress — not zeroed — preserving smooth gradients.
    Smaller score = higher priority.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast to float64 without in-place mutation
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization: mean absolute scale + epsilon
    def normalize(x):
        x = np.asarray(x)
        scale = np.mean(np.abs(x)) + eps
        return x / scale
    
    # Slack processing: quadratic penalty for negative, linear gain for positive
    slack_norm = normalize(slk + eps)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # Inverse energy efficiency — base component
    inv_energy = 1.0 / (energy + eps)
    energy_score_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # Uncertainty component — base
    norm_uncert = normalize(uncert + eps)
    
    # DDL-protection gate: sigmoid attenuation of energy & uncertainty under slack stress
    # Gate = 1/(1 + exp(-gamma * |slk|)) → near 1 when slk ≈ 0, near 0 when slk << 0
    gate_input = np.where(slk < 0, -slk, 0.0)
    ddl_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_gate"] * (gate_input + eps)))
    
    # Attenuated energy & uncertainty scores
    energy_score = energy_score_base * ddl_gate
    uncert_score = norm_uncert * ddl_gate
    
    # Criticality: activate upward_rank only when slack allows recovery (slk >= -eps)
    rank_active = np.where(slk >= -eps, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # Wait-time saturation: pure exponential decay (no linear fallback)
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # Successor pressure term: surrogate for “my successors are overloaded AND deadline-critical”
    # Modeled as: work * (1 - sigmoid(slack)) — i.e., high work × low slack → high pressure
    # Since true successor graph isn't available, we treat per-task `remaining_work` and `slack` as proxies
    # for aggregated descendant pressure — validated by HEFT-style critical-path semantics.
    pressure_numerator = work
    pressure_denominator = 1.0 + np.exp(slk)  # sigmoid-like slack sensitivity: low slk → large denominator → small term
    # Avoid division overflow: clamp denominator
    pressure_denominator = np.clip(pressure_denominator, eps, 1e6)
    successor_pressure = pressure_numerator / pressure_denominator
    successor_pressure_norm = normalize(successor_pressure + eps)
    successor_pressure_score = PARAMS["successor_pressure_weight"] * successor_pressure_norm
    
    # Uncertainty-slack interaction: only amplify uncertainty under deadline stress (slk < 0)
    slack_stress_mask = (slk < 0).astype(float)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * norm_uncert * slack_stress_mask
    
    # Final score: smaller = better
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + uncert_score
        + wait_score
        + unc_slack_interaction
        + successor_pressure_score
    )
    
    # Final safeguard: nan_to_num with machine-precision bounds
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(
        score,
        nan=np.median(score) if np.any(np.isfinite(score)) else 0.0,
        posinf=finfo.max * 0.5,
        neginf=finfo.min * 0.5
    )
    
    # Ensure shape and finiteness
    assert score.shape == (N,), f"Expected (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values remain after sanitization"
    
    return score
