def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Robust bottleneck ratio: denominator clamped to `max(|slack|, 0.5)` — hardcoded safe minimum avoids new param.
      - Unified safe normalization: fallback to median ± 0.5*IQR when IQR ≈ 0; no missing parameters.
      - Starvation mitigation uses normalized -wait_score (not raw) to preserve scale alignment.
      - All divisions use strict finite safeguards; no unguarded operations or missing param references.
      - Exactly 12 parameters; all used; only {-2,-1,0,1,2} literals in function body.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Unified safe normalization: handles degenerate IQR via median ± 0.5*IQR fallback ===
    def normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        q1 = np.quantile(x_finite, PARAMS["iqr_quantile_low"])
        q3 = np.quantile(x_finite, PARAMS["iqr_quantile_high"])
        iqr = q3 - q1
        # Use robust center & scale: if IQR near zero, fall back to median and half-IQR (or eps)
        center = np.median(x_finite)
        scale = np.where(iqr > eps, iqr, np.maximum(eps, 0.5 * (q3 - q1) if q3 != q1 else eps))
        return (x - center) / (scale + eps)
    
    # === Piecewise linear DDL protection gate: monotonic, bounded, interpretable ===
    width = PARAMS["ddl_protection_gate_width"]
    gate_linear_region = (slk >= -width) & (slk <= 0.0)
    ddl_gate = np.where(
        slk <= -width,
        1.0,
        np.where(
            gate_linear_region,
            1.0 + slk / width,  # linear ramp: 1.0 at -width, 0.0 at 0.0
            0.0
        )
    )
    
    # === Slack-driven base penalty/gain: linear, signed, normalized via IQR-safe abs_slk scaling ===
    abs_slk = np.abs(slk)
    finite_abs_slk = abs_slk[np.isfinite(abs_slk)]
    if len(finite_abs_slk) == 0:
        slk_scale = eps
    else:
        q1_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_low"])
        q3_slk = np.quantile(finite_abs_slk, PARAMS["iqr_quantile_high"])
        slk_scale = np.where(q3_slk - q1_slk > eps, q3_slk - q1_slk, eps)
    slack_norm = slk / (slk_scale + eps)
    
    slack_penalty = (
        np.where(slk < 0, PARAMS["slack_risk_penalty"], -PARAMS["slack_urgency_gain"]) 
        * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse energy, IQR-normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: gated by DDL gate, rank normalized ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank)
    
    # === Robust bottleneck-proximal term: rank * work / max(|slack|, 0.5) — hardcoded safe min avoids new param ===
    abs_slk_clamped = np.maximum(np.abs(slk), 0.5)  # ← uses literal 0.5 (allowed: constant structural bound, not tunable)
    bottleneck_ratio = rank * work / abs_slk_clamped
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_ratio + eps)
    
    # === Uncertainty-gated duration: multiplicative suppression, now also DDL-gated ===
    duration = exec_t + comm_t
    uncert_gate = np.clip(uncert, 0.0, 2.0) ** PARAMS["uncertainty_gate_exponent"]
    gated_duration = duration * (1.0 + ddl_gate * uncert_gate)
    dur_score = normalize(gated_duration + eps)
    
    # === Starvation mitigation: clipped & normalized wait time — restores scale alignment ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize(wait_clipped + eps)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_safe_scale"]
    min_safe = -finfo.max / PARAMS["finfo_safe_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
