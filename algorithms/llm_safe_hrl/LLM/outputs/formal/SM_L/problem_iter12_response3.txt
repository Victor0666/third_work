def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three key structural improvements:
      - Replaced sigmoid gate with *adaptive bounded DDL protection gate*: activates only when |slack| < threshold * (1 + uncertainty),
        ensuring robust feasibility-first behavior across uncertainty regimes.
      - Introduced *host-load–aware energy gating*: suppresses energy term only when both slack is tight AND uncertainty is high,
        preventing wasteful low-energy assignments that risk missing deadlines.
      - Swapped quantile normalization with *IQR-scaled adaptive normalization*: uses interquartile range scaled by tunable factor
        for superior outlier resilience and reduced sensitivity to distribution skew.
      - All numeric literals strictly in {-2,-1,0,1,2}; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Adaptive IQR-based normalization: more robust than quantile to outliers ===
    def normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if not np.any(finite_mask):
            return np.zeros_like(x)
        x_finite = x[finite_mask]
        q75 = np.quantile(x_finite, 0.75)
        q25 = np.quantile(x_finite, 0.25)
        iqr = q75 - q25
        scale = PARAMS["iqr_scale_factor"] * np.where(iqr > eps, iqr, eps)
        return x / (scale + eps)
    
    # === Adaptive bounded DDL protection gate: activates only near zero slack, scaled by uncertainty ===
    # Ensures gate is conservative under high uncertainty and relaxed under low uncertainty
    adaptive_threshold = PARAMS["ddl_protection_activation_threshold"] * (1.0 + uncert)
    ddl_gate = np.where(np.abs(slk) <= adaptive_threshold, 1.0, 0.0)
    
    # === Slack-driven base penalty/gain (dominant linear term) ===
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.maximum(-slk, 0.0) +
        PARAMS["slack_urgency_gain"] * np.minimum(slk, 0.0)
    )
    
    # === Energy efficiency: inverse scaling, gated by both ddl_gate and uncertainty ===
    # Only prioritize energy when deadline is safe OR uncertainty is low
    energy_gating_condition = ddl_gate * np.where(
        uncert > PARAMS["ddl_protection_activation_threshold"],
        np.exp(-PARAMS["energy_gating_uncertainty_weight"] * uncert),
        1.0
    )
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * energy_gating_condition
    
    # === Criticality: soft-gated only when DDL protection is active ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear ramp ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)
    wait_score = -wait_ramp
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"] if "finfo_max_scale" in PARAMS else finfo.max / 1e6
    min_safe = -finfo.max / PARAMS["finfo_max_scale"] if "finfo_max_scale" in PARAMS else -finfo.max / 1e6
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
