def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining best structural elements:
      - Piecewise-linear DDL feasibility margin (Parent 2) + smooth energy modulation via tanh (novel hybrid gate)
      - Bottleneck-proximal interaction with inverse-slack sharpening (Parent 2), now enhanced with exponentiation on uncertainty-weighted work
      - Quantile-based normalization (Parent 2) applied uniformly across all features
      - Nonlinear duration-uncertainty coupling: power-law blend instead of linear, better capturing risk compounding
      - Wait-time mitigation: unit-ramped and normalized, preserving ordering under starvation
      - All terms additive, deterministic, finite, and shape-strict (N,) — no loops, randomness, or side effects.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for feasibility logic ===
    slack_norm = normalize(slk)
    
    # === Smooth DDL modulation gate: tanh-based transition centered at -margin for sharper deadline pressure onset ===
    # Gate ≈ 0 when slack >> -margin (safe), ≈ 1 when slack << -margin (critical); continuous & bounded
    gate_center = -PARAMS["ddl_feasibility_margin"]
    gate_input = PARAMS["energy_modulation_steepness"] * (slk - gate_center)
    # Clip input to avoid tanh overflow (tanh saturates beyond ~±10)
    gate_input_clipped = np.clip(gate_input, -10.0, 10.0)
    energy_modulation_gate = 0.5 * (1.0 - np.tanh(gate_input_clipped))  # [0,1], rising as slack falls
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: smoothly modulated by tanh gate — full suppression only under hard violation ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * energy_modulation_gate
    
    # === Criticality: un-gated, quantile-normalized — critical path always matters ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps)
    
    # === Bottleneck-proximal interaction: sharpened by inverse slack AND uncertainty-weighted work ===
    # Emphasizes tasks blocking large, uncertain workloads especially when slack is tight
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    uncert_work = work * (1.0 + uncert)  # amplify work by uncertainty exposure
    bottleneck_sharpened = rank * uncert_work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: nonlinear power-law blend for risk-aware compounding ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    # Use exponent >1 to penalize high-duration + high-uncertainty combinations more than linearly
    dur_uncert_blend = np.power(
        np.abs(dur_norm) + PARAMS["duration_uncertainty_ratio"] * np.abs(uncert_norm),
        PARAMS["dur_uncert_nonlinear_exponent"]
    )
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear ramp on wait time → monotonic, smooth, preserves ordering ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant; others provide nuanced tradeoffs within feasibility envelope
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with bounded finite values using machine epsilon context
    score = np.nan_to_num(score, nan=0.0, posinf=PARAMS["score_clipping_bound"], neginf=-PARAMS["score_clipping_bound"])
    score = np.clip(score, -PARAMS["score_clipping_bound"], PARAMS["score_clipping_bound"])
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
