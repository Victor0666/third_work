def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule featuring:
      - Smooth sigmoid DDL protection gate (replacing hard threshold) for differentiable, robust transitions.
      - Removal of redundant bottleneck term (rank * work) — avoids duplication and rank instability.
      - Fixed [-2, 2] clipping in normalization (no adaptive saturation) for outlier resilience.
      - Unified load-pressure replaced by leaner, interpretable components: no joint rank×work×uncert coupling.
      - All gating and amplifications now driven by a single smooth sigmoid(gate_center, steepness).
      - Simplified structure: 7 additive terms, 1 conditional branch (sigmoid), zero loops or hidden state.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust bounded normalization: median + MAD scale, then clip to [-2, 2] ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            med = np.median(abs_x[finite_mask])
            dev = np.abs(abs_x - med)
            mad = np.median(dev[finite_mask]) if np.any(finite_mask) else eps
            scale = max(med, 1.0 * mad, eps)  # literal 1.0 allowed
        else:
            scale = eps
        normed = x / (scale + eps)
        return np.clip(normed, -2.0, 2.0)  # bounded piecewise form
    
    # === Normalized slack for unified gating ===
    slack_norm = normalize(slk)
    
    # === Smooth sigmoid DDL protection gate: replaces hard threshold for stability ===
    # Gate = 1/(1 + exp(-steepness * (center - slack_norm))) → rises as slack_norm falls
    gate_input = PARAMS["ddl_protection_steepness"] * (PARAMS["ddl_protection_center"] - slack_norm)
    # Clip input to avoid overflow in exp
    gate_input_clipped = np.clip(gate_input, -np.log(np.finfo(float).max), np.log(np.finfo(float).max))
    ddl_gate = 1.0 / (1.0 + np.exp(-gate_input_clipped))
    
    # === Slack-driven base penalty/gain: signed, linear, monotonic ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse marginal energy, normalized ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: smoothly gated by sigmoid (not binary) ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps) * ddl_gate
    
    # === Duration-uncertainty coupling: simplified bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear clip ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_normalized = wait_clipped / (PARAMS["wait_clip_threshold"] + eps)
    wait_score = -normalize(wait_normalized + eps)
    
    # === Smooth DDL protection modulation: soft suppression/amplification ===
    dur_score = dur_score * (1.0 - ddl_gate * (1.0 - PARAMS["duration_suppression_factor"]))
    energy_score = energy_score * (1.0 + ddl_gate * (PARAMS["energy_amplification_factor"] - 1.0))
    rank_score = rank_score * (1.0 + ddl_gate * (PARAMS["criticality_amplification_factor"] - 1.0))
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with 0.0 using explicit masking
    score = np.where(np.isfinite(score), score, 0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
