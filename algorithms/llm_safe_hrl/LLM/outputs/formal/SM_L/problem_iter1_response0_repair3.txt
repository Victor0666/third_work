def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL risk containment first, then energy-aware balancing.
    
    Key innovations:
    - Piecewise slack handling: strong convex penalty for negative slack (via exponent + threshold)
    - Critical-path activation gate: upward_rank only boosts priority above a normalized threshold
    - Saturation-based anti-starvation: ready_wait_time contributes sublinearly (tanh-scaled)
    - Uncertainty-coupled risk: uncertainty amplifies slack penalty *only* when slack < 0
    - Balanced duration-energy tradeoff: combines exec+comm and incremental energy via robust ratio
    - All normalizations use median absolute deviation (MAD) for outlier resilience, not mean abs
    """
    epsilon = PARAMS["epsilon"]
    N = len(slack)
    
    # Defensive copy & cast
    min_exec_time = np.asarray(min_exec_time, dtype=float).copy()
    min_comm_time = np.asarray(min_comm_time, dtype=float).copy()
    min_incremental_energy = np.asarray(min_incremental_energy, dtype=float).copy()
    slack = np.asarray(slack, dtype=float).copy()
    upward_rank = np.asarray(upward_rank, dtype=float).copy()
    remaining_work = np.asarray(remaining_work, dtype=float).copy()
    ready_wait_time = np.asarray(ready_wait_time, dtype=float).copy()
    uncertainty = np.asarray(uncertainty, dtype=float).copy()
    
    # Robust normalization: MAD-based (more outlier-resistant than mean abs)
    def robust_normalize(x):
        x = np.clip(x, -1e12, 1e12)  # prevent inf propagation
        center = np.median(x)
        dev = np.median(np.abs(x - center))
        scale = np.maximum(dev, epsilon)
        return (x - center) / scale
    
    # Normalize base features
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)
    
    # === SLACK RISK PRIORITIZATION (DDL-hard constraint enforcement) ===
    # Strong convex penalty for negative slack; zero penalty for slack >= threshold
    slack_risk_mask = (slack < PARAMS["critical_slack_threshold"])
    slack_penalty = np.zeros(N)
    slack_penalty[slack_risk_mask] = (
        np.abs(norm_slack[slack_risk_mask]) ** PARAMS["slack_risk_exponent"]
        * (1.0 + PARAMS["uncertainty_slack_coupling"] * norm_uncert[slack_risk_mask])
    )
    
    # === CRITICAL-PATH ACTIVATION GATE ===
    # Only amplify priority for high upward_rank (prevents over-prioritizing trivial nodes)
    rank_boost = np.zeros(N)
    rank_threshold_norm = np.clip(
        (PARAMS["rank_activation_threshold"] - np.min(norm_rank)) 
        / (np.maximum(np.ptp(norm_rank), epsilon) + epsilon),
        0.0, 1.0
    )
    rank_active_mask = (norm_rank > rank_threshold_norm)
    rank_boost[rank_active_mask] = norm_rank[rank_active_mask]
    
    # === ANTI-STARVATION: SATURATED WAIT TIME CONTRIBUTION ===
    # tanh scaling prevents runaway priority from long waits; saturates near ±1
    wait_contribution = np.tanh(norm_wait / PARAMS["wait_saturation_point"])
    
    # === ENERGY-AWARE BALANCING IN FEASIBLE REGION ===
    # Only apply energy sensitivity when slack is non-negative (DDL-safe zone)
    energy_weight_mask = (slack >= PARAMS["critical_slack_threshold"])
    energy_contribution = np.zeros(N)
    energy_contribution[energy_weight_mask] = (
        PARAMS["energy_sensitivity"] * norm_energy[energy_weight_mask]
    )
    
    # === DURATION-ENERGY TRADEOFF RATIO (robust sign-preserving) ===
    # Avoid division by zero; use signed ratio with bounded denominator
    denom = np.abs(norm_duration) + np.abs(norm_energy) + epsilon
    duration_energy_ratio = (norm_duration - norm_energy) / denom
    balanced_duration_energy = PARAMS["duration_balance_factor"] * duration_energy_ratio
    
    # === FINAL SCORE: smaller = higher priority ===
    # Slack risk dominates (penalty → increases score → lowers priority → correct)
    # Other beneficial terms subtracted to lower score → raise priority
    score = (
        slack_penalty  # high = bad → high score → low priority
        - 0.5 * rank_boost  # high rank → boost priority → subtract
        + 0.3 * balanced_duration_energy  # balance: prefer shorter duration & lower energy → lower score
        - 0.4 * wait_contribution  # longer wait → higher contribution → subtract → higher priority
        + energy_contribution  # in safe region, lower energy → lower score
        + 0.1 * norm_work  # smaller remaining work → less critical → slightly raise score (lower priority)
    )
    
    # Final safeguard: ensure finite, replace NaN/inf using declared bounds
    finfo = np.finfo(float)
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=PARAMS["clip_upper_bound"],
        neginf=-PARAMS["clip_upper_bound"]  # symmetric clip avoids log-bound violation
    )
    
    # Clamp to avoid extreme outliers
    score = np.clip(
        score,
        -PARAMS["clip_upper_bound"],
        PARAMS["clip_upper_bound"]
    )
    
    return score
