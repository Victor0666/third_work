def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three key structural innovations:
      - Added host-load-aware energy suppression: reduces energy priority when local load is high (via uncertainty-triggered gating)
      - Replaced linear wait ramp with piecewise-linear saturation: sharper response at low wait times, smoother asymptote
      - Introduced robust slack-aware criticality: upward_rank weighted by normalized |slack|, enabling dynamic criticality scaling
      - Removed finfo_max_scale (inactive per diagnostics) and simplified safeguards using np.finfo directly
      - All normalizations now use max-abs for better stability under sparse outliers
      - Energy term is conditionally suppressed only under high uncertainty AND tight slack, preserving DDL-first behavior
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust max-abs normalization: more stable than quantile under sparse extremes ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling with quantile-normalized inv-energy, gated by load/uncertainty ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # Host-load suppression: reduce energy priority when uncertainty is high AND slack is tight
    # Uses piecewise-linear suppression: active only when uncert > 0.5 * median and |slack| < median_slack
    finite_uncert = uncert[np.isfinite(uncert)]
    finite_slack = slk[np.isfinite(slk)]
    median_uncert = np.median(finite_uncert) if len(finite_uncert) > 0 else eps
    median_slack = np.median(np.abs(finite_slack)) if len(finite_slack) > 0 else eps
    load_suppress_mask = (
        (uncert > PARAMS["host_load_suppression_factor"] * median_uncert) &
        (np.abs(slk) < median_slack)
    )
    energy_score = np.where(load_suppress_mask, energy_score * (1.0 - PARAMS["host_load_suppression_factor"]), energy_score)
    
    # === Criticality: dynamically scaled by normalized |slack| — less critical when slack large, more when tight ===
    # Avoids over-prioritizing critical tasks when deadline is relaxed
    slack_magnitude_norm = normalize(np.abs(slk) + eps)
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps) * (1.0 + slack_magnitude_norm)
    
    # === Successor-release interaction: sharpened bottleneck term with inverse slack exponentiation ===
    # Prioritizes tasks blocking large workloads *especially* when slack is tight
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, max-abs-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: piecewise-linear wait ramp — faster initial rise, then smooth saturation ===
    # Ramps from 0→1 over [0, wait_ramp_threshold * wait_ramp_linear_fraction], then stays flat
    wait_normalized = wait / (PARAMS["wait_ramp_threshold"] + eps)
    wait_ramp = np.where(
        wait_normalized <= PARAMS["wait_ramp_linear_fraction"],
        wait_normalized / PARAMS["wait_ramp_linear_fraction"],
        1.0
    )
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max * 0.5, neginf=finfo.min * 0.5)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
