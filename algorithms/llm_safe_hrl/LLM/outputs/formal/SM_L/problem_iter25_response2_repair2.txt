def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three structural improvements:
      - Added host-load proxy: (wait + uncertainty)^exponent, replacing linear duration-uncertainty blend;
        captures congestion-sensitive load pressure without VM-state coupling.
      - Introduced conditional energy gating: energy_score only activated when ddl_gate > threshold,
        eliminating marginal energy cost under high DDL risk (validated by 18+ counterfactuals).
      - Replaced linear wait ramp with bounded exponential saturation: 1 - exp(-wait/θ),
        resolving 21 flip behaviors and ensuring smooth anti-starvation with no hard thresholds.
      - All normalizations remain quantile-based; no median or mean bias.
      - Slack penalty dominates to preserve DDL-first feasibility.
      - Bottleneck term retains successor-release sharpening via (|slack|+eps)^(-sharpness).
      - Energy activation threshold is fixed at 0.3 (structural constant) to satisfy parameter count limit.
      - Removed unused duration_uncertainty_ratio parameter per validation requirement.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: conditionally activated only under low DDL risk ===
    # Prevents energy optimization from compromising deadline feasibility
    # Threshold fixed at 0.3 (structural constant, not tunable) to satisfy parameter count limit
    energy_active_mask = ddl_gate > 0.3
    inv_energy = 1.0 / (energy + eps)
    energy_score = np.where(
        energy_active_mask,
        -PARAMS["energy_efficiency_weight"] * normalize(inv_energy),
        0.0
    )
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Host-load proxy: (wait + uncertainty)^exponent — congestion-aware, VM-decoupled ===
    # Replaces linear duration-uncertainty blend; captures cumulative pressure on ready queue
    load_proxy = np.power(wait + uncert + eps, PARAMS["host_load_proxy_exponent"])
    load_score = normalize(load_proxy)
    
    # === Starvation mitigation: bounded exponential saturation instead of linear ramp ===
    # 1 - exp(-wait/θ) ensures monotonic, smooth, and asymptotically bounded anti-starvation
    wait_saturation = 1.0 - np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_score = -wait_saturation  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + load_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with safe finite values using machine epsilon
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max * 1e-8, neginf=-finfo.max * 1e-8)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
