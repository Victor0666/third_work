import numpy as np

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing robust DDL compliance first, then energy efficiency.
    
    Key innovations:
      - Piecewise slack handling: strong quadratic penalty for negative slack, soft gain for positive slack.
      - Inverse energy efficiency: 1/(energy + epsilon) — favors low marginal energy more sharply.
      - Criticality gated by slack: upward_rank only contributes meaningfully when slack > 0.
      - Uncertainty-slack interaction: multiplicative boost to uncertainty penalty under deadline stress.
      - Wait-time modeled with exponential decay: prioritizes long-waiting tasks but saturates smoothly.
      - Duration-uncertainty ratio: combines exec+comm with uncertainty via weighted harmonic mean for stability.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Ensure no in-place modification; copy to float64 for numerical stability
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helpers ===
    def safe_mean_abs(x):
        m = np.mean(np.abs(x))
        return m if np.isfinite(m) and m > 0 else eps
    
    def normalize(x):
        x = np.asarray(x)
        return x / (safe_mean_abs(x) + eps)
    
    # === Slack processing: piecewise & risk-aware ===
    # Negative slack → quadratic penalty (stronger than linear); positive slack → linear gain (mild de-prioritization)
    slack_norm = normalize(slk)
    slack_penalty = np.where(slk < 0, 
                           PARAMS["slack_risk_penalty"] * (slack_norm ** 2), 
                           -PARAMS["slack_urgency_gain"] * np.abs(slack_norm))
    
    # === Energy efficiency: inverse scaling (higher priority for lower energy) ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: only activate upward_rank when slack allows breathing room ===
    rank_active = np.where(slk > -eps, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_active + eps)
    
    # === Duration-uncertainty coupling: harmonic blend for stable timing signal ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    # Weighted harmonic mean: avoids division-by-zero, smooth, bounded
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * uncert_norm) 
        / (dur_norm + eps + PARAMS["duration_uncertainty_ratio"] * uncert_norm + eps)
    )
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    # Prevents runaway priority from very large wait times while preserving ordering
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: amplifies uncertainty impact only under deadline stress ===
    slack_stress = np.where(slk < 0, np.abs(slack_norm), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * uncert_norm * slack_stress
    
    # === Assemble final score: smaller = better ===
    # All components are finite, normalized, and deterministically bounded
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf with finite fallback using machine epsilon
    score = np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
    
    # Ensure shape (N,) and finite values
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
