def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with absolute DDL gate, dynamic-range min-max normalization, and soft sigmoid criticality fading.
    
    Key structural improvements:
      - Replaced relative DDL gate with ABSOLUTE margin (`ddl_feasibility_absolute_margin`) for robustness across heterogeneous deadline scales.
      - Swapped clipped-percentile scaling with dynamic-range min-max: computes [p-th, 99.9-th] bounds per feature using tunable percentiles.
      - Upward rank now uses smooth sigmoid fade: `sigmoid(slack / max_slack)` instead of hard zeroing → preserves gradient & avoids priority cliffs.
      - All normalizations are bounded, finite, and deterministic; no reliance on global statistics that break under single-task or skewed distributions.
      - `remaining_work` remains unused (not aligned with energy or deadline objective), but signature preserved per contract.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs to float64; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Hard DDL feasibility gate: absolute margin ensures scale-invariance ===
    ddl_feasible = slk >= -PARAMS["ddl_feasibility_absolute_margin"]
    feasible_mask = ddl_feasible.astype(np.float64)
    infeasible_mask = (1.0 - feasible_mask)
    
    # === Dynamic-range min-max normalization: robust to outliers & small-N ===
    def dynamic_minmax_scale(x):
        x = np.asarray(x)
        if not np.any(np.isfinite(x)):
            return np.zeros_like(x)
        # Use asymmetric clipping: lower bound at p-th percentile, upper at fixed 99.9-th percentile
        lo = np.percentile(x, PARAMS["minmax_clip_percentile"]) if np.any(np.isfinite(x)) else 0.0
        hi = np.percentile(x, 99.9) if np.any(np.isfinite(x)) else 1.0
        rng = hi - lo
        rng_safe = np.where(rng > eps, rng, eps)
        clipped = np.clip(x, lo + eps, hi - eps)
        return 2.0 * (clipped - lo - eps) / rng_safe - 1.0
    
    # Duration = exec + comm
    duration = exec_t + comm_t
    norm_duration = dynamic_minmax_scale(duration)
    
    # Energy: inverse efficiency score, masked by feasibility
    inv_energy = 1.0 / (energy + eps)
    norm_inv_energy = dynamic_minmax_scale(inv_energy)
    energy_score = -PARAMS["energy_efficiency_weight"] * norm_inv_energy * feasible_mask
    
    # Slack: quadratic penalty for negative, linear gain for positive — always computed
    norm_slack = dynamic_minmax_scale(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (norm_slack ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(norm_slack)
    )
    
    # Soft criticality gate: sigmoid fade instead of hard zeroing → smooth gradient near deadline boundary
    # Compute max slack magnitude for scaling, avoiding division-by-zero
    max_slack_mag = np.max(np.abs(slk)) if np.any(np.isfinite(slk)) else 1.0
    max_slack_mag_safe = np.where(max_slack_mag > eps, max_slack_mag, eps)
    # Sigmoid: 1 / (1 + exp(-x)) → maps slack/max to [0,1]; decays smoothly as slack → -∞
    slack_normalized_for_sigmoid = slk / max_slack_mag_safe
    sigmoid_fade = 1.0 / (1.0 + np.exp(-PARAMS["criticality_fade_slope"] * slack_normalized_for_sigmoid))
    rank_active = rank * sigmoid_fade  # soft gating, not hard zeroing
    norm_rank = dynamic_minmax_scale(rank_active + eps)
    rank_score = -PARAMS["criticality_weight"] * norm_rank * feasible_mask
    
    # Duration-uncertainty coupling: harmonic blend with clipped inputs
    norm_uncert = dynamic_minmax_scale(uncert + eps)
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * norm_uncert) 
        / (np.abs(norm_duration) + eps + PARAMS["duration_uncertainty_ratio"] * np.abs(norm_uncert) + eps)
    )
    dur_score = dynamic_minmax_scale(dur_uncert_blend)
    
    # Starvation mitigation: exponential saturation (robust, parameter-efficient)
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    norm_wait_sat = dynamic_minmax_scale(wait_sat + eps)
    wait_score = -norm_wait_sat * feasible_mask
    
    # Uncertainty-slack interaction: only under deadline stress (negative slack AND feasible)
    slack_stress = np.where((slk < 0) & ddl_feasible, np.abs(norm_slack), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * norm_uncert * slack_stress
    
    # === Assemble base score for feasible tasks only ===
    base_score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Hard penalty for infeasible tasks: dominates all other terms
    finfo = np.finfo(np.float64)
    inf_penalty = finfo.max / PARAMS["inf_scale"]
    score = base_score * feasible_mask + inf_penalty * infeasible_mask
    
    # Final safeguard: replace NaN/inf using declared scale and median fallback
    score = np.nan_to_num(
        score,
        nan=np.median(score) if np.any(np.isfinite(score)) else 0.0,
        posinf=inf_penalty,
        neginf=-inf_penalty
    )
    
    # Ensure shape (N,) and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
