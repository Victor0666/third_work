def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining DDL-hard constraint enforcement, robust criticality gating,
    inverse energy efficiency, exponential wait saturation, and MAD-based normalization.
    
    Key improvements over parents:
      - Uses median absolute deviation (MAD) with tunable scale for robust normalization,
        more stable than mean-abs under skewed ready-task distributions.
      - Criticality strictly gated: upward_rank contributes *only* if slack > -epsilon (hard DDL feasibility).
      - Slack penalty: quadratic for negative slack, linear gain for positive — matches urgency semantics.
      - Wait bias uses exponential saturation (1 - exp(-rate * wait)) to bound starvation correction.
      - Uncertainty-slack interaction is multiplicative and only active under deadline stress.
      - All operations safeguarded against NaN/inf using np.finfo and finite fallbacks.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Copy inputs with float64 precision; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization using median and MAD (more stable than mean/std under skew/outliers)
    def robust_normalize(x):
        x = np.asarray(x)
        med = np.median(x)
        mad = np.median(np.abs(x - med))
        scale = PARAMS["robustness_mad_scale"] * (mad + eps)
        return (x - med) / scale
    
    # Duration: execution + communication
    duration = exec_t + comm_t
    
    # === Slack processing: piecewise & risk-aware ===
    # Negative slack → quadratic penalty (stronger than linear); positive slack → linear gain (mild de-prioritization)
    slack_norm = robust_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (slack_norm ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling (higher priority for lower energy) ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * robust_normalize(inv_energy)
    
    # === Criticality: strict slack-gated activation (only if DDL-feasible path remains) ===
    rank_active = np.where(slk > -eps, rank, 0.0)
    rank_score = -PARAMS["criticality_weight"] * robust_normalize(rank_active + eps)
    
    # === Duration-uncertainty coupling: harmonic blend for stable timing signal ===
    dur_norm = robust_normalize(duration + eps)
    uncert_norm = robust_normalize(uncert + eps)
    # Weighted harmonic blend: avoids singularity, smooth, bounded in [0, ~2]
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * uncert_norm) 
        / (dur_norm + eps + PARAMS["duration_uncertainty_ratio"] * uncert_norm + eps)
    )
    dur_score = robust_normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -robust_normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: amplifies uncertainty impact only under deadline stress ===
    slack_stress = np.where(slk < 0, np.abs(slack_norm), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * uncert_norm * slack_stress
    
    # === Assemble final score: smaller = better ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf with finite fallback using machine epsilon bounds
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max, neginf=finfo.min)
    
    # Ensure shape (N,) and finite values
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
