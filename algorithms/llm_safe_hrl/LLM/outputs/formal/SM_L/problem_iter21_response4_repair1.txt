def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with three structural improvements:
      - Replaces linear `slack_urgency_gain` with bounded inverse-slack exponentiation for stable urgency scaling.
      - Introduces robust min-max normalization for rank/work (instead of quantile) to improve cross-seed stability.
      - Adds soft feasibility gate: sigmoid(slack) applied to *all non-slack contributions*, gently suppressing them under extreme deadline pressure—preserving DDL dominance without hard gating.
      - Removed `soft_feasibility_gate_slope` to comply with 12-parameter limit; reused `ddl_protection_sigmoid_slope` for soft gate to preserve intent and reduce count.
    """
    eps = np.finfo(np.float64).tiny
    N = len(min_exec_time)
    
    # Cast safely without in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust min-max normalization for structural features (rank, work, bottleneck) ===
    # Uses finite support and avoids quantile sensitivity across seeds
    def normalize_minmax_pos(x, eps=eps):
        x = np.asarray(x)
        finite_mask = np.isfinite(x) & (x >= 0)
        if np.any(finite_mask):
            x_clean = x[finite_mask]
            x_min = np.min(x_clean)
            x_max = np.max(x_clean)
            range_val = x_max - x_min + eps
        else:
            x_min, range_val = 0.0, eps
        return (x - x_min) / range_val
    
    # === Slack: signed temporal → abs-quantile (preserved) ===
    def normalize_abs_quantile(x, q=PARAMS["robust_normalization_quantile"]):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], q)
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    slack_norm = normalize_abs_quantile(slk)
    
    # === Slack-driven base penalty: negative slack penalized, positive slack uses smooth inverse scaling ===
    # Avoids pathological de-prioritization near zero slack
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -np.power(np.abs(slk) + eps, -PARAMS["slack_inverse_urgency_exponent"])
    )
    
    # === Conditional sigmoid DDL protection gate: active ONLY when slack < 0 ===
    sigmoid_gate = np.where(
        slk < 0,
        1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_sigmoid_slope"] * slk)),
        0.0
    )
    
    # === Soft feasibility gate: reuses ddl_protection_sigmoid_slope for stability and parameter count compliance ===
    # Smoothly reduces influence of non-slack terms as slack becomes highly negative
    soft_gate = 1.0 / (1.0 + np.exp(-PARAMS["ddl_protection_sigmoid_slope"] * slk))
    
    # === Energy efficiency: inv_energy is positive → use abs-quantile (consistent with temporal scaling) ===
    slack_magnitude = np.abs(slk) + eps
    slack_coupling_factor = np.power(slack_magnitude, -PARAMS["energy_ddl_coupling_exponent"])
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize_abs_quantile(inv_energy) * sigmoid_gate * slack_coupling_factor * soft_gate
    
    # === Criticality: upward_rank ≥ 0 → min-max normalized, gated by DDL and soft gate ===
    rank_sharpened = np.power(rank + eps, PARAMS["rank_sharpening_exponent"])
    rank_norm = normalize_minmax_pos(rank_sharpened)
    rank_score = -PARAMS["criticality_weight"] * rank_norm * sigmoid_gate * soft_gate
    
    # === Bottleneck term: rank * work → min-max normalized pre-interaction, no slack inversion ===
    bottleneck_base = rank * work
    bottleneck_norm = normalize_minmax_pos(bottleneck_base + eps)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * bottleneck_norm * soft_gate
    
    # === Duration & uncertainty: signed temporal → abs-quantile ===
    duration = exec_t + comm_t
    dur_norm = normalize_abs_quantile(duration + eps)
    uncert_norm = normalize_abs_quantile(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize_abs_quantile(dur_uncert_blend) * soft_gate
    
    # === Starvation mitigation: wait ≥ 0 → abs-quantile on clipped input, softened ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize_abs_quantile(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)
    wait_score = -wait_ramp * soft_gate
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with bounded finite values
    score = np.nan_to_num(score, nan=0.0, posinf=PARAMS["score_clipping_bound"], neginf=-PARAMS["score_clipping_bound"])
    score = np.clip(score, -PARAMS["score_clipping_bound"], PARAMS["score_clipping_bound"])
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
