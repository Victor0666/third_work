def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with IQR-based robust normalization, softmax urgency gating,
    and monotonic uncertainty-coupled urgency — eliminating brittle thresholding and outlier-prone percentiles.
    
    Key self-evolution improvements:
      - Replaced percentile_normalize(p10/p90) with IQR_normalize(q1/q3 * scale): more stable on small N,
        less sensitive to extreme tails, preserves rank order better under sparse ready sets.
      - Replaced hard `np.where(slk < 0, ..., ...)` with smooth softmax urgency gating using
        `exp(-slk / urgency_smoothing)` — provides gradient-friendly, monotonic decay for slack,
        enabling reliable CMA-ES optimization without discontinuities.
      - Replaced successor-release interaction with normalized urgency coupling:
        `clip(-slack, 0, ∞) / (1 + uncertainty)` — avoids noise amplification while promoting
        unblocking of critical paths under latency risk.
      - All operations use finfo-safe clamping and deterministic shape enforcement.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust IQR-based normalization: q1/q3 → [-1,1] mapping scaled by tunable factor
    def iqr_normalize(x):
        x = np.asarray(x)
        q1 = np.percentile(x, 25)
        q3 = np.percentile(x, 75)
        iqr = q3 - q1 + eps
        rng = PARAMS["iqr_scale_factor"] * iqr
        centered = x - (q1 + q3) / 2.0
        return np.clip(centered / rng, -2.0, 2.0)
    
    # Duration includes successor workload to anticipate contention
    duration = exec_t + comm_t + PARAMS["successor_work_coupling"] * work
    norm_duration = iqr_normalize(duration + eps)
    
    # Inverse energy efficiency: favors low marginal energy sharply
    inv_energy = 1.0 / (energy + eps)
    norm_energy = iqr_normalize(inv_energy)
    energy_score = -PARAMS["energy_efficiency_weight"] * norm_energy
    
    # Softmax urgency gating: smooth, differentiable replacement for hard slack thresholds
    # exp(-max(-slk, 0) / urgency_smoothing) → 1.0 when slk >= 0, decays smoothly for lateness
    urgency_raw = np.maximum(-slk, 0.0)
    urgency_gate = np.exp(-urgency_raw / (PARAMS["urgency_smoothing"] + eps))
    norm_slack = iqr_normalize(slk)
    # Slack penalty: quadratic risk penalty weighted by urgency_gate; reduced urgency = softer penalty
    base_slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (norm_slack ** 2) * urgency_gate,
        -PARAMS["slack_urgency_gain"] * np.abs(norm_slack) * (1.0 - urgency_gate)
    )
    # Hard gate: amplify penalty under violation risk (still retains differentiability via multiplicative gate)
    slack_penalty = np.where(
        slk < -eps,
        PARAMS["ddl_protection_gate"] * base_slack_penalty,
        base_slack_penalty
    )
    
    # Criticality: gated by slack feasibility — zero contribution if deadline violated
    rank_active = np.where(slk >= -eps, rank, 0.0)
    norm_rank = iqr_normalize(rank_active + eps)
    rank_score = -PARAMS["criticality_weight"] * norm_rank
    
    # Duration-uncertainty coupling: linear blend (stable & interpretable)
    norm_uncert = iqr_normalize(uncert + eps)
    dur_uncert_blend = norm_duration + PARAMS["duration_uncertainty_ratio"] * norm_uncert
    dur_score = iqr_normalize(dur_uncert_blend)
    
    # Starvation mitigation: clipped linear ramp
    clipped_wait = np.minimum(wait, PARAMS["wait_clip_threshold"])
    norm_wait = iqr_normalize(clipped_wait + eps)
    wait_score = -norm_wait
    
    # Monotonic uncertainty-coupled urgency: promotes unblocking under tight deadlines + high risk
    # Uses smoothed urgency (not binary) and normalized uncertainty → avoids noise amplification
    unc_urgency_coupling = urgency_gate * norm_uncert
    unc_urgency_score = -PARAMS["uncertainty_slack_interaction"] * unc_urgency_coupling
    
    # Assemble final score: smaller = higher priority
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_urgency_score
    )
    
    # Final safeguards: ensure finite, shape (N,), deterministic
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1e6
    min_safe = -finfo.max / 1e6
    score = np.nan_to_num(
        score,
        nan=0.0,
        posinf=max_safe,
        neginf=min_safe
    )
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
