def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three key structural improvements:
      - Added host-load-sensitive energy gating: energy term is suppressed by (1 + normalized_uncertainty)^exponent
        only when ddl_gate < 0.5 (i.e., slack is safe), avoiding interference with critical DDL enforcement.
      - Replaced linear slack gain/penalty with piecewise-linear DDL gating: smooth but interpretable transition
        between hard penalty (slk < 0), neutral zone (0 <= slk <= threshold), and soft de-prioritization (slk > threshold).
      - Introduced bounded exponential starvation mitigation: 1 - exp(-wait/θ) ensures monotonic increase,
        avoids flip behavior observed in counterfactual-dominated choices.
      - All normalizations remain quantile-based and epsilon-guarded.
      - Criticality and bottleneck terms preserved, now conditioned on same DDL gate for consistency.
      - Duration-uncertainty coupling simplified to sum (no ratio parameter needed).
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Piecewise-linear DDL gating: hard penalty for negative slack, neutral buffer, soft de-prioritization beyond threshold ===
    # Threshold derived from normalized slack scale — ensures consistent behavior across traces
    slack_threshold = np.quantile(np.abs(slack_norm), 0.75) + eps
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        np.where(
            slk <= slack_threshold,
            0.0,  # neutral zone: no explicit gain or penalty
            -PARAMS["slack_urgency_gain"] * (slk - slack_threshold) / (slack_threshold + eps)
        )
    )
    
    # === Energy efficiency: inverse scaling, now gated by host-load sensitivity only when DDL is safe (ddl_gate < 0.5) ===
    inv_energy = 1.0 / (energy + eps)
    energy_norm = normalize(inv_energy)
    # Load sensitivity: uncertainty acts as proxy for host load variability; suppress energy term when load is high AND slack is safe
    load_suppression = np.where(
        ddl_gate < 0.5,
        np.power(1.0 + normalize(uncert), PARAMS["host_load_sensitivity_exponent"]),
        1.0
    )
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_norm / (load_suppression + eps)
    
    # === Criticality & bottleneck: both gated identically by ddl_gate for coherence ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term with inverse slack exponentiation ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * ddl_gate * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: simplified to sum (no ratio parameter) ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_score = dur_norm + uncert_norm
    
    # === Starvation mitigation: bounded exponential ramp (1 - exp(-wait/θ)) → monotonic, smooth, avoids saturation artifacts ===
    wait_exp_ramp = 1.0 - np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_score = -wait_exp_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant; all other terms are conditionally gated or suppressed
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1e6
    min_safe = -finfo.max / 1e6
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
