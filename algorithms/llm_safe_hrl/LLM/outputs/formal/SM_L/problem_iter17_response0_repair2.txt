def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule combining Parent 2's arctan slack penalty and MAD-based normalization with Parent 1's bottleneck sharpness and energy-DDL coupling.
    Key innovations:
      - Bottleneck term sharpened via slack-magnitude exponentiation (Parent 1) *and* sigmoid DDL gating (Parent 2)
      - Energy term uses power-law coupling with ddl_gate (Parent 1's concept, but with Parent 2's superior sigmoid)
      - Retains robust [-2,2] clipped MAD/median normalization from Parent 2 for outlier resilience
      - Uses arctan slack mapping for saturation (Parent 2), not linear (Parent 1)
      - All normalizations apply floor before scaling to prevent division-by-zero degeneracy
      - Sigmoid gate input is clipped using tunable PARAMS["sigmoid_gate_clip_bound"] instead of literal 700.0
      - No unbounded operations; all divisions guarded; no infinite loops or side effects
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust bounded normalization: median + MAD scale, then clip to [-2, 2] ===
    def normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            med = np.median(abs_x[finite_mask])
            dev = np.abs(abs_x - med)
            mad = np.median(dev[finite_mask]) if np.any(finite_mask) else eps
            # Use max(med, mad, eps) to ensure scale > 0 even for near-zero signals
            scale = max(med, 1.0 * mad, eps)  # literal 1.0 allowed
        else:
            scale = eps
        normed = x / (scale + eps)
        return np.clip(normed, -2.0, 2.0)  # bounded piecewise form
    
    # === Normalized slack for unified gating ===
    slack_norm = normalize(slk)
    
    # === Smooth sigmoid DDL protection gate: rises as slack decreases ===
    gate_input = PARAMS["ddl_protection_steepness"] * (PARAMS["ddl_protection_center"] - slack_norm)
    # Clip gate input to avoid overflow in exp using tunable bound
    gate_input_clipped = np.clip(gate_input, -PARAMS["sigmoid_gate_clip_bound"], PARAMS["sigmoid_gate_clip_bound"])
    ddl_gate = 1.0 / (1.0 + np.exp(-gate_input_clipped))
    
    # === Saturating slack penalty using arctan with fixed π scaling ===
    slack_arctan = (2.0 / np.pi) * np.arctan(slack_norm)  # maps ℝ → (-1,1)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (1.0 + slack_arctan),  # (-1,1) → (0,2)
        -PARAMS["slack_urgency_gain"] * (1.0 - slack_arctan)  # (-1,1) → (0,2)
    )
    
    # === Energy efficiency: inverse marginal energy, power-law gated by ddl_gate ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * np.power(ddl_gate, PARAMS["energy_ddl_coupling_exponent"])
    
    # === Criticality: smoothly gated by sigmoid, rank normalized with floor ===
    rank_stable = rank + PARAMS["upward_rank_normalization_floor"]
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_stable) * ddl_gate
    
    # === Bottleneck-proximal term: sharpened via inverse slack magnitude exponentiation AND ddl_gate ===
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["bottleneck_sharpness_exponent"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps) * ddl_gate
    
    # === Duration-uncertainty coupling: simplified bounded linear blend ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: sharpened power-law ramp — monotonic, bounded, stronger bias than linear ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_sharpened = np.power(wait_clipped + eps, PARAMS["starvation_exponent"])
    wait_score = -normalize(wait_sharpened)
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with 0.0 using explicit masking
    score = np.where(np.isfinite(score), score, 0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
