def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with key structural changes:
      - Replaced sigmoid DDL gate with piecewise-linear feasibility margin for smoother cross-seed consistency
      - Added explicit ddl_feasible_mask to conditionally activate energy scoring only when slack >= margin
      - Sharpened successor-release interaction using absolute slack magnitude with clamped exponentiation
      - Used bounded exponential saturation for wait-time (1 - exp(-wait/θ)) instead of linear ramp — eliminates flip behavior
      - All normalizations use robust quantile scaling; no mean/median bias
      - Strict DDL-first ordering preserved via dominant slack_penalty and conditional energy activation
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Piecewise-linear DDL feasibility margin (replaces sigmoid gate) ===
    # Feasibility region: slack >= -margin → full energy priority; otherwise suppressed
    ddl_feasibility_margin = PARAMS["ddl_feasibility_margin"]
    ddl_feasible_mask = np.where(slk >= -ddl_feasibility_margin, 1.0, 0.0)
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: gated by ddl_feasible_mask — only active when DDL risk is low ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * ddl_feasible_mask
    
    # === Criticality: un-gated but normalized — critical path matters regardless of slack sign ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened using clamped |slack|⁻ˢʰᵃʳᵖⁿᵉˢˢ with floor ===
    # Avoids extreme amplification near zero slack by clamping inverse slack magnitude
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    # Clamp slack_magnitude_inv to prevent explosion when |slack| ~ 0
    slack_magnitude_inv_clamped = np.clip(slack_magnitude_inv, 0.0, 1.0 / eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv_clamped, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential saturation (1 - exp(-wait/θ)) → smooth, monotonic, no flip ===
    wait_saturation = 1.0 - np.exp(-wait / (PARAMS["wait_ramp_threshold"] + eps))
    wait_normalized = normalize(wait_saturation + eps)
    wait_score = -wait_normalized  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with finite values using machine epsilon scaled bound
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finite_safeguard_scale"]
    min_safe = -finfo.max / PARAMS["finite_safeguard_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
