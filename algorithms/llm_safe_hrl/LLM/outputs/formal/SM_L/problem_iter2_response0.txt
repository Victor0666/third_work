def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Improved priority rule combining robust deadline risk containment (Parent 2),
    median-stabilized normalization (Performance Analysis), slack-gated criticality,
    and successor-release risk coupling.
    
    Key improvements:
      - Uses median absolute deviation (MAD) normalization for all features → stable under skew
      - Criticality gating uses slack margin relative to median positive slack → adaptive deadline awareness
      - Successor-release interaction: tanh-saturated product of normalized successor work and inverse slack
      - Preserves inverse-energy efficiency and exponential wait saturation from Parent 2
      - All operations guarded against NaN/inf/zero; deterministic and finite output.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs to float64 and ensure no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Robust normalization using median and MAD (Perf Analysis): stable under skew/outliers
    def mad_normalize(x):
        x = np.asarray(x)
        med = np.median(x)
        mad = np.median(np.abs(x - med)) + eps
        return (x - med) / mad
    
    # Duration: execution + communication
    duration = exec_t + comm_t
    norm_duration = mad_normalize(duration)
    
    # Energy: inverse efficiency — higher priority for lower energy
    inv_energy = 1.0 / (energy + eps)
    norm_inv_energy = mad_normalize(inv_energy)
    energy_score = -PARAMS["energy_efficiency_weight"] * norm_inv_energy
    
    # Slack processing: quadratic penalty for negative, linear gain for positive
    norm_slack = mad_normalize(slk)
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (norm_slack ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(norm_slack)
    )
    
    # Criticality gating: activate only when slack > -margin, where margin = median_slack_gate * median_pos_slack
    pos_slack = slk[slk > eps]
    median_pos_slack = np.median(pos_slack) if len(pos_slack) > 0 else 1.0
    slack_margin = PARAMS["median_slack_gate"] * max(median_pos_slack, eps)
    rank_gate = np.where(slk > -slack_margin, 1.0, 0.0)
    norm_rank = mad_normalize(rank + eps)
    rank_score = -PARAMS["criticality_weight"] * norm_rank * rank_gate
    
    # Duration-uncertainty coupling: weighted harmonic blend (Parent 2), stabilized
    norm_uncert = mad_normalize(uncert + eps)
    dur_uncert_blend = (
        (1.0 + PARAMS["duration_uncertainty_ratio"] * norm_uncert) 
        / (np.abs(norm_duration) + eps + PARAMS["duration_uncertainty_ratio"] * np.abs(norm_uncert) + eps)
    )
    dur_score = mad_normalize(dur_uncert_blend)
    
    # Starvation mitigation: exponential saturation on wait time
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    norm_wait_sat = mad_normalize(wait_sat + eps)
    wait_score = -norm_wait_sat
    
    # Uncertainty-slack interaction: only amplifies under deadline stress (negative slack)
    slack_stress = np.where(slk < 0, np.abs(norm_slack), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * norm_uncert * slack_stress
    
    # Successor-release risk coupling: penalize tasks whose immediate successors have high work & low slack
    # Since we lack direct successor arrays, we use `remaining_work` as proxy for aggregate descendant load
    # and couple with inverse slack (high risk = high work * low slack); tanh saturates interaction
    work_slack_product = work * np.where(slk > eps, 1.0 / (slk + eps), 0.0)
    norm_work_slack = mad_normalize(work_slack_product + eps)
    successor_risk = PARAMS["successor_risk_coupling"] * np.tanh(norm_work_slack)
    
    # Final score: smaller = higher priority
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + successor_risk
    )
    
    # Final safeguard: replace NaN/inf with finite values; preserve ordering via median fallback
    score = np.nan_to_num(
        score,
        nan=np.median(score) if np.any(np.isfinite(score)) else 0.0,
        posinf=np.finfo(np.float64).max / 1e6,
        neginf=-np.finfo(np.float64).max / 1e6
    )
    
    # Ensure shape (N,) and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
