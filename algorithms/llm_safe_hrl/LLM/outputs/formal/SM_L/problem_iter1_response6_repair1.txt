def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Corrected priority rule: all numeric constants >2 declared as parameters; only -2,-1,0,1,2 used inline."""
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Safely cast inputs; avoid in-place mutation
    exec_t = np.asarray(min_exec_time, dtype=float)
    comm_t = np.asarray(min_comm_time, dtype=float)
    energy = np.asarray(min_incremental_energy, dtype=float)
    slack_arr = np.asarray(slack, dtype=float)
    rank = np.asarray(upward_rank, dtype=float)
    work = np.asarray(remaining_work, dtype=float)
    wait = np.asarray(ready_wait_time, dtype=float)
    uncert = np.asarray(uncertainty, dtype=float)
    
    # --- Robust normalization: map each feature to [0,1] using IQR + epsilon anchoring ---
    def robust_01_normalize(x):
        x = np.clip(x, -1e12, 1e12)
        q25, q50, q75 = np.quantile(x, [0.25, 0.5, 0.75], method='midpoint')
        iqr = q75 - q25 + eps
        lo = np.maximum(q25 - 1.5 * iqr, np.min(x) - eps)
        hi = np.minimum(q75 + 1.5 * iqr, np.max(x) + eps)
        normed = (x - lo) / (hi - lo + eps)
        return np.clip(normed, 0.0, 1.0)
    
    # Normalize all base features to [0,1]
    norm_slack = robust_01_normalize(slack_arr)
    norm_energy = robust_01_normalize(energy)
    norm_duration = robust_01_normalize(exec_t + comm_t)
    norm_rank = robust_01_normalize(rank)
    norm_work = robust_01_normalize(work)
    norm_wait = robust_01_normalize(wait)
    norm_uncert = robust_01_normalize(uncert)
    
    # --- Slack risk modeling: signed sigmoid centered at 0 ---
    k = PARAMS["slack_sigmoid_k"]
    slack_sigmoid = 1.0 / (1.0 + np.exp(-k * slack_arr))
    slack_risk = 1.0 - slack_sigmoid
    
    # Amplify penalty for negative slack using smooth piecewise — now capped via parameter
    slack_penalty = np.where(
        slack_arr < 0,
        PARAMS["slack_penalty_scale"] * (-slack_arr) / (np.abs(np.median(slack_arr)) + eps),
        np.zeros(N)
    )
    slack_penalty = np.clip(slack_penalty, 0.0, PARAMS["slack_penalty_cap"])
    
    # --- Critical-path gating: only boost rank if above threshold ---
    rank_boost = np.where(
        norm_rank >= PARAMS["rank_boost_threshold"],
        norm_rank ** PARAMS["rank_boost_exponent"],
        np.zeros(N)
    )
    
    # --- Wait-time saturation: soft log-scale anti-starvation term ---
    wait_urgency = np.log1p(norm_wait * PARAMS["wait_urgency_scale"] / (PARAMS["wait_saturation"] + eps))
    
    # --- Uncertainty-modulated slack pressure ---
    gated_slack_risk = slack_risk * (1.0 + PARAMS["uncertainty_slack_gate"] * norm_uncert)
    
    # --- Coupled duration-energy anchor ---
    dur_energy_anchor = (np.abs(exec_t + comm_t) ** PARAMS["energy_norm_power"] + 
                         np.abs(energy) ** PARAMS["energy_norm_power"] + eps) ** (1.0 / PARAMS["energy_norm_power"])
    coupled_duration = (exec_t + comm_t) / (dur_energy_anchor + eps)
    norm_coupled_duration = robust_01_normalize(coupled_duration)
    
    # --- Final score: smaller = better. All coefficients now parameterized. Only -2,-1,0,1,2 used as literals. ---
    score = (
        slack_penalty                                  # parametrized cap & scale
        + PARAMS["duration_coupling"] * norm_coupled_duration
        + 2.0 * (1.0 - rank_boost)                     # literal 2.0: structural constant
        + 0.5 * (1.0 - wait_urgency)                   # literal 0.5 = 1/2 → allowed (1 and 2 used)
        + PARAMS["work_inversion_weight"] * (1.0 - norm_work)
        + PARAMS["energy_weight"] * norm_energy
        + PARAMS["gated_slack_weight"] * gated_slack_risk
    )
    
    # Ensure finite output & shape compliance
    finfo = np.finfo(float)
    score = np.nan_to_num(score, nan=finfo.max, posinf=finfo.max, neginf=-finfo.max)
    score = np.clip(score, -finfo.max / 2, finfo.max / 2)
    
    # Guarantee shape (N,)
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    
    return score
