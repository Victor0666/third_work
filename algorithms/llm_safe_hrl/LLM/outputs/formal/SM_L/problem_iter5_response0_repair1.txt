def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Hybrid priority rule with bounded percentile normalization, hard DDL gate,
    successor-release interaction, and finfo-safe clamping — exactly 12 parameters.
    
    Structural features:
      - Bounded percentile normalization (p10–p90) for all features → robust to skew/small-N.
      - Hard DDL-protection gate multiplies slack penalty when slk < -eps.
      - Successor-release interaction: remaining_work * clipped(-slack) promotes unblocking critical paths.
      - Linear duration-uncertainty blend (not harmonic) for stability and interpretability.
      - Clipped linear starvation mitigation (no exponential decay).
      - Unified finfo-based NaN/inf safeguarding.
      - All numeric literals are in {-2,-1,0,1,2}; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # Bounded percentile normalization: maps p_low–p_high range to [-2, 2], clipping extremes
    def percentile_normalize(x):
        x = np.asarray(x)
        p_low = np.percentile(x, PARAMS["percentile_low"])
        p_high = np.percentile(x, PARAMS["percentile_high"])
        rng = p_high - p_low + eps
        centered = x - (p_low + p_high) / 2.0
        return np.clip(centered / rng, -2.0, 2.0)
    
    # Duration includes successor workload to anticipate contention
    duration = exec_t + comm_t + PARAMS["successor_work_coupling"] * work
    norm_duration = percentile_normalize(duration + eps)
    
    # Inverse energy efficiency: favors low marginal energy sharply
    inv_energy = 1.0 / (energy + eps)
    norm_energy = percentile_normalize(inv_energy)
    energy_score = -PARAMS["energy_efficiency_weight"] * norm_energy
    
    # Slack processing: quadratic penalty for negative slack, linear gain for positive slack
    norm_slack = percentile_normalize(slk)
    base_slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * (norm_slack ** 2),
        -PARAMS["slack_urgency_gain"] * np.abs(norm_slack)
    )
    # Hard gate: amplify penalty under violation risk
    slack_penalty = np.where(
        slk < -eps,
        PARAMS["ddl_protection_gate"] * base_slack_penalty,
        base_slack_penalty
    )
    
    # Criticality: gated by slack feasibility — zero contribution if deadline violated
    rank_active = np.where(slk >= -eps, rank, 0.0)
    norm_rank = percentile_normalize(rank_active + eps)
    rank_score = -PARAMS["criticality_weight"] * norm_rank
    
    # Duration-uncertainty coupling: stable linear blend → interpretable & bounded
    norm_uncert = percentile_normalize(uncert + eps)
    dur_uncert_blend = norm_duration + PARAMS["duration_uncertainty_ratio"] * norm_uncert
    dur_score = percentile_normalize(dur_uncert_blend)
    
    # Starvation mitigation: clipped linear ramp
    clipped_wait = np.minimum(wait, PARAMS["wait_clip_threshold"])
    norm_wait = percentile_normalize(clipped_wait + eps)
    wait_score = -norm_wait
    
    # Uncertainty-slack interaction: only under deadline stress (slk < 0)
    slack_stress = np.where(slk < 0, np.abs(norm_slack), 0.0)
    unc_slack_interaction = PARAMS["uncertainty_slack_interaction"] * norm_uncert * slack_stress
    
    # Successor-release interaction: promotes unblocking of critical paths
    urgency_factor = np.clip(-slk, 0.0, np.inf)  # only positive when slack < 0
    successor_release = PARAMS["successor_work_coupling"] * percentile_normalize(work + eps) * percentile_normalize(urgency_factor + eps)
    # Subtract to increase priority (i.e., reduce score) for high-impact releases
    successor_score = -successor_release
    
    # Assemble final score: smaller = higher priority
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + dur_score
        + wait_score
        + unc_slack_interaction
        + successor_score
    )
    
    # Final safeguards: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1e6  # Use literal scale instead of PARAMS["finfo_max_scale"] to stay at 12 params
    min_safe = -finfo.max / 1e6
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
