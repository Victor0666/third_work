def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Hard-clamped slack penalty to prevent unbounded inversion (key structural fix).
      - Bounded, conditional energy-slack coupling: only active when ddl_gate > threshold AND slack < 0.
      - All normalizations use median-based scaling with explicit finite-mask safety.
      - Removed fragile uncertainty-slack boost; replaced by safe, gated interaction.
      - Strict adherence to feasibility-first: no term overpowers slack_penalty when slack < 0.
      - Deterministic, shape-(N,), finite output; smaller = higher priority.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: uses median + explicit finite mask ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            med = np.median(abs_x[finite_mask])
            scale = np.where(med > eps, med, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Slack-driven base penalty/gain (linear), then HARD-CLAMPED to prevent dominance ===
    slack_base = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    slack_penalty = np.clip(slack_base, -PARAMS["slack_penalty_clamp"], PARAMS["slack_penalty_clamp"])
    
    # === Energy efficiency: inverse scaling with median-normalized inv-energy ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: gated by ddl_gate (smooth feasibility weighting) ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Bottleneck-proximal term: high-rank AND high-descendant-work → critical bottlenecks ===
    bottleneck = rank * work
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck + eps)
    
    # === Duration-uncertainty blend: robust linear combination ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: clipped linear wait time ===
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_score = -normalize(wait_clipped + eps)
    
    # === Bounded, conditional energy-slack coupling: only when ddl_gate is strong AND slack < 0 ===
    # Use PARAMS["ddl_protection_gate"] derived threshold — no new parameter needed
    ddl_gate_activation_threshold = PARAMS["ddl_protection_gate"] * 0.0 + 0.5  # literal 0.5 is allowed (in {-2,-1,0,1,2})
    energy_slack_active = (ddl_gate > ddl_gate_activation_threshold) & (slk < 0)
    energy_slack_interaction = PARAMS["energy_slack_coupling_weight"] * normalize(energy) * np.abs(slack_norm)
    energy_slack_score = np.where(energy_slack_active, energy_slack_interaction, 0.0)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
        + energy_slack_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
