def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three structural improvements:
      - Added host-load-aware energy suppression: energy optimization only activated when DDL safety margin exceeds threshold
      - Replaced linear wait ramp with smooth power-law saturation: preserves monotonicity while reducing sensitivity to outliers
      - Introduced piecewise-linear robust slack gating: replaces sigmoid with interpretable, bounded transition around slack=0
      - Removed finfo_max_scale (inactive per diagnostics) and simplified normalization safeguards
      - All features now strictly normalized using quantile scaling with epsilon fallback
      - Critical-path terms retain dominance via slack penalty + bottleneck interaction
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Piecewise-linear DDL protection gate: crisp but bounded transition at slack=0 ===
    # Linear ramp from 0 to 1 over [-ddl_gate, +ddl_gate] → interpretable, differentiable, no asymptotes
    ddl_gate_width = PARAMS["ddl_protection_gate"]
    ddl_gate = np.clip(0.5 + 0.5 * (slk / (ddl_gate_width + eps)), 0.0, 1.0)
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk),
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: gated by DDL safety margin to suppress energy optimization under risk ===
    # Only activate energy scoring when slack > 0 and sufficiently large relative to execution time
    safe_margin = slk - (exec_t + comm_t)
    energy_active = (safe_margin > eps).astype(np.float64)
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy) * energy_active
    
    # === Host-load-aware suppression: further reduce energy priority when system load is high ===
    # Load proxy: normalized uncertainty + normalized execution time → higher values indicate congestion
    load_proxy = normalize(uncert + eps) + normalize(exec_t + eps)
    load_suppression = 1.0 - PARAMS["host_load_suppression_factor"] * np.clip(load_proxy, 0.0, 1.0)
    energy_score = energy_score * load_suppression
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Successor-release interaction: sharpened bottleneck term with inverse slack exponentiation ===
    # Prioritizes tasks blocking large workloads *especially* when slack is tight
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: power-law wait saturation → smoother than linear ramp ===
    # Ensures monotonic priority boost with diminishing returns as wait time grows
    wait_normalized = normalize(wait + eps)
    wait_saturation = np.power(np.clip(wait_normalized, 0.0, 1.0), PARAMS["host_load_suppression_factor"])
    wait_score = -wait_saturation  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with zero (finite, neutral)
    score = np.nan_to_num(score, nan=0.0, posinf=0.0, neginf=0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
