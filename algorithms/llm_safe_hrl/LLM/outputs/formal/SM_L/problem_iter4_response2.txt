def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with smooth DDL protection gate and successor-release interaction.
    
    Key structural mutations:
      - Replaces hard `slack_norm >= 0` gating with *smooth sigmoid gate* centered at slack=0,
        controlled by `ddl_protection_gate_width` — eliminates discontinuities while preserving
        strong deadline-driven activation near zero slack.
      - Adds `load_successor_release_weight`-scaled term: computes normalized product of
        `upward_rank` and `remaining_work`, then applies smooth gate — promotes tasks that
        release large downstream critical work early under deadline pressure.
      - Retains robust normalization, exponential wait saturation, and bounded linear duration-uncertainty blend.
      - All gates are monotonic and differentiable; no conditionals on raw magnitudes.
      - Uses `np.tanh` for numerically stable, bounded smooth step instead of fragile piecewise logic.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    rem_work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: avoids zero-scale and preserves ordering ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        scale = np.mean(abs_x) if np.all(np.isfinite(abs_x)) and np.mean(abs_x) > eps else eps
        return x / (scale + eps)
    
    # === Smooth slack-based gating: tanh-based transition around slack = 0 ===
    # Gate value ∈ (-1, 1), symmetric, differentiable, stable — replaces hard threshold
    # Width controls steepness: smaller → sharper transition; larger → smoother activation
    gate_center = 0.0
    gate_width = PARAMS["ddl_protection_gate_width"] + eps
    slack_gate = np.tanh((slk - gate_center) / gate_width)
    
    # === Slack-driven components: linear penalty/gain modulated by smooth gate ===
    # Penalty dominates for negative slack (gate < 0); gain dominates for positive slack (gate > 0)
    slack_penalty = (
        PARAMS["slack_risk_penalty"] * np.where(slk < 0, np.abs(slk), 0.0) 
        - PARAMS["slack_urgency_gain"] * np.where(slk > 0, slk, 0.0)
    )
    # Normalize penalty to avoid magnitude explosion from raw slack
    slack_penalty = normalize(slack_penalty + eps)
    
    # === Energy efficiency: inverse scaling → sharper preference for low marginal energy ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Criticality: smooth-gated upward_rank (active near deadline, not binary) ===
    rank_score = -PARAMS["criticality_weight"] * normalize(rank + eps) * (1.0 + slack_gate) / 2.0
    
    # === Successor-release interaction: promotes tasks that unblock high-criticality downstream work ===
    # Product captures 'load release potential'; smoothed by same gate to activate only under stress
    successor_release_potential = rank * rem_work
    successor_release_score = (
        -PARAMS["load_successor_release_weight"] 
        * normalize(successor_release_potential + eps) 
        * (1.0 - slack_gate) / 2.0  # Activates strongest when slack < 0 (gate < 0)
    )
    
    # === Duration-uncertainty coupling: bounded linear blend (stable & interpretable) ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: exponential saturation on wait time ===
    wait_sat = 1.0 - np.exp(-PARAMS["wait_decay_rate"] * wait)
    wait_score = -normalize(wait_sat + eps)
    
    # === Uncertainty-slack interaction: activates only under normalized deadline stress ===
    unc_slack_interaction = (
        PARAMS["uncertainty_slack_interaction"] 
        * uncert_norm 
        * np.maximum(0.0, -slack_gate)  # Stronger when gate < 0 (i.e., slack < 0)
    )
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + successor_release_score
        + dur_score
        + wait_score
        + unc_slack_interaction
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values scaled by finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
