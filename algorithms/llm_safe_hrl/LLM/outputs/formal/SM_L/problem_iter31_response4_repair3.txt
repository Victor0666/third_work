def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with two structural innovations:
      - Introduces hard DDL feasibility mode: when normalized slack falls below `min_slack_feasibility_margin`,
        all non-DDL terms are suppressed and priority is driven purely by slack + criticality + bottleneck terms.
      - Replaces linear wait ramp with smoothed saturating sigmoid ramp for monotonic starvation mitigation without abrupt thresholds.
      - Uses stricter quantile-based normalization with explicit finite masking to improve cross-scenario stability.
      - Energy scoring is conditionally activated only when ddl_gate > energy_activation_threshold.
      - All numeric literals are restricted to {-2,-1,0,1,2}; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization with explicit finite masking ===
    def normalize(x):
        x = np.asarray(x)
        finite_mask = np.isfinite(x)
        if np.any(finite_mask):
            # Use absolute values for scale to handle sign-sensitive features consistently
            abs_x = np.abs(x)
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating and margin logic ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Feasibility margin flag: binary-like but smooth via tanh transition ===
    # Activates strict DDL-first mode when slack_norm <= min_slack_feasibility_margin
    feasibility_margin = np.tanh((PARAMS["min_slack_feasibility_margin"] - slack_norm) / eps)
    strict_ddl_mode = np.where(feasibility_margin > 0.0, 1.0, 0.0)
    
    # === Slack-driven base penalty/gain (linear) — always active ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: conditionally activated only when ddl_gate > energy_activation_threshold ===
    inv_energy = 1.0 / (energy + eps)
    energy_score_base = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    energy_score = np.where(
        ddl_gate > PARAMS["energy_activation_threshold"],
        energy_score_base,
        0.0
    )
    
    # === Criticality: always active under strict mode; gated otherwise ===
    rank_score_base = -PARAMS["criticality_weight"] * normalize(rank + eps)
    rank_score = np.where(
        strict_ddl_mode > 0.0,
        rank_score_base,
        ddl_gate * rank_score_base
    )
    
    # === Successor-release interaction: sharpened bottleneck term ===
    # Prioritizes tasks blocking large workloads *especially* when slack is tight
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: smoothed saturating sigmoid ramp (no hard threshold) ===
    # Monotonic, differentiable, avoids discontinuities of clipping or piecewise linear ramps
    wait_normalized = normalize(wait + eps)
    wait_ramp = 1.0 / (1.0 + np.exp(-wait_normalized))  # sigmoid from 0→1
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # In strict DDL mode, suppress non-DDL terms except bottleneck & criticality
    score = (
        slack_penalty
        + rank_score
        + bottleneck_score
        + wait_score
    )
    
    # Only add energy and duration terms when not in strict DDL mode
    score = np.where(
        strict_ddl_mode > 0.0,
        score,
        score + energy_score + dur_score
    )
    
    # Final safeguard: replace NaN/inf with safe finite values using np.finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / 1e6
    min_safe = -finfo.max / 1e6
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
