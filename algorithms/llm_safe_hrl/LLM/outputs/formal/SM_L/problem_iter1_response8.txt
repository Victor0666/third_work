def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule emphasizing DDL-risk gating, robust normalization, and uncertainty-aware slack coupling.
    
    Key innovations:
      - Uses sign-preserving power-law slack penalty: negative slack penalized quadratically,
        positive slack softly rewarded (smoothed via exponent >1).
      - Replaces mean-based normalization with IQR-based robust scaling to suppress outlier distortion.
      - Introduces uncertainty-coupled slack gate: penalty amplified only where both slack < 0 AND uncertainty is high.
      - Criticality (upward_rank) weighted *relative* to duration via ratio-based stabilization.
      - Ready wait time applied as bounded linear boost, capped by a soft sigmoid to avoid domination.
      - All operations guarded against NaN/inf/zero using epsilon and np.nan_to_num.
    """
    eps = PARAMS["epsilon"]
    
    # Robust normalization: use IQR instead of mean/std for stability under skew/outliers
    def robust_normalize(x):
        x = np.asarray(x, dtype=float)
        q75 = np.quantile(x, 0.75, method='midpoint')
        q25 = np.quantile(x, 0.25, method='midpoint')
        iqr = q75 - q25 + eps
        center = np.median(x)
        return (x - center) / iqr
    
    # Slack processing: strong penalty for negative slack, mild reward for surplus
    # Use power law: |slack|^p * sign(slack), with p > 1 for asymmetry
    slack_abs = np.abs(slack)
    slack_sign = np.sign(slack)
    slack_powered = np.power(slack_abs + eps, PARAMS["slack_sensitivity"]) * slack_sign
    
    # Negative slack penalty: only activate when slack < 0, scaled by factor and uncertainty
    negative_slack_mask = (slack < 0).astype(float)
    uncertainty_coupled_penalty = (
        PARAMS["slack_penalty_factor"] 
        * np.abs(slack_powered) 
        * negative_slack_mask 
        * np.tanh(PARAMS["uncertainty_slack_coupling"] * uncertainty)
    )
    
    # Normalize core features robustly
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(min_exec_time + min_comm_time)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    
    # Stabilized criticality term: upward_rank weighted *against* duration to avoid biasing short tasks
    # Ratio avoids division-by-zero and captures relative importance per unit time
    duration_safe = min_exec_time + min_comm_time + eps
    rank_per_duration = upward_rank / duration_safe
    norm_rank_per_duration = robust_normalize(rank_per_duration)
    
    # Bounded anti-starvation: wait time contributes linearly but saturated via tanh
    wait_boost = PARAMS["wait_fairness"] * np.tanh(ready_wait_time / (1.0 + eps))
    
    # Final score: minimize → prioritize urgent, low-energy, critical-but-efficient, non-starved tasks
    score = (
        negative_slack_penalty
        + PARAMS["energy_scale"] * norm_energy
        + PARAMS["duration_balance"] * norm_duration
        - PARAMS["criticality_weight"] * norm_rank_per_duration
        - wait_boost
    )
    
    # Ensure finite output, no inf/nan, shape (N,)
    return np.nan_to_num(score, nan=0.0, posinf=np.finfo(float).max, neginf=-np.finfo(float).max)
