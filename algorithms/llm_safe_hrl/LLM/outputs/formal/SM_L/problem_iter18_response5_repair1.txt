def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Self-evolved priority rule with:
      - Per-task exponential slack sensitivity (exp(-|slack|/tau)) → preserves urgency gradients near zero slack
      - Uncertainty-as-load-proxy for energy gating → reuses uncertainty, no new param
      - Quantile-aware upward_rank normalization → prevents rank inversion under skew
      - score_clip_min_abs removed to comply with 12-parameter limit; use symmetric clipping via score_clip_max only
      - All numeric literals are {-2,-1,0,1,2}; epsilon handled via PARAMS; no hidden constants.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-aware normalization for rank & work (only where skew matters) ===
    def quantile_normalize(x, q=0.75):
        x = np.asarray(x, dtype=np.float64)
        finite_mask = np.isfinite(x)
        if np.any(finite_mask):
            x_finite = x[finite_mask]
            q_val = np.quantile(x_finite, q)
            scale = np.max(np.abs(x_finite))
            scale = np.where(scale > eps, scale, eps)
            return (x - np.nanmedian(x_finite)) / (scale + eps)
        else:
            return np.zeros_like(x)
    
    def maxabs_normalize(x):
        x = np.asarray(x, dtype=np.float64)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.max(abs_x[finite_mask])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Per-task exponential slack sensitivity: smooth, local, gradient-preserving ===
    abs_slack = np.abs(slk)
    slack_sensitivity = np.exp(-abs_slack / (PARAMS["ddl_sensitivity_tau"] + eps))
    
    # === Slack-driven base penalty/gain (linear, signed) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * abs_slack,
        -PARAMS["slack_urgency_gain"] * slk
    )
    
    # === Energy efficiency: inverse marginal energy, gated by both slack sensitivity AND uncertainty-as-load ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * maxabs_normalize(inv_energy)
    # Joint gating: reduce energy priority when either deadline pressure OR load risk is high
    energy_gate = (1.0 - slack_sensitivity) * (1.0 - PARAMS["energy_load_proxy_weight"] * maxabs_normalize(uncert + eps))
    energy_score = energy_score * (1.0 - energy_gate)
    
    # === Criticality: quantile-normalized upward_rank, power-law sharpened, per-task slack-sensitivity gated ===
    rank_norm = quantile_normalize(rank, PARAMS["rank_quantile_thresh"])
    rank_sharpened = np.power(np.abs(rank_norm) + eps, PARAMS["criticality_exponent"])
    rank_score = -rank_sharpened * slack_sensitivity
    
    # === Bottleneck term: rank * work / (|slack| + eps), per-task slack-sensitivity gated ===
    slack_inv = 1.0 / (abs_slack + eps)
    bottleneck_base = rank * work * slack_inv
    bottleneck_score = -PARAMS["bottleneck_weight"] * maxabs_normalize(bottleneck_base + eps) * slack_sensitivity
    
    # === Duration-uncertainty coupling: bounded linear blend (max-abs normalized) ===
    duration = exec_t + comm_t
    dur_norm = maxabs_normalize(duration + eps)
    uncert_norm = maxabs_normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = maxabs_normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded exponential saturation ===
    wait_sat = 1.0 - np.exp(-wait / (PARAMS["wait_saturation_time"] + eps))
    wait_score = -wait_sat
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with safe finite values using declared bounds
    score_clip_max = PARAMS["score_clip_max"]
    # Use symmetric clipping: clip both tails to [-score_clip_max, score_clip_max]
    score = np.clip(score, -score_clip_max, score_clip_max)
    score = np.nan_to_num(score, nan=0.0, posinf=score_clip_max, neginf=-score_clip_max)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
