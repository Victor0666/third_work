def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with two structural innovations:
      - Upward rank raised to tunable power to avoid saturation on long critical paths.
      - Energy term gated by |slack|^exponent to suppress under deadline pressure without sigmoid instability.
      - Removed unused ddl_protection_gate parameter and associated logic.
      - All normalizations use robust quantile scaling; no mean/median bias.
      - Slack penalty remains dominant to enforce DDL feasibility first.
      - Wait ramp preserved as bounded linear unit-scale monotonic mitigation.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(normalize(slk)),
        -PARAMS["slack_urgency_gain"] * np.abs(normalize(slk))
    )
    
    # === Energy efficiency: gated by slack magnitude to suppress under deadline pressure ===
    # Uses |slack|^exponent: stronger suppression for more negative slack, smoother than sigmoid
    inv_energy = 1.0 / (energy + eps)
    energy_norm = normalize(inv_energy)
    # Gate: power-based decay using |slk| as base
    slack_abs = np.abs(slk) + eps
    energy_gate = np.power(slack_abs, -PARAMS["upward_rank_power"])
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_norm * energy_gate
    
    # === Criticality: upward_rank raised to tunable power, then normalized ===
    rank_pow = np.power(rank + eps, PARAMS["upward_rank_power"])
    rank_score = -PARAMS["criticality_weight"] * normalize(rank_pow)
    
    # === Successor-release interaction: sharpened bottleneck term with inverse slack exponentiation ===
    # Prioritizes tasks blocking large workloads *especially* when slack is tight
    slack_magnitude_inv = 1.0 / (np.abs(slk) + eps)
    bottleneck_sharpened = rank_pow * work * np.power(slack_magnitude_inv, PARAMS["successor_release_sharpness"])
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_sharpened + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear ramp instead of hard clip → monotonic & smooth ===
    # Ramps linearly from 0 to 1 up to threshold, then saturates — preserves ordering, avoids discontinuities
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with zero (safe fallback per evidence)
    score = np.nan_to_num(score, nan=0.0, posinf=0.0, neginf=0.0)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
