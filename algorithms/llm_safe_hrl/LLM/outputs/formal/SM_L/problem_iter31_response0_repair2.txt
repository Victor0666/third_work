import numpy as np

def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three structural improvements:
      - Adds host-load-aware energy gating: suppresses energy optimization when system load proxy (duration + uncertainty) is high,
        aligning with 'avoidable marginal energy' counterfactual evidence.
      - Replaces linear slack penalty/gain with smooth, differentiable piecewise-linear approximation using tanh-based saturation,
        avoiding kinks and improving gradient stability near slack=0.
      - Introduces *normalized bottleneck urgency* as a distinct term: (upward_rank * remaining_work) / (|slack| + eps),
        decoupled from exponentiation to improve interpretability and reduce fragility.
      - Removes finfo_max_scale (inactive per diagnostics) and replaces with robust np.clip-based finite safeguard.
      - All normalizations remain quantile-based; no median/mean bias.
      - DDL feasibility remains lexically dominant via slack-driven base term.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating ===
    slack_norm = normalize(slk)
    
    # === Sigmoid DDL protection gate: soft activation near slack=0 ===
    ddl_gate = 1.0 / (1.0 + np.exp(PARAMS["ddl_protection_gate"] * slack_norm))
    
    # === Smooth piecewise-linear slack scoring via tanh saturation ===
    # Avoids hard kinks; saturates smoothly at ±1 for robustness
    # Negative slack → strong penalty; positive slack → mild de-prioritization
    slack_score = PARAMS["slack_risk_penalty"] * np.tanh(-slk / (np.abs(np.median(slk)) + eps))
    slack_score = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slk) / (np.abs(np.median(slk)) + eps),
        -PARAMS["slack_urgency_gain"] * np.abs(slk) / (np.abs(np.median(slk)) + eps)
    )
    # Apply smooth saturation to prevent unbounded growth
    slack_score = np.clip(slack_score, -PARAMS["slack_saturation_limit"], PARAMS["slack_saturation_limit"])
    
    # === Host-load proxy: duration + uncertainty → higher values indicate heavier system load ===
    host_load_proxy = exec_t + comm_t + uncert
    load_norm = normalize(host_load_proxy + eps)
    
    # === Energy efficiency: conditionally gated by host load — suppress when load is high ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    # Apply load-sensitive attenuation: full weight when load is low, reduced when high
    load_attenuation = 1.0 - PARAMS["host_load_sensitivity"] * np.clip(load_norm, 0.0, 1.0)
    energy_score = energy_score * load_attenuation
    
    # === Criticality: gated by ddl_gate for smooth feasibility-aware weighting ===
    rank_score = -PARAMS["criticality_weight"] * ddl_gate * normalize(rank + eps)
    
    # === Normalized bottleneck urgency: critical path importance weighted by urgency (1 / |slack|) without exponentiation ===
    # More stable than power law; avoids numeric explosion near slack=0 while preserving ordering
    slack_abs = np.abs(slk) + eps
    bottleneck_urgency = (rank * work) / slack_abs
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_urgency + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear ramp instead of hard clip → monotonic & smooth ===
    # Ramps linearly from 0 to 1 up to threshold, then saturates — preserves ordering, avoids discontinuities
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_ramp_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_ramp = np.clip(wait_normalized, 0.0, 1.0)  # unit-scaled ramp
    wait_score = -wait_ramp  # higher wait → higher priority
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_score
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with safe finite values using clipping
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max / 1e6, neginf=-finfo.max / 1e6)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
