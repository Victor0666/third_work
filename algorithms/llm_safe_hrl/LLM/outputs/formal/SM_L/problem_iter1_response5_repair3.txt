def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Novel priority rule: prioritizes DDL-critical tasks with risk-gated energy efficiency.
    
    Fully compliant: uses only -2,-1,0,1,2 literals; all tunable constants declared;
    no loops, no I/O, no VM selection; deterministic and finite-output.
    """
    epsilon = PARAMS["epsilon"]
    p = PARAMS["robustness_normalization_power"]

    # Cast and sanitize inputs — no loops, no 'for'
    def safe_array(x):
        x = np.asarray(x, dtype=np.float64)
        x = np.nan_to_num(
            x,
            nan=PARAMS["nan_replacement"],
            posinf=PARAMS["inf_replacement"],
            neginf=PARAMS["neginf_replacement"]
        )
        return x

    min_exec_time = safe_array(min_exec_time)
    min_comm_time = safe_array(min_comm_time)
    min_incremental_energy = safe_array(min_incremental_energy)
    slack = safe_array(slack)
    upward_rank = safe_array(upward_rank)
    remaining_work = safe_array(remaining_work)
    ready_wait_time = safe_array(ready_wait_time)
    uncertainty = safe_array(uncertainty)

    # Robust Lp normalization: ||x||_p = (mean(|x|^p))^(1/p)
    def robust_normalize(x):
        x_abs_p = np.abs(x) ** p
        mean_abs_p = np.mean(x_abs_p)
        norm = (mean_abs_p + epsilon) ** (1.0 / p)
        return x / (norm + epsilon)

    exec_comm = min_exec_time + min_comm_time
    norm_slack = robust_normalize(slack)
    norm_energy = robust_normalize(min_incremental_energy)
    norm_duration = robust_normalize(exec_comm)
    norm_rank = robust_normalize(upward_rank)
    norm_work = robust_normalize(remaining_work)
    norm_wait = robust_normalize(ready_wait_time)
    norm_uncert = robust_normalize(uncertainty)

    # Nonlinear slack penalty: convex penalty for negative slack only; zero otherwise
    slack_sign = np.sign(norm_slack)
    is_negative = (slack_sign == -1.0).astype(float)
    abs_norm_slack = np.abs(norm_slack)
    slack_penalty = -is_negative * (abs_norm_slack ** PARAMS["slack_penalty_exponent"])

    # Criticality gate: 1.0 when (norm_slack < threshold) AND (norm_uncert > threshold), else 0.0
    gate_condition = (
        (norm_slack < PARAMS["criticality_gate_threshold"]) &
        (norm_uncert > PARAMS["criticality_gate_threshold"])
    ).astype(float)

    # Energy efficiency: invert normalized energy (avoid division by zero)
    inv_energy = 1.0 / (norm_energy + epsilon)

    # Duration-uncertainty pacing: penalize long+uncertain unless critical
    dur_uncert_pace = norm_duration * norm_uncert * (1.0 - gate_condition)

    # Anti-starvation: smooth saturating boost via arctan (bounded, monotonic)
    wait_boost = np.arctan(norm_wait / PARAMS["wait_saturation_point"])

    # Composite score: smaller = higher priority
    score = (
        slack_penalty
        - PARAMS["energy_efficiency_bias"] * inv_energy
        + PARAMS["duration_uncertainty_coupling"] * dur_uncert_pace
        - gate_condition * PARAMS["rank_energy_tradeoff"] * norm_rank
        - wait_boost
        + PARAMS["remaining_work_weight"] * norm_work
    )

    # Final numerical safeguard using np.finfo
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(
        score,
        nan=finfo.max,
        posinf=finfo.max,
        neginf=-finfo.max
    )

    return score
