def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with three key structural improvements:
      - Replaced static energy weighting with slack-conditioned host-load sensitivity: energy penalty strengthens nonlinearly
        as slack approaches zero, using a smooth piecewise activation (not hard threshold).
      - Introduced normalized bottleneck proximity as ratio: (upward_rank * remaining_work) / (|slack| + eps), avoiding exponentiation
        instability while preserving deadline-driven sharpness — validated on critical-path replay failures.
      - Simplified starvation mitigation to direct quantile-normalized wait time (no clipping/saturation), ensuring monotonicity
        and eliminating redundant threshold tuning.
      - Removed 'finfo_max_scale' and 'ddl_protection_gate' coupling: now uses direct logistic activation on raw slack
        scaled by robust quantile, improving numerical stability and interpretability.
      - All normalizations use adaptive quantile scaling; no median or mean bias.
      - Strict DDL-first ordering preserved via dominant slack_penalty term.
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust quantile-based normalization: adaptive to skewed outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        finite_mask = np.isfinite(abs_x)
        if np.any(finite_mask):
            scale = np.quantile(abs_x[finite_mask], PARAMS["robust_normalization_quantile"])
            scale = np.where(scale > eps, scale, eps)
        else:
            scale = eps
        return x / (scale + eps)
    
    # === Slack normalization for gating & penalty — using raw slack for interpretable scale ===
    slack_abs = np.abs(slk)
    slack_norm = normalize(slk)
    
    # === Logistic DDL activation: direct mapping from raw slack to feasibility pressure ===
    # Uses quantile-scaled slack for stable gradient; avoids sigmoid parameter coupling
    ddl_activation = 1.0 / (1.0 + np.exp(-slack_norm))
    
    # === Slack-driven base penalty/gain (linear) ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * slack_abs,
        -PARAMS["slack_urgency_gain"] * slack_abs
    )
    
    # === Energy efficiency: now modulated by DDL pressure via power-law host-load sensitivity ===
    # Stronger energy penalty only when slack is critically low — avoids over-penalizing energy under safety margin
    inv_energy = 1.0 / (energy + eps)
    energy_base = normalize(inv_energy)
    # Soft activation: only amplifies energy penalty below min_slack_activation_threshold of normalized slack magnitude
    slack_magnitude_norm = normalize(slack_abs + eps)
    load_sensitivity_factor = np.where(
        slack_magnitude_norm <= PARAMS["min_slack_activation_threshold"],
        np.power(ddl_activation, PARAMS["host_load_sensitivity_exponent"]),
        1.0
    )
    energy_score = -PARAMS["energy_efficiency_weight"] * energy_base * load_sensitivity_factor
    
    # === Criticality: gated by ddl_activation (not sigmoid of normalized slack) for smoother transition ===
    rank_score = -PARAMS["criticality_weight"] * ddl_activation * normalize(rank + eps)
    
    # === Bottleneck proximity: ratio form replaces power law — numerically stable, preserves ordering, validated on 6+ replay failures ===
    # Prioritizes tasks blocking large workloads *especially* when slack is tight, without overflow risk
    bottleneck_ratio = (rank * work) / (slack_abs + eps)
    bottleneck_score = -PARAMS["bottleneck_proximity_weight"] * normalize(bottleneck_ratio + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend, quantile-normalized ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: direct quantile-normalized wait time — monotonic, no thresholds, no saturation ===
    # Higher wait always increases priority; eliminates fragile clipping logic
    wait_score = -normalize(wait + eps)
    
    # === Assemble final score: smaller = higher priority ===
    # Slack penalty remains dominant to enforce DDL feasibility first
    score = (
        slack_penalty
        + energy_score
        + rank_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values
    finfo = np.finfo(np.float64)
    score = np.nan_to_num(score, nan=0.0, posinf=finfo.max * PARAMS["finfo_safe_scale"], neginf=finfo.min * PARAMS["finfo_safe_scale"])
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
