def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Mutated priority rule with:
       - Bounded linear wait-time clipping instead of exponential saturation (more stable & interpretable)
       - Slack-gated criticality using upward_rank * remaining_work (bottleneck-aware importance)
       - DDL protection gate: smooth activation of criticality only when slack is near zero or negative
       - Robust median-based normalization to resist outliers (replaces mean-abs scaling)
       - Removed uncertainty-slack interaction: diagnosed inactive; replaced with cleaner ddl_protection_gate
       - All divisions guarded; all outputs finite; shape strictly (N,)
    """
    eps = PARAMS["epsilon"]
    N = len(min_exec_time)
    
    # Cast inputs safely; no in-place modification
    exec_t = np.asarray(min_exec_time, dtype=np.float64)
    comm_t = np.asarray(min_comm_time, dtype=np.float64)
    energy = np.asarray(min_incremental_energy, dtype=np.float64)
    slk = np.asarray(slack, dtype=np.float64)
    rank = np.asarray(upward_rank, dtype=np.float64)
    work = np.asarray(remaining_work, dtype=np.float64)
    wait = np.asarray(ready_wait_time, dtype=np.float64)
    uncert = np.asarray(uncertainty, dtype=np.float64)
    
    # === Robust normalization helper: uses median absolute deviation scale to resist outliers ===
    def normalize(x):
        x = np.asarray(x)
        abs_x = np.abs(x)
        # Use median of absolute values as robust scale; fallback to eps if degenerate
        scale = np.median(abs_x) if np.all(np.isfinite(abs_x)) and np.median(abs_x) > eps else eps
        return x / (scale + eps)
    
    # === Slack gating: use raw slack for sign-sensitive activation, normalized for scaling ===
    slack_norm = normalize(slk)
    
    # === Slack-driven components: linear penalty/gain — unchanged semantics ===
    slack_penalty = np.where(
        slk < 0,
        PARAMS["slack_risk_penalty"] * np.abs(slack_norm),
        -PARAMS["slack_urgency_gain"] * np.abs(slack_norm)
    )
    
    # === Energy efficiency: inverse scaling → sharper preference for low marginal energy ===
    inv_energy = 1.0 / (energy + eps)
    energy_score = -PARAMS["energy_efficiency_weight"] * normalize(inv_energy)
    
    # === Bottleneck-criticality: upward_rank * remaining_work, gated by DDL protection ===
    # DDL protection gate smoothly activates criticality as slack approaches zero or turns negative
    # Gate = 1 - exp(-|slk|/gate_width), but here we use bounded linear ramp: max(0, 1 - |slk|/threshold)
    # This gives strong criticality boost for tight deadlines, none for ample slack
    abs_slack = np.abs(slk)
    ddl_gate = np.clip(1.0 - abs_slack / (PARAMS["ddl_protection_gate"] + eps), 0.0, 1.0)
    bottleneck_importance = rank * work
    # Apply gate only to avoid over-prioritizing non-urgent critical paths
    bottleneck_active = ddl_gate * bottleneck_importance
    bottleneck_score = -PARAMS["criticality_weight"] * normalize(bottleneck_active + eps)
    
    # === Duration-uncertainty coupling: bounded linear blend (stable & interpretable) ===
    duration = exec_t + comm_t
    dur_norm = normalize(duration + eps)
    uncert_norm = normalize(uncert + eps)
    dur_uncert_blend = dur_norm + PARAMS["duration_uncertainty_ratio"] * uncert_norm
    dur_score = normalize(dur_uncert_blend)
    
    # === Starvation mitigation: bounded linear clipping instead of exponential decay ===
    # More stable, predictable, and avoids underflow/overflow; saturates at wait_clip_threshold
    wait_clipped = np.clip(wait, 0.0, PARAMS["wait_clip_threshold"])
    wait_normalized = normalize(wait_clipped + eps)
    wait_score = -wait_normalized  # Higher wait → higher priority (negative sign)
    
    # === Assemble final score: smaller = higher priority ===
    score = (
        slack_penalty
        + energy_score
        + bottleneck_score
        + dur_score
        + wait_score
    )
    
    # Final safeguard: replace NaN/inf with machine-safe finite values scaled by finfo
    finfo = np.finfo(np.float64)
    max_safe = finfo.max / PARAMS["finfo_max_scale"]
    min_safe = -finfo.max / PARAMS["finfo_max_scale"]
    score = np.nan_to_num(score, nan=0.0, posinf=max_safe, neginf=min_safe)
    
    # Enforce shape and finiteness
    assert score.shape == (N,), f"Expected shape (N,), got {score.shape}"
    assert np.all(np.isfinite(score)), "Non-finite values detected in priority score"
    
    return score
